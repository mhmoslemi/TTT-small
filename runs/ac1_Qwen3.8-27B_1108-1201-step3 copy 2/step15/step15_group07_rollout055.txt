        H_aug[n, :n] = 1

    conv = fast_autocorr(a)
    max_val = conv.max()
    tol = max_val * tol_rel + tol_abs
    active = np.where(conv >= max_val - tol)[0]

    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    return active, conv, max_val


def build_hessian_and_gradient_sparse(a, active_lags):
    """Build the Hessian and gradient for the active-set subproblem using sparse matrices."""
    n = len(a)
    H_rows = []
    H_cols = []
    H_vals = []
    g = np.zeros(n)

    for k in active_lags:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if i == j:
                H_rows.extend([i, i])
                H_cols.extend([i, i])
                H_vals.extend([1.0, 1.0])
                g[i] += a[i]
                g[i] += a[i]
            else:
                if i < j:
                    H_rows.extend([i, j, j, i])
                    H_cols.extend([j, i, i, j])
                    H_vals.extend([0.5, 0.5, 0.5, 0.5])
                    g[i] += a[j]
                    g[j] += a[i]

    H = csc_matrix((H_vals, (H_rows, H_cols)), shape=(n, n))
    return H, g


def active_set_newton(a0, time_limit, verbose=True, max_iters=200):
    """
    Active-set Newton method for minimizing max_k c_k(a) over the simplex.
    Converges quadratically once the active set is stable.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Track active set stability
    prev_active = None
    stable_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()

        # Find active lags with tighter tolerance as we converge
        tol_rel = max(1e-10, 1e-8 * (max_val - best_score) / max_val)
        tol_abs = 1e-14 * max_val
        tol = max_val * tol_rel + tol_abs
        active = np.where(conv >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Check active set stability
        active_set = frozenset(active.tolist())
        if prev_active is not None and active_set == prev_active:
            stable_count += 1
        else:
            stable_count = 0
            if stable_count == 0 and it > 5:
                # Active set changed, reduce tolerance
                pass
        prev_active = active_set

        # Build Hessian and gradient (dense for solving, but built from sparse)
        H, g = build_hessian_and_gradient_sparse(a, active)
        H_dense = H.toarray()

        # Solve the KKT system:
        # [2*H  1] [da]   [-2*g]
        # [1^T  0] [lam] = [  0 ]
        # (factor of 2 because we're minimizing sum c_k, gradient is 2*H*a)
        # Actually, c_k(a) = a^T P_k a, grad c_k = 2 P_k a
        # So grad of (1/|L|) sum c_k = (2/|L|) H a where H = (1/|L|) sum P_k
        # Newton step: solve H * da = -grad + lam*1, with 1^T da = 0
        # Let's use: [H  1; 1^T 0] [da; lam] = [-g; 0]
        # where g = (1/|L|) sum 2 P_k a = 2 * (1/|L|) H a... 
        # Actually let's just use: minimize (1/|L|) sum_{k in L} a^T P_k a
        # = a^T ((1/|L|) sum P_k) a = a^T H_eff a
        # grad = 2 * H_eff * a
        # Hessian = 2 * H_eff
        # Newton: [2*H_eff  1; 1^T  0] [da; lam] = [-2*H_eff*a; 0]

        n_active = len(active)
        H_eff = H_dense / n_active
        grad = 2.0 * H_eff @ a

        # Build augmented system
        n_aug = n + 1
        A_sys = np.zeros((n_aug, n_aug))
        A_sys[:n, :n] = 2.0 * H_eff
        A_sys[:n, n] = 1.0
        A_sys[n, :n] = 1.0
        rhs = np.zeros(n_aug)
        rhs[:n] = -grad

        try:
            sol = np.linalg.solve(A_sys, rhs)
            da = sol[:n]
        except np.linalg.LinAlgError:
            # Try regularized
            try:
                A_sys_reg = A_sys.copy()
                A_sys_reg[:n, :n] += 1e-10 * np.eye(n)
                sol = np.linalg.solve(A_sys_reg, rhs)
                da = sol[:n]
            except:
                break

        # Line search
        best_alpha = 0.0
        best_obj = max_val
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.01, 0.005]
        for alpha in alphas:
            a_try = a + alpha * da
            if (a_try < -1e-10).any():
                # Project
                a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * norm / s_try
            obj_try = fast_autocorr(a_try).max()
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_alpha = alpha

        if best_alpha == 0.0:
            # Try smaller or break
            if it > 20:
                break
            # Perturb
            a = a + 1e-6 * np.random.default_rng(it).normal(0, 1, n)
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            continue

        a = a + best_alpha * da
        a = np.maximum(a, 0.0)
        a = a * norm / a.sum()

        if best_obj < best_score:
            best_score = best_obj
            best_a = a.copy()

        if verbose and (it % 10 == 0 or best_obj < best_score + 1e-12):
            elapsed = time.monotonic() - t0
            print(f"    ASN iter={it} ({elapsed:.0f}s): max={best_obj:.12f}, best={best_score:.12f}, |L|={n_active}, alpha={best_alpha:.3f}")

        # Convergence check
        if np.abs(max_val - best_score) < 1e-14 and stable_count > 5:
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Fine-grained coordinate descent for final polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break

        conv = fast_autocorr(a)
        max_val = conv.max()
        top_lags = np.argsort(conv)[-15:]

        improved = False
        perm = rng.permutation(n)

        for idx_in_perm in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx_in_perm]
            if a[i] < norm * 1e-12:
                continue

            # Try moving small amount from i to various j
            for offset in [1, -1, 2, -2, 3, -3, 5, -5, 7, -7]:
                j = (i + offset) % n
                if j == i:
                    continue

                # Compute the delta in convolution efficiently
                # conv_new[k] = conv[k] + d*(a holds at i vs j)
                # Moving d from a[i] to a[j]:
                # delta_conv[k] = d * (contribution of a[j] at k - contribution of a[i] at k)
                # Contribution of a[i] to conv at position k: a[k-i] (if valid)
                # So delta_conv[k] = d * (a[k-j] - a[k-i]) for valid positions

                delta = np.zeros(size)
                # Effect of decreasing a[i] by d: -d * a shifted by i
                lo = i
                hi = min(size, i + n)
                delta[lo:hi] -= a[:hi - lo]
                # Effect of increasing a[j] by d: +d * a shifted by j
                lo = j
                hi = min(size, j + n)
                delta[lo:hi] += a[:hi - lo]
                # Self-terms: a[i]^2 changes, a[j]^2 changes
                # conv[i+i] had a[i]^2, now (a[i]-d)^2, change = -2*d*a[i] + d^2
                # conv[j+j] had a[j]^2, now (a[j]+d)^2, change = 2*d*a[j] + d^2
                # These are already included in the linear approximation above for d<<a[i]
                # For exact computation, we need to add the quadratic terms:
                # The above is exact for the linear part. The quadratic part is d^2 at positions 2*i and 2*j.

                best_d = 0.0
                best_new_max = max_val
                # Try a few values of d
                max_d = a[i] * 0.8
                n_samples = 8
                for d in np.linspace(max_d * 0.05, max_d, n_samples):
                    new_conv = conv + d * delta
                    # Add quadratic correction
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    new_max = new_conv.max()
                    if new_max < best_new_max - 1e-14:
                        best_new_max = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        improved = True
                        no_improve = 0

        if not improved:
            no_improve += 1

        if verbose and (idx_in_perm > 0):
            if it == 0:
                pass

    return best_a, best_score


def try_qp_solve(a_init, n, time_limit, verbose=True):
    """Try to solve the convex QP directly using cvxpy (for smaller n)."""
    if n > 800:
        if verbose:
            print(f"    QP: n={n} too large, skipping")
        return None

    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1]

    size = 2 * n - 1
    # Build constraints using sparse quadratic forms
    for k in range(size):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        rows = []
        cols = []
        vals = []
        for i in range(i_min, i_max + 1):
            j = k - i
            if i == j:
                rows.append(i)
                cols.append(i)
                vals.append(1.0)
            elif i < j:
                rows.extend([i, j])
                cols.extend([j, i])
                vals.extend([0.5, 0.5])
        P_k = cp.cvxpy.cvxpy_core.sparse.Sparse((vals, (rows, cols)), shape=(n, n)) if False else None

        # Simpler: just write out the quadratic form directly
        terms = []
        for i in range(i_min, i_max + 1):
            j = k - i
            if i == j:
                terms.append(cp.square(a[i]))
            elif i < j:
                terms.append(2.0 * a[i] * a[j])
            # else: already counted when i > j (we skip to avoid double counting)
        # Wait, this isn't right. Let me just use cp.quad_form with a dense matrix for small n.
        pass

    # Actually, for n <= 800, let's use dense matrices
    for k in range(size):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        P_k = np.zeros((n, n))
        for i in range(i_min, i_max + 1):
            j = k - i
            if i == j:
                P_k[i, i] = 1.0
            else:
                P_k[i, j] = 0.5
                P_k[j, i] = 0.5
        constraints.append(cp.quad_form(a, P_k) <= t)

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    t_start = time.monotonic()
    solver = get_solver()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        if verbose:
            print(f"    QP solve failed: {e}")
        return None

    if time.monotonic() - t_start > time_limit:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    a_val = np.maximum(np.asarray(a.value, dtype=np.float64), 0.0)
    s = a_val.sum()
    if s < 1e-15:
        return None
    a_val = a_val / s
    return a_val


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_parent = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP Nesterov on parent n (40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    n1 = n_parent
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, patience_threshold=80)
        update_best(a_out, f"lp_n{n1}")

    # Phase 2: Active-set Newton polish (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Active-set Newton n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = active_set_newton(a_norm, t2, verbose=True)
        update_best(a_out, f"asn_n{n_best}")

    # Phase 3: LP Nesterov round 2 (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, patience_threshold=60)
        update_best(a_out, f"lp2_n{n_best}")

    # Phase 4: Active-set Newton again (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.10
    n_best = len(best_seq)
    print(f"\n=== Phase 4: ASN R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = active_set_newton(a_norm, t4, verbose=True, max_iters=300)
        update_best(a_out, f"asn2_n{n_best}")

    # Phase 5: Try different n values (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.15
    print(f"\n=== Phase 5: Multi-n exploration (budget={t5:.0f}s) ===")
    if t5 > 60:
        n_alt_list = [n_parent - 100, n_parent + 100, n_parent + 200]
        for n_alt in n_alt_list:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_sub = min(remaining * 0.4, t5 / 3)
            if t_sub < 20:
                continue
            print(f"  Trying n={n_alt} (budget={t_sub:.0f}s)")
            # Initialize from parent (interpolate)
            a_parent = best_seq
            if n_alt != len(a_parent):
                # Resample
                x_old = np.linspace(0, 1, len(a_parent))
                x_new = np.linspace(0, 1, n_alt)
                a_alt = np.interp(x_new, x_old, a_parent)
            else:
                a_alt = a_parent.copy()
            a_alt = a_alt / a_alt.sum()
            a_norm = a_alt * np.sqrt(2.0 * n_alt) / a_alt.sum()

            # Quick LP
            if t_sub > 30:
                a_out, _ = nesterov_lp_optimize(a_norm, t_sub * 0.6, solver, verbose=False, patience_threshold=50)
                # Quick ASN
                remaining2 = time_budget - (time.monotonic() - t_global)
                t_asn = min(remaining2 * 0.3, t_sub * 0.3)
                if t_asn > 5 and a_out is not None:
                    a_norm2 = a_out * np.sqrt(2.0 * n_alt) / a_out.sum()
                    a_out2, _ = active_set_newton(a_norm2, t_asn, verbose=False, max_iters=100)
                    update_best(a_out2, f"multi_n{n_alt}")
                update_best(a_out, f"multi_n{n_alt}")

    # Phase 6: Final CD polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.4, 60.0)
    print(f"\n=== Phase 6: CD Polish n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t6, verbose=False)
        update_best(a_out, f"cd_n{n_best}")

    # Phase 7: Final ASN with tight tolerance
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.5, 30.0)
    if t7 > 5:
        print(f"\n=== Phase 7: Final ASN n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = active_set_newton(a_norm, t7, verbose=True, max_iters=500)
        update_best(a_out, f"final_asn_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
