    size = 2 * n - 1

def cross_corr_sliding(w, a):
    """Compute (w ⊛ a)_i = sum_k w_k * a_{k-i} for i = 0,...,n-1"""
    n = len(a)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    fs = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fs)
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(np.conj(W) * A, fs)[:n]

def proj_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# --- Evaluation ---

def score_shape(a):
    """2n * max(conv(a,a)) for a on simplex (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_conv_full(a).max()

def score_full(a):
    """Full evaluate_sequence-compatible score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * len(a) * fast_conv_full(a).max() / (s * s))

# --- Nesterov Projected Gradient Descent ---

def nesterov_pgd(a0, time_limit, verbose=True, seed=0):
    """
    Nesterov-accelerated projected gradient descent on simplex.
    Minimizes 2n * soft_max(conv(a,a)) with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_shape(a)
    y = a.copy()
    t0 = time.monotonic()
    it = 0

    # Annealing phases: (T_start, T_end, lr_start, lr_end, time_fraction)
    phases = [
        (80.0, 10.0, 0.08, 0.02, 0.25),
        (10.0, 3.0, 0.02, 0.005, 0.25),
        (3.0, 1.0, 0.005, 0.001, 0.25),
        (1.0, 0.3, 0.001, 0.0002, 0.25),
    ]

    for pi, (Ts, Te, lrs, lre, frac) in enumerate(phases):
        phase_budget = frac * time_limit
        t_start = time.monotonic()

        while time.monotonic() - t_start < phase_budget:
            if time.monotonic() - t0 > time_limit:
                break

            it += 1
            elapsed_frac = min((time.monotonic() - t_start) / max(phase_budget, 1e-9), 1.0)
            T = Ts + (Te - Ts) * elapsed_frac
            lr = lrs + (lre - lrs) * elapsed_frac

            # Nesterov momentum
            if it > 1:
                beta = (it - 1) / (it + 2.0)
                beta = min(beta, 0.95)
            else:
                beta = 0.0
            y = a + beta * (y - a)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

            # Compute gradient
            conv_y = fast_conv_full(y)
            Tconv = T * conv_y
            Tconv -= Tconv.max()
            exp_tc = np.exp(Tconv)
            w = exp_tc / exp_tc.sum()
            grad = 4.0 * n * cross_corr_slide(w, y)

            # Gradient step
            v = y - lr * grad
            a_new = proj_simplex(v)

            # Evaluate
            sc = score_shape(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

            a = a_new

            if verbose and it % 2000 == 0 and pi == 0:
                el = time.monotonic() - t0
                print(f"      PGD it={it} ({el:.1f}s): T={T:.1f}, score={best_score:.10f}")

    return best_a, best_score

# --- LP-based polish (simplified from previous SOTA) ---

def build_active_T(a, active_idx):
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T

def lp_best_response(a, M, solver, active_idx):
    """Solve max sum(b) s.t. T@b <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_active_T(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val / val.sum()

def nesterov_lp_simplex(a0, time_limit, solver, verbose=True, seed=0):
    """LP-based Nesterov on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_shape(a)
    y = a.copy()
    t0 = time.monotonic()
    it = 0

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_full(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Active set
        tol = 0.05 + 0.05 * np.exp(-t / 200.0)
        threshold = M * (1.0 - tol)
        active = np.where(conv_y >= threshold)[0]
        if len(active) < 10:
            active = np.where(conv_y >= M * 0.7)[0]
        if len(active) > 0.85 * (2 * n - 1):
            active = np.where(conv_y >= M * 0.9)[0]

        b = lp_best_response(y, M, solver, active)
        if b is None:
            b = lp_best_response(y, M, solver, np.arange(2 * n - 1))
        if b is None:
            continue

        # Nesterov
        alpha = max(2.0 / (t + 2.0), 0.05)
        if t == 1:
            a_new = b.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * b
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

        y = y if t == 1 else (
            (1.0 - alpha) * y + alpha * b
        )
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        a = a_new

        sc = score_shape(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and t % 500 == 0:
            el = time.monotonic() - t0
            print(f"      LP t={t} ({el:.1f}s): M={M:.10f}, score={best_score:.10f}")

    return best_a, best_score

# --- Initialization and utilities ---

def init_structured(n, style, rng):
    x = np.linspace(0.0, 1.0, n)
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 0.5
    elif style == 2:
        a = 0.3 + 0.7 * (x * (1 - x)) ** 0.3
        a = a[::-1]  # U-shape
    elif style == 3:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif style == 4:
        a = (1.0 - 2.0 * np.abs(x - 0.5)) ** 1.5
    elif style == 5:
        # W-shape
        a = 0.4 + 0.6 * np.cos(4 * np.pi * x) ** 2
    elif style == 6:
        a = np.exp(-12 * (x - 0.1)**2) + np.exp(-12 * (x - 0.9)**2) + 0.3 * np.exp(-4 * (x - 0.5)**2)
    else:
        a = 1.0 + 0.4 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def perturb(a, rng, scale=0.03):
    noise = 1.0 + scale * rng.standard_normal(len(a))
    noise = np.maximum(noise, 0.05)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s

def get_solver():
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    return cp.SCS
