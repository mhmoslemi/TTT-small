import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():

        try:

            prob.solve(solver=s)

                print(f"  Using solver: {s}")

        except Exception:

    _solver = cp.ECOS

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a, rows=None):
    """Build convolution matrix T where (T @ b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    if rows is None:
        rows = np.arange(size)
    T = np.zeros((len(rows), n))
    k_arr = rows[:, None]
    i_arr = np.arange(n)[None, :]
    j = k_arr - i_arr
    valid = (j >= 0) & (j < n)
    T[valid] = a[j[valid]]
    return T


def lp_solve_constraints(a, M, solver, active_indices):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0."""
    n = len(a)
    if len(active_indices) == 0:
        return None
    T = build_conv_matrix(a, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_best_response(a, M, solver, verbose=False):
    """Find best response b with constraint generation."""
    n = len(a)
    size = 2 * n - 1
    conv_a = fast_conv_self(a)

    # Start with top 30% peaks
    threshold = M * 0.85
    active = np.where(conv_a >= threshold)[0]
    if len(active) < 20:
        # Use top 20 lags
        top_idx = np.argsort(conv_a)[::-1][:20]
        active = top_idx

    b = lp_solve_constraints(a, M, solver, active)
    if b is None:
        # Fallback: use all constraints
        active = np.arange(size)
        b = lp_solve_constraints(a, M, solver, active)
        if b is None:
            return None

    # Check for violations of dropped constraints
    for _ in range(3):
        if len(active) >= size:
            break
        check_conv = fast_conv_cross(a, b)
        violated = np.where(check_conv > M * 1.005)[0]
        # Only consider violations that are actually significant
        violated = violated[check_conv[violated] > M * 1.001]
        if len(violated) == 0:
            break
        active = np.union1d(active, violated)
        b = lp_solve_constraints(a, M, solver, active)
        if b is None:
            return None

    return b


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent for fast initial progress."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n).bit_length()

    n_stages = 10
    temps = np.geomspace(50.0/n, 0.1/n, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)
            grad = grad - grad.mean()  # Project to preserve sum
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = 1.0 / (T_val * grad_norm * np.sqrt(n))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.6f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_iters=100000):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_prev = a.copy()
    size = 2 * n - 1
    no_improve = 0
    best_t = 0

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = lp_best_response(y, M, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        # Extrapolation
        momentum = (t - 1.0) / (t + 2.0)
        if momentum > 1.0:
            momentum = 1.0
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s_a

        y = y_new
        a = a_new

        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
            best_t = t
        else:
            no_improve += 1

        sc_a = fast_conv_self(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
            best_t = t

        # Reset Nesterov if stuck too long
        if no_improve > 500:
            y = best_a.copy()
            a = best_a.copy()
            no_improve = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

                        best_val = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = np.maximum(a, 0.0)
                a = a / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve >= 2:
                n_perturb = max(10, n // 30)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.2 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 300 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fast_peak_cd(a0, time_limit, rng, verbose=True):
    """Faster peak CD using vectorized operations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv_self(a).max()
    a = a / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        peak_threshold = cur_max * 0.94
        peak_lags = np.where(conv >= peak_threshold)[0]

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue

            # Vectorized: compute gain for moving from i to all j at once
            # gain[k] = sum_j_delta * a[k-j] - a[k-i] (net change in conv[k])
            # We find the best (j, delta) pair

            # For a given i, the "source" vector is:
            src = np.zeros(size)
            for k in range(max(0, i - n + 1), min(size, i + n)):
                j_idx = k - i
                if 0 <= j_idx < n:
                    src[k] = a[j_idx]

            # Find top peaks and their positions
            if len(peak_lags) > 0:
                top_peaks = peak_lags[:8]
            else:
                top_peaks = np.array([np.argmax(conv)])

            # Try a few j candidates
            j_candidates = [int(rng.integers(n)) for _ in range(5)]
            for kl in top_peaks[:4]:
                jc = (kl - i) % n
                if jc != i:
                    j_candidates.append(jc)
                jc2 = (i + kl) % n
                if jc2 != i:
                    j_candidates.append(jc2)

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in set(j_candidates):
                if j == i:
                    continue
                aj = a[j]

                # dest vector
                dst = np.zeros(size)
                for k in range(max(0, j), min(size, j + n)):
                    j_idx = k - j
                    if 0 <= j_idx < n:
                        dst[k] = a[j_idx]

                g = dst - src

                # Binary search for optimal delta
                lo, hi = 0.0, ai * 0.98
                for _ in range(12):
                    mid = (lo + hi) / 2
                    if mid < 1e-14:
                        break
                    new_conv = conv + mid * g
                    new_conv[2 * i] = conv[2 * i] - 2 * ai * mid + mid * mid
                    new_conv[2 * j] = conv[2 * j] + 2 * aj * mid + mid * mid
                    v = new_conv.max()
                    if v < best_val:
                        best_val = v
                        best_delta = mid
                        best_j = j
                        lo = mid
                    else:
                        hi = mid

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    total_improvements += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
            if no_improve >= 2:
                n_perturb = max(8, n // 40)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.15 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    FPCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_score(a):
    """Compute the evaluation score for a sequence (normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_conv_self(a)
    n = len(a)
    return float(2 * n * conv.max() / (s ** 2))


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        # Try multiple structured initializations
        best_init = None
        best_init_score = float('inf')
        for trial in range(5):
            x = np.linspace(0.0, 1.0, n_base)
            choice = rng.integers(0, 6)
            if choice == 0:
                a_init = np.cos(np.pi * x / 2.0) ** 1.5
            elif choice == 1:
                a_init = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
            elif choice == 2:
                a_init = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
            elif choice == 3:
                a_init = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2) + 0.1
            elif choice == 4:
                a_init = np.ones(n_base) + 0.1 * np.sin(4 * np.pi * x)
            else:
                a_init = 0.4 + 0.6 * np.exp(-2.0 * (x - 0.5) ** 2)
            a_init = np.maximum(a_init, 0.0) + 0.005 * rng.random(n_base)
            a_init = a_init / a_init.sum()
            sc = compute_score(a_init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_init.copy()
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp1")

    # Phase 3: Fast peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Fast Peak CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = fast_peak_cd(best_seq, t3, rng, verbose=True)
        try_update(a_out, "fpcd1")

    # Phase 4: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: LP Nesterov R2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: Try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(n_base + 800, 4000)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: Larger n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 40 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t5a = t5 * 0.6
        a_out, _ = nesterov_lp_optimize(init_long, t5a, solver, verbose=False)
        try_update(a_out, f"lp_long{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5b = min(remaining * 0.3, 60.0)
            a_out2, _ = fast_peak_cd(best_seq, t5b, rng, verbose=False)
            try_update(a_out2, "fpcd_long")

    # Phase 6: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t6a = min(remaining * 0.4, 80.0)
        a_out, _ = fast_peak_cd(best_seq, t6a, rng, verbose=True)
        try_update(a_out, "fpcd_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.5, 60.0)
            a_out, _ = lse_gradient_descent(best_seq, t6b, verbose=False)
            try_update(a_out, "lse_final")

    # Phase 7: Random restarts
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Random restarts (budget={remaining:.0f}s) ===")
    if remaining > 30:
        n_restarts = max(2, int(remaining // 60))
        for ri in range(n_restarts):
            if time.monotonic() - t_global > time_budget:
                break
            t_restart = min((time_budget - (time.monotonic() - t_global)) * 0.7, 120.0)
            if t_restart < 15:
                break

            # Perturb best
            a_pert = best_seq.copy()
            n_pert = max(20, len(a_pert) // 10)
            idx_pert = rng.choice(len(a_pert), size=n_pert, replace=False)
            a_pert[idx_pert] *= (1.0 + 0.3 * rng.standard_normal(n_pert))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            # Quick LP optimization
            t_sub = t_restart * 0.6
            a_out, _ = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
            try_update(a_out, f"restart{ri}")

            # Quick CD
            remaining_r = time_budget - (time.monotonic() - t_global)
            if remaining_r > 10:
                t_cd = min(remaining_r * 0.3, 40.0)
                a_out, _ = fast_peak_cd(a_pert, t_cd, rng, verbose=False)
                try_update(a_out, f"restart_cd{ri}")

    # Final evaluation
    if best_seq is not None:
        final_score = compute_score(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
