import numpy as np

from scipy.signal import correlate
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * len(a) * conv.max() / (s * s))

def normalize_sum(a, target):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target / len(a)
    return a * target / s


def compute_sgrad(a, weights):
    """Smoothed gradient: grad[j] = 2 * sum_k weights[k]*a[k-j] for valid indices."""
    n = len(a)
    return 2.0 * correlate(weights, a, mode='valid')


def smoothed_descent(a0, time_limit, tau_init=0.3, tau_final=0.005, lr_init=0.05,
                     verbose=True, seed=0, use_adam=True, momentum_blend=0.85):
    """Log-sum-exp smoothed subgradient descent with Adam and momentum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_sum(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    momentum_vec = np.zeros(n)

    t0 = time.monotonic()
    t_iter = 0
    no_improve = 0
    best_iter_gap = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))

        # Adaptive smoothing: start wide, narrow down
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.7)

        conv = fast_autocorr(a)
        conv_max = conv.max()
        if conv_max < 1e-15:
            break

        # Softmax weights (numerically stable)
        scaled = (conv - conv_max) / tau
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        weights = exp_scaled / exp_scaled.sum()

        # Gradient
        grad = compute_sgrad(a, weights)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a_norm = np.linalg.norm(a)
        lr = lr_init * (a_norm / (grad_norm * n)) ** 0.5 * (1.0 - 0.5 * frac)
        lr = min(lr, 0.5 * a_norm / max(grad_norm, 1e-10))

        if use_adam:
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
            m_hat = m_adam / (1 - beta1 ** t_iter)
            v_hat = v_adam / (1 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
        else:
            step = lr * grad

        # Momentum
        momentum_vec = momentum_blend * momentum_vec + (1 - momentum_blend) * step
        a_new = a - momentum_vec

        # Project: non-negative + normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, norm)

        # Evaluate
        score_new = fast_autocorr(a_new).max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Restart if stalled
        if no_improve > 2000:
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = normalize_sum(a, norm)
            momentum_vec = np.zeros(n)
            no_improve = 0

        if verbose and t_iter % 2000 == 0:
            print(f"    SSD iter={t_iter} ({elapsed:.0f}s): tau={tau:.5f}, "
                  f"score={conv_max:.8f}, best={best_score:.8f}")

    return best_a, best_score


def paired_cd(a0, time_limit, rng, verbose=True, max_evals=50000):
    """Aggressive paired coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_sum(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0
    n_evals = 0
    phi = 0.6180339887498949

    while time.monotonic() - t0 < time_limit and n_evals < max_evals:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for p in range(n // 2):
            if time.monotonic() - t0 > time_limit or n_evals >= max_evals:
                break
            i = perm[2 * p % n]
            j = perm[(2 * p + 1) % n]
            if i == j:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-10 and aj < 1e-10:
                continue

            # Golden section on mixing parameter
            # new_a[i] = ai*(1-t) + aj*t, new_a[j] = aj*(1-t) + ai*t
            # t in [0, 1]
            lo, hi = 0.0, 1.0

            def obj_at_t(t_val):
                a_try = a.copy()
                a_try[i] = ai * (1 - t_val) + aj * t_val
                a_try[j] = aj * (1 - t_val) + ai * t_val
                c = fast_autocorr(a_try)
                return c.max()

            c1 = lo + (1 - phi) * (hi - lo)
            c2 = lo + phi * (hi - lo)
            f1 = obj_at_t(c1)
            f2 = obj_at_t(c2)
            n_evals += 2

            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = lo + (1 - phi) * (hi - lo)
                    a_try = a.copy()
                    a_try[i] = ai * (1 - c1) + aj * c1
                    a_try[j] = aj * (1 - c1) + ai * c1
                    f1 = fast_autocorr(a_try).max()
                    n_evals += 1
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + phi * (hi - lo)
                    a_try = a.copy()
                    a_try[i] = ai * (1 - c2) + aj * c2
                    a_try[j] = aj * (1 - c2) + ai * c2
                    f2 = fast_autocorr(a_try).max()
                    n_evals += 1

            t_opt = c1 if f1 < f2 else c2
            a_try = a.copy()
            a_try[i] = ai * (1 - t_opt) + aj * t_opt
            a_try[j] = aj * (1 - t_opt) + ai * t_opt
            sc = fast_autocorr(a_try).max()

            if sc < best_score - 1e-13:
                a = a_try
                best_score = sc
                best_a = a.copy()
                pass_improved = True
                no_improve = 0

        if not pass_improve if False else not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

            if ai < 1e-12:

            # Try scaling a[i] by factor in [0.5, 1.5]

def dual_fit_direction(a, conv, n_top=20):
    """
    Compute a direction based on dual fit flattening.
    At optimum, the weighted sum of shifted a should be constant where a>0.
    We compute the residual and use it as a direction.
    """
    n = len(a)
    size = 2 * n - 1
    conv_max = conv.max()

    # Top-k peaks
    top_idx = np.argsort(conv)[-n_top:]
    top_vals = conv[top_idx]
    top_vals = top_vals / top_vals.sum()

    # Weighted sum of shifted a: result[j] = sum_k top_vals[k] * a[k-j] (valid)
    weight_vec = np.zeros(size)
    weight_vec[top_idx] = top_vals
    dual = correlate(weight_vec, a, mode='valid')

    # Where a > 0, dual should be constant (= conv_max / sum(a) at optimum)
    positive = a > 1e-10 * a.max()
    if positive.sum() < 2:
        return np.zeros(n)

    target = np.mean(dual[positive])
    residual = target - dual

    # Direction: increase a where residual > 0 (dual is too low), decrease where residual < 0
    direction = np.zeros(n)
    direction[positive] = residual[positive]

    # Make it zero-sum (so normalization is roughly preserved)
    direction -= np.mean(direction[positive]) * positive

    return direction


def dual_flatten_descent(a0, time_limit, verbose=True, seed=0):
    """Descent using dual-fit flattening direction."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_sum(np.maximum(np.asarray(a0, dtype=np.float64), 0.0), norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    t_iter = 0
    momentum_vec = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        conv = fast_autocorr(a)
        conv_max = conv.max()

        # Compute dual-fit direction
        direction = dual_fit_direction(a, conv, n_top=min(30, n // 10))
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break

        # Also compute smoothed gradient as fallback
        tau = 0.1 * (1 - frac) + 0.005
        scaled = (conv - conv_max) / tau
        exp_scaled = np.exp(scaled - scaled.max())
        weights = exp_scaled / exp_scaled.sum()
        sgrad = compute_sgrad(a, weights)

        # Blend directions
        blended = 0.5 * direction / dir_norm * np.linalg.norm(sgrad) + 0.5 * sgrad
        bl_norm = np.linalg.norm(blended)
        if bl_norm < 1e-15:
            break

        # Adaptive step
        lr = 0.02 * (1 - 0.3 * frac)

        a_new = a - lr * momentum_vec

        if score_new < best_score:

        else:

        if verbose and t_iter % 1000 == 0:

def structured_init(n, rng, style=0):
    x = np.linspace(0.0, 1.0, n)

        a = np.sin(np.pi * x) ** 1.2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2

        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8

        a = 1.0 + 0.25 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = x ** 0.5 * (1 - x) ** 0.3
    elif style == 8:
        a = (x + 0.01) ** 1.5
    elif style == 9:
        a = 1.0 / (x + 0.1)
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.05):

    noise = np.maximum(noise, 0.05)

    if s < 1e-15:

    return ap / s

        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = paired_cd(a_norm, t3, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Single CD (10% time) ===
    print(f"\n=== Phase 4: Single CD n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = single_cd(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"scd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Multi-restart with perturbations (20% time) ===
    print(f"\n=== Phase 5: Multi-restart ===")
    t5_total = min(remaining * 0.20, 300.0)
    n_restarts = 4
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t5 = min(t5_each, remaining - 50)
        if t5 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.015 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = smoothed_descent(a_norm, t5, tau_init=0.3, tau_final=0.003,
                                    lr_init=0.06, verbose=False, seed=100 + r * 13)
        try_update(a_out, f"restart{r}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Different lengths (10% time) ===
    print(f"\n=== Phase 6: Length exploration ===")
    for dl in [200, -200, 400]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        n_try = len(best_seq) + dl
        if n_try < 500:
            continue
        t6 = min(remaining * 0.15, 120.0)
        if t6 < 20:
            break
        print(f"  Trying n={n_try}")
        init = interpolate_seq(best_seq, n_try)
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = smoothed_descent(a_norm, t6, tau_init=0.4, tau_final=0.005,
                                    lr_init=0.07, verbose=False, seed=500 + n_try)
        try_update(a_out, f"len{n_try}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final combined polish (remaining time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} ===")
    t7 = remaining * 0.5
    if t7 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_descent(a_norm, t7, tau_init=0.2, tau_final=0.002,
                                    lr_init=0.04, verbose=True, seed=999)
        try_update(a_out, f"final_ssg_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final paired CD ===
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final PCD n={n_best} ===")
        t8 = min(remaining * 0.6, 150.0)
        if t8 > 20:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = paired_cd(a_norm, t8, rng, verbose=True)
            try_update(a_out, f"final_pcd_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: Final dual-flatten ===
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final DFD n={n_best} ===")
        t9 = min(remaining * 0.7, 100.0)
        if t9 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = dual_flatten_descent(a_norm, t9, verbose=True, seed=777)
            try_update(a_out, f"final_dfd_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure all values are in valid range
    result = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in result]
