                i2 = j - k
                if 0 <= i2 < n:
                    grad[j] += weight * a[i2]
        
        grad /= n_active  # average
        
        # Projected gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new * norm / a_new.sum()
        
        score_new = fast_conv(a_new).max()

            best_score = score_new

            patience = 0
            step = min(step * 1.005, 5.0)

            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            s_others = a.sum() - aj
            
            # Compute contribution of a[j] to each convolution value
            # (a*a)[k] = bg[k] + a[j] * (a[k-j] if valid) + a[j] * (a[j-k] if valid) + a[j]^2 * delta_{k,2j}
            # More precisely:
            # When we change a[j] from aj to x:
            # (a_new*a_new)[k] = conv[k] - aj*c_j[k] - aj*d_j[k] - aj^2*delta_{k,2j} + x*c_j[k] + x*d_j[k] + x^2*delta_{k,2j}
            # where c_j[k] = a[k-j] if 0<=k-j<n and k-j != j
            #       d_j[k] = a[j-k] if 0<=j-k<n and j-k != j
            
            c_j = np.zeros(size)
            d_j = np.zeros(size)
            for k in range(size):
                i1 = k - j
                if 0 <= i1 < n and i1 != j:
                    c_j[k] = a[i1]
                i2 = j - k
                if 0 <= i2 < n and i2 != j:
                    d_j[k] = a[i2]
            
            delta = (2 * j < size)
            
            # bg[k] = conv[k] - aj*c_j[k] - aj*d_j[k] - aj^2*delta (for k=2j)
            bg = conv - aj * c_j - aj * d_j
            if delta:
                bg[2*j] -= aj * aj
            
            # We want to minimize max_k (bg[k] + x*c_j[k] + x*d_j[k] + x^2*delta_{k,2j}) / (s_others + x)^2
            # Try a grid of x values
            x_candidates = np.linspace(0, norm, 20)
            best_x = aj
            best_val = float('inf')
            
            for x in x_candidates:
                new_conv = bg + x * (c_j + d_j)
                if delta:
                    new_conv[2*j] += x * x
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                val = new_conv.max() / (new_sum * new_sum)
                if val < best_val:
                    best_val = val
                    best_x = x
            
            # Bisection refinement
            if best_x != aj:
                lo = min(aj, best_x) * 0.9
                hi = max(aj, best_x) * 1.1
                lo = max(0, lo)
                hi = min(norm, hi)
                for _ in range(10):
                    mid = (lo + hi) / 2
                    new_conv = bg + mid * (c_j + d_j)
                    if delta:
                        new_conv[2*j] += mid * mid
                    new_sum = s_others + mid
                    if new_sum > 1e-15:
                        val = new_conv.max() / (new_sum * new_sum)
                        if val < best_val:
                            best_val = val
                            best_x = mid
                            lo = mid
                        else:
                            hi = mid
                
                if abs(best_x - aj) > 1e-10:
                    a[j] = best_x
                    a = a * norm / a.sum()
                    new_score = fast_conv(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = aj
                        a = a * norm / a.sum()
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def spectral_flatten(a0, time_limit, verbose=False):
    """
    Iterative spectral flattening:
    1. Compute DFT of current sequence
    2. Modify amplitude toward flat spectrum
    3. Inverse DFT
    4. Project to non-negative, fixed sum
    5. Repeat
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a * norm / a.sum()
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Use a larger FFT to approximate linear convolution in frequency domain
    fft_size = 2 * n
    A = rfft(a, fft_size)
    target_mag = np.ones(len(A))
    
    relaxation = 0.5
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Interpolate amplitude toward target
        A = rfft(a, fft_size)
        new_mag = (1 - relaxation) * np.abs(A) + relaxation * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = a_new * norm / a_new.sum()
        
        score = fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def perturb_and_reoptimize(a, rng, scale, time_budget, solver):
    """Perturb the sequence and re-optimize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.2)
    a_pert = a * noise
    a_pert = normalize(a_pert)
    
    # Quick optimization
    a_out, _ = nesterov_lp(a_pert, time_budget, solver, seed=int(rng.integers(10000)), verbose=False)
    return a_out

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n)
    return a_interp / s
