def fft_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def max_conv(a):
    return float(fft_conv_self(a).max())


def norm_seq(a, target=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    return a * target / s

def build_conv_rows(a, lags):
    """Build rows of conv matrix: row l corresponds to conv(a,b)[lags[l]] as linear func of b."""
    n = len(a)
    rows = []
    for k in lags:
        row = np.zeros(n)
        # conv(a,b)[k] = sum_j b[j] * a[k-j] where 0<=k-j<n
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def solve_lp(a, M, lags, solver):
    """Solve max sum(b) s.t. conv(a,b)[k]<=M for k in lags, b>=0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_rows(a, lags)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None, None
    val = np.maximum(val, 0.0)
    # Try to get dual variables
    duals = None
    try:
        duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
        if duals is not None and duals.shape[0] == len(lags):
            duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    return val, duals


def lp_nesterov(a0, time_budget, solver, verbose=True):
    """LP Nesterov with adaptive active sets and dual-guided refinement."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = norm_seq(a0, target)
    best_a = a.copy()
    best_mc = max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    no_improve = 0
    k_scale = 1.0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_budget:
            break

        conv_y = fft_conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        # Adaptive active set
        # Include top peaks and some spread
        sorted_idx = np.argsort(conv_y)[::-1]
        top_k = min(size, max(30, int(n * 0.02 * k_scale)))
        active = sorted_idx[:top_k]

        # Also add some medium peaks for diversity
        med_start = top_k
        med_end = min(size, top_k * 3)
        if med_end > med_start:
            med_candidates = sorted_idx[med_start:med_end]
            n_med = min(20, len(med_candidates))
            if n_med > 0:
                med_selected = med_candidates[np.random.choice(len(med_candidates), n_med, replace=False)]
                active = np.unique(np.concatenate([active, med_selected]))

        b, duals = solve_lp(y, M, active, solver)

        if b is None:
            # Retry with just top peaks
            b, duals = solve_lp(y, M, sorted_idx[:max(10, top_k // 2)], solver)

        if b is None:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-13:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        g = b * target / b_sum

        # Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-13:
            y_new = y.copy()
        else:
            y_new = y_new * target / sy

        if t == 1:
            a_new = g.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-13:
                a_new = y_new.copy()
            else:
                a_new = a_new * target / sa

        y = y_new
        a = a_new

        mc_y = max_conv(y)
        if mc_y < best_mc - 1e-10:
            best_mc = mc_y
            best_a = y.copy()
            no_improve = 0
            k_scale = max(0.3, k_scale * 0.9)
        else:
            no_improve += 1

        mc_a = max_conv(a)
        if mc_a < best_mc - 1e-10:
            best_mc = mc_a
            best_a = a.copy()
            no_improve = 0

        if verbose and t % 50 == 0:
            el = time.monotonic() - t0
            print(f"    LP-Ne t={t} ({el:.0f}s): best_mc={best_mc:.8f}, M={M:.6f}, k={len(active)}, no_imp={no_improve}")

            no_improve = 0

def peak_pair_cd(a0, time_budget, rng, verbose=True):
    """
    Peak-focused pair coordinate descent.
    For each top peak, find the best (i,j) pair to move mass that reduces it.
    """
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = norm_seq(a0, target)
    best_a = a.copy()
    best_mc = max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    total_moves = 0
    stalled = 0

    while time.monotonic() - t0 < time_budget:
        conv = fft_conv_self(a)
        cur_max = conv.max()

        # Focus on top peaks
        n_peaks = min(10, size)
        peak_lags = np.argsort(conv)[::-1][:n_peaks]

        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            i = perm[idx]
            ai = a[i]
            if ai < target * 1e-12:
                continue

            # Candidate j's: bias towards positions that interact with peaks
            j_candidates = set()
            for kl in peak_lags:
                # j such that conv[kl] involves a[i]*a[j]
                # conv[kl] = sum_p a[p]*a[kl-p], so a[i]*a[kl-i] contributes
                j1 = kl - i
                if 0 <= j1 < n and j1 != i:
                    j_candidates.add(j1)
                j2 = kl - (n - 1 - i)  # for the other direction
                if 0 <= j2 < n and j2 != i:
                    j_candidates.add(j2)

            # Add random candidates
            for _ in range(5):
                j_candidates.add(int(rng.integers(n)))
            j_candidates.discard(i)

            if not j_candidates:
                continue

            best_v = cur_max
            best_delta = 0.0
            best_j = -1

            for j in list(j_candidates)[:15]:
                aj = a[j]

                # Compute delta in conv when moving d from i to j
                # conv_new[k] = conv[k] + d*(contrib_j[k] - contrib_i[k])
                # where contrib_j[k] = a[k-j] (if valid) + correction for k=2j
                # Simpler: just compute the actual change for top peaks only

                # For each peak lag, compute the effect
                # Moving d from i to j:
                # a[i] -> a[i]-d, a[j] -> a[j]+d
                # conv[k] = sum_p a[p]*a[k-p]
                # Change = d * (sum over p where k-p==j: a[p] + sum over p==j: a[k-p])
                #        - d * (sum over p where k-p==i: a[p] + sum over p==i: a[k-p])
                #        + correction for p=i and k-p=i (both change)
                # Simplified:
                # delta_conv[k] = d * (a[k-j] + a[k-j reversed]) - d * (a[k-i] + a[k-i reversed])
                # where "reversed" accounts for the symmetric term

                # Actually for linear conv: conv[k] = sum_{p=0}^{min(k,n-1)} a[p] * a[k-p] (with k-p < n)
                # The terms involving a[i]: a[i]*a[k-i] (if 0<=k-i<n) and a[k-i]*a[i] (same term, counted once in sum)
                # Wait, conv[k] = sum_p a[p]*a[k-p]. The term a[i] appears when p=i (giving a[i]*a[k-i]) 
                # and when k-p=i, i.e., p=k-i (giving a[k-i]*a[i]). These are the same term, so:
                # conv[k] contains a[i]*a[k-i] once (if both i and k-i are in [0,n-1])
                # UNLESS i == k-i, i.e., k=2i, in which case a[i]^2 appears.

                # So delta_conv[k] from moving d from i to j:
                # = d * (a[k-j] if valid) - d * (a[k-i] if valid)  [cross terms]
                # + d * (a[j] if k==2i and i==k-i... this gets complicated)
                # Let me just use the simple cross-term approximation for candidate selection

                improved = 0

def lse_adam(a0, time_budget, verbose=False):
    """LSE-smoothed Adam optimizer for the max-conv objective."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = norm_seq(a0, target)
    best_a = a.copy()
    best_mc = max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_ad = 0

    n_stages = 15
    temps = np.geomspace(50.0, 0.1, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(20, min(100, int(remaining * 2)))
        lr = 0.3 / np.sqrt(n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_budget:
                break
            t_ad += 1

            conv = fft_conv_self(a)
            mc = conv.max()
            if mc < 1e-14:
                break

            # Softmax weights over conv values
            shifted = (conv - mc) / T
            shifted = np.maximum(shifted, -60.0)
            exp_c = np.exp(shifted)
            w = exp_c / (exp_c.sum() + 1e-30)

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            # Gradient of sum_k w[k]*conv[k] w.r.t. a
            # conv[k] = sum_p a[p]*a[k-p]
            # d(conv[k])/d(a[i]) = a[k-i] + a[i] (if both valid, counting carefully)
            # Using FFT: gradient = irfft(conj(A)*W)[:n] + irfft(A*W)[:n]
            ca = irfft(np.conj(A) * W, fft_size)[:n]
            aw = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (ca + aw)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_ad)
            v_hat = v / (1.0 - beta2 ** t_ad)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-13:
                a_new = a.copy()
            else:
                a_new = a_new * target / s_new
            a = a_new

            sc = max_conv(a)
            if sc < best_mc:
                best_mc = sc
                best_a = a.copy()

    return best_a, best_mc

    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        f1 = rng.integers(1, 5)
        f2 = rng.integers(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    elif choice == 3:
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        a = np.exp(-40.0 * (x - c1) ** 2) + np.exp(-40.0 * (x - c2) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x)
    elif choice == 6:
        n_peaks = rng.integers(4, 12)
        centers = rng.uniform(0.05, 0.95, n_peaks)
        a = np.zeros(n)
        for c in centers:
            w = rng.uniform(0.02, 0.08)
            a += np.exp(-(x - c) ** 2 / (2 * w ** 2))
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x)
    elif choice == 8:
        a = np.sinc(2 * (x - 0.5)) ** 2 + 0.15
    else:
        a = 0.5 + 0.25 * np.cos(4 * np.pi * x) + 0.15 * np.cos(8 * np.pi * x)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a + 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetrize(a):
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a
    return a_sym * (a.sum() / s)


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = detect_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = get_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = get_score(best_seq)
        print(f"Init: n={n_init}, score={best_score:.6f}")
    else:
        n_init = len(best_seq)

    def update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = get_score(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (was {old:.6f}, n={len(a)}, t={el:.0f}s)")

    # === Phase 1: LSE-Adam for quick initial improvement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 1: LSE-Adam n={n_init} ({t1:.0f}s) ===")
    if t1 > 20:
        a_out, _ = lse_adam(best_seq, t1, verbose=False)
        update(a_out, "lse1")

    # === Phase 2: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 2: LP Nesterov n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov(best_seq, t2, solver, verbose=True)
        update(a_out, "lp1")

    # === Phase 3: Peak CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: Peak CD n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_pair_cd(best_seq, t3, rng, verbose=True)
        update(a_out, "cd1")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP round 2 ({t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = lp_nesterov(best_seq, t4, solver, verbose=False)
        update(a_out, "lp2")

    # === Phase 5: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: Final CD ({t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = peak_pair_cd(best_seq, t5, rng, verbose=True)
        update(a_out, "cd_final")

    # === Phase 6: Symmetrization and sparsification attempts ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 6: Post-processing ===")
        a_sym = symmetrize(best_seq)
        update(a_sym, "sym")

        # Try slight sparsification
        a_sp = best_seq.copy()
        thresh = 0.005 * a_sp.max()
        a_sp[a_sp < thresh] = 0.0
        update(a_sp, "sparse")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Multi-length (remaining={remaining:.0f}s) ===")
        for n_try in [n_init + 200, n_init - 200, n_init + 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 400 or n_try > 4000:
                continue
            t_try = min(remaining * 0.4, 50.0)
            print(f"  Trying n={n_try} ({t_try:.0f}s)")
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = lse_adam(init_try, min(t_try, 25.0), verbose=False)
            update(a_out, f"ml_lse_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = peak_pair_cd(a_out, min(remaining * 0.5, 20.0), rng, verbose=False)
                update(a_out, f"ml_cd_n{n_try}")

    # === Final: LSE-Adam from best with low temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final LSE refinement ({remaining:.0f}s) ===")
        a_out, _ = lse_adam(best_seq, min(remaining, 40.0), verbose=False)
        update(a_out, "final_lse")

    # Final evaluation
    if best_seq is not None:
        final_score = get_score(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
