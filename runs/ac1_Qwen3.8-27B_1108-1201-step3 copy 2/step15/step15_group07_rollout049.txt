def lp_best_response_step(a, M, solver_info):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)(k)<=M for all k. Returns b or None."""

    if solver is None or cp is None:
        return None
    n = len(a)
    size = 2 * n - 1
    # Build T matrix where T[k,j] = a[j] if 0<=k-j<n else 0
    # conv(a,b)(k) = sum_j a[k-j]*b[j] = sum_j T[k,j]*b[j]
    # T[k,j] = a[k-j] if max(0,k-n+1)<=j<=min(k,n-1) else 0

    for j in range(n):

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_optimize(a0, time_limit, solver_info, verbose=True):
    """LP best-response with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_conv = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0
    lr_base = 0.3

    while time.monotonic() - t0 < time_limit:
        t_elapsed = time.monotonic() - t0
        if t_elapsed > time_limit - 5:
            break

        conv = fast_autocorr(a)
        M = conv.max()
        if M < 1e-14:
            break

        # LP solve
        b = lp_best_response_step(a, M, solver_info)
        if b is None:
            stuck += 1
            if stuck > 10:
                break
            continue
        stuck = 0
        n_iters += 1

        # Normalize b to same sum as a
        b_sum = b.sum()
        g = b / b_sum

        # Line search
        best_t = 0.0
        best_val = best_conv
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_autocorr(a_try)
            max_try = conv_try.max()
            if max_try < best_val - 1e-14:
                best_val = max_try
                best_t = t

        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            a_new = a_new / s_new
            a = a_new
            if best_val < best_conv - 1e-14:
                best_conv = best_val
                best_a = a.copy()
            # Update base learning rate
            lr_base = best_t
        else:
            # No improvement from LP direction, try smaller steps
            improved = False
            for t in [0.01, 0.02, 0.03]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_autocorr(a_try)
                max_try = conv_try.max()
                if max_try < best_conv - 1e-14:
                    best_conv = max_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                stuck += 1
                if stuck > 20:
                    break

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): score={2*n*best_conv:.8f}, stuck={stuck}")

    return best_a, 2.0 * n * best_conv


def pairwise_cd(a0, time_limit, verbose=True):
    """Pairwise coordinate descent with exact delta computation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv = fast_autocorr(a)
    best_conv = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    n_passes = 0
    no_improve = 0

    # Precompute the "delta" contribution of each element
    # If we move amount d from i to j:
    # conv[2i] decreases by 2*a[i]*d - d^2
    # conv[2j] increases by 2*a[j]*d + d^2
    # For other lags k, the change is:
    #   Remove: d * (a shifted by i) from conv
    #   Add: d * (a shifted by j) to conv
    # Specifically, conv[k] changes by:
    #   -d * (a[k-i] if valid) [removing from position i]
    #   +d * (a[k-j] if valid) [adding to position j]
    # But also the cross terms: -d*a[k-i] appears in the "i removal" 
    # and +d*a[k-j] appears in the "j addition"
    # Actually, let's think of it as:
    # new_a = a with a[i]-=d, a[j]+=d
    # new_conv[k] = conv[k] - d*(a[k-i]_valid + a[i]_at_k_i) + d*(a[k-j]_valid + a[j]_at_k_j) ...
    # This is getting complicated. Let me just use the direct formula.
    #
    # The change in conv when we set a[i] -> a[i]-d and a[j] -> a[j]+d:
    # delta_conv[k] = -d * (contribution of a[i] to conv[k]) + d * (contribution of a[j] to conv[k])
    # where "contribution of a[i] to conv[k]" = a[k-i] (if 0<=k-i<n) + a[i] (if i+k-i... 
    # 
    # Actually, conv[k] = sum_m a[m]*a[k-m]
    # The terms involving a[i]: a[i]*a[k-i] (if 0<=k-i<n) and a[k-i]*a[i] (if 0<=k-i<n, same term counted once in the sum)
    # Wait, conv[k] = sum_{m: 0<=m<n, 0<=k-m<n} a[m]*a[k-m]
    # The terms where m=i: a[i]*a[k-i] (if 0<=k-i<n)
    # The terms where k-m=i, i.e., m=k-i: a[k-i]*a[i] (if 0<=k-i<n)
    # If i != k-i (i.e., k != 2i), these are two separate terms, each contributing a[i]*a[k-i]
    # If i == k-i (i.e., k = 2i), it's a single term a[i]^2
    #
    # So the "sensitivity" of conv[k] to a[i] is:
    # d(conv[k])/d(a[i]) = a[k-i] * [0<=k-i<n] + a[k-i] * [0<=k-i<n and k-i != i]
    # = a[k-i] * [0<=k-i<n] * (1 + [k != 2i])
    # = 2*a[k-i] if 0<=k-i<n and k != 2i
    # = a[i] if k = 2i (since k-i=i, and it's a[i]*a[i], derivative is 2*a[i]... wait)
    #
    # Let me redo: conv[k] = sum_m a[m]*a[k-m]
    # d/d(a[i]) conv[k] = sum_m [delta(m,i)*a[k-m] + a[m]*delta(k-m,i)]
    # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n and k-i != i)
    # Hmm, the second term: delta(k-m,i) means m=k-i, and we need 0<=m<n and 0<=k-m<n
    # m=k-i, k-m=i. So we need 0<=k-i<n and 0<=i<n. The second condition is always true.
    # Value: a[m]*1 = a[k-i] (if 0<=k-i<n)
    # So: d/d(a[i]) conv[k] = a[k-i]*[0<=k-i<n] + a[k-i]*[0<=k-i<n] = 2*a[k-i]*[0<=k-i<n]
    #
    # Wait, that's the same as before! Let me verify with k=2i:
    # d/d(a[i]) conv[2i] = 2*a[2i-i]*[0<=2i-i<n] = 2*a[i]*[0<=i<n] = 2*a[i]
    # But conv[2i] contains the term a[i]*a[i] = a[i]^2, whose derivative is 2*a[i]. ✓
    # And also terms a[i]*a[2i-m] for m != i... those don't involve a[i] as the first factor
    # unless m=i. So the total derivative is indeed 2*a[i]. ✓
    #
    # Great, so the sensitivity is simply: sens[i,k] = 2*a[k-i] if 0<=k-i<n else 0
    #
    # For moving d from i to j:
    # delta_conv[k] = -d * 2*a[k-i]*[0<=k-i<n] + d * 2*a[k-j]*[0<=k-j<n]
    # Plus the second-order term at k=2i: +d^2 (since a[i]^2 -> (a[i]-d)^2 = a[i]^2 - 2a[i]*d + d^2)
    # And at k=2j: +d^2 (since a[j]^2 -> (a[j]+d)^2 = a[j]^2 + 2a[j]*d + d^2)
    #
    # Wait, the first-order formula already accounts for the a[i]^2 and a[j]^2 terms via the sensitivity.
    # Let me check: for k=2i, sensitivity = 2*a[i]. So -d*2*a[i] is the first-order change in conv[2i].
    # But the actual change is: (a[i]-d)^2 - a[i]^2 = -2*a[i]*d + d^2.
