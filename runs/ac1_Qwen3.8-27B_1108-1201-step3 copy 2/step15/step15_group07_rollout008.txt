def active_set_lp(a, M, top_k, solver):
    """Solve LP with only top_k constraints active."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    active_idx = np.argsort(conv)[-top_k:]
    
    # Build reduced constraint matrix
    T = np.zeros((top_k, n))
    for row, k in enumerate(active_idx):
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(int(k), n - 1)
        j_lo = int(k) - i_hi
        j_hi = int(k) - i_lo
        T[row, i_lo:i_hi+1] = a[j_lo:j_hi+1][::-1]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    return b_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration, adaptive active-set, and smart restarts
2. Active-set LP (top-K constraints) for speed when full LP is too slow
3. Coordinate descent polish on active lags
4. Multi-initialization: parent sequence + random restarts
Key improvements: adaptive active set sizing, multiplicative perturbation restarts,
and a dedicated final polish phase.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix_full(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_active(a, active_indices):
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi >= i_lo:
            j_lo = k - i_hi
            j_hi = k - i_lo
            T[row, i_lo:i_hi + 1] = a[j_lo:j_hi + 1][::-1]
    return T


def lp_best_response_full(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_full(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_active(a, M, active_indices, solver):
    n = len(a)
    T = build_conv_matrix_active(a, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, active_set_size=0):
    """
    Nesterov-accelerated LP best-response optimization.
    active_set_size: if > 0, use only top-K active constraints in LP
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    patience = 0
    momentum_t = 1.0
    lp_time_avg = 0.0

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_lp_start = time.monotonic()
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if active_set_size > 0:
            conv_y = fast_autocorr(y)
            size = 2 * n - 1
            k_active = min(active_set_size, size)
            active_idx = np.argsort(conv_y)[-k_active:]
            b = lp_best_response_active(y, M_curr, active_idx, solver)
        else:
            b = lp_best_response_full(y, M_curr, solver)

        t_lp = time.monotonic() - t_lp_start
        lp_time_avg = 0.9 * lp_time_avg + 0.1 * t_lp

        if b is None:
            patience += 1
            if patience > 50:
                break
            continue

        patience = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if momentum_t < 2.0:
            a_new = g.copy()
        else:
            momentum = (momentum_t - 1.0) / (momentum_t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        momentum_t += 1.0

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Restart logic
        if n_iters % 30 == 0:
            current_max = fast_autocorr(y).max()
            if current_max > best_score * 1.005:
                y = best_a.copy()
                a = best_a.copy()
                momentum_t = 1.0
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    NLP restart at iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, lp_avg={lp_time_avg*1000:.1f}ms")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_active=20, verbose=True):
    """
    Coordinate descent: shift mass between positions to reduce peak autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(54321)
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        conv = fast_autocorr(a)
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Find the best j to shift mass to
            best_gain = 0.0
            best_j = -1
            best_d = 0.0

            # Try nearby positions
            for offset in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 20, -20, 50, -50]:
                j = (i + offset) % n
                if j == i:
                    continue

                # Compute change in max autocorrelation
                # delta_conv[k] = change in c[k] when we move d from i to j
                # c[k] changes by: -d*a[i offset adjustments] + d*a[j offset adjustments]
                # Approximate: try a few d values
                for d in [a[i] * 0.1, a[i] * 0.3, a[i] * 0.5]:
                    if d > a[i]:
                        d = a[i] * 0.9
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    a[i] += d
                    a[j] -= d

                    if new_max < best_score - best_gain:
                        best_gain = best_score - new_max
                        best_j = j
                        best_d = d

            if best_j >= 0 and best_d > 0:
                a[i] -= best_d
                a[best_j] += best_d
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    no_improve += 1

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturb
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0
            elif no_improve >= 20:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multiplicative_perturb(a, rng, strength=0.1):
    """Apply multiplicative perturbation and renormalize."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    a_norm = a * norm / s
    mult = 1.0 + strength * rng.normal(0, 1, n)
    mult = np.maximum(mult, 0.01)
    a_pert = a_norm * mult
    a_pert = np.maximum(a_pert, 0)
    s_pert = a_pert.sum()
    if s_pert > 1e-15:
        a_pert = a_pert * norm / s_pert
    return a_pert
