```python
            print(f"    CD pass {pass_count} ({elapsed:.0f}s): max={best_max:.10f}")
    
    return best_a, best_max
```

Hmm, the coordinate descent above recomputes the entire base array and evaluates the objective many times per coordinate. For n=1600, this might be slow. Let me optimize.

Actually, the key observation is:
- For coordinate i, the affected k values are k in [i, i+n-1] (at most n values)
- Among these, k=2i is quadratic, the rest are linear
- The "unaffected" k values have constant values

So the objective as a function of x = a[i] is:
max(
    max_{k not in [i, i+n-1]} base[k],  (constant)
    base[2i] + x^2,  (quadratic, if 2i in range)
    max_{k in [i, i+n-1], k != 2i} (base[k] + 2*a[k-i]*x)  (max of linear functions)
)

The max of linear functions is a convex piecewise-linear function. Its minimum is at the intersection of two "upper" lines.

So the overall objective is:
max(constant, x^2 + c, PL(x))

where PL(x) is the piecewise-linear max.

The minimum of the max of these is found by:
1. Finding the minimum of PL(x) alone (at a breakpoint)
2. Finding where x^2 + c = PL(x) or = constant
3. Taking the best

This is more complex to implement exactly. For practical purposes, I'll use a combination:
- Compute the "critical points" of PL(x) (the breakpoints)
- At each critical point, evaluate the full objective
- Also check where the quadratic crosses the PL
- Take the best

But computing the upper envelope of n lines is O(n log n), and there are at most n breakpoints. So the total cost per coordinate is O(n log n), and for a full pass it's O(n^2 log n). For n=1600, that's about 1600^2 * 11 = 28 million operations, which should be fine.

Actually, let me simplify. For the 1D problem, I'll just use a refined grid + ternary search approach. The objective is convex (max of convex functions), so ternary search works.

But I need to be careful: the objective is the max of:
- A constant (the max of unaffected constraints)
- A quadratic (x^2 + c)
- Several linear functions (2*a[k-i]*x + base[k])

The max of these is convex, and ternary search will find the global minimum.

The range of x is [0, ∞), but practically, the optimum won't be more than a few times the current value. I'll use [0, max(ai*3, norm/n*5)] as the search range.

Let me also think about the LP approach. The key issue with the LP is that it can be slow if done naively with all constraints. The constraint screening (using only a subset of active constraints) is important.

The coordinate descent is faster per iteration but might get stuck in less-optimal local minima. However, it's very precise.

One more idea: **block coordinate descent**. Instead of updating one coordinate at a time, update a block of coordinates (e.g., 5-10 consecutive coordinates). This can escape local minima that single-coordinate updates can't.

For a block of coordinates I = {i1, i2, ..., im}, the subproblem is:
minimize max_k conv(k)(a_I, a_{-I})
subject to a_I >= 0

where a_{-I} is fixed. This is a small non-convex problem (quadratic constraints), but for small m (like 5-10), we can solve it with a simple grid search or interior point method.

Actually, for m=2, it's a 2D problem that we can solve with a 2D grid search. For m=3-5, we can use a nested search or a more sophisticated method.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral-inspired initialization (1/sqrt(i+1) profile)
2. LSE smooth gradient descent for fast convergence
3. LP best-response with Nesterov acceleration
4. Exact coordinate descent with ternary search (precise local optimization)
5. Multi-length exploration with interpolation
6. Final high-precision polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  Using solver (fallback): ECOS")
    return _solver


def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m + n, m] = a
    return T


def lp_solve(a, M, solver, active=None):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    if active is not None and 0 < len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        n_steps = max(20, min(500, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -100.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)

            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (T_val * np.sqrt(n)) * (1.0 / (1.0 + 0.02 * it))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.10f}")

    return best_a, best_score
