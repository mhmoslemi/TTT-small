Actually, I want to add one more idea that I think could be quite effective: **the "water-filling" approach**.

The idea is: we want all lags to have approximately equal autocorrelation values. This is like "water-filling" - we pour water (autocorrelation energy) into all lags equally.

Algorithm:
1. Start with current a.
2. Compute conv = autocorrelation(a).
3. Find the target level T (e.g., the 95th percentile of conv).
4. For lags where conv > T, we want to reduce the contribution.
5. For lags where conv < T, we want to increase the contribution.
6. Update a in the direction that achieves this.

The update direction can be computed as:
- For each "hot" lag k (conv[k] > T), the gradient w.r.t. a is related to the cross-correlation with a shifted version.
- We want to move a in the direction that reduces hot lags and increases cold lags.

This is essentially what the LSE gradient descent does, but with a sharper "target."

_solver = None

def get_solver():

        try:

            prob.solve(solver=s)

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(y, rows):
    """Build only the specified rows of the convolution matrix T."""
    n = len(y)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if hi > lo:
            T[idx, lo:hi] = y[lo:hi]
    return T


def lp_solve_screened(y, M, solver, active_indices):
    """Solve the LP best-response problem with given active constraints."""
    n = len(y)
    T = build_T_rows(y, active_indices)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)


def lp_solve_full(y, M, solver):
    """Solve LP with all constraints."""
    n = len(y)
    size = 2 * n - 1
    return lp_solve_screened(y, M, solver, np.arange(size))


def compute_pressure(a, conv, n_peaks=30):
    """Compute per-coordinate pressure: how much each coord contributes to top peaks."""
    n = len(a)
    size = 2 * n - 1
    threshold = conv.max() * 0.85
    peak_lags = np.where(conv >= threshold)[0]
    peak_lags = peak_lags[:n_peaks]
    
    pressure = np.zeros(n)
    for k in peak_lags:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if hi > lo:
            pressure[lo:hi] += a[lo:hi]
    return pressure


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent to minimize max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 14
    temps = np.geomspace(100.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(40, min(400, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -80.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            decay = 1.0 / (1.0 + 0.03 * it)
            step = norm / (T_val * np.sqrt(n)) * decay
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

                    b = b_full
                    b_sum = b.sum()
                    g = b * norm / b_sum

        # Nesterov update with adaptive momentum
        alpha_t = 2.0 / (t + 2.0)

        sc_a = fast_conv_self(a).max()

            best_a = a.copy()
            stall_count = 0

        # Backtracking: if stalled for too long, reduce momentum

def pressure_perturb(a, rng, strength=0.1):
    """Perturb the sequence based on pressure distribution to escape local optima."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.asarray(a, dtype=np.float64).copy()
    conv = fast_conv_self(a)
    pressure = compute_pressure(a, conv)
    
    # Normalize pressure
    p_max = pressure.max()
    if p_max < 1e-15:
        return a
    
    pressure_norm = pressure / p_max
    
    # Reduce high-pressure coords, increase low-pressure ones
    target = np.ones(n) / n
    delta = strength * (target - pressure_norm)
    
    n_perturb = max(20, n // 10)
    idx = rng.choice(n, size=n_perturb, replace=False)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    return a

                    a[j] = a_save_j
                    
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                a = a * norm / s_new
                new_max = fast_conv_self(a).max()
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
            if no_improve == 2:
                n_pert = max(10, n // 25)
                idx_p = rng.choice(n, size=n_pert, replace=False)
                a[idx_p] *= (1.0 + 0.12 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
        else:
            no_improve = 0

        if verbose and idx % 500 == 0 and idx > 0:
            elapsed = time.monotonic() - t0
            print(f"    CPCD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.0
    elif choice == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.3 + 0.3 * np.sin(4 * np.pi * x)
    elif choice == 7:
        # Fejer-kernel inspired
        a = np.maximum(1.0 - x, 0) * np.maximum(1.0 - x, 0)
    elif choice == 8:
        # Multi-peak
        centers = rng.uniform(0.1, 0.9, 3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-20.0 * (x - c) ** 2)
    elif choice == 9:
        # Checkbox pattern (alternating high/low)
        a = np.ones(n)
        a[::2] = 0.5
    elif choice == 10:
        a = 1.0 / (1.0 + 5.0 * np.abs(x - 0.5) ** 2)
    else:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / (0.1 + 0.15 * rng.random()))
    
    a = np.maximum(a, 0.0)
    a = a + 0.02 * rng.random(n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def eval_fn(a):
    """Evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if len(a_list) == 0:
            return float('inf')
        for val in a_list:
            if val < 0 or not np.isfinite(val):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. LSE warmup
    2. LP Nesterov (main)
    3. Target-level gradient
    4. Pressure perturbation + re-optimization
    5. Multi-length exploration
    6. Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 1: LSE GD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: Target-level descent (novel)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 2: Target-Level n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = target_level_descent(best_seq, t2, verbose=True)
        try_update(a_out, "target_lvl")

    # Phase 3: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 4: Pressure perturbation + LP re-optimization
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 4: Perturb + LP (budget={t4:.0f}s) ===")
    if t4 > 60:
        a_pert = pressure_perturb(best_seq, rng, strength=0.12)
        t4a = t4 * 0.5
        a_out, _ = nesterov_lp_optimize(a_pert, t4a, solver, verbose=False)
        try_update(a_out, "perturb_lp1")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_pert2 = pressure_perturb(best_seq, rng, strength=0.18)
            t4b = min(remaining * 0.5, 100.0)
            a_out2, _ = nesterov_lp_optimize(a_pert2, t4b, solver, verbose=False)
            try_update(a_out2, "perturb_lp2")

    # Phase 5: Target-level + LP hybrid
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Hybrid TL+LP (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = target_level_descent(best_seq, t5 * 0.4, verbose=False)
        try_update(a_out, "tl_hybrid")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out2, _ = nesterov_lp_optimize(a_out, min(remaining * 0.6, 80.0), solver, verbose=False)
            try_update(a_out2, "lp_hybrid")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 6: Multi-length (budget={t6:.0f}s) ===")
    if t6 > 60:
        n_base = len(best_seq)
        lengths_to_try = [n_base - 200, n_base + 200, n_base + 400]
        for n_try in lengths_to_try:
            if time.monotonic() - t_global > time_budget - 120:
                break
            if n_try < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            t_sub = min(remaining * 0.3, 50.0)
            if t_sub < 15:
                break
            init_try = interpolate_seq(best_seq, n_try)
            a_out, _ = nesterov_lp_optimize(init_try, t_sub, solver, verbose=False)
            try_update(a_out, f"len_{n_try}")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final LP polish n={len(best_seq)} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=True)
        try_update(a_out, "lp_final")

    # Phase 8: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 8: Coord Descent (budget={t8:.0f}s) ===")
    if t8 > 15:
        a_out, _ = coordinated_pair_cd(best_seq, t8, rng, verbose=True)
        try_update(a_out, "cd_final")

    # Final: one more quick LSE on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.7, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t9, verbose=False)
        try_update(a_out, "lse_very_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
