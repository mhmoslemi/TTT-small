import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_name = None

def _get_solver():
    global _solver_name
    if _solver_name is not None:
        return _solver_name
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_name = s
                return s
        except Exception:
            continue
    _solver_name = cp.ECOS
    return _solver_name

def _fft_size(size):
    return 1 << (size - 1).bit_length()

def _conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fs = _fft_size(size)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]

def _conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fs = _fft_size(size)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]

def _build_T(a):
    """Build the convolution matrix T where (T @ b)[i] = (conv(a,b))[i]"""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for m in range(n):
        T[m:m+n, m] = a
    return T

def _normalize(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _lp_solve(a, M, solver):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    size = 2 * n - 1
    T = _build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        if constraints[0].dual_value is not None:
            duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
            duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    return val, duals

def _lp_solve_screened(a, M, solver, threshold=0.7):
    """LP with constraint screening"""
    n = len(a)
    size = 2 * n - 1
    T = _build_T(a)
    conv_a = _conv_self(a)
    active = np.where(conv_a > M * threshold)[0]
    if len(active) < 10:
        active = np.where(conv_a > M * 0.3)[0]
    
    b_var = cp.Variable(n, nonneg=True)
    if len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    
    # Verify dropped constraints
    if len(active) < size:
        check = _conv_cross(a, val)
        violated = check > M * (1 + 1e-6)
        if np.any(violated):
            # Fall back to full LP
            return _lp_solve_full(a, M, solver)
    return val

def _nesterov_lp(a0, time_limit, solver, verbose=False):
    """LP-based mirror descent with Nesterov acceleration"""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = _normalize(a, norm)
    
    best_a = a.copy()
    best_score = _conv_self(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov lookahead
    momentum_buf = np.zeros(n)
    no_improve = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M from lookahead point
        conv_y = _conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break
        
        # Solve LP
        b = _lp_solve(a, M, solver)  # Use a (not y) for the LP constraints
        if b is None:
            b = _lp_solve_full(a, M, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-14:
            continue
        g = b * norm / b_sum  # Best response, normalized
        
        # Nesterov momentum
        # Standard: y_{t+1} = (1-alpha)*y_t + alpha*g, a_{t+1} = y_{t+1} + beta*(y_{t+1} - y_t)
        # Simpler: use momentum on the direction
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = _normalize(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            beta = np.sqrt(t) / (np.sqrt(t) + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        sc_y = _conv_self(y).max()
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        sc_a = _conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            no_improve = 0
        
        # Periodic perturbation
        if no_improve == 200:
            rng = np.random.default_rng(t)
            n_pert = max(20, n // 10)
            idx = rng.choice(n, size=n_pert, replace=False)
            y[idx] *= (1.0 + 0.3 * rng.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            y = _normalize(y, norm)
            no_improve = 0
        
        # Convergence check
        if t > 500 and no_improve > 5000:
            break
        
        if verbose and t % 500 == 0:
            print(f"      LP-ND t={t} ({time.monotonic()-t0:.0f}s): M={best_score:.8f}")
    
    return best_a, best_score

def _lse_descent(a0, time_limit, verbose=False):
    """LSE smooth approximation gradient descent"""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = _normalize(a, norm)
    
    best_a = a.copy()
    best_score = _conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = _fft_size(size)
    
    # Annealing schedule
    stages = [(200.0, 80), (80.0, 100), (30.0, 120), (10.0, 150), (3.0, 200), (1.0, 300), (0.3, 500), (0.1, 800)]
    
    for T_val, max_steps in stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        steps = min(max_steps, int(remaining * 2))
        
        for it in range(steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = _conv_self(a)
            max_c = conv.max()
            if max_c < 1e-14:
                break
            
            # LSE approximation of max
            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -300, 0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k])
            # = 2 * (w cross-correlated with a)
            W = rfft(w, fs)
            A = rfft(a, fs)
            g1 = irfft(np.conj(A) * W, fs)[:n]
            g2 = irfft(A * W, fs)[:n]
            grad = 2.0 * (g1 + g2)
            
            # Adaptive step size with backtracking
            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.01 * it))
            for trial in range(20):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = _normalize(a_new, norm)
                sc = _conv_self(a_new).max()
                if sc <= max_c + 0.1 * T_val:
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0.0)
            a = _normalize(a, norm)
            
            sc = _conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if verbose:
            print(f"      LSE T={T_val}: M={best_score:.8f}")
    
    return best_a, best_score

def _structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 12)

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _fft_size(size):

def _conv_self(a):

def _conv_cross(a, b):

    fs = _fft_size(size)
    A = rfft(a, fs)

def _eval(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def _normalize(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * (target_sum / s)

def _lp_solve(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = _build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    duals = None
    try:
        if constraints[0].dual_value is not None:
            duals = np.asarray(constraints[0].dual_value, dtype=np.float64)
            duals = np.maximum(duals, 0.0)
    except Exception:
        pass
    return val, duals

def _lp_solve_screened(a, M, solver, threshold):
    n = len(a)
    size = 2 * n - 1
    T = _build_T(a)
    conv_a = _conv_self(a)
    active = np.where(conv_a > M * threshold)[0]
    if len(active) < 20:
        active = np.where(conv_a > M * 0.3)[0]
    b_var = cp.Variable(n, nonneg=True)
    if len(active) < size:
        constraints = [T[active, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    if len(active) < size:
        check = _conv_cross(a, val)
        if check.max() > M * (1 + 1e-5):
            val2, _ = _lp_solve(a, M, solver)
            if val2 is not None:
                return val2
    return val

def _nesterov_lp(a0, time_limit, solver, use_screening, verbose):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = _normalize(a, norm)
    best_a = a.copy()
    best_score = _conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    no_improve = 0
    t_last_print = t0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = _conv_self(y)
        M = conv_y.max()
        if M < 1e-14:
            break

        if use_screening:
            thr = 0.5 if t < 100 else (0.7 if t < 500 else 0.8)
            b = _lp_solve_screened(a, M, solver, thr)
        else:
            b, _ = _lp_solve(a, M, solver)
        if b is None:
            b, _ = _lp_solve(a, M, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-14:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = _normalize(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, norm)

        y_prev = y.copy()
        y = y_new
        a = a_new

        sc_y = _conv_self(y).max()
        improved = False
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            improved = True
        sc_a = _conv_self(a).max()
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 150:
            rng_p = np.random.default_rng(t)
            n_pert = max(20, n // 15)
            idx_p = rng_p.choice(n, size=n_pert, replace=False)
            y[idx_p] *= (1.0 + 0.25 * rng_p.standard_normal(n_pert))
            y = np.maximum(y, 0.0)
            y = _normalize(y, norm)
            no_improve = 0

        if t > 1000 and no_improve > 10000:
            break

        if verbose and time.monotonic() - t_last_print > 15:
            print(f"      LP-ND t={t} ({time.monotonic()-t0:.0f}s): M={best_score:.8f}")
            t_last_print = time.monotonic()

    return best_a, best_score

def _dual_gradient_refine(a0, time_limit, solver, verbose):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = _normalize(a, norm)
    best_a = a.copy()
    best_score = _conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = _fft_size(size)

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        conv = _conv_self(a)
        M = conv.max()
        if M < 1e-14:
            break

        b, duals = _lp_solve(a, M, solver)
        if duals is None or b is None:
            break

        duals = np.maximum(duals, 0.0)
        d_sum = duals.sum()
        if d_sum < 1e-15:
            duals = np.zeros(size)
            duals[int(np.argmax(conv))] = 1.0
        else:
            duals = duals / d_sum

        A = rfft(a, fs)
        D = rfft(duals, fs)
        g1 = irfft(np.conj(A) * D, fs)[:n]
        g2 = irfft(A * D, fs)[:n]
        grad = 2.0 * (g1 + g2)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = 0.5
        accepted = False
        for trial in range(25):
            a_new = a - step * grad / grad_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = _normalize(a_new, norm)
            sc = _conv_self(a_new).max()
            if sc < M - 1e-10:
                a = a_new
                accepted = True
                break
            step *= 0.5

        if accepted:
            sc = _conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        elif it % 50 == 0 and it > 0:
            break

        if verbose and it % 50 == 0:
            print(f"      DualGrad it={it} ({time.monotonic()-t0:.0f}s): M={best_score:.8f}")

    return best_a, best_score

def _lse_descent(a0, time_limit, verbose, t_start):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = _normalize(a, norm)
    best_a = a.copy()
    best_score = _conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fs = _fft_size(size)

    if t_start > 1:
        stages = [(1.0, 200), (0.3, 400), (0.1, 600), (0.03, 800)]
    elif t_start > 0.1:
        stages = [(10.0, 100), (3.0, 150), (1.0, 200), (0.3, 300), (0.1, 500), (0.03, 800)]
    else:
        stages = [(100.0, 50), (30.0, 80), (10.0, 100), (3.0, 150), (1.0, 200), (0.3, 300), (0.1, 500), (0.03, 800)]

    for T_val, max_steps in stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        steps = min(max_steps, max(20, int(remaining * 1.5)))

        for it in range(steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = _conv_self(a)
            max_c = conv.max()
            if max_c < 1e-14:
                break

            shifted = (conv - max_c) / T_val
            shifted = np.clip(shifted, -300.0, 0.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            W = rfft(w, fs)
            A = rfft(a, fs)
            g1 = irfft(np.conj(A) * W, fs)[:n]
            g2 = irfft(A * W, fs)[:n]
            grad = 2.0 * (g1 + g2)

            step = norm / (T_val * np.sqrt(n) * (1.0 + 0.005 * it))
            for trial in range(20):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = _normalize(a_new, norm)
                sc = _conv_self(a_new).max()
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a = a - step * grad
            a = np.maximum(a, 0.0)
            a = _normalize(a, norm)

            sc = _conv_self(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"      LSE T={T_val}: M={best_score:.8f}")

    return best_a, best_score

def _structured_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 14)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.05)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-6.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 1.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-2.0 * x) + 0.5 * np.exp(-2.0 * (1 - x))
    elif choice == 7:
        a = 0.5 * np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
        a = a * (1.0 + 0.3 * np.cos(20 * np.pi * x))
        a = np.maximum(a, 0.01)
    elif choice == 8:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.8
    elif choice == 9:
        a = 0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(11 * np.pi * x)
        a = np.maximum(a, 0.05)
    elif choice == 10:
        k = rng.integers(3, 8)
        freqs = rng.uniform(1, 15, k)
        phases = rng.uniform(0, 2 * np.pi, k)
        amps = rng.uniform(0.1, 0.5, k)
        a = np.ones(n)
        for j in range(k):
            a += amps[j] * np.cos(2 * np.pi * freqs[j] * x + phases[j])
        a = np.maximum(a, 0.01)
    elif choice == 11:
        a = np.zeros(n)
        n_spikes = max(5, n // 15)
        positions = rng.choice(n, n_spikes, replace=False)
        for p in positions:
            width = rng.integers(2, max(3, n // 40))
            a[max(0, p - width):min(n, p + width + 1)] += 1.0
        a = np.maximum(a, 0.0)
    elif choice == 12:
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 2 * x) ** 2
        a = a * (1.0 + 0.2 * np.cos(2 * np.pi * 13 * x))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
        a[rng.choice(n, n // 4, replace=False)] = 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _eval(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_init = None
        best_init_score = float('inf')
        for trial in range(12):
            cand = _structured_init(n_base, rng)
            sc = _eval(cand)
            if sc < best_init_score:
                best_init_score = sc
                best_init = cand
        best_seq = best_init
        best_score = best_init_score
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t1 = min(remaining * 0.02, 12.0)
        a_out, _ = _lse_descent(best_seq, t1, False, 10.0)
        sc = _eval(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  LSE init: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
