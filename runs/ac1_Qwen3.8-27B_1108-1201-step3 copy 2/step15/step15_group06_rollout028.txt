def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + 1.0 - cssv
    idx = np.where(cond > 0)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_grad(a, t):
    """Compute gradient of soft-max objective (1/t)*log(sum exp(t*c_k)) where c=conv(a,a)."""
    n = len(a)
    c = fast_autocorr(a)
    c_shifted = c - c.max()
    exp_c = np.exp(t * c_shifted)
    Z = exp_c.sum()
    if Z < 1e-300:
        return np.zeros(n)
    w = exp_c / Z
    # grad[j] = sum_m a[m] * w[m+j] = conv(a_flip, w)[n-1:2n-1]
    a_flip = a[::-1]
    conv_size = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(A * W, fft_size)[:conv_size]
    grad = conv_result[n - 1:2 * n - 1]
    return grad


def soft_max_obj_val(a, t):
    """Compute soft-max objective value."""
    n = len(a)
    c = fast_autocorr(a)
    c_shifted = c - c.max()
    exp_c = np.exp(t * c_shifted)
    Z = exp_c.sum()
    if Z < 1e-300:
        return c.max()
    return (c.max() + np.log(Z)) / t

def build_T_rows(a, active_idx):
    """Build rows of the convolution matrix for given active indices."""
    n = len(a)
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                T[row, j] = a[idx]
    return T


def lp_solve_max_sum(a, M, solver, active_idx):
    """Solve max sum(b) s.t. (a*b)(k) <= M for k in active_idx, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = build_T_rows(a, active_idx)
    cons = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b.value is None:
        return None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def lp_solve_min_max(a, solver, active_idx, normalize_sum=1.0):
    """Solve min t s.t. (a*b)(k) <= t for k in active, sum(b)=normalize_sum, b>=0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    T = build_T_rows(a, active_idx)
    cons = [T @ b <= t, cp.sum(b) == normalize_sum, b >= 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b.value is None:
        return None, None
    val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    t_val = float(t.value) if t.value is not None else float('inf')
    if val.sum() < 1e-15:
        return None, t_val
    return val, t_val

def coord_descent_fast(a0, time_limit, rng, verbose=True):
    """Fast coordinate descent - only adjust top candidates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            ai = a[i]
            if ai < 1e-10 * norm:
                continue

            # Try a few delta values
            best_delta = 0.0
            best_val = best_score

            for frac in [0.1, 0.25, 0.5, 0.75, -0.1, -0.25, -0.5]:
                delta = ai * frac
                new_val = ai + delta
                if new_val < 0:
                    continue
                a_try = a.copy()
                a_try[i] = new_val
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                cv = fast_autocorr(a_try)
                val = cv.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_delta = delta

            if abs(best_delta) > 1e-14:
                a[i] += best_delta
                s_try = a.sum()
                a = a * norm / s_try
                cv = fast_autocorr(a)
                val = cv.max()
                if val < best_score - 1e-13:
                    best_score = val
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.8
    elif style == 6:
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif style == 7:
        a = np.sin(np.pi * x) ** 0.6 * (1.0 + 0.15 * np.cos(10 * np.pi * x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def perturb_seq(a, rng, scale=0.03):
    """Perturb a normalized sequence."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    ap = a * noise
    s = ap.sum()
    if s < 1e-15:
        return a.copy()
    return ap / s


def propose_candidate():
    """
    Main search function. Uses multiple optimization strategies:
    1. Nesterov LP with small adaptive active set (primary)
    2. Soft-max first-order Nesterov (diversification)
    3. Coordinate descent and pair swaps (polishing)
    4. Multi-length exploration
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_best_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        n_init = 1600
        for sty in range(8):
            cand = structured_init(n_init, sty)
            sc = evaluate_safe(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            old_score = best_score
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, delta={old_score-sc:.6f})")

    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)

    # === Phase 1: Soft-max Nesterov (diversify, 15% time) ===
    print(f"\n=== Phase 1: Soft-max Nesterov n={n_base} ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 40:
        a_norm = best_seq / best_seq.sum()  # normalize to sum=1
        a_out, _ = nesterov_softmax(a_norm, t1, verbose=True, seed=0)
        # Convert back: evaluate at any scale
        try_update(a_out, f"softmax_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 2: LP Nesterov with small active set (40% time) ===
    print(f"\n=== Phase 2: LP Nesterov (K=40) n={len(best_seq)} ===")
    t2 = min(remaining * 0.40, 500.0)
    if t2 > 60:
        n_cur = len(best_seq)
        norm = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_fast(a_norm, t2 * 0.6, solver, K_active=40, verbose=True, seed=0)
        try_update(a_out, f"lp40_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            n_cur = len(best_seq)
            norm = np.sqrt(2.0 * n_cur)
            a_norm = best_seq * norm / best_seq.sum()
            a_out2, _ = nesterov_lp_fast(a_norm, min(t2 * 0.4, remaining - 30), solver, K_active=30, verbose=True, seed=100)
            try_update(a_out2, f"lp30_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 3: Perturbation + LP restart (15% time) ===
    print(f"\n=== Phase 3: Perturbation restarts ===")
    t3_total = min(remaining * 0.15, 200.0)
    n_restarts = 3
    t3_each = t3_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t3 = min(t3_each, remaining - 40)
        if t3 < 30:
            break
        n_cur = len(best_seq)
        norm = np.sqrt(2.0 * n_cur)
        scale = 0.02 + 0.01 * r
        pert = perturb_seq(best_seq, rng, scale)
        a_norm = pert * norm / pert.sum()
        K = 30 + 10 * r
        a_out, _ = nesterov_lp_fast(a_norm, t3, solver, K_active=K, verbose=False, seed=200 + r * 13)
        try_update(a_out, f"pert{r}_K{K}_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Pair CD polishing (10% time) ===
    print(f"\n=== Phase 4: Pair CD n={len(best_seq)} ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        n_cur = len(best_seq)
        norm = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Coordinate descent (8% time) ===
    print(f"\n=== Phase 5: Coord Descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        n_cur = len(best_seq)
        norm = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = coord_descent_fast(a_norm, t5, rng, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 6: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    t6_total = min(remaining * 0.10, 150.0)
    lengths_to_try = [n_base - 200, n_base + 200, n_base + 400]
    t6_each = t6_total / max(1, len(lengths_to_try))
    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        n_try = max(500, min(3000, n_try))
        if n_try == len(best_seq):
            continue
        t6 = min(t6_each, remaining - 30)
        if t6 < 20:
            break
        print(f"  Trying n={n_try}")
        init_try = interpolate_seq(best_seq, n_try)
        norm_try = np.sqrt(2.0 * n_try)
        a_norm = init_try * norm_try / init_try.sum()
        a_out, _ = nesterov_lp_fast(a_norm, t6, solver, K_active=40, verbose=False, seed=500)
        try_update(a_out, f"len{n_try}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final LP polish (12% time) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP polish n={n_best} ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30:
        norm = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_fast(a_norm, t7, solver, K_active=60, verbose=True, seed=999)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 8: Final CD polish ===
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_best} ===")
        t8 = min(remaining * 0.10, 60.0)
        if t8 > 10:
            norm = np.sqrt(2.0 * n_best)
            a_norm = best_seq * norm / best_seq.sum()
            a_out, _ = coord_descent_fast(a_norm, t8, rng, verbose=False)
            try_update(a_out, f"cd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 9: One more LP with larger active set ===
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 9: LP K=80 final n={n_best} ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 20:
            norm = np.sqrt(2.0 * n_best)
            a_norm = best_seq * norm / best_seq.sum()
            a_out, _ = nesterov_lp_fast(a_norm, t9, solver, K_active=80, verbose=True, seed=777)
            try_update(a_out, f"lp80_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
