1. Compute c = conv(a, a)
2. Find the top K peak positions p_1, ..., p_K
3. For each peak position p_k, compute the subgradient g_k[i] = 2*a[p_k - i] (valid indices)
4. Combined subgradient: g = sum_k w_k * g_k where w_k are weights (e.g., proportional to how much each peak exceeds the average)
5. Update: a[i] = a[i] - eta * g[i] + eta * sum(g)/n (to maintain sum)
6. Project to non-negative: a = max(a, 0)
7. Renormalize: a = a * S / sum(a)
8. Repeat

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    if v.sum() <= S:
        return v + (S - v.sum()) / n
    # Binary search for threshold
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n + 1) > (css - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if weights.sum() < 1e-15:
            break
        weights = weights / weights.sum()
        
        # Compute combined subgradient
        # At peak p, subgrad[i] = 2*a[(p-i) mod n]... wait, not mod n.
        # For full convolution, (a*a)[p] = sum_j a[j]*a[p-j] where 0<=j<n and 0<=p-j<n.
        # d(a*a)[p]/da[i] = 2*a[p-i] if 0<=p-i<n (and i != p-i), or a[i] if 2i=p
        # Simplified: subgrad[i] = sum over peaks p of weight_p * 2 * a[p-i] (if valid)
        
        grad = np.zeros(n)
        for idx in range(K):
            p = peak_indices[idx]
            w = weights[idx]
            # For position i in [0, n-1], the contribution is w * 2 * a[p-i] if 0 <= p-i < n
            valid_start = max(0, p - (n - 1))
            valid_end = min(n - 1, p)
            for i in range(valid_start, valid_end + 1):
                j = p - i
                if 0 <= j < n:
                    grad[i] += w * 2.0 * a[j]
        
        # Also account for the fact that we need to preserve the sum
        # Subgradient direction: decrease a[i] where grad[i] is large
        # To preserve sum: a_new[i] = a[i] - eta * (grad[i] - grad.mean())
        
        grad_centered = grad - grad.mean()
        a_new = a - eta * grad_centered
        a_new = np.maximum(a_new, 0)
        a_new = normalize_to(a_new, norm_val)
        
        conv_new = fast_autocorr(a_new)
        score_new = conv_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            n_stuck = 0
            # Decay eta slowly
            eta = max(0.1 / n, eta * 0.999)
        else:
            # Try smaller step
            a_try = a - eta * 0.1 * grad_centered
            a_try = np.maximum(a_try, 0)
            a_try = normalize_to(a_try, norm_val)
            conv_try = fast_autocorr(a_try)
            if conv_try.max() < best_score - 1e-12:
                best_score = conv_try.max()
                best_a = a_try.copy()
                a = a_try
                conv = conv_try
                n_stuck = 0
                eta = max(0.01 / n, eta * 0.5)
            else:
                n_stuck += 1
                eta = eta * 0.9
                if n_stuck > 200:
                    # Perturb and reset
                    noise = np.random.default_rng(time.time_ns() % 2**31).normal(0, 0.01 * norm_val / n, n)
                    a_pert = a + noise
                    a = normalize_to(a_pert, norm_val)
                    conv = fast_autocorr(a)
                    eta = 1.0 / n
                    n_stuck = 0
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_score:.8f}, peak={conv_new.max():.4f}")
    
    return best_a, best_score

```python
def mirror_subgradient_descent(a0, time_limit, norm_val, K=5, verbose=True):
    n = len(a0)
    a = normalize_to(np.asarray(a0, dtype=np.float64), norm_val)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    eta = 2.0 / n
    n_stuck = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Find top K peaks
        K_actual = min(K, len(conv))
        peak_indices = np.argpartition(conv, -K_actual)[-K_actual:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
        
        # Weight peaks
        avg_cv = conv.sum() / len(conv)
        weights = np.maximum(conv[peak_indices] - avg_cv, 0)
        if weights.sum() < 1e-15:
            break
        weights = weights / weights.sum()
        
        # Compute combined subgradient
        grad = np.zeros(n)
        for idx in range(K_actual):
            p = int(peak_indices[idx])
            w = weights[idx]
            i_lo = max(0, p - n + 1)
            i_hi = min(n - 1, p)
            if i_lo > i_hi:
                continue
            indices = np.arange(i_lo, i_hi + 1)
            # At position i, gradient contribution is 2 * a[p - i]
            a_idx = p - indices
            grad[indices] += w * 2.0 * a[a_idx]
        
        # Center the gradient to preserve sum
        grad_mean = grad.mean()
        a_new = a - eta * (grad - grad_mean)
        a_new = np.maximum(a_new, 0)
        a_new = normalize_to(a_new, norm_val)
        
        conv_new = fast_autocorr(a_new)
        score_new = conv_new.max()
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            n_stuck = 0
            eta *= 0.999
        else:
            n_stuck += 1
            if n_stuck % 50 == 0:
                eta *= 0.8
            if n_stuck > 500:
                # Random perturbation to escape
                noise = rng.normal(0, 0.05 * norm_val / n, n)
                a_pert = np.maximum(a + noise, 0)
                a = normalize_to(a_pert, norm_val)
                conv = fast_autocorr(a)
                eta = 2.0 / n
                n_stuck = 0
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
```

This should be very fast. Each iteration is O(n) for the gradient + O(n log n) for the FFT. For n=1600, each iteration is ~10μs for gradient + ~50μs for FFT = ~60μs. So we can do ~16,000 iterations per second. In 400 seconds, that's ~6.4M iterations. More than enough.
