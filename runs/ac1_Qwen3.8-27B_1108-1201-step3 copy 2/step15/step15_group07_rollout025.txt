```python
    # So (w ⋆ a)[j] = conv(w, a_rev)[j + n_a - 1]
    
    a_rev = a[::-1]
    size_conv = n_w + n_a - 1
    fft_size2 = 1 << (size_conv - 1).bit_length()
    W2 = rfft(w, fft_size2)
    AR = rfft(a_rev, fft_size2)
    conv_result = irfft(W2 * AR, fft_size2)[:size_conv]
    return conv_result[n_a-1:n_a-1+n_w]  # (w ⋆ a)[j] for j = 0, ..., n_w-1
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft

# --- Helper functions ---

def fast_autocorr(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgrad(a, conv, top_k=30):
    """Subgradient of max(conv) using top-k active set."""
    n = len(a)
    size = len(conv)
    max_val = conv.max()
    
    # Get top-k indices
    if top_k >= size:
        idx = np.arange(size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            # g[j] += 2 * a[k-j] for j in [j_lo, j_hi]
            # a[k-j] for j in [j_lo, j_hi] = a[k-j_hi : k-j_lo+1][::-1]
            g[j_lo:j_hi+1] += 2.0 * a[k-j_hi:k-j_lo+1][::-1]
    
    return g, max_val

def normalize(a, target_sum):
    """Normalize a to have the target sum, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def mirror_descent_step(a, g, eta):
    """Multiplicative (mirror) update."""
    log_a = np.log(a + 1e-30) - eta * g
    # Shift for numerical stability
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    return a_new

def accelerated_mirror_descent(a0, time_budget, n_topk=30, verbose=True):
    """Main optimization: accelerated mirror descent."""
    n = len(a0)
    target_sum = 1.0  # Normalize to sum=1 for simplicity
    a = normalize(a0.copy(), target_sum)
    
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    best_score = 2 * n * best_max  # score with sum=1
    
    t0 = time.monotonic()
    y = a.copy()  # Nesterov extrapolate
    y_prev = a.copy()
    t_iter = 1.0
    eta_base = 5.0 / np.sqrt(n)
    patience = 0
    n_success = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Compute convergence at y (the extrapolated point)
        y = normalize(y, target_sum)
        conv_y = fast_autocorr(y)
        M_y = conv_y.max()
        
        # Compute subgradient at y
        g, _ = compute_subgrad(y, conv_y, top_k=n_topk)
        
        # Normalize gradient for step size
        g_norm = g.max()
        if g_norm < 1e-15:
            break
        g = g / g_norm  # Scale gradient so max is 1
        
        # Mirror descent step
        alpha = 2.0 / (t_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * mirror_descent_step(y, g, eta_base / np.sqrt(t_iter))
        y_new = normalize(y_new, target_sum)
        
        # Nesterov extrapolation
        a_new = y_new + ((t_iter - 1.0) / (t_iter + 2.0)) * (y_new - y)
        a_new = normalize(a_new, target_sum)
        
        y = y_new
        a = a_new
        t_iter += 1.0
        
        # Check improvement
        conv_a = fast_autocorr(a)
        M_a = conv_a.max()
        score_a = 2 * n * M_a
        
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            patience = 0
            n_success += 1
        else:
            patience += 1
        
        if patience > 200:
            # Reset to best and reduce step size
            y = best_a.copy()
            a = best_a.copy()
            t_iter = 1.0
            eta_base *= 0.95
            patience = 0
            
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD iter={it} ({elapsed:.0f}s): best_score={best_score:.8f}, eta={eta_base/np.sqrt(t_iter):.6f}")
    
    return best_a, best_score
```

Wait, I realize there's an issue with the Nesterov + mirror descent combination. Nesterov acceleration is designed for Euclidean (additive) updates. For mirror descent (multiplicative), the acceleration is different.

The correct approach for accelerated mirror descent (also known as "Nesterov's method in the mirror map") is:

1. y_t = (1 - alpha_t) * x_t + alpha_t * (x_t - x_{t-1}) [in the primal]
   Or in the mirror geometry: y_t = argmin <g, y> + D_ψ(y, (1-α)x_t + α x_{t-1})

2. x_{t+1} = argmin <g_t, x> + (1/η_t) D_ψ(x, y_t)

where D_ψ is the Bregman divergence of the mirror map.

For the simplex with entropy mirror map:
x_{t+1}[j] = y_t[j] * exp(-η_t * g_t[j]) / Z

This is the multiplicative weights update centered at y_t.

So the algorithm is:
1. y_t = extrapolate(x_t, x_{t-1}) [in the simplex geometry]
2. g_t = subgradient at y_t
3. x_{t+1}[j] = y_t[j] * exp(-η_t * g_t[j]) / Z

For the extrapolation in the simplex geometry, the natural choice is:
y_t[j] = (1-α) * x_t[j] + α * x_{t-1}[j] (linear combination, which stays in the simplex)

Or using the mirror map:
log y_t[j] = (1-α) * log x_t[j] + α * log x_{t-1}[j]

The log version is more natural for the entropy mirror map. But the linear version is simpler and often works well in practice.

Let me use the linear version for simplicity.

Actually, for the mirror descent on the simplex, the standard accelerated version is:

```
y_1 = x_0
for t = 1, 2, ...:
    g_t = ∇f(y_t) (or subgradient)
    x_{t+1} = y_t * exp(-η_t * g_t) / Z  (multiplicative weights)
    y_{t+1} = (t-1)/(t+2) * (x_1 + ... + x_t) / t + 1/(t+2) * x_{t+1}  (weighted average)
```

Wait, that's not quite right either. Let me look up the correct formulation.

For Nesterov's accelerated gradient method in the mirror geometry:
- Let ψ(x) = -sum x[i] log x[i] (negative entropy)
- Bregman divergence: D_ψ(x, y) = ψ(x) - ψ(y) - <∇ψ(y), x-y>
- The update is:
  y_t = argmin_x <g_{t-1}, x> + (1/η_{t-1}) D_ψ(x, x_{t-1} + β_{t-1}(x_{t-1} - x_{t-2}))
  x_t = argmin_x <g_t, x> + (1/η_t) D_ψ(x, y_t)

For the entropy mirror map, the update simplifies to:
x_t[j] = y_t[j] * exp(-η_t * g_t[j]) / Z

And the extrapolation is:
y_t = x_{t-1} + β_{t-1} * (x_{t-1} - x_{t-2}) (in the primal)

where β_t = t / (t+2) (or (t-1)/(t+2) depending on indexing).
