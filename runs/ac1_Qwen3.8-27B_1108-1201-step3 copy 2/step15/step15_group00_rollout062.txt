def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200):
    """LP-based Frank-Wolfe with top-K constraint selection and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    iter_count = 0

    # Precompute the constraint matrix structure
    # T[j, i] such that (y*b)[k] = sum_i T[k,i]*b[i]
    # T[k, i] = y[k-i] if 0<=k-i<n, else 0
    T_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_full[j:j + n, j] = y

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Update T_full with current y
        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K most binding constraints
        constraint_vals = T_full @ y
        # Find the constraints closest to M_curr (most binding)
        margins = M_curr - constraint_vals
        # We want constraints where margin is smallest (most binding)
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top contributing lags
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity: for each position d, how much does adding mass there
            # increase the peak autocorrelation?
            # Sensitivity at position d = sum over top lags k2 of a[d relative to k2]
            n_sens = min(10, n_top)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:n_sens]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            # Find best destination: low sensitivity, not in peak
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Score destinations by their "cost" (how much they'd increase peaks)
            scores = np.zeros(n)
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    scores[d] += add

            # Among non-peak positions, find the one with lowest score
            candidate_scores = scores[candidates]
            if len(candidate_scores) > 0:
                best_dst_idx = candidates[np.argmin(candidate_scores)]
            else:
                best_dst_idx = int(rng.integers(n))

            dst = best_dst_idx
            if dst == src:
                continue

            # Try multiple fractions
            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 50:
                # Forced random move
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_linear_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_bimodal(n, center_sep=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.1) ** 2) + np.exp(-50 * (x - 0.9) ** 2) + 0.01
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    # Generate a smooth random function via low-frequency Fourier series
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Try multiple initializations and AMD on each ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 480.0)

    if t_phase1 > 60:
        # Build list of initializations
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Linear ramp
        inits.append(make_linear_ramp(n0, power=1.0))
        labels.append("linear")

        # Quadratic ramp
        inits.append(make_linear_ramp(n0, power=2.0))
        labels.append("quadratic")

        # Bimodal
        inits.append(make_bimodal(n0))
        labels.append("bimodal")

        # Random smooth
        inits.append(make_smooth_random(n0, seed=7))
        labels.append("smooth_rand")

        # Blend parent with linear ramp
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_linear_ramp(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove duplicates (keep first occurrence)
        seen_sums = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0//100)], 4))
            if sig not in seen_sums:
                seen_sums.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        # Allocate time per init
        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=50, beta=5.0, c0=4.0,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 3500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=200)
        try_update(a_out, f"lp_n{n_best}")
    else:
        print("  Skipping LP (time or size constraint)")

    # === Phase 3: AMD polish after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMD polish ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        # Use higher top_k for polishing
        a_out, _ = amd_optimize(a_in, t3, top_k=80, beta=4.0, c0=3.0,
                                 restart_every=10000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 400, n_best + 800, n_best + 1600]
    longer_lengths = [l for l in longer_lengths if l <= 5000 and remaining > 120]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: Scale to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.30, 150.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=50, beta=5.0, c0=3.0,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3500:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=True, top_K=150)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t6 = min(remaining * 0.35, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=300)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 3000:
        print(f"\n=== Phase 8: Extra LP (n={n_best}) ===")
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=400)
            try_update(a_out, f"lp_extra_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n_fb = 2400
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
