Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with active set reduction and backtracking step sizes
2. Mirror descent with entropy-regularized subgradient updates
3. Multi-phase strategy: coarse LP optimization -> fine LP -> subgradient -> CD polish
4. Random restarts with adaptive perturbation for escaping local minima

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    _global_solver = 'highs'
    return _global_solver


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), size)
    B = rfft(np.asarray(b, dtype=np.float64), size)
    return irfft(A * B, size)[:n]


def fft_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2 * n - 1]


def obj_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fft_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def build_T_sub(a, active_idx):
    n = len(a)
    K = len(active_idx)
    T = np.zeros((K, n))
    for row, i in enumerate(active_idx):
        i = int(i)
        j_lo = max(0, i - n + 1)
        j_hi = min(i, n - 1)
        if j_hi >= j_lo:
            T[row, j_lo:j_hi + 1] = a[i - j_hi:i - j_lo + 1][::-1]
    return T


def lp_best_response(a, M, active_idx):
    n = len(a)
    T_sub = build_T_sub(a, active_idx)
    K = len(active_idx)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    res = linprog(c_obj, A_ub=T_sub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def normalize(a, target_sum):
    s = a.sum()

def lp_nesterov_phase(a0, time_limit, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    best_max = fft_autocorr(a).max()

    restart_count = 0

    y = a.copy()
    rng = np.random.default_rng(77)

            if patience > 50:
                break
            continue

        b = normalize(b, norm)
        n_iters += 1

        alpha = 2.0 / (momentum_t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b

        if momentum_t < 2.0:
            a_new = b.copy()
        else:

            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)

        a = a_new
        momentum_t += 1.0

        score_y = fft_autocorr(y).max()

            best_a = y.copy()
            patience = 0

        if score_a < best_max:

def mirror_descent_phase(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_max = fft_autocorr(a).max()
    t0 = time.monotonic()
    eta = 0.1 / np.sqrt(n)
    top_k = 50
    n_iters = 0
    patience = 0
    rng = np.random.default_rng(314)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_autocorr(a)
        M_curr = c.max()
        active_idx = np.argsort(c)[-top_k:]

        # Compute subgradient: average of gradients at active indices
        grad = np.zeros(n)
        for i in active_idx:
            i = int(i)
            j_lo = max(0, i - n + 1)
            j_hi = min(i, n - 1)
            if j_hi >= j_lo:
                grad[j_lo:j_hi + 1] += 2.0 * a[i - j_hi:i - j_lo + 1][::-1]
        grad /= len(active_idx)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Try multiple step sizes
        best_a_try = a.copy()
        best_conv_max = M_curr
        for factor in [0.5, 1.0, 2.0, 3.0, 5.0]:
            eta_try = eta * factor
            log_a_try = np.log(a + 1e-30) - eta_try * grad
            a_try = np.exp(log_a_try - log_a_try.max())
            a_try = normalize(a_try, norm)
            conv_try = fft_autocorr(a_try).max()
            if conv_try < best_conv_max:
                best_conv_max = conv_try
                best_a_try = a_try
                eta = eta_try

        a = best_a_try
        n_iters += 1

        if best_conv_max < best_max - 1e-12:
            best_max = best_conv_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.85

        if patience > 200:
            noise = rng.normal(0, norm * 0.01, n)
            a_pert = np.maximum(a + noise, 0)
            a = normalize(a_pert, norm)
            patience = 0
            eta = 0.1 / np.sqrt(n)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={n_iters} ({elapsed:.0f}s): best_max={best_max:.10f}, eta={eta:.6f}")

            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            # Try moving small amounts to nearby positions
            for off in [1, -1, 2, -2, 3, -3, 5, -5, 10, -10]:
                j = (i + off) % n
                if j == i:
                    continue

                # Compute delta in convolution when moving d from i to j
                # delta[i] changes: -2*a[i]*d + d^2 at position 2i (if valid)
                # delta[j] changes: +2*a[j]*d + d^2 at position 2j
                # Cross terms change at positions involving i and j

                # For efficiency, just try a small fraction and check
                d = a[i] * 0.1
                if d < 1e-15:
                    continue

                a[i] -= d
                a[j] += d
                new_c = fft_autocorr(a)
                new_max = new_c.max()

                if new_max < best_max - 1e-13:
                    best_max = new_max
                    best_a = a.copy()
                    c = new_c
                    pass_improved = True
                    no_improve = 0
                else:
                    a[i] += d
                    a[j] -= d

        if not pass_improved:
            no_improve += 1
            if no_improve >= 20:
                # Perturb to escape
                noise = rng.normal(0, norm * 0.003, n)
                a_pert = np.maximum(a + noise, 0)
                a = normalize(a_pert, norm)
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_max={best_max:.10f}")

    return best_a, best_max


def targeted_cd_phase(a0, time_limit, top_lags=15, verbose=True):
    """CD focusing on the lags where convolution is highest."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)

    best_a = a.copy()
    best_max = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(888)
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if n_passes > 3000 and no_improve > 30:
            break

        c = fft_autocorr(a)
        active_lags = np.argsort(c)[-top_lags:]

        # For each active lag, try to reduce it by redistributing mass
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for off in range(1, min(30, n // 4) + 1):
                for j in [(i + off) % n, (i - off) % n]:
                    if j == i:
                        continue
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        new_max = fft_autocorr(a).max()
                        if new_max < best_max - 1e-13:
                            best_max = new_max
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                        else:
                            a[i] += d
                            a[j] -= d
                    if improved:
                        break
                if improved and off > 5:
                    break

        if not improved:
            no_improve += 1
            if no_improve >= 30:
                noise = rng.normal(0, norm * 0.005, n)
                a_pert = np.maximum(a + noise, 0)
                a = normalize(a_pert, norm)
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best_max={best_max:.10f}")

    return best_a, best_max


def random_restart_phase(a0, time_limit, n_restarts=3, verbose=True):
    """Perturb and re-optimize with quick LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_max = fft_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    time_per_restart = time_limit / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        scale = norm * 0.03 * (r + 1)
        noise = rng.laplace(0, scale, n)
        a_pert = np.maximum(best_a + noise, 0)
        a_pert = normalize(a_pert, norm)

        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_restart * 0.8, t_remaining * 0.7)
        if t_sub < 10:
            break

        a_out, sc = lp_nesterov_phase(a_pert, t_sub, verbose=False)
        if sc < best_max - 1e-10:
            best_max = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    RR restart {r}: improved to max={best_max:.10f}")

    return best_a, best_max


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_best = len(best_seq)

    # Phase 1: LP Nesterov (30% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.30
    print(f"\n=== Phase 1: LP Nesterov n={n_best} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_phase(a_in, t1, verbose=True)
        try_update(a_out, "lp1")

    # Phase 2: Mirror Descent (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.15
    print(f"\n=== Phase 2: Mirror Descent n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mirror_descent_phase(a_in, t2, verbose=True)
        try_update(a_out, "md")

    # Phase 3: LP Nesterov round 2 (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP Nesterov R2 n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_phase(a_in, t3, verbose=True)
        try_update(a_out, "lp2")

    # Phase 4: Subgradient Hedge (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.10
    print(f"\n=== Phase 4: Subgradient Hedge n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_hedge_phase(a_in, t4, verbose=True)
        try_update(a_out, "hedge")

    # Phase 5: Targeted CD (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: Targeted CD n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_cd_phase(a_in, t5, verbose=True)
        try_update(a_out, "tcd")

    # Phase 6: Random restarts (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t6 = remaining * 0.5
        print(f"\n=== Phase 6: Random Restarts n={n_best} (budget={t6:.0f}s) ===")
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_restart_phase(a_in, t6, n_restarts=2, verbose=True)
        try_update(a_out, "rr")

    # Phase 7: Final CD polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t7 = remaining * 0.6
        print(f"\n=== Phase 7: Final CD n={n_best} (budget={t7:.0f}s) ===")
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_phase(a_in, t7, verbose=True)
        try_update(a_out, "cd_final")

    # Final LP polish with whatever time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t8 = min(remaining * 0.8, 120.0)
        print(f"\n=== Phase 8: Final LP polish n={n_best} (budget={t8:.0f}s) ===")
        a_in = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_phase(a_in, t8, verbose=False)
        try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
