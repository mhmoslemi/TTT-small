            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            a = a_new
            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and (si % 5 == 0 or si == n_stages - 1):
            print(f"    LSE-Adam stage {si}/{n_stages} T={T_val:.4f}: best={best_score:.8f}")

    return best_a, best_score

def build_constraint_matrix(a, active_indices):
    """Build sparse constraint matrix for LP.
    T[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    T = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            T[idx, j] = a[k - j]
    return T

def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. b >= 0, conv(a,b)_k <= M for k in active_set."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is None or len(active_set) >= size:
        active_set = np.arange(size)

    T = build_constraint_matrix(a, active_set)
    constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    return np.maximum(val, 0.0)
