"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Novel Hankel-QP Frank-Wolfe with iterative peak reweighting
2. Nesterov subgradient with top-K peak weighting
3. LP-based refinement (short bursts)

5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def hankel_matvec(lam, a):
    """Compute (La)_i = sum_j lam[i+j] * a_j efficiently via FFT."""
    n = len(a)
    # (La)_i = sum_j lam[i+j] * a_j
    # Let rev_lam[k] = lam[2n-2-k]
    # Then (La)_i = sum_j rev_lam[2n-2-i-j] * a_j = (rev_lam * a)_{2n-2-i}
    rev_lam = lam[::-1]  # rev_lam[k] = lam[2n-2-k]
    # Standard convolution: (rev_lam * a)_m = sum_j rev_lam[m-j] * a_j
    # We want sum_j rev_lam[2n-2-i-j] * a_j = (rev_lam * a)_{2n-2-i}
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    R = rfft(rev_lam.astype(np.float64), fft_size)
    A = rfft(a.astype(np.float64), fft_size)
    conv = irfft(R * A, fft_size)[:size]
    # (La)_i = conv[2n-2-i]
    return conv[2*n-2::-1]  # reverse to get index i -> 2n-2-i

def project_simplex(v, cap=1.0):
    """Project v onto {x: sum(x)=cap, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - cap))[0][-1]
    theta = (cssv[rho] - cap) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def hankel_matvec(lam, a):
    """Compute (La)_i = sum_j lam[i+j] * a_j via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    rev_lam = lam[::-1]
    R = rfft(rev_lam, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(R * A, fft_size)[:size]
    return conv[n-1:2*n-1][::-1]


def project_simplex(v, cap=1.0):
    """Project v onto {x: sum(x)=cap, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - cap)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - cap) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def qp_peak_equalize(a0, time_limit, peak_count=10, fw_iters=100, fista_iters=50):

        a = a / s

            a_new = a

            a = a_new

    t = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg = a.copy()
    count = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Compute gradient from top-K peaks
        b = fast_autocorr(y)
        sort_idx = np.argsort(b)[::-1][:topk]
        
        # Weighted subgradient
        grad = np.zeros(n)
        max_b = b[sort_idx[0]]
        for i, k in enumerate(sort_idx):
            w = np.exp(-i * 0.2)  # weight decay
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w * 2.0 * y[k - idx]
        
        # Project gradient onto tangent space of simplex
        grad = grad - grad.mean()
        
        # Nesterov step
        alpha = 2.0 / (it + 2.0)
        y_new = y - alpha * grad
        y_new = np.maximum(y_new, 0.0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / y_sum
        
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_new = y_new + ((t - 1.0) / t_new) * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / a_sum
        
        y = y_new
        a = a_new
        
        # Update running average
        avg = (count * avg + a) / (count + 1)
        count += 1
        
        # Check score periodically
        if (it + 1) % 50 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            score_avg = 2.0 * n * fast_autocorr(avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg.copy()
    
    # Final check
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_a = a.copy()
        best_score = score
    score_avg = 2.0 * n * fast_autocorr(avg).max()
    if score_avg < best_score:
        best_a = avg.copy()
        best_score = score_avg
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Peak shaving with global verification."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        top_peaks = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find safe destinations (positions not contributing to top peaks)
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks:
                jsp = max(0, kp - n + 1)
                jep = min(n - 1, kp)
                danger[jsp:jep + 1] = True
            
            safe = np.where(~danger)[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            
            for dst in safe_sorted[:5]:
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        amt = a_trial[i] * rng.uniform(0.005, 0.05)
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    if 2.0 * n * fast_autocorr(a_trial).max() < best_score:
                        a = a_trial
                        best_score = 2.0 * n * fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: for each coordinate, find optimal value."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # For coordinate j, if we change a_j by delta (and adjust a_l by -delta to keep sum),
    # the change in b_k is: delta * (a[k-j] - a[k-l]) + a_j * ... (gets complex)
    # Simpler: for each j, try moving mass from j to l (the least "dangerous" position)
    
    order = np.random.default_rng(123).permutation(n)
    
    for _ in range(5):  # sweep
        if time.monotonic() - t0 > time_limit - 2:
            break
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            if a[j] < 1e-10:
                continue
            
            b = fast_autocorr(a)
            bmax = b.max()
            
            # Find the position l to move mass to
            # Score each potential destination by the increase in max_b
            # Approximate: move small amount from j to l, check if max_b decreases
            for frac in [0.05, 0.1, 0.2, 0.4]:
                amt = a[j] * frac
                if amt < 1e-12:
                    continue
                # Find best destination
                best_l = -1
                best_new_max = bmax
                # Check a few candidates
                candidates = np.argsort(a)[:10]  # positions with smallest values
                for l in candidates:
                    if l == j:
                        continue
                    a_trial = a.copy()
                    a_trial[j] -= amt
                    a_trial[l] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    nm = fast_autocorr(a_trial).max()
                    if nm < best_new_max:
                        best_new_max = nm
                        best_l = l
                
                if best_l >= 0 and best_new_max < bmax * (1 - 1e-9):
                    a[j] -= amt
                    a[best_l] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-12:
                        a = a / st
                        sc = 2.0 * n * fast_autocorr(a).max()
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                    break
    
    return best_a, best_score
