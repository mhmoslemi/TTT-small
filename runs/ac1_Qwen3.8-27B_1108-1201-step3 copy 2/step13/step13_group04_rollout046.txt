import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csr_matrix, diags
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def lse_weights(b, T):
    """Compute normalized softmax weights for LSE gradient."""
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


def lse_gradient(a, T):
    """
    Compute gradient of LSE(a, T) = T*log(sum exp(b_k/T)) where b = conv(a,a).
    grad[m] = (w * a_rev)[m+n-1] + (w * a)[m]
    """
    n = len(a)
    b = fast_autocorr(a)
    w = lse_weights(b, T)

    fft_size = 1 << (3 * n).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def lse_nesterov(a0, time_budget, T_start=10.0, T_end=0.1, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_budget:
                break

            if it == 0:
                grad = lse_gradient(y, T)
                y_new = y - (1.0 / 3.0) * grad
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_ex = y_extrap.sum()
                if s_ex > 1e-15:
                    y_extrap = y_extrap * norm / s_ex
                grad = lse_gradient(y_extrap, T)
                y_new = y_extrap - (1.0 / (it + 3.0)) * grad

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if verbose and it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix T where (T @ b)[j] = (conv(a,b))[j]."""
    n = len(a)
    T = lil_matrix((2 * n - 1, n))
    for k in range(n):
        # T[j, k] = a[j-k] for 0 <= j-k < n
        # j ranges from k to k+n-1
        js = np.arange(k, k + n)
        T[js, k] = a[np.arange(k, k + n) - np.arange(k, k + n)]  # a[0..n-1]
        # Actually T[j, k] = a[j-k], so for j=k..k+n-1, j-k = 0..n-1
        T[js, k] = a[np.arange(n)]
    return T.tocsr()


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_optimize(a0, time_budget, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_nest = a.copy()
    n_iters = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_budget:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_nest = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
            a_nest = a_new

        y = y_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a_nest).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_nest.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_search_annealed(a0, time_budget, seed=42, verbose=True):
    """
    Peak-targeting local search with simulated annealing-style acceptance.
    Tries mass transfers from peak-contributing positions to non-peak positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0
    stall = 0
    T_anneal = best_score * 0.01

    while time.monotonic() - t0 < time_budget:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]
        top_vals = b[top_ks]

        improved = False
        # Try multiple random perturbation directions
        for attempt in range(20):
            # Pick a peak lag
            k_star = top_ks[rng.integers(0, min(5, len(top_ks)))]

            # Find contributors to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]

            if contrib.max() < 1e-20:
                continue

            # Pick a source position (high contributor)
            src_positions = idx[np.argsort(contrib)[::-1][:5]]
            src = src_positions[rng.integers(0, len(src_positions))]

            if a[src] < norm * 1e-10:
                continue

            # Pick destination: prefer positions not contributing to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_ks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            # Also mask the source
            peak_mask[src] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Weight toward low-value positions
            weights = 1.0 / (a[candidates] + norm * 0.001)
            weights /= weights.sum()
            dst = candidates[rng.choice(len(candidates), p=weights)]

            # Try different transfer amounts
            fracs = [0.001, 0.003, 0.01, 0.03, 0.07, 0.15]
            rng.shuffle(fracs)

            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()

                # Accept if improvement, or with probability exp(-(score-best)/T_anneal)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    T_anneal = max(best_score * 0.003, 1e-15)
                    improved = True
                    stall = 0
                    it_count += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / (T_anneal + 1e-30)):
                    a = a_trial
                    improved = True
                    stall = 0
                    it_count += 1
                    break

            if improved:
                break

        if not improved:
            stall += 1
            T_anneal *= 0.95
            if stall > 300:
                break

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    f_padded = np.concatenate([f_norm * 0.5, f_norm, f_norm * 0.5])
    f_padded = f_padded / f_padded.sum()
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_padded)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        # Asymmetric ramp with taper
        x = np.linspace(0.0, 1.0, n)
        a = x * np.sin(np.pi * x) + 0.1
    elif choice == 2:
        # Raised cosine
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    elif choice == 3:
        # Skewed Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.6) ** 2) + 0.3 * np.exp(-8.0 * (x - 0.2) ** 2)
    elif choice == 4:
        # Power law
        x = np.linspace(0.01, 1.0, n)
        a = x ** 2 + 0.5
    elif choice == 5:
        # Multi-peak
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.5) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
    elif choice == 6:
        # Asymmetric triangular
        x = np.linspace(0.0, 1.0, n)
        a = np.where(x < 0.6, x / 0.6, (1 - x) / 0.4)
    else:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def mirror_descent_lse(a0, time_budget, T_start=8.0, T_end=0.2, n_stages=8, verbose=True):

    # ============ Phase 2: LP Nesterov (main workhorse) ============
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 480.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_work, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # ============ Phase 3: LSE refinement ============
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE refinement n={len(best_seq)} (budget={t3:.0f}s) ===")
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_nesterov(a_work, t3, T_start=5.0, T_end=0.1, n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_ref_n{n_cur}")

    # ============ Phase 4: Mirror descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: MirrorDescent n={len(best_seq)} (budget={t4:.0f}s) ===")
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent_lse(a_work, t4, T_start=5.0, T_end=0.1, n_stages=8, verbose=False)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # ============ Phase 5: Local search / annealing ============
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        print(f"\n=== Phase 5: LocalSearch n={len(best_seq)} (budget={t5:.0f}s) ===")
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_search_annealed(a_work, t5, seed=12345, verbose=True)
        do_try_update(a_out, f"local_n{n_cur}")

    # ============ Phase 6: Longer length exploration ============
    n2 = n_base + 300
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        print(f"\n=== Phase 6: Longer length n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        # Quick LP at longer length
        t6a = t6 * 0.5
        a_out, _ = lp_nesterov_optimize(a_work, t6a, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n2}")
        # Then LSE polish
        remaining_sub = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining_sub * 0.5, t6 * 0.4)
        if t6b > 10:
            n_cur = len(best_seq)
            a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_nesterov(a_work, t6b, T_start=3.0, T_end=0.1, n_stages=6, verbose=False)
            do_try_update(a_out, f"lse_n{n2}_polish")

    # ============ Phase 7: Even longer length ============
    n3 = n_base + 600
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 20:
        print(f"\n=== Phase 7: Longer length n={n3} (budget={t7:.0f}s) ===")
        init_long3 = interpolate_sequence(best_seq, n3)
        a_work = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = lp_nesterov_optimize(a_work, t7, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n3}")

    # ============ Phase 8: Final LP polish on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 20:
        print(f"\n=== Phase 8: Final LP polish n={n_best} (budget={t8:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_work, t8, solver, verbose=True)
        do_try_update(a_out, "lp_final")

    # ============ Phase 9: Final local search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.30, 100.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final local search n={n_best} (budget={t9:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = local_search_annealed(a_work, t9, seed=99999, verbose=True)
        do_try_update(a_out, "local_final")

    # ============ Phase 10: Ultra-fine LSE ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 5:
            print(f"\n=== Phase 10: Ultra-fine LSE n={n_best} (budget={t10:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lse_nesterov(a_work, t10, T_start=1.0, T_end=0.01, n_stages=5, verbose=True)
            do_try_update(a_out, "lse_ultra")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
