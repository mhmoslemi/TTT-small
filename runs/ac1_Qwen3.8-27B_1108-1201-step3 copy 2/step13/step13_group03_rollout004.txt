def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)

    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max() / norm
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max() / norm
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max() / norm
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a * norm, best_score


def _grad_descent_nesterov(a0, time_limit, verbose=False):
    """Pure gradient descent with Nesterov on LSE, no LP needed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    t_k = 1

    T_start = 2.0
    T_end = 0.005
    decay = (T_end / T_start) ** (1.0 / 50000.0)

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start * (decay ** it)
        grad = _lse_grad(y, T)
        L = max(n / (T * 2.0), n / 20.0)
        step = 1.0 / L

        z = y - step * grad
        a_new = _proj_simplex(z)
        a_new = a_new * norm / a_new.sum()

        if t_k == 1:
            y_new = a_new.copy()
        else:
            mom = (t_k - 1.0) / (t_k + 2.0)
            y_new = a_new + mom * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            else:
                y_new = a_new.copy()

        y = y_new
        a = a_new
        t_k += 1

        if it % 500 == 0:
            sc = _fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            if verbose and it % 50000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Grad-Nest it={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f} (J={2*n*best_score/norm**2:.6f})")

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    """Systematic peak reduction by redistributing mass from peak-contributing positions."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:12]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find all positions that contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            # Candidates: positions NOT contributing to peaks, with low values
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Sort by (value ascending, distance from src descending)
            cands_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(5, len(cands_sorted) // 20)
            cands_sorted = cands_sorted[:n_try]

            for dst in cands_sorted:
                if dst == src:
                    continue
                for frac in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-18 or amt >= a[src]:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-16:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_perturb = max(1, n // 200)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break

            if stall > 300:
                # Larger random restart
                noise = rng.normal(0, 0.002 * norm / n, n)
                a_trial = np.maximum(a + noise, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score + 0.001:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                stall = 0
            elif stall > 2000:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _hybrid_lp_gradient(a0, time_limit, solver, verbose=False):
    """Hybrid: alternate between LP best-response and gradient steps."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    lp_counter = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Every 5 iterations, do an LP step; otherwise gradient
        if t % 5 == 0:
            M_curr = _fast_autocorr(y).max()
            if M_curr < 1e-12:
                break
            b = _lp_best_response(y, M_curr, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    g = b * norm / b_sum
                    alpha = 2.0 / (t + 2.0)
                    y_new = (1.0 - alpha) * y + alpha * g
                    y_new = np.maximum(y_new, 0.0)
                    s_y = y_new.sum()
                    if s_y > 1e-15:
                        y_new = y_new * norm / s_y
                    y = y_new
                    n_iters += 1
                    lp_counter += 1
        else:
            T = max(0.01, 1.0 / np.sqrt(t + 1.0))
            grad = _lse_grad(y, T)
            L = max(n / (T * 3.0), 1.0)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)
            a_new = a_new * norm / a_new.sum()
            y_new = (1.0 - 0.5/(t+1.0)) * y + (0.5/(t+1.0)) * a_new
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            y = y_new

        if t % 100 == 0:
            sc = _fast_autocorr(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()

            if verbose and t % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Hybrid iter={t} LPs={lp_counter} ({elapsed:.0f}s): best={best_score:.8f} (J={2*n*best_score/norm**2:.6f})")

    return best_a, best_score


def propose_candidate():
    """Main search: multi-phase optimization with aggressive cooling and hybrid methods."""
    t_global = time.monotonic()
    time_budget = 1750.0  # Leave 50s margin

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1400
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Aggressive LSE-FISTA (deep cooling) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [40.0, 8.0, 2.0, 0.5, 0.1, 0.03, 0.008]
        a_out, _ = _lse_fista(best_seq, T_sched, 12000, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Gradient Nesterov (fast, no LP) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 15:
        print(f"\n=== Phase 2: Grad-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _grad_descent_nesterov(a_norm, t2, verbose=True)
        _try_update(a_out / np.sqrt(2.0 * n_cur), f"grad_n{n_cur}")

    # === Phase 3: LP-Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t3, solver, verbose=True)
        _try_update(a_out / np.sqrt(2.0 * n_cur), f"lp1_n{n_cur}")

    # === Phase 4: Hybrid LP+Gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.18, 280.0)

        print(f"\n=== Phase 4: Hybrid LP+Grad n={n_cur} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _hybrid_lp_gradient(a_norm, t4, solver, verbose=True)

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)

        a_out, _ = _peak_shave(a_norm, t5, seed=99999)
        _try_update(a_out / np.sqrt(2.0 * n_cur), "peak1")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)

        a_out, _ = _peak_shave(a_norm, t8, seed=12321)

    remaining = time_budget - (time.monotonic() - t_global)

    if t9 > 20:
        print(f"\n=== Phase 9: Final LP n={n_cur} (budget={t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t9, solver, verbose=True)
        _try_update(a_out / np.sqrt(2.0 * n_cur), "lp_final")

    # === Phase 10: Ultimate peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 10:
        t10 = remaining * 0.8
        t10 = min(t10, 80.0)

        a_out, _ = _peak_shave(a_norm, t10, seed=55555)

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
