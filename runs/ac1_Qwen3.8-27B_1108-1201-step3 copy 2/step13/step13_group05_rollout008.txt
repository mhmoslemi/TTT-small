6. Efficient FFT-based autocorrelation
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  Using LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        for x in a_list:
            if np.isnan(x) or np.isinf(x) or x < 0:
                return float('inf')
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_subgrad_exact(a, ac):
    """Exact subgradient at the single maximum lag."""
    n = len(a)
    k_star = int(np.argmax(ac))
    grad = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad


def compute_subgrad_soft(a, ac, top_k=5, beta=3.0):
    """Weighted subgradient using top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        i_start = max(0, int(k) - n + 1)
        i_end = min(n - 1, int(k))
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * wi * a[int(k) - idx]
    return grad


def lp_best_response(a, M, solver, b_warm=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        if b_warm is not None:
            b_var.value = np.asarray(b_warm, dtype=np.float64)
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def mirror_descent_adaptive(a0, time_limit, solver=None, use_lp_every=50, verbose=True):
    """Adaptive mirror descent with optional LP polish every N steps."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a * c / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    gamma = 1.0
    step_decay = 0.9995
    current_step = gamma
    lp_warm = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        # Blend exact and soft subgradients
        g_exact = compute_subgrad_exact(a, ac)
        g_soft = compute_subgrad_soft(a, ac, top_k=4, beta=2.5)
        g = 0.5 * g_exact + 0.5 * g_soft

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - current_step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new

        # Adaptive step size: reduce if no improvement in recent steps
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
            # Slight step increase on improvement
            current_step = min(gamma, current_step * 1.01)
        else:
            current_step *= step_decay

        # Periodic LP polish
        if solver is not None and use_lp_every > 0 and t % use_lp_every == 0:
            ac_cur = fast_autocorr(a)
            M_cur = ac_cur.max()
            b = lp_best_response(a, M_cur, solver, b_warm=lp_warm)
            if b is not None:
                lp_warm = b
                b_sum = b.sum()
                if b_sum > 1e-15:
                    g_dir = b * c / b_sum
                    # Blend with current
                    blend = 0.3
                    a_lp = (1 - blend) * a + blend * g_dir
                    a_lp = np.maximum(a_lp, 0.0)
                    s_lp = a_lp.sum()
                    if s_lp > 1e-15:
                        a_lp = a_lp * c / s_lp
                        M_lp = fast_autocorr(a_lp).max()
                        if M_lp < best_m:
                            best_m = M_lp
                            best_a = a_lp.copy()
                            a = a_lp.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a * c / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    lp_warm = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, b_warm=lp_warm)
        if b is None:
            continue

        lp_warm = b
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x

        y = y_new
        x = x_new

        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()

        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Aggressive coordinate descent: move mass from peak contributors to safe spots."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a * c / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:6]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destination: minimize increase to all top peaks
            impact = np.zeros(n)
            for k2 in top_k:
                k2_int = int(k2)
                js2 = max(0, k2_int - n + 1)
                je2 = min(n - 1, k2_int)
                if js2 <= je2:
                    idx2 = np.arange(js2, je2 + 1)
                    impact[idx2] += a[k2_int - idx2]

            # Destinations with low impact and low current value
            safe_score = a + 0.1 * impact
            candidates = np.where(idx != src)[0]
            if len(candidates) == 0:
                continue
            candidates = candidates[safe_score[candidates] < np.median(safe_score)]
            if len(candidates) == 0:
                candidates = np.where(idx != src)[0]

            # Try several destinations
            cands_sorted = candidates[np.argsort(safe_score[candidates])[:min(5, len(candidates))]]
            for dst in cands_sorted:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_m - 1e-12:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a * c / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c / a.sum()

        if verbose and (stall % 50 == 0 or time.monotonic() - t0 > 30):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    CD ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_init(n, rng, style=0):
    """Generate a diverse random initialization."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Sinusoidal
        a = np.sin(np.pi * x) ** 2
    elif style == 1:
        # Cosine bump
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 2:
        # Two bumps
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif style == 3:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    elif style == 4:
        # Random smooth
        a = np.abs(np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28)))
    else:
        # Uniform with noise
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with progressive scaling."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_default = 1600
        best_seq = random_init(n_default, rng, style=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Quick MD on parent ===
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Fast MD n={n0} ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 10:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_adaptive(a_norm, t1, solver=solver, use_lp_every=100, verbose=True)
        try_update(a_out, f"md1_n{n0}")

    # === Phase 2: LP Nesterov on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n0} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # === Phase 3: Scale to n=2400 ===
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Scale to n={n1} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_long = interpolate_to_length(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()
        # First MD pass
        t3a = t3 * 0.4
        a_out, _ = mirror_descent_adaptive(a_norm, t3a, solver=solver, use_lp_every=80, verbose=True)
        try_update(a_out, f"md_n{n1}")
        # Then LP polish
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.12, 200.0)
        if t3b > 20:
            a_norm2 = best_seq * c1 / best_seq.sum() if len(best_seq) == n1 else (interpolate_to_length(best_seq, n1) * c1 / interpolate_to_length(best_seq, n1).sum())
            a_out2, _ = nesterov_lp(a_norm2, t3b, solver, verbose=True)
            try_update(a_out2, f"lp_n{n1}")

    # === Phase 4: Scale to n=3200 ===

            a_out2, _ = nesterov_lp(a_norm2, t4b, solver, verbose=True)
            try_update(a_out2, f"lp_n{n2}")

    # === Phase 5: Scale to n=4800 ===
    n3 = 4800

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Random restarts n={n_best} ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 20:
        c_best = np.sqrt(2.0 * n_best)
        for restart in range(3):
            elapsed = time.monotonic() - t_global
            t_remaining = time_budget - elapsed
            if t_remaining < 15:

            t_alloc = min(t6 / max(1, 3 - restart), t_remaining * 0.5)
            if t_alloc < 10:
                break
            # Initialize near best but with perturbation
            a_rand = best_seq.copy()
            noise = rng.standard_normal(n_best) * 0.1
            a_rand = a_rand + noise
            a_rand = np.maximum(a_rand, 0.0)
            a_rand = a_rand * c_best / a_rand.sum()
            a_out, _ = nesterov_lp(a_rand, t_alloc, solver, verbose=(restart == 0))
            try_update(a_out, f"restart{restart}_n{n_best}")
            # Also try a completely different init for the last restart
            if restart == 2:
                t_remaining = time_budget - (time.monotonic() - t_global)
                if t_remaining > 20:
                    a_diff = random_init(n_best, rng, style=rng.integers(5))
                    a_diff = a_diff * c_best / a_diff.sum()
                    a_out3, _ = nesterov_lp(a_diff, min(t_remaining * 0.4, 60.0), solver, verbose=True)
                    try_update(a_out3, f"diffinit_n{n_best}")

        try_update(a_out, f"cd_final_n{n_final}")

    # === Phase 8: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_final} ===")
    t8 = min(remaining * 0.5, 100.0)
    if t8 > 10:
        c_final = np.sqrt(2.0 * n_final)
        a_norm = best_seq * c_final / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t8, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_final}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
