        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=200000):
    """Nesterov-accelerated LP best-response with geometrically weighted averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Geometrically weighted average (exponential moving average with slow decay)
    geo_avg = a.copy()
    geo_alpha = 0.01  # slow decay for long-term average
    # Short-term average
    short_avg = a.copy()
    short_alpha = 0.1

    stall_count = 0
    warm_b = None
    last_print = 0
    no_improve = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update geometric average
        geo_avg = (1 - geo_alpha) * geo_avg + geo_alpha * a
        short_avg = (1 - short_alpha) * short_avg + short_alpha * a

        # Check multiple candidates
        for cand in [a, y, geo_avg, short_avg]:
            score_c = 2.0 * n * fast_autocorr(cand).max()
            if score_c < best_score:
                best_score = score_c
                best_a = cand.copy()
                stall_count = 0
                no_improve = 0
            else:
                no_improve += 1

        stall_count += 1

        # Adaptive restart
        if stall_count > 800 and t > 200:
            a = best_a.copy()
            y = best_a.copy()
            geo_avg = best_a.copy()
            short_avg = best_a.copy()
            stall_count = 0
            no_improve = 0
            warm_b = None

        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.7f}, M={M_val:.9f}")
            last_print = t

    return best_a, best_score


def subgradient_mirror_descent(a0, time_limit, seed=42):
    """Mirror descent using exact subgradients of max(conv(a,a))."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0
    eta_min = 1e-5

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        # Find all peaks within 0.5% of max
        threshold = bmax * 0.995
        peak_indices = np.where(b >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(b)])

        # Compute subgradient: g[j] = sum over active peaks k of 2*a[k-j]
        g = np.zeros(n)
        for k in peak_indices:
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    g[j] += 2.0 * a[m]

        # Normalize gradient
        g_norm = g.sum()
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Mirror descent step: a_new[j] = a[j] * exp(-eta * g[j] / a[j] * a[j])
        # Actually for the simplex mirror descent:
        # a_new = a * exp(-eta * grad) where grad is the subgradient
        # But we need to be careful with the constraint.
        # Standard: a_new[j] proportional to a[j] * exp(-eta * g[j])
        updates = a * np.exp(-eta * g)
        updates = np.maximum(updates, 1e-30)
        su = updates.sum()
        if su < 1e-15:
            break
        a_new = updates / su

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score - 1e-10:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.05, 0.5)  # slightly increase step
        else:
            no_improve += 1
            eta *= 0.9
            if eta < eta_min:
                # Random restart
                if no_improve > 200:
                    noise = rng.normal(0, 0.02, n)
                    a_pert = a * (1 + noise)
                    a_pert = np.maximum(a_pert, 0)
                    sp = a_pert.sum()
                    if sp > 1e-10:
                        a = a_pert / sp
                        best_a = a.copy()
                        best_score = 2.0 * n * fast_autocorr(a).max()
                    eta = 0.05
                    no_improve = 0
                if no_improve > 500:
                    break

    return best_a, best_score


def local_peak_shaving(a0, time_limit, seed=123):
    """Intensive local peak shaving with systematic search."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        n_peaks = min(20, len(b))
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # Precompute which positions contribute to which peaks
        # For peak k, position j contributes a[j]*a[k-j]
        improved = False

        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            partner = k - src

            # Find safe destination positions (low contribution to all top peaks)
            danger_score = np.zeros(n)
            for pk in top_peaks[:10]:
                for j in range(n):
                    m = pk - j
                    if 0 <= m < n:
                        danger_score[j] += a[m]

            # Sort by danger_score (want low)
            safe_positions = np.argsort(danger_score)
            # Exclude src and partner
            exclude = {src, partner}
            candidates = [p for p in safe_positions if p not in exclude and a[p] < a[src] * 0.5]

            found = False
            for dst in candidates[:12]:
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Try swapping between two positions
            if stall % 5 == 0:
                # Find two positions with different "danger" levels
                danger_score = np.zeros(n)
                for pk in top_peaks[:8]:
                    for j in range(n):
                        m = pk - j
                        if 0 <= m < n:
                            danger_score[j] += a[m]

                high_danger = np.argsort(danger_score)[::-1][:5]
                low_danger = np.argsort(danger_score)[:5]
                for hi in high_danger:
                    for lo in low_danger:
                        if hi == lo or a[hi] < 1e-12:
                            continue
                        for frac in [0.01, 0.02, 0.05]:
                            amt = a[hi] * frac
                            a_trial = a.copy()
                            a_trial[hi] -= amt
                            a_trial[lo] += amt
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-12:
                                continue
                            a_trial = a_trial / st
                            new_max = fast_autocorr(a_trial).max()
                            if new_max < bmax * (1 - 1e-9):
                                a = a_trial
                                new_score = 2.0 * n * new_max
                                if new_score < best_score:
                                    best_score = new_score
                                    best_a = a.copy()
                                stall = 0
                                break

            # Random perturbation
            if stall % 20 == 0 and stall > 0:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=777):
    """Coordinate descent: optimize one coordinate at a time."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    # Pre-compute which peaks are active
    b = fast_autocorr(a)
    bmax = b.max()

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        active_peaks = np.where(b >= threshold)[0]

        improved = False
        # Try optimizing each coordinate
        order = np.random.default_rng(int(time.monotonic() * 1000) % 10000).permutation(n)

        for j in order[:min(200, n)]:
            if time.monotonic() - t0 > time_limit - 3:
                break
            if a[j] < 1e-12:
                continue

            # Convex optimization: minimize max over active peaks of (conv[k] as function of a[j])
            # conv[k] = sum_i a[i]*a[k-i]
            # As function of a[j]: conv[k] = a[j]*(a[k-j] + (a[j] if k=2j else 0)) + rest[k]
            # = a[j] * c1[k] + a[j]^2 * c2[k] + c0[k]
            # where c1[k] = a[k-j] (if valid), c2[k] = 1 if k=2j else 0, c0[k] = conv[k] - a[j]*c1[k] - a[j]^2*c2[k]

            # For each active peak k:
            # f_k(x) = x * c1[k] + x^2 * c2[k] + c0[k] where x = a[j] (adjusted for normalization)
            # But normalization makes this tricky. Let's use a simpler approach:
            # Try reducing a[j] and adding to multiple low-danger positions.

            # Compute how much a[j] contributes to each active peak
            contrib_to_peaks = np.zeros(len(active_peaks))
            for ii, k in enumerate(active_peaks):
                m = k - j
                if 0 <= m < n:
                    contrib_to_peaks[ii] = a[m]
                # Also if k-j == j (i.e., k = 2j), there's a self-contribution
                if k == 2 * j:
                    contrib_to_peaks[ii] += a[j]  # the a[j]^2 term derivative

            # Try reducing a[j] by trying to redistribute
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[j] * frac
                if amt < 1e-14:
                    continue
                # Distribute to positions with lowest danger
                danger = np.zeros(n)
                for k in active_peaks[:10]:
                    for i in range(n):
                        m = k - i
                        if 0 <= m < n:
                            danger[i] += a[m]
                danger[j] = np.inf  # exclude self

                safe = np.argsort(danger)[:max(3, n // 200)]
                safe = [s2 for s2 in safe if s2 != j]
                if not safe:
                    continue

                # Try putting all mass in one safe position
                for dst in safe[:4]:
                    a_trial = a.copy()
                    a_trial[j] -= amt
                    a_trial[dst] += amt
                    if a_trial[j] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.6 * np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 1.5)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (x - 1) ** 2)
    elif choice == 9:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 10:
        # Asymmetric with peak off-center
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2) + 0.3 * np.exp(-10 * (x - 0.7) ** 2)
    else:
        # Multi-scale
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.sin(9 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP game, subgradient mirror descent,
    local peak shaving, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 5.0, 1790.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
        else:
            raise ValueError("Invalid parent")
    except Exception:
        parent = None
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"No valid parent, initialized n={n_init}, score={best_score:.7f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP Nesterov (main workhorse) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")
        print(f"  LP done: score={sc_out:.7f}")

    # === Phase 2: Subgradient mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Subgradient mirror descent n={n_base} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, sc_out = subgradient_mirror_descent(best_seq, t2, seed=42)
        try_update(a_out, f"mirror_n{n_base}")
        print(f"  Mirror done: score={sc_out:.7f}")

    # === Phase 3: Local peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Peak shaving n={n_base} ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 15:
        a_out, sc_out = local_peak_shaving(best_seq, t3, seed=777)
        try_update(a_out, "peak_shave_1")
        print(f"  Peak shave done: score={sc_out:.7f}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Coordinate descent n={n_base} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_out, sc_out = coordinate_descent_polish(best_seq, t4, seed=555)
        try_update(a_out, "coord_1")
        print(f"  Coord done: score={sc_out:.7f}")

    # === Phase 5: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Multi-length exploration ===")
    lengths_to_try = []
    if n_base >= 100:
        lengths_to_try = [n_base - 100, n_base + 100, n_base + 200]

    for nl in lengths_to_try:
        if nl < 50:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(remaining * 0.2, 120.0)
        if t5 < 30:
            break
        print(f"  Trying n={nl}")
        init_nl = interpolate_sequence(best_seq, nl)
        # Quick LP at this length
        a_out, sc_out = nesterov_lp_optimize(init_nl, t5, solver)
        try_update(a_out, f"lp_n{nl}")

    # === Phase 6: Run peak shaving again on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Second peak shaving ===")
    t6 = min(remaining * 0.25, 100.0)
    if t6 > 10:
        a_out, sc_out = local_peak_shaving(best_seq, t6, seed=999)
        try_update(a_out, "peak_shave_2")

    # === Phase 7: Extra mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.3, 60.0)
        if t7 > 10:
            print(f"=== Phase 7: Extra mirror descent ===")
            a_out, sc_out = subgradient_mirror_descent(best_seq, t7, seed=2000)
            try_update(a_out, "mirror_2")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.5, 40.0)
        if t8 > 5:
            print(f"=== Phase 8: Final coordinate polish ===")
            a_out, sc_out = coordinate_descent_polish(best_seq, t8, seed=3000)
            try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
