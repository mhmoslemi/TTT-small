def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T_cache=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    if T_cache is None:
        T = build_conv_matrix(a, n)
    else:
        T = T_cache
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_val


def normalize_seq(a, norm_val):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_val / len(a)
    return a * norm_val / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_adaptive=True, restart_threshold=60.0):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    else:

    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 2000000):

            break

        if b_sum < 1e-15:

        n_iters += 1

        if use_adaptive:
            if best_score > 0:
                ratio = M_curr / best_score
                alpha = 2.0 / (t + 2.0)
                alpha = min(alpha * min(max(ratio, 0.1), 3.0), 1.0)
            else:
                alpha = 2.0 / (t + 2.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = normalize_seq(y_new, norm)

        if t == 1:
            a_new = g.copy()
            momentum = 0.0
        else:
            new_momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_seq(a_new, norm)
            momentum = new_momentum

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()

            last_improve = time.monotonic()

            if score_avg < best_score:

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")

def lse_objective_and_grad(a, n, T_temp):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    obj = 2.0 * n * lse / (s * s)

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad


def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        stage_time = min(remaining / (n_stages - si), 100.0)
        if stage_time < 2:
            break

        lr = 1.0 / (T * n)
        best_lr = lr
        t_stage = time.monotonic()
        n_inner = 0

        while time.monotonic() - t_stage < stage_time:
            obj, grad = lse_objective_and_grad(a, n, T)
            if not np.isfinite(obj):
                break

            a_new = a - lr * grad
            a_new = normalize_seq(a_new, norm)
            score_new = fast_autocorr(a_new).max()

            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                best_lr = min(best_lr * 1.1, 1.0 / (T * 10))
                lr = best_lr
                n_inner += 1
            else:
                lr *= 0.7
                if lr < 1e-10:
                    break

        if n_inner > 0 and si % 2 == 0:
            print(f"    LSE T={T:.1f}: {n_inner} steps, best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(np.asarray(a0, dtype=np.float64), norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        improved = False

        for k_peak in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if improved:
                    break
                if a[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 3:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]

                for dst in candidates_sorted[:5]:
                    if improved:
                        break
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = normalize_seq(a_trial, norm)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                for _ in range(50):
                    if time.monotonic() - t0 > time_limit:
                        break
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.005, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize_seq(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score

            momentum = 0.0
        else:
            new_momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_seq(a_new, norm)
            momentum = new_momentum

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = normalize_seq(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * (1.0 + 0.3 * np.cos(4.0 * np.pi * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.2 * np.sin(2.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1400, 1600, 1800, 2000]
    scan_results = {}
    scan_budget = time_budget * 0.12

    for idx_scan, n_scan in enumerate(scan_lengths):
        elapsed = time.monotonic() - t_global
        if elapsed > scan_budget:
            break
        remaining_scan = scan_budget - elapsed
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - idx_scan), 100.0)

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        print(f"    Best scan: n={best_scan_n}, score={scan_best_sc:.8f}")
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP optimization at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, sc_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_adaptive=True, restart_threshold=45.0)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: Try a different length (n_base + 200 or n_base - 200)
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 200
    print(f"\n=== Phase 2: LP n={n_alt} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out_alt, sc_out_alt = nesterov_lp_optimize(a_norm_alt, t2, solver, verbose=True, use_adaptive=True)
        do_try_update(a_out_alt, f"lp_alt_n{n_alt}")

    # Phase 3: LSE gradient descent at best n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE GD n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out_lse, _ = lse_gradient_descent(a_norm, t3, T_start=80.0, T_end=1.0, n_stages=10)
        do_try_update(a_out_lse, f"lse_gd_n{n_best}")

    # Phase 4: Targeted peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Targeted peak n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out_peak, _ = targeted_peak_reduction(a_norm, t4, seed=555)
        do_try_update(a_out_peak, f"peak_reduce_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out_cd, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        do_try_update(a_out_cd, "coord_polish")

    # Phase 6: Final LP pass (short) to clean up
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out_final, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, use_adaptive=True, restart_threshold=30.0)
        do_try_update(a_out_final, f"lp_final_n{n_best}")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.5
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out_fin_cd, _ = coordinate_descent_polish(a_norm, t7, seed=77777)
        do_try_update(a_out_fin_cd, "coord_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
