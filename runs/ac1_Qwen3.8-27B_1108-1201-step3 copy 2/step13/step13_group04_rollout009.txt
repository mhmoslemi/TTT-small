"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Projected L-BFGS on LSE surrogate with multi-temperature cooling

3. Pairwise coordinate descent for final refinement

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):
    """Compute gradient of T*log(sum_k exp(b_k/T)) where b = autocorr(a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # First, project onto {x >= 0, sum(x) = 1}, then scale
    u = v / target_sum if target_sum > 0 else v
    # Standard simplex projection
    d = np.sort(u)[::-1]
    cssv = np.cumsum(d)
    rho = np.where(d * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    proj = np.maximum(u - theta, 0)
    return proj * target_sum

def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def lbfgs_projected(a0, time_limit, T_schedule, m=20, verbose=True):
    """
    Projected L-BFGS on LSE surrogate.
    a0: initial point (will be normalized)
    time_limit: seconds
    T_schedule: list of temperatures to use (decreasing)
    m: number of L-BFGS memory pairs
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # L-BFGS memory
    s_mem = []  # step directions
    y_mem = []  # gradient differences
    rho_mem = []  # 1 / (y^T s)

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Reset L-BFGS memory at each temperature change
        s_mem.clear()
        y_mem.clear()
        rho_mem.clear()

        g = lse_gradient(a, T)
        it = 0
        max_iters = max(100, int(remaining * 5.0))

        while it < max_iters and time.monotonic() - t0 < time_limit:
            # Compute search direction using two-loop recursion
            if len(s_mem) == 0:
                d = -g
            else:
                alpha = np.zeros(len(s_mem))
                q = g.copy()
                # First loop (reverse)
                for i in range(len(s_mem) - 1, -1, -1):
                    alpha[i] = rho_mem[i] * np.dot(s_mem[i], q)
                    q -= alpha[i] * y_mem[i]
                # Scale by gamma = s^T y / y^T y (approximately)
                if len(s_mem) > 0:
                    gamma = np.dot(s_mem[-1], y_mem[-1]) / max(np.dot(y_mem[-1], y_mem[-1]), 1e-30)
                else:
                    gamma = 1.0
                r = gamma * q
                # Second loop (forward)
                for i in range(len(s_mem)):
                    beta = rho_mem[i] * np.dot(y_mem[i], r)
                    r += (alpha[i] - beta) * s_mem[i]
                d = -r

            # Projected line search (backtracking)
            step = 1.0
            f_curr = lse_value(a, T)
            # Projected gradient direction
            d_proj = d - np.mean(d)  # Make d sum-zero for the simplex
            # Actually, for projected methods, we use the direction as-is
            # and project after the step

            success = False
            for ls_iter in range(30):
                a_trial = a + step * d
                a_trial = project_simplex(a_trial, norm)
                f_trial = lse_value(a_trial, T)
                if f_trial < f_curr - 1e-4 * step * np.dot(g, d):
                    success = True
                    break
                step *= 0.5
                if step < 1e-12:
                    break

            if not success:
                # Try a smaller step or break
                if step < 1e-14:
                    break
                a_trial = a + step * d
                a_trial = project_simplex(a_trial, norm)

            # Update L-BFGS memory
            g_new = lse_gradient(a_trial, T)
            s_new = a_trial - a
            y_new = g_new - g
            sy = np.dot(s_new, y_new)
            if sy > 1e-12:
                s_mem.append(s_new)
                y_mem.append(y_new)
                rho_mem.append(1.0 / sy)
                if len(s_mem) > m:
                    s_mem.pop(0)
                    y_mem.pop(0)
                    rho_mem.pop(0)

            a = a_trial
            g = g_new
            it += 1

            # Track best
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    L-BFGS T={T:.3f} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """
    Exact subgradient method for max(conv(a,a)).
    Uses the active set (lags achieving the maximum) to compute the subgradient.
    Adaptive step size based on Polyak's rule.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Lower bound: by averaging, max_b >= (sum(a))^2 / (2n-1) = 2n/(2n-1) * (norm^2/(2n-1))
    # Actually with sum(a) = norm = sqrt(2n): max_b >= norm^2 / (2n-1) = 2n/(2n-1)
    lower_bound = 2.0 * n / (2 * n - 1)

    step_size = 1.0 / np.sqrt(n)
    decay = 0.999
    stall = 0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Compute subgradient: average of gradients of active terms
        # Active terms are those with b_k close to max_b
        active_mask = b >= max_b - 1e-6 * max_b
        n_active = active_mask.sum()

        # Subgradient of max(b) at a: (1/|A|) * sum_{k in A} db_k/da
        # db_k/da_m = a[k-m] + a[m-k] (with boundary handling)
        # This is the same as lse_gradient with uniform weights on active set
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / n_active
        # Normalize w so sum = 1
        w /= w.sum()

        fft_size = 1 << (3 * n - 2).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        A_fwd = rfft(a, fft_size)
        W = rfft(w, fft_size)
        conv1 = irfft(A_rev * W, fft_size)
        conv2 = irfft(A_fwd * W, fft_size)
        r1 = conv1[n - 1:n - 1 + n]
        r2 = conv2[:n]
        grad = r1 + r2

        # Polyak step size
        gap = max_b - lower_bound
        if gap > 1e-10:
            step = step_size * gap / (np.dot(grad, grad) + 1e-12)
        else:
            break

        a_new = a - step * grad
        a_new = project_simplex(a_new, norm)

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        step_size *= decay

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad it={it} ({elapsed:.0f}s): best={best_score:.8f}, gap={gap:.8f}")

        if stall > 5000:
            break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Raised cosine with varying width
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    elif choice == 6:
        # Multi-peak
        a = np.zeros(n)
        for _ in range(rng.integers(3, 6)):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-0.5 * ((np.arange(n) / n - center) / width) ** 2)
    else:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):

    except Exception:
