def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    x_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Solve LP best response
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            # Fallback: use uniform
            g = np.ones(n) * norm / n
        else:
            b_sum = b.sum()
            if b_sum < 1e-15:
                g = np.ones(n) * norm / n
            else:
                g = b * norm / b_sum

        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            x_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - x_prev)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x

        y = y_new
        x_prev = x_new

        # Check scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_x = fast_autocorr(x_new).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()
            last_improve = time.monotonic()

        # Running average
        avg_a += x_new
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 20:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def mirror_descent_set(a0, time_limit, eta_scale=1.0, verbose=False):
    """Mirror descent (multiplicative weights) for the autocorrelation game."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Use log-space for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_sum = np.logsumexp(log_a)  # Wait, I need to import this or implement it
    
    # Actually, let me work in linear space but with careful scaling
    log_a = np.log(np.maximum(a, 1e-300))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_curr = np.exp(log_a - log_a.max())  # work in scaled space
        a_curr = a_curr / a_curr.sum() * norm  # normalize
        
        b = fast_autocorr(a_curr)
        k_star = int(np.argmax(b))
        
        # Subgradient: grad_i = sum_{j: i+j=k*} 2 * a_curr[j]
        grad = np.zeros(n)
        valid_i = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        valid_j = k_star - valid_i
        grad[valid_i] = 2.0 * a_curr[valid_j]
        
        # Mirror descent update in log space
        eta = eta_scale / np.sqrt(t + 1.0)
        log_a = log_a - eta * grad
        log_a = log_a - log_a.max()  # renormalize for stability
        
        a_curr = np.exp(log_a)
        s = a_curr.sum()
        if s < 1e-15:
            log_a = np.zeros(n)
        else:
            a_curr = a_curr * norm / s
        
        score = fast_autocorr(a_curr).max()
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient
    w = exp_vals / exp_vals.sum()
    
    # cross = conv(a, w) (linear convolution, take first n elements)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    conv_grad = 4.0 * cross
    obj_val = obj
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return obj, grad.ravel()

def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    """LSE smooth optimization with scipy L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)

        x0 = a.copy()

        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score

def coord_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent with peak-targeted moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_k = min(10, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        improved = False
        
        for k_peak in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Sort by contribution (descending)
            order = np.argsort(contrib)[::-1]
            top_idx = idx[order[:min(8, len(idx))]]
            
            # Build peak influence mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                
                # Target positions: low influence on peaks
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                
                # Pick a few good targets (low value, low peak influence)
                cand_scores = a[candidates] + 0.1 * b[np.minimum(candidates + k_peak, 2*n-2)]
                # Actually, let's just pick randomly from non-peak positions
                n_try = min(5, len(candidates))
                if n_try == 0:
                    continue
                targets = candidates[rng.choice(len(candidates), size=n_try, replace=False)]
                
                for dst in targets:
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 5 == 0:
                for _ in range(50):
                    if time.monotonic() - t0 > time_limit:
                        break
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            
            if stall > 300:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Multi-frequency
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for freq, amp in [(7, 0.2), (13, 0.1), (23, 0.05), (41, 0.03)]:
            a += amp * np.cos(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
    else:
        # Random with structure
        a = np.ones(n)
        for _ in range(3):
            freq = rng.uniform(3, 30)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.1 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    phase0_time = time_budget * 0.08
    scan_lengths = [1400, 1600, 1800, 2000]
    scan_results = {}

    for n_scan in scan_lengths:
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.12:
            break
        remaining_scan = phase0_time - elapsed
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 60.0)
        if scan_time < 10:
            break

        if best_seq is not None and len(best_seq) > 0:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: raw_score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    # Pick best from scan
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 1b: Try a slightly different length
    n_alt = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1b: LP n={n_alt} (budget={remaining:.0f}s) ===")
    t1b = min(remaining * 0.15, 180.0)
    if t1b > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp(a_norm, t1b, solver, verbose=False)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 2: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_curr} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = mirror_descent_set(a_norm, t2, eta_scale=2.0, verbose=False)
        do_try_update(a_out, f"mirror_n{n_curr}")

    # Phase 3: LSE scipy
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_curr} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=30.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_curr}")

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_curr} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.30, 400.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t4, seed=12345)
        do_try_update(a_out, "coord1")

    # Phase 5: Another round of LP with fresh momentum
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 5: LP restart n={n_curr} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 15:
        # Add small random perturbation to escape local minimum
        a_pert = best_seq + rng.normal(0, 0.005, len(best_seq))
        a_pert = np.maximum(a_pert, 0.0)
        a_norm = a_pert * np.sqrt(2.0 * n_curr) / a_pert.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=False)
        do_try_update(a_out, f"lp_restart_n{n_curr}")

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 6: Final coord n={n_curr} (budget={remaining:.0f}s) ===")
