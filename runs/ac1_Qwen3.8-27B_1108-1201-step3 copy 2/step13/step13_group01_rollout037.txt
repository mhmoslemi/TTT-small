```python
class LPContext:
    def __init__(self, n):
        self.n = n
        self.size = 2 * n - 1
        # Precompute row/col indices
        rows = []
        cols = []
        data_indices = []  # index into a for each (row, col)
        for i in range(n):
            for k in range(i, min(i + n, self.size)):
                rows.append(k)
                cols.append(i)
                data_indices.append(k - i)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.data_indices = np.array(data_indices, dtype=np.int32)
    
    def build_matrix(self, a):
        data = a[self.data_indices]
        return sp.csr_matrix((data, (self.rows, self.cols)), shape=(self.size, self.n))
```

def lp_best_response(a, M, solver, ctx):
    n = ctx['n']
    size = ctx['size']
    rows = ctx['rows']
    cols = ctx['cols']
    data_idx = ctx['data_idx']
    
    data = a[data_idx]
    T = sp.csr_matrix((data, (rows, cols)), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val
