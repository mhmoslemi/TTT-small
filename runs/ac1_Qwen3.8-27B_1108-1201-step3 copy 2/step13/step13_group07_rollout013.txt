def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Binary search on threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 1e-15:
            result = result / s * target_sum
    return result

    stall_count = 0
    stall_threshold = 200
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Fast best response (approximate)
        n_grad_steps = 15 if t < 1000 else 8
        b_star = fast_best_response(y, n_steps=n_grad_steps)
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)
        
        y = y_new
        a = a_new
        
        # Check scores (less frequently for speed)
        if t % 5 == 0 or t < 100:
            score_a = internal_score(a)
            if score_a < best_score:
                if score_a < best_score - 1e-10:
                    stall_count = 0
                best_score = score_a
                best_a = a.copy()
            
            score_y = internal_score(y)
            if score_y < best_score:
                if score_y < best_score - 1e-10:
                    stall_count = 0
                best_score = score_y
                best_a = y.copy()
        
        # Time-averaged iterate check
        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        stall_count += 1
        if stall_count > stall_threshold:
            # Try a restart with perturbation
            rng = np.random.default_rng(t)
            a_pert = a + 0.01 * rng.normal(0, 1, n)
            a_pert = project_simplex(a_pert)
            y = a_pert.copy()
            a = a_pert.copy()
            stall_count = 0
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NF t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def exact_lp_best_response(a_fixed, solver=None):
    """Solve exact LP: min M s.t. (a*b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix using Toeplitz structure
    # (a*b)_k = sum_j a_{k-j} * b_j for valid indices
    # We can express this as T @ b where T[k,j] = a[k-j] if valid
    # But for LP, we just need the constraint (a*b)_k <= M
    
    # Use scipy to compute the convolution matrix efficiently
    # T is (2n-1) x n, T[k,j] = a[k-j] if 0<=k-j<n else 0
    # Instead of building the full matrix, use the fact that it's Toeplitz
    
    # For the LP, we need T @ b_var <= M_var
    # Let's build T efficiently
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        T[start:start + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]
    solvers = [s for s in solvers if s is not None]
    solvers = list(dict.fromkeys(solvers))  # remove duplicates, preserve order
    
    for sol in solvers:
        try:
            prob.solve(solver=sol, max_iter=500)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                b_val = b_val / s
                return b_val
        except Exception:
            continue
    return None

def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Spectral shaping to flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=30, lr=0.005):
    """LSE-smoothed Adam optimization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Estimate iterations per stage
        time_per_stage = time_limit / n_stages
        est_time_per_iter = 2 * n / 1e6  # rough estimate
        iters = max(20, int(remaining / max(1, (n_stages - si)) / time_per_stage * time_per_stage))
        iters = max(20, min(500, iters))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            obj, grad_a, w, b = lse_objective_and_grad(a, stage_T)
            
            # Convert gradient from a-space to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / (1 + 0.05 * si)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 50 == 0:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}): best={best_score:.6f}")
    
    return best_a, best_score
