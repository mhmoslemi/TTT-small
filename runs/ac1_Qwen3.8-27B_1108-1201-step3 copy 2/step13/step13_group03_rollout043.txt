import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Helper functions (all top-level)

def _fft_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    """Project onto probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        return out / s
    return np.ones(n) / n

def _lse_val(c, T):
    """LSE-smoothed max of c."""
    c_shifted = np.clip((c - c.max()) / T, -50, 50)
    return T * (c.max() / T + np.log(np.exp(c_shifted).sum()))
    # Simplified: T * log(sum(exp(c/T))) but with numerical stability

def _lse_gradient(a, T):
    """Gradient of LSE-smoothed max self-convolution."""
    n = len(a)
    c = _fft_conv(a)
    c_shifted = np.clip((c - c.max()) / T, -50, 50)
    w = np.exp(c_shifted)
    w /= w.sum()
    
    # FFT size for cross-correlation and convolution
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv = irfft(A * W, fft_size)[:n]

    grad = cross[:n] + conv
    return grad

    a = irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
