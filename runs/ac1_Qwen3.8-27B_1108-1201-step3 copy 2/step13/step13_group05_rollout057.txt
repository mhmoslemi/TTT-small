def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def compute_full_subgrad(a, ac, threshold_frac=0.01):
    """Compute subgradient using all lags within threshold of max."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    # All lags within threshold of max
    active = np.where(ac >= M * (1.0 - threshold_frac))[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])
    
    grad = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        j = k_int - idx
        grad[idx] += 2.0 * a[j]
    return grad

def compute_topk_subgrad(a, ac, top_k=10, beta=2.0):
    """Compute weighted subgradient using top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start >= i_end + 1:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix more efficiently
    # conv(a,b)[k] = sum_j a[k-j]*b[j] for valid indices
    # We need: for each k in 0..2n-2, sum_j a[k-j]*b[j] <= M
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        print(f"    LP solve error: {e}")
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    t0 = time.monotonic()
    
    y = a.copy()  # Average iterate
    x = a.copy()  # Current iterate

        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(x).max()

            g_norm = np.linalg.norm(g, ord=1)
            if g_norm < 1e-15:

            b = np.maximum(c - g * c / (n * g_norm), 0)
            b = b / b.sum() * c
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def create_asymmetric_init(n, rng):
    """Create an asymmetric initialization that naturally has low center correlation."""

    f = np.power(x, 1.0/alpha) * np.power(1-x, alpha)
    # Add some noise
    f += rng.uniform(0, 0.1, n) * f.mean()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f

            break
        ac = fast_autocorr(a)
        M = ac.max()
        if use_full:
            g = compute_full_subgrad(a, ac, threshold)
        else:
            g = compute_topk_subgrad(a, ac, top_k, beta)
        g_norm = np.linalg.norm(g, ord=np.inf)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = 1.0 / (np.sqrt(t + 1.0) * np.sqrt(n))
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def frank_wolfe_optimize(a0, time_limit, verbose, threshold):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = a.copy()
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M = ac.max()
        g = compute_full_subgrad(a, ac, threshold)
        if np.max(g) - np.min(g) < 1e-14:
            break
        # Frank-Wolfe: linear minimizer over simplex is vertex with min g
        lw = np.zeros(n)
        lw[np.argmin(g)] = c
        # Step size: use exact line search or gamma = 2/(t+2)
        gamma = 2.0 / (t + 2.0)
        a_new = (1.0 - gamma) * a + gamma * lw
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        avg_a += a
        if (t + 1) % 100 == 0:
            avg_check = avg_a / (t + 1)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_best_response_dual(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    cons = T @ b_var <= M
    constraints = [cons]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    # Try to get dual variables
    dual = None
    try:
        dual_vals = cons.dual_value
        if dual_vals is not None:
            dual = np.maximum(np.asarray(dual_vals, dtype=np.float64), 0.0)
            if dual.sum() < 1e-15:
                dual = None
    except Exception:
        dual = None
    return b_val, dual

def nesterov_lp_optimize(a0, time_limit, solver, verbose, max_iters):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    # Pre-build matrix (a changes each iter, so we rebuild)
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(x)
        M_curr = ac.max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(x, M_curr, solver)
        if b is None:
            g = compute_full_subgrad(x, ac, 0.0)
            g_l1 = np.sum(g)
            if g_l1 < 1e-15:
                break
            b = np.maximum(1.0 - g / g_l1, 0)
            b_sum = b.sum()
            if b_sum < 1e-15:
                b = np.ones(n) / n
            else:
                b = b / b_sum
            b = b * c
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b * c / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        y = y_new
        x = x_new
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.6f}")
    return best_a, best_m

def hybrid_lp_mirror(a0, time_limit, solver, verbose, lp_every):
    """Alternate: every lp_every mirror steps, do one LP best response."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    last_b = None
    step_count = 0
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M = ac.max()
        if t % lp_every == 1:
            # Do LP
            b = lp_best_response(a, M, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    last_b = b * c / b_sum
        if last_b is not None:
            # Move towards LP solution with Nesterov-like step
            alpha = 2.0 / (step_count + 2.0)
            y_new = (1.0 - alpha) * y + alpha * last_b
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * c / s_y
            if step_count == 0:
                a_new = last_b.copy()
            else:
                beta = (step_count - 1.0) / (step_count + 2.0)
                a_new = y_new + beta * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_x = a_new.sum()
                if s_x < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * c / s_x
            y = y_new
            a = a_new
        else:
            # Mirror descent step using subgradient
            g = compute_full_subgrad(a, ac, 0.01)
            g_norm = np.linalg.norm(g, ord=np.inf)
            if g_norm < 1e-15:
                break
            eta = 1.0 / (np.sqrt(step_count + 1.0) * np.sqrt(n))
            log_a = np.log(a + 1e-30)
            log_a_new = log_a - eta * (g / g_norm)
            max_log = log_a_new.max()
            a_new = np.exp(log_a_new - max_log)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * c / s_new
            a = a_new
        step_count += 1
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    Hybrid t={t} ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if len(contrib) == 0:
                continue
            src = idx[np.argmax(contrib)]
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            cands_sorted = candidates[np.argsort(a[candidates])]
            n_picks = max(1, min(10, len(cands_sorted) // 3))
            picks = cands_sorted[:n_picks]
            dst = picks[rng.integers(len(picks))]
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * c / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_m * (1.0 - 1e-10):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            n_pert = rng.integers(1, 6)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-7:
                    frac = rng.uniform(0.003, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
            if stall > 40:
                break
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
