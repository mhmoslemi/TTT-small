def build_conv_matrix(a):
    """Build the Toeplitz matrix T where (T @ b)[k] = sum_i a[i]*b[k-i]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, warm_start=None):
    """Solve LP: maximize sum(b) subject to T@b <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)

    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def compute_lse_gradient(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k using FFT."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_sum_nonneg(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum} approximately."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T_curr = max(1e-4, T * (0.01) ** progress)

        # LSE weights
        shifted = (c - max_c) / T_curr
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient
        grad = compute_lse_gradient(a, w)

        # Backtracking line search
        accepted = False
        local_step = step_size
        for _ in range(15):
            a_new = a - local_step * grad
            a_new = project_sum_nonneg(a_new, norm)
            c_new = fast_autocorr(a_new)
            max_c_new = c_new.max()
            if max_c_new < max_c * (1 + 1e-12) or local_step < 1e-15:
                a = a_new
                c = c_new
                max_c = max_c_new
                accepted = True
                break
            local_step *= 0.5

        if not accepted:
            # Try a random perturbation to escape
            rng = np.random.default_rng(steps)
            idx = rng.choice(n, size=max(1, n // 100), replace=False)
            a_pert = a.copy()
            a_pert[idx] *= (1 + rng.uniform(-0.1, 0.1, len(idx)))
            a = project_sum_nonneg(a_pert, norm)
            c = fast_autocorr(a)
            max_c = c.max()

        step_size *= 1.01  # slowly increase step size
        step_size = min(step_size, 1.0 / (n * 0.01))

        steps += 1

        if max_c < best_score:
            best_score = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart if stuck
        if no_improve > 5000:
            rng = np.random.default_rng(steps * 3 + 1)
            a_pert = a * (1 + rng.normal(0, 0.02, n))
            a = project_sum_nonneg(a_pert, norm)
            c = fast_autocorr(a)
            max_c = c.max()
            no_improve = 0
            step_size = 1.0 / (n * 0.1)

        if verbose and steps % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T_curr:.6f}, best={best_score:.8f}")

    return best_a, best_score


def pair_swap_refine(a0, time_limit, seed=42, verbose=True):
    """Local refinement via pair swaps targeting peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    c = fast_autocorr(a)
    best_score = c.max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_sa = max(0.1, best_score * 0.02)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T_curr = T_sa * (0.01) ** elapsed_frac

        # Find top peaks
        top_k = np.argsort(c)[::-1][:min(10, 2 * n - 1)]

        improved = False
        for k in top_k:
            if c[k] < best_score * 0.9:
                continue

            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]
            partner = k - src

            # Find good destination: low value, not contributing to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            peak_mask[src] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick destination with lowest value
            dst = candidates[np.argmin(a[candidates])]

            # Try different fractions
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                c_trial = fast_autocorr(a_trial)
                score = c_trial.max()

                if score < best_score:
                    a = a_trial
                    c = c_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T_curr, 1e-10)):
                    a = a_trial
                    c = c_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-12:
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        c_trial = fast_autocorr(a_trial)
                        score = c_trial.max()
                        if score < best_score:
                            a = a_trial
                            c = c_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T_curr, 1e-10)):
                            a = a_trial
                            c = c_trial

            if n_stall > 500:
                # Larger perturbation
                idx_pert = rng.choice(n, size=max(5, n // 30), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.05, 0.3)
                    j2 = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j2] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    c_trial = fast_autocorr(a_trial)
                    if c_trial.max() < best_score * 1.02:
                        a = a_trial
                        c = c_trial
                        if c_trial.max() < best_score:
                            best_score = c.max()
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PairSwap improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def antisymmetric_init(n, rng):
    """Create an initialization with antisymmetric structure."""
    base = rng.random(n) + 0.1
    # Add antisymmetric component
    half = n // 2
    anti = np.zeros(n)
    anti[:half] = rng.random(half)
    anti[half:] = -anti[:half][::-1]
    if n % 2 == 1:
        anti[half] = 0
    a = base + 0.5 * anti
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization with multiple lengths."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    best_n = 0

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            best_n = len(best_seq)
            print(f"Parent: n={best_n}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 2000
        best_seq = antisymmetric_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        best_n = n_base
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = best_n

    # Define candidate lengths to try
    candidate_lengths = sorted(set([
        n_base,
        n_base + 200,
        n_base + 400,
        max(1000, n_base - 200),
    ]))

    # Filter to reasonable range
    candidate_lengths = [nl for nl in candidate_lengths if 500 <= nl <= 5000]
    if not candidate_lengths:
        candidate_lengths = [n_base]

    print(f"Candidate lengths: {candidate_lengths}")

    # For each candidate length, run the optimization pipeline
    for n_try in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break

        print(f"\n{'='*60}")
        print(f"Processing n={n_try} (remaining={remaining:.0f}s)")
        print(f"{'='*60}")

        # Get initial point for this length
        if n_try == n_base:
            init = best_seq.copy()
        else:
            init = interpolate_sequence(best_seq, n_try)

        # Also try an antisymmetric initialization
        init_anti = antisymmetric_init(n_try, np.random.default_rng(seed + n_try))

        # Pick the better initial
        sc1 = evaluate_safe(init)
        sc2 = evaluate_safe(init_anti)
        if np.isfinite(sc1) and (not np.isfinite(sc2) or sc1 <= sc2):
            init = init
        else:
            init = init_anti

        do_try_update(init, f"init_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Phase A: LP Nesterov (most powerful for global optimization)
        t_lp = min(remaining * 0.30, 400.0)
        if t_lp > 30:
            print(f"\n  Phase A: LP Nesterov n={n_try} (budget={t_lp:.0f}s)")
            a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t_lp, solver, verbose=True)
            do_try_update(a_out, f"lp_n{n_try}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Phase B: LSE gradient descent (fast local optimization)
        t_lse = min(remaining * 0.25, 300.0)
        if t_lse > 15:
            print(f"\n  Phase B: LSE-GD n={len(best_seq)} (budget={t_lse:.0f}s)")
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_gradient_descent(a_norm, t_lse, verbose=True)
            do_try_update(a_out, f"lse_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Phase C: Pair-swap refinement (fine polishing)
        t_ps = min(remaining * 0.20, 200.0)
        if t_ps > 10:
            print(f"\n  Phase C: PairSwap n={len(best_seq)} (budget={t_ps:.0f}s)")
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = pair_swap_refine(a_norm, t_ps, seed=seed + n_cur, verbose=True)
            do_try_update(a_out, f"ps_n{n_cur}")

    # Final phases: intensive polishing on the best found
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)

        # Final LP pass
        t_final_lp = min(remaining * 0.40, 300.0)
        if t_final_lp > 20:
            print(f"\n{'='*60}")
            print(f"Final LP polish n={n_best} (budget={t_final_lp:.0f}s)")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t_final_lp, solver, verbose=True)
            do_try_update(a_out, f"lp_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Final LSE pass
        t_final_lse = min(remaining * 0.35, 200.0)
        if t_final_lse > 10:
            print(f"\nFinal LSE polish n={len(best_seq)} (budget={t_final_lse:.0f}s)")
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_gradient_descent(a_norm, t_final_lse, verbose=True)
            do_try_update(a_out, f"lse_final_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)

        # Final pair-swap
        t_final_ps = min(remaining * 0.50, 150.0)
        if t_final_ps > 5:
            print(f"\nFinal PairSwap n={len(best_seq)} (budget={t_final_ps:.0f}s)")
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = pair_swap_refine(a_norm, t_final_ps, seed=999, verbose=True)
            do_try_update(a_out, f"ps_final_n{n_cur}")

    # Final verification
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence!")
        return [1.0, 1.0, 1.0, 1.0]
