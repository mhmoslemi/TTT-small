def convolve_with_a(a, b):
    """Compute T @ b where T is the convolution matrix with a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lp_solve_fast(a, M, solver, warm_start=None, max_active=100):
    """Solve max sum(b) s.t. T@b <= M, b >= 0, with constraint reduction."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    constraints = [convolve_with_a(a, b_var) <= M]  # Can't do this with cvxpy vectorized

        circ = convolve_with_a(a, warm_start)
        tol = 1e-8 * max(M, 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())

                vals[active] = -np.inf
                additional = np.argsort(vals)[::-1][:max_active - n_active]
                active = np.zeros(2*n-1, dtype=bool)
                active[:2*n-1] = False
                active_indices = np.where(circ >= M - tol)[0].tolist() + additional.tolist()
                active[active_indices] = True
        
        T_sub = _build_T_rows(a, active)
        M_sub = M[active]
        constraints = [T_sub @ b_var <= M_sub]
    else:

        top_idx = np.argsort(self_conv)[::-1][:max_active]
        active = np.zeros(2*n-1, dtype=bool)
        active[top_idx] = True
        T_sub = _build_T_rows(a, active)
        M_sub = M[active]
        constraints = [T_sub @ b_var <= M_sub]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:

        return None

        return None

def _build_T_rows(a, active_mask):
    """Build rows of T corresponding to active constraints."""
    n = len(a)
    size = 2 * n - 1
    active_indices = np.where(active_mask)[0]
    n_active = len(active_indices)
    
    T = np.zeros((n_active, n))
    for row, i in enumerate(active_indices):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        T[row, j_start:j_end+1] = a[j_start:j_end+1]
    return T

def subgradient_polish(a0, time_limit, verbose=True):
    """Polish using exact subgradient of max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Subgradient of max(conv(a,a)) at a is 2 * T^T * e_k where k is argmax
    # T^T * e_k has entries (T^T e_k)[j] = T[k,j] = a[j] if max(0,k-n+1) <= j <= min(k, n-1)
    # So the subgradient component for coordinate j is 2 * a[j] * (number of peak positions k where j is in range)
    # More precisely: subgrad[j] = 2 * sum_{k in peaks} a[j] * 1[k-j in [0,n-1] and j in [0,n-1]]
    # = 2 * a[j] * count of peak positions k in [j, j+n-1]
    
    steps = 0
    lr = 1.0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peaks = np.where(np.isclose(b, max_b, atol=1e-10 * max_b))[0]
        
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in peaks:
            j_start = max(0, int(k) - n + 1)
            j_end = min(n - 1, int(k))
            subgrad[j_start:j_end+1] += 2.0 * a[j_start:j_end+1]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break
        
        # Adaptive learning rate (diminishing)
        lr = 1.0 / (np.sqrt(steps + 1) * sg_norm)
        
        a_new = a - lr * subgrad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            n_rounds += 1
            if n_rounds % 5000 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    CD round={n_rounds} ({elapsed:.0f}s): best={best_score:.8f}, improve={n_improve}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def lse_gradient_descent_v2(a0, time_limit, verbose=True):
    """Improved LSE gradient descent with better step size and annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.01, best_score * 0.005)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: grad_i = 2 * sum_k w[k] * a_rev[k-i] (valid range)
        # = 2 * (w * a_rev)_i where * is convolution, evaluated at position i+n-1... 
        # Actually: sum_k w[k] * a[k-i] for valid k-i
        # Let a_rev[j] = a[n-1-j]
        # sum_k w[k] * a[k-i] = sum_k w[k] * a_rev[n-1-(k-i)] = sum_k w[k] * a_rev[n-1-k+i]
        # = (w * a_rev)[n-1+i] ... let me just use the direct computation
        
        # Direct: grad_i = 2 * sum_{k: 0<=k<n+i, 0<=k-i<n} w[k] * a[k-i]
        # = 2 * sum_{j: max(0,i-n+1)<=k<=min(i,n-1)} w[i+j] * a[j]  (substituting k=i+j, j=k-i)
        # Wait, k-i = j, so j >= 0 and j < n, and k = i+j, so 0 <= i+j <= 2n-2
        # grad_i = 2 * sum_{j=max(0,-i)}^{min(n-1, 2n-2-i)} w[i+j] * a[j] for i+j in [0, 2n-2]
        # Since i >= 0: grad_i = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
        
        # Use FFT: this is the cross-correlation of w (length 2n-1) with a (length n)
        # corr[i] = sum_j w[i+j] * a[j] = (w * a_rev)[i + n - 1]
        a_rev = a[::-1]
        fft_size = 1 << ((2*n - 1 + n - 1) - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Backtracking line search (simplified)
        step = 2.0 / (n * (1.0 + steps * 0.001))
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new * norm / s_new
        
        # Check if step was too large
        score_new = fast_autocorr(a_new).max()
        if score_new > best_score * 1.5:
            # Halve the step
            step *= 0.1
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a
            else:
                a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE2 step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov with FFT-based constraint eval, aggressive constraint reduction,
   and amortized LP solves (solve every K iterations for speed)
2. Subgradient polishing with exact max subgradient and adaptive step sizes
3. LSE (LogSumExp) smooth gradient descent via FFT
4. Multi-length exploration (1600, 1800, 2000) with interpolation transfer
5. Diverse initialization and adaptive restart strategies
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def convolve_with_a(a, b):
    """Compute T @ b where T is the convolution matrix structure with a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a, active_mask):
    """Build rows of T corresponding to active constraints."""
    n = len(a)
    active_indices = np.where(active_mask)[0]
    n_active = len(active_indices)
    if n_active == 0:
        return np.zeros((0, n))
    T = np.zeros((n_active, n))
    for row, i in enumerate(active_indices):
        j_start = max(0, int(i) - n + 1)
        j_end = min(n - 1, int(i))
        T[row, j_start:j_end + 1] = a[j_start:j_end + 1]
    return T


def lp_solve_reduced(a, M, solver, warm_start=None, max_active=80):
    """Solve max sum(b) s.t. T@b <= M, b >= 0 using constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if warm_start is not None and np.all(np.isfinite(warm_start)):
        circ = convolve_with_a(a, warm_start)
        tol = 1e-7 * max(M.max(), 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())

        if n_active == 0:
            active = np.zeros(size, dtype=bool)
            active[np.argmax(circ)] = True

        if n_active < 20:
            vals = circ.copy()
            vals[active] = -np.inf
            extra = np.argsort(vals)[::-1][:max(1, 30 - n_active)]
            full_active = np.zeros(size, dtype=bool)
            full_active[np.where(circ >= M - tol)[0]] = True
            full_active[extra] = True
            active = full_active

        if n_active > max_active:
            # Keep only the most active (tightest) constraints
            active_indices = np.where(active)[0]
            tightness = M[active_indices] - circ[active_indices]
            # Actually keep those closest to or exceeding M
            sorted_idx = active_indices[np.argsort(circ[active_indices])[::-1]][:max_active]
            active = np.zeros(size, dtype=bool)
            active[sorted_idx] = True

        T_sub = build_T_rows(a, active)
        M_sub = M[active]
        constraints = [T_sub @ b_var <= M_sub, b_var >= 0]
    else:
        # No warm start: use constraints based on a's own autocorrelation
        self_conv = fast_autocorr(a)
        top_idx = np.argsort(self_conv)[::-1][:max_active]
        active = np.zeros(size, dtype=bool)
        active[top_idx] = True
        T_sub = build_T_rows(a, active)
        M_sub = M[active]
        constraints = [T_sub @ b_var <= M_sub, b_var >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)

    # Verify with full constraints (FFT-based, fast)
    circ = convolve_with_a(a, b_val)
    if circ.max() > M.max() * (1 + 1e-3):
        return None

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_v2(a0, time_limit, solver, verbose=True, max_active=60, amortize=3):
    """
    LP Nesterov with amortized LP solves.
    Solve LP every 'amortize' iterations, reuse direction in between.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    prev_b = None
    g_cache = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Solve LP (amortized)
        if t % amortize == 1 or g_cache is None:
            b = lp_solve_reduced(y, M_curr, solver, warm_start=prev_b, max_active=max_active)
            if b is None:
                b = lp_solve_reduced(y, M_curr, solver, warm_start=None, max_active=max_active)
                if b is None:
                    continue
            prev_b = b.copy()
            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g_cache = b * norm / b_sum
        else:
            g = g_cache

        g = g_cache
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        a_new = y_new.copy()
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        improved = False
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            improved = True

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            improved = True

        if not improved:
            stall += 1

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP2 iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        # Adaptive restart
        if stall > 300:
            rng = np.random.default_rng(n_iters * 17 + 3)
            strat = rng.integers(4)
            if strat == 0:
                noise = rng.normal(0, 0.03 * a.mean(), n)
                y = np.maximum(y + noise, 0)
                a = np.maximum(a + noise, 0)
            elif strat == 1:
                # Blend with parent shape
                x = np.linspace(0, 1, n)
                shape = np.sin(np.pi * x) ** rng.uniform(0.5, 3.0)
                shape = shape / shape.sum() * norm
                y = 0.7 * y + 0.3 * shape
                a = np.maximum(a + rng.normal(0, 0.02 * a.mean(), n), 0)
            elif strat == 2:
                # Random redistribution
                n_move = max(1, n // 30)
                src = rng.choice(n, size=n_move, replace=False)
                dst = rng.choice(n, size=n_move, replace=False)
                for si, di in zip(src, dst):
                    amt = y[si] * 0.15
                    y[si] -= amt
                    y[di] += amt
                    amt2 = a[si] * 0.15
                    a[si] -= amt2
                    a[di] += amt2
            else:
                # Large random restart (rare)
                y = np.maximum(y + rng.normal(0, 0.1 * a.mean(), n), 0)
                a = np.maximum(a + rng.normal(0, 0.1 * a.mean(), n), 0)

            y = y * norm / max(y.sum(), 1e-15)
            a = a * norm / max(a.sum(), 1e-15)
            stall = 0
            prev_b = None
            g_cache = None
            if verbose:
                print(f"    [Restart iter={n_iters}, strat={strat}]")

    return best_a, best_score


def subgradient_polish_v2(a0, time_limit, verbose=True):
    """Polish using exact subgradient of max autocorrelation with averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Polyak-type step size: lr = (f(a) - f*) / ||g||^2, but f* unknown
    # Use diminishing: lr = c / sqrt(t)
    c = 1.0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peaks = np.where(b >= max_b - 1e-9 * max_b)[0]

        # Compute subgradient: subgrad[j] = 2 * sum_{k in peaks} a[j] * 1[j in [k-n+1, k]]
        subgrad = np.zeros(n)
        for k in peaks:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            subgrad[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1]

        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break

        # Adaptive step with averaging
        lr = c / (np.sqrt(steps + 1) * sg_norm)
        a_new = a - lr * subgrad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad2 step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gd_v2(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with better annealing and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    stall = 0

    T_start = max(0.3, best_score * 0.03)
    T_end = max(0.005, best_score * 0.003)
    base_step = 1.0 / (n * 0.3)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # FFT-based gradient
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Line search (simplified backtracking)
        step = base_step
        a_new = None
        for trial in range(5):
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s_c = a_candidate.sum()
            if s_c < 1e-15:
                step *= 0.5
                continue
            a_candidate = a_candidate * norm / s_c
            score_c = fast_autocorr(a_candidate).max()
            if score_c <= max_b + 1e-10:
                a_new = a_candidate
                break
            step *= 0.5

        if a_new is None:
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Restart temperature
            T_start = max(0.1, best_score * 0.02)
            stall = 0
            base_step = 1.0 / (n * 0.2)

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE2 step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave_v2(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with simulated annealing focusing on top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    stall = 0

    T = max(0.1, best_score * 0.02)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_k = np.argsort(b)[::-1][:min(12, 2 * n - 1)]

        improved = False
        for k in top_k[:6]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Find good destinations (low autocorrelation contribution)
            # Try moving mass to positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick destinations with lowest current value
            sorted_cand = candidates[np.argsort(a[candidates])][:min(8, len(candidates))]
            dst = sorted_cand[rng.integers(len(sorted_cand))]

            frac = rng.choice([0.02, 0.05, 0.1, 0.15, 0.25])
            amt = a[src] * frac
            if amt < 1e-18:
                continue

            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t

            score = fast_autocorr(a_trial).max()

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                T *= 0.98
                break
            elif score < b[ki] and rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.1, 0.3)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                            a = a_trial
                stall = 0
            T *= 1.01  # Slightly increase temperature to escape

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave2 improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP Nesterov (amortized) at base length
    2. Subgradient polish
    3. LSE gradient descent
    4. Peak shaving with SA
    5. LP Nesterov at longer length
    6. Final polishing rounds
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # Phase 1: Main LP Nesterov (amortized)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov amortized n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_v2(a_norm, t1, solver, verbose=True, max_active=60, amortize=2)
        do_try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SubGrad n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.06, 60.0)
    if t2 > 5:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish_v2(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sg_n{n_cur}")

    # Phase 3: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 80.0)
    if t3 > 5:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gd_v2(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 4: Second LP Nesterov round (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_v2(a_norm, t4, solver, verbose=True, max_active=100, amortize=1)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 80.0)
    if t5 > 5:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_v2(a_norm, t5, seed=123, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # Phase 6: Try longer length (n+200)
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_v2(a_norm, t6, solver, verbose=True, max_active=60, amortize=2)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 7: Final LSE polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.08, 70.0)
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gd_v2(a_norm, t7, verbose=False)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Phase 8: Final LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.20, 160.0)
    if t8 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_v2(a_norm, t8, solver, verbose=True, max_active=80, amortize=1)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Phase 9: Final subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        print(f"\n=== Phase 9: Final SubGrad n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish_v2(a_norm, remaining, verbose=False)
        do_try_update(a_out, f"sg_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
