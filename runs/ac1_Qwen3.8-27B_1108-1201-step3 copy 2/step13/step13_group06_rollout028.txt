import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def score_from_autocorr(a):
    """Compute score directly from sequence (normalized)."""
    n = len(a)
    conv_max = fast_autocorr(a).max()
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv_max / (s * s)


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_constrained(y, T, active_idx, solver, warm_b=None):
    """Solve LP: min M s.t. T[active_idx]@b <= M, sum(b)=1, b>=0.
    Only enforce constraints at active_idx (constraint generation)."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n * 2)
    T_active = T[active_idx]
    constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccuratic'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response with constraint generation.
    Only enforces top-K convolution constraints (adaptively chosen).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    WINDOW_MED = 20
    med_buf = [None] * WINDOW_MED
    med_idx = 0
    avg_med = np.zeros(n)
    n_med = 0

    stall_count = 0
    warm_b = None
    last_print = 0

    # Adaptive K: start small, grow over time
    K_base = max(10, n // 4)
    K_max = 2 * n - 1

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution with y to find active constraints
        conv_y = fast_autocorr(y)
        # Top K indices (adaptive)
        K = min(K_max, int(K_base + 0.5 * (K_max - K_base) * min(1.0, t / 5000.0)))
        active_idx = np.argsort(conv_y)[::-1][:K]

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_constrained(y, T, active_idx, solver, warm_b)
        if b_star is None:
            warm_b = None
            # Fallback: use uniform over top-K positions
            b_star = np.zeros(n)
            b_star[active_idx[:n]] = 1.0 / min(n, len(active_idx))
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t <= 3:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Track averages
        avg_long += a
        n_long += 1
        if med_buf[med_idx] is not None:
            avg_med -= med_buf[med_idx]
        med_buf[med_idx] = a.copy()
        avg_med += a
        n_med += 1
        med_idx = (med_idx + 1) % WINDOW_MED

        # Evaluate candidates
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long % 10 == 0 and n_long >= 20:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if n_med >= WINDOW_MED and n_long % 10 == 0:
            avg_m = avg_med / WINDOW_MED
            score_m = 2.0 * n * fast_autocorr(avg_m).max()
            if score_m < best_score:
                best_score = score_m
                best_a = avg_m.copy()
                stall_count = 0

        # Restart if stalled
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_med = np.zeros(n)
            n_med = 0
            med_buf = [None] * WINDOW_MED
            med_idx = 0
            stall_count = 0
            warm_b = None

        if t - last_print > 1000:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, K={K}")
            last_print = t

    return best_a, best_score


def subgrad_adam(a0, time_limit, topk=8, lr_init=0.05, seed=123):
    """
    Subgradient Adam method.
    Gradient of conv(a,a)_k w.r.t a[m] = 2*a[k-m] if 0<=k-m<n else 0.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    stall = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit - 2:
        conv = fast_autocorr(a)
        # Find top-k peaks
        top_idx = np.argsort(conv)[::-1][:topk]
        top_vals = conv[top_idx]

        # Compute subgradient: average gradient of top-k peaks
        grad = np.zeros(n)
        for k in top_idx:
            # grad_k[m] = 2*a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                m_range = np.arange(lo, hi + 1)
                grad[m_range] += 2.0 * a[k - m_range]
        grad /= len(top_idx)

        # Adam update
        t_adam += 1
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad ** 2
        m_hat = m_vec / (1 - beta1 ** t_adam)
        v_hat = v_vec / (1 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still move (for exploration)
            stall += 1
            if stall % 100 == 0:
                lr *= 0.7
                if lr < 1e-5:
                    lr = lr_init
                    a = best_a.copy()
                    m_vec = np.zeros(n)
                    v_vec = np.zeros(n)
                    t_adam = 0

        if t_adam % 200 == 0:
            # Periodic random perturbation
            if rng.random() < 0.3:
                a = best_a.copy()
                noise = rng.normal(0, 0.02, n)
                a = a * (1 + noise)
                a = np.maximum(a, 0)
                sn = a.sum()
                if sn > 1e-10:
                    a = a / sn
                m_vec = np.zeros(n)
                v_vec = np.zeros(n)
                t_adam = 0

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_cycles=50, seed=42):
    """
    Cyclic coordinate descent with 1D bisection for each coordinate.
    For coordinate m, minimize max_k conv(a,a)_k over a[m] >= 0.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # For efficiency, precompute the "base" convolution (without coordinate m)
    # and update it as we modify coordinates

    coords = list(range(n))

    for cycle in range(n_cycles):
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Shuffle for diversity
        rng.shuffle(coords)

        cycle_improved = 0
        for m in coords:
            if time.monotonic() - t0 > time_limit - 2:
                break

            # Current value of a[m]
            x0 = a[m]
            if x0 < 1e-12:
                continue

            # Compute conv with a[m] = 0 (base)
            a[m] = 0.0
            conv_base = fast_autocorr(a)
            a[m] = x0

            # The full conv when a[m] = x:
            # conv_k = conv_base_k + 2*x*a[k-m] (for k != 2m)
            # conv_{2m} = conv_base_{2m} + x^2 (if 2m < 2n-1)
            # We want to minimize max_k conv_k over x >= 0

            # Compute the "slope" for each k: d_k = 2*a[k-m] if 0<=k-m<n and k!=2m
            # Special: for k=2m, the term is x^2 (quadratic)
            d = np.zeros(2 * n - 1)
            lo = 0
            hi = min(n - 1, 2 * n - 2)
            for k in range(lo, hi + 1):
                jm = k - m
                if 0 <= jm < n:
                    if k == 2 * m:
                        d[k] = 0  # quadratic case handled separately
                    else:
                        d[k] = 2.0 * a[jm]

            has_quad = (2 * m < 2 * n - 1)

            # Bisection for optimal x in [0, x0 * 3] (allow some growth)
            x_lo = 0.0
            x_hi = max(x0 * 3.0, 0.01)

            def obj_at_x(x):
                if x < 0:
                    x = 0
                vals = conv_base + d * x
                if has_quad:
                    vals[2 * m] += x * x
                return vals.max()

            # Find minimum via ternary/bisection on the convex function
            # Use golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            c = x_hi - gr * (x_hi - x_lo)
            d_pt = x_lo + gr * (x_hi - x_lo)
            fc = obj_at_x(c)
            fd = obj_at_x(d_pt)

            for _ in range(30):
                if fc < fd:
                    x_hi = d_pt
                    d_pt = c
                    fd = fc
                    c = x_hi - gr * (x_hi - x_lo)
                    fc = obj_at_x(c)
                else:
                    x_lo = c
                    c = d_pt
                    fc = fd
                    d_pt = x_lo + gr * (x_hi - x_lo)
                    fd = obj_at_x(d_pt)

            x_star = (x_lo + x_hi) / 2.0
            # Also check x=0
            obj_star = obj_at_x(x_star)
            obj_zero = obj_at_x(0.0)
            if obj_zero < obj_star:
                x_star = 0.0
                obj_star = obj_zero

            # Apply
            a[m] = x_star
            new_score = 2.0 * n * obj_star
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                cycle_improved += 1

        # Renormalize after each cycle
        sn = a.sum()
        if sn < 1e-15:
            a = best_a.copy()
        else:
            a = a / sn
        # Re-check score after normalization
        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if cycle % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD cycle {cycle} ({elapsed:.0f}s): best={best_score:.6f}, improved={cycle_improved}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Peak shaving: identify top peaks and try to reduce them by
    moving mass from contributing positions to non-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        conv = fast_autocorr(a)
        bmax = conv.max()
        # Top peaks
        top_peaks = np.argsort(conv)[::-1][:8]

        improved = False
        for k in top_peaks:
            # Find the pair (j, k-j) that contributes most to conv[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (not contributing to top peaks)
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:5]:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                danger[js2:je2 + 1] = True
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:min(10, max(3, len(safe_sorted) // 5))]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                a = best_a.copy()
                n_pert = rng.integers(1, 4)
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-10:
                    a = a / st
                    score = 2.0 * n * fast_autocorr(a).max()
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    if choice == 0:
        # Asymmetric peak
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.3 * np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 1:
        # Two peaks (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.25) ** 2) + 0.5 * np.exp(-30 * (x - 0.75) ** 2)
    elif choice == 2:
        # Edge concentrated
        x = np.linspace(0, 1, n)
        a = np.exp(-40 * x) + 0.1 * np.exp(-40 * (x - 1) ** 2)
    elif choice == 3:
        # Random smooth
        k = rng.integers(2, 8)
        coeffs = rng.normal(0, 1, k)
        x = np.linspace(0, 2 * np.pi, n)
        a = np.ones(n)
        for i, c in enumerate(coeffs):
            a += c * np.cos((i + 1) * x)
        a = np.abs(a) + 0.1
    elif choice == 4:
        # Cosine with phase
        x = np.linspace(0, 1, n)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1 + 0.5 * np.cos(3 * np.pi * x + phase)
    elif choice == 5:
        # Random sparse
        a = np.zeros(n)
        n_nonzero = rng.integers(n // 4, n // 2)
        idx = rng.choice(n, n_nonzero, replace=False)
        a[idx] = rng.uniform(0.1, 1.0, n_nonzero)
        a += 0.01
    elif choice == 6:
        # Geometric
        r = rng.uniform(0.3, 0.8)
        x = np.arange(n)
        a = r ** (x - n // 2) ** 2
        a = np.concatenate([a, a[::-1]])[:n]
    elif choice == 7:
        # Sinc-like
        x = np.linspace(-3, 3, n)
        a = np.sinc(x / 3) ** 2
    elif choice == 8:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(0.5, 2.0)
        a = x ** alpha + (1 - x) ** alpha
    elif choice == 9:
        # Alternating
        a = np.ones(n)
        a[::2] *= rng.uniform(2, 5)
    elif choice == 10:
        # Triangular with asymmetry
        x = np.linspace(0, 1, n)
        peak = rng.uniform(0.2, 0.8)
        a = np.maximum(0, 1 - 3 * np.abs(x - peak))
    else:
        # Broader Gaussian
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.2, 0.8)
        a = np.exp(-5 * (x - center) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP game, subgradient Adam, coordinate descent, and peak shaving.
    """
    t_global = time.monotonic()
    total_budget = 1700.0  # Leave 100s buffer

    solver = find_solver()
    print(f"LP solver: {solver}, total budget: {total_budget:.0f}s")

    # Get initial sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n = 1600
        rng0 = np.random.default_rng(0)
        best_seq = structured_init(n, rng0)
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    rng = np.random.default_rng(42)

    # === Phase 1: LP Nesterov with constraint generation ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Subgradient Adam ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Subgrad Adam n={n_base} ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_out, _ = subgrad_adam(best_seq, t2, topk=8, lr_init=0.03, seed=77)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 3: Coordinate Descent ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Coordinate Descent n={n_base} ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_out, _ = coordinate_descent(best_seq, t3, n_cycles=200, seed=99)
        try_update(a_out, f"cd_n{n_base}")

    # === Phase 4: Peak Shaving ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Peak Shaving n={n_base} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = peak_shaving(best_seq, t4, seed=314)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 5: Multi-length exploration ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Multi-length exploration ===")
    if remaining > 100:
        for n_try in [n_base + 100, n_base + 200, n_base - 100]:
            if n_try < 100:
                continue
            if time.monotonic() - t_global > t_global + total_budget - 80:
                break
            t5_each = min(remaining * 0.25, 150.0)
            if t5_each < 30:
                continue
            init_long = interpolate_sequence(best_seq, n_try)
            print(f"  Trying n={n_try} (budget={t5_each:.0f}s)")
            a_out, _ = nesterov_lp_optimize(init_long, t5_each, solver)
            try_update(a_out, f"mlp_n{n_try}")
            # Quick subgrad polish
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5b = min(remaining * 0.15, 60.0)
                if t5b > 15:
                    a_out2, _ = subgrad_adam(interpolate_sequence(best_seq, n_try),
                                             t5b, topk=6, lr_init=0.02, seed=n_try % 1000)
                    try_update(a_out2, f"mls_n{n_try}")
            remaining = total_budget - (time.monotonic() - t_global)

    # === Phase 6: Multi-start with different inits ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Multi-start ({remaining:.0f}s) ===")
    if remaining > 60:
        n_starts = 3
        t6_each = remaining / (n_starts * 2)
        for si in range(n_starts):
            if time.monotonic() - t_global > t_global + total_budget - 50:
                break
            init_s = structured_init(n_base, rng)
            # Blend with best
            blend = 0.5 * init_s + 0.5 * best_seq
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            t6a = min(t6_each, 80.0)
            a_out, _ = subgrad_adam(blend, t6a, topk=10, lr_init=0.05, seed=100 + si)
            try_update(a_out, f"ms_subgrad_{si}")
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t6b = min(remaining * 0.3, 50.0)
                if t6b > 15:
                    a_out2, _ = coordinate_descent(best_seq, t6b, n_cycles=50, seed=200 + si)
                    try_update(a_out2, f"ms_cd_{si}")

    # === Phase 7: Final peak shaving polish ===
    remaining = total_budget - (time.monotonic() - t_global)
    print(f"=== Phase 7: Final polish ({remaining:.0f}s) ===")
    if remaining > 15:
        t7 = min(remaining * 0.4, 100.0)
        a_out, _ = peak_shaving(best_seq, t7, seed=2718)
        try_update(a_out, "final_peak")

        # One more CD pass
        remaining = total_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.4, 80.0)
            a_out, _ = coordinate_descent(best_seq, t7b, n_cycles=50, seed=5555)
            try_update(a_out, "final_cd")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
