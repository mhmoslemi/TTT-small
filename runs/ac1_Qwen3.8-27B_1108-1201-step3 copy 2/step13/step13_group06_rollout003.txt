import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * len(a) * c.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def grad_conv_at_lags(a, lags):
    """Compute average gradient of conv_k for given lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in lags:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= max(len(lags), 1)
    return grad


def subgradient_momentum(a0, time_limit, lr_init=0.02, momentum=0.9, active_frac=0.02):
    """Subgradient descent with momentum on max-conv objective."""
    n = len(a0)
    a = normalize(np.array(a0, dtype=np.float64))
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    lr = lr_init
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_limit - 2:
        c = fast_conv(a)
        max_c = c.max()
        threshold = max_c * (1.0 - active_frac)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 30:
            top_idx = np.argsort(c[active])[::-1][:15]
            active = active[top_idx]

        g = grad_conv_at_lags(a, active)
        g -= g.mean()

        v = momentum * v - lr * g
        a_new = a + v
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_conv(a_new).max()
        step += 1

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > 50:
            lr *= 0.8

                v = np.zeros(n)
                lr = lr_init

                a = simplex_project(a + noise * a)

        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit * 0.7:
                break
            # Anneal active_frac
            progress = elapsed / time_limit
            active_frac = max(0.005, 0.02 * (1 - progress))

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=0.1, topk=10, n_stages=30):
    """Top-k LSE smooth approximation with Adam optimizer."""
    n = len(a0)
    a = normalize(np.array(a0, dtype=np.float64))
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_of(a)

    temps = np.geomspace(T_start, T_end, n_stages)
    base_lr = 5e-3 * min(1.0, 200.0 / n)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 2:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        lr = base_lr * (0.9 ** si)
        m = np.zeros(n)
        v_ = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_iter = max(10, int(min(remaining, 40.0) * 4))

        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit - 2:
                break

            a_cur = np.exp(z - z.max())
            a_cur /= a_cur.sum()

            c = fast_conv(a_cur)
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            # LSE value
            max_tv = top_vals.max()
            exp_tv = np.exp(top_vals - max_tv)
            lse_val = (max_tv + np.log(exp_tv.sum())) / T
            w = exp_tv / exp_tv.sum()

            # Gradient
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wK = w[idx]
                js = max(0, K - n + 1)
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a_cur[K - idx_range]

            # Convert to z-gradient
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v_ = beta2 * v_ + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v_ / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a_check = np.exp(z - z.max())
                a_check /= a_check.sum()
                sc = score_of(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    if (it + 1) % 40 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m = np.zeros(n)
                        v_ = np.zeros(n)
                        t_adam = 0

        a_stage = np.exp(z - z.max())
        a_stage /= a_stage.sum()
        sc = score_of(a_stage)
        if sc < best_score:
            best_score = sc
            best_a = a_stage.copy()

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=50000):
    """
    Targeted coordinate descent: repeatedly find the best single-position
    move that reduces the convolution max.
    """
    n = len(a0)
    a = normalize(np.array(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 2:
            break

        c = fast_conv(a)
        max_c = c.max()
        # Find top lags within 1%
        threshold = max_c * 0.995
        top_lags = np.where(c >= threshold)[0]
        if len(top_lags) == 0:
            top_lags = np.array([np.argmax(c)])
        if len(top_lags) > 20:
            top_lags = top_lags[np.argsort(c[top_lags])[::-1][:10]]

        # Build danger map: positions that contribute to top lags
        danger = np.zeros(n, dtype=np.float64)
        for k in top_lags[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            danger[idx] += a[k - idx]

        # Find source: highest danger position
        src = int(np.argmax(danger * (a > 1e-10)))
        if a[src] < 1e-10:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = normalize(a)
            continue

        # Find safe destinations (low danger, low value)
        safe_scores = danger + 0.1 * a  # lower is safer
        safe_scores[src] = 1e10
        dest_candidates = np.argsort(safe_scores)[:max(10, n // 10)]

        found = False
        for dst in dest_candidates:
            if dst == src:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-12:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                new_max = fast_conv(a_trial).max()
                if new_max < max_c * (1 - 1e-9):
                    a = a_trial
                    new_score = 2.0 * n * new_max
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    stall = 0
                    found = True
                    break
            if found:
                break

        if not found:
            stall += 1
            # Try random move
            if stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize(a_trial)
                    sc = score_of(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                # Reset to best and perturb
                a = best_a.copy()
                noise = rng.normal(0, 0.01, n)
                a = simplex_project(a * (1 + noise))
                stall = 0

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_start=0.001, T_end=1e-7, seed=42):
    """Simulated annealing with targeted move generation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(np.array(a0, dtype=np.float64))
    cur_score = score_of(a)
    best_a = a.copy()
    best_score = cur_score
    t0 = time.monotonic()
    total_time = time_limit
    step = 0

    while time.monotonic() - t0 < time_limit - 2:
        progress = (time.monotonic() - t0) / total_time
        T = T_start * (T_end / T_start) ** progress
        step += 1

        c = fast_conv(a)
        max_c = c.max()
        top_lag = int(np.argmax(c))

        # Generate move: move mass from high-contribution to low-contribution
        js = max(0, top_lag - n + 1)
        je = min(n - 1, top_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[top_lag - idx]

        if contrib.max() < 1e-15:
            # Random move
            i = rng.integers(n)
            j = rng.integers(n)
        else:
            i = idx[np.argmax(contrib)]

        if rng.random() < 0.7:
            j = rng.integers(n)
        else:
            # Target low-contribution position
            j = idx[np.argmin(a[idx])]

        if i == j or a[i] < 1e-12:
            continue

        # Move amount: geometric distribution
        max_move = min(a[i], 0.05)
        move = rng.uniform(0.0001, max_move) if max_move > 1e-8 else 0.001
        direction = 1 if rng.random() < 0.6 else -1

        a_trial = a.copy()
        a_trial[i] -= direction * move
        a_trial[j] += direction * move
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-12:
            continue
        a_trial /= s

        new_score = 2.0 * n * fast_conv(a_trial).max()
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_trial.copy()

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate normalized sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp


def multi_start_optimize(a0, time_limit, n_starts=3, seed=99):
    """Try multiple perturbed starts and return the best."""
    rng = np.random.default_rng(seed)
    best_a = a0.copy()
    best_score = score_of(a0)
    t_per = time_limit / n_starts

    for i in range(n_starts):
        if time.monotonic() - (time.monotonic() - t_per * (i + 1)) > time_limit:
            break
        t_remaining = time_limit - (time.monotonic() - (time.monotonic() - t_per * (i + 1)))
        if t_remaining < 10:
            break

        scale = 0.02 * (i + 1)
        noise = rng.normal(0, scale, len(a0))
        a_pert = simplex_project(a0 * (1 + noise))

        # Quick LSE refinement
        a_out, sc = lse_gradient_descent(a_pert, t_per * 0.6, T_start=15.0, T_end=0.5, topk=8, n_stages=10)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

        # Quick coord descent
        a_out2, sc2 = coordinate_descent(a_out, t_per * 0.4)
        if sc2 < best_score:
            best_score = sc2
            best_a = a_out2.copy()

    return best_a, best_score


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. Subgradient + momentum (fast broad descent)
    2. LSE gradient with annealing (smooth refinement)
    3. Coordinate descent (targeted local search)
    4. Simulated annealing (escape local minima)
    5. Multi-length exploration
    6. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        # Generate initial sequence
        n_init = 1600
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 2 + 0.3 * np.exp(-20 * (x - 0.3) ** 2)
        best_seq = normalize(best_seq)
        best_score = score_of(best_seq)
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)
    print(f"Starting optimization: n={n_base}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Subgradient + momentum (broad fast descent) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Subgradient momentum ({t1:.0f}s) ===")
        a_out, sc = subgradient_momentum(best_seq, t1, lr_init=0.03, momentum=0.85, active_frac=0.03)
        try_update(a_out, "subgrad_p1")

    # === Phase 2: LSE gradient (smooth refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        print(f"\n=== Phase 2: LSE gradient ({t2:.0f}s) ===")
        a_out, sc = lse_gradient_descent(best_seq, t2, T_start=30.0, T_end=0.1, topk=12, n_stages=35)
        try_update(a_out, "lse_p2")

    # === Phase 3: Coordinate descent (targeted) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Coordinate descent ({t3:.0f}s) ===")
        a_out, sc = coordinate_descent(best_seq, t3)
        try_update(a_out, "coord_p3")

    # === Phase 4: Simulated annealing (escape local minima) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Simulated annealing ({t4:.0f}s) ===")
        a_out, sc = simulated_annealing(best_seq, t4, T_start=0.002, T_end=1e-7, seed=777)
        try_update(a_out, "anneal_p4")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration ===")
    if remaining > 60:
        for delta_n in [50, -50, 100]:
            if time.monotonic() - t_global > t_global + time_budget - 60:
                break
            n_try = n_base + delta_n
            if n_try <= 100:
                continue
            t5_each = min(remaining * 0.15, 100.0)
            if t5_each < 15:
                continue
            print(f"  Trying n={n_try} ({t5_each:.0f}s)")
            a_interp = interpolate_to_length(best_seq, n_try)
            # Quick subgradient
            a_out, sc = subgradient_momentum(a_interp, t5_each * 0.5, lr_init=0.02, momentum=0.85)
            try_update(a_out, f"subgrad_n{n_try}")
            # Quick coord descent
            t5b = t5_each * 0.5 - (time.monotonic() - t_global)
            if t5b > 5:
                remaining2 = time_budget - (time.monotonic() - t_global)
                if remaining2 > 20:
                    a_out2, sc2 = coordinate_descent(a_out, min(t5b, remaining2 * 0.3))
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 6: Multi-start refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 6: Multi-start ({min(remaining*0.15, 150):.0f}s) ===")
        t6 = min(remaining * 0.15, 150.0)
        a_out, sc = multi_start_optimize(best_seq, t6, n_starts=3, seed=555)
        try_update(a_out, "multi_p6")

    # === Phase 7: Final subgradient polish (very low lr) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Final subgradient polish ===")
        t7 = min(remaining * 0.25, 100.0)
        a_out, sc = subgradient_momentum(best_seq, t7, lr_init=0.005, momentum=0.9, active_frac=0.01)
        try_update(a_out, "subgrad_polish")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Final coordinate descent ===")
        t8 = min(remaining * 0.70, 80.0)
        a_out, sc = coordinate_descent(best_seq, t8)
        try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
