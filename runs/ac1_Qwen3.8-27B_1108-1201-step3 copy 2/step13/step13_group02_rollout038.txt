def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_solve(a, M, solver, active_rows=None):
    """Solve the LP: max sum(b) s.t. T(a)@b <= M, b >= 0.
    If active_rows is given, only use those constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    if active_rows is not None and len(active_rows) < 2 * n - 1:
        constraints = [T[active_rows, :] @ b_var <= M[active_rows]]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def lse_gradient_descent(a0, time_limit, verbose=True):

        T = T_start * (T_end / T_start) ** progress

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Subgradient descent with adaptive step
    step_scale = 0.1
    for _ in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = np.argmax(b)
        
        # Subgradient: grad_i = a[k_star - i] + a[i - k_star] (with 0 padding)
        grad = np.zeros(n)

            j1 = k_star - i
            j2 = i - k_star
            if 0 <= j1 < n:
                grad[i] += a[j1]

                grad[i] += a[j2]

        a = a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization with LP Nesterov
2. LP Nesterov with line search and cutting-plane constraints
3. Subgradient-based polish for exact max reduction
4. LSE smooth gradient descent
5. Multi-destination peak shaving with simulated annealing
6. Adaptive length exploration (1600, 1800, 2000)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver, active_rows=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    if active_rows is not None and 0 < len(active_rows) < 2 * n - 1:
        constraints = [T[active_rows, :] @ b_var <= M[active_rows]]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    patience = 250

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M_curr = b.max()
        if M_curr < 1e-12:
            break

        # Adaptive constraint set: use top 80% of constraints after initial phase
        if t < 20:
            active_rows = None
        else:
            threshold = M_curr * 0.85
            active_idx = np.where(b >= threshold)[0]
            if len(active_idx) < 15:
                active_rows = None
            else:
                active_rows = active_idx

        bl = lp_solve(a, M_curr, solver, active_rows)
        if bl is None:
            continue

        bl_sum = bl.sum()
        if bl_sum < 1e-15:
            continue
        g = bl * norm_val / bl_sum
        n_iters += 1

        # Line search: try several alpha values to find best
        alphas = [2.0 / (t + 2.0), 2.0 / (t + 2.0) * 1.5, 2.0 / (t + 2.0) * 0.5, 0.1, 0.15, 0.2, 0.3]
        best_alpha = 2.0 / (t + 2.0)
        best_obj = float('inf')
        for alpha in alphas:
            alpha = min(max(alpha, 0.005), 0.95)
            trial = (1.0 - alpha) * a + alpha * g
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial * norm_val / s_t
            obj = fast_autocorr(trial).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = np.ones(n) * norm_val / n
        else:
            a = a * norm_val / s_a

        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and n_iters % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        # Adaptive restart
        if stall > patience:
            rng = np.random.default_rng(n_iters * 13 + 7)
            scale = 0.02 * a.mean()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm_val / s_a
            stall = 0
            patience = max(100, patience - 20)
            if verbose:
                print(f"    [Restart at iter {n_iters}, score={best_score:.8f}]")

    return best_a, best_score


def compute_subgradient(a, k_star):
    n = len(a)
    grad = np.zeros(n)
    # First term: a[k_star - i] for valid i
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    if i_start <= i_end:
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += a[k_star - indices]
    # Second term: a[i - k_star] for valid i
    i_start2 = max(0, k_star)
    i_end2 = n - 1
    if i_start2 <= i_end2:
        indices2 = np.arange(i_start2, i_end2 + 1)
        grad[indices2] += a[indices2 - k_star]
    return grad


def subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n

        a = a * norm_val / s0

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_scale = 0.05

    for _ in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        grad = compute_subgradient(a, k_star)

        # Try a few step sizes
        best_trial = None
        best_trial_score = float('inf')
        for frac in [0.5, 1.0, 2.0, 4.0]:
            step = step_scale * frac / np.sqrt(steps + 1)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:

            score = fast_autocorr(a_new).max()
            if score < best_trial_score:

            best_score = best_trial_score
            best_a = a.copy()

            if best_trial is not None:
                a = best_trial

        steps += 1
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.3, best_score * 0.04)
    T_end = max(0.0005, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Backtracking line search
        step = 1.0 / (T * n * 0.3)
        a_new = a
        for _ in range(8):
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s_new = a_candidate.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_candidate = a_candidate * norm_val / s_new
            new_score = fast_autocorr(a_candidate).max()
            if new_score < max_b:
                a_new = a_candidate
                break
            step *= 0.5

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def make_initialization(n, kind, rng):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'hill':
        seq = np.sin(np.pi * x) ** 1.5
    elif kind == 'hill_sq':
        seq = np.sin(np.pi * x) ** 2.0
    elif kind == 'broad':
        seq = np.sin(np.pi * x) ** 0.8
    elif kind == 'valley':
        seq = 0.1 + 0.9 * (2 * x - 1) ** 2
    elif kind == 'biphasic':
        seq = 0.3 + 0.7 * np.sin(2 * np.pi * x) ** 2
    elif kind == 'random_smooth':
        base = rng.random(n)
        # Smooth it
        kernel = np.ones(21) / 21
        seq = np.convolve(base, kernel, mode='same')
        seq = np.maximum(seq, 0.01)
    else:
        seq = np.sin(np.pi * x) ** 1.5
    seq = np.maximum(seq, 1e-8)
    return seq / seq.sum()

    # Phase 0: Multi-start if parent score is poor or no parent
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is None or best_score > 1.51:
        print(f"\n=== Phase 0: Multi-start (remaining={remaining:.0f}s) ===")
        n_start = 1600
        inits = ['hill', 'hill_sq', 'broad', 'random_smooth', 'biphasic']
        t_per = min(remaining * 0.04, 40.0)
        for kind in inits:
            if time.monotonic() - t_global > time_budget - 300:
                break
            a_init = make_initialization(n_start, kind, rng)
            a_norm = a_init * np.sqrt(2.0 * n_start)
            if t_per > 20:
                a_out, _ = nesterov_lp_optimize(a_norm, t_per, solver, verbose=False)
            else:
                a_out = a_init
            sc = evaluate_safe(a_out)
            label = f"init_{kind}"
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out / a_out.sum()
                print(f"  Multi-start: {label} -> {sc:.6f}")

    if best_seq is None:
        n_base = 1600
        best_seq = make_initialization(n_base, 'hill', rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: score={best_score:.6f}")

    def try_update(a_arr, label):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_norm = a_arr / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.32, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Subgradient polish n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.08, 120.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur)
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        try_update(a_out, f"sg_n{n_cur}")

    # Phase 3: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LSE-GD n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 1
20.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur)
        a_out, _ = lse_gradient_descent(a_norm, t3, verbose=True)
        try_update(a_out, f"lse_n{n_cur}")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur)
        a_out, _ = peak_shave(a_norm, t4, seed=123, verbose=True)
        try_update(a_out, f"ps_n{n_cur}")

    # Phase 5: Expand to n=1800
    n2 = max(n_base + 200, 1800)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.22, 350.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2)
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

        # Quick polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        n_cur = len(best_seq)
        t5b = min(remaining * 0.05, 60.0)
        if t5b > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_cur)
            a_out, _ = subgradient_polish(a_norm, t5b, verbose=False)
            try_update(a_out, f"sg_n{n_cur}_p5")

    # Phase 6: Expand to n=2000
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3)
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        try_update(a_out, f"lp_n{n3}")

    # Phase 7: Final peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best)
        a_out, _ = peak_shave(a_norm, t7, seed=777, verbose=True)
        try_update(a_out, f"ps_final_n{n_best}")

    # Phase 8: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best)
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
