"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov with constraint reduction for initial optimization
2. Fast Nesterov method with top-k weighted autocorrelation gradients
3. LP-guided peak shaving with larger steps and simulated annealing
4. LSE gradient descent with backtracking line search
5. Multi-length exploration and final polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix_rows(a, active_indices):
    """Build only the rows of the convolution matrix corresponding to active indices."""
    n = len(a)
    rows = []
    for k in active_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def lp_best_response_fast(a, M, solver, max_iters_reduce=3):
    """LP best response with constraint reduction for speed."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build full constraint matrix
    T = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            T[k, j] = a[k - j]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Reduce to active constraints
    for _ in range(max_iters_reduce):
        circ = T @ b_val
        tol = 1e-8 * max(M, 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 10 or n_active < 20:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        if t < 30:
            b = lp_best_response_fast(y, M_curr, solver, max_iters_reduce=3)
        else:
            b = lp_best_response_fast(y, M_curr, solver, max_iters_reduce=2)
        
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
        
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            pert = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            stall = 0
    
    return best_a, best_score


def top_k_gradient(a, k_values, weights):
    """Compute gradient of sum_k weights[k] * b_k where b = conv(a,a)."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(k_values, weights):
        if abs(w) < 1e-15:
            continue
        # grad_i += w * (a[k-i] + a[i-k])
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1+1] += w * a[k - idx]
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[lo2:hi2+1] += w * a[idx - k]
    return grad


def nesterov_topk(a0, time_limit, verbose=True):
    """Nesterov accelerated gradient descent using top-k peak weighting."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    steps = 0
    stall = 0
    
    T_start = max(0.1, best_score * 0.3)
    T_end = max(0.01, best_score * 0.01)
    
    # Adaptive step size
    step = 1.0 / (n * 0.1)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(y)
        max_b = b.max()
        
        # Find top peaks
        n_top = min(8, len(b))
        top_k = np.argsort(b)[::-1][:n_top]
        
        # Softmax weights
        bk = b[top_k]
        w = np.exp((bk - bk[0]) / T)
        w = w / w.sum()
        
        # Compute gradient
        grad = top_k_gradient(y, top_k, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        y_try = y.copy()
        step_trial = step
        f_curr = max_b
        for ls in range(20):
            y_try = y - step_trial * grad / norm
            y_try = np.maximum(y_try, 0.0)
            s_try = y_try.sum()
            if s_try < 1e-15:
                break
            y_try = y_try * norm / s_try
            f_try = fast_autocorr(y_try).max()
            if f_try <= f_curr - 1e-4 * step_trial * grad_norm:
                break
            step_trial *= 0.5
        else:
            step_trial = 0
        
        if step_trial < 1e-10:
            # Can't improve, try smaller
            step *= 0.9
            continue
        
        step = min(step * 1.05, 1.0 / (n * 0.01))
        
        y_new = y_try
        steps += 1
        
        # Nesterov momentum
        alpha = 2.0 / (steps + 2.0)
        y_ns = (1.0 - alpha) * y + alpha * y_new
        y_ns = np.maximum(y_ns, 0.0)
        s_ns = y_ns.sum()
        if s_ns > 1e-15:
            y_ns = y_ns * norm / s_ns
        else:
            y_ns = y_new.copy()
        
        if steps <= 1:
            a_new = y_new.copy()
        else:
            momentum = (steps - 1.0) / (steps + 2.0)
            a_new = y_ns + momentum * (y_ns - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_ns.copy()
        
        y = y_ns
        a = a_new
        
        score = max(fast_autocorr(y).max(), fast_autocorr(a).max())
        if score < best_score:
            best_score = score
            best_a = (y if fast_autocorr(y).max() <= fast_autocorr(a).max() else a).copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and steps % 500 == 0:
            print(f"    TopK-Nest step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best={best_score:.8f}")
        
        if stall > 500:
            rng = np.random.default_rng(steps * 3 + 1)
            pert = 0.01 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            stall = 0
            step = 1.0 / (n * 0.1)
    
    return best_a, best_score


def lp_guided_peak_shave(a0, time_limit, solver, seed=42, verbose=True):
    """Peak shaving guided by LP best response."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    T_start = max(0.3, best_score * 0.05)
    T_end = best_score * 0.001
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min((time.monotonic() - t0) / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find the worst peak and its contributors
        k_worst = int(np.argmax(b))
        js = max(0, k_worst - n + 1)
        je = min(n - 1, k_worst)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_worst - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 10
            continue
        
        src = idx[np.argmax(contrib)]
        
        # Use LP to find best response direction (if time allows)
        use_lp = (time.monotonic() - t0) < time_limit * 0.5
        if use_lp:
            b_resp = lp_best_response_fast(a, max_b, solver, max_iters_reduce=1)
            if b_resp is not None:
                b_sum = b_resp.sum()
                g = b_resp * norm / b_sum
                # Take a step toward g
                for t in [0.3, 0.5, 0.7, 0.1, 0.2]:
                    a_trial = (1 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                        break
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial
                        n_stall = 0
                        break
                else:
                    n_stall += 1
            else:
                n_stall += 1
        else:
            # Direct peak shaving without LP
            # Move mass from src to a position that reduces the peak
            peak_mask = np.zeros(n, dtype=bool)
            top_peaks = np.argsort(b)[::-1][:5]
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]
            
            # Choose destination with lowest value
            dst = non_peak[np.argmin(a[non_peak])]
            
            for frac in [0.1, 0.05, 0.2, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    n_stall = 0
                    break
            else:
                n_stall += 1
        
        if n_stall > 100:
            # Perturbation
            idx_pert = rng.choice(n, size=max(1, n // 100), replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.05, 0.2)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score * 1.005:
                    a = a_trial
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
            n_stall = 0
        
        if verbose and n_improve > 0 and n_improve % 1000 == 0:
            print(f"    PeakShave improve={n_improve} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

def lse_backtracking(a0, time_limit, verbose=True):
    """LSE gradient descent with backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.005, best_score * 0.002)
    step = 1.0 / (n * 0.2)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient using FFT
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        f_curr = max_b
        step_trial = step
        accepted = False
        for ls in range(30):
            a_try = a - step_trial * grad / norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try * norm / s_try
            f_try = fast_autocorr(a_try).max()
            if f_try <= f_curr - 1e-5 * step_trial * grad_norm / norm:
                a = a_try
                accepted = True
                break
            step_trial *= 0.5
        
        if accepted:
            step = min(step * 1.1, 5.0 / n)
        else:
            step *= 0.5
        
        steps += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 100 == 0:
            print(f"    LSE-BT step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def lp_best_response_fast(a, M, solver, n_active_target=50):

    size = 2 * n - 1

    T = np.zeros((size, n))

        lo = max(0, k - n + 1)

        T[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    prob.solve(solver=solver, verbose=False)

        return None

        tol = 1e-7 * M
        active = circ >= M - tol
        n_act = int(active.sum())
        if n_act < 10 or n_act > size - 20:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov with constraint reduction for solid initial optimization
2. Fast Nesterov with top-k weighted autocorrelation gradients + backtracking line search
3. LP-guided peak shaving with simulated annealing and larger steps
4. LSE gradient descent with FFT-based gradients and backtracking
5. Multi-length exploration and final LP polish
Key improvements over previous: backtracking line search, corrected gradient formula,
more aggressive step sizes, LP-guided moves in peak shaving, better time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            T[k, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]
    return T


def lp_best_response(a, M, solver, n_reduce=3):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(n_reduce):
        circ = T @ b_val
        tol = 1e-7 * max(M, 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active < 15 or n_active > (2 * n - 1) - 10:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if (T @ b_new).max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break

    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        n_red = 3 if t < 50 else 2
        b = lp_best_response(y, M_curr, solver, n_reduce=n_red)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 200 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            pert = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            stall = 0

    return best_a, best_score


def topk_gradient_direct(a, k_values, weights):

def nesterov_topk(a0, time_limit, verbose=True):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    steps = 0
    stall = 0
    step = 2.0 / n

    T_start = max(0.05, best_score * 0.2)
    T_end = max(0.005, best_score * 0.005)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(y)
        max_b = b.max()

        n_top = min(10, len(b))
        top_k = np.argsort(b)[::-1][:n_top]
        bk = b[top_k]
        w = np.exp((bk - bk[0]) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = topk_gradient_direct(y, top_k, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        f_curr = max_b
        step_trial = step
        accepted = False
        y_accepted = None
        for ls in range(25):
            y_try = y - step_trial * grad / norm
            y_try = np.maximum(y_try, 0.0)
            s_try = y_try.sum()
            if s_try < 1e-15:
                break
            y_try = y_try * norm / s_try
            f_try = fast_autocorr(y_try).max()
            if f_try <= f_curr - 1e-6 * step_trial * grad_norm / norm:
                y_accepted = y_try
                accepted = True
                break
            step_trial *= 0.5

        if not accepted:
            step *= 0.8
            if step < 1e-10:
                break
            continue

        step = min(step * 1.08, 10.0 / n)
        y_new = y_accepted
        steps += 1

        # Nesterov momentum
        alpha = 2.0 / (steps + 2.0)
        y_ns = (1.0 - alpha) * y + alpha * y_new
        y_ns = np.maximum(y_ns, 0.0)
        s_ns = y_ns.sum()
        if s_ns > 1e-15:
            y_ns = y_ns * norm / s_ns
        else:
            y_ns = y_new.copy()

        if steps <= 1:
            a_new = y_new.copy()
        else:
            momentum = (steps - 1.0) / (steps + 2.0)
            a_new = y_ns + momentum * (y_ns - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_ns.copy()

        y = y_ns
        a = a_new

        score_y = fast_autocorr(y).max()
        score_a = fast_autocorr(a).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        elif score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and steps % 500 == 0:
            print(f"    TopK-Nest step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best={best_score:.8f}")

        if stall > 600:
            rng = np.random.default_rng(steps * 3 + 1)
            pert = 0.01 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            stall = 0
            step = 2.0 / n

    return best_a, best_score

                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                        improved = True
                        break
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial
                        n_stall = 0
                        improved = True
                        break
        else:
            # Direct peak shaving
            peak_mask = np.zeros(n, dtype=bool)
            top_peaks = np.argsort(b)[::-1][:6]
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != idx[np.argmax(contrib)])[0]

            src = idx[np.argmax(contrib)]
            dst_candidates = non_peak[np.argsort(a[non_peak])][:20]

            for dst in dst_candidates[:5]:
                for frac in [0.1, 0.05, 0.15, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                        improved = True
                        break
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial
                        n_stall = 0
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            n_stall += 1

        if n_stall > 80:
            idx_pert = rng.choice(n, size=max(2, n // 80), replace=False)
            a_trial = a.copy()
            for idx2 in idx_pert:
                amt = a[idx2] * rng.uniform(0.05, 0.2)
                j = rng.integers(n)
                a_trial[idx2] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score * 1.003:
                    a = a_trial
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 1000 == 0:
            print(f"    PeakShave improve={n_improve} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_backtracking(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step = 2.0 / n

    T_start = max(0.3, best_score * 0.08)
    T_end = max(0.002, best_score * 0.001)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = lse_gradient_fft(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking
        f_curr = max_b
        step_trial = step
        accepted = False
        for ls in range(30):
            a_try = a - step_trial * grad / norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                break
            a_try = a_try * norm / s_try
            f_try = fast_autocorr(a_try).max()
            if f_try <= f_curr - 1e-6 * step_trial * grad_norm / norm:
                a = a_try
                accepted = True
                break
            step_trial *= 0.5

        if accepted:
            step = min(step * 1.06, 10.0 / n)
        else:
            step *= 0.7

        steps += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            print(f"    LSE-BT step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def normalize_for_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return None
    return a / s
