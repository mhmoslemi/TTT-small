from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Gradient of T*log(sum exp(b_k/T)) where b = conv(a,a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1].copy()
    A_rev = rfft(a_rev, fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    
    return r1 + r2


def max_subgradient(a, n_top=1):
    """
    Compute (average) subgradient of max_k (a*a)_k.
    Returns the average gradient over the top n_top peaks.
    """
    n = len(a)
    b = fast_autocorr(a)
    M = b.max()
    
    # Find positions near the max
    threshold = M - 1e-10  # essentially exact max positions
    top_positions = np.where(b >= threshold)[0]
    
    if len(top_positions) == 0:
        top_positions = np.array([np.argmax(b)])
    
    # For robustness, average over top n_top positions
    if len(top_positions) > n_top:
        # Sort by value descending, take top n_top
        order = np.argsort(b[top_positions])[::-1]
        top_positions = top_positions[order[:n_top]]
    
    # Compute average gradient
    # grad of (a*a)_k w.r.t. a_m = a_{k-m} + a_{m-k} (for valid indices)
    # For a position k, the gradient vector g has:
    #   g[m] = a[k-m] (if 0 <= k-m < n) + a[m-k] (if 0 <= m-k < n)
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A_fwd = rfft(a, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)
    
    grad = np.zeros(n)
    for k in top_positions:
        # g[m] = a[k-m] for valid m (0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(k, n-1))
        # g[m] += a[m-k] for valid m (0 <= m-k < n, i.e., k <= m <= k+n-1, intersected with [0,n-1])
        
        # First term: a[k-m] for m in [max(0, k-n+1), min(k, n-1)]
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += a[k - idx]
        
        # Second term: a[m-k] for m in [max(k, 0), min(k+n-1, n-1)]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += a[idx - k]
    
    grad /= len(top_positions)
    return grad, M

    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            if it == 0:
                y_new = y - (1.0 / 3.0) * lse_gradient(y, T)
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e > 1e-15:
                    y_extrap = y_extrap * norm / s_e
                y_new = y_extrap - (1.0 / (it + 3.0)) * lse_gradient(y_extrap, T)
            
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:

            else:

def nesterov_max_grad_optimize(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated gradient descent using the direct max subgradient.
    The subgradient of max_k (a*a)_k is computed at the top peak positions.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    # Use a fixed number of iterations with decreasing step size
    max_iters = 5000
    it = 0
    
    while it < max_iters and (time.monotonic() - t0) < time_limit:
        # Nesterov extrapolation
        if it == 0:
            x = y.copy()
        else:
            momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 1.0))
            x = y + momentum * (y - y_prev)
            x = np.maximum(x, 0.0)
            s_x = x.sum()
            if s_x > 1e-15:
                x = x * norm / s_x
        
        # Compute subgradient at x
        grad, M_x = max_subgradient(x, n_top=3)
        
        # Step size: 2/(L*t) where L is Lipschitz constant of gradient
        # L ~ 2 for the individual quadratic forms, use L=2
        step = 2.0 / (2.0 * (it + 1.0))  # = 1/(it+1)
        
        y_prev = y.copy()
        y = x - step * grad
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = y_prev.copy()
        else:
            y = y * norm / s_y
        
        it += 1
        
        if it % 20 == 0:
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MaxGrad it={it} ({elapsed:.0f}s): best={best_score:.10f}, M={M_x:.6f}")
    
    # Final check
    score = fast_autocorr(y).max()
    if score < best_score:
        best_score = score
        best_a = y.copy()
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    """LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = np.zeros(n)
    
    for t in range(1, 200):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        M_curr = b.max()
        if M_curr < 1e-12:
            break
        
        b_opt = lp_best_response(y, M_curr, solver)
        if b_opt is None:
            continue
        
        b_sum = b_opt.sum()
        if b_sum < 1e-15:
            continue
        g = b_opt * norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y.copy()
        y = y_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search with multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M = b.max()
        # Find top peaks
        top_ks = np.argsort(b)[::-1][:8]
        
        improved = False
        for k_star in top_ks:
            if b[k_star] < M - 1e-10:
                break
            
            # Decompose the peak: (a*a)_{k_star} = sum_i a_i * a_{k_star - i}
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            total = contrib.sum()
            
            if total < 1e-15:
                continue
            
            # Try reducing the top contributing pairs
            top_contrib_order = np.argsort(contrib)[::-1][:5]
            
            for ci in top_contrib_order:
                src = idx[ci]
                if a[src] < norm * 1e-10:
                    continue
                
                # Find destination: positions that minimize increase in other peaks
                # Heuristic: positions with low a value that are far from other peaks
                peak_mask = np.zeros(2 * n - 1, dtype=bool)
                for k2 in top_ks:
                    if b[k2] > M * 0.9:  # only strong peaks
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        # This position contributes to peaks at k2 - m + something... 
                        # Simpler: just mask positions that are in the support of top peaks
                        peak_mask[js2:je2 + 1] = True
                
                candidates = np.where(~peak_mask[:n])[0]
                if len(candidates) < 10:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Pick low-value candidates
                candidates = candidates[np.argsort(a[candidates])[:20]]
                
                for dst in candidates[:5]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if rng.integers(5) == 0:
                n_mods = rng.integers(1, 4)
                i = rng.integers(n)
                if a[i] > norm * 1e-8:
                    for _ in range(n_mods):
                        j = rng.integers(n)
                        if i != j:
                            frac = rng.uniform(0.005, 0.05)
                            amt = a[i] * frac
                            a_trial = a.copy()
                            a_trial[i] -= amt
                            a_trial[j] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t > 1e-15:
                                a_trial = a_trial * norm / s_t
                                score = fast_autocorr(a_trial).max()
                                if score < best_score:
                                    a = a_trial
                                    best_score = score
                                    stall = 0
                                    break
            
            if stall > 100:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.10f} ({time.monotonic()-t0:.0f}s)")
                break
        
        if verbose and it_count % 200 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Sum of Gaussians
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(3, 7)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.2)
            amp = rng.uniform(0.3, 1.0)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif choice == 6:
        # Polynomial
        x = np.linspace(0.0, 1.0, n)
        a = 1.0
        for _ in range(rng.integers(2, 5)):
            a *= (x - rng.uniform(0.1, 0.9)) ** 2 + 0.1
    else:
        # Cosine window with random parameters
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.5 * np.cos(np.pi * (x - rng.uniform(0, 1)) * rng.uniform(1, 3))
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
