def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if v.sum() == S:
        return np.maximum(v, 0)
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

        return v

            best_score = score_y

            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL it={n_iters} ({elapsed:.0f}s): best={best_score:.8f} stall={stall}")

        if stall > 400:
            rng = np.random.default_rng(n_iters * 17 + 3)
            ps = 0.015 * a.mean()
            y = np.maximum(y + rng.normal(0, ps, n), 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y * norm / sy
            a = np.maximum(a + rng.normal(0, ps, n), 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a * norm / sa
            stall = 0

    return best_a, best_score


def compute_grad_fft(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    rng = np.random.default_rng(99999)

    step_scale = 1.0 / (n * 0.3)
    patience = 0
    best_patience = best_score

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        threshold_frac = 0.998
        threshold = max_b * threshold_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        w = np.zeros(2 * n - 1)
        w[active] = 1.0
        w /= w.sum()

        grad = compute_grad_fft(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = step_scale
        improved = False
        for _ in range(8):
            a_new = a - step * grad
            a_new = project_simplex(a_new, norm)
            score = fast_autocorr(a_new).max()
            if score < max_b - 1e-7:
                a = a_new
                improved = True
                break
            step *= 0.5

        steps += 1

        if improved:
            step_scale = min(step_scale * 1.05, 5.0 / n)
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:
            step_scale = max(step_scale * 0.7, 0.01 / n)
            patience += 1

        if patience > 150:
            ps = rng.normal(0, 0.003 * a.mean(), n)
            a = project_simplex(a + ps, norm)
            patience = 0
            step_scale = 1.0 / (n * 0.3)

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD st={steps} ({elapsed:.0f}s): best={best_score:.8f} pat={patience}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    T = max(0.01, best_score * 0.01)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = 0.01 * (0.001) ** progress

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]

            for frac in [0.03, 0.08, 0.12, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                n_dest = min(5, len(non_peak))
                candidates = non_peak[np.argsort(a[non_peak])][:max(n_dest * 3, 12)]
                selected = candidates[rng.choice(len(candidates), min(n_dest, len(candidates)), replace=False)]

                a_trial = a.copy()
                a_trial[src] -= amt
                dst_vals = a[selected] + 1e-12
                weights = 1.0 / dst_vals
                weights /= weights.sum()
                for di, dst in enumerate(selected):
                    a_trial[dst] += amt * weights[di]

                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif score - best_score < T and rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            if n_stall > 30:
                n_pert = max(1, n // 80)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.05, 0.4)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score * 1.01:
                        a = a_trial
                n_stall = 0

        if verbose and n_improve % 300 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS imp={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length (most impactful)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: SubGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sgd_n{n_cur}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t3, seed=555, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # Phase 4: Try longer length
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.06, 60.0)
        if t4b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t4b, verbose=False)
            do_try_update(a_out, f"sgd_n{n_cur}")

    # Phase 5: Try even longer
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 6: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 250.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_fin_n{n_best}")

    # Phase 7: Final subgradient + peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final SGD+PS n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.35, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t7, verbose=True)
        do_try_update(a_out, f"sgd_fin_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.40, 100.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t8, seed=777, verbose=True)
        do_try_update(a_out, f"ps_fin_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
