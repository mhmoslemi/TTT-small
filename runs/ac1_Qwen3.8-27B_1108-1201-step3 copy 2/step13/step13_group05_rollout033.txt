    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    best_val = fast_autocorr(a).max()
    
    y = a.copy()  # Nesterov iterate
    x_prev = a.copy()
    
    t0 = time.monotonic()
    t = 0
    last_restart = 0
    prev_val = best_val
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute b and gradient at y
        b = fast_autocorr(y)
        M = b.max()
        
        # Softmax weights
        b_shifted = (b - M) / tau
        exp_b = np.exp(b_shifted)
        p = exp_b / exp_b.sum()
        
        # Gradient
        grad = compute_grad(y, p)  # as above
        
        # Step size
        L = 2.0  # smoothness (spectral norm of 2*sum p_k S_k <= 2)
        step = 1.0 / L

        x_new = y - step * grad
        x_new = proj_simplex(x_new, 1.0)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        M_new = b_new.max()
        if M_new < best_val:
            best_val = M_new
            best_a = x_new.copy()

        if t == 1:
            beta = 0.0
        else:
            beta = (t - 1) / (t + 2.0)
        
        y_new = x_new + beta * (x_new - x_prev)
        
        # Restart: if the function value at y_new is worse, restart
        # (simple restart rule)
        if t - last_restart > 50 and M_new > prev_val * 1.01:
            beta = 0.0
            last_restart = t
        
        y = y_new
        x_prev = x_new
        prev_val = M_new
        
        if verbose and t % 500 == 0:

            score = 2.0 * n * best_val
            print(f"  FISTA t={t} ({elapsed:.1f}s): score={score:.6f}")

    R = irfft(A * np.conj(P), fft_size)
    c = np.zeros(n)
    c[0] = R[0]
    if n > 1:
        c[1:] = R[fft_size - 1 : fft_size - n : -1]
    return 2.0 * c


def mirror_descent(a0, time_limit, top_k=5, beta=3.0, verbose=False):
    """Mirror descent with exponential weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 1.0 / (np.sqrt(t) + 1.0)
        ac = fast_autocorr(a)
        
        # Soft subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            i_start = max(0, int(k) - n + 1)
            i_end = min(n, int(k) + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j = int(k) - idx
            grad[idx] += 2.0 * wi * a[j]
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
