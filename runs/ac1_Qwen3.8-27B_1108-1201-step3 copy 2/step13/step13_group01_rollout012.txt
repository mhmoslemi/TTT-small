def build_T_sparse(a):
    """Build sparse convolution matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        for k in range(start, end):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response(a, M, solver, T_cache=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    if T_cache is not None:
        T = T_cache
    else:
        T = build_T_sparse(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_adaptive=True,
                         alpha_candidates=None, max_iters=500000):
    """LP Nesterov with optional adaptive step size via mini line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    if alpha_candidates is None:
        alpha_candidates = [0.1, 0.3, 0.5, 0.7, 1.0]

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        base_alpha = 2.0 / (t + 2.0)

        if use_adaptive and n_iters > 20:
            # Try multiple step sizes, pick best
            best_alpha = base_alpha
            best_trial_score = best_score
            for alpha_mult in [0.5, 1.0, 2.0, 4.0, 8.0]:
                alpha_try = min(base_alpha * alpha_mult, 1.0)
                y_trial = (1.0 - alpha_try) * y + alpha_try * g
                y_trial = np.maximum(y_trial, 0.0)
                s_tr = y_trial.sum()
                if s_tr < 1e-15:
                    continue
                y_trial = y_trial * norm / s_tr
                sc = fast_autocorr(y_trial).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_alpha = alpha_try
            alpha = best_alpha
        else:
            alpha = base_alpha

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        stall_check_interval = 50 if n_iters < 500 else 200
        if n_iters % stall_check_interval == 0:
            if time.monotonic() - last_improve > 60.0:
                stall_count += 1
                if stall_count > 5:
                    if verbose:
                        print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                    break
            else:
                stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def power_sum_objective(a_flat, n, p):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    # Power sum: (sum b_i^p)^(1/p) approximates max as p->inf
    ps = (np.power(b, p).sum()) ** (1.0 / p)
    obj = 2.0 * n * ps / (s * s)
    return obj


def power_sum_grad(a_flat, n, p):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)

    # Gradient of power sum: d/dx_i (sum x_j^p)^(1/p) = x_i^(p-1) / (sum x_j^p)^(1 - 1/p)
    b_p = np.power(b, p)
    sum_bp = b_p.sum()
    if sum_bp < 1e-30:
        return np.zeros(n)
    ps = sum_bp ** (1.0 / p)
    w = np.power(b, p - 1) / (sum_bp ** (1.0 - 1.0 / p))

    # Gradient of b = conv(a,a) w.r.t. a: db/da[i] = sum_k w[k] * db[k]/da[i]
    # db[k]/da[i] = a[k-i] + a[i-k] (if valid)
    # So grad[i] = sum_k w[k] * (a[k-i] + a[i-k])

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 2.0 * cross
    obj_val = 2.0 * n * ps / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def power_sum_scipy_optimize(a0, time_limit, p_start=50.0, p_end=5.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    for si, p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 180.0)

        x0 = a.copy()

        def obj_func(x):
            return power_sum_objective(x, n, p)

        def grad_func(x):
            return power_sum_grad(x, n, p)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                 jac=grad_func, options={'maxiter': 300,
                                 'maxfun': 800, 'time_limit': stage_time,
                                 'maxls': 50})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:6]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(3, len(candidates) // 4)]

                for dst in candidates_sorted[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)

    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        best_seq = None
        best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick scan at parent length to verify and potentially improve
    n_base = len(best_seq) if best_seq is not None else 1600
    print(f"\n=== Phase 0: Quick LP scan n={n_base} ===")
    t_phase0 = min(time_budget * 0.08, 120.0)
    if best_seq is not None:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t_phase0, solver, verbose=False,
                                             use_adaptive=True)
        do_try_update(a_out, f"scan_n{n_base}")

    # Phase 1: Deep LP Nesterov with adaptive step (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_base
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.50
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_adaptive=True,
                                    alpha_candidates=[0.5, 1.0, 2.0, 4.0, 8.0])
    do_try_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: Power-sum smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Power-sum n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = power_sum_scipy_optimize(a_norm, t2, p_start=80.0, p_end=10.0, n_stages=5)
        do_try_update(a_out, f"psum_n{n_best}")

    # Phase 3: LP Nesterov again from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_adaptive=True,
                                        alpha_candidates=[0.25, 0.5, 1.0, 2.0])
        do_try_update(a_out, f"lp_r2_n{n_best}")

    # Phase 4: Try a slightly different length
    n_alt = n_best + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP n={n_alt} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False, use_adaptive=True)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 5: Coordinate descent polish at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=99999)
        do_try_update(a_out, "coord_polish")

    # Phase 6: Final power-sum polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Power-sum polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = power_sum_scipy_optimize(a_norm, t6, p_start=20.0, p_end=3.0, n_stages=4)
        do_try_update(a_out, "psum_polish")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.6
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=12321)
        do_try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
