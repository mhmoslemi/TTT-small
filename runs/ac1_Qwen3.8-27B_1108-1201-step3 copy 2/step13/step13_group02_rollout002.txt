def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    momentum_arr = np.zeros(n)

    T_start = max(0.5, best_score * 0.08)
    T_end = max(0.01, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step with momentum
        base_step = norm / (grad_norm * 5.0)
        base_step = min(base_step, 0.5 / n)
        momentum_arr = 0.9 * momentum_arr + (1.0 - 0.9) * (base_step * grad / grad_norm)
        a_new = a - momentum_arr
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    old_x = old_x[:-1]
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            score = fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
            elif score < c[k] * 0.995:
                # Accept if it reduces this specific peak significantly
                a = a_trial
                improved = True
                break

        if not improved:
            # Random swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score <= best_score * 1.001:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()

        iters += 1
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TPR iter={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, full_iters=200)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Targeted peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Peak Reduction n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = targeted_peak_reduction(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"tpr_n{n_cur}")

    # === Phase 3: LP Nesterov at n=2000 ===
    n_try = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n_try} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, full_iters=150)
        do_try_update(a_out, f"lp_n{n_try}")

    # === Phase 4: LSE gradient descent polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE-GD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 130.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # === Phase 5: LP Nesterov at n=2400 ===
    n_try2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n_try2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long2 = interpolate_sequence(best_seq, n_try2)
        a_norm = init_long2 * np.sqrt(2.0 * n_try2) / init_long2.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, full_iters=100)
        do_try_update(a_out, f"lp_n{n_try2}")

    # === Phase 6: Final LP Nesterov polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, full_iters=100)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Peak Reduction n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_peak_reduction(a_norm, t7, solver, verbose=False)
        do_try_update(a_out, f"tpr_final_n{n_best}")

    # === Phase 8: Quick LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Quick LSE n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 60.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t8, verbose=False)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
