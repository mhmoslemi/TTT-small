    A = rfft(a_pad, fft_size)
    W = rfft(w_rev, fft_size)
    conv = irfft(A * W, fft_size)
    
    # Extract positions 2n-2, 2n-3, ..., n-1 (reversed to get i=0,1,...,n-1)
    grad = 2.0 * conv[n-1 : 2*n-1][::-1].copy()
    
    return grad

def lse_value(c, tau):
    """Compute (1/tau)*log(sum_k exp(tau*c_k)) = max(c) + (1/tau)*log(sum exp(tau*(c-max(c))))."""
    c_max = c.max()
    return c_max + (1.0 / tau) * np.log(np.sum(np.exp(tau * (c - c_max))) + 1e-300)

def topk_subgrad(a, c, top_k=8, beta=4.0):
    """Compute subgradient using weighted top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def nesterov_smoothed_optimize(a0, time_limit, tau_start=10.0, tau_end=500.0, verbose=True):
    """
    Nesterov-accelerated projected gradient on log-sum-exp smoothed objective.
    Uses adaptive temperature and backtracking line search.
    """
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    a = np.maximum(a, 1e-15)
    a = a * c_norm / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_prev = a.copy()
    
    # Initial step size estimate
    c_y = fast_autocorr(y)
    g_y = compute_lse_gradient(y, c_y, tau_start)
    L_est = 2.0 * tau_start * np.linalg.norm(a, ord=np.inf) * 2.0 + 1e-6
    eta = 1.0 / L_est
    eta_min = 1e-10
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature: increase over first 3000 iterations
        progress = min(1.0, t / 3000.0)
        tau = tau_start + (tau_end - tau_start) * progress * progress  # Quadratic ramp
        
        # Compute gradient at extrapolated point y
        c_y = fast_autocorr(y)
        g_y = compute_lse_gradient(y, c_y, tau)
        
        # Take step with backtracking
        f_y = lse_value(c_y, tau)
        step_accepted = False
        
        for bt in range(15):
            candidate = y - eta * g_y
            candidate = proj_simplex(candidate, c_norm)
            
            if eta < eta_min:
                break
            
            c_cand = fast_autocorr(candidate)
            f_cand = lse_value(c_cand, tau)
            
            # Sufficient decrease condition
            grad_dot = np.dot(g_y, y - candidate)
            if f_cand <= f_y - 0.4 * grad_dot + 1e-14:
                step_accepted = True
                break
            
            eta *= 0.5
        
        if not step_accepted or not step_accepted:
            # Use a very small step
            candidate = y - eta * g_y
            candidate = proj_simplex(candidate, c_norm)
        
        x_new = candidate
        
        # Track best
        M_x = fast_autocorr(x_new).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x_new.copy()
        
        # Nesterov extrapolation for next y
        if t == 1:
            y_new = x_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y_new = x_new + beta * (x_new - x_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = x_new.copy()
            else:
                y_new = y_new * c_norm / s_y
        
        # Also check y
        M_y = fast_autocorr(y_new).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y_new.copy()
        
        x_prev = x_new
        y = y_new
        
        # Occasionally increase eta if we've been shrinking too much
        if bt > 5 and eta < 1.0 / (L_est * 0.1):
            eta = 1.0 / L_est * 0.5
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    NS t={t} tau={tau:.1f} eta={eta:.2e} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def subgrad_nesterov_optimize(a0, time_limit, top_k=8, beta=4.0, verbose=True):
    """
    Nesterov-accelerated projected subgradient with top-k peak weighting.
    """
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_prev = a.copy()
    eta = c_norm / (2.0 * top_k * beta + 1)  # Initial step
    
    # Running average
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c_y = fast_autocorr(y)
        g_y = topk_subgrad(y, c_y, top_k=top_k, beta=beta)
        
        # Adaptive step size: eta_t = c / sqrt(t)
        eta_t = eta / (np.sqrt(t) + 1.0)
        
        candidate = y - eta_t * g_y
        candidate = proj_simplex(candidate, c_norm)
        
        x_new = candidate
        
        # Track best
        M_x = fast_autocorr(x_new).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x_new.copy()
        
        # Running average
        avg_a += x_new
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c_norm / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        # Nesterov extrapolation
        if t == 1:
            y_new = x_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y_new = x_new + beta * (x_new - x_prev)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = x_new.copy()
            else:
                y_new = y_new * c_norm / s_y
        
        M_y = fast_autocorr(y_new).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y_new.copy()
        
        x_prev = x_new
        y = y_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    SN t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_best_response(a, M, solver, T=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if T is None:
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M, cp.sum(b_var) >= 0.01]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    T_cache = None
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        if T_cache is None:
            T_cache = np.zeros((2 * n - 1, n))
            for j in range(n):
                T_cache[j:j + n, j] = y
        else:
            # Update T since y changed
            for j in range(n):
                T_cache[j:j + n, j] = y
        
        b = lp_best_response(y, M_curr, solver, T_cache)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c_norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c_norm / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c_norm / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent: targeted mass redistribution from peak contributors
    to low-impact positions. Uses peak-level analysis for smarter moves.
    """
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    ac = fast_autocorr(a)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find all peaks above threshold
        threshold = M * 0.95
        peak_lags = np.where(ac >= threshold)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.argsort(ac)[::-1][:5]
        
        improved = False
        
        for k in peak_lags[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            src_j = k_int - src
            
            # Find best destination: minimize impact on all peaks
            # A position d affects peak k if d is in [k-n+1, k] or k-d is in [0, n-1]
            # i.e., d affects peak k if |d - some_j| creates a term in c_k
            
            # Score each candidate destination by its total contribution to peaks
            peak_contrib = np.zeros(n)
            for k2 in peak_lags:
                k2_int = int(k2)
                js2 = max(0, k2_int - n + 1)
                je2 = min(n - 1, k2_int)
                for d in range(js2, je2 + 1):
                    partner = k2_int - d
                    if 0 <= partner < n:
                        peak_contrib[d] += a[d] * a[partner]
            
            # We want to move mass to a position with low peak_contrib
            # and also low current value (to avoid creating new peaks)
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            mask[src_j] = False
            
            # Use a combined score
            combined_score = peak_contrib + 0.1 * a
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue
            
            # Pick from the lowest-scoring candidates
            cand_scores = combined_score[candidates]
            n_candidates = min(50, len(candidates))
            idx_sorted = np.argsort(cand_scores)[:n_candidates]
            dst = candidates[idx_sorted[rng.integers(n_candidates)]]
            
            # Try several transfer fractions
            best_frac = None
            best_trial_m = best_m
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * c_norm / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_frac = frac
            
            if best_frac is not None and best_trial_m < best_m - 1e-15:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = a * c_norm / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            n_pert = rng.integers(1, 5)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c_norm * 1e-6:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a * c_norm / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        # Revert
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a * c_norm / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c_norm * c_norm)
        print(f"    CoordCD ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape (area-normalized)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    # Normalize to unit area for interpolation
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    # Rescale to have the right total sum for the new length
    target_sum = np.sqrt(2.0 * new_n)
    return f_interp * target_sum / ss

def spectral_init(n, pattern='bell'):
    """Generate initial sequence using spectral shaping ideas."""
    if pattern == 'bell':
        x = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif pattern == 'sinc':
        x = np.linspace(0, 1, n)
        a = np.sinc(6.0 * (x - 0.5))
        a = np.maximum(a, 0)
    elif pattern == 'cos':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif pattern == 'flat_mod':
        a = np.ones(n)
        # Add small sinusoidal modulation
        x = np.linspace(0, 1, n)
        a = a + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s

def propose_candidate(budget_s=1800.0):
    """
    Main search function: multi-phase optimization with smoothed gradients,
    LP refinement, coordinate descent, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        best_seq = spectral_init(n_base, 'bell') / spectral_init(n_base, 'bell').sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Nesterov smoothed gradient at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:

    # === Phase 2: Subgradient Nesterov at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)

        c0 = np.sqrt(2.0 * n_parent)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = subgrad_nesterov_optimize(a_norm, t2, top_k=10, beta=5.0, verbose=True)
        try_update(a_out, f"sng_n{n_parent}")
    
    # === Phase 3: LP best-response at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)

    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Optimize n={n1} ({t5:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n1)
        a_norm = a_long / a_long.sum()
        # First: smoothed gradient
        t5a = t5 * 0.5
        c1 = np.sqrt(2.0 * n1)
        a_norm_c = a_long * c1 / a_long.sum()
        a_out, _ = nesterov_smoothed_optimize(a_norm_c, t5a, tau_start=5.0, tau_end=150.0, verbose=False)
        try_update(a_out, f"nsm_n{n1}")
        
        # Then: coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.08, 100.0)
        if t5b > 10:
            c1 = np.sqrt(2.0 * n1)
            a_norm_c = best_seq * c1 / best_seq.sum()  # Use current best
            a_interp = interpolate_sequence(a_norm_c, n1)
            a_out, _ = coordinate_descent(a_interp, t5b, verbose=True)
            try_update(a_out, f"ccd_n{n1}")
    
    # === Phase 6: Scale to n=2400, optimize ===
    n2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 30:
        print(f"\n=== Phase 6: Optimize n={n2} ({t6:.0f}s) ===")
        c2 = np.sqrt(2.0 * n2)
        a_norm_c = best_seq * c2 / best_seq.sum()
        a_interp = interpolate_sequence(a_norm_c, n2)
        a_out, _ = nesterov_smoothed_optimize(a_interp, t6 * 0.6, tau_start=5.0, tau_end=100.0, verbose=False)
        try_update(a_out, f"nsm_n{n2}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.05, 80.0)
        if t6b > 10:
            a_out, _ = coordinate_descent(interpolate_sequence(best_seq * c2 / best_seq.sum(), n2), t6b, verbose=True)
            try_update(a_out, f"ccd_n{n2}")
    
    # === Phase 7: LP refinement at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2000:  # LP is slow for large n
        t7 = min(remaining * 0.10, 120.0)
        if t7 > 20:
            print(f"\n=== Phase 7: LP refinement n={n_best} ({t7:.0f}s) ===")
            c_best = np.sqrt(2.0 * n_best)
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
            try_update(a_out, f"lp_refine_n{n_best}")
    
    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final CoordCD n={n_best} ({t8:.0f}s) ===")
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t8, verbose=True)
        try_update(a_out, f"ccd_final_n{n_best}")
    
    # === Phase 9: Final smoothed gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.15, 150.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final Smoothed n={n_best} ({t9:.0f}s) ===")
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_smoothed_optimize(a_norm, t9, tau_start=20.0, tau_end=500.0, verbose=True)
        try_update(a_out, f"nsm_final_n{n_best}")
    
    # === Phase 10: One more coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t10 = min(remaining * 0.4, 100.0)
        if t10 > 10:
            print(f"\n=== Phase 10: Final Final CoordCD n={n_best} ({t10:.0f}s) ===")
            c_best = np.sqrt(2.0 * n_best)
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = coordinate_descent(a_norm, t10, verbose=True)
            try_update(a_out, f"ccd_v2_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
