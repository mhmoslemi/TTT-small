**Pair-based local search (detailed):**

Given a (normalized so sum = sqrt(2n)), compute c = conv(a,a), M = max(c).

Step 5: Apply the best (src, dst, delta) combination.

Step 6: Repeat until no improvement or time runs out.

- Current c[k] values
- Direction d[k] = change in c[k] per unit delta
- Find delta* = argmin max_k (c[k] + delta * d[k]) subject to 0 <= delta <= a[src]

The LP is:

This is equivalent to:
max 1'b

     b >= 0

where * denotes convolution (truncated to valid range).

In the frequency domain, a * b = ifft(fft(a) * fft(b)). But the constraint is in the time domain, so this doesn't directly help.

However, the constraint matrix T is a (2n-1) x n matrix where T[k,j] = a[j] if 0 <= k-j < n. This is a "convolution matrix" or "Toeplitz-like" matrix.

For solving the LP, the key is that the dual has a simple structure:
min M * sum(z)
s.t. sum_k T[k,j] * z[k] >= 1 for all j
     z >= 0

The dual constraint for variable j is: sum_k a[j] * z[k] * [0 <= k-j < n] >= 1
i.e., a[j] * sum_{k: |k-j| < n} z[k] >= 1
i.e., sum_{k: max(0,j-n+1) <= k <= min(j+n-1, 2n-2)} z[k] >= 1/a[j]

This is a covering problem! We want to cover the "demand" 1/a[j] for each j using a "weighted sum" of z[k] over a window.

This structure might be exploitable, but let's not over-engineer it. The cvxpy solver should handle it fine.

def build_conv_matrix(a):
    """Build the convolution matrix T where T[k,j] = a[j] if valid."""
    n = len(a)

    for j in range(n):

def lp_solve_constraint_gen(a, M, solver, max_rounds=15, initial_n_constraints=200):
    """Solve the best-response LP using constraint generation."""
    n = len(a)

    try:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(max_rounds):

        violations = circ - M
        if violations.max() < 1e-8 * M:

        # Find violated constraints not already included

        new_violated = np.where(violated & ~np.isin(np.arange(size), active_idx))[0]
        if len(new_violated) == 0:

def compute_direction_c(k, j, a, n):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov with constraint generation for fast best-response computation
2. Exact 1D pair-based local search with binary-search minimax optimization
3. Multi-length exploration with interpolation transfer
4. Polyak averaging and adaptive restarts for robust convergence
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_constraint_gen(a, M, solver, max_rounds=20, init_constraints=300):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    if size <= init_constraints:
        active_idx = np.arange(size)
    else:
        active_idx = np.unique(np.concatenate([
            np.linspace(0, size - 1, init_constraints, dtype=int),
            np.array([0, size // 2, size - 1])
        ]))

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                      [T[active_idx, :] @ b_var <= M[active_idx]])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(max_rounds):
        circ = T @ b_val
        max_viol = (circ - M).max()
        if max_viol < 1e-8 * max(M, 1e-15):
            break
        violated = circ - M > 1e-10 * max(M, 1e-15)
        all_idx = np.arange(size)
        not_active = ~np.isin(all_idx, active_idx)
        new_violated = np.where(violated & not_active)[0]
        if len(new_violated) == 0:
            break
        n_add = min(150, len(new_violated))
        worst = new_violated[np.argsort((circ - M)[new_violated])[::-1]][:n_add]
        active_idx = np.union1d(active_idx, worst)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                          [T[active_idx, :] @ b_var <= M[active_idx]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    circ = T @ b_val
    if circ.max() > M * (1 + 1e-5):
        return None
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_iters = 0
    stall = 0
    t0 = time.monotonic()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_constraint_gen(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        pw = 1.0 / math.sqrt(n_iters)
        avg_a = (1.0 - pw) * avg_a + pw * a

        candidates = [y, a]
        if n_iters % 10 == 0:
            candidates.append(avg_a)

        for cand in candidates:
            sc = fast_autocorr(cand).max()
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                stall = 0
            else:
                stall += 1

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if stall > 500:
            rng = np.random.default_rng(n_iters * 7 + 3)
            ps = 0.015 * a.mean()
            y = np.maximum(y + rng.normal(0, ps, n), 0)
            y = y * norm / y.sum()
            a = y.copy()
            avg_a = y.copy()
            stall = 0

    return best_a, best_score


def compute_contrib(j, k, a, n):
    k_j = k - j
    if 0 <= k_j < n:
        if k_j == j:
            return a[j]
        return 2.0 * a[k_j]
    return 0.0


def optimal_delta_1d(c_full, d_full, max_delta):
    if max_delta <= 1e-15:
        return 0.0
    m = min(len(c_full), 80)
    idx = np.argsort(c_full)[::-1][:m]
    cv = c_full[idx]
    dv = d_full[idx]
    max_cv = cv.max()
    active0 = cv >= max_cv - 1e-12
    if dv[active0].min() >= -1e-12:
        return 0.0
    lo = 0.0
    hi = max_delta
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = cv + mid * dv
        mv = vals.max()
        act = vals >= mv - 1e-12
        md = dv[act].min()
        xD = dv[act].max()
        if md >= 0:
            hi = mid
        elif xD <= 0:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5


def pair_local_search(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M = c.max()
        if M < 1e-15:
            break

        n_peaks = min(15, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        best_gain = 0.0
        best_src = -1
        best_dst = -1
        best_new_a = None

        for k in peak_idx[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_r = np.arange(js, je + 1)
            if len(idx_r) == 0:
                continue
            contribs = a[idx_r] * a[k - idx_r]
            top_n = min(5, len(idx_r))
            top_c = np.argsort(contribs)[::-1][:top_n]

            for tc in top_c:
                i = idx_r[tc]
                j = k - i
                if a[i] >= a[j]:
                    src = i
                else:
                    src = j
                if a[src] < 1e-15:
                    continue

                low_thresh = 0.4 * np.median(a)
                safe_mask = a < low_thresh
                safe_mask[src] = False
                safe_cands = np.where(safe_mask)[0]
                if len(safe_cands) == 0:
                    safe_cands = np.where(a < a.mean())[0]
                    safe_cands = safe_cands[safe_cands != src]
                if len(safe_cands) == 0:
                    continue

                n_safe = min(8, len(safe_cands))
                sel = safe_cands[rng.choice(len(safe_cands), n_safe, replace=False)]

                for dst in sel:

                    for kk in peak_idx:
                        d_full[kk] = compute_contrib(dst, kk, a, n) - compute_contrib(src, kk, a, n)
                    # Also check positions where d is large positive (could create new peaks)
                    extra = np.where(np.abs(d_full) > 0.1 * np.abs(d_full[peak_idx]).max() + 1e-15)[0]
                    extra = np.setdiff1d(extra, peak_idx)
                    if len(extra) > 0:
                        extra_sel = extra[rng.choice(len(extra), min(20, len(extra)), replace=False)]
                        for ek in extra_sel:
                            d_full[ek] = compute_contrib(dst, ek, a, n) - compute_contrib(src, ek, a, n)

                    delta = optimal_delta_1d(c, d_full, a[src])
                    if delta > 1e-12:
                        a_new = a.copy()
                        a_new[src] -= delta
                        a_new[dst] += delta
                        a_new = np.maximum(a_new, 0.0)
                        sn = a_new.sum()
                        if sn > 1e-15:
                            a_new = a_new * norm / sn
                            sc_new = fast_autocorr(a_new).max()
                            gain = M - sc_new
                            if gain > best_gain:
                                best_gain = gain
                                best_src = src
                                best_dst = dst
                                best_new_a = a_new

        if best_gain > 1e-10 and best_new_a is not None:
            a = best_new_a
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            n_improve += 1
            n_stall = 0
            if verbose and n_improve % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PairLS imp={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
        else:
            n_stall += 1
            if n_stall > 80:
                n_pert = max(3, n // 50)
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.05, 0.25)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                n_stall = 0

    return best_a, best_score
