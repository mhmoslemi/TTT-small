        if n_med >= WINDOW_MED and n_long % 20 == 0:
            avg_m = avg_med / WINDOW_MED
            score_m = 2.0 * n * fast_autocorr(avg_m).max()
            if score_m < best_score:
                best_score = score_m
                best_a = avg_m.copy()
                stall_count = 0

        if stall_count > 400 and t > 80:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_med = np.zeros(n)
            n_med = 0
            med_buf = [None] * WINDOW_MED
            med_idx = 0
            stall_count = 0
            warm_b = None

        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, eta0=0.02, max_iters=500000):
    """
    Multiplicative weights (mirror descent) optimization.
    Faster than LP but slightly less effective. Good for Phase 2.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        bmax_idx = np.argmax(b)
        bmax = b[bmax_idx]

        # Subgradient of max: gradient of (a*a)_{bmax_idx} w.r.t. a
        js = max(0, bmax_idx - n + 1)
        je = min(n - 1, bmax_idx)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[bmax_idx - idx]

        # Mirror descent: multiplicative update
        # a_new[j] = a[j] * exp(-eta * grad[j]) / Z
        log_update = -eta * grad
        log_update -= log_update.max()
        weights = np.exp(log_update)
        a_new = a * weights
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                eta *= 0.95
                if eta < 1e-6:
                    eta = eta0
                    a = best_a.copy()

        if t - last_print > 50000:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.2e}")
            last_print = t

        if stall > 10000:
            break

    return best_a, best_score


def topk_lse_objective_and_grad(a, T, topk):
    """Compute top-k LogSumExp of autocorrelation and its gradient."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for i in range(k_use):
        K = top_idx[i]
        wK = w[i]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a, top_idx, w


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=20, topk=15):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 5e-3 * min(1.0, 200.0 / n)
        lr = base_lr * (0.9 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(10, int(min(remaining, 40.0) * 8))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break

            a = softmax_fn(z)
            obj, grad_a, _, _ = topk_lse_objective_and_grad(a, stage_T, topk)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=42):
    """
    Aggressive peak shaving with multiple strategies:
    1. Direct transfer from peak contributors to safe positions
    2. Split transfer (reduce two contributors, increase one safe position)
    3. Random local search with acceptance criterion
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    last_print = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit - 2:
        iteration += 1
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        top_peaks = top_peaks[:20]

        # Compute danger mask: positions that contribute to any top peak
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        # Compute contribution score for each position
        contrib_score = np.zeros(n)
        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib_score[idx] += a[idx] * a[k - idx]

        improved = False

        # Strategy 1: Direct transfers
        safe_positions = np.where(~danger)[0]
        if len(safe_positions) > 0:
            # Sort safe positions by contribution (prefer those with zero contribution)
            safe_sorted = safe_positions[np.argsort(contrib_score[safe_positions])]
            safe_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]

            # Find top contributors to peaks
            for k in top_peaks[:8]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) == 0:
                    continue
                contrib = a[idx] * a[k - idx]
                top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

                for src in top_contrib_idx:
                    if a[src] < 1e-10:
                        continue
                    for dst in safe_pool[:6]:
                        if dst == src:
                            continue
                        for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                            amt = a[src] * frac
                            if amt < 1e-14:
                                continue
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            if a_trial[src] < -1e-14:
                                continue
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-12:
                                continue
                            a_trial = a_trial / st
                            new_max = fast_autocorr(a_trial).max()
                            if new_max < bmax * (1 - 1e-10):
                                a = a_trial
                                new_score = 2.0 * n * new_max
                                if new_score < best_score:
                                    best_score = new_score
                                    best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 2: Random perturbation if stuck
        if not improved:
            stall += 1
            if stall % 20 == 0:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True

            if stall > 500:
                # Reset to best and try larger perturbations
                a = best_a.copy()
                noise_scale = 0.02
                a_trial = a * (1 + rng.normal(0, noise_scale, n))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score * 1.01:
                        a = a_trial
                        stall = 0
                if stall > 1000:
                    break

        if iteration - last_print > 5000:
            elapsed = time.monotonic() - t0
            print(f"    PS iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}")
            last_print = iteration

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=123):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce the max autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        peak_idx = np.argmax(b)

        # Find the pair that contributes most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[peak_idx - idx]
        src = idx[np.argmax(contrib)]

        # Try moving from src to various destinations
        improved_this_round = False
        # Destination: position with minimum total contribution to all top peaks
        top_peaks = np.argsort(b)[::-1][:10]
        dest_score = np.zeros(n)
        for k in top_peaks:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            idx2 = np.arange(js2, je2 + 1)
            dest_score[idx2] += 1.0

        # Prefer destinations with low dest_score and low a value
        candidates = np.argsort(dest_score * (a + 1e-10))
        for dst in candidates[:20]:
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.01, 0.03]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                new_max = fast_autocorr(a_trial).max()
                if new_max < bmax * (1 - 1e-12):
                    a = a_trial
                    new_score = 2.0 * n * new_max
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    improved_this_round = True
                    stall = 0
                    break
            if improved_this_round:
                break

        if not improved_this_round:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init_asymmetric(n, rng):
    """Generate diverse asymmetric initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = x ** 1.5
    elif choice == 1:
        a = np.exp(-3 * x)
    elif choice == 2:
        a = np.exp(-5 * x) + 0.1 * x
    elif choice == 3:
        a = (1 - x) ** 2 * x
    elif choice == 4:
        a = np.exp(-8 * (x - 0.1) ** 2) + 0.3 * np.exp(-8 * (x - 0.6) ** 2)
    elif choice == 5:
        a = x * np.sin(np.pi * x / 2)
    elif choice == 6:
        a = np.exp(-2 * x) * (1 + 0.3 * np.cos(20 * np.pi * x))
    elif choice == 7:
        a = x ** 2 * (1 - x)
    elif choice == 8:
        a = np.exp(-4 * x) + 0.2 * np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 9:
        a = (1 / (x + 0.1)) ** 0.5
    elif choice == 10:
        a = np.sin(np.pi * x ** 0.5) ** 2
    else:
        a = np.exp(-6 * (1 - x) ** 2) * x
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP game, mirror descent, LSE gradient,
    aggressive peak shaving, coordinate polish, and length exploration.
    """

    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    # If no parent, generate initializations and pick best
    if best_seq is None:
        n_init = 1600
        for _ in range(5):

            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        print(f"Generated init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)

        if s < 0.01:
            return
        a = a / s

            best_seq = a.copy()

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: Aggressive peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)

            a_out, _ = adam_lse_optimize(pert, t8_each, T_start=15.0, T_end=1.0, n_stages=8, topk=12)

    # === Phase 9: Final LSE polish (very low temperature) ===

    if remaining > 15:
        t10a = min(remaining * 0.4, 60.0)
        t10b = min(remaining * 0.3, 40.0)
        if t10a > 8:
            print(f"=== Phase 10a: Final peak shave ===")
            a_out, _ = peak_shaving_aggressive(best_seq, t10a, seed=2001)
            try_update(a_out, "final_ps")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 8:
            t10b = min(remaining2 * 0.5, 30.0)
            if t10b > 5:
                print(f"=== Phase 10b: Final coordinate polish ===")
                a_out, _ = coordinate_descent_polish(best_seq, t10b, seed=3001)
                try_update(a_out, "final_cd")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
