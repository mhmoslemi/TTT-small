Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (top-k log-sum-exp) on the probability simplex
2. Adam optimizer with cosine learning rate schedule
3. Multi-scale LSE annealing (from smooth to sharp)
4. Local refinement with random perturbations and coordinate moves
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def full_conv(a):
    """Full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_lse_grad(a, topk, T):
    """Compute top-k LSE objective and gradient."""
    n = len(a)
    c = full_conv(a)
    k_use = min(topk, 2*n-1)
    top_idx = np.argpartition(c, -k_use)[-k_use:]
    top_vals = c[top_idx]
    
    max_tv = top_vals.max()
    exp_vals = np.exp(T * (top_vals - max_tv))
    w_top = exp_vals / exp_vals.sum()
    
    # Build weight vector (sparse)
    w = np.zeros(2*n-1)
    w[top_idx] = w_top
    
    # Gradient: 2 * cross_corr(w, a)
    grad = np.zeros(n)
    for idx in range(k_use):
        k = top_idx[idx]
        wk = w_top[idx]
        if wk < 1e-15:
            continue
        js = max(0, k - (n-1))
        je = min(n-1, k)
        i_range = np.arange(js, je+1)
        grad[i_range] += 2.0 * wk * a[k - i_range]
    
    lse_val = (max_tv + np.log(exp_vals.sum())) / T
    return lse_val, grad, c


def grad_optimize(a0, time_limit, T_start=50.0, T_end=0.5, topk_start=50, topk_end=5,
                  lr_start=0.01, n_restarts=3, seed=42):
    """
    Gradient-based optimization with LSE smoothing and Adam.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * full_conv(a).max()
    t0 = time.monotonic()
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        if restart == 0:
            a = best_a.copy()
        else:
            # Perturb best
            noise = rng.normal(0, 0.02 * (restart), n)
            a = best_a * (1 + noise)
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s if s > 1e-15 else best_a.copy()
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        n_steps = max(100, int((time_limit - (time.monotonic() - t0)) / n_restarts * 3))
        
        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            # Progress in [0, 1]
            progress = step / max(n_steps, 1)
            
            # Anneal T and topk
            T = T_start * (T_end / T_start) ** progress
            topk = max(topk_end, int(topk_start * (1 - progress) + topk_end * progress))
            
            # Learning rate with cosine decay
            lr = lr_start * 0.5 * (1 + np.cos(np.pi * progress))
            
            # Compute gradient
            lse_val, grad, c = compute_lse_grad(a, topk, T)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex
            a_new = project_simplex(a_new)
            
            # Check improvement
            score = 2.0 * n * full_conv(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
            
            if step % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Grad opt: restart={restart}, step={step}, T={T:.1f}, topk={topk}, best={best_score:.6f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def local_refine(a0, time_limit, seed=123):
    """
    Local refinement: try small random perturbations and coordinate moves.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * full_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        # Try random pair moves
        n_moves = 50
        improved = False
        
        for _ in range(n_moves):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-15:
                continue
            
            # Try moving a small fraction from i to j
            for frac in [0.001, 0.005, 0.01, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * full_conv(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Nesterov-accelerated LP zero-sum game (relaxation)
2. Smoothed gradient descent (top-k log-sum-exp) with Adam
3. Exact simplex projection
4. Local refinement via random coordinate moves
5. Multi-length exploration with interpolation
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def full_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v - theta, 0)
    # Ensure exact sum = 1 (fix floating point)
    s = result.sum()
    if s > 0:
        result = result / s
    return result


def compute_lse_grad(a, topk, T_param):
    """
    Compute top-k LSE smoothed max of conv(a,a) and its gradient.
    Returns (lse_value, gradient, full_convolution).
    """
    n = len(a)
    c = full_conv(a)
    k_use = min(topk, 2 * n - 1)
    top_idx = np.argpartition(c, -k_use)[-k_use:]
    top_vals = c[top_idx]
    
    max_tv = top_vals.max()
    exp_vals = np.exp(T_param * (top_vals - max_tv))
    w_top = exp_vals / exp_vals.sum()
    
    # Gradient: 2 * sum_k w[k] * a[k-i] for valid indices
    grad = np.zeros(n)
    for idx in range(k_use):
        k = top_idx[idx]
        wk = w_top[idx]
        if wk < 1e-15:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        i_range = np.arange(js, je + 1)
        grad[i_range] += 2.0 * wk * a[k - i_range]
    
    lse_val = (max_tv + np.log(exp_vals.sum())) / T_param
    return lse_val, grad, c


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    m = 2 * n - 1
    i = np.arange(m)[:, None]
    jj = np.arange(n)[None, :]
    diff = i - jj
    mask = (diff >= 0) & (diff < n)
    idx = np.clip(diff, 0, n - 1)
    T = np.where(mask, a_fixed[idx], 0.0)
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
