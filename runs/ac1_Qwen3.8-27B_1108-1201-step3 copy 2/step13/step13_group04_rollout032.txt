def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient_corrected(a, T, top_k=None):
    """
    Compute CORRECTED gradient of LSE surrogate: T*log(sum_k exp(b_k/T))
    where b = autocorr(a).
    
    dL/da[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
             = 2 * (a_rev * w)[m + n - 1]
    
    The previous version incorrectly used (a_rev * w)[m+n-1] + (a*w)[m].
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    if top_k is not None and top_k < len(w):
        # Zero out all but top-k entries
        thresholds = np.sort(w)[::-1][top_k]
        w[w < thresholds] = 0
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(w))
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    
    return 2.0 * r1


def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum


def nesterov_lse_corrected(a0, time_limit, T_start=15.0, T_end=0.3, n_stages=12, top_k=None, verbose=True):
    """Nesterov-accelerated LSE gradient descent with CORRECTED gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_extrap = y
            else:
                momentum = (np.sqrt(1.0 + 4.0 * it * it) - 1.0) / (2.0 * it)
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_ye = y_extrap.sum()
                if s_ye > 1e-15:
                    y_extrap = y_extrap * norm / s_ye

            grad = lse_gradient_corrected(y_extrap, T, top_k=top_k)

            # Step size: use T as learning rate (standard for LSE)
            eta = 1.0 / T
            y_new = y - eta * grad

            y_prev = y.copy()

            # Project: clip to non-negative, rescale to norm
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            # Track best (check periodically to save time)
            if it_count % 5 == 0 or it == n_iters_stage - 1:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

            if it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE-Nest stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check
    score = fast_autocorr(y).max()
    if score < best_score:
        best_score = score
        best_a = y.copy()

    return best_a, best_score

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(12.0 * np.pi * x) + 0.05 * np.cos(20.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x)
    elif choice == 6:
        # Smaller peaks at edges
        a = np.sin(np.pi * x) ** 2 * (1.0 + 0.2 * np.cos(4.0 * np.pi * x))
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(A_rev * W, fft_size)
    r1 = conv_result[n - 1:n - 1 + n]

    return 2.0 * r1


def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum


def nesterov_lse_corrected(a0, time_limit, T_start=20.0, T_end=0.2, n_stages=14, top_k=None, verbose=True):
    """Nesterov-accelerated LSE gradient descent with CORRECTED gradient and averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    # Running average for Nesterov
    avg_a = np.zeros(n)
    n_avg = 0
    best_avg_score = best_score
    best_avg_a = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(40, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_extrap = y
            else:
                beta = (np.sqrt(1.0 + 4.0 * it * it) - 1.0) / (2.0 * it)
                y_extrap = y + beta * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_ye = y_extrap.sum()
                if s_ye > 1e-15:
                    y_extrap = y_extrap * norm / s_ye

            grad = lse_gradient_corrected(y_extrap, T, top_k=top_k)

            # Step size
            eta = 1.0 / T
            y_new = y - eta * grad

            y_prev = y.copy()

            # Project: clip to non-negative, rescale
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            # Update running average
            avg_a += y
            n_avg += 1

            # Periodic evaluation
            if it_count % 5 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

            if n_avg % 50 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
                    if score_avg < best_avg_score:
                        best_avg_score = score_avg
                        best_avg_a = avg_check.copy()

            if verbose and it_count % 300 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Nest stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check
    score = fast_autocorr(y).max()
    if score < best_score:
        best_score = score
        best_a = y.copy()

    return best_a, best_score


def mirror_descent_corrected(a0, time_limit, T_start=10.0, T_end=0.2, n_stages=10, verbose=True):
    """Mirror descent (multiplicative weights) with CORRECTED gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(40, int(remaining * 2.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient_corrected(a, T)
            eta = 1.0 / (T * (it + 1) ** 0.35)
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            it_count += 1

            if it_count % 5 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

            if verbose and it_count % 300 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MirrorDes stage={si}/{n_stages} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    score = fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T_mat = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T_mat[j, k] = a[i]
    T_mat = T_mat.tocsr()

    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_fixed_point(a0, time_limit, solver, verbose=True):
    """LP best-response fixed-point iteration with diminishing step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    n_iters = 0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        y = y_new

        # Track average
        avg_a += y
        n_avg += 1

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search: move mass from high-contributing to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:4]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:6]]

            for src in top_contrib_idx:
                if a[src] < norm * 1e-8:
                    continue

                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 5:
                    candidates = np.where(np.arange(n) != src)[0]

                candidates = candidates[np.argsort(a[candidates])[:15]]

                for dst in candidates[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-10:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(4) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.003, 0.06)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 400:
                break

        if verbose and stall % 100 == 0 and stall > 0:
            print(f"    PeakSearch stall={stall} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(12.0 * np.pi * x) + 0.05 * np.cos(20.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x)
    elif choice == 6:
        a = np.sin(np.pi * x) ** 2 * (1.0 + 0.2 * np.cos(4.0 * np.pi * x))
    elif choice == 7:
        a = np.exp(-3.0 * (x - 0.5) ** 2) + 0.5
    elif choice == 8:
        a = 1.0 / (1.0 + 10.0 * x * (1.0 - x))
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses corrected LSE gradient (2*r1 fix), LP fixed-point,
    and peak targeting at multiple lengths with random restarts.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            prev_score = best_score
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (prev={prev_score:.6f}, n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: CORRECTED LSE Nesterov at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov (CORRECTED GRAD) n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.32, 500.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse_corrected(a_work, t1, T_start=25.0, T_end=0.2, n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_corr_n{n_base}")

    # ============ Phase 2: LP fixed-point ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP fixed-point n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_fixed_point(a_work, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # ============ Phase 3: LSE with different params (aggressive cooling) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE aggressive cooling n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse_corrected(a_work, t3, T_start=5.0, T_end=0.05, n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_aggr_n{n_best}")

    # ============ Phase 4: LSE at longer length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.14, 220.0)

        init_long = interpolate_sequence(best_seq, n2)

        do_try_update(a_out, f"lse_n{n2}")

    # ============ Phase 5: Peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)

        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeting_search(a_work, t5, seed=12345, verbose=True)

    # ============ Phase 6: Final LSE polish (very low T) ============
    remaining = time_budget - (time.monotonic() - t_global)
