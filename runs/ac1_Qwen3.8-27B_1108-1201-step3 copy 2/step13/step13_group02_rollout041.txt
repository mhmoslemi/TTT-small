Let me try a different approach: **primal-dual hybrid gradient (PDHG)** method.

Or even simpler: **use the fact that the LP is a packing LP** and apply the multiplicative weights algorithm.

The packing LP is: `max 1^T b subject to sum_i T_{k,i} b_i <= M_k for all k, b >= 0`.

The MWU algorithm:
1. Initialize `b_i = 1` for all `i`
2. For each iteration:
   a. Compute `w_k = exp(-eta * sum_i T_{k,i} b_i / M_k)` for each constraint `k`
   b. For each `i`: `b_i <- b_i * (1/n) * sum_k w_k * T_{k,i} / M_k`
   c. After `T` iterations, output `b / max_k (sum_i T_{k,i} b_i / M_k)`

This gives a (1-epsilon) approximate solution in `O(n^2 / epsilon^2)` iterations (each iteration being O(n^2)).

For our problem, `T_{k,i} = a_{k-i}` (or 0), so the computation can be done with convolutions (FFT), making each iteration O(n log n).

This is much faster than a full LP solver!

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast MWU (Multiplicative Weights Update) best response with FFT
2. Nesterov acceleration with adaptive restart
3. LSE (LogSumExp) smooth gradient descent via FFT-based gradients
4. Multi-destination peak shaving with simulated annealing
5. Multi-length exploration with interpolation transfer
Key innovation: Replaces expensive LP solves with O(n log n) MWU iterations,
allowing far more Nesterov iterations within the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a, b=None):
    """Compute full convolution using FFT."""
    n = len(a)
    if b is None:
        b = a
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def mwu_best_response(a, M, n_iter=100):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0
    using Multiplicative Weights Update (each iteration O(n log n) via FFT).
    """
    n = len(a)
    b = np.ones(n)
    a_rev = a[::-1]
    eta = 3.0 / np.sqrt(n)

    for _ in range(n_iter):
        ab = fast_conv(a, b)
        max_ab = ab.max()
        if max_ab < 1e-30:
            break
        scale = max(M, max_ab)
        v = ab / scale
        w = np.exp(-eta * v)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # s_i = sum_j a[j] * w[i+j] = conv(a_rev, w)[n-1+i]
        s = fast_conv(a_rev, w)[n - 1:2 * n - 1]
        s = np.maximum(s, 1e-30)

        b = b * s
        b_sum = b.sum()
        if b_sum < 1e-30:
            break
        b = b / b_sum * n

    ab = fast_conv(a, b)
    max_ab = ab.max()
    if max_ab > 1e-30:
        b = b * M / max_ab
    return b

                break

        if not improved:
            n_stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.03, 0.15)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 150:
                idx_pert = rng.choice(n, size=max(1, n // 80), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.05, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 1500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def cosine_basis_init(n, K=20):
    """Initialize using a cosine basis with random coefficients."""
    rng = np.random.default_rng(12345)
    x = np.arange(n)
    coeffs = rng.normal(0, 1.0, K)
    coeffs[0] = np.abs(coeffs[0]) + 2.0  # Ensure positive mean
    a = np.zeros(n)
    for k in range(K):
        a += coeffs[k] * np.cos(np.pi * k * (x + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses MWU-based Nesterov, LSE-GD, peak shaving,
    and multi-length exploration to minimize the objective.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    # Fallback initialization
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        # Try cosine basis init
        best_seq = cosine_basis_init(n_base, K=30)
        sc1 = evaluate_safe(best_seq)
        # Try sine init
        x = np.linspace(0.0, 1.0, n_base)
        alt_seq = np.sin(np.pi * x) ** 1.3
        alt_seq = alt_seq / alt_seq.sum()
        sc2 = evaluate_safe(alt_seq)
        if sc2 < sc1:
            best_seq = alt_seq
            best_score = sc2
        else:
            best_score = sc1
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # Phase 1: Nesterov-MWU at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov-MWU n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_mwu_optimize(a_norm, t1, n_mwu_iter=80, verbose=True)
        do_try_update(a_out, f"nmu_n{n_cur}")

    # Phase 2: LSE gradient descent at current best length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 130.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t3, seed=123, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # Phase 4: Try longer length
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov-MWU n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_mwu_optimize(a_norm, t4, n_mwu_iter=80, verbose=True)
        do_try_update(a_out, f"nmu_n{n2}")
        # Quick LSE polish
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.06, 60.0)
        if t4b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_gradient_descent(a_norm, t4b, verbose=False)
            do_try_update(a_out, f"lse_polish_n{n_cur}")

    # Phase 5: Try even longer
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Nesterov-MWU n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_mwu_optimize(a_norm, t5, n_mwu_iter=60, verbose=True)
        do_try_update(a_out, f"nmu_n{n3}")

    # Phase 6: Final Nesterov-MWU on best found
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Nesterov-MWU final n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_mwu_optimize(a_norm, t6, n_mwu_iter=60, verbose=True)
        do_try_update(a_out, f"nmu_final_n{n_best}")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: PeakShave final n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t7, seed=777, verbose=True)
        do_try_update(a_out, f"ps_final_n{n_best}")

    # Phase 8: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: LSE final n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.10, 80.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t8, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
