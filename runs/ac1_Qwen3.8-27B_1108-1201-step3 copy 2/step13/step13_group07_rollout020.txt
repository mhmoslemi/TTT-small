def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):

def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 1e-15:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:

        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:

            best_a = a_new.copy()
        a = a_new

    return best_a

def solve_lp_best_response(a_fixed):

    T = build_convolution_matrix(a_fixed)

    # Variables: [b_0, ..., b_{n-1}, M]
    # min M
    c = np.zeros(n + 1)
    c[n] = 1.0

    # (2n-1) x (n+1) inequality matrix

    b_ub = np.zeros(2 * n - 1)

                break
            continue
        consecutive_failures = 0

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t <= 2:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_old)
            a_new = project_simplex(a_new)

        y_old = y
        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall_count = 0
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.7f}, LP_M={M_val:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Direct subgradient optimization with adaptive smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    base_lr = 0.05
    t = 0

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        T_smooth = max(0.5, 30.0 * np.exp(-t / 5000.0))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        lr_t = base_lr / (1.0 + 0.005 * t)

        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with adaptive target selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        sorted_idx = np.argsort(b)[::-1]
        top_k = sorted_idx[:min(20, len(sorted_idx))]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and a[j] < np.median(a)]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            accepted = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    accepted = True
                    break
            if accepted:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20, lr=0.004):
    """LogSumExp-smoothed optimization with Adam optimizer (simplified)."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(100, int(remaining * 40 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 15 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Uses fast scipy-based LP for Nesterov game, diverse spectral initializations,
    and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.7f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.7f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Multi-strategy spectral initialization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral initialization n={n_base} ===")
    t1 = min(remaining * 0.05, 50.0)
    if t1 > 10:
        inits = [best_seq.copy()]

        # Different alpha values
        for alpha_val in [0.3, 0.5, 0.8]:
            a_spec = spectral_shaping(best_seq, n_iter=400, alpha=alpha_val)
            inits.append(a_spec)

        # Aggressive version
        a_agg = spectral_shaping_aggressive(best_seq, n_iter=300, alpha=0.5)
        inits.append(a_agg)

        # Sinc-like initialization
        x = np.linspace(-1, 1, n_base)
        a_sinc = np.sinc(x * 2.0) ** 2
        a_sinc = np.maximum(a_sinc, 0)
        a_sinc = a_sinc / a_sinc.sum()
        inits.append(a_sinc)

        # Fejer kernel
        k = np.arange(n_base) - (n_base - 1) / 2.0
        a_fejer = np.maximum(1.0 - np.abs(k) / n_base, 0)
        a_fejer = a_fejer / a_fejer.sum()
        inits.append(a_fejer)

        # Keep all inits that improve over parent
        for i, a_init in enumerate(inits):
            sc = internal_score(a_init)
            if sc < best_score:
                best_score = sc
                best_seq = a_init.copy()
                print(f"  Init {i}: score={sc:.7f} *** NEW BEST")

        best_init = best_seq.copy()
    else:
        best_init = best_seq.copy()

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.7f} (t={elapsed:.0f}s)")

    # === Phase 2: Nesterov LP (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} ===")
    t2 = min(remaining * 0.40, 680.0)
    if t2 > 30:
        # Warm up with quick LP to check speed
        t_warmup = min(5.0, t2 * 0.05)
        a_warm, sc_warm = nesterov_lp_fast(best_init, t_warmup, verbose=False)
        sc = evaluate_safe(a_warm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_init = np.maximum(a_warm, 0)
            s = best_init.sum()
            if s > 0:
                best_init = best_init / s
            print(f"  Warmup: {best_score:.7f}")

        # Main LP run
        t2_main = t2 - t_warmup - (time.monotonic() - t_global_start - (t_global_start + elapsed))
        t2_main = max(30.0, t2 - (time.monotonic() - t_global_start))
        a_out, sc_out = nesterov_lp_fast(best_init, t2_main, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.7f}")

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.7f} (t={elapsed:.0f}s)")

    # === Phase 3: Subgradient optimization (different direction) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient n={n_base} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_out, _ = subgradient_optimize(best_seq, t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:

            s = best_seq.sum()
            if s > 0:

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.7f} (t={elapsed:.0f}s)")

    # === Phase 4: Try longer length ===
    n_long = min(int(round(n_base * 1.3)), 2400)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Longer n={n_long} ===")
    t4 = min(remaining * 0.20, 340.0)

            if s > 0:

            print(f"  Spectral_n{n_long}: {best_score:.7f} *** NEW BEST")

        # LP at longer length
        t4_lp = t4 - (time.monotonic() - t_global_start)

        if t4_lp > 30:
            init_lp_l = interpolate_sequence(best_seq, n_long) if len(best_seq) != n_long else best_seq.copy()
            a_out_l, _ = nesterov_lp_fast(init_lp_l, t4_lp, verbose=True)

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving n={len(best_seq)} ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave_1]: {best_score:.7f}")

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.7f} (t={elapsed:.0f}s)")

        s = best_seq.sum()

        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
