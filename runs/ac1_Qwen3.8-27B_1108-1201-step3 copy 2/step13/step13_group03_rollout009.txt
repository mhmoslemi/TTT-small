import numpy as np
import time
from scipy.fft import rfft, irfft


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _power_spectrum(a, L):
    """Power spectrum of zero-padded sequence."""
    n = len(a)
    if L < n:
        L = 1 << (n - 1).bit_length()
    A = rfft(np.pad(a, (0, L - n)), L)
    return np.abs(A) ** 2


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad_conv(y, T):
    """Gradient of LSE-smoothed max of linear autocorrelation.
    Returns gradient with respect to y (on simplex)."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    # d/dy_j [sum_k w_k * ac_k] where ac_k = sum_i y_i * y_{k-i}
    # grad_j = 2 * sum_k w_k * (y_{k-j} + y_{j+k-n} terms) ... 
    # More efficiently: grad = 2 * sum_k w_k * (convolve indicator at position related to k)
    # ac_k = sum_i y_i * y_{k-i}, so d(ac_k)/dy_j = y_{k-j} + y_{j-(k)} ... 
    # Actually: d/dy_j [sum_i y_i y_{k-i}] = y_{k-j} (if 0<=k-j<n) + y_{j-k+n} ... 
    # Let's use the FFT approach:
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    # The gradient of sum_k w_k * ac_k w.r.t. y is 2 * (w * y) circular conv, then extract
    cross = irfft(np.conj(A) * W, fft_size)
    # For linear autocorrelation, the gradient is:
    # grad[j] = 2 * sum_k w_k * (y[k-j] if valid + y[j-k] if valid)
    # This equals 2 * (circular_conv(w, y))[j] approximately for zero-padded
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _spectral_flatness_fista(a0, time_limit, verbose=False):
    """Minimize variance of power spectrum (promotes flat autocorrelation)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # We want to minimize: variance of |F(a)|^2 = E[|F|^4] - (E[|F|^2])^2
    # E[|F|^2] is fixed (by Parseval = ||a||_2^2 * L), so minimize E[|F|^4]
    # = sum |F|^4 = sum |F|^2 squared
    # This is equivalent to minimizing the "peak" of the circular autocorrelation
    
    L = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1
    staging = a.copy()
    L_step = n / 10.0

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Gradient of sum |F(a)|^4 w.r.t. a
        # |F|^4 = |F|^2 * |F|^2 = (F conj(F)) (F conj(F))
        # In time domain: sum_k (circular_ac[k])^2 where circular_ac = ifft(|F|^2)
        # = sum_k (c_k)^2 where c = circular autocorrelation
        # d/d a_j sum_k c_k^2 = 2 * sum_k c_k * d(c_k)/d(a_j)
        # d(c_k)/d(a_j) = a[(j+k) mod L] + a[(j-k) mod L] (for zero-padded)
        # = 2 * circular_conv(c, a)[j] approximately
        
        A = rfft(y, L)
        ps = np.abs(A) ** 2
        circ_ac = irfft(ps, L)
        # grad[j] = 2 * sum_k circ_ac[k] * (y[(j+k)%L] + y[(j-k)%L])
        # = 4 * circ_conv(circ_ac, y)[j] for zero-padded
        W = rfft(circ_ac, L)
        grad_full = irfft(np.conj(A) * 4.0 * W, L)
        grad = grad_full[:n]
        
        # Also need to account for the zero-padding region
        # For zero-padded, the gradient in the first n positions is:
        # grad[j] = 2 * sum_k circ_ac[k] * (y[(j+k)%L] + y[(j-k)%L])
        # Since y is zero outside [0,n-1], this simplifies
        
        step = 1.0 / L_step
        z = y - step * grad
        a_new = _proj_simplex(z)

        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - staging)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
            else:
                y_new = a_new.copy()

        staging = a_new.copy()
        a = a_new
        y = y_new
        t_fista += 1

        if it % 500 == 0:
            sc = _fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent: move mass between pairs to reduce peak."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        peak_idx = int(np.argmax(b))
        
        # Find the pair that most contributes to the peak
        # b[peak_idx] = sum_i a[i] * a[peak_idx - i]
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-20:
            break
            
        src = idx[np.argmax(contrib)]
        
        # Find destination: position that minimally increases other peaks
        # Heuristic: move to a position with low current value
        candidates = np.where(a < np.median(a))[0]
        if len(candidates) == 0:
            candidates = np.where(a < a.max())[0]
        if len(candidates) == 0:
            break
        
        dst = candidates[rng.integers(len(candidates))]
        if dst == src:
            continue

        # Try several fractions
        improved = False
        for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-20:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t  # renormalize
            score = _fast_autocorr(a_trial).max()
            if score < best_score - 1e-18:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            stall += 1
            if stall > 2000:
                break

    return best_a, best_score

                    a_trial = a_trial / a_trial.sum()
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if not improved:
                stall += 1
                if stall > 3000:
                    break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _generate_initializer(n, kind, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)

    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'sinusoidal':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'sin15':
        a = np.sin(np.pi * x) ** 1.5
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif kind == 'gaussian':
        a = np.exp(-((x - 0.5) / 0.2) ** 2)
    elif kind == 'two_bumps':
        a = np.exp(-((x - 0.25) / 0.12) ** 2) + np.exp(-((x - 0.75) / 0.12) ** 2)
    elif kind == 'linear_ramp':
        a = x ** 0.5
    elif kind == 'random_smooth':
        a = rng.uniform(0.5, 1.0, n)
        # Smooth with a kernel
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif kind == 'random_sparse':
        a = rng.uniform(0, 1, n)
        # Make it sparser
        a[a < 0.3] = 0
        a = np.convolve(a, np.ones(5)/5, mode='same')
    elif kind == 'fourier_mix':
        a = np.ones(n)
        for k in range(1, 6):
            a += 0.3 / k * np.sin(2 * np.pi * k * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-method, multi-init, multi-length optimization."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1680.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    # Main search over multiple lengths and initializations
    lengths_to_try = [1600, 2000, 2400, 3000]
    if parent is not None:
        # Put parent length first
        lengths_to_try = [len(parent)] + [l for l in lengths_to_try if l != len(parent)]

    inits = ['sin15', 'sinusoidal', 'gaussian', 'random_smooth', 'fourier_mix', 'cosine']

    # Phase allocation: give more time to shorter/familiar lengths
    # Strategy: for each length, try a few inits with LSE-FISTA, then polish
    num_phases = len(lengths_to_try) * 2  # ~2 sub-phases per length
    time_per_unit = time_budget / (num_phases + 3)

    for length_idx, n in enumerate(lengths_to_try[:4]):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break

        # Decide time for this length (decreasing for longer lengths)
        if n <= 1600:
            t_alloc = min(remaining * 0.35, time_per_unit * 8)
        elif n <= 2400:
            t_alloc = min(remaining * 0.25, time_per_unit * 6)
        else:
            t_alloc = min(remaining * 0.20, time_per_unit * 5)
        t_alloc = max(t_alloc, 40)

        print(f"\n=== Length {n} (budget={t_alloc:.0f}s) ===")

        # Get initializations for this length
        inits_to_try = inits[:3] if t_alloc > 100 else inits[:2]
        if parent is not None and n == len(parent):
            inits_to_try = ['parent'] + inits_to_try

        t_per_init = t_alloc / (len(inits_to_try) + 1)  # +1 for polish

        for init_idx, init_name in enumerate(inits_to_try):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_init = min(t_per_init * 0.7, remaining * 0.4)
            if t_init < 10:
                continue

            if init_name == 'parent' and parent is not None:
                a0 = _interpolate_sequence(parent, n) if n != len(parent) else parent.copy()
            else:
                a0 = _generate_initializer(n, init_name, seed=seed + length_idx * 100 + init_idx)

            # LSE-FISTA
            T_sched = [15.0, 4.0, 1.0, 0.3, 0.1]
            if t_init > 60:
                T_sched = [20.0, 5.0, 1.5, 0.4, 0.1, 0.05]

            a_out, _ = _lse_fista_multi(a0, T_sched, t_init, verbose=False)
            _try_update(a_out, f"lse_{init_name}_n{n}")

        # Polish the best so far for this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and best_seq is not None:
            n_best = len(best_seq)
            if n_best == n or t_alloc > 60:
                t_polish = min(remaining * 0.3, t_per_init * 0.5)
                if t_polish > 5:
                    print(f"  Polishing n={n_best} (t={t_polish:.0f}s)")
                    a_out, _ = _mass_redistribute(best_seq, t_polish, seed=seed + 999)
                    _try_update(a_out, f"polish_n{n_best}")

    # Phase: Spectral flatness (new objective, different landscape)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        t_spec = min(remaining * 0.15, 120.0)
        if t_spec > 15:
            print(f"\n=== Spectral flatness n={n_best} (budget={t_spec:.0f}s) ===")
            a_out, _ = _spectral_flatness_fista(best_seq, t_spec, verbose=True)
            _try_update(a_out, f"spectral_n{n_best}")

    # Phase: Try a completely different length with spectral approach
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_new = len(best_seq) + 400
        t_new = min(remaining * 0.15, 100.0)
        if t_new > 15:
            print(f"\n=== Spectral flatness n={n_new} (budget={t_new:.0f}s) ===")
            a0 = _interpolate_sequence(best_seq, n_new)
            a_out, _ = _spectral_flatness_fista(a0, t_new, verbose=False)
            _try_update(a_out, f"spectral_n{n_new}")
            # Quick LSE polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out2, _ = _lse_fista_multi(a_out, [2.0, 0.5, 0.1], min(remaining * 0.3, 40.0), verbose=False)
                _try_update(a_out2, f"lse_after_spectral_n{n_new}")

    # Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and best_seq is not None:
        t_final = min(remaining * 0.5, 60.0)
        if t_final > 5:
            print(f"\n=== Final coord descent n={len(best_seq)} (budget={t_final:.0f}s) ===")
            a_out, _ = _coord_descent_polish(best_seq, t_final, seed=seed + 777777, verbose=False)
            _try_update(a_out, "final_coord")

    # Final mass redistribution
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and best_seq is not None:
        t_final2 = min(remaining * 0.6, 40.0)
        if t_final2 > 5:
            print(f"\n=== Final mass redistribute n={len(best_seq)} (budget={t_final2:.0f}s) ===")
            a_out, _ = _mass_redistribute(best_seq, t_final2, seed=seed + 888888)
            _try_update(a_out, "final_mass")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n = 1600
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
        a = a / a.sum()
        return [float(v) for v in a]

    return [float(x) for x in best_seq]
