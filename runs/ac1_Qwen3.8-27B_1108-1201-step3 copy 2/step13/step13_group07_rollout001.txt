"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping for initialization
2. Multi-stage L-BFGS-B on smoothed objective (main workhorse)
3. Nesterov LP with fast multiplicative-weights approximate best response
4. Multi-scale peak shaving / coordinate descent
5. Multi-length exploration with interpolation transfer
Key: L-BFGS-B with curvature info is more effective than Adam for this problem,

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 0:
            result = result / ss * target_sum
    return result


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_objective_and_grad(z, T):
    """
    Compute smoothed objective and gradient w.r.t. z.
    a = softmax(z), obj = (1/T)*log(sum_k exp(T*(a*a)_k))
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    obj = (max_Tb + np.log(exp_Tb.sum())) / T

    # Gradient: grad_a_j = 2 * sum_k w_k * a_{k-j}
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # Chain rule: grad_z = a * (grad_a - a . grad_a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbfgs_smoothed_optimize(a0, time_limit, T_start=50.0, T_end=0.005, n_stages=30, verbose=True):
    """Multi-stage L-BFGS-B optimization on the smoothed objective."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    t0 = time.monotonic()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Determine max iterations for this stage based on remaining time
        # Each L-BFGS-B iteration costs roughly 2 FFT evaluations
        est_time_per_iter = 0.003 * (n / 1000.0)
        max_iters = max(10, min(500, int(remaining * 80 / (n_stages - si))))

        def obj_fn(z_in):
            obj, _ = lse_objective_and_grad(z_in, stage_T)
            return obj

        def grad_fn(z_in):
            _, grad = lse_objective_and_grad(z_in, stage_T)
            return grad

        try:
            res = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': max_iters, 'maxfun': max_iters * 2,
                         'gtol': 1e-8, 'ftol': 1e-12}
            )
            z = res.x
        except Exception:
            pass

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LBFGS stage {si}/{n_stages} (T={stage_T:.4f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def multiplicative_weights_lp(a_fixed, n_inner=200, eta=0.1):
    """
    Fast approximate LP: min M s.t. (a*b)_k <= M, sum(b)=1, b>=0.
    Uses multiplicative weights on the dual.
    Dual: max lambda s.t. (T^T mu)_j >= lambda, sum(mu)=1, mu>=0
    Equivalently: max_{mu in Simplex} min_j sum_k mu_k * a[k-j]
    """
    n = len(a_fixed)
    num_lags = 2 * n - 1

    # Dual: find mu over lags to maximize min_j (T^T mu)_j
    # (T^T mu)_j = sum_k mu_k * T[k,j] = sum_k mu_k * a[k-j] (when 0<=k-j<n)
    # = sum_{d=max(0,j-n+1)}^{min(j,n-1)}... let me compute directly.
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    # (T^T mu)_j = sum_k mu[k] * T[k,j] = sum_k mu[k] * a[k-j] (if 0<=k-j<n)
    # Let d = k-j, so k = j+d, d ranges over [max(0,-j), min(num_lags-1-j, n-1)]
    # = sum_d mu[j+d] * a[d] for valid d

    # Initialize mu uniformly
    mu = np.ones(num_lags) / num_lags

    # Precompute T columns for efficiency: T[k,j] = a[k-j] if valid
    # For speed, compute (T^T mu)_j via cross-correlation
    # (T^T mu)_j = sum_k mu[k] * a[k-j] = (mu * a_reversed)_j ... let me think.
    # Actually (T^T mu)_j = sum_k mu[k] * a[k-j] for 0<=k-j<n
    # = sum_{d: 0<=d<n, 0<=j+d<2n-1} mu[j+d] * a[d]
    # This is a cross-correlation of mu and a.

    best_b = None
    best_M = float('inf')

    for it in range(n_inner):
        # Compute c_j = (T^T mu)_j for all j
        # c_j = sum_k mu[k] * a[k-j] (valid)
        # This is the cross-correlation: c = np.correlate(mu, a, mode='full') offset
        # Actually: c_j = sum_k mu[k] * a[k-j] = (mu * a_flipped)[j + n - 1] ... let me use FFT
        fft_size = 1 << (num_lags + n - 1 - 1).bit_length()
        MU = rfft(mu, fft_size)
        A_rev = rfft(a_fixed[::-1], fft_size)
        c = irfft(MU * A_rev, fft_size)
        # c[m] = sum_k mu[k] * a_rev[m-k] = sum_k mu[k] * a[n-1-(m-k)] = sum_k mu[k] * a[n-1-m+k]
        # We want c_j = sum_k mu[k] * a[k-j] for 0<=k-j<n
        # Let's just compute it directly for clarity, or use the right offset
        # a[k-j] is defined for 0<=k-j<n, i.e., j<=k<=j+n-1
        # c_j = sum_{k=j}^{j+n-1} mu[k] * a[k-j] (with mu[k]=0 for k>=2n-1)
        # This is the cross-correlation (mu ∘ a)[j] = sum_k mu[k] a[k-j]
        # In numpy: np.correlate(mu, a, mode='full')[n-1:] gives sum_k mu[k] a[k]
        # Actually let me just use the FFT approach properly.
        # cross_corr[j] = sum_k mu[k] * conj(a[k-j]) = sum_k mu[k] * a[k-j] (real)
        # Using FFT: IFFT(FFT(mu) * conj(FFT(a))) gives circular cross-correlation
        # For linear: zero-pad to size >= (2n-1) + n - 1

        fft_size2 = 1 << (num_lags + n - 1 - 1).bit_length()
        MU2 = rfft(mu, fft_size2)
        A2 = rfft(a_fixed, fft_size2)
        # cross-correlation: sum_k mu[k] * a[k-j]
        # = IFFT(conj(FFT(mu)) * FFT(a))[j] ... let me verify
        # If A = FFT(a), M = FFT(mu), then IFFT(M * conj(A))[j] = sum_k mu[k] conj(a[k-j])
        # For real: = sum_k mu[k] a[k-j]
        # But we need the right offset for linear correlation
        xcorr = irfft(MU2 * np.conj(A2), fft_size2)
        # xcorr[j] = sum_k mu[k] a[k-j] (circular)
        # For our range j=0..n-1, and k=0..2n-2, this should be correct with proper padding
        c = xcorr[:n]  # c_j for j=0..n-1

        min_c = c.min()
        j_star = c.argmin()

        # Multiplicative weights update on mu
        # We want to increase mu at lags that contribute to c[j_star]
        # c[j_star] = sum_k mu[k] * a[k-j_star]
        # So lags k where a[k-j_star] > 0 (i.e., 0 <= k-j_star < n) are "active"
        weights = np.zeros(num_lags)
        valid_idx = np.arange(j_star, j_star + n)
        valid_idx = valid_idx[valid_idx < num_lags]
        if len(valid_idx) > 0:
            weights[valid_idx] = a_fixed[valid_idx - j_star]

        # Update: mu_k *= (1 + eta * weights_k / (sum weights * mu))
        norm = np.dot(mu, weights)
        if norm > 1e-15:
            mu = mu * (1.0 + eta * weights / norm)
        mu = np.maximum(mu, 1e-30)
        mu = mu / mu.sum()

        # Periodically convert dual solution to primal
        if it % 50 == 49 or it == n_inner - 1:
            # Primal b: b_j proportional to exp(-eta2 * c_j)
            # Actually, the primal solution is related to the dual via complementary slackness
            # Simpler: just use b_j = exp(-c_j) / sum(exp(-c))
            c_clipped = np.clip(c, c.max() - 50, c.max())
            b = np.exp(-c_clipped)
            b = b / b.sum()
            M_val = fast_autocorr(a_fixed)  # (a*b) not (a*a)
            # Actually M = max_k (a*fixed * b)_k
            fft_sz = 1 << (2 * n - 1 - 1).bit_length()
            AF = rfft(a_fixed, fft_sz)
            BF = rfft(b, fft_sz)
            conv_ab = irfft(AF * BF, fft_sz)[:2 * n - 1].max()
            if conv_ab < best_M:
                best_M = conv_ab
                best_b = b.copy()

    if best_b is None:
        return np.ones(n) / n
    return best_b


def nesterov_lp_optimize(a0, time_limit, n_inner=100, eta=0.15, verbose=True):
    """Nesterov-accelerated approximate LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star = multiplicative_weights_lp(y, n_inner=n_inner, eta=eta)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:

            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if len(idx) < 2:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find good destination: low contribution to all peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            # Sort by current value (prefer moving to low-value positions)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_polish(a0, time_limit, n_top=5):
    """Subgradient method with top-K peak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    beta = 0.85
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Softmax weights on top peaks
        T_smooth = max(1.0, 30.0 * (1.0 - t / 5000.0))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        if conv_len > fft_size:
            fft_size *= 2
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        lr_t = 0.5 / (1 + 0.005 * t)
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.25, 0.15]:
            a_out = spectral_shaping(best_seq, n_iter=300, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After Phase 1: best={best_score:.6f}")

    # Phase 2: Nesterov MW-LP (fast approximate game solver)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov MW-LP n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, n_inner=80, eta=0.12)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [Nesterov_MW]: {best_score:.6f}")
        else:
            # Still use as starting point for next phase even if not better
            pass
    print(f"  After Phase 2: best={best_score:.6f}")

    # Phase 3: L-BFGS-B multi-stage (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: L-BFGS-B n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.35, 550.0)
    if t3 > 30:
        a_out, sc_out = lbfgs_smoothed_optimize(
            best_seq, t3, T_start=30.0, T_end=0.003, n_stages=35, verbose=True
        )
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LBFGS_main]: {best_score:.6f}")
    print(f"  After Phase 3: best={best_score:.6f}")

    # Phase 4: Second L-BFGS-B round with lower temperatures
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: L-BFGS-B round 2 n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, sc_out = lbfgs_smoothed_optimize(
            best_seq, t4, T_start=5.0, T_end=0.001, n_stages=20, verbose=True
        )
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LBFGS_r2]: {best_score:.6f}")
    print(f"  After Phase 4: best={best_score:.6f}")

    # Phase 5: Subgradient polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Subgradient n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_out, sc_out = subgradient_polish(best_seq, t5)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [SubGrad]: {best_score:.6f}")
    print(f"  After Phase 5: best={best_score:.6f}")

    # Phase 6: Try longer length
    n_long = int(round(n_base * 1.25))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 300.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.2)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  *** NEW BEST [spectral_n{n_long}]: {best_score:.6f}")
        # L-BFGS-B at longer length
        t6b = t6 * 0.70
        a_out, _ = lbfgs_smoothed_optimize(
            best_seq, t6b, T_start=20.0, T_end=0.005, n_stages=25, verbose=False
        )
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LBFGS_n{n_long}]: {best_score:.6f}")
    print(f"  After Phase 6: best={best_score:.6f}")

    # Phase 7: Peak shaving (final refinement at best length)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_out, _ = peak_shaving(best_seq, t7, seed=456)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [peak_shave_1]: {best_score:.6f}")
    print(f"  After Phase 7: best={best_score:.6f}")

    # Phase 8: Another L-BFGS-B round for final polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final L-BFGS-B n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        a_out, _ = lbfgs_smoothed_optimize(
            best_seq, t8, T_start=3.0, T_end=0.001, n_stages=15, verbose=False
        )
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LBFGS_final]: {best_score:.6f}")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t9 = min(remaining * 0.40, 80.0)
        if t9 > 10:
            a_out, _ = peak_shaving(best_seq, t9, seed=789)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [peak_shave_final]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
