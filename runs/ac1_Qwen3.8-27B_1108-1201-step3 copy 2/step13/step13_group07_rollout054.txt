"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Game-based optimization with QP best response (Hankel structure + FFT matvec + CG)
2. Multiplicative weights for the dual variable
3. Subgradient polish with Nesterov acceleration
4. Peak shaving for final refinement
5. Multi-length exploration (n=1600, 2000, 2400)

Key innovation: Exploits the Hankel structure of the autocorrelation matrix
to solve the QP best-response in O(n*log(n)*sqrt(kappa)) per iteration using
CG with FFT-based matrix-vector products, enabling ~100k+ iterations in budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def hankel_matvec(q, x):
    """Compute Q(q) @ x where Q(q)_{ij} = q_{i+j}. q has length 2n-1, x has length n."""
    n = len(x)
    # result_i = sum_j q[i+j] * x_j = (x_rev * q)[i + n - 1]
    conv_len = n + len(q) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    X_rev = rfft(x[::-1], fft_size)
    Q_fft = rfft(q, fft_size)
    conv = irfft(X_rev * Q_fft, fft_size)
    return conv[n - 1: 2 * n - 1]


def solve_equality_qp(q, max_cg_iters=200, tol=1e-14):
    """
    Solve min a^T Q(q) a s.t. 1^T a = 1 (equality only, no non-negativity).
    Returns a normalized solution. Uses CG with FFT-based Hankel matvec.
    """
    n_a = (len(q) + 1) // 2
    b = np.ones(n_a)
    x = np.ones(n_a) / n_a  # initial guess: uniform
    r = b - hankel_matvec(q, x)
    p = r.copy()
    rs_old = r @ r
    if rs_old < 1e-30:
        return x

    for _ in range(max_cg_iters):
        Ap = hankel_matvec(q, p)
        pAp = p @ Ap
        if pAp < 1e-30:
            break
        alpha = rs_old / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = r @ r
        if rs_new < tol * rs_old:
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new

    s = x.sum()
    if abs(s) < 1e-30:
        return np.ones(n_a) / n_a
    x = x / s
    return x


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n * target_sum
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        result = np.maximum(result + diff / n, 0)
        result = result / result.sum() * target_sum
    return result


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def game_optimize(a_init, time_limit, n_dual_smooth=50, verbose=True):
    """
    Game-based optimization:
    - Column player (a): solves QP best response using CG + Hankel matvec
    - Row player (q): multiplicative weights update
    Returns best a found.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize q: start with peaks of a*a
    b0 = fast_autocorr(a)
    tau_init = 0.1 * b0.max()
    q = np.exp(b0 / tau_init)
    q = q / q.sum()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    # Dual running average for better convergence
    q_avg = q.copy()
    n_avg = 0

    # Learning rate schedule for multiplicative weights
    total_expected_iter = 200000
    eta_base = np.sqrt(2.0 * np.log(2 * n - 1) / max(total_expected_iter, 100))

    iter_count = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        # Solve QP best response for a given q
        a_new = solve_equality_qp(q, max_cg_iters=100, tol=1e-16)

        # Check non-negativity; if violated, project
        if a_new.min() < -1e-12:
            a_new = project_simplex(a_new, 1.0)

        # Update q using multiplicative weights
        b_new = fast_autocorr(a_new)
        eta_t = eta_base * min(1.0, 5000.0 / iter_count)
        log_q = np.log(np.maximum(q, 1e-300)) + eta_t * b_new
        log_q = log_q - log_q.max()
        q_new = np.exp(log_q)
        q_new = q_new / q_new.sum()

        # Smooth q to prevent concentration (mix with previous)
        mix = min(1.0, 10.0 / iter_count)
        q = (1.0 - mix) * q_new + mix * q
        q = np.maximum(q, 0)
        qs = q.sum()
        if qs > 1e-30:
            q = q / qs

        # Update dual average
        q_avg = q_avg + q
        n_avg += 1

        a = a_new

        # Track best
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check dual average periodically
        if n_avg >= 200 and n_avg % 500 == 0:
            q_check = q_avg / n_avg
            a_dual = solve_equality_qp(q_check, max_cg_iters=150)
            if a_dual.min() < -1e-12:
                a_dual = project_simplex(a_dual, 1.0)
            score_dual = internal_score(a_dual)
            if score_dual < best_score:
                best_score = score_dual
                best_a = a_dual.copy()

        if verbose and time.monotonic() - last_print > 5.0:
            elapsed = time.monotonic() - t0
            print(f"    Game n={n} iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}")
            last_print = time.monotonic()

    return best_a, best_score


def subgradient_optimize(a_init, time_limit, verbose=True):
    """
    Subgradient optimization with Nesterov acceleration and
    softmax-weighted subgradient for smoothness.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.7
    base_lr = 0.05
    t = 0

    # FFT setup for gradient computation
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    last_print = t0
    stall = 0
    best_score_hist = best_score

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Adaptive temperature: start high, decay
        frac = min(1.0, t / 50000.0)
        T_smooth = 100.0 * (1.0 - frac) + 1.0
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Compute subgradient via FFT
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1: n - 1 + n]

        # Learning rate with decay
        lr_t = base_lr / (1.0 + 0.02 * t)

        # Nesterov momentum
        v = beta * v + (1.0 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new, 1.0)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if verbose and time.monotonic() - last_print > 5.0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad n={n} t={t} ({elapsed:.0f}s): best={best_score:.8f}")
            last_print = time.monotonic()

        if stall > 100000:
            break

    return best_a, best_score


def peak_shaving(a_init, time_limit, seed=123):
    """Targeted peak shaving via coordinate moves."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:20]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: low contribution to all top peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            if not candidates:
                continue

            # Pick candidate with lowest "risk" (min a[j] to minimize impact)
            cands_arr = np.array(candidates)
            dst_pool = cands_arr[np.argsort(a[cands_arr])[:max(1, len(cands_arr) // 4)]]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = internal_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local optima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 2000:
                break

    return best_a, best_score


def spectral_shaping(a, n_iter=200, alpha=0.25):
    """Spectral shaping to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_polish(a_init, time_limit, T_start=50.0, T_end=1.0, n_stages=20, lr=0.002):
    """LogSumExp-smoothed optimization with Adam."""
    n = len(a_init)
    a_init = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    last_print = t0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        iters = max(20, min(300, int(remaining * 80 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()

            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient of LSE w.r.t. a
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1: n - 1 + n]

            # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1.0 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 50 == 0:
                a = np.exp(z - z.max())
                a = a / a.sum()
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = np.exp(z - z.max())
        a = a / a.sum()
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if time.monotonic() - last_print > 5.0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} T={stage_T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")
            last_print = time.monotonic()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses game-based QP optimization as the primary method,
    combined with subgradient, LSE, and peak shaving for polish.
    Explores multiple sequence lengths.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 70.0, 1730.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: Quick spectral shaping on parent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.3, 0.15]:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")

    # Phase 2: Game-based QP optimization at parent length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Game QP n={n_parent} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_out, sc_out = game_optimize(best_seq, t2)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Game_n{n_parent}]: {best_score:.8f}")
        print(f"  Game internal: {sc_out:.8f}")

    # Phase 3: Game at longer length (2000)
    n_long = 2000
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Game QP n={n_long} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral at longer length
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.2)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  Spectral n={n_long}: {best_score:.8f}")

        a_out, sc_out = game_optimize(init_long, t3 * 0.85)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Game_n{n_long}]: {best_score:.8f}")
        print(f"  Game internal: {sc_out:.8f}")

    # Phase 4: Subgradient polish at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Subgradient n={n_best} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, sc_out = subgradient_optimize(best_seq, t4)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad]: {best_score:.8f}")

    # Phase 5: LSE polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE polish n={n_best} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_out, sc_out = lse_polish(best_seq, t5, T_start=30.0, T_end=2.0, n_stages=15, lr=0.001)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LSE]: {best_score:.8f}")

    # Phase 6: Another game round at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Game round 2 n={n_best} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 20:
        a_out, sc_out = game_optimize(best_seq, t6, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Game_r2]: {best_score:.8f}")

    # Phase 7: Peak shaving (final refinement)
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Peak shaving n={n_best} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 5:
        a_out, sc_out = peak_shaving(best_seq, t7, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave]: {best_score:.8f}")

    # Phase 8: Final subgradient with very small steps
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t8 = min(remaining * 0.30, 100.0)
        print(f"\n=== Phase 8: Final subgrad ({t8:.0f}s) ===")
        a_out, sc_out = subgradient_optimize(best_seq, t8, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 1e-15:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [final_subgrad]: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 1e-15:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
