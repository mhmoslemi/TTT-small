from scipy.fft import rfft, irfft, fft, ifft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient_step(a, w, fft_size):
    """
    Compute gradient of LSE objective w.r.t. a.
    w = softmax weights over convolution values (length 2n-1).
    Returns gradient of length n.
    """
    n = len(a)
    size = 2 * n - 1
    w_rev = w[::-1]
    
    # g1[i] = sum_j a[j] * w[i+j] = (a * w_rev)[2n-2+i]
    A = rfft(a, fft_size)
    Wr = rfft(w_rev, fft_size)
    conv_awr = irfft(A * Wr, fft_size)
    g1 = conv_awr[2*n-2 : 2*n-2+n]

    W = rfft(w, fft_size)
    conv_aw = irfft(A * W, fft_size)
    g2 = conv_aw[0:n]
    
    return g1 + g2

        remaining = time_limit - (time.monotonic() - t0)

            break

        m = np.zeros(n)
        v = np.zeros(n)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_gradient(a, w, fft_size):
    n = len(a)
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv_a_rev_w = irfft(A_rev * W, fft_size)
    conv_a_w = irfft(A * W, fft_size)
    g1 = conv_a_rev_w[n - 1: n - 1 + n]
    g2 = conv_a_w[0: n]
    return g1 + g2

def lse_adam_optimize(a0, time_limit, rng=None, verbose=True, max_iters=50000):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    if rng is None:
        rng = np.random.default_rng(0)

    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    it_count = 0

    # Adaptive temperature: start at spread*3, end at spread*0.05
    # Use geometric decay over iterations
    n_total_stages = max(1, max_iters)
    n_stages = 10
    stage_iters = max(50, max_iters // n_stages)

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_autocorr(a)
        max_conv = conv.max()
        mean_conv = conv.mean()
        spread = max(max_conv - mean_conv, 1e-8)

        # Temperature for this stage
        frac = stage / max(1, n_stages - 1)
        T = spread * (3.0 * (1 - frac) + 0.05 * frac)
        T = max(T, 1e-4)

        # Reset Adam per stage for better convergence at new scale
        m = np.zeros(n)
        v = np.zeros(n)
        it_count = 0

        stage_limit = stage_iters
        if remaining > 5.0:
            stage_limit = min(stage_limit, max(50, int(remaining * 3)))

        for _ in range(stage_limit):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()

            # Softmax weights
            shifted = (conv - max_conv) / T
            exp_conv = np.exp(shifted)
            w = exp_conv / exp_conv.sum()

            grad = compute_lse_gradient(a, w, fft_size)

            it_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** it_count)
            v_hat = v / (1 - beta2 ** it_count)

            # Learning rate: scale with norm, decay with iterations
            lr = 0.8 * norm / (n * (T + 1e-8)) / (1.0 + 0.005 * it_count)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {stage+1}/{n_stages} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)

        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)

    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:

        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Symmetric bump
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    else:
        # Random smooth
        k = rng.integers(3, 10)
        coeffs = rng.standard_normal(k)
        x = np.linspace(0.0, 1.0, n)
        a = np.zeros(n)

        a = np.abs(a) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a

    if active_set is not None and len(active_set) < size:
        constraints = [T[active_set, :] <= M]
    else:
        active_set = np.where(conv_current > M * 0.7)[0]
        if len(active_set) < 20:
            active_set = np.where(conv_current > M * 0.4)[0]
        constraints = [T[active_set, :] <= M]

    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set

    # Check for violated constraints
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) > 0.85 * size:
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T @ b_var2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active
            else:
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T[new_active, :] <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active

def cyclical_cd(a, time_limit, rng, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try 2 random partners
            for _ in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector for transferring delta from i to j
                g = np.zeros(size)
                ai = a[i]

                # Add: shifting mass to j adds to conv
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[max(0, lo - j):max(0, hi - j)]
                lo2 = max(0, n - 1 - j)
                hi2 = min(size, n + j)
                if hi2 > lo2:
                    idx2 = np.arange(lo2, hi2)
                    src2 = n - 1 - idx2
                    valid = (src2 >= 0) & (src2 < n)
                    g[lo2:hi2][valid] += a[src2[valid]]

                # Subtract: removing mass from i
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[max(0, lo3 - i):max(0, hi3 - i)]
                lo4 = max(0, n - 1 - i)
                hi4 = min(size, n + i)
                if hi4 > lo4:
                    idx4 = np.arange(lo4, hi4)
                    src4 = n - 1 - idx4
                    valid4 = (src4 >= 0) & (src4 < n)
                    g[lo4:hi4][valid4] -= a[src4[valid4]]

                # Sample delta values
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                best_d = 0.0
                best_val = conv.max()
                for d in deltas[1:]:
                    new_max = (conv + d * g).max()
                    if 2 * i < size:
                        nm = conv[2 * i] - 2 * ai * d + d * d
                        new_max = max(new_max, nm)
                    if 2 * j < size:
                        nm = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = max(new_max, nm)
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved and n_passes > 2:
            break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_adam_optimize_restarts(a0, time_limit, rng, verbose=True, n_restarts=3):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    total_per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        # Perturb the current best
        a_start = best_a.copy()
        if r > 0:
            # Add noise proportional to current magnitude
            noise_scale = 0.1 / (r + 1)
            noise = rng.standard_normal(n) * noise_scale * a_start.max()
            a_start = a_start + noise
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            if s < 1e-15:
                a_start = best_a.copy()
            else:
                a_start = a_start * norm / s

        this_limit = min(total_per_restart, remaining - 2.0)
        if this_limit < 3.0:
            continue

        a_out, sc = lse_adam_optimize(a_start, this_limit, rng=rng, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE restart {r+1}/{n_restarts} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        best_n = n_base
        print(f"Default init: n={best_n}, score={best_score:.6f}")

    # Phase 1: Multi-length LSE sweep (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase1_budget = min(remaining * 0.22, 330.0)
    print(f"\n=== Phase 1: Multi-length LSE sweep (budget={phase1_budget:.0f}s) ===")

    lengths_to_try = [800, 1200, 1600, 2000, 2400, 3000, 4000]
    # Always include the parent length
    if best_n not in lengths_to_try:
        lengths_to_try.append(best_n)
    lengths_to_try = sorted(set(lengths_to_try))

    n_lengths = len(lengths_to_try)
    per_length = phase1_budget / n_lengths

    for li, n_try in enumerate(lengths_to_try):
        if time.monotonic() - t_global > time_budget - 120.0:
            break
        remaining_here = time_budget - 120.0 - (time.monotonic() - t_global)
        this_budget = min(per_length, remaining_here)
        if this_budget < 5.0:
            break

        # Initialize
        if best_seq is not None and abs(n_try - best_n) <= best_n:
            init = interpolate_sequence(best_seq, n_try)
        else:
            init = structured_init(n_try, rng)

        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = lse_adam_optimize(a_norm, this_budget, rng=rng, verbose=False)
        try_update(a_out, f"lse_n{n_try}")

    # Phase 2: Extended LSE on top lengths (28% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase2_budget = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: Extended LSE (budget={phase2_budget:.0f}s) ===")

    # Try the best length and one neighbor
    top_lengths = [best_n]
    for nl in lengths_to_try:
        if nl != best_n and nl not in top_lengths:
            top_lengths.append(nl)
            break
    # Also try a slightly different length
    alt_n = best_n + 200
    if alt_n not in top_lengths:
        top_lengths.append(alt_n)

    n_top = len(top_lengths)
    per_top = phase2_budget / n_top

    for tl in top_lengths[:3]:
        if time.monotonic() - t_global > time_budget - 180.0:
            break
        remaining_here = time_budget - 180.0 - (time.monotonic() - t_global)
        this_budget = min(per_top, remaining_here)
        if this_budget < 10.0:
            break

        init = interpolate_sequence(best_seq, tl)
        a_norm = init * np.sqrt(2.0 * tl) / init.sum()
        a_out, _ = lse_adam_optimize_restarts(a_norm, this_budget, rng=rng, verbose=True, n_restarts=3)
        try_update(a_out, f"lse_ext_n{tl}")

    # Phase 3: LP Nesterov on best (25% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase3_budget = min(remaining * 0.25, 400.0)
    n_lp = best_n
    print(f"\n=== Phase 3: LP Nesterov n={n_lp} (budget={phase3_budget:.0f}s) ===")

    if phase3_budget > 20.0 and n_lp <= 4000:
        a_norm = best_seq * np.sqrt(2.0 * n_lp) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, phase3_budget, solver, verbose=True)
        try_update(a_out, f"lp_n{n_lp}")

    # Phase 4: Cyclical CD (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase4_budget = min(remaining * 0.18, 250.0)
    print(f"\n=== Phase 4: Cyclical CD n={best_n} (budget={phase4_budget:.0f}s) ===")

    if phase4_budget > 10.0:
        a_norm = best_seq * np.sqrt(2.0 * best_n) / best_seq.sum()
        a_out, _ = cyclical_cd(a_norm, phase4_budget, rng, verbose=True)
        try_update(a_out, f"cd_n{best_n}")

    # Phase 5: Fresh multi-start at best length (12% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase5_budget = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Multi-start n={best_n} (budget={phase5_budget:.0f}s) ===")

    if phase5_budget > 15.0:
        n_starts = min(5, max(2, int(phase5_budget / 20)))
        per_start = phase5_budget / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global > time_budget - 60.0:
                break
            remaining_here = time_budget - 60.0 - (time.monotonic() - t_global)
            if remaining_here < 5.0:
                break
            init = structured_init(best_n, rng)
            # Mix with current best
            mix = 0.5 * init + 0.5 * best_seq
            mix = np.maximum(mix, 0.0)
            s_mix = mix.sum()
            if s_mix < 1e-15:
                mix = init
            else:
                mix = mix / s_mix

            a_norm = mix * np.sqrt(2.0 * best_n) / mix.sum()
            this_budget = min(per_start, remaining_here)
            if this_budget < 3.0:
                continue

            a_out, _ = lse_adam_optimize(a_norm, this_budget * 0.7, rng=rng, verbose=False)
            # Short CD polish
            this_cd = this_budget * 0.3
            if this_cd > 2.0:
                a_out2, _ = cyclical_cd(a_out, this_cd, rng, verbose=False)
                a_out = a_out2

            try_update(a_out, f"start{si+1}_n{best_n}")

    # Phase 6: Final polish (5% time)
    remaining = time_budget - (time.monotonic() - t_global)
    phase6_budget = min(remaining * 0.3, 60.0)
    print(f"\n=== Phase 6: Final polish n={best_n} (budget={phase6_budget:.0f}s) ===")

    if phase6_budget > 5.0:
        a_norm = best_seq * np.sqrt(2.0 * best_n) / best_seq.sum()
        a_out, _ = cyclical_cd(a_norm, phase6_budget, rng, verbose=False)
        try_update(a_out, f"final_cd_n{best_n}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
