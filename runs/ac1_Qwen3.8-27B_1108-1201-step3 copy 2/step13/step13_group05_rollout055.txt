    M = ac.max()
    w = np.exp(beta * (ac - M))
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    N = 1 << (n + len(w) - 2).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    S = irfft(W * np.conj(A), N)[:n]
    return 2.0 * S / w_sum

def compute_topk_subgrad(a, ac, top_k=5, beta=3.0):
    """Compute weighted top-k subgradient (sharper, more focused)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        i_start = max(0, int(k) - n + 1)
        i_end = min(n, int(k) + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = int(k) - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def mirror_descent_soft(a0, time_limit, beta_start=8.0, beta_end=0.5,
                        gamma=1.0, noise_level=0.005, noise_period=1500,
                        verbose=True):
    """Mirror descent with FFT soft subgradient, annealing, and noise restarts."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s0 = a.sum()
    a = a * c / s0

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(t / 8000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * frac

        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m:
            best_m = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        g = compute_soft_subgrad(a, ac, beta)
        g_max = g.max()
        if g_max > 1e-10:
            g = g / g_max

        step = gamma / (np.sqrt(t) + 1.0)

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new

        # Noise injection to escape local minima
        if t % noise_period == 0:
            if no_improve > noise_period // 2:
                strength = noise_level * (1.0 + no_improve / float(noise_period))
                noise = rng.normal(0, strength / np.sqrt(n), size=n)
                a_noisy = a + noise
                a_noisy = np.maximum(a_noisy, 0)
                s_n = a_noisy.sum()
                if s_n > 1e-15:
                    a_noisy = a_noisy * c / s_n
                    M_noisy = fast_autocorr(a_noisy).max()
                    if M_noisy < best_m * 1.02:
                        a = a_noisy
                        no_improve = 0
                        if M_noisy < best_m:
                            best_m = M_noisy
                            best_a = a.copy()

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.2f}, no_imp={no_improve}")

    return best_a, best_m

def build_constraint_matrix(a):
    """Build T where T[k,j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        T[j:k_end, j] = a[:k_end - j]
    return T

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_constraint_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.shape != (n,):
        b_val = b_val.flatten()
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True, max_iter=None):
    """Nesterov-accelerated LP best-response optimization."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        if max_iter is not None and n_iter >= max_iter:
            break
        n_iter += 1

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * c / b_sum

        alpha = 2.0 / (n_iter + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y

        if n_iter == 1:
            x_new = g.copy()
        else:
            beta_m = (n_iter - 1.0) / (n_iter + 2.0)
            x_new = y_new + beta_m * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x

        y = y_new
        x = x_new

        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()

        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()

        if verbose and n_iter % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={n_iter} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent with simulated annealing acceptance."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    temperature = 0.01

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[int(k) - idx]
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            cands_sorted = candidates[np.argsort(a[candidates])]
            n_cand = len(cands_sorted)
            dst = cands_sorted[rng.integers(max(1, n_cand // 4))]

            if dst == src:
                continue

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * c / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()

                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    ac = ac_trial
                    improved = True
                    stall = 0
                    temperature = max(0.001, temperature * 0.95)
                    break
                elif M_trial < best_m + temperature:
                    # Simulated annealing: accept worse with some probability
                    if rng.random() < np.exp(-(M_trial - best_m) / max(temperature, 1e-6)):
                        a = a_trial
                        ac = ac_trial
                        stall = 0
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            perturbation_count = rng.integers(2, 6)
            for _ in range(perturbation_count):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a * c / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = a * c / a.sum()
            temperature = max(0.0005, temperature * 0.9)

        if verbose and stall > 0 and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def multi_start_md(base_seq, n, time_limit, solver, n_starts=4, verbose=True):
    """Run mirror descent from multiple different initializations."""
    c = np.sqrt(2.0 * n)
    rng = np.random.default_rng(42)
    best_a = base_seq.copy()
    best_m = fast_autocorr(base_seq).max()

    # Generate diverse initializations
    inits = [base_seq.copy()]

    # Sine-like
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x) ** 2)

    # Cosine bump
    inits.append(np.cos(np.pi * (x - 0.5)) ** 4)

    # Two-bump
    inits.append(np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2))

    # Band-limited random
    mask = np.ones(n, dtype=bool)
    for _ in range(n // 4):
        mask[rng.integers(n)] = False
    rand_sig = np.abs(irfft(rfft(rng.standard_normal(n), n) * mask[0:n], n)[:n])
    inits.append(np.maximum(rand_sig, 0))

    time_per_start = time_limit / len(inits)
    for init_idx, init in enumerate(inits):
        if time.monotonic() - (t_start - time_limit) > time_limit:
            break
        a_norm = np.maximum(init, 0)
        s = a_norm.sum()
        if s < 1e-15:
            continue
        a_norm = a_norm * c / s
        a_out, m_out = mirror_descent_soft(a_norm, time_per_start,
                                            beta_start=6.0, beta_end=1.0,
                                            gamma=0.8, noise_level=0.003,
                                            noise_period=1000, verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                J = 2.0 * n * best_m / (c * c)
                print(f"      Multi-start {init_idx}: J={J:.6f}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search function: multi-phase optimization with soft subgradient, LP, and multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Soft subgradient MD with annealing at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Soft MD n={n0} ({remaining:.0f}s remaining) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_soft(a_norm, t1, beta_start=10.0, beta_end=0.3,
                                        gamma=1.2, noise_level=0.008,
                                        noise_period=1000, verbose=True)
        try_update(a_out, f"softmd_n{n0}")

    # === Phase 2: LP Nesterov at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n0} ({remaining:.0f}s remaining) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # === Phase 3: Second round of soft MD (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Soft MD round 2 n={n0} ({remaining:.0f}s remaining) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_soft(a_norm, t3, beta_start=4.0, beta_end=0.2,
                                        gamma=0.6, noise_level=0.01,
                                        noise_period=800, verbose=True)
        try_update(a_out, f"softmd2_n{n0}")

    # === Phase 4: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n0} ({remaining:.0f}s remaining) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n0}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration ({remaining:.0f}s remaining) ===")

    for n_target in [2000, 2400, 2800]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  --- Trying n={n_target} ---")
        t_alloc = min(remaining * 0.35, 200.0)
        if t_alloc < 20:
            break

        a_long = interpolate_sequence(best_seq, n_target)
        c_t = np.sqrt(2.0 * n_target)
        a_norm = a_long * c_t / a_long.sum()

        # Quick MD at new length
        t_md = t_alloc * 0.5
        t_lp = t_alloc * 0.5
        a_out, _ = mirror_descent_soft(a_norm, t_md, beta_start=6.0, beta_end=0.5,
                                        gamma=1.0, noise_level=0.005,
                                        noise_period=800, verbose=False)
        try_update(a_out, f"md_n{n_target}")

        # Quick LP at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_norm2 = best_seq_copy = a_out * c_t / a_out.sum() if a_out.sum() > 0 else a_norm
            a_out2, _ = nesterov_lp(a_norm2, min(remaining * 0.3, 120.0), solver, verbose=False)
            try_update(a_out2, f"lp_n{n_target}")

    # === Phase 6: Final LP polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP polish n={n_best} ({remaining:.0f}s remaining) ===")
    if remaining > 20:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, min(remaining * 0.5, 150.0), solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord polish n={n_best} ({remaining:.0f}s remaining) ===")
    if remaining > 15:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coordinate_polish(a_norm, min(remaining * 0.6, 100.0), verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 8: Final LP with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, min(remaining * 0.7, 80.0), solver, verbose=False)
        try_update(a_out, f"lp_last_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
