import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation (convolution with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, w, fft_size):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    grad[i] = 2 * (sum_j a[j]*w[j+i] + sum_j a[j]*w[i-j])
    where w is non-negative weights on the convolution lags.
    """
    n = len(a)
    L = fft_size
    A = rfft(a, L)
    W = rfft(w, L)
    # T1[i] = sum_j a[j] * w[j+i] (cross-correlation)
    T1 = irfft(A * np.conj(W), L)[:n]
    # T2[i] = (w*a)[i] = sum_j w[j] * a[i-j] (first n elems of full conv)
    T2 = irfft(W * A, L)[:n]
    return 2.0 * (T1 + T2)

    """Compute smooth approximation of max using log-sum-exp."""
    m = conv.max()
    if m < 1e-300:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated optimization:
1. Multi-stage LSE gradient descent with backtracking line search (primary)
2. LP best-response Frank-Wolfe with Nesterov acceleration (polish)

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad(a, w, L):
    n = len(a)
    A = rfft(a, L)
    W = rfft(w, L)
    T1 = irfft(A * np.conj(W), L)[:n]
    T2 = irfft(W * A, L)[:n]
    return 2.0 * (T1 + T2)

def lse_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    size = 2 * n - 1
    L = 1 << (3 * n - 2).bit_length()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stages = [(50.0, 10.0), (10.0, 3.0), (3.0, 0.8), (0.8, 0.15), (0.15, 0.03)]
    for si, (T_s, T_e) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 3:
            break
        T = T_s
        n_in = 0
        lr = 1.0 / (n ** 0.6)
        patience = 0
        max_pat = 80
        prev_smooth = None
        while time.monotonic() - t0 < time_limit:
            if n_in > 300:
                T = T_e
            else:
                T = T_s * (T_e / T_s) ** (n_in / 300.0)
            conv = fast_autocorr(a)
            cmax = conv.max()
            if T < 1e-5:
                w = np.zeros(size)
                w[np.argmax(conv)] = 1.0
                ws = 1.0
            else:
                w = np.exp((conv - cmax) / T)
                ws = w.sum()
                w = w / ws
            smooth_curr = cmax + T * np.log(ws) if T >= 1e-5 else cmax
            if prev_smooth is not None:
                if smooth_curr > prev_smooth + 1e-10:
                    pass
            prev_smooth = smooth_curr
            grad = compute_grad(a, w, L)
            step = lr
            a_old = a.copy()
            improved = False
            for bt in range(10):
                a_t = a_old - step * grad
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    step *= 0.5
                    continue
                a_t = a_t * norm / st
                ct = fast_autocorr(a_t)
                ct_max = ct.max()
                if T < 1e-5:
                    loss_t = ct_max
                else:
                    wt = np.exp((ct - ct_max) / T)
                    loss_t = ct_max + T * np.log(wt.sum())
                if loss_t < smooth_curr - 1e-12:
                    a = a_t
                    improved = True
                    break
                step *= 0.5
            if not improved:
                patience += 1
                if patience >= max_pat:
                    rng_k = np.random.default_rng()
                    pert = rng_k.uniform(0, 1, n) * 0.2 * norm / n
                    a = a_old + pert
                    a = np.maximum(a, 0.0)
                    a = a * norm / a.sum()
                    lr = max(lr * 3, 5.0 / n ** 0.5)
                    patience = 0
                continue
            patience = 0
            n_in += 1
            exact = a.max() * 0 + fast_autocorr(a).max()
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
            if verbose and n_in % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LSE s{si} it={n_in} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
            if T <= T_e and n_in > 300:
                break
    return best_a, best_score

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    import cvxpy as cp
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    active_set = None
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        T_mat = build_T(y)
        conv_curr = T_mat @ y
        if active_set is None or t % 50 == 0:
            thresh = M_curr * 0.95
            active_set = np.where(conv_curr > thresh)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_curr > M_curr * 0.5)[0]
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat[active_set, :] @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = None
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            active_set = None
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        g = b_val * norm / bs
        full_conv = T_mat @ g
        viol = np.where(full_conv > M_curr * (1 + 1e-6))[0]
        if len(viol) > 0 and len(active_set) < size:
            active_set = np.union1d(active_set, viol)
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        if t == 1:
            a_new = g.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa
        y = y_new
        a = a_new
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def coord_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(777)
    no_improve = 0
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        i = int(rng.integers(n))
        if a[i] < norm * 1e-9:
            continue
        conv = fast_autocorr(a)
        ai = a[i]
        g = np.zeros(size)
        lo1 = i
        hi1 = min(size, i + n)
        if hi1 > lo1:
            g[lo1:hi1] += a[:hi1 - lo1]
        lo2 = max(0, n - 1 - i)
        hi2 = min(size, n - 1 + i + 1)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            src = n - 1 - idx
            valid = (src >= 0) & (src < n)
            if valid.any():
                g[lo2:hi2][valid] += a[src[valid]]
        best_d = 0.0
        best_val = conv.max()
        for d in np.linspace(0, ai, 15):
            if d == 0:
                continue
            nm = (conv - d * g).max()
            if 2 * i < size:
                nm = max(nm, conv[2 * i] - 2 * ai * d + d * d)
            if nm < best_val - 1e-12:
                best_val = nm
                best_d = d
        if best_d > 0:
            a[i] -= best_d
            a = a * norm / a.sum()
            cv = fast_autocorr(a)
            if cv.max() < best_score:
                best_score = cv.max()
                best_a = a.copy()
                no_improve = 0
            else:
                a[i] += best_d
                a = a * norm / a.sum()
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 200:
            break
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def structured_init(n, rng):

    elif choice == 4:

        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7

        x = np.linspace(0.0, 1.0, n)

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

        try_update(a_out, f"lse_n{n_short}")

    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 7: LSE fresh init n={n_base} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 15:
        init_fresh = structured_init(n_base, rng)
        a_out, _ = lse_optimize(init_fresh, t7, verbose=True)
        try_update(a_out, f"lse_fresh_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 8: Final CD + LSE (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t8a = min(remaining * 0.4, 60.0)
    if t8a > 5:
        a_out, _ = coord_descent(best_seq, t8a, verbose=False)
        try_update(a_out, f"cd_final_n{n_best}")
    remaining = time_budget - (time.monotonic() - t_start)
    t8b = min(remaining * 0.6, 80.0)
    if t8b > 10:
        a_out, _ = lse_optimize(best_seq, t8b, verbose=False)
        try_update(a_out, f"lse_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_start)
    n3 = n_base + 800
    print(f"\n=== Phase 9: LSE n={n3} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.5, 120.0)
    if t9 > 20:
        init_3 = interp_seq(best_seq, n3)
        a_out, _ = lse_optimize(init_3, t9, verbose=True)
        try_update(a_out, f"lse_n{n3}")

    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
