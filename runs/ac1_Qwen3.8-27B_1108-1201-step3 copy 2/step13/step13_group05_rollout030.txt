"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Adaptive temperature (beta) schedule
3. LP best-response with Nesterov acceleration
4. Targeted coordinate descent polish

Key insight: uses FFT-based O(n log n) gradient computation via

import numpy as np

from scipy.fft import rfft, irfft

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def compute_lse_grad(a, c, beta):
    """Compute gradient of logsumexp(beta*conv(a,a)) using FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv_result = irfft(W * A_rev, fft_size)[:size]
    
    grad = beta * conv_result[n-1 : 2*n-1]
    return grad

def proj_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def mirror_descent_lse(a0, time_limit, beta_init=10.0, beta_max=500.0, 
                       gamma=1.0, check_every=50, verbose=True):
    """Mirror descent with logsumexp-smoothed subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: start low, increase to concentrate on peaks
        progress = min(1.0, t / 5000.0)
        beta = beta_init + (beta_max - beta_init) * progress
        # Or use a schedule: beta = beta_init * (1 + t/1000)^0.5
        beta = min(beta_max, beta_init * np.sqrt(1.0 + t / 200.0))
        
        c = fast_conv(a)
        M = c.max()
        
        grad = compute_lse_grad(a, c, beta)
        
        # Mirror descent step
        # a_new[i] ∝ a[i] * exp(-eta * grad[i])
        L = np.abs(grad).max() + 1e-10
        eta = gamma / (L * np.sqrt(t))
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Running average
        avg_a += a
        n_avg += 1
        
        if t % check_every == 0:
            # Check current
            M_cur = fast_conv(a).max()
            score_cur = 2.0 * n * M_cur
            if score_cur < best_score:
                best_score = score_cur
                best_a = a.copy()
            
            # Check average
            avg = avg_a / n_avg
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_conv(avg).max()
                score_avg = 2.0 * n * M_avg
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MD-LSE t={t} beta={beta:.1f} ({elapsed:.0f}s): "
                      f"score={best_score:.6f}, M={M:.2e}")
        
        # Reset average periodically (exponential moving average)
        if t % 2000 == 0:
            avg_a = a.copy()
            n_avg = 1
    
    return best_a, best_score

def lp_best_response(a, M, solver):
    """LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix efficiently
    # conv(a,b)_k = sum_j a_{k-j} * b_j for valid j
    # This is a Toeplitz-like matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c_y = fast_conv(y)
        M_y = c_y.max()
        
        b = lp_best_response(y, M_y, solver)
        if b is None:
            continue
        b = b / b.sum()
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            x_new = b.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        y = y_new
        x = x_new
        
        score_y = 2.0 * n * fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_x = 2.0 * n * fast_conv(x).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def coordinate_polish(a0, time_limit, verbose=True):
    """Targeted coordinate descent: move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        top_lags = np.argsort(c)[::-1][:10]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partners = k - idx
            contrib = a[idx] * a[partners]
            total_contrib = contrib.sum()
            if total_contrib < 1e-30:
                continue
            src_local = np.argmax(contrib)
            src = int(idx[src_local])
            
            # Find destination: position not in any top peak's support
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_lags:
                k2 = int(k2)
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                in_peak[js:je+1] = True
            
            if in_peak[src]:
                candidates = np.where(~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
            else:
                candidates = np.where((np.arange(n) != src) & ~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
            
            if len(candidates) == 0:
                continue
            
            # Pick candidate with lowest value
            cands_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(1, len(cands_sorted) // 4)
            dst = int(cands_sorted[rng.integers(n_try)])
            
            if dst == src:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    M = M_trial
                    score = 2.0 * n * M
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < M:
                        best_a = a.copy()
                        best_score = 2.0 * n * M_new
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord polish ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def make_initializations(n):
    """Create diverse initializations of length n."""
    inits = []
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    # Ramp
    x = np.arange(n, dtype=np.float64)
    ramp = (x + 1) / (n * (n + 1) / 2.0)
    inits.append(('ramp', ramp))
    # Inverted V (peak in middle)
    if n % 2 == 1:
        mid = n // 2
        tri = 1.0 - np.abs(x - mid) / (mid + 1)
    else:
        mid = n / 2.0
        tri = 1.0 - np.abs(x - mid) / (mid + 1)
    tri = np.maximum(tri, 0.1)
    inits.append(('triangle', tri / tri.sum()))
    # Cosine
    cos = np.cos(np.pi * x / n) ** 2
    inits.append(('cosine', cos / cos.sum()))
    # Two peaks
    peak1 = np.exp(-((x - n * 0.25) ** 2) / (2 * (n * 0.05) ** 2))
    peak2 = np.exp(-((x - n * 0.75) ** 2) / (2 * (n * 0.05) ** 2))
    two = peak1 + peak2 + 1e-6
    inits.append(('two_peaks', two / two.sum()))
    return inits

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
