1. Smoothed mirror descent with log-sum-exp gradient and temperature annealing
2. Gradient computation via FFT (O(n log n) per step)

4. Multi-length strategy: optimize at parent length, then upscale
5. Multiple initializations for robustness

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_grad(a, ac, T):
    """Compute gradient of soft-max(ac, T) w.r.t. a, using FFT."""
    n = len(a)
    # Soft-max weights
    shifted = (ac - ac.max()) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # = 2 * conv(w, a[::-1])[n-1 + m]
    b = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    B = rfft(b, fft_size)
    conv_result = irfft(W * B, fft_size)[:3*n-2]
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    return grad

def mirror_descent_smooth(a0, time_limit, T_init=1.0, T_final=0.05, 
                          lr_scale=1.0, verbose=True):
    """Smoothed mirror descent with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    n_total_steps = 50000  # target steps
    step = 0
    
    while True:
        if time.monotonic() - t0 > time_limit:
            break
        step += 1
        if step > 500000:
            break
        
        # Anneal temperature
        progress = min(step / 5000.0, 1.0)
        T = T_init * (T_final / T_init) ** progress
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute smooth gradient
        grad = compute_smooth_grad(a, ac, T)
        
        # Mirror descent step
        lr = lr_scale / (np.sqrt(step) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - lr * grad
        # Shift for numerical stability
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        # Track best
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"  MD[{step}] T={T:.4f} score={score:.6f} t={elapsed:.0f}s")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find top peak lags
        top_lags = np.argsort(ac)[::-1][:min(16, len(ac))]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break

            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Destination: position that minimizes impact on all top lags
            # Score each position by its total contribution to top lags

def fast_autocorr(a):
    """Compute linear autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, T):
    """
    Compute gradient of soft-max(conv(a,a), T) w.r.t. a using FFT.
    Uses: grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    where w = normalized softmax of the autocorrelation.
    """
    n = len(a)
    ac = fast_autocorr(a)
    shifted = (ac - ac.max()) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    # grad[m] = 2 * conv(w, a[::-1])[n-1+m]
    b = a[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    W = rfft(w, fft_size)
    B = rfft(b, fft_size)
    conv_result = irfft(W * B, fft_size)[:3 * n - 2]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad, ac


def compute_hard_grad(a, top_k=5):
    """Compute subgradient using top-k peak lags (averaged)."""
    n = len(a)
    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals / top_vals.sum()

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        i_start = max(0, int(k) - n + 1)
        i_end = min(n - 1, int(k))
        idx = np.arange(i_start, i_end + 1)
        j = int(k) - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad, ac

                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall_count = 0

    if verbose:
        elapsed = time.monotonic() - t0
        score = 2.0 * n * best_m
        print(f"    CD done ({elapsed:.0f}s): score={score:.6f}")

    return best_a, best_m


def pair_descent(a0, time_limit, verbose=True, seed=456):
    """Optimize pairs of coordinates simultaneously for better progress."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:8]
        improved = False

        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            k = int(top_lags[rng.integers(len(top_lags))])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            partners = k - idx
            contrib = a[idx] * a[partners]

            src = int(idx[np.argmax(contrib)])
            impact = np.zeros(n)
            for k2 in top_lags:
                k2 = int(k2)
                i2s = max(0, k2 - n + 1)
                i2e = min(n - 1, k2)
                idx2 = np.arange(i2s, i2e + 1)
                p2 = k2 - idx2
                impact[idx2] += a[p2]
            impact[src] = 1e10
            candidates = np.where(impact < 1e9)[0]
            if len(candidates) == 0:
                candidates = np.array([i for i in range(n) if i != src])
            if len(candidates) == 0:
                continue
            dst = int(candidates[rng.integers(len(candidates))])
            if src == dst or a[src] < 1e-15:
                continue

            for frac in [0.01, 0.03, 0.07, 0.15]:
                delta = a[src] * frac
                if delta < 1e-18:
                    break
                a_t = a.copy()
                a_t[src] -= delta
                a_t[dst] += delta
                if a_t[src] < -1e-18:
                    continue
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m - 1e-15:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                delta = a[i] * rng.uniform(0.05, 0.2)
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    if fast_autocorr(a).max() < best_m:
                        best_m = fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PairCD ({elapsed:.0f}s): score={2.0*n*best_m:.6f}")
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search function: multi-phase, multi-length optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    rng = np.random.default_rng(42)

    # Track best
    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)
    print(f"Starting optimization with n={n0}, score={best_score:.6f}")

    # Phase 1: Smoothed mirror descent with high->low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth MD (anneal) n={n0} ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_out, _ = mirror_descent_smooth(best_seq, t1, T_init=1.0, T_final=0.05,
                                         lr_scale=0.8, anneal_steps=2000, verbose=True)
        do_update(a_out, f"smooth_md1_n{n0}")

    # Phase 2: Hard subgradient mirror descent (focused on peaks)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Hard MD n={n_cur} ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 20:
        a_out, _ = mirror_descent_hard(best_seq, t2, top_k=3, lr_scale=0.6, verbose=True)
        do_update(a_out, f"hard_md_n{n_cur}")

    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Coord Descent n={n_cur} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(best_seq, t3, verbose=True, seed=123)
        do_update(a_out, f"cd_n{n_cur}")

    # Phase 4: Smooth MD round 2 (lower temperature, finer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Smooth MD round 2 n={n_cur} ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 20:
        a_out, _ = mirror_descent_smooth(best_seq, t4, T_init=0.3, T_final=0.01,
                                         lr_scale=0.4, anneal_steps=3000, verbose=True)
        do_update(a_out, f"smooth_md2_n{n_cur}")

    # Phase 5: Scale to n=2400
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Scale to n={n1} ===")
    t5 = min(remaining * 0.18, 280.0)
    if t5 > 40:
        a_long = interpolate_seq(best_seq, n1)
        a_out, _ = mirror_descent_smooth(a_long, t5 * 0.55, T_init=0.3, T_final=0.02,
                                         lr_scale=0.5, anneal_steps=2000, verbose=True)
        do_update(a_out, f"smooth_md_n{n1}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = coordinate_descent(best_seq if len(best_seq) != n1 else a_out,
                                           min(remaining * 0.35, 100.0), verbose=True, seed=789)
            do_update(a_out2, f"cd_n{n1}")

    # Phase 6: Scale to n=3200
    n2 = 3200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Scale to n={n2} ===")
    t6 = min(remaining * 0.18, 280.0)
    if t6 > 40:
        a_long2 = interpolate_seq(best_seq, n2)
        a_out, _ = mirror_descent_smooth(a_long2, t6 * 0.55, T_init=0.2, T_final=0.01,
                                         lr_scale=0.4, anneal_steps=2000, verbose=True)
        do_update(a_out, f"smooth_md_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            a_out2, _ = coordinate_descent(a_out if len(a_out) == n2 else best_seq,
                                           min(remaining * 0.4, 100.0), verbose=True, seed=321)
            do_update(a_out2, f"cd_n{n2}")

    # Phase 7: Final coordinate descent at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} ===")
    t7 = min(remaining * 0.35, 200.0)
    if t7 > 15:
        a_out, _ = coordinate_descent(best_seq, t7 * 0.5, verbose=True, seed=999)
        do_update(a_out, f"cd_final_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_descent(best_seq, min(remaining * 0.5, 100.0), verbose=True, seed=555)
            do_update(a_out, f"pair_final_n{n_best}")

    # Phase 8: Last chance smooth MD with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final smooth MD n={n_best} ===")
    t8 = min(remaining * 0.6, 150.0)
    if t8 > 10:
        a_out, _ = mirror_descent_smooth(best_seq, t8, T_init=0.1, T_final=0.005,
                                         lr_scale=0.2, anneal_steps=5000, verbose=False)
        do_update(a_out, f"smooth_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
