"""

2. LSE (log-sum-exp) gradient descent with Nesterov for warm-up
3. Peak-targeting coordinate descent for final polish

5. Multi-length exploration with interpolation transfer
Key improvements over previous: dual LP formulation for faster solves,

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.sparse import lil_matrix, csr_matrix, dia_matrix

_global_solver = None


def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    _global_solver = cp.ECOS

def fast_conv(a, b=None):
    """Compute full convolution of a with b (or a with a if b is None) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    return fast_conv(a, a)


def cross_corr_fft(a, y):
    """Compute cross-correlation: v[i] = sum_j a[j]*y[j+i] for i=0..len(a)-1."""
    n = len(a)
    ny = len(y)
    size = n + ny - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    Y = rfft(y, fft_size)
    cc = irfft(np.conj(A) * Y, fft_size)
    return cc[:n]


def lse_gradient(a, T):
    """Compute gradient of T*log(sum exp(b_k/T)) where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a, a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    # Gradient: dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k]) for valid indices
    # = (rev_a * w)[m+n-1] + (a * w)[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2


def nesterov_lse(a0, time_limit, T_start=10.0, T_end=0.2, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        n_iters = max(30, int(remaining * 4.0 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if it == 0:
                y_new = y - (1.0 / (it + 3.0)) * lse_gradient(y, T)
            else:
                mom = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + mom * (y - y_prev)
                y_new = y_extrap - (1.0 / (it + 3.0)) * lse_gradient(y_extrap, T)
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1
            score = fast_conv(y, y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            if verbose and it_count % 200 == 0:
                print(f"    LSE s={si}/{n_stages} T={T:.3f} it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def build_T_matrix(a):
    """Build the convolution constraint matrix T where T[k,i] = a[k-i] if valid."""
    n = len(a)
    T = lil_matrix((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = a[:]
    return T.tocsr()


def lp_best_response(a, M, solver, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_topk(a, M, solver, k_top=200, verbose=False):
    """Solve LP with only top-k active constraints for speed."""
    n = len(a)
    b = fast_conv(a, a)
    top_k = np.argsort(b)[::-1][:k_top]
    T = build_T_matrix(a)
    T_reduced = T[top_k, :]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_reduced @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, use_topk=False, k_top=300, verbose=True):

    n = len(a0)

    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_acc = np.zeros(n)
    n_acc = 0
    n_iters = 0
    prev_score = best_score
    stall_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        if M_curr < 1e-12:

        if use_topk:

        else:
            b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()

        else:

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_conv(y, y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_conv(a, a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        a_acc += a
        n_acc += 1
        if n_acc % 30 == 0:
            avg = a_acc / n_acc
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg * norm / s_avg
                score_avg = fast_conv(avg, avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
                    stall_count = 0

        if verbose and n_iters % 50 == 0:
            print(f"    LP-Nes iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} stall={stall_count}")

        if stall_count > 500:
            break

    # Return best of y, a, and average
    score_y = fast_conv(y, y).max()
    if score_y < best_score:
        best_score = score_y
        best_a = y.copy()

    if n_acc > 0:
        avg = (a_acc / n_acc) * norm / (a_acc / n_acc).sum()
        score_avg = fast_conv(avg, avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg.copy()

def dual_subgradient(a, M, time_limit, verbose=True):
    """
    Solve the dual of the LP best-response problem using projected subgradient.
    Dual: min M*sum(y) + sum_i max(0, 1 - (T^T y)[i]) s.t. y >= 0
    Returns the optimal primal b recovered from complementary slackness.
    """
    n = len(a)
    m = 2 * n - 1
    t0 = time.monotonic()

    # Initialize y
    y = np.ones(m) * 0.01 / m

    # Step size schedule
    base_step = 1.0 / (n * 0.5)

    best_obj = np.inf
    best_y = y.copy()
    n_iters = 0

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute v = T^T y (cross-correlation of a and y)
        v = cross_corr_fft(a, y[:m])[:n]

        # Compute mask
        mask = (v < 1.0).astype(np.float64)

        # Compute gradient: grad[k] = M - (a * mask)[k]
        c = fast_conv(a, mask)  # length 2n-1 = m
        grad = M * np.ones(m) - c

        # Subgradient step
        step = base_step / (np.sqrt(it + 1))
        y_new = y - step * grad
        y = np.maximum(y_new, 0.0)
        n_iters += 1

        # Compute objective
        v_new = cross_corr_fft(a, y[:m])[:n]
        obj = M * y.sum() + np.sum(np.maximum(0.0, 1.0 - v_new))
        if obj < best_obj:
            best_obj = obj
            best_y = y.copy()

        if verbose and n_iters % 200 == 0:
            print(f"    Dual-Subgrad it={n_iters} ({time.monotonic()-t0:.0f}s): obj={obj:.6f}")

    # Recover primal from dual: b[i] = 1 if (T^T y)[i] < 1 (active constraint)
    # More precisely, use complementary slackness
    v_final = cross_corr_fft(a, best_y[:m])[:n]
    b = np.zeros(n)
    # Where v[i] < 1, the dual constraint is inactive, so b[i] > 0 is possible
    # Actually, the KKT condition is: b[i] * (T^T y + z - 1)[i] = 0
    # At optimum, z[i] = max(0, 1 - (T^T y)[i])
    # So b[i] > 0 only when T^T y[i] = 1 (complementary slackness)
    # This is tricky to recover exactly. Let's just solve a small LP for the active set.
    active = np.where(v_final >= 0.999)[0]
    if len(active) == 0:
        active = np.where(v_final >= v_final.max() - 1e-6)[0]

    if len(active) > 0 and len(active) < n:
        # Solve restricted LP
        T = build_T_matrix(a)
        b_active = cp.Variable(len(active), nonneg=True)
        T_sub = T[:, active]
        prob = cp.Problem(cp.Maximize(cp.sum(b_active)), [T_sub @ b_active <= M, cp.sum(b_active) <= np.sum(np.ones(n))])
        try:
            prob.solve(solver=get_solver(), verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b[active] = np.maximum(np.asarray(b_active.value, dtype=np.float64), 0)
        except Exception:
            pass

    if b.sum() < 1e-15:
        return None
    return b


def peak_targeting(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search with multiple move types."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a, a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_star - idx]
            sorted_idx = idx[np.argsort(contrib)[::-1]]

            for src in sorted_idx[:4]:
                if a[src] < norm * 1e-9:
                    continue

                # Find safe destination positions
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.setdiff1d(np.arange(n), [src])

                if len(candidates) == 0:
                    continue

                # Sort by current value (move to low-value positions)
                candidates = candidates[np.argsort(a[candidates])[:min(5, len(candidates))]]

                for dst in candidates:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial, a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 50 == 0:
                if rng.integers(5) == 0:
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_conv(a_trial, a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0

            if stall > 300:
                break

        if verbose and it_count % 200 == 0:
            print(f"    PeakTg it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} stall={stall}")

    return best_a, best_score


def two_swap_search(a0, time_limit, seed=7, verbose=True):
    """Try pairs of swaps to reduce the max convolution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a, a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a, a)
        peak_k = np.argmax(b)
        peak_val = b[peak_k]

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[peak_k - idx]
        top_srcs = idx[np.argsort(contrib)[::-1][:5]]

        # Find bottom contributors (positions with low total contribution to any peak)
        total_contrib = np.zeros(n)
        for k2 in np.argsort(b)[::-1][:5]:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            if len(idx2) > 0:
                total_contrib[idx2] += a[idx2] * a[k2 - idx2]

        low_positions = np.argsort(total_contrib)[:10]

        improved = False
        for src in top_srcs:
            if a[src] < norm * 1e-9:
                continue
            for dst in low_positions:
                if dst == src:
                    continue
                # Try moving small amount
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial, a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it += 1
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0 and a[j] > 0:
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
                a_trial = a_trial * norm / a_trial.sum()
                score = fast_conv(a_trial, a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    it += 1

        if verbose and it % 500 == 0:
            print(f"    TwoSwap it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

        if it > 50000:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(5.0 * x) + 0.05 * np.cos(11.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        # Chirp-like
        x = np.arange(n) / n
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * 3 + 0.5 * x * n)
    elif choice == 6:
        # Multi-scale
        a = np.ones(n) * 0.3
        for _ in range(rng.integers(4, 10)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.03, n * 0.2)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    else:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Solver: {solver}, Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # ============ Phase 1: LSE warm-up ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE warm-up n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=8.0, T_end=0.3, n_stages=10)
        do_try_update(a_out, f"lse_warmup_n{n_base}")

    # ============ Phase 2: Main LP Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 600.0)
    if t2 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t2, solver, use_topk=False, verbose=True)
        do_try_update(a_out, f"lp_main_n{n_best}")

    # ============ Phase 3: Peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: PeakTargeting n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeting(a_work, t3, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # ============ Phase 4: LP polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t4, solver, use_topk=True, k_top=200, verbose=True)
        do_try_update(a_out, f"lp_polish_n{n_best}")

    # ============ Phase 5: LSE at longer length ============
    n_long = n_best + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE longer n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 100.0)
    if t5 > 10:
        init_long = interpolate_sequence(best_seq, n_long)
        a_work = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lse(a_work, t5, T_start=10.0, T_end=0.2, n_stages=8)
        do_try_update(a_out, f"lse_long_n{n_long}")

    # ============ Phase 6: Final peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final PeakTargeting n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeting(a_work, t6, seed=99999, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_best}")

    # ============ Phase 7: Two-swap search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: TwoSwap n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 5:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = two_swap_search(a_work, t7, seed=777, verbose=True)
        do_try_update(a_out, f"twoswap_n{n_best}")

    # ============ Phase 8: Final LP with topk ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.8, 120.0)
        print(f"\n=== Phase 8: Final LP-topk n={n_best} (budget={t8:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t8, solver, use_topk=True, k_top=150, verbose=True)
        do_try_update(a_out, f"lp_topk_final_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
