def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad(a, w):
    """
    Compute gradient of sum_k w_k * conv(a,a)_k with respect to a.
    grad_j = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    = 2 * conv(a_rev, w)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    # conv length: n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    AR = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(AR * W, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)_k = conv(a,b)_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """LP: maximize sum(b) subject to T@a <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        y = y_new
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def adam_lse_optimize(a0, time_limit, verbose=True):
    """
    Adam-based LSE smooth gradient descent for minimizing max autocorrelation.
    Normalizes a to sum=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01

    # Temperature schedule parameters
    T_start = max(0.01, best_score * 0.1)
    T_end = max(1e-4, best_score * 0.003)
    total_time_target = time_limit

    # Track for adaptive LR
    prev_score = best_score
    no_improve = 0
    lr_patience = 500

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / total_time_target
        progress = min(1.0, elapsed_frac)
        # Cosine annealing for temperature
        T = T_start + (T_end - T_start) * 0.5 * (1 + np.cos(np.pi * progress))
        # Learning rate decay
        lr_curr = lr * (1.0 - 0.7 * progress)

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # LSE weights (softmax of autocorrelation / T)
        # Use a stable computation
        shifted = (b - max_b) / T
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            nmid = n - 1
            w[nmid] = 1.0
        else:
            w = w / w_sum

        # Compute gradient
        grad = compute_grad(a, w)

        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        a_new = a - lr_curr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        score = fast_autocorr(a).max()

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive LR: reduce if stuck
        if no_improve > lr_patience:
            lr *= 0.7
            no_improve = 0
            if verbose and steps % 2000 == 0:
                print(f"      LR reduced to {lr:.6f}")

        # Periodic small perturbation to escape local minima
        if no_improve > lr_patience * 3:
            rng = np.random.default_rng(steps * 13 + 7)
            n_pert = max(1, n // 100)
            idx = rng.choice(n, size=n_pert, replace=False)
            a = a.copy()
            for i in idx:
                j = rng.integers(n)
                if i != j:
                    amt = a[i] * 0.1
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0

        if verbose and steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE step={steps} ({elapsed:.0f}s): T={T:.5f}, lr={lr_curr:.5f}, best={best_score:.8f}")

    return best_a, best_score


def lse_gd_simple(a0, time_limit, lr_init=0.005, verbose=True):
    """
    Simple gradient descent with LSE, adaptive step size via backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.005, best_score * 0.05)
    T_end = max(1e-4, best_score * 0.002)
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start + (T_end - T_start) * 0.5 * (1 + np.cos(np.pi * progress))

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        shifted = (b - max_b) / T
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            continue
        w = w / w_sum

        grad = compute_grad(a, w)
        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-15:
            break

        step = lr / grad_norm
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        steps += 1
        score = fast_autocorr(a).max()

        if score < best_score:
            best_score = score
            best_a = a.copy()
            if lr < 0.02:
                lr *= 1.02
        else:
            if steps % 100 == 0:
                lr *= 0.9

    return best_a, best_score


def stochastic_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Stochastic peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_start = max(0.1, best_score * 0.05)
    T_end = best_score * 0.001

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        current_max = b.max()

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Find good destinations: positions with low values
            non_src = np.where(np.arange(n) != src)[0]
            n_dest = min(3, len(non_src))
            # Pick from lowest-valued positions
            sorted_dst = non_src[np.argsort(a[non_src])][:min(n_dest * 4, len(non_src))]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]

            frac = rng.choice([0.01, 0.02, 0.05, 0.1, 0.15])
            amt = a[src] * frac
            if amt < 1e-18:
                continue

            dst_vals = a[selected_dst] + 1e-12
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()
            amounts = amt * weights

            a_trial = a.copy()
            a_trial[src] -= amt
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amounts[di]

            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

            score = fast_autocorr(a_trial).max()

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif score < current_max and rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            n_stall += 1
            # Random swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 150:
                # Larger random perturbation
                idx_p = rng.choice(n, size=max(1, n // 80), replace=False)
                a_trial = a.copy()
                for idx in idx_p:
                    amt = a[idx] * rng.uniform(0.05, 0.2)
                    j2 = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j2] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 3000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and lengths to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # --- Initialization ---
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Brief LP at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Brief LP n={n_base} ===")
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 15:
        # Determine how many LP iterations to do
        n_lp_iters = max(10, min(40, int(t1 / 3)))
        a_out, _ = lp_nesterov_brief(best_seq, n_lp_iters, solver)
        do_try_update(a_out, f"lp_init_n{n_base}")

    # === Phase 2: Adam LSE at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam-LSE n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_out, _ = adam_lse_optimize(best_seq, t2, verbose=True)
        do_try_update(a_out, f"adam_n{n_cur}")

    # === Phase 3: Simple LSE-GD (different dynamics) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-GD n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_out, _ = lse_gd_simple(best_seq, t3, lr_init=0.003, verbose=False)
        do_try_update(a_out, f"gds_n{n_cur}")

    # === Phase 4: Peak shaving at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: PeakShave n={len(best_seq)} ===")
    t4 = min(remaining * 0.08, 90.0)
    if t4 > 10:
        n_cur = len(best_seq)
        a_out, _ = stochastic_peak_shave(best_seq, t4, seed=99, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # === Phase 5: Try longer lengths ===
    for n_try in [n_base + 200, n_base + 400, n_base + 600]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n=== Phase 5: Length exploration n={n_try} ===")
        init_long = interpolate_sequence(best_seq, n_try)

        # Brief LP
        t_lp = min(remaining * 0.15, 80.0)
        if t_lp > 10:
            n_lp_iters = max(5, min(20, int(t_lp / 3)))
            a_out, _ = lp_nesterov_brief(init_long, n_lp_iters, solver)
            do_try_update(a_out, f"lp_n{n_try}")

        # Adam LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_adam = min(remaining * 0.30, 250.0)
        if t_adam > 20:
            a_out, _ = adam_lse_optimize(best_seq, t_adam, verbose=True)
            do_try_update(a_out, f"adam_n{n_try}")

    # === Phase 6: Final Adam LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Adam-LSE n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 250.0)
    if t6 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t6, verbose=True)
        do_try_update(a_out, f"adam_final_n{n_best}")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PeakShave n={n_best} ===")
    t7 = min(remaining * 0.20, 100.0)
    if t7 > 10:
        a_out, _ = stochastic_peak_shave(best_seq, t7, seed=555, verbose=True)
        do_try_update(a_out, f"ps_final_n{n_best}")

    # === Phase 8: Quick LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    t8 = min(remaining * 0.25, 120.0)
    if t8 > 10:
        n_lp_iters = max(5, min(30, int(t8 / 3)))
        a_out, _ = lp_nesterov_brief(best_seq, n_lp_iters, solver)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: One more Adam pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final Adam n={n_best} ({remaining:.0f}s) ===")
        a_out, _ = adam_lse_optimize(best_seq, remaining, verbose=True)
        do_try_update(a_out, f"adam_final2_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Return a fallback
        n_fb = 1600
        x = np.linspace(0, 1, n_fb)
        fb = np.sin(np.pi * x) ** 1.2
        fb = fb / fb.sum()
        return [float(x) for x in fb]

    return [float(x) for x in best_seq]
