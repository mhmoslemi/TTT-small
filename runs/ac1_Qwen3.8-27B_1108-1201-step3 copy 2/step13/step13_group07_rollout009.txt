def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) * (target_sum / n)
    # Efficient simplex projection
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) * (target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s2 = result.sum()
        if s2 > 0:
            result = result / s2 * target_sum
    return result


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce convolution peaks."""
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        if non_dc.max() < 1e-30:
            break
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = target / (non_dc + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    T_matrix = build_conv_matrix(a)
    T_matrix_cache = {}

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Rebuild T matrix periodically (a changes)
        if t % 50 == 0 or t == 1:
            T_matrix = build_conv_matrix(y)

        b_star, M_val = solve_lp_best_response(y, solver, T_matrix)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score - 1e-12:
            stall_count = 0
            best_score = score_a
            best_a = a.copy()
        else:
            stall_count += 1

        score_y = internal_score(y)
        if score_y < best_score - 1e-12:
            stall_count = 0
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score - 1e-12:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > 3000:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_advanced(a0, time_limit, seed=42):
    """Advanced peak shaving with line search and smart candidate selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    best_consecutive_fail = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        top_k = np.argsort(b)[::-1][:min(20, len(b))]

        improved = False

        # Strategy 1: Source from peak contributors, destination to low-impact locations
        for k in top_k[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Find the index that contributes most to this peak
            if contrib.sum() < 1e-30:
                continue
            weights = contrib / contrib.sum()
            # Sample source indices weighted by contribution
            n_samples = min(5, len(idx))
            src_candidates = []
            for _ in range(n_samples):
                src_local = rng.choice(len(idx), p=weights)
                src_candidates.append(idx[src_local])

            for src in src_candidates:
                if a[src] < 1e-15:
                    continue

                # Find good destination: minimize impact on peaks
                # Compute "impact score" for each potential destination
                peak_set = set(int(kk) for kk in top_k[:10])
                best_dst = -1
                best_impact = float('inf')
                worst_impact = 0

                # Sample destinations
                n_dst_samples = min(30, n // 10)
                dst_samples = rng.choice(n, size=n_dst_samples, replace=False)
                for dst in dst_samples:
                    if dst == src:
                        continue
                    # Compute change in max convolution from moving eps from src to dst
                    # First order: delta_k = -eps*a[k-src] + eps*a[k-dst]
                    # We want this to be <= 0 for all peak k
                    max_increase = 0
                    for kk in peak_set:
                        dec = a[max(0, min(n-1, kk - src))] if 0 <= kk - src < n else 0
                        inc = a[max(0, min(n-1, kk - dst))] if 0 <= kk - dst < n else 0
                        delta = -dec + inc
                        max_increase = max(max_increase, delta)
                    if max_increase < worst_impact or (best_dst == -1 and max_increase < best_impact):
                        if max_increase < best_impact:
                            best_impact = max_increase
                            best_dst = dst

                if best_dst == -1:
                    continue

                # Binary search on transfer amount
                lo = 0.0
                hi = a[src] * 0.9
                for _ in range(12):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    if a_trial[src] < 0:
                        hi = mid
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = internal_score(a_trial)
                    if sc < best_score:
                        lo = mid
                    else:
                        hi = mid

                if lo > 1e-15:
                    a_trial = a.copy()
                    a_trial[src] -= lo
                    a_trial[dst] += lo
                    a_trial = a_trial / a_trial.sum()
                    sc = internal_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved:
            stall += 1
            best_consecutive_fail += 1

            # Random perturbation to escape local minima
            if stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.001, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = internal_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            best_consecutive_fail = 0
                        elif sc < best_score + 0.001:
                            # Accept slightly worse moves (annealing)
                            a = a_trial
                            stall = 0

            if best_consecutive_fail > 2000:
                break
        else:
            best_consecutive_fail = 0

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient optimization with adaptive smoothing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.05
    t = 0
    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Adaptive smoothing temperature
        T_smooth = max(0.5, 30.0 * np.exp(-t / 500.0))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Compute gradient via FFT
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        lr_t = base_lr / (1 + 0.005 * t)
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def pairwise_optimize(a0, time_limit, seed=99):
    """Jointly optimize pairs of indices that co-occur in peak-producing terms."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        top_k = np.argsort(b)[::-1][:min(5, len(b))]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.sum() < 1e-30 or len(idx) < 2:
                continue

            # Find the two largest contributors
            top2_local = np.argsort(contrib)[::-1][:2]
            if len(top2_local) < 2:
                continue
            i1 = idx[top2_local[0]]
            i2 = idx[top2_local[1]]

            # Try to reduce both by moving mass to a third location
            total_to_move = (a[i1] + a[i2]) * 0.05
            candidates = [j for j in range(n) if j != i1 and j != i2 and a[j] < np.median(a)]
            if not candidates:
                continue
            dst = candidates[rng.integers(len(candidates))]

            # Try different splits
            for ratio in [0.3, 0.5, 0.7]:
                amt1 = total_to_move * ratio
                amt2 = total_to_move * (1 - ratio)
                if amt1 > a[i1] or amt2 > a[i2]:
                    continue
                a_trial = a.copy()
                a_trial[i1] -= amt1
                a_trial[i2] -= amt2
                a_trial[dst] += amt1 + amt2
                if a_trial[i1] < 0 or a_trial[i2] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = internal_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

    return best_a, best_score
