```python
def proj_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cv - c))
    theta = (cv[rho] - c) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def sparse_lp_refine(a, M, solver, max_iters=20):
    n = len(a)
    size = 2 * n - 1
    c = np.sqrt(2.0 * n)
    
    for it in range(max_iters):
        # Solve LP: max sum(b) s.t. conv_matrix(a) @ b <= M, b >= 0
        # Build sparse constraint matrix
        rows, cols, vals = [], [], []
        for k in range(size):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
        
        T = coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()
        
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(b_var.value, 0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        
        g = b_val * c / b_sum
        a = 0.5 * a + 0.5 * g
        a = np.maximum(a, 0)
        a = a * c / a.sum()
        M = fast_autocorr(a).max()
    
    return a
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA on LSE (log-sum-exp) smooth approximation with FFT-based gradient - O(n log n) per step
2. Nesterov-accelerated LP best-response with sparse constraint matrix
3. Pair coordinate descent for final polishing
4. Multi-length, multi-restart strategy with adaptive time allocation
Key innovations: FFT-based gradient for LSE (much faster than LP per iteration),
sparse LP construction, and aggressive multi-restart with diverse initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import coo_matrix, csr_matrix
import cvxpy as cp

_solver_cache = None

def get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.SCS
    print(f"  Fallback LP solver: SCS")
    return _solver_cache


def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution a*a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - c)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, c / n)
    theta = (cv[rho[0]] - c) / (rho[0] + 1)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(w, a):
    """
    Compute gradient of LSE objective w.r.t. a.
    w: softmax weights over lags (length 2n-1)
    a: sequence (length n)
    Returns: gradient of length n.
    
    grad[i] = sum_k w[k]*(a[k-i] + a[i-k])
            = cross_corr[w,a][i] + conv[w,a][i]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()
    
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    
    # conv_wa[i] = sum_j w[j] * a[i-j] = (w * a)[i]
    conv_wa = irfft(W * A, fft_size)
    
    # cross[i] = sum_j w[j] * a[(j-i) mod N] = sum_j w[j+i] * a[j] (for linear, with padding)
    cross = irfft(W * np.conj(A), fft_size)
    
    grad = cross[:n] + conv_wa[:n]
    return grad


def fista_lse(a0, time_limit, tau_start=5.0, tau_end=0.5, verbose=True):
    """FISTA on LSE smooth approximation of max autocorrelation."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    t = 1.0
    tau = tau_start
    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, int(time_limit * 5)))
    
    n_iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_y = fast_autocorr(y)
        max_cv = conv_y.max()
        if max_cv < 1e-12:
            break
        
        # LSE weights
        shifted = conv_y - max_cv
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv_y))
            w[np.argmax(conv_y)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient via FFT
        grad = compute_lse_grad(w, y)
        
        # Lipschitz estimate
        L = 2.0 * np.sum(y * y) / tau
        if L < 1e-10:
            L = 1.0
        step = 1.0 / L
        
        # Cap step to prevent instability
        max_step = c / (n * 0.1)
        step = min(step, max_step)
        
        # FISTA update
        z = y - step * grad
        x_new = proj_simplex(z, c)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + (t - 1.0) / t_new * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new * c / s_y
        
        a = x_new
        y = y_new
        t = t_new
        tau = max(tau * tau_decay, tau_end)
        n_iters += 1
        
        # Check improvement
        score = fast_autocorr(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA LSE iter={n_iters} (t={elapsed:.0f}s): best={best_score:.8f}, tau={tau:.3f}")
        
        if no_improve > 2000:
            break
    
    return best_a, best_score


def build_sparse_conv_indices(n):
    """Precompute row/col index pattern for convolution matrix (independent of a)."""
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    k_idx = 0
    for k in range(size):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        nnz = i_hi - i_lo + 1
        rows[k_idx:k_idx + nnz] = k
        cols[k_idx:k_idx + nnz] = np.arange(i_lo, i_hi + 1)
        k_idx += nnz
    return rows[:k_idx], cols[:k_idx]


def fill_conv_matrix(a, rows, cols):
    """Fill values into precomputed sparse structure. T[k,i] = a[k-i]."""
    idx = cols - np.repeat(np.arange(2 * len(a) - 1, dtype=np.int64), 
                           np.diff(np.concatenate([[0], 
                           np.array([min(len(a)-1, k) - max(0, k-len(a)+1) + 1) for k in range(2*len(a)-1)]])))]
    return a[idx.astype(int)]


def build_conv_matrix(a):
    """Build sparse convolution matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        nnz = i_hi - i_lo + 1
        rows.extend([k] * nnz)
        cols.extend(range(i_lo, i_hi + 1))
        vals.extend(a[k - np.arange(i_lo, i_hi + 1)].tolist())
    return coo_matrix((vals, (rows, cols)), shape=(size, n)).tocsr()


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    t = 1.0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-10:
            break
        
        # Solve LP: max sum(b) s.t. T_a @ b <= M, b >= 0
        T = build_conv_matrix(y)
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            prob.solve(solver=cp.SCS, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Fallback: use gradient step
            idx_max = np.argmax(conv_y)
            grad = np.zeros(n)
            for i in range(n):
                j1 = idx_max - i
                if 0 <= j1 < n:
                    grad[i] += y[j1]
                j2 = i - idx_max
                if 0 <= j2 < n:
                    grad[i] += y[j2]
            grad *= 2.0
            i_min = np.argmin(grad)
            g = np.zeros(n)
            g[i_min] = c
        else:
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum < 1e-15:
                continue
            g = b_val * c / b_sum
        
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * c / s_a
        
        y = y_new
        a = a_new
        
        # Check both y and a
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and n_iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} (t={elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent: transfer mass between pairs to reduce peak."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * c / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    n_samples = 16
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < c * 1e-7:
                continue
            
            # Try 5 random partners
            for _ in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Compute gain vector for transferring delta from i to j
                g = np.zeros(size)
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    src = np.arange(lo1, hi1) - i
                    g[lo1:hi1] += a[src]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    src2 = (n - 1) - np.arange(lo2, hi2)
                    valid = (src2 >= 0) & (src2 < n)
                    g[lo2:hi2][valid] += a[src2[valid]]
                
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    src3 = np.arange(lo3, hi3) - j
                    g[lo3:hi3] -= a[src3]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    src4 = (n - 1) - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    g[lo4:hi4][valid4] -= a[src4[valid4]]
                
                # Find best delta
                deltas = np.linspace(0.0, a[i], n_samples + 1)[1:]
                best_d = 0.0
                best_val = conv.max()
                
                for d in deltas:
                    new_max = (conv + d * g).max()
                    if 2 * i < size:
                        val_2i = conv[2 * i] - 2 * a[i] * d + d * d
                        new_max = max(new_max, val_2i)
                    if 2 * j < size:
                        val_2j = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = max(new_max, val_2j)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * c / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * c / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} (t={elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        # Two peaks
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
    elif choice == 7:
        # Power law
        x = np.linspace(0.01, 1.0, n)
        a = x ** 0.3
    elif choice == 8:
        # Oscillating
        x = np.linspace(0.0, 1.0, n)
        a = 0.6 + 0.4 * np.cos(4 * np.pi * x) ** 2
    else:
        # Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-15.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def lse_grad_optimize(a0, time_limit, verbose=True):
    """Plain gradient descent on LSE (no Nesterov) - sometimes more stable."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = 5.0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        max_cv = conv.max()
        if max_cv < 1e-12:
            break
        
        shifted = conv - max_cv
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum
        
        grad = compute_lse_grad(w, a)
        
        # Adaptive step size
        L = 2.0 * np.sum(a * a) / tau
        if L < 1e-10:
            L = 1.0
        step = 0.5 / L
        step = min(step, c / (n * 0.05))
        
        a_new = a - step * grad
        a_new = proj_simplex(a_new, c)
        
        # Check if it improved, if not try smaller step
        score_new = fast_autocorr(a_new).max()
        if score_new > best_score * 1.01:
            # Revert
            a_new = a.copy()
            tau *= 0.9  # Reduce temperature
        else:
            a = a_new
        
        tau = max(tau * 0.9995, 0.3)
        n_iters += 1
        
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-grad iter={n_iters} (t={elapsed:.0f}s): best={best_score:.8f}, tau={tau:.3f}")
    
    return best_a, best_score
