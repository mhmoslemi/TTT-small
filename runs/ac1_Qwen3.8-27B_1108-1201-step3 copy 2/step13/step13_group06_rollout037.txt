def fast_autocorr(a, fft_size):
    """Full convolution a*a via FFT."""
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:len(a) * 2 - 1]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_score_normalized(a, n, fft_size):
    """Compute score assuming sum(a)=1."""
    b = fast_autocorr(a, fft_size)
    return 2.0 * n * b.max()

def normalize_seq(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def topk_lse_grad(a, b, T, topk, n):
    """Compute top-k LSE objective value and gradient w.r.t. a."""
    k = min(topk, len(b))
    top_idx = np.argpartition(b, -k)[-k:]
    top_vals = b[top_idx]
    
    max_tv = top_vals.max()
    shifted = (top_vals - max_tv) / T
    exp_shifted = np.exp(shifted)
    lse_val = max_tv + T * np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()
    
    grad_a = np.zeros(n)
    for ii in range(k):
        K = top_idx[ii]
        wK = w[ii]
        js = max(0, K - n + 1)
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    return lse_val, grad_a

def full_lse_grad_fft(a, b, T, n, conv_fft_size):
    """Compute full LSE gradient using FFT-based cross-correlation."""
    max_c = b.max()
    shifted = (b - max_c) / T
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    w_flip = w[::-1]
    A = rfft(a, conv_fft_size)
    W = rfft(w_flip, conv_fft_size)
    conv_result = irfft(A * W, conv_fft_size)[:3 * n - 2]
    
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad

def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def lp_direction(a0, n_iters=20, time_limit=60.0):

            continue

        return best_a, best_score
    
    y = a.copy()
    warm_b = None

        if time.monotonic() - t0 > time_limit:

        M_var = cp.Variable(value=1.0 / n)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        bs = b_val.sum()
        if bs < 1e-15:
            continue
        b_val = b_val / bs
        warm_b = b_val
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_val

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            try:
                p_score = evaluate_sequence([float(x) for x in parent])
            except Exception:
                p_score = float('inf')
            print(f"Parent: n={n_base}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = structured_init(n_base, 0)
        print(f"No parent available, using structured init n={n_base}")
        best_seq = parent.copy()
    
    fft_size = 1 << (2 * n_base - 2).bit_length()
    
    # === Phase 1: Extended Adam LSE (main workhorse, ~45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 800.0)
    print(f"\n=== Phase 1: Adam LSE n={n_base}, budget={t1:.0f}s ===")
    if t1 > 30:
        a_out, sc = adam_lse_extended(best_seq, t1, topk=25, T_start=100.0,
                                      T_end=0.05, n_stages=50, lr_base=0.004, seed=123)
        try_update(a_out, f"adam_lse_n{n_base}")
    
    # === Phase 2: LP direction finding (limited, ~8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)
    print(f"=== Phase 2: LP direction n={n_base}, budget={t2:.0f}s ===")
    if t2 > 30:
        a_out, sc = lp_direction(best_seq, n_iters=30, time_limit=t2)
        try_update(a_out, f"lp_dir_n{n_base}")
        # Run Adam LSE from LP result
        if remaining > t2 + 60:
            a_out2, sc2 = adam_lse_extended(a_out, 60.0, topk=20, T_start=20.0,
                                            T_end=0.1, n_stages=15, lr_base=0.003, seed=456)
            try_update(a_out2, "post_lp_lse")
    
    # === Phase 3: Aggressive peak shaving (~12% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    print(f"=== Phase 3: Peak shaving n={n_base}, budget={t3:.0f}s ===")
    if t3 > 15:
        a_out, sc = peak_shaving_aggressive(best_seq, t3, seed=777)
        try_update(a_out, f"peak_shave_n{n_base}")
    
    # === Phase 4: Mirror descent (~8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 120.0)
    print(f"=== Phase 4: Mirror descent n={n_base}, budget={t4:.0f}s ===")
    if t4 > 15:
        a_out, sc = mirror_descent(best_seq, t4, eta_start=0.015, seed=555)
        try_update(a_out, f"mirror_n{n_base}")
    
    # === Phase 5: Second round of peak shaving + LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 180.0)
    print(f"=== Phase 5: Peak shave round 2, budget={t5:.0f}s ===")
    if t5 > 15:
        a_out, sc = peak_shaving_aggressive(best_seq, t5 * 0.6, seed=999)
        try_update(a_out, "peak_shave_2")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, sc2 = adam_lse_extended(best_seq, min(remaining * 0.3, 60.0),
                                            topk=15, T_start=10.0, T_end=0.05,
                                            n_stages=15, lr_base=0.002, seed=111)
            try_update(a_out2, "lse_round2")
    
    # === Phase 6: Multi-length exploration (~10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6_total = min(remaining * 0.10, 180.0)
    print(f"=== Phase 6: Multi-length, budget={t6_total:.0f}s ===")
    if t6_total > 60:
        for dn in [100, 200]:
            if time.monotonic() - t_global > t_global + time_budget - 80:
                break
            n_new = n_base + dn
            t6_each = t6_total / 2
            if t6_each < 30:
                continue
            print(f"  Trying n={n_new}...")
            init_long = interpolate_sequence(best_seq, n_new)
            a_out, sc = adam_lse_extended(init_long, t6_each * 0.6, topk=20,
                                          T_start=50.0, T_end=0.1, n_stages=30,
                                          lr_base=0.003, seed=200 + dn)
            try_update(a_out, f"multilength_n{n_new}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and a_out is not None:
                a_out2, _ = peak_shaving_aggressive(a_out, min(remaining * 0.3, 30.0), seed=300)
                try_update(a_out2, f"multilength_shave_n{n_new}")
    
    # === Phase 7: Diverse restarts (~5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.05, 80.0)
    print(f"=== Phase 7: Diverse restarts, budget={t7:.0f}s ===")
    if t7 > 30:
        for vi in range(3):
            if time.monotonic() - t_global > t_global + time_budget - 40:
                break
            t7_each = t7 / 3
            # Perturbed version of best
            a_pert = best_seq * (1 + rng.normal(0, 0.05, n_base))
            a_pert = normalize_seq(a_pert)
            a_out, sc = adam_lse_extended(a_pert, t7_each, topk=20, T_start=30.0,
                                          T_end=0.5, n_stages=20, lr_base=0.003, seed=500 + vi)
            try_update(a_out, f"restart_{vi}")
    
    # === Phase 8: Final ultra-low-temp LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.3, 60.0)
        print(f"=== Phase 8: Final polish, budget={t8:.0f}s ===")
        a_out, sc = adam_lse_extended(best_seq, t8, topk=10, T_start=5.0,
                                      T_end=0.01, n_stages=20, lr_base=0.001, seed=9999)
        try_update(a_out, "final_polish")
    
    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.5, 40.0)
        print(f"=== Phase 9: Final peak shave, budget={t9:.0f}s ===")
        a_out, sc = peak_shaving_aggressive(best_seq, t9, seed=2001)
        try_update(a_out, "final_peak")
    
    # Final validation
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100
    
    return [float(x) for x in best_seq]
