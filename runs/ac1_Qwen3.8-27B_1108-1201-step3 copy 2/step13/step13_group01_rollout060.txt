Now, for the Nesterov acceleration, the key idea is:

                cols.append(j)

def nesterov_lp(a0, time_limit, solver, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s0 < 1e-15:

        a = a * norm / s0

    best_conv_max = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            if (t % 100 == 0) and verbose:
                print(f"    LP failed at iter {t}")
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Try multiple step sizes
        alpha = 2.0 / (t + 2.0)
        candidates = [alpha, alpha * 0.5, alpha * 1.5, 1.0 / (t + 1.0)]
        best_alpha = alpha
        best_trial_score = best_conv_max
        
        for a_try in candidates:
            a_try = min(a_try, 1.0)
            y_new = (1.0 - a_try) * y + a_try * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                continue
            y_new = y_new * norm / s_y
            sc = fast_autocorr(y_new).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_alpha = a_try
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        now = time.monotonic()
        
        sc_y = fast_autocorr(y).max()
        if sc_y < best_conv_max:
            best_conv_max = sc_y
            best_a = y.copy()
            last_improve = now
        
        sc_a = fast_autocorr(a).max()
        if sc_a < best_conv_max:
            best_conv_max = sc_a
            best_a = a.copy()
            last_improve = now
        
        if now - last_improve > 30.0:
            stall += 1
            if stall > 15:
                if verbose:

                break
        else:

            elapsed = now - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_conv_max:.8f}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
