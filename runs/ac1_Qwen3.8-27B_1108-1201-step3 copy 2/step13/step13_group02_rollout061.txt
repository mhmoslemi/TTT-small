"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. O(n) greedy dual LP for duality gap computation and adaptive steps

6. Running-average tracking for robust best-point identification

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def project_simplex(a, S):
    """Project vector a onto {x >= 0, sum(x) = S}."""
    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def compute_subgradient(a, K_active):
    """Compute subgradient of max_k (a*a)_k at positions K_active.
    grad_j = 2 * sum_{k in K} a_{k-j} for valid indices.
    """
    n = len(a)
    size = 2 * n - 1
    w = np.zeros(size)
    w[K_active] = 1.0
    # grad_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * (w_convolve_a_reversed)[j + n - 1] approximately
    a_rev = a[::-1]
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def compute_lse_weights(b, T):
    """Compute softmax weights for LSE approximation."""
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[b == max_b] = 1.0 / max(1, (b == max_b).sum())
        return w
    return w / s


def compute_lse_gradient(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k using FFT."""
    n = len(a)
    a_rev = a[::-1]
    size_w = len(w)
    fft_size = 1 << (size_w + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def greedy_dual_lp_value(a, M):
    """Compute optimal LP value using greedy algorithm on the dual.
    The dual is: min sum(lambda) s.t. sum_{k=j}^{j+n-1} lambda_k >= 1/a[j], lambda >= 0.
    Returns the optimal value (sum of lambda).
    """
    n = len(a)
    lam = np.zeros(2 * n - 1)
    active = 0.0
    for j in range(n):
        if a[j] < 1e-30:
            target = 0.0
        else:
            target = 1.0 / a[j]
        if active < target:
            deficit = target - active
            lam[j + n - 1] += deficit
            active += deficit
        # Advance window
        active -= lam[j]
        if j + n < 2 * n - 1:
            active += lam[j + n]
    return lam.sum() * M


def lp_best_response_fast(a, M, solver):
    """Solve the LP best response using cvxpy. Returns b or None."""
    n = len(a)
    if n > 3000:
        return None
    b_var = cp.Variable(n, nonneg=True)
    # Build constraints: (a*b)_k <= M for all k
    # (a*b)_k = sum_j a_j * b_{k-j}
    # Use the banded structure
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_subgradient(a0, time_limit, verbose=True, use_lp_every=2000):
    """Fast Nesterov subgradient method with FFT-based gradients.
    Minimizes max_k (a*a)_k subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    # Nesterov initialization
    y = a.copy()
    x = a.copy()

    # Adaptive step size parameters
    G_est = 2.0 * best_score / S  # rough gradient bound
    R = S * np.sqrt(2.0) / np.sqrt(n)  # effective radius
    c_step = R / max(G_est, 1e-10)

    # Running average for subgradient method
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation
        b = fast_autocorr(y)
        M_curr = b.max()

        # Find active constraints (near-max positions)
        threshold = M_curr * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.where(b == M_curr)[0]

        # Compute gradient
        grad = compute_subgradient(y, active)

        # Nesterov step size (optimal for convex subgradient)
        eta = c_step / np.sqrt(t)

        # Update y (the "look-ahead" point)
        y_new = y - eta * grad
        y_new = project_simplex(y_new, S)

        # Update x with Nesterov momentum
        if t == 1:
            x_new = y_new
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * S / s_x

        y = y_new
        x = x_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_x = fast_autocorr(x).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()

        # Running average
        avg_a += x
        n_avg += 1
        if n_avg % 50 == 0:
            avg_candidate = avg_a / n_avg
            avg_candidate = np.maximum(avg_candidate, 0.0)
            s_avg = avg_candidate.sum()
            if s_avg > 1e-15:
                avg_candidate = avg_candidate * S / s_avg
                score_avg = fast_autocorr(avg_candidate).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_candidate.copy()

        n_iters += 1

        # Periodic LP correction
        if use_lp_every > 0 and n_iters % use_lp_every == 0:
            pass  # LP correction handled externally

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M_y={M_curr:.8f}")

    return best_a, best_score

def lse_polish(a0, time_limit, verbose=True):
    """LSE gradient descent with adaptive temperature for final polishing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    b_init = fast_autocorr(a)
    M_init = b_init.max()
    T_start = M_init * 0.05
    T_end = M_init * 0.001

    # Use Adam-like adaptive steps
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = S / (n * 0.5)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = compute_lse_weights(b, T)
        grad = compute_lse_gradient(a, w)

        steps += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** steps)
        v_hat = v_v / (1 - beta2 ** steps)
        step = lr_base / np.sqrt(steps)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step * update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * S / s_new

        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_random_init(n, seed, style=0):
    """Generate different types of random initializations."""
    rng = np.random.default_rng(seed)
    if style == 0:
        # Uniform
        return np.ones(n) / n
    elif style == 1:
        # Exponential decay from edges
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
        a += 0.1 * rng.standard_normal(n)
        return np.maximum(a, 0) / np.maximum(a, 0).sum()
    elif style == 2:
        # Edge-peaked
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 20 * np.sin(np.pi * x) ** 2)
        a += 0.05 * rng.standard_normal(n)
        return np.maximum(a, 0) / np.maximum(a, 0).sum()
    elif style == 3:
        # Sine wave with random phase
        x = np.linspace(0, 1, n)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x + phase)
        a += 0.05 * rng.standard_normal(n)
        return np.maximum(a, 0) / np.maximum(a, 0).sum()
    elif style == 4:
        # Random smooth
        a = np.abs(np.cumsum(rng.standard_normal(n)))
        a += 0.1 * rng.standard_normal(n)
        return np.maximum(a, 0) / np.maximum(a, 0).sum()
    else:
        # Pure random
        a = rng.random(n)
        return a / a.sum()

        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = hybrid_nesterov(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"hybrid_n{n_cur}")

    # === Phase 3: LP Nesterov refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # === Phase 4: LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE Polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_polish(a_norm, t4, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 5: Try longer length ===
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Hybrid n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = hybrid_nesterov(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"hybrid_n{n2}")

    # === Phase 6: Try shorter length ===
    n3 = max(1000, n_base - 200)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Hybrid n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        init_short = interpolate_sequence(best_seq, n3)
        a_norm = init_short * np.sqrt(2.0 * n3) / init_short.sum()
        a_out, _ = hybrid_nesterov(a_norm, t6, solver, verbose=False)
        do_try_update(a_out, f"hybrid_n{n3}")

    # === Phase 7: Final LSE polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_polish(a_norm, t7, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # === Phase 8: Final hybrid on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Hybrid n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 250.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = hybrid_nesterov(a_norm, t8, solver, verbose=True)
        do_try_update(a_out, f"hybrid_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
