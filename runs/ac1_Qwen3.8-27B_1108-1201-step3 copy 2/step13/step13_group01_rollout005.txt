            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_toeplitz(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, verbose=False):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_toeplitz(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        if dual_val is not None:
            dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, restart_threshold=45.0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    a_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_time = 0.0
    restart_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, dual = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size: accelerate when improving, decelerate when stalling
        if best_score > 0:
            ratio = M_curr / best_score
            base_alpha = 2.0 / (t + 2.0)
            if ratio < 1.0:
                alpha = min(base_alpha * (1.0 + 0.5 * (1.0 - ratio)), 1.0)
            elif ratio < 1.2:
                alpha = base_alpha
            else:
                alpha = base_alpha * 0.5
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Restart mechanism: if stalled, perturb and reset momentum
        time_since_improve = time.monotonic() - last_improve
        if time_since_improve > restart_threshold:
            stall_time += 1
            if stall_time > 8:
                restart_count += 1
                rng_r = np.random.default_rng(seed_hash(restart_count))
                perturbation = rng_r.normal(0, 0.01, size=n)
                a_restart = best_a + perturbation * best_score
                a_restart = np.maximum(a_restart, 0.0)
                s_r = a_restart.sum()
                if s_r > 1e-15:
                    a_restart = a_restart * norm / s_r
                y = a_restart.copy()
                a = a_restart.copy()
                a_prev = a_restart.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall_time = 0
                if verbose:
                    print(f"    [RESTART #{restart_count}] at iter={n_iters}, t={time.monotonic()-t0:.0f}s")

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M_curr={M_curr:.8f}")

    return best_a, best_score


def seed_hash(count):
    return 12345 + count * 7919


def dual_guided_coordinate_descent(a0, time_limit, solver, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:12]

        # Get dual variables from LP to identify critical peaks
        M_curr = b.max()
        b_lp, dual = lp_best_response(a, M_curr, solver)
        if dual is not None and len(dual) == len(b):
            # Weight peaks by dual variable strength
            dual_weighted = dual * b
            critical_peaks = np.argsort(dual_weighted)[::-1][:8]
        else:
            critical_peaks = top_peaks

        improved = False
        for k_peak in critical_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of coordinates involved in top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(2, len(candidates) // 4)]

                for dst in candidates_sorted[:4]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall % 5 == 1:
                for _ in range(30):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = float(rng.uniform(0.005, 0.08))
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def lse_scipy_optimize(a0, time_limit, T_start=40.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 180.0)

        def obj_func(x):
            return _lse_objective(x, n, T)

        def grad_func(x):
            return _lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func, options={'maxiter': 300,
                                  'maxfun': 800, 'factr': 1e4, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def _lse_objective(x_flat, n, T):
    a = np.maximum(np.asarray(x_flat, dtype=np.float64).reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    scaled = T * (b - max_b)
    scaled = np.clip(scaled, -500, 500)
    lse = max_b + np.log(np.exp(scaled).sum()) / T
    obj = 2.0 * n * lse / (s * s)
    return obj


def _lse_gradient(x_flat, n, T):
    a = np.maximum(np.asarray(x_flat, dtype=np.float64).reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    scaled = T * (b - max_b)
    scaled = np.clip(scaled, -500, 500)
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T

    # Compute gradient of LSE w.r.t. a via FFT
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 7))
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2.0 * np.pi * x) + 0.15 * np.cos(4.0 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-8.0 * x ** 2)
    elif choice == 5:
        a = (1.0 - x) ** 0.8
    else:
        a = np.exp(-3.0 * (x - 0.4) ** 2) * (1.0 + 0.3 * np.sin(4.0 * np.pi * x))
        a = np.maximum(a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            parent = p / p.sum()

        parent = None
        print("No parent sequence available, using structured init.")

        nonlocal best_seq, best_score
        if a is None:
            return

        s = a.sum()
        if s < 0.01:
            return

            best_score = sc
            best_seq = a.copy()

        remaining_scan = time_budget * 0.12 - (time.monotonic() - t_global)

        print(f"\n=== Phase 7: LSE final n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, remaining, T_start=5.0, T_end=0.3, n_stages=4)
        do_try_update(a_out, "lse_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
