       A = rfft(a, n=fft_size)
       return irfft(A * A, n=fft_size)[:size]

       """Compute LSE objective and gradient."""
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       
       conv = fast_autocorr(a)
       max_c = conv.max()
       shifted = T * (conv - max_c)
       exp_shifted = np.exp(shifted)
       Z = exp_shifted.sum()
       obj = max_c + np.log(Z) / T
       w = exp_shifted / Z

       A = rfft(a, n=fft_size)
       cross = irfft(W * np.conj(A), n=fft_size)
       grad = 2.0 * cross[:n]
       
       return obj, grad, conv

   def waterfill_project(v, S):
       """Project v onto {x >= 0, sum(x) = S} (Euclidean projection)."""
       n = len(v)
       if n == 0:
           return v
       # Sort in decreasing order
       u = np.sort(v)[::-1]
       # Find tau using the water-filling formula
       cv = np.cumsum(u)
       rho = np.arange(1, n + 1)
       cond = u - (cv - S) / rho > 0
       if not cond.any():

           return np.full(n, S / n)
       rho_hat = np.max(rho[cond])
       tau = (cv[rho_hat - 1] - S) / rho_hat
       return np.maximum(v - tau, 0)
