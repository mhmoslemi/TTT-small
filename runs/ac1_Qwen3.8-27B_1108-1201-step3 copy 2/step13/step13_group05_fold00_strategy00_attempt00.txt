def fast_autocorr(a):
    """Compute linear convolution a*a using FFT efficiently."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    t = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(t * t, fft_size)[:size]

def proj_simplex(v, s):
    """Project v onto the set {x >= 0, sum(x) = s}."""
    if s <= 0:
        return v.astype(float)
    if v.shape[0] == 0:
        return np.ones(1) * (s / 1.0 if 1 > 0 else 0)
    v_flat = np.array(v).flatten()
    if s < 1e-30:

        if s < 1e-15:
            return np.ones(len(v_flat))
        return v_flat / s
    if v_flat.shape[0] == 1:

        return np.array([coef]) * v_flat
