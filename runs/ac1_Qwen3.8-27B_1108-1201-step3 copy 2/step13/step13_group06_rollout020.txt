import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def compute_lse_and_grad(a, T, topk):
    """Compute top-k LSE of autocorrelation and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    if max_Tc < -1e300:
        return 0.0, np.zeros(n)
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def lse_grad_z(z, T, temp, topk):
    """Objective and gradient in z-space for softmax parameterization."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_lse_and_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def multi_temp_lse(a0, time_limit, T_start=80.0, T_end=0.1, n_stages=40, topk_list=None):
    """
    Multi-temperature, multi-topk LSE Adam gradient descent.
    Uses adaptive top-k selection and periodic restarts.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    if topk_list is None:
        topk_list = [50, 30, 20, 15, 10, 8, 5, 3]

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        # Choose topk based on stage progress
        frac = si / max(1, n_stages - 1)
        topk_idx = min(len(topk_list) - 1, int(frac * len(topk_list)))
        topk = topk_list[topk_idx]

        base_lr = 5e-3 * min(1.0, 50.0 / n)
        lr = base_lr * (0.92 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.75, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 90.0) * 8))

        local_best = best_score
        local_best_a = best_a.copy()
        stall = 0

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    local_best = score
                    local_best_a = a.copy()
                    stall = 0
                else:
                    stall += 1

                # Periodic restart from best if stalled
                if stall > 80 and (it + 1) % 40 == 0:
                    z = np.log(np.maximum(local_best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_iter_max=50000):
    """
    Exact subgradient method with projected step onto simplex.
    Uses averaging of iterates for convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0

    eta0 = 0.02
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_autocorr(a)
        bmax = b.max()

        # Find all peaks within 0.5% of max
        threshold = bmax * 0.995
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])

        # Compute average subgradient over all active peaks
        grad = np.zeros(n)
        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adaptive step size (decreasing)
        eta = eta0 / np.sqrt(it + 1)

        # Take step and project
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        # Line search: try a few step sizes
        best_trial = a_new
        best_trial_score = 2.0 * n * fast_autocorr(a_new).max()

        for frac in [0.5, 0.25, 0.1]:
            a_trial = a - eta * frac * grad
            a_trial = project_simplex(a_trial)
            sc = 2.0 * n * fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = a_trial

        a = best_trial
        avg_a += a
        n_avg += 1

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Check running average
        if n_avg % 50 == 0 and n_avg >= 100:
            avg_check = avg_a / n_avg
            avg_score = 2.0 * n * fast_autocorr(avg_check).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_check.copy()
                stall = 0

        if stall > 200:
            # Restart from best with smaller eta
            a = best_a.copy()
            eta0 *= 0.5
            avg_a = np.zeros(n)
            n_avg = 0
            stall = 0
            if eta0 < 1e-5:
                break

    return best_a, best_score


def aggressive_peak_shave(a0, time_limit, seed=42):
    """
    Aggressive multi-peak shaving with move and swap operations.
    Considers global peak constraint.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()

        # Get top peaks
        n_peaks = min(20, len(b))
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # Compute danger mask: positions that contribute to any top peak
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False

        # Try move operations: move mass from a dangerous position to a safe one
        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Get top contributing positions
            order = np.argsort(contrib)[::-1]
            for oi in order[:6]:
                src = idx[oi]

                # Find safe destinations (not in danger zone)
                safe = np.where(~danger)[0]
                safe = safe[safe != src]
                if len(safe) == 0:
                    continue

                # Sort safe positions by current value (prefer moving to small positions)
                safe_sorted = safe[np.argsort(a[safe])]
                dst_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]

                for dst in dst_pool[:12]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:

                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-9):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        # Try swap operations: exchange mass between two positions
        if not improved:
            # Find pairs of positions where swapping might help
            peak_positions = set()
            for k in top_peaks[:8]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                peak_positions.update(range(js, je + 1))

            peak_list = list(peak_positions)
            if len(peak_list) >= 2:
                # Try swapping some mass between a high-contributing peak position
                # and a low-contributing position
                for _ in range(min(10, len(peak_list))):
                    i = peak_list[rng.integers(len(peak_list))]
                    j = rng.integers(n)
                    if i == j:
                        continue
                    a_i, a_j = a[i], a[j]
                    for frac in [0.01, 0.02, 0.05, 0.1]:
                        delta = min(a_i, a_j) * frac
                        if delta < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[i] = a_i - delta
                        a_trial[j] = a_j + delta
                        if a_trial[i] < -1e-12 or a_trial[j] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-9):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 20 == 0:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def find_solver():

    try:
        import cvxpy as cp
        for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):

            except Exception:

        return cp.ECOS
    except Exception:
        return None


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
