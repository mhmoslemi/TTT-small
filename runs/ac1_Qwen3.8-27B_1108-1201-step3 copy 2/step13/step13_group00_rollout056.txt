def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, size=None):
    """Full LP best response: max sum(b) s.t. b>=0, T(a)@b <= M"""
    n = len(a)
    if size is None:
        size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    return b_val


def lp_best_response_screened(a, M, solver, n_active_frac=0.3):
    """Screened LP: only use the most active constraints."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a
    
    # Select active constraints
    threshold = M * (1.0 - 0.05)
    active = np.where(conv_current > threshold)[0]
    if len(active) < max(10, int(size * 0.05)):
        active = np.where(conv_current > M * 0.5)[0]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Verify full constraint satisfaction
    full_conv = T @ b_val
    if full_conv.max() > M * (1 + 1e-5):
        # Fallback to full LP
        return lp_best_response(a, M, solver, size)
    
    return b_val

def block_coordinate_descent(a, time_limit, rng, block_size=4, verbose=True):
    """Block coordinate descent: optimize groups of consecutive elements."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Try blocks starting at random positions
        n_blocks = max(10, n // block_size)
        for _ in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            start = int(rng.integers(0, n - block_size + 1))
            idx = np.arange(start, start + block_size)
            
            # Try redistributing mass within the block
            block_val = a[idx].copy()
            block_sum = block_val.sum()
            if block_sum < 1e-12:
                continue
            
            # Try several redistribution patterns
            for pattern_type in range(3):
                if pattern_type == 0:
                    # Spread uniformly
                    new_block = np.full(block_size, block_sum / block_size)
                elif pattern_type == 1:
                    # Linear gradient
                    x = np.linspace(0, 1, block_size)
                    new_block = x * block_sum / (x.sum() + 1e-15)
                else:
                    # Uniform + noise
                    new_block = np.full(block_size, block_sum / block_size)
                    new_block += rng.normal(0, block_sum / block_size * 0.1, block_size)
                    new_block = np.maximum(new_block, 0)
                    new_block = new_block * block_sum / (new_block.sum() + 1e-15)
                
                # Evaluate
                a_trial = a.copy()
                a_trial[idx] = new_block
                a_trial = a_trial * norm / a_trial.sum()
                trial_conv = fast_autocorr(a_trial)
                trial_max = trial_conv.max()
                
                if trial_max < best_score - 1e-10:
                    a = a_trial
                    conv = trial_conv
                    best_score = trial_max
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def gradient_flow_optimize(a0, time_limit, verbose=True):
    """
    Optimize using a smooth surrogate (LSE) with adaptive temperature.
    Uses FFT for gradient computation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature annealing
    temps = [10.0, 5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1]
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Number of steps proportional to remaining time
        n_steps = max(20, min(200, int(remaining * 2)))
        
        A_fft = rfft(a, fft_size)
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            max_conv = conv.max()
            
            # LSE weights
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            w = np.maximum(w, 0)  # Ensure non-negative
            
            # Gradient of LSE w.r.t. a_i
            # LSE = T * log(sum_k exp(conv_k / T))
            # dLSE/da_i = (1/T) * sum_k exp(conv_k/T) * dconv_k/da_i
            # dconv_k/da_i = a_{k-i} + a_{i-k} (for valid indices)
            # So grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k}) / ... 
            # Actually: grad_i = sum_k w_k * (a_{k-i} + a_{i-k})
            # This is the cross-correlation of w with a, plus w with flipped a
            
            # Using FFT: 
            # sum_k w_k * a_{k-i}: this is the cross-correlation
            # cross1[i] = sum_k w[k] * a[k-i] (= conv(w, a) at position... )
            # Actually, let's think of it differently.
            # sum_k w[k] * a[k-i]: let m = k-i, so k = m+i
            # = sum_m w[m+i] * a[m]
            # This is the cross-correlation of w with a at lag i.
            
            # For the FFT approach:
            # ifft(FFT(w) * conj(FFT(a))) gives cross-correlation
            W_fft = rfft(w, fft_size)
            # cross[i] = sum_k w[k] * a[k+i] (circular)
            # We want sum_k w[k] * a[k-i] = cross at lag -i = cross at (fft_size - i)
            
            cross = irfft(W_fft * np.conj(A_fft), fft_size)
            # cross[m] = sum_k w[k]*a[k+m] (circular correlation)
            # We want sum_k w[k]*a[k-i] = cross[fft_size - i] for i > 0, cross[0] for i = 0
            
            g1 = np.zeros(n)
            for i in range(n):
                # sum_k w[k] * a[k-i]
                # In circular: cross[(fft_size - i) % fft_size]
                g1[i] = cross[(fft_size - i) % fft_size]
            
            # sum_k w[k] * a[i-k] = sum_k a[i-k] * w[k]
            # Let j = i-k: sum_j a[j] * w[i-j]
            # = cross of a with w at lag i = cross_aw[i]
            cross_aw = irfft(A_fft * np.conj(W_fft), fft_size)
            # cross_aw[m] = sum_k a[k] * w[k+m]
            # sum_j a[j] * w[i-j]: let k=j, = sum_k a[k] * w[i-k] = cross_aw[fft_size - i]
            g2 = np.zeros(n)
            for i in range(n):
                g2[i] = cross_aw[(fft_size - i) % fft_size]
            
            grad = g1 + g2
            
            # Projected gradient step
            step = norm / (n * (1.0 + 0.1 * si)) / (T + 0.1)
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-15:
                step = min(step, norm / (grad_norm * 10))
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            # Update FFT
            A_fft = rfft(a, fft_size)
            
            # Check improvement
            if it % 5 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Asymmetric: high on one side
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * x) + 0.5 * np.exp(-3.0 * (x - 0.5)**2)
    else:
        # Binary-like
        a = np.where(rng.random(n) > 0.3, 1.0, 0.1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

import numpy as np

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver, active_indices=None):
    """Solve LP: max sum(b) s.t. b>=0, T(a)@b <= M"""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_indices is not None and len(active_indices) < size:
        constraints = [T[active_indices, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    return b_val, None


def lp_solve_screened(a, M, solver, prev_active=None):
    """LP with adaptive constraint screening and verification."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_cur = T @ a

    if prev_active is not None:
        active = prev_active.copy()
    else:
        thresh = np.percentile(conv_cur, 70)
        active = np.where(conv_cur > thresh)[0]
        if len(active) < 20:
            active = np.where(conv_cur > M * 0.5)[0]

    b_val, _ = lp_solve(a, M, solver, active)
    if b_val is None:
        return None, None

    # Verify all constraints
    full_conv = T @ b_val
    violated = np.where(full_conv > M * (1.0 + 1e-6))[0]
    if len(violated) > 0:
        new_active = np.union1d(active, violated)
        if len(new_active) > 0.85 * size:
            b_val, _ = lp_solve(a, M, solver, None)
            active = None
        else:
            b_val, _ = lp_solve(a, M, solver, new_active)
            if b_val is not None:
                active = new_active
            else:
                b_val, _ = lp_solve(a, M, solver, None)
                active = None

    if b_val is None:
        return None, None
    return b_val, active


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov with adaptive screening and restart logic."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    active = None
    stall_count = 0
    last_best_score = best_score

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Solve LP (use full for first 20 iters, then screened)
        if t <= 20:
            b, _ = lp_solve(y, M_curr, solver, None)
            active = None
        else:
            b, active = lp_solve_screened(y, M_curr, solver, active)
            if b is None:
                b, _ = lp_solve(y, M_curr, solver, None)
                active = None

        if b is None or b.sum() < 1e-15:
            continue

        g = b * norm / b.sum()
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Also check y
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Restart if stalled for too long
        if stall_count > 100:
            y = best_a.copy()
            a = best_a.copy()
            active = None
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall_count}")

    return best_a, best_score


def targeted_cd(a, time_limit, rng, verbose=True):
    """
    Targeted coordinate descent: focus on coordinates that contribute
    most to the peak convolution lag.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Find peak lag
        k_star = int(np.argmax(conv))

        # Find coordinates contributing most to conv[k_star]
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Process high-contribution coordinates first
        order = np.argsort(-contrib)

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx)
            if a[i] < norm * 1e-10:
                continue

            # Try transferring small amounts to random other coordinates
            ai = a[i]
            n_tries = max(3, min(10, int(remaining_time_fraction(n_passes, time_limit, t0) * 20)))

            for _ in range(n_tries):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Try several delta values (geometric spacing)
                best_d = 0.0
                best_val = conv.max()
                base_d = ai * 0.1
                for mult in [0.01, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                    d = mult * ai
                    if d >= ai:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    a_trial = a_trial * norm / a_trial.sum()
                    trial_conv = fast_autocorr(a_trial)
                    trial_max = trial_conv.max()
                    if trial_max < best_val - 1e-13:
                        best_val = trial_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def remaining_time_fraction(n_passes, time_limit, t0):
    elapsed = time.monotonic() - t0
    return max(0.0, 1.0 - elapsed / time_limit)


def lse_refine(a0, time_limit, verbose=True):
    """LSE smooth gradient with FFT-based gradient computation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Multi-stage temperature
    stages = [(20.0, 0.15), (10.0, 0.15), (5.0, 0.15), (2.0, 0.15), (1.0, 0.15), (0.5, 0.15), (0.3, 0.15), (0.2, 0.15)]

    for T, frac in stages:
        if time.monotonic() - t0 > time_limit:
            break
        stage_time = time_limit * frac
        stage_start = time.monotonic()
        n_steps = max(10, int(stage_time * 2))

        for it in range(n_steps):
            if time.monotonic() - stage_start > stage_time or time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_c = conv.max()
            exp_c = np.exp((conv - max_c) / T)
            w = exp_c / exp_c.sum()

            # Compute gradient: grad_j = sum_k w[k] * (a[k-j] + a[j-k])
            # Using FFT cross-correlation
            A_f = rfft(a, fft_size)
            W_f = rfft(w, fft_size)

            # cross1[j] = sum_k w[k] * a[k-j] = circular cross-corr at lag -j
            cross1 = irfft(W_f * np.conj(A_f), fft_size)
            # cross1[m] = sum_k w[k]*a[(k+m) mod fft_size]
            # We want sum_k w[k]*a[k-j] = cross1[fft_size - j] for j > 0

            # cross2[j] = sum_k w[k] * a[j-k]
            # = sum_k a[j-k] * w[k], let m=j-k: sum_m a[m]*w[j-m]
            # = cross of a with w at lag j
            cross2 = irfft(A_f * np.conj(W_f), fft_size)
            # cross2[m] = sum_k a[k]*w[(k+m) mod fft_size]
            # sum_k a[j-k]*w[k]: let n=j-k, k=j-n: sum_n a[n]*w[j-n]
            # = sum_n a[n]*w[-(n-j)] = cross2 at lag (j) ... 
            # cross2[m] = sum_k a[k]*w[k+m]
            # We want sum_k a[j-k]*w[k]. Let p=j-k, k=j-p: sum_p a[p]*w[j-p]
            # = sum_p a[p]*w[-(p-j)] = this is cross2 at m = ... 
            # cross2[m] = sum_k a[k]*w[k+m]. Set k=p: sum_p a[p]*w[p+m]
            # We want sum_p a[p]*w[j-p] = sum_p a[p]*w[-(p-j)] = cross2 at m = -(p-j)... 
            # This is cross2[fft_size - (p-j)] summed over p... no that doesn't work.
            # 
            # Actually: sum_p a[p]*w[j-p] is the (j)-th element of the linear convolution of a and w.
            # lin_conv(a,w)[j] = sum_p a[p]*w[j-p]
            # Using FFT: ifft(FFT(a)*FFT(w)) gives circular convolution
            # If fft_size >= 2n, then circular = linear for indices 0..2n-2

            conv_aw = irfft(A_f * W_f, fft_size)
            # conv_aw[k] = sum_p a[p]*w[k-p] (circular, but = linear if fft_size >= 2n)

            grad = np.zeros(n)
            for j in range(n):
                # Part 1: sum_k w[k]*a[k-j]
                # = cross1[(fft_size - j) % fft_size] but need to handle wrap
                # cross1[m] = sum_k w[k]*a[(k+m) mod F] where F=fft_size
                # For m = F-j: sum_k w[k]*a[(k-F+j) mod F] = sum_k w[k]*a[(k+j) mod F]
                # Hmm that gives a[k+j] not a[k-j]
                # For m = j: sum_k w[k]*a[(k+j) mod F] -- this is a[k+j]
                # We want a[k-j] = a[(k-j) mod F] which is cross1 at m = F-j
                # Wait: cross1[m] = ifft(FFT(w)*conj(FFT(a)))[m]
                # = sum_k w[k] * a[(k-m) mod F]  (by the cross-correlation formula)
                # Actually the standard formula: ifft(X*conj(Y))[n] = sum_k x[k]*y[k-n]
                # So cross1[m] = sum_k w[k]*a[k-m]
                # We want sum_k w[k]*a[k-j] = cross1[j]
                grad[j] += cross1[j % fft_size]

                # Part 2: sum_k w[k]*a[j-k]
                # = lin_conv(a, w)[j] = conv_aw[j]
                # (since fft_size >= 2n, circular = linear for j in [0, n-1])
                grad[j] += conv_aw[j % fft_size]

            # Projected gradient step
            lr = norm * 0.5 / (n * (T + 0.5) * (1.0 + 0.01 * it))
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            if it % 10 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0.0)
    ss = f_i.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_i / ss


def diverse_init(n, rng):

        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(9 * x)
    elif choice == 3:

    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.exp(-5.0 * x) + 0.3 * np.exp(-2.0 * (x - 0.6) ** 2)
    elif choice == 7:
        a = np.ones(n)
        a[::2] += 0.3
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * 3)) ** 0.5 + 0.2
    else:
        a = np.ones(n)
        mask = rng.random(n) > 0.3
        a[~mask] *= 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_init = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp(a_init, t1, solver, verbose=True)
        do_update(a_out, f"lp_n{n1}")

    # Phase 2: LSE refinement
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    print(f"\n=== Phase 2: LSE refine n={nb} ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_init = best_seq * np.sqrt(2.0 * nb) / best_seq.sum()
        a_out, _ = lse_refine(a_init, t2, verbose=True)
        do_update(a_out, f"lse_n{nb}")

    # Phase 3: Targeted CD
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    print(f"\n=== Phase 3: Targeted CD n={nb} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_init = best_seq * np.sqrt(2.0 * nb) / best_seq.sum()
        a_out, _ = targeted_cd(a_init, t3, rng, verbose=True)
        do_update(a_out, f"tcd_n{nb}")

    # Phase 4: Try longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        init_len = interpolate_seq(best_seq, n2)
        a_init = init_len * np.sqrt(2.0 * n2) / init_len.sum()
        a_out, _ = nesterov_lp(a_init, t4, solver, verbose=True)
        do_update(a_out, f"lp_n{n2}")

    # Phase 5: Targeted CD on current best
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    print(f"\n=== Phase 5: TCD polish n={nb} ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_init = best_seq * np.sqrt(2.0 * nb) / best_seq.sum()
        a_out, _ = targeted_cd(a_init, t5, rng, verbose=True)
        do_update(a_out, f"tcd2_n{nb}")

    # Phase 6: Perturbation restarts
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    print(f"\n=== Phase 6: Perturbation restarts ===")
    n_restarts = 2
    t_per = min(remaining / (n_restarts + 0.5), 120.0)
    for r in range(n_restarts):
        if time.monotonic() - t_global > time_budget - 60:
            break
        noise_scale = 0.03 + 0.02 * r
        a_pert = best_seq + rng.normal(0, noise_scale / np.sqrt(nb), nb)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_init = a_pert * np.sqrt(2.0 * nb) / a_pert.sum()
        t_r = min(t_per, time_budget - (time.monotonic() - t_global) - 60)
        if t_r > 20:
            a_out, _ = nesterov_lp(a_init, t_r, solver, verbose=False)
            do_update(a_out, f"pert{r}_n{nb}")

    # Phase 7: Final targeted CD
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    print(f"\n=== Phase 7: Final TCD n={nb} ===")
    t7 = min(remaining * 0.5, 100.0)
    if t7 > 10:
        a_init = best_seq * np.sqrt(2.0 * nb) / best_seq.sum()
        a_out, _ = targeted_cd(a_init, t7, rng, verbose=True)
        do_update(a_out, f"final_tcd_n{nb}")

    # Phase 8: Final LSE
    remaining = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    t8 = min(remaining * 0.4, 60.0)
    if t8 > 10:
        a_init = best_seq * np.sqrt(2.0 * nb) / best_seq.sum()
        a_out, _ = lse_refine(a_init, t8, verbose=False)
        do_update(a_out, f"final_lse_n{nb}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
