def softmax_weights(ac, beta):
    ac_max = ac.max()
    exponents = beta * (ac - ac_max)
    exponents = np.maximum(exponents, -700.0)
    w = np.exp(exponents)
    w_sum = w.sum()

            peak_influence = np.zeros(n)

                a = np.maximum(a, 0.0)
                a = normalize_a(a, c)
                if fast_autocorr(a).max() < best_m:
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    PairwiseSwap ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Uniform
    inits.append(np.ones(n))
    # Sinusoidal
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x) ** 2 + 0.01)
    # Triangular
    inits.append(1.0 - 2.0 * np.abs(x - 0.5) + 0.01)
    # Bump (Gaussian)
    inits.append(np.exp(-20.0 * (x - 0.5) ** 2) + 0.01)
    # Two bumps
    inits.append(np.exp(-40.0 * (x - 0.3) ** 2) + np.exp(-40.0 * (x - 0.7) ** 2) + 0.01)
    # Random smooth
    r = rng.standard_normal(n // 10 + 1)
    smooth = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(r)), r)
    inits.append(np.abs(smooth) + 0.1)
    # Cosine window
    inits.append(np.cos(np.pi * x) ** 2 + 0.01)
    return [np.maximum(i, 0.01) for i in inits]
