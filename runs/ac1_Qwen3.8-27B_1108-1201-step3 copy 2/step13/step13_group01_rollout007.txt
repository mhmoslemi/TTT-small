def lse_objective_and_grad(a_flat, n, T_temp):
    """Compute LSE objective and gradient for the autocorrelation problem."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    shifted = T_temp * (b - max_b)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    
    # Gradient of LSE w.r.t. b: softmax weights
    w = exp_vals / exp_vals.sum()
    
    # Gradient of b = convolve(a,a) w.r.t. a
    # db/d a[i] = 2 * (a correlated with w)
    # Use FFT for efficiency
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    # Gradient of objective = 2n * lse / s^2
    obj_val = 2.0 * n * lse / (s * s)
    conv_grad = 4.0 * cross / (s * s) - 2.0 * obj_val * a / s
    
    return obj_val, conv_grad.ravel()

def lse_anneal(a0, time_limit):
    """LSE optimization with geometric temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = [50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.8, 0.4]
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (len(temps) - si), 180.0)

        def obj_func(x):
            obj, _ = lse_objective_and_grad(x, n, T)
            return obj

        def grad_func(x):
            _, grad = lse_objective_and_grad(x, n, T)
            return grad

        x0 = a.copy()
        max_iter = min(300, max(50, int(stage_time * 2)))
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                          'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
                        print(f"    LSE T={T}: new best={best_score:.8f}")
        except Exception as e:
            print(f"    LSE T={T} failed: {e}")

    return best_a, best_score

                            peak_positions.add(j)

                        j = k2 - i
                        if 0 <= j < n:
                            peak_positions.add(i)

                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t

                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:

                    continue

                a_trial[i] -= amt

                if s_t < 1e-10:

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

                break
            if improved:
                stall = 0

    return best_a, best_score
