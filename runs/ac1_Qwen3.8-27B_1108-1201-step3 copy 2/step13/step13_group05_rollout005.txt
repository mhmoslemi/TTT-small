        b = lp_best_response_sparse(y, M_curr, active_k, solver)
        if b is None:
            # Fallback: use all constraints
            b = lp_best_response_sparse(y, M_curr, np.arange(len(ac)), solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_M:
            best_M = score_x
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

def mirror_descent_fast(a0, time_limit, top_k=5, beta=3.0, verbose=True):
    """Fast mirror descent with running average."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    n_avg = 0
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 1.0 / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        M = ac.max()

        k_max = int(np.argmax(ac))
        js = max(0, k_max - n + 1)
        je = min(n, k_max + 1)
        idx = np.arange(js, je)
        g = np.zeros(n)
        g[idx] = 2.0 * a[k_max - idx]

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        avg_a += a
        n_avg += 1

        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = avg_check.copy()

def coordinate_descent_polish(a0, time_limit, verbose=True):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP best-response (scipy HiGHS) with Nesterov acceleration and sparse active constraints
2. Mirror descent complement
3. Coordinate descent polish
4. Multi-length scaling and multiple restarts
5. Randomization for escaping local minima (problem is non-convex)
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def lp_best_response(a, M, active_k):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for k in active_k, b >= 0.
    Uses scipy HiGHS solver for speed."""
    n = len(a)
    K = len(active_k)
    if K == 0:
        return None
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraint matrix: K x n
    A_ub = np.zeros((K, n))
    for i, k in enumerate(active_k):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            A_ub[i, idx] = a[k_int - idx]
    
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    b_val = np.maximum(res.x, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    x = a.copy()
    iter_count = 0

        ac_y = fast_autocorr(y)

            if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        
        # Nesterov update
        t = float(iter_count)
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if iter_count == 1:
            x_new = g.copy()
        else:
            beta = (iter_count - 1.0) / (iter_count + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        prev_y = y.copy()
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_M:
            best_M = score_x
            best_a = x.copy()
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP iter={iter_count} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def mirror_descent_fast(a0, time_limit, top_k=3, beta=2.0, verbose=True):
    """Fast mirror descent with running average on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        step = 1.0 / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Use top-k subgradient
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            idx = np.arange(js, je)
            g[idx] += 2.0 * wi * a[k_int - idx]
        
        # Mirror descent
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        avg_a += a
        n_avg += 1
        
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = avg_check.copy()
            
            M_cur = fast_autocorr(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        top_peaks = np.argsort(ac)[::-1][:5]
        
        improved = False
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            idx = np.arange(js, je)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            
            # Try moving to random destinations
            candidates = rng.choice(n, size=min(100, n), replace=False)
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = a[src] - amt
                    a_trial[dst] = a[dst] + amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.02, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if
