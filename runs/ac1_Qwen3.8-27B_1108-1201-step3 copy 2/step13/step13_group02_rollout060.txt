from scipy.fft import rfft, irfft

def fast_autocorr(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def multiplicative_subgradient(a0, time_limit, verbose=True):
    """Multiplicative weighted descent on the max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)

For k* >= n-1: j ranges over [k*-(n-1), n-1]. The pairs are (j, k*-j) for j=k*-(n-1),...,n-1.

Now, `b_{k*}` as a function of `a`:
- It's a sum of products `a_j * a_{k*-j}`
- If k* < n-1: the pairs (j, k*-j) for j=0,...,k* involve indices 0, 1, ..., k* (each index appears in at most 2 pairs)

This is a crucial structural insight! The peak at lag k* only depends on a window of the sequence.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with candidate evaluation
2. Targeted peak shaving with structured moves
3. LSE-smoothed multiplicative gradient descent
4. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize(a, target_sum=None):
    """Normalize a to have the target sum (or sqrt(2n) if None)."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) * target_sum / n
    return a * target_sum / s

def coordinate_descent(a0, time_limit, verbose=True):
    """
    Exact/semi-exact coordinate descent.
    For each coordinate j, find the best a_j by evaluating a set of candidates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a)
    
    best_a = a.copy()
    best_b = fast_conv(a)
    best_max = best_b.max()
    t0 = time.monotonic()
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        b = best_b  # current convolution
        
        for j in range(n):
            s_other = a.sum() - a[j]
            if s_other < 1e-15:
                continue
            
            # Compute c_k = coefficient of a_j in b_k (linear part)
            # c_k = 2 * a_{k-j} for valid k (k-j in [0,n-1]) and k != 2j
            # c_k = a_j for k = 2j (quadratic term, approximate as linear: a_j)
            
            # For efficiency, compute the affected b_k values
            # b_k involves a_j when max(0,k-n+1) <= j <= min(n-1,k)
            # i.e., k in [j, j+n-1] intersect [0, 2n-2]
            k_lo = max(0, j)
            k_hi = min(2*n-2, j+n-1)
            
            # Coefficients for a_j in b_k for k in [k_lo, k_hi]
            # For k in [j, j+n-1]: k-j is in [0, n-1]
            # c_k = 2*a[k-j] if k != 2j, else a[j]
            
            # Current b_k values (with a_j included)
            # b_k = sum_{j'} a_{j'} * a_{k-j'}
            # The terms involving a_j: a_j * a[k-j] (as first factor) + a[k-j] * a_j (as second factor, if k-j != j)
            # = a_j * a[k-j] + (a[k-j] * a_j if k-j != j else a_j * a_j)
            # For k != 2j: 2 * a_j * a[k-j]
            # For k = 2j: a_j^2
            
            # If we set a_j = 0:
            b_base = b.copy()  # current b with a_j included
            # Subtract the contribution of a_j to get b_base (a_j = 0)
            # For k in [k_lo, k_hi]:
            for k in range(k_lo, k_hi + 1):
                kj = k - j  # = k - j, should be in [0, n-1]
                if kj < 0 or kj >= n:
                    continue
                if k == 2 * j:
                    b_base[k] -= a[j] * a[j]
                else:
                    b_base[k] -= 2 * a[j] * a[kj]
            
            # Now b_k(a_j) = b_base[k] + c_k * a_j (+ a_j^2 if k=2j)
            # For simplicity, ignore the quadratic term (k=2j)
            # c_k = 2 * a[k-j] for k in [k_lo, k_hi], k != 2j
            # c_k = a[j] for k = 2j (approximation)
            
            # Candidate values for a_j:
            candidates = set()
            candidates.add(0.0)
            candidates.add(max(0.0, a[j]))
            candidates.add(max(0.0, a[j] * 0.1))
            candidates.add(max(0.0, a[j] * 0.5))
            candidates.add(max(0.0, a[j] * 2.0))
            
            # For the current peak lag and top peaks:
            top_peaks = np.argsort(b_base)[::-1][:5]
            for k in top_peaks:
                if k < k_lo or k > k_hi:
                    continue
                kj = k - j
                if kj < 0 or kj >= n:
                    continue
                if k == 2 * j:
                    c_k = a[j]  # approximate
                else:
                    c_k = 2 * a[kj]
                if c_k > 1e-15:
                    # Optimal a_j for this k: minimize (b_base[k] + c_k * a_j) / (s_other + a_j)^2
                    # d/da_j = 0 => c_k * s_other - c_k * a_j - 2*b_base[k] = 0
                    # a_j = s_other - 2*b_base[k]/c_k
                    aj_cand = s_other - 2.0 * b_base[k] / c_k
                    if aj_cand > 0:
                        candidates.add(aj_cand)
                    # Also try the value that makes b_k = current max (or slightly less)
                    target = b.max() * 0.95
                    if b_base[k] < target:
                        aj_cand2 = (target - b_base[k]) / c_k
                        if aj_cand2 > 0:
                            candidates.add(aj_cand2)
            
            # Evaluate candidates
            best_cand_val = a[j]
            best_cand_score = b.max()
            
            for aj in candidates:
                aj = max(0.0, min(aj, s_other))  # keep reasonable
                if abs(aj - a[j]) < 1e-12:
                    continue
                # Compute new b
                b_new = b_base.copy()
                for k in range(k_lo, k_hi + 1):
                    kj = k - j
                    if kj < 0 or kj >= n:
                        continue
                    if k == 2 * j:
                        b_new[k] += aj * aj
                    else:
                        b_new[k] += 2 * aj * a[kj]
                # Also add the quadratic correction for k=2j if we used linear approx
                # (already handled above)
                
                max_new = b_new.max()
                # The actual score is 2n * max_new / (s_other + aj)^2
                # But since we're comparing relative to the current best,
                # and n is fixed, we can just compare max_new / (s_other+aj)^2
                score_cand = max_new / (s_other + aj) ** 2
                score_cur = best_cand_score / (a.sum()) ** 2
                
                if score_cand < score_cur:
                    best_cand_score = max_new
                    best_cand_val = aj
                    # Track actual score
                    actual_score = max_new / (s_other + aj) ** 2
                    cur_actual = b.max() / (a.sum()) ** 2
                    if actual_score < cur_actual:
                        improved = True
            
            if best_cand_val != a[j]:
                a[j] = best_cand_val
                a = normalize(a)
                b = fast_conv(a)
                if b.max() < best_max:
                    best_max = b.max()
                    best_a = a.copy()
        
        if verbose and n_passes % 5 == 0:
            print(f"    CD pass={n_passes} ({time.monotonic()-t0:.0f}s): best={best_max:.8f}")
        
        if not improved and n_passes > 1:
            break
    
    return best_a, best_max

"""

1. LP Nesterov acceleration (inherited approach, optimized)
2. Exact coordinate descent with incremental convolution updates (NEW)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a, target_sum=None):
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) * target_sum / n
    return a * target_sum / s


def build_conv_matrix_sparse(n):
    """Return (rows, cols) for the convolution matrix structure."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-30:
        return None
    return b_val


def lp_best_response_reduced(a, M, solver, T):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(4):
        circ = T @ b_val
        tol = 1e-8 * max(M, 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= (2 * n - 1) - 10:
            break
        if n_active < 30:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-6):
            b_val = b_new
        else:
            break

    s = b_val.sum()
    if s < 1e-30:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    best_a = a.copy()
    best_score = fast_conv(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break

        # Update T since y changed
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        if t < 30:
            b = lp_best_response(y, M_curr, solver)
        else:
            b = lp_best_response_reduced(y, M_curr, solver, T)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.15)
    T_end = max(0.001, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_conv(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        step = 1.0 / (T * n * 0.3)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_conv(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, max_passes=500):
    """
    Efficient coordinate descent with incremental convolution updates.
    For each coordinate j, finds the best a_j by evaluating candidates
    based on the top peaks. Uses O(n) incremental b updates per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a)
    b = fast_conv(a)
    total_sum = a.sum()

    best_a = a.copy()
    best_max = b.max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and n_passes < max_passes:
        n_passes += 1
        pass_improved = 0
        b = fast_conv(a)

        # Process coordinates in random order for diversity
        order = np.random.permutation(n)

        for j in order:
            s_other = total_sum - a[j]
            if s_other < 1e-15:
                continue

            # Affected lags: k in [j, j+n-1] ∩ [0, 2n-2]
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            n_affected = k_hi - k_lo + 1

            # b_base[k] = b[k] - contribution of a[j]
            # Contribution: for k in [k_lo, k_hi], k-j in [0, n-1]
            # If k != 2j: contrib = 2 * a[j] * a[k-j]
            # If k == 2j: contrib = a[j]^2

            kj_range = np.arange(k_lo - j, k_hi - j + 1)  # = 0 to (k_hi-j)
            b_base = b[k_lo:k_hi + 1].copy()

            # Subtract linear contributions
            quad_mask = (kj_range == j)  # k - j == j => k == 2j
            b_base[~quad_mask] -= 2.0 * a[j] * a[kj_range[~quad_mask]]
            if quad_mask.any():
                b_base[quad_mask] -= a[j] * a[j]

            # Coefficients for linear part
            c_vals = 2.0 * a[kj_range]
            if quad_mask.any():
                c_vals[quad_mask] = a[j]

            # Generate candidates
            candidates = [0.0, a[j] * 0.3, a[j] * 0.7, a[j] * 1.4]

            # From top peaks in b_base
            if n_affected > 1:
                top_idx = np.argpartition(b_base, -4)[::-1][:4]
            else:
                top_idx = np.array([np.argmax(b_base)])

            for idx in top_idx:
                c_k = c_vals[idx]
                bb = b_base[idx]
                if c_k > 1e-15:
                    # a_j that minimizes (bb + c_k * a_j) / (s_other + a_j)^2
                    # = s_other - 2*bb/c_k (from derivative = 0)
                    aj_opt = s_other - 2.0 * bb / c_k
                    if 0 < aj_opt < 10 * total_sum / n:
                        candidates.append(aj_opt)
                    # a_j that makes b_k at the current global max level
                    target = b.max()
                    if bb < target and c_k > 1e-15:
                        aj_target = (target - bb) / c_k
                        if 0 < aj_target:
                            candidates.append(aj_target)

            # Evaluate candidates
            best_aj = a[j]
            best_local_score = b.max() / (total_sum ** 2)

            # Precompute max outside affected range
            if k_lo > 0:
                max_left = b[:k_lo].max()
            else:
                max_left = -np.inf
            if k_hi < 2 * n - 2:
                max_right = b[k_hi + 1:].max()
            else:
                max_right = -np.inf
            max_outside = max(max_left, max_right)

            for aj in candidates:
                aj = max(0.0, min(aj, 5.0 * total_sum / n))
                if abs(aj - a[j]) < 1e-12 * max(1.0, a[j]):
                    continue

                if aj == 0.0:
                    b_new_aff = b_base
                else:
                    b_new_aff = b_base + aj * c_vals
                    if quad_mask.any():
                        b_new_aff = b_new_aff.copy()
                        b_new_aff[quad_mask] = b_base[quad_mask] + aj * aj

                max_aff = b_new_aff.max()
                max_new = max(max_aff, max_outside) if max_outside > -np.inf else max_aff
                new_sum = s_other + aj
                if new_sum < 1e-15:
                    continue
                score = max_new / (new_sum ** 2)

                if score < best_local_score:
                    best_local_score = score
                    best_aj = aj

            if best_aj != a[j]:
                # Update a[j] and b incrementally
                old_val = a[j]
                a[j] = best_aj
                total_sum += (best_aj - old_val)

                # Update b for affected lags
                delta = best_aj - old_val
                for offset, kj in enumerate(kj_range):
                    k = k_lo + offset
                    if kj == j:
                        # Quadratic: b_k += (aj^2 - old^2)
                        b[k] += best_aj * best_aj - old_val * old_val
                    else:
                        # Linear: b_k += 2 * delta * a[kj]
                        b[k] += 2.0 * delta * a[kj]

                if b.max() < best_max:
                    best_max = b.max()
                    best_a = a.copy()
                    pass_improved += 1

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_max:.8f}, improved={pass_improved}")

        if pass_improved == 0:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    # Final normalize and verify
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a * np.sqrt(2.0 * n) / s
        best_max = fast_conv(best_a).max()

    return best_a, best_max


def targeted_peak_shave(a0, time_limit, seed=42, verbose=True):
    """
    Targeted peak shaving: identifies peak-contributing coordinates and
    redistributes mass to low-impact positions using structured moves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_start = max(0.3, best_score * 0.05)
    T_end = max(0.01, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac

        b = fast_conv(a)
        current_max = b.max()
        # Find all lags near the peak (within 2% of max)
        peak_threshold = current_max * 0.98
        peak_lags = np.where(b >= peak_threshold)[0]

        improved = False
        for k in peak_lags[:6]:
            # Find contributing pairs for this lag
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            idx = np.arange(j_lo, j_hi + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Find low-impact destination positions
            # A position i is "low impact" for peak lag k if a[i] * a[k-i] is small
            # and i doesn't contribute to other peak lags
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_lags:
                j_lo2 = max(0, k2 - n + 1)
                j_hi2 = min(n - 1, k2)
                peak_mask[j_lo2:j_hi2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]

            # Select destinations: prefer low-value, non-peak positions
            n_dest = min(4, len(non_peak))
            # Weight by inverse value + randomness
            weights = 1.0 / (a[non_peak] + 1e-10) + rng.random(len(non_peak)) * 0.1
            top_dst = non_peak[np.argsort(weights)[::-1][:n_dest * 4]]
            selected_dst = top_dst[rng.choice(len(top_dst), min(n_dest, len(top_dst)), replace=False)]

            # Try different fraction amounts
            best_trial = None
            best_trial_score = current_max
            for frac in [0.03, 0.07, 0.12, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[src] -= amt
                # Distribute to destinations proportionally to inverse value
                dst_w = 1.0 / (a[selected_dst] + 1e-10)
                dst_w = dst_w / dst_w.sum()
                for di, dst in enumerate(selected_dst):
                    a_trial[dst] += amt * dst_w[di]

                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_conv(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial

            if best_trial is not None:
                if best_trial_score < best_score:
                    a = best_trial
                    best_score = best_trial_score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif best_trial_score < current_max and rng.random() < np.exp(-(best_trial_score - best_score) / max(T, 1e-10)):
                    a = best_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            # Random pairwise swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 150:
                # Larger structured perturbation
                n_pert = max(3, n // 80)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.15, 0.5)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                n_stall = 0

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def multiplicative_descent(a0, time_limit, verbose=True):
    """Multiplicative weighted descent using LSE-smoothed subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = normalize(a)

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.01, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_conv(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Multiplicative update
        lr = 1.0 / (n * T * 0.5 + 1.0)
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-10)
        a = normalize(a)
        steps += 1

        score = fast_conv(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score
