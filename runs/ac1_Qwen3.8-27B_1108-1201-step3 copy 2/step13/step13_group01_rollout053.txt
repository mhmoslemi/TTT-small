def build_conv_matrix(a, n):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns (b, dual)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_val


def lp_best_response_sparse(a, M, solver):

def normalize_to(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and backtracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0
    use_sparse = False

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Choose between dense and sparse based on timing
        if use_sparse:
            b, dual = lp_best_response_sparse(y, M_curr, solver)
        else:
            b, dual = lp_best_response(y, M_curr, solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Backtracking line search for step size
        best_alpha = 2.0 / (t + 2.0)
        best_trial_score = best_score
        for trial_alpha in [1.0, 0.5, 0.3, 2.0 / (t + 2.0), 0.1, 0.05]:
            y_trial = (1.0 - trial_alpha) * y + trial_alpha * g
            y_trial = np.maximum(y_trial, 0.0)
            y_trial = normalize_to(y_trial, norm)
            sc = fast_autocorr(y_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_alpha = trial_alpha
            if sc < fast_autocorr(y).max() * 0.999:
                break

        # Apply Nesterov momentum
        y_new = (1.0 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, norm)

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = normalize_to(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                last_improve = time.monotonic()

        # Stall detection with restart
        if time.monotonic() - last_improve > 20.0:
            stall_count += 1
            if stall_count > 5:
                # Perturb and reset
                rng = np.random.default_rng(t)
                y = np.maximum(y + rng.normal(0, 0.05 * norm / n, n), 0)
                y = normalize_to(y, norm)
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall_count = 0
                last_improve = time.monotonic()
                if verbose:
                    print(f"    Restart at iter={n_iters}, best={best_score:.8f}")
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mwu_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with multiplicative weights update."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, dual = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        n_iters += 1

        # MWU: use dual as "loss" signal
        # The gradient of the peak at y is: grad_i = y[k*-i] + y[i-k*]
        k_star = int(np.argmax(fast_autocorr(y)))
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += y[j2]
            grad[i] = 2.0 * val

        eta = 2.0 / (t + 2.0)
        # Multiplicative update: a_i *= exp(-eta * grad_i)
        log_w = -eta * grad
        log_w -= log_w.max()  # numerical stability
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(n) / n
        else:
            w = w / w_sum

        a_new = y * w
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, norm)

        y = a_new

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 30.0:
            stall_count += 1
            if stall_count > 8:
                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_gradient(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    # Compute gradient via FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):

        def obj_func(x):
            return lse_objective(x, n, T)

            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func, bounds=[(0, None)] * n,

def gradient_local_search(a0, time_limit, n_top_peaks=5, batch_size=10):
    """Fast gradient-based local search targeting top peaks with batch updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_top_peaks]
        peak_values = b[top_peaks]

        # Compute combined gradient (weighted by peak heights)
        combined_grad = np.zeros(n)
        for k_peak in top_peaks:
            weight = peak_values[list(top_peaks).index(k_peak)]
            for i in range(n):
                val = 0.0
                j1 = k_peak - i
                if 0 <= j1 < n:
                    val += a[j1]
                j2 = i - k_peak
                if 0 <= j2 < n:
                    val += a[j2]
                combined_grad[i] += 2.0 * weight * val

        # Batch update: move mass from top gradient positions to bottom
        n_move = min(batch_size, n // 4)
        if n_move < 1:
            break

        sorted_idx = np.argsort(combined_grad)
        sources = sorted_idx[::-1][:n_move]  # highest gradient
        targets = sorted_idx[:n_move]        # lowest gradient

        # Try moving a fraction of mass
        moved = False
        for frac in [0.02, 0.05, 0.01, 0.03, 0.08]:
            a_trial = a.copy()
            for src, dst in zip(sources, targets):
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial[src] -= amt
                a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_to(a_trial, norm)
            score = fast_autocorr(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                moved = True
                break

        if not moved:
            # Try individual moves for the worst peak
            k_star = top_peaks[0]
            grad_single = np.zeros(n)
            for i in range(n):
                val = 0.0
                j1 = k_star - i
                if 0 <= j1 < n:
                    val += a[j1]
                j2 = i - k_star
                if 0 <= j2 < n:
                    val += a[j2]
                grad_single[i] = 2.0 * val

            sorted_idx = np.argsort(grad_single)
            for frac in [0.01, 0.02, 0.03]:
                for trial_n in [3, 5, 8]:
                    a_trial = a.copy()
                    for src, dst in zip(sorted_idx[::-1][:trial_n],
                                        sorted_idx[:trial_n]):
                        amt = a_trial[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = normalize_to(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        moved = True
                        break
                if moved:
                    break

        if not moved:
            stall += 1
            if stall > 100:
                break
            # Random perturbation to escape
            if stall % 20 == 0:
                rng = np.random.default_rng(stall)
                a = np.maximum(a + rng.normal(0, 0.01 * norm / n, n), 0)
                a = normalize_to(a, norm)

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:6]

        improved = False
        for k_peak in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            # Find positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != top_contrib_idx[0])[0]
            candidates_sorted = candidates[np.argsort(-a[candidates])][:max(1, len(candidates) // 4)]

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                for dst in candidates_sorted[:4]:
                    for frac in [0.01, 0.03, 0.06, 0.12]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = normalize_to(a_trial, norm)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = normalize_to(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = (1 + 0.4 * np.cos(2 * np.pi * x)) * np.sin(np.pi * x) ** 0.5
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.7 * (1 + 0.2 * np.cos(4 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_length_scan(best_seq, solver, time_budget, rng):
    """Quick scan of multiple lengths to find the best one."""
    results = {}
    scan_lengths = [1200, 1600, 2000, 2400, 2800]

    # Filter to reasonable lengths based on what we already have
    if best_seq is not None:
        base_n = len(best_seq)
        scan_lengths = sorted(set([
            base_n,
            base_n - 200,
            base_n + 200,
            base_n + 400,
            base_n - 400,
        ]))
        scan_lengths = [x for x in scan_lengths if 800 <= x <= 3200]
        if not scan_lengths:
            scan_lengths = [base_n]

    for n_scan in scan_lengths:
        elapsed = time.monotonic() - t_multi_start
        remaining = time_budget - elapsed
        if remaining < 30:
            break
        scan_time = min(remaining / max(1, len(scan_lengths) - list(scan_lengths).index(n_scan)), 120.0)
        if scan_time < 15:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = normalize_to(init_seq, np.sqrt(2.0 * n_scan))
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        results[n_scan] = (a_out, sc_out)
        print(f"    Scan n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_multi_start:.0f}s)")

    return results


t_multi_start = 0.0


def propose_candidate(seed=42, budget_s=1800.0):
    global t_multi_start
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan (15% of time)
    print("\n=== Phase 0: Multi-length scan ===")
    t_multi_start = time.monotonic()
    scan_time = time_budget * 0.15
    scan_results = multi_length_scan(best_seq, solver, scan_time, rng)

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP optimization at base length (45% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: Gradient local search (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Grad local search n={n_best} (budget={remaining:.0f}s) ===")
    t2 = remaining * 0.15
    if t2 > 20:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = gradient_local_search(a_norm, t2, n_top_peaks=5, batch_size=8)
        do_try_update(a_out, f"grad_local_n{n_best}")

    # Phase 3: LSE smooth optimization (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=40.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Try a longer length (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 4: LP n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 200.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = normalize_to(init_long, np.sqrt(2.0 * n_long))
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 5: MWU optimization (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: MWU n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = mwu_lp_optimize(a_norm, t5, solver, verbose=False)
        do_try_update(a_out, f"mwu_n{n_best}")

    # Phase 6: Coordinate descent polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.5
    if t6 > 10:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = coordinate_descent_polish(a_norm, t6, seed=12345)
        do_try_update(a_out, "coord_polish")

    # Phase 7: Final LSE polish with low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.4, 100.0)
    if t7 > 10:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_start=8.0, T_end=0.5, n_stages=3)
        do_try_update(a_out, "lse_polish")

    # Phase 8: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.5
    if t8 > 5:
        a_norm = normalize_to(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = coordinate_descent_polish(a_norm, t8, seed=77777)
        do_try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
