def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # If all elements are already non-negative and sum is close
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-10:

        return np.full(n, target_sum / n)
    
    theta = np.sort(v)[::-1]
