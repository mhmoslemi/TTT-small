def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def lse_gradient_full(z, T_temp, n):
    """
    Compute LSE objective and gradient in z-space.
    a = softmax(z), objective = (1/T) log(sum_k exp(T * b_k))
    Returns (objective, gradient_z)
    """
    a = softmax_fn(z)
    b = fast_autocorr(a)
    
    # Compute LSE
    T = T_temp
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of sum_k w_k * b_k w.r.t. a
    # db_k / da_j = 2 * a_{k-j} if 0 <= k-j < n
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return lse_val, grad_z

        top_idx = np.argsort(b)[::-1][:n_peaks]
        top_vals = b[top_idx]
        
        # Compute the "danger" for each position: how much it contributes to top peaks
        danger = np.zeros(n)
        for k_idx in range(n_peaks):
            k = top_idx[k_idx]
            weight = top_vals[k_idx] / top_vals[0]  # Relative importance
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # Contribution of position j to peak k is a[j] * a[k-j]
            danger[idx] += weight * a[k - idx]
        
        # Find the most dangerous position
        src = np.argmax(danger)
        
        # Find safe destinations (low danger)
        safe_mask = danger < danger.max() * 0.1
        if not safe_mask.any():
            safe_mask = danger < danger.max() * 0.5
        safe_positions = np.where(safe_mask)[0]
        safe_positions = safe_positions[safe_positions != src]
        
        if len(safe_positions) == 0:
            safe_positions = np.where(np.arange(n) != src)[0]
        
        # Sort safe positions by how little they contribute to peaks
        safe_sorted = safe_positions[np.argsort(danger[safe_positions])]
        dst_pool = safe_sorted[:min(10, len(safe_sorted))]
        
        if len(dst_pool) == 0:
            break
        
        # Try moves
        found = False
        for dst in dst_pool:
            if found:
                break
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                new_max = fast_autocorr(a_trial).max()
                if new_max < bmax * (1 - 1e-9):
                    a = a_trial
                    new_score = 2.0 * n * new_max
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    found = True
                    stall = 0
                    break
        
        if not found:
            stall += 1
            # Random perturbation
            if stall % 5 == 0:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 150:
                break
    
    return best_a, best_score


def spectral_smooth(a, alpha=0.1):
    """Apply light spectral smoothing to reduce high-frequency oscillations."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length() - 1
    # Actually, let's just use a simple approach
    A = rfft(a, 2 * n)
    # Apply a gentle low-pass: reduce high frequencies
    freqs = np.arange(len(A))
    # Keep frequencies below n//2 mostly intact, smooth the rest
    mask = np.ones(len(A))
    high_start = len(A) // 3
    mask[high_start:] = alpha
    A_smooth = A * mask
    a_smooth = irfft(A_smooth, 2 * n)[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s > 1e-15:
        a_smooth = a_smooth / s
    return a_smooth
