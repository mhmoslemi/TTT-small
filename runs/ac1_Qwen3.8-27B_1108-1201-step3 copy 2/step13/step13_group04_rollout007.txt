def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = autocorr(a).
    dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k])
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

def mirror_descent_peaks(a0, time_limit, top_k=5, eta_start=1.0, decay=0.995, verbose=True):
    """Mirror descent (multiplicative weights) using peak subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0
    eta = eta_start

    while time.monotonic() - t0 < time_limit:
        g = peak_subgradient(a, top_k)
        
        # Mirror descent step: a <- a * exp(-eta * g), then normalize
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        a = a_new
        it_count += 1
        eta *= decay

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-Peaks it={it_count} eta={eta:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=12, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        a = np.ones(n) * norm / n

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(50, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:

            T_local = T * (1.0 + 0.5 * np.cos(np.pi * it / max(1, n_iters_stage)))

            grad = lse_gradient(y, T_local)

            if it_count == 1:

            else:
                momentum = (it_count - 1.0 + np.sqrt(1.0 + 4.0 * (it_count - 1.0) ** 2)) / (2.0 * (it_count + 2.0))

                if s_ye > 1e-15:
                    y_extrap = y_extrap * norm / s_ye
                grad_extrap = lse_gradient(y_extrap, T_local)
                y_new = y + momentum * (y - y_prev) - (1.0 / (it_count + 3.0)) * grad_extrap

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)

            if s_y < 1e-15:

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()
    n_iters = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
            
        # Use a slightly relaxed M to allow some flexibility
        M_relaxed = M_curr * 1.001
        
        b = lp_best_response(y, M_relaxed, solver)
        if b is None:
            # Try tighter
            b = lp_best_response(y, M_curr, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation
        y_extrap = y_new + (t / (t + 2.0)) * (y_new - y)
        y_extrap = np.maximum(y_extrap, 0.0)
        s_ye = y_extrap.sum()
        if s_ye > 1e-15:
            y_extrap = y_extrap * norm / s_ye

        prev_y = y.copy()
        y = y_new

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        score_e = fast_autocorr(y_extrap).max()
        if score_e < best_score:
            best_score = score_e
            best_a = y_extrap.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Identify top peaks
        top_ks = np.argsort(b)[::-1][:10]
        
        improved = False
        
        # Strategy 1: Move mass from peak-contributing positions to non-peak positions
        for k_star in top_ks[:5]:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            # Build peak mask (positions that contribute to ANY top peak)
            peak_contrib = np.zeros(n, dtype=bool)
            for k2 in top_ks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_contrib[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < norm * 1e-6:
                    continue
                # Try moving to various destinations
                # Prioritize destinations that don't contribute to peaks
                dest_candidates = np.where(~peak_contrib)[0]
                if len(dest_candidates) > 0:
                    # Sort by current value (prefer low-value positions)
                    dest_candidates = dest_candidates[np.argsort(a[dest_candidates])[:10]]
                else:
                    dest_candidates = np.array([i for i in range(n) if i != src][:10])

                for dst in dest_candidates:
                    # Try different transfer fractions
                    for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-14:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        # Strategy 2: Pair swap (exchange values between two positions)
        if not improved and rng.integers(3) == 0:
            # Find pair of positions whose swap reduces the peak
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0 and a[j] > 0:
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a[j], a[i]
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it_count += 1

        # Strategy 3: Random perturbation
        if not improved:
            stall += 1
            if stall % 5 == 0 or rng.integers(10) == 0:
                # Gaussian perturbation
                pert = rng.normal(0, norm * 0.001, n)
                a_trial = a + pert
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        it_count += 1

            if stall > 500:
                # Bigger perturbation
                pert = rng.normal(0, norm * 0.01, n)
                a_trial = a + pert
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

                if stall > 1000:
                    if verbose:
                        print(f"    LocalSearch: stalled at {best_score:.8f}")
                    break

        if verbose and it_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Cauchy-like
        x = np.linspace(-2, 2, n)
        a = 1.0 / (1.0 + x ** 2)
    elif choice == 6:
        # Flat with small oscillation
        a = np.ones(n) * 0.9 + 0.1 * np.cos(2 * np.pi * np.arange(n) * 3 / n)
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
