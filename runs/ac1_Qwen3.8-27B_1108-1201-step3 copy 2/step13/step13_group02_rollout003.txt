"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start LP Nesterov with diverse initializations
2. Adaptive active-set LP for speed
3. Coordinate descent polish (1D minimization per coordinate)

5. Length exploration around the best found n

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global solver
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCIP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, x)) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lp_best_response_reduced(a, M, solver, max_rounds=3):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # First full solve
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Iteratively reduce to active constraints
    for _ in range(max_rounds):
        circ = T @ b_val
        tol = max(1e-8 * M, 1e-12)
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 5 or n_active < 10:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-6):
            b_val = b_new
        else:
            break
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        if t < 30:
            b = lp_best_response(y, M_curr, solver)
        else:
            b = lp_best_response_reduced(y, M_curr, solver)
        
        if b is None:
            stall += 1
            if stall > 100:
                break
            continue
        
        n_iters += 1
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Coordinate descent: optimize each coordinate to minimize max autocorr."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break
            
            v = a[i]
            if v < 1e-15:
                continue
            
            # Try moving v up and down
            # Compute the effect on autocorrelation
            # b_new[k] = b_old[k] + delta_k
            # For k in [i, i+n-1] (valid range): delta = 2*(v'-v)*a[k-i] if k != 2i
            # For k = 2i: delta = v'^2 - v^2
            
            b = fast_autocorr(a)
            M_curr = b.max()
            
            # Try several candidate values
            candidates = [v * 0.9, v * 1.1, v * 0.8, v * 1.2, v * 0.95, v * 1.05]
            # Also try adding/subtracting a small amount
            delta = 0.01 * norm / n
            candidates.extend([v - delta, v + delta, v - 2*delta, v + 2*delta])
            candidates = [max(0.0, c) for c in candidates]
            
            best_v = v
            best_M = M_curr
            
            for v_try in candidates:
                if abs(v_try - v) < 1e-15:
                    continue
                a[i] = v_try
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = v
                    continue
                a = a * norm / s
                M_try = fast_autocorr(a).max()
                if M_try < best_M:
                    best_M = M_try
                    best_v = v_try
                a[i] = v  # restore for next candidate (will renormalize after)
            
            if best_v != v and best_M < M_curr:
                a[i] = best_v
                s = a.sum()
                a = a * norm / s
                if best_M < best_score:
                    best_score = best_M
                    best_a = a.copy()
                improved = True
        
        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, improved={improved}")
        
        if not improved and n_passes > 10:
            # Stalled, try larger perturbations
            for _ in range(5):
                i = rng.integers(n)
                v = a[i]
                if v < 1e-10:
                    continue
                frac = rng.uniform(0.2, 0.5)
                a[i] = v * (1 - frac)
                j = rng.integers(n)
                a[j] += v * frac
                a = np.maximum(a, 0)
                s = a.sum()
                a = a * norm / s
                M_try = fast_autocorr(a).max()
                if M_try < best_score:
                    best_score = M_try
                    best_a = a.copy()
                    break
                else:
                    # revert
                    a[i] = v * (1 - frac)
                    a[j] += v * frac
                    a = a * norm / a.sum()
            break
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient via FFT
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Adaptive step size (Barzilai-like)
        step = 1.0 / (T * max(1.0, np.max(np.abs(grad))))
        step = min(step, 5.0 / n)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def make_gaussian_init(n, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / width) ** 2)
    return a / a.sum()

def make_cosine_init(n, power=1.0):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_plateau_init(n, frac=0.6):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    mask = x < (1 - frac) / 2 or x > 1 - (1 - frac) / 2
    a[mask] = 0
    # Smooth edges
    edge_width = int(n * 0.05)
    for i in range(edge_width):
        t = i / edge_width
        start = int(n * (1 - frac) / 2)
        if 0 <= start - edge_width + i < n:
            a[start - edge_width + i] = max(a[start - edge_width + i], t)
        if 0 <= start + frac * n - i < n:
            a[start + frac * n - i] = max(a[start + frac * n - i], 1 - t)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

Wait, I realize the coordinate descent as I wrote it is O(n) per coordinate evaluation (because of the FFT autocorrelation), and we try ~8 candidates per coordinate, so it's O(8n) per coordinate, O(8n^2) per pass. For n=1600, that's about 20M operations per pass. Each FFT is O(n log n) ≈ 1600 * 11 ≈ 17K operations. So per coordinate: 8 * 17K = 136K. Per pass: 1600 * 136K = 218M. That's too slow!

I need to make the coordinate descent faster. The key is: instead of computing the full autocorrelation for each candidate, I should compute it incrementally.

When we change a[i] from v to v', the change in autocorrelation is:
- For k in [i, i+n-1] (clamped to [0, 2n-2]): b[k] changes by 2*(v'-v)*a[k-i] (if k != 2i)
- For k = 2i: b[k] changes by v'^2 - v^2

So I can maintain the autocorrelation array and update it in O(n) per coordinate change (not O(n log n)).

Let me rewrite the coordinate descent to be O(n^2) per pass instead of O(n^2 * log n).

Actually, even O(n^2) per pass with 8 candidates is 8 * 1600 * 1600 = 20M per pass. At ~1 ns per operation (which is optimistic for Python), that's 20 seconds per pass. We can only afford a few passes in 200 seconds.

Better approach: instead of trying 8 candidates, do a smart 1D optimization. The function `max_k b_new[k]` as a function of v' is convex (piecewise linear + one quadratic piece). We can find its minimum by:
1. Computing the slopes of each piece: slope_k = 2*a[k-i] for k in the range
2. The minimum is where the maximum slope crosses zero
3. This can be found by sorting the slopes and finding the median-like point

Actually, let me think more carefully. The function is:
`f(v) = max( max_{k: 2i != k, k in range} (b_k + 2*(v-v0)*s_k), b_{2i} + v^2 - v0^2, max_{k: k not in range} b_k )`

where `s_k = a[k-i] >= 0` and `v0 = a[i]`.

The first term is `max_k (b_k + 2*s_k * v - 2*s_k*v0)` = `max_k (2*s_k * v + c_k)` where `c_k = b_k - 2*s_k*v0`.

This is the max of linear functions, hence convex and piecewise linear. The minimum is where the "upper envelope" has a kink (or where it crosses the quadratic piece).

A simpler approach: binary search on v. For a given v, compute f(v) in O(n) (recompute the affected b values). Binary search takes O(log(precision)) ≈ 30 iterations. Total: O(30n) per coordinate, O(30n^2) per pass. For n=1600: 30 * 2.5M = 75M per pass. Still too slow in Python.

Let me think of a faster approach. What if I just do a gradient step?

The "gradient" of `max_k b[k]` at the current point is: for the peak lag k*, the gradient component at position i is `a[k*-i]` (or `2*a[k*-i]` if there are two contributions). 

So a simple coordinate descent step is:
`a[i] <- max(0, a[i] - step * a[k* - i])`

where k* is the current argmax of b. This is O(1) per coordinate! And since we're just doing a subgradient step, it should work.

But this is essentially the same as the LSE gradient with T -> 0 (only the peak lag contributes).

Let me use a hybrid: maintain the full autocorrelation array, and for each coordinate, compute the "pressure" (contribution to the max) and do a small step. The autocorrelation update after changing a[i] is O(n), so a full pass is O(n^2).

For n=1600, O(n^2) = 2.5M operations per pass. In numpy, this should take about 2-5ms per pass. We can do thousands of passes in 200 seconds.
