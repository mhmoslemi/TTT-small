def fast_autocorr(a):

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, tight_factor=1.0, use_adaptive=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        M_use = M_curr * tight_factor
        b = lp_best_response(y, M_use, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if use_adaptive and best_score > 0:
            ratio = M_curr / best_score
            alpha = 2.0 / (t + 2.0)
            alpha = min(alpha * min(max(ratio, 0.5), 2.0), 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 1:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg

                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 15:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_power_value_and_grad(a_flat, n, p):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    B_max = b.max()
    if B_max < 1e-15:
        return 1e10, np.zeros(n)
    r = b / B_max
    r_clipped = np.maximum(r, 1e-30)
    r_p = r_clipped ** p
    C_p_sum = r_p.sum()
    C = C_p_sum ** (1.0 / p)
    lp_val = B_max * C

    s_pow = r_clipped ** (p - 1)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    s_pad = np.zeros(fft_size)
    s_pad[:2 * n - 1] = s_pow
    A_f = rfft(a_pad, fft_size)
    S_f = rfft(s_pad, fft_size)
    cross = irfft(np.conj(A_f) * S_f, fft_size)[:n]

    coeff = 2.0 * C ** (1.0 - p)
    conv_grad = coeff * cross

    obj_val = 2.0 * n * lp_val / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return float(obj_val), grad.ravel()


def lp_scipy_optimize(a0, time_limit, p_values=None, verbose=True):

def multi_peak_gradient_step(a, n_peaks=5, step_size=0.01):
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if s < 1e-10:
        return a.copy()
    top_k = np.argsort(b)[::-1][:n_peaks]
    weights = b[top_k] ** 2
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return a.copy()
    weights = weights / w_sum

    grad = np.zeros(n)
    for w_k, k in zip(weights, top_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k - idx]
        grad[idx] += 2.0 * w_k * contrib / max(b[k], 1e-15)

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a.copy()
    norm = np.sqrt(2.0 * n)
    a_new = a_new * norm / s_new
    return a_new


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        num_peaks = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(num_peaks):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-((x - center) / width) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 4) * x) + 0.2 * np.cos(2 * np.pi * rng.uniform(4, 8) * x)
        a = np.maximum(a, 0.001)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.05):
    n = len(a)
    a_pert = a.copy()
    mask = rng.random(n) < strength
    a_pert[mask] *= rng.uniform(0.5, 1.5, mask.sum())
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    norm = np.sqrt(2.0 * n)
    return a_pert * norm / s
