Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Corrected LSE gradient descent with Nesterov acceleration (was using wrong gradient)
2. LP best-response Frank-Wolfe with vectorized matrix construction
3. Multi-peak coordinated local search
4. Multi-length exploration
Key fix: The LSE gradient was incorrectly computed as r1+r2; the correct formula
is 2*conv(softmax(b/T), rev(a))[n-1:2n-1], giving a better descent direction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver


def fast_conv(a):
    """Compute full conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def lse_gradient_correct(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = conv(a,a).
    Correct formula: dL/da[m] = 2 * sum_j w[m+j] * a[j] = 2 * conv(w, rev(a))[m+n-1]
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    
    # Compute conv(w, rev(a)) and extract [n-1 : 2n-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)
    conv_result = irfft(W * A_rev, fft_size)
    r = conv_result[n - 1:2 * n - 1]
    
    return 2.0 * r


def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.3, n_stages=14, verbose=True):
    """Nesterov-accelerated LSE gradient descent with correct gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    it_count = 0
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters_stage = max(50, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov extrapolation
            if it == 0:
                y_extrap = y
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e > 1e-15:
                    y_extrap = y_extrap * norm / s_e
            
            grad = lse_gradient_correct(y_extrap, T)
            step = 1.0 / (it + 3.0)
            y_new = y - step * grad
            
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            y = y_new
            it_count += 1
            
            score = fast_conv(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            
            if it_count % 200 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    j = np.arange(n)
    rows = np.repeat(j, n) + np.tile(np.arange(n), n)
    cols = np.repeat(j, n)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = build_conv_matrix(a)
    M_vec = np.ones(2 * n - 1) * M
    constraints = [T_mat @ b_var <= M_vec]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP best-response Frank-Wolfe with acceleration and averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    it_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        it_count += 1
        
        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            y_extrap = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_extrap = y_new + momentum * (y_new - y)
            y_extrap = np.maximum(y_extrap, 0.0)
            s_e = y_extrap.sum()
            if s_e < 1e-15:
                y_extrap = y_new.copy()
            else:
                y_extrap = y_extrap * norm / s_e
        
        y_prev = y.copy()
        y = y_new
        # Use extrapolated point for next LP
        y_search = y_extrap
        
        # Update best
        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_ye = fast_conv(y_extrap).max()
        if score_ye < best_score:
            best_score = score_ye
            best_a = y_extrap.copy()
        
        # Averaging
        avg_a += y
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_conv(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and it_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def multi_peak_local_search(a0, time_limit, seed=42, verbose=True):
    """Multi-peak coordinated local search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_b = b.max()
        threshold = max_b * 0.999  # Consider all peaks within 0.1% of max
        peak_indices = np.where(b >= threshold)[0]
        
        improved = False
        # For each significant peak, try to reduce it
        for k_star in peak_indices[:10]:  # Limit to top 10 peaks
            if improved:
                break
            
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-16:
                continue
            
            # Top contributors
            top_n = min(5, len(idx))
            top_pos = idx[np.argsort(contrib)[::-1][:top_n]]
            
            for src in top_pos:
                if a[src] < norm * 1e-10:
                    continue
                
                # Find "safe" positions (not contributing to any significant peak)
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in peak_indices[:10]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                
                safe_positions = np.where(~peak_mask)[0]
                if len(safe_positions) < 3:
                    safe_positions = np.where(np.arange(n) != src)[0]
                
                # Pick lowest-value safe positions
                candidates = safe_positions[np.argsort(a[safe_positions])[:5]]
                
                for dst in candidates:
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if rng.integers(8) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.005, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            
            if stall > 300:
                if verbose:
                    print(f"    LocalSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break
        
        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
