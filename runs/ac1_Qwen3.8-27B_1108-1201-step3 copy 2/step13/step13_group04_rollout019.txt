    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_value(a, T):
    """Compute LSE value: T * log(sum exp(conv_k / T))."""
    c = fast_conv(a)
    max_c = c.max()
    return T * (max_c / T + np.log(np.sum(np.exp((c - max_c) / T))))


def lse_grad(a, T):
    """
    Gradient of LSE surrogate T*log(sum exp(conv_k/T)).
    Returns gradient vector of length n.

    A = rfft(a, pad_size)
    W = rfft(w, pad_size)

    T_end = max_b0 * 0.005

        a = y.copy()

        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            cur_score = fast_conv(a).max()
            print(f"  FISTA stage {si}/{n_stages} T={T:.6f} ({elapsed:.0f}s): "
                  f"cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Subgradient method for min max_k conv(a,a)_k subject to sum(a)=1, a>=0.
    Uses random active constraint selection + Polyak averaging.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    c = fast_conv(a)
    max_b = c.max()
    # Polyak step size: eta_t = (max_b_target - f_t) / ||g_t||^2
    # We use max_b * 0.99 as target
    target = max_b * 0.995

    avg_a = np.zeros(n)
    n_avg = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        max_b = c.max()

        # Select top-k active constraints randomly
        sorted_idx = np.argsort(c)[::-1]
        k_top = min(5, 2 * n - 1)
        active = sorted_idx[:k_top]
        k_star = active[rng.integers(k_top)]

        # Subgradient: d conv[k_star]/da[m] = a[k_star - m] + a[m - k_star]
        m_idx = np.arange(n)
        g1_idx = k_star - m_idx
        g2_idx = m_idx - k_star

        g = np.zeros(n)
        mask1 = (g1_idx >= 0) & (g1_idx < n)
        g[mask1] += a[g1_idx[mask1]]
        mask2 = (g2_idx >= 0) & (g2_idx < n)
        g[mask2] += a[g2_idx[mask2]]

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break

        # Polyak step size
        gap = max_b - target
        if gap < 1e-12:
            eta = 1.0 / np.sqrt(it_count + 1) * 0.01
        else:
            eta = gap / g_norm_sq
            eta = min(eta, 0.1)  # cap step size

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        it_count += 1

        # Update Polyak average
        avg_a += a
        n_avg += 1

        # Check best (both current and average)
        if it_count % 10 == 0:
            score = fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if n_avg >= 50 and it_count % 50 == 0:
                avg_check = avg_a / n_avg
                score_avg = fast_conv(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Subgrad it={it_count} ({elapsed:.0f}s): "
                  f"max_b={max_b:.8f} best={best_score:.8f}")

    return best_a, best_score


def local_search_1swap(a0, time_limit, verbose=True):
    """
    Exhaustive 1-swap local search: for each source position i, try moving
    a fraction to all other positions j, keep the best improvement.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, fast_conv(a).max()
    a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall_count = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        max_val = c.max()
        # Find positions contributing to peak
        peak_positions = np.where(c > max_val * 0.99)[0]

        improved = False
        # Try moving from high-mass positions to low-mass positions
        src_candidates = np.argsort(a)[::-1][:min(50, n // 4)]

        for src in src_candidates:
            if a[src] < 1e-10:
                continue
            amt = a[src] * 0.05

            # Precompute the "influence" of source on convolution
            # Moving amt from src to dst changes conv[k] by:
            # amt * (a[k-src] + a[src-k] - a[k-dst] - a[dst-k]) (linear approx)

            best_dst = -1
            best_new_max = max_val

            # Try a subset of destinations
            dst_candidates = np.argsort(a)[:min(50, n // 4)]
            # Also include some random positions
            extra = rng.integers(n, size=min(20, n))
            dst_candidates = np.unique(np.concatenate([dst_candidates, extra]))

            for dst in dst_candidates:
                if dst == src:
                    continue
                # Compute change in convolution
                k_all = np.arange(2 * n - 1)
                # a[k-src]: non-zero when 0<=k-src<n
                delta = np.zeros(2 * n - 1)
                m1 = np.arange(2 * n - 1) - src
                mask1 = (m1 >= 0) & (m1 < n)
                delta[mask1] += a[m1[mask1]]

                m2 = src - np.arange(2 * n - 1)
                mask2 = (m2 >= 0) & (m2 < n)
                delta[mask2] += a[m2[mask2]]

                m3 = np.arange(2 * n - 1) - dst
                mask3 = (m3 >= 0) & (m3 < n)
                delta[mask3] -= a[m3[mask3]]

                m4 = dst - np.arange(2 * n - 1)
                mask4 = (m4 >= 0) & (m4 < n)
                delta[mask4] -= a[m4[mask4]]

                new_c = c + amt * delta
                new_max = new_c.max()

                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_dst = dst

            if best_dst >= 0 and best_new_max < max_val - 1e-14:
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                actual_max = fast_conv(a).max()
                if actual_max < best_score:
                    best_score = actual_max
                    best_a = a.copy()
                    improved = True
                    stall_count = 0

        if not improved:
            # Try smaller amounts
            for frac in [0.01, 0.02, 0.03, 0.015, 0.005]:
                for src in rng.integers(n, size=min(20, n)):
                    if a[src] < 1e-10:
                        continue
                    dst = rng.integers(n)
                    if dst == src:
                        continue
                    amt = a[src] * frac
                    a_tr = a.copy()
                    a_tr[src] -= amt
                    a_tr[dst] += amt
                    a_tr = np.maximum(a_tr, 0.0)
                    s = a_tr.sum()
                    if s < 1e-15:
                        continue
                    a_tr = a_tr / s
                    sc = fast_conv(a_tr).max()
                    if sc < best_score - 1e-12:
                        a = a_tr
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall_count = 0
                        break
                if improved:
                    break

        if not improved:
            stall_count += 1
            if stall_count > 30:
                break
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

        sweep += 1
        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LocalSearch sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) on LSE surrogate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    c0 = fast_conv(a)
    max_b0 = c0.max()
    T_start = max_b0 * 0.3
    T_end = max_b0 * 0.01
    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)

    it_count = 0
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters = max(30, int(remaining * 2.5 / max(1, n_stages - si)))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = lse_grad(a, T)
            eta = 1.0 / (T * (it + 1) ** 0.4)
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new
            it_count += 1

            if it_count % 10 == 0:
                score = fast_conv(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed):
    """Generate diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 + 0.3 * np.cos(2 * np.pi * x)) / 1.3
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = x ** 0.5 * (1 - x) ** 0.3
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8 * x ** 2) + 0.5 * np.exp(-8 * (x - 0.8) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(12 * np.pi * x)
    elif choice == 5:
        a = np.ones(n)
        a[:n // 2] *= 1.3
        a[n // 2:] *= 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = x ** 0.7
    else:
        a = np.ones(n) * 0.8
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.02, n * 0.15)
            amp = rng.uniform(-0.3, 0.4)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining FISTA-LSE, subgradient,
    mirror descent, and exhaustive local search at multiple lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: FISTA-LSE at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA-LSE n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = fista_lse(a_work, t1, verbose=True)
        do_try_update(a_out, f"fista_n{n_base}")

    # ============ Phase 2: Subgradient at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Subgradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 15:
        a_work = best_seq.copy()
        a_out, _ = subgradient_optimize(a_work, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # ============ Phase 3: Mirror descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MirrorDescent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_work = best_seq.copy()
        a_out, _ = mirror_descent(a_work, t3, verbose=True)
        do_try_update(a_out, f"mirror_n{n_best}")

    # ============ Phase 4: Local search on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LocalSearch n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 15:
        a_work = best_seq.copy()
        a_out, _ = local_search_1swap(a_work, t4, verbose=True)
        do_try_update(a_out, f"local_n{n_best}")

    # ============ Phase 5: Try longer length ============
    n2 = n_best + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: FISTA n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        init_long = interpolate_seq(best_seq, n2)
        a_out, _ = fista_lse(init_long, t5, verbose=True)
        do_try_update(a_out, f"fista_n{n2}")
        # Local search on longer
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.10, 100.0)
        if t5b > 10:
            a_work = best_seq if len(best_seq) == n2 else interpolate_seq(best_seq, n2)
            a_out2, _ = local_search_1swap(a_work, t5b, verbose=False)
            do_try_update(a_out2, f"local_n{n2}")

    # ============ Phase 6: Try shorter length ============
    n3 = max(n_best - 100, 500)
    if n3 != n_best:
        remaining = time_budget - (time.monotonic() - t_global)
        print(f"\n=== Phase 6: FISTA n={n3} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.10, 100.0)
        if t6 > 10:
            init_short = interpolate_seq(best_seq, n3)
            a_out, _ = fista_lse(init_short, t6, verbose=False)
            do_try_update(a_out, f"fista_n{n3}")

    # ============ Phase 7: Final local search polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 5:
            print(f"\n=== Phase 7: Final LocalSearch n={n_best} (budget={t7:.0f}s) ===")
            a_work = best_seq.copy()
            a_out, _ = local_search_1swap(a_work, t7, verbose=True)
            do_try_update(a_out, "local_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
