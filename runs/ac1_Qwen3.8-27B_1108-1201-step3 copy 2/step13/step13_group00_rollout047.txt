        full_conv = fast_autocorr(a)
        # Actually need conv(a, b_val)
        # Compute via: for each i in b, add a shifted by i
        # Use the matrix
        T_full_rows = build_conv_matrix_fast(a, np.setdiff1d(np.arange(size), active_indices))
        if len(T_full_rows) > 0:
            viol = T_full_rows @ b_val
            if viol.max() > M * (1 + 1e-4):
                # Expand active set
                violated_idx = np.where(viol > M * (1 + 1e-4))[0]
                # Map back to original indices
                dropped = np.setdiff1d(np.arange(size), active_indices)
                new_active = np.union1d(active_indices, dropped[violated_idx])
                if len(new_active) > 0.95 * size:
                    # Just use all
                    b_var2 = cp.Variable(n, nonneg=True)
                    T2 = build_conv_matrix_fast(a, np.arange(size))
                    constraints2 = [T2 @ b_var2 <= M]
                    prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                    try:
                        prob2.solve(solver=solver, verbose=False)
                    except Exception:
                        pass
                    if prob2.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    return b_val, np.arange(size)
                # Re-solve with expanded set
                b_var2 = cp.Variable(n, nonneg=True)
                T2 = build_conv_matrix_fast(a, new_active)
                constraints2 = [T2 @ b_var2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    pass
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_indices = new_active

    return b_val, active_indices


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_set = None

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            b, _ = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_guided_cd(a, time_limit, rng, verbose=True):
    """
    Coordinate descent with gradient-informed pair selection.
    Uses the subgradient of max autocorrelation to identify promising moves,
    then performs exact 2-variable line search.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_streak = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Compute subgradient at current max
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += a[j2]
            grad[i] = 2.0 * v

        # Select promising pairs: high grad -> low grad
        # Top sources (high grad) and top sinks (low grad)
        n_top = max(5, n // 40)
        sources = np.argsort(grad)[-n_top:]
        sinks = np.argsort(grad)[:n_top]

        # Also add random pairs for exploration
        n_random = max(3, n // 50)

        candidates = []
        for i in sources:
            if a[i] < norm * 1e-9:
                continue
            for j in sinks:
                if j != i and a[j] < norm * 100.0:
                    candidates.append((int(i), int(j)))

        for _ in range(n_random):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                candidates.append((i, j))

        rng.shuffle(candidates)

        for (i, j) in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-9:
                continue

            # Compute the effect of transferring delta from i to j
            # Change in conv_k = -delta*(a_{k-i} + a_{i-k}) + delta*(a_{k-j} + a_{j-k})
            #                    + second-order terms at k=2i and k=2j
            # For efficiency, compute the "gain vector" g_k
            g = np.zeros(size)
            # Subtract contribution from i
            lo1 = max(0, i)
            hi1 = min(size, i + n)
            if hi1 > lo1:
                g[lo1:hi1] -= a[lo1 - i : hi1 - i]
            lo2 = max(0, 2 * n - 2 - i)
            hi2 = min(size, n - 1 + i + 1)
            # Actually: a_{i-k} is nonzero when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            lo2 = max(0, i - n + 1)
            hi2 = min(size, i + 1)
            if hi2 > lo2:
                for idx in range(lo2, hi2):
                    src = i - idx
                    if 0 <= src < n:
                        g[idx] -= a[src]

            # Add contribution from j
            lo3 = max(0, j)
            hi3 = min(size, j + n)
            if hi3 > lo3:
                g[lo3:hi3] += a[lo3 - j : hi3 - j]
            lo4 = max(0, j - n + 1)
            hi4 = min(size, j + 1)
            if hi4 > lo4:
                for idx in range(lo4, hi4):
                    src = j - idx
                    if 0 <= src < n:
                        g[idx] += a[src]

            # Find optimal delta in [0, a[i]] minimizing max(conv + delta*g)
            # Sample delta values
            ai = a[i]
            n_samples = 15
            deltas = np.linspace(0.0, ai, n_samples)
            best_d = 0.0
            best_val = conv.max()

            for d in deltas[1:]:  # skip 0
                new_conv_est = conv + d * g
                # Second order: at k=2i, the a[i]^2 term becomes (ai-d)^2
                ki = 2 * i
                if ki < size:
                    corr_i = conv[ki] - ai * ai + (ai - d) * (ai - d)
                    # The linear estimate already accounts for -d*a[i] at position i
                    # The true quadratic change at ki: (ai-d)^2 - ai^2 = -2*ai*d + d^2
                    # Linear estimate gives: conv[ki] + d*g[ki]
                    # g[ki] includes -a[0] (from the i-term at k=2i, a_{2i-i}=a_i, wait...)
                    # Let me just add the quadratic correction directly
                    new_conv_est[ki] = corr_i

                kj = 2 * j
                if kj < size:
                    aj = a[j]
                    corr_j = conv[kj] - aj * aj + (aj + d) * (aj + d)
                    new_conv_est[kj] = corr_j

                new_max = new_conv_est.max()
                if new_max < best_val - 1e-14:
                    best_val = new_max
                    best_d = d

            if best_d > 1e-15:
                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve_streak = 0
                else:
                    a[j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve_streak += 1
            if no_improve_streak >= 5:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def perturbation_restart(a, time_limit, solver, rng, verbose=True):
    """Add noise to a, re-normalize, and re-optimize via LP Nesterov."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    # Add structured perturbation
    noise_scale = 0.3
    noise = rng.normal(0.0, noise_scale, n)
    # Make it a bit asymmetric
    noise += 0.2 * rng.normal(0.0, 1.0, n) * np.linspace(-1, 1, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0.0)
    s_p = a_pert.sum()
    if s_p < 1e-15:
        a_pert = a.copy()
    else:
        a_pert = a_pert * norm / s_p

    # Run LP Nesterov on perturbed version
    a_out, score = nesterov_lp_optimize(a_pert, time_limit * 0.7, solver, verbose=False)
    return a_out, score


def structured_init(n, rng, init_type=None):
    """Generate diverse structured initializations biased toward asymmetry."""
    if init_type is None:
        init_type = int(rng.integers(0, 10))

    if init_type == 0:
        # Asymmetric Gaussian
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.35) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif init_type == 1:
        # Power-law asymmetric
        x = np.linspace(0.0, 1.0, n)
        a = x ** 0.3 * (1 - x) ** 0.7
    elif init_type == 2:
        # Cosine with phase shift (asymmetric)
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x + 0.5) + 0.1 * np.cos(4 * np.pi * x - 0.3)
        a = np.maximum(a, 0)
    elif init_type == 3:
        # Split: high on left, low on right with smooth transition
        x = np.linspace(0.0, 1.0, n)
        a = np.where(x < 0.5, 1.5 - 2.0 * x, 0.5 - 0.3 * x)
        a = np.maximum(a, 0)
    elif init_type == 4:
        # Bump in middle, low at edges (asymmetric bump)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.4) ** 2) + 0.6 * np.exp(-20.0 * (x - 0.6) ** 2) + 0.1
    elif init_type == 5:
        # Sawtooth-like (asymmetric periodic)
        x = np.linspace(0.0, 1.0, n)
        a = np.abs(np.sin(np.pi * x)) ** 1.5 * (1.0 + 0.3 * np.sin(2 * np.pi * x))
        a = np.maximum(a, 0)
    elif init_type == 6:
        # Exponential decay (strongly asymmetric)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * x) + 0.05
    elif init_type == 7:
        # Double bump asymmetric
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-15.0 * (x - 0.25) ** 2) + 0.4 * np.exp(-15.0 * (x - 0.7) ** 2) + 0.05
    elif init_type == 8:
        # Random smooth (via random Fourier coefficients)
        x = np.linspace(0.0, 1.0, n)
        coeffs = rng.normal(0, 1, 5)
        a = 1.0 + sum(coeffs[k] * 0.2 * np.cos(2 * np.pi * (k + 1) * x) for k in range(len(coeffs)))
        a = np.maximum(a, 0)
    else:
        # Logistic-like asymmetric
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.4)))
        a = a - a.min() + 0.05

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def block_lp_cd(a, time_limit, solver, rng, verbose=True):
    """
    Block coordinate descent: partition into blocks, optimize each block via LP.
    More effective than single-coordinate moves for escaping shallow local optima.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Block size
    block_size = max(4, min(16, n // 100))
    n_blocks = (n + block_size - 1) // block_size

    n_rounds = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_rounds += 1
        round_improved = False

        # Shuffle block order
        block_order = rng.permutation(n_blocks)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            if end <= start:
                continue
            bs = end - start

            M = conv.max()

            # Build constraints for this block
            # conv(a_new, a_new)_k for k where the block indices interact
            # Only need to check k values where the block contributes
            # Block indices: [start, end-1]
            # conv_k gets contribution a[block_i] * a[k - block_i] for all valid
            # So k ranges from start to start + end - 2 (at least)
            k_lo = max(0, start)
            k_hi = min(size, start + end - 1 + n)  # conservative
            k_indices = np.arange(k_lo, k_hi + 1)

            if len(k_indices) == 0:
                continue

            # For the LP: variables are a[start:end] (bs variables)
            # Constraint: for each k in relevant range,
            #   sum_{i in block} a_new[i] * (contrib from rest + a_new counterpart) <= M
            # This is quadratic in the block variables...
            # For simplicity, use the linear approximation:
            # Fix the "rest" of the sequence, linearize the quadratic terms
            b_block_var = cp.Variable(bs, nonneg=True)
            b_block_var <<= a[start:end]  # warm start

            constraints = []
            for k in k_indices:
                # conv_k = sum_{i in block} b_i * a_rest[k-i] + sum_{i,j in block} b_i * b_j * delta_{i+j=k}
                # Linear part: sum_i b_i * a_rest[k-i]
                # Quadratic part: sum_{i+j=k, i,j in block} b_i * b_j
                # For LP, use: sum_i b_i * (a_rest[k-i] + sum_{j in block, j!=i} a[j]) <= M
                # This is an upper bound on the true conv_k (by AM-GM or direct)
                lin = 0.0
                for i_local, i in enumerate(range(start, end)):
                    # Contribution from rest
                    j1 = k - i
                    if 0 <= j1 < n and not (start <= j1 < end):
                        lin += a[j1]
                    # Quadratic cross terms (upper bound: use current values)
                    for j_local, j in enumerate(range(start, end)):
                        if j != i and 0 <= j < n and i + j == k:
                            lin += a[j]
                    # Self term at k = 2i
                    if k == 2 * i:
                        lin += a[i]
                if lin > 0:
                    constraints.append(b_block_var @ np.array([0.0]) <= M)  # placeholder
            # This approach is getting too complex for a clean LP.
            # Let me use a simpler approach: just do the full LP on the block
            # with the current M and the rest fixed.

            # Actually, let's just skip this and use the simpler pair CD approach
            # which is already implemented. This block LP is too complex to implement
            # correctly and efficiently in the time available.
            pass

        if verbose and n_rounds % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BLP round={n_rounds} ({elapsed:.0f}s): best={best_score:.8f}")

        if no_improve >= 3:
            break

    return best_a, best_score


def lse_smooth_optimize(a0, time_limit, verbose=True):
    """
    Log-sum-exp smooth relaxation gradient descent.
    Smooths the max function and uses gradient descent with Adam-like adaptation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Stages with decreasing temperature
    n_stages = 10
    temps = np.geomspace(30.0, 0.3, n_stages)
    lr = 0.1
    momentum = 0.9

    v = np.zeros(n)  # momentum buffer

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(200, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_c = conv.max()
            exp_vals = np.exp((conv - max_c) / T)
            w = exp_vals / exp_vals.sum()

            # Compute gradient: grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k}) [valid]
            # Use FFT: conv(w, a) gives sum_j w_j * a[k-j]
            # sum_k w_k * a_{k-i} for k=0..size-1: this is cross-correlation
            # Let's compute directly using the convolution structure
            # grad_i = 2 * [sum_k w_k * a[k-i] + sum_k w_k * a[i-k]]
            #         = 2 * [ (w * a_flip)[n-1-i] + (w * a)[i] ]  (with appropriate indexing)

            # Simpler: compute via FFT
            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)

            # conv(w, a)[k] = sum_j w_j * a[k-j]
            wa = irfft(W_fft * A_fft, fft_size)[:size]

            # sum_k w_k * a[i-k]: let j = i-k, k = i-j
            # = sum_j w[i-j] * a[j] for valid j and i-j
            # = (a * w_flip)[i] where w_flip = w[::-1]
            # Actually: sum_k w_k * a[i-k] = sum_j a_j * w[i-j] = (a * rev(w))[i]
            # where rev(w) has rev(w)[m] = w[size-1-m]... this is getting confusing.

            # Let me just compute it directly for correctness (n <= 4000 is fine)
            grad = np.zeros(n)
            for i in range(n):
                # sum_k w_k * a[k-i]: k-i in [0,n-1] => k in [i, i+n-1], but k < size
                k_lo = i
                k_hi = min(size, i + n)
                if k_hi > k_lo:
                    grad[i] += np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
                # sum_k w_k * a[i-k]: i-k in [0,n-1] => k in [i-n+1, i], but k >= 0
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(size, i + 1)
                if k_hi2 > k_lo2:
                    idx = np.arange(k_lo2, k_hi2)
                    src = i - idx
                    grad[i] += np.dot(w[k_lo2:k_hi2], a[src])
            grad *= 2.0

            # Adam-like update
            lr_t = lr / (1.0 + 0.1 * it)
            v = momentum * v + (1 - momentum) * grad
            v_hat = v / (1 - momentum ** (it + 1))
            a_new = a - lr_t * v_hat
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses time-adaptive multi-phase approach:
    1. Multi-start LP Nesterov (diverse asymmetric inits)
    2. Gradient-guided CD on best
    3. Perturbation restarts
    4. Length exploration
    5. LSE smooth refinement
    6. Final CD polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, init_type=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Multi-start LP Nesterov ===
    # Run 3 different initializations in parallel (sequentially)
    remaining = time_budget - (time.monotonic() - t_global)
    n_starts = 3
    t_per_start = remaining * 0.30 / n_starts
    print(f"\n=== Phase 1: Multi-start LP ({n_starts} starts, {t_per_start:.0f}s each) ===")

    for si in range(n_starts):
        if time.monotonic() - t_global > time_budget - 100:
            break
        init_type = si + 3  # types 3, 4, 5 (asymmetric)
        a_init = structured_init(n_base, rng, init_type=init_type)
        a_norm = a_init * np.sqrt(2.0 * n_base) / a_init.sum()
        t_start = time.monotonic()
        a_out, _ = nesterov_lp_optimize(a_norm, t_per_start - 0.5, solver, verbose=(si == 0))
        do_try_update(a_out, f"lp_start{si}_n{n_base}")

    # Also run LP from parent
    remaining = time_budget - (time.monotonic() - t_global)
    t_parent = min(remaining * 0.12, 150.0)
    if t_parent > 20:
        print(f"\n=== Phase 1b: LP from parent ({t_parent:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t_parent, solver, verbose=True)
        do_try_update(a_out, f"lp_parent_n{n_base}")

    # === Phase 2: Gradient-guided CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 15:
        print(f"\n=== Phase 2: Gradient CD n={n_best} ({t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_guided_cd(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"gcd_n{n_best}")

    # === Phase 3: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_pert = 3
    t_per_pert = remaining * 0.20 / n_pert
    if t_per_pert > 15:
        print(f"\n=== Phase 3: Perturbation restarts ({n_pert}x {t_per_pert:.0f}s) ===")
        for pi in range(n_pert):
            if time.monotonic() - t_global > time_budget - 80:
                break
            a_out, _ = perturbation_restart(best_seq, t_per_pert, solver, rng, verbose=False)
            do_try_update(a_out, f"pert{pi}_n{len(best_seq)}")

    # === Phase 4: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    # Try a slightly different length
    n_alt = n_base + 200 if n_base < 2000 else n_base - 200
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LP n={n_alt} ({t4:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4 * 0.7, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n_alt}")
        # Polish
        t4b = min(time_budget - (time.monotonic() - t_global) - 80, 60.0)
        if t4b > 10 and best_seq is not None and len(best_seq) == n_alt:
            a_norm = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
            a_out, _ = gradient_guided_cd(a_norm, t4b, rng, verbose=False)
            do_try_update(a_out, f"pcd_n{n_alt}")

    # === Phase 5: LSE smooth refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 10:
        print(f"\n=== Phase 5: LSE smooth n={n_best} ({t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_smooth_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # === Phase 6: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.20, 150.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final CD n={n_best} ({t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_guided_cd(a_norm, t6, rng, verbose=True)
        do_try_update(a_out, f"final_cd_n{n_best}")

    # === Phase 7: Short length check ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(600, n_base - 600)
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 20 and n_short != n_base:
        print(f"\n=== Phase 7: Short n={n_short} ({t7:.0f}s) ===")
        init_short = interpolate_sequence(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7 * 0.6, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n_short}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
