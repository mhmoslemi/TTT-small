def penalized_objective(a_sequence, gamma=0.01):
    """Compute the penalized objective to enforce numerical stability."""
    n_val = len(a_sequence)
    s_val = np.sum(a_sequence)
    if s_val < 1e-10:
        raise ValueError("Sum is too small")
    s_val = s_val  # Avoid normalization by tiny numbers
    conv_result = np.convolve(a_sequence, a_sequence)
    max_b = np.max(conv_result)
    return 2.0 * n_val * max_b / (s_val**2)

def evaluate_safe_alt_core(a_norm: np.ndarray):
    """Direct core evaluation function to avoid function wrapper overhead."""
    n_val = a_norm.shape[0]
    s_val = np.sum(a_norm)
    if s_val < 1e-10:
        return float('inf')
    
    # Rescale to stay within [0, 1000] after min/max-normalization
    max_val = 1000.0
    scale_val = (max_val-1) / np.max(a_norm) if np.max(a_norm) > 1 else 1.0
    a_scaled = np.clip(a_norm * scale_val + 0.0 * np.random.randn(n_val) * 1e-3, 0.0000, max_val)  # Small noise for derandomization
    
    conv_result = np.convolve(a_scaled, a_scaled)
    max_b = np.max(conv_result)
    
    true_sum = np.sum(a_scaled)
    if true_sum < 1e-10:
        return float('inf')
    
    obj_val = 2.0 * n_val * max_b / (true_sum**2)
    return obj_val
