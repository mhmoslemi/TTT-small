def lse_gradient(a, T):
    """Compute gradient of LSE-smoothed max autocorrelation.
    
    Returns gradient of T * log(sum_k exp((a*a)_k / T))
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), b, 0.0
    w = w / w_sum
    # grad_j = (2/T) * sum_k w_k * a_{k-j}
    # = (2/T) * (w * a_rev)[j + n - 1]
    a_rev = a[::-1]
    size = (2*n-1) + n - 1  # = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = (2.0 / T) * conv[n-1:2*n-1]
    return grad, b, max_b


def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """LSE-smoothed gradient descent with BB step sizes and simplex projection."""

    s0 = a.sum()

    b = fast_autocorr(a)
    best_score = b.max()  # This is max_b when sum(a)=1, score = 2n*max_b
    best_a = a.copy()

    if T_start is None:
        T_start = max(0.01, best_score / n * 0.5)
    if T_end is None:
        T_end = max(0.001, best_score / n * 0.01)

    steps = 0
    stall = 0
    prev_grad = None
    prev_a = None
    bb_step = 1.0 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** min(elapsed_frac, 1.0)
        
        grad, b, max_b = lse_gradient(a, T)
        
        if max_b < best_score * (1 - 1e-10):
            best_score = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # BB step size
        if prev_grad is not None and prev_a is not None:
            dy = a - prev_a
            dg = grad - prev_grad
            dy_norm = np.dot(dy, dy)
            dg_norm = np.dot(dg, dg)
            if dg_norm > 1e-20 and dy_norm > 1e-20:
                yd = np.dot(dy, dg)
                if yd > 1e-10:
                    bb_step = dy_norm / yd
                    bb_step = np.clip(bb_step, 0.01 / n, 10.0 / n)
        
        # Gradient step with projection
        a_new = a - bb_step * grad
        a_new = project_simplex(a_new)
        
        # Ensure non-negativity (projection should handle it, but double-check)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        b_new = fast_autocorr(a_new)
        max_b_new = b_new.max()
        
        if max_b_new < best_score:
            best_score = max_b_new
            best_a = a_new.copy()
            stall = 0
        
        prev_grad = grad
        prev_a = a
        a = a_new
        steps += 1
        
        if verbose and steps % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_score
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.6f}, best_score={score:.6f}, stall={stall}")
        
        if stall > 2000:
            # Restart with perturbation
            rng = np.random.default_rng(steps)
            a = project_simplex(a + rng.normal(0, 0.01/n, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            prev_grad = None
            prev_a = None

    return best_a, 2 * n * best_score

            n_top = min(n, int(0.7 * (2 * n - 1)))
            top_idx = np.argsort(b_y)[::-1][:n_top]
            b = conv.solve_lp_reduced(M_curr, top_idx, solver)
            if b is None:
                b = conv.solve_lp(np.full(2 * n - 1, M_curr), solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_maxb:
            best_maxb = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_maxb:
            best_maxb = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={2*n*best_maxb/norm**2:.6f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm / a.sum()
            conv.b_prev = None
            stall = 0

    return best_a, 2.0 * n * best_maxb / (best_a.sum() ** 2)


def lse_fista(a0, time_limit, verbose=True):
    """FISTA with LSE smoothing and backtracking on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    b = fast_autocorr(a)
    best_maxb = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    steps = 0
    stall = 0

    # LSE parameters
    T_current = max(0.5 / n, best_maxb / n * 2)
    T_min = max(0.05 / n, best_maxb / n * 0.1)
    L = 4.0 / (T_current * n)  # initial Lipschitz estimate
    t_fista = 1.0
    y = a.copy()
    a_prev = a.copy()

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_current * (T_min / T_current) ** (progress ** 0.7)

        # Compute LSE gradient at y
        b_y = fast_autocorr(y)
        max_b_y = b_y.max()
        w = np.exp((b_y - max_b_y) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        a_rev = y[::-1]
        size = 3 * n - 2
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_res = irfft(W * AR, fft_size)
        grad = (2.0 / T) * conv_res[n - 1:2 * n - 1]

        # Backtracking line search
        step = 1.0 / L
        a_trial = a.copy()
        for _ in range(20):
            candidate = y - step * grad
            candidate = np.maximum(candidate, 0)
            s_c = candidate.sum()
            if s_c > 1e-15:
                candidate = candidate / s_c
            else:
                candidate = a_trial
            b_cand = fast_autocorr(candidate)
            f_cand = T * np.log(np.exp((b_cand - max_b_y) / T).sum()) + max_b_y
            f_y = T * np.log(np.exp((b_y - max_b_y) / T).sum()) + max_b_y
            quad_bound = f_y + np.dot(grad, candidate - y) + (L / 2) * np.dot(candidate - y, candidate - y)
            if f_cand <= quad_bound + 1e-12:
                a_trial = candidate
                break
            step *= 0.5
            L *= 2
        L = max(L * 0.9, 4.0 / (T * n * 10))

        # Fista momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        y_new = a_trial + ((t_fista - 1.0) / t_new) * (a_trial - y)
        y_new = np.maximum(y_new, 0)
        s_yn = y_new.sum()
        if s_yn < 1e-15:
            y_new = a_trial.copy()
        else:
            y_new = y_new / s_yn
        y = y_new
        a = a_trial
        t_fista = t_new
        steps += 1

        b_a = fast_autocorr(a)
        max_b_a = b_a.max()
        if max_b_a < best_maxb:
            best_maxb = max_b_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA step={steps} ({elapsed:.0f}s): T={T:.6f}, L={L:.1f}, best={2*n*best_maxb:.6f}, stall={stall}")

        if stall > 1500:
            rng = np.random.default_rng(steps * 7 + 3)
            a = project_simplex(a + rng.normal(0, 0.005 / n, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            t_fista = 1.0
            stall = 0

    return best_a, 2.0 * n * best_maxb


def subgradient_random_peak(a0, time_limit, verbose=True):
    """Random peak subgradient with averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_maxb = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    steps = 0
    rng = np.random.default_rng(12345)

    c = 0.1  # step size constant

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < best_maxb:
            best_maxb = max_b
            best_a = a.copy()

        # Pick random peak with probability proportional to excess
        top_k = np.argsort(b)[::-1][:min(20, n)]
        # Choose weighted random
        weights = b[top_k]
        weights = weights / weights.sum()
        k = top_k[rng.choice(len(top_k), p=weights)]

        # Subgradient: 2 * a_{k-j} for j=0..n-1 (with a_{k-j}=0 if out of range)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k - idx]

        eta = c / np.sqrt(t)
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        a = a_new
        avg_a = avg_a + a
        steps += 1

        # Check averaged iterate
        if steps % 100 == 0:
            avg_check = avg_a / steps
            b_avg = fast_autocorr(avg_check)
            if b_avg.max() < best_maxb:
                best_maxb = b_avg.max()
                best_a = avg_check.copy()

        if verbose and steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={2*n*best_maxb:.6f}")

    return best_a, 2.0 * n * best_maxb


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_best_seq, _global_best_score
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        x = np.linspace(0.0, 1.0, n_init)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_base}")

    # Phase 2: FISTA at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FISTA n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_out, _ = lse_fista(best_seq, t2, verbose=True)
        try_update(a_out, f"fista_n{n_cur}")

    # Phase 3: LP Nesterov at n+200
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # Phase 4: LP Nesterov at n+400
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp_n{n3}")

    # Phase 5: FISTA polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FISTA polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_out, _ = lse_fista(best_seq, t5, verbose=True)
        try_update(a_out, f"fista_polish_n{n_best}")

    # Phase 6: Subgradient with averaging
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SubGrad n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_out, _ = subgradient_random_peak(best_seq, t6, verbose=True)
        try_update(a_out, f"subgrad_n{n_best}")

    # Phase 7: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 250.0)

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t7, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
