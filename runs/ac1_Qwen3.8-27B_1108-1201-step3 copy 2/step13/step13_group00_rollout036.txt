2. Exact simplex projection for constraint handling

4. Pair coordinate descent with adaptive step size for final polish
5. Peak-shaving via targeted pair adjustments
6. Random perturbation restarts to escape local optima

import numpy as np
import time
from scipy.fft import rfft, irfft

_global_best = None
_global_best_score = float('inf')


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, c):
    """Project vector v onto {a >= 0, sum(a) = c} Euclidean projection."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - c)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        result = np.zeros(n)
        result[0] = c
        return result
    rho = rho_candidates[-1]
    theta = (cssv[rho] - c) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(a, T, fft_size):
    """
    Compute LSE_T(conv(a,a)) and its gradient w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / T)
    val = max_c + T * np.log(exp_c.sum())
    w = exp_c / exp_c.sum()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # S1[m] = sum_k w[k] * a[k-m] (cross-correlation)
    S1 = irfft(W * np.conj(A), fft_size)[:n]
    # S2[m] = sum_k w[k] * a[m-k] (convolution)
    S2 = irfft(W * A, fft_size)[:n]
    grad = 2.0 * (S1 + S2)
    return val, grad


def lse_projected_gradient(a0, T_schedule, c, time_limit, verbose=True):
    """
    LSE smoothed optimization via projected gradient method.
    a0: initial point (will be normalized)
    T_schedule: list of temperatures from high to low
    c: target sum (typically sqrt(2n))
    """
    n = len(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    total_iters = 0

    avg_a = np.zeros(n)
    n_avg = 0

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Number of iterations for this temperature
        n_iters = max(20, min(300, int(remaining * 2)))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            val, grad = lse_value_and_grad(a, T, fft_size)

            # Backtracking line search
            lr = T / (n * 0.1)
            if it < 5:
                lr = T / (n * 0.5)

            improved = False
            for backtrack in range(10):
                a_trial = a - lr * grad
                a_trial = project_simplex(a_trial, c)
                val_trial = fast_autocorr(a_trial).max() * (T / max_c_safe(a_trial))
                # Use the actual LSE for the line search
                val_trial_lse, _ = lse_value_and_grad(a_trial, T, fft_size)
                if val_trial_lse < val - 1e-12 * max(1.0, abs(val)) or backtrack == 9:
                    a = a_trial
                    improved = True
                    break
                lr *= 0.5

            total_iters += 1

            # Track best
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Running average
            avg_a += a
            n_avg += 1
            if n_avg % 30 == 0:
                avg_check = project_simplex(avg_a / n_avg, c)
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def max_c_safe(a):
    """Safe max of autocorrelation."""
    if a.sum() < 1e-15:
        return 1.0
    return max(fast_autocorr(a).max(), 1e-15)


def pair_cd(a0, c, time_limit, rng, n_passes_max=200, verbose=True):
    """
    Pair coordinate descent: for each pair (i,j), transfer mass from i to j (or vice versa)
    to reduce the max autocorrelation peak.
    """
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * c / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve_count = 0

    for pass_num in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        n_trials = 0

        # Try multiple random pairs
        n_pairs_per_pass = min(200, n)
        for _ in range(n_pairs_per_pass):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < c * 1e-10:
                continue

            n_trials += 1
            # Compute direction: transfer delta from i to j
            # Change in conv[k] = delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
            # Use first-order approximation
            d = np.zeros(size)

            # Add contribution from j (positive)
            for k in range(j, min(size, j + n)):
                d[k] += a[k - j]
            for k in range(max(0, n - 1 - j), min(size, n + j)):
                idx_in_a = n - 1 - k
                if 0 <= idx_in_a < n:
                    d[k] += a[idx_in_a]

            # Subtract contribution from i (negative)
            for k in range(i, min(size, i + n)):
                d[k] -= a[k - i]
            for k in range(max(0, n - 1 - i), min(size, n + i)):
                idx_in_a = n - 1 - k
                if 0 <= idx_in_a < n:
                    d[k] -= a[idx_in_a]

            # Current max
            conv = fast_autocorr(a)
            cur_max = conv.max()

            # Try a few delta values (transferring from i to j)
            max_d = a[i] * 0.5
            best_delta = 0.0
            best_trial = cur_max
            n_samples = 8
            for fr in np.linspace(0.05, 1.0, n_samples):
                delta = max_d * fr
                new_conv = conv + delta * d
                # Second order at 2i and 2j
                if 2 * i < size:
                    new_conv[2 * i] = conv[2 * i] - 2 * a[i] * delta + delta * delta
                if 2 * j < size:
                    new_conv[2 * j] = conv[2 * j] + 2 * a[j] * delta + delta * delta
                trial = new_conv.max()
                if trial < best_trial - 1e-14:
                    best_trial = trial
                    best_delta = delta

            if best_delta > 0:
                a[i] -= best_delta
                a[j] += best_delta
                a = np.maximum(a, 0.0)
                a = a * c / a.sum()

                # Verify with actual autocorrelation
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[j] -= best_delta
                    a[i] += best_delta
                    a = a * c / a.sum()

        if verbose and (pass_num + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={pass_num+1} ({elapsed:.0f}s): best={best_score:.8f}")

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                break
        else:
            no_improve_count = 0

    return best_a, best_score


def peak_shaving(a0, c, time_limit, rng, verbose=True):
    """
    Targeted peak shaving: identify top autocorrelation peaks and reduce them
    by adjusting contributing pairs.
    """
    n = len(a0)
    size = 2 * n - 1
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * c / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_rounds = 50
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        # Find top peaks (excluding lag 0 which is hardest to reduce)
        peak_indices = np.argsort(conv)[::-1]
        n_peaks = min(10, size)

        for k in peak_indices[:n_peaks]:
            if k == 0:
                # For lag 0, reduce the most "concentrated" elements
                # Find the largest a_i and redistribute
                i_max = int(np.argmax(a))
                if a[i_max] < c / n * 1.5:
                    continue
                # Try spreading from i_max
                for _ in range(5):
                    j = int(rng.integers(n))
                    if j == i_max:
                        continue
                    delta = a[i_max] * 0.05
                    a[i_max] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0.0)
                    a = a * c / a.sum()
                    sc = fast_autocorr(a).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    else:
                        a[j] -= delta
                        a[i_max] += delta
                        a = a * c / a.sum()
            else:
                # For lag k, reduce products a_i * a_{i+k}
                # Find the pair with largest product at this lag
                valid_i = np.arange(0, n - k)
                if len(valid_i) == 0:
                    continue
                products = a[valid_i] * a[valid_i + k]
                i_best = int(valid_i[np.argmax(products)])
                # Reduce the larger of the two
                if a[i_best] >= a[i_best + k]:
                    src, dst = i_best, i_best + k
                else:
                    src, dst = i_best + k, i_best

                # Try transferring a small amount
                delta = a[src] * 0.03
                # Find where to put it (somewhere that doesn't worsen peaks)
                j = int(rng.integers(n))
                if j == src:
                    continue

                a[src] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a * c / a.sum()
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                else:
                    a[j] -= delta
                    a[src] += delta
                    a = a * c / a.sum()

        if verbose and (rnd + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rnd+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def upsample_sequence(a, new_n):
    """Upsample sequence to new length using linear interpolation."""
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s_new


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        # Tukey window with r=0.5
        r = 0.5
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        mask_low = x < r / 2
        mask_high = x > 1 - r / 2
        a[mask_low] = 0.5 * (1 + np.cos(np.pi * (2 * x[mask_low] / r - 1)))
        a[mask_high] = 0.5 * (1 + np.cos(np.pi * (2 * x[mask_high] / r - 1)))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.8 + 0.1 * np.sin(2 * np.pi * x) ** 2

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and returns the best sequence found.
    """
    global _global_best, _global_best_score

    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Determine base length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE projected gradient at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    c1 = np.sqrt(2.0 * n1)
    print(f"\n=== Phase 1: LSE PG n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        # Temperature schedule
        T_schedule = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1]
        a_init = best_seq * c1 / best_seq.sum()
        a_out, _ = lse_projected_gradient(a_init, T_schedule, c1, t1, verbose=True)
        try_update(a_out, f"lse_n{n1}")

    # === Phase 2: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c2 = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 10:
        a_init = best_seq * c2 / best_seq.sum()
        a_out, _ = pair_cd(a_init, c2, t2, rng, n_passes_max=500, verbose=True)
        try_update(a_out, f"pcd_n{n_best}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c3 = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 3: Peak Shaving n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_init = best_seq * c3 / best_seq.sum()
        a_out, _ = peak_shaving(a_init, c3, t3, rng, verbose=True)
        try_update(a_out, f"peak_n{n_best}")

    # === Phase 4: LSE again (refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c4 = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 4: LSE refinement n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        T_schedule2 = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05]
        a_init = best_seq * c4 / best_seq.sum()
        a_out, _ = lse_projected_gradient(a_init, T_schedule2, c4, t4, verbose=True)
        try_update(a_out, f"lse2_n{n_best}")

    # === Phase 5: Try different length (longer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = n_base + 400
    c5 = np.sqrt(2.0 * n5)
    print(f"\n=== Phase 5: LSE n={n5} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_init = upsample_sequence(best_seq, n5) * c5
        T_schedule3 = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        a_out, _ = lse_projected_gradient(a_init, T_schedule3, c5, t5, verbose=True)
        try_update(a_out, f"lse_n{n5}")

    # === Phase 6: Try shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n6 = max(800, n_base - 400)
    c6 = np.sqrt(2.0 * n6)
    print(f"\n=== Phase 6: LSE n={n6} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 30:
        a_init = upsample_sequence(best_seq, n6) * c6
        T_schedule4 = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        a_out, _ = lse_projected_gradient(a_init, T_schedule4, c6, t6, verbose=True)
        try_update(a_out, f"lse_n{n6}")

    # === Phase 7: Pair CD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c7 = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 7: Pair CD final n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 10:
        a_init = best_seq * c7 / best_seq.sum()
        a_out, _ = pair_cd(a_init, c7, t7, rng, n_passes_max=500, verbose=True)
        try_update(a_out, f"pcd_final_n{n_best}")

    # === Phase 8: Random perturbation + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c8 = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 8: Perturbation restarts n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        n_restarts = min(5, int(remaining / 40))
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 30:
                break
            # Add random perturbation
            noise_scale = 0.1 / (r + 1)
            a_pert = best_seq.copy()
            a_pert += noise_scale * rng.standard_normal(n_best)
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p < 1e-15:
                continue
            a_pert = a_pert * c8 / s_p

            t_r = min(remaining - 30 - r * 40, 50.0)
            if t_r < 10:
                break
            T_schedule_r = [3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.1]
            a_out, _ = lse_projected_gradient(a_pert, T_schedule_r, c8, t_r, verbose=False)
            try_update(a_out, f"pert{r}_n{n_best}")

    # === Phase 9: Final pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    c9 = np.sqrt(2.0 * n_best)
    if remaining > 15:
        print(f"\n=== Phase 9: Final PCD n={n_best} (budget={remaining:.0f}s) ===")
        t9 = min(remaining - 10, 100.0)
        if t9 > 5:
            a_init = best_seq * c9 / best_seq.sum()
            a_out, _ = pair_cd(a_init, c9, t9, rng, n_passes_max=300, verbose=False)
            try_update(a_out, f"pcd_ult_n{n_best}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
