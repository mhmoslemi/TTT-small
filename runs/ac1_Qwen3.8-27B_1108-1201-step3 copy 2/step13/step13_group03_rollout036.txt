def _lse_grad_and_weights(a, T):

def _project_nonneg_sum(v, target_sum):
    n = len(v)
    if n == 0:
        return np.array([])
    # First project to sum=1, then scale
    v_pos = np.maximum(v, 0.0)
    s = v_pos.sum()
    if s < 1e-30:
        return np.ones(n) * target_sum / n
    # Now we need the exact simplex projection
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:

    if s_out < 1e-30:

def _lp_best_response(a, M):
    """Find b >= 0 maximizing sum(b) subject to (a*b)[k] <= M for all k."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint matrix: (T @ b)[k] = sum_j T[k,j] * b[j] <= M
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    # For efficiency, use the structure
    # (a*b)[k] = sum_{j=0}^{n-1} a[max(0, k-j):min(n, k-j+n)]... 
    # Actually, the full convolution (a*b)[k] = sum_{j=0}^{min(k,n-1)} a[j] * b[k-j] for k in [0, 2n-2]
    # Wait, full convolution: (a*b)[k] = sum_{j} a[j] * b[k-j] where both indices are in [0,n-1]
    # So for a given k, j ranges from max(0, k-(n-1)) to min(k, n-1)
    
    # The constraint is: for all k in [0, 2n-2], (a*b)[k] <= M
    # This is equivalent to: for all k, sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * b[k-j] <= M
    
    # Let's build the matrix T where T[k,j] = a[k-j] if valid
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j), min(2*n-1, j+n)):
            T[k, j] = a[k - j]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _graft(a1, a2, rng):
    """Try to improve a1 by incorporating parts of a2."""
    n = len(a1)
    if len(a2) != n:
        # Interpolate a2 to length n
        old_n = len(a2)
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, n)
        a2_interp = np.interp(x_new, x_old, a2 / a2.sum())
        a2 = a2_interp * a1.sum() / a2_interp.sum()
    
    norm = np.sqrt(2.0 * n)
    best = a1.copy()
    best_score = _fft_autocorr(a1).max()
    
    # Try different mixing ratios
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
        mixed = (1 - alpha) * a1 + alpha * a2
        mixed = np.maximum(mixed, 0.0)
        if mixed.sum() < 1e-15:
            continue
        mixed = mixed * norm / mixed.sum()
        sc = _fft_autocorr(mixed).max()
        if sc < best_score:
            best_score = sc
            best = mixed.copy()
    
    return best, best_score

def _interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

Actually, (a*a)[k] = sum_j a[j]*a[k-j]. If we write this as a^T B_k a where B_k is a matrix, then B_k has B_k[j, k-j] = 1/2 and B_k[k-j, j] = 1/2 (for the off-diagonal terms to give the right cross terms). This matrix is not PSD in general (it has negative eigenvalues for n > 2).

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam with aggressive temperature annealing (primary method)
2. Multi-start with diverse initializations to escape local minima
3. LP dual refinement (warm-started Nesterov)
4. Multi-length expansion with linear interpolation
5. Final low-temperature LSE-Adam polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def _fft_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _lse_grad_and_weights(a, T):
    """Compute gradient of LSE-smoothed max autocorrelation and softmax weights."""
    n = len(a)
    ac = _fft_autocorr(a)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -100.0, 100.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad, w, ac


def _project_nonneg_sum(v, target_sum):
    """Project vector onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) * target_sum / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s_out = out.sum()
    if s_out < 1e-30:
        return np.ones(n) * target_sum / n
    return out * target_sum / s_out


def _lse_adam(a0, temps, iters_per_stage, time_limit, lr_init=0.1, beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """LSE-Adam optimization with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_max = _fft_autocorr(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    for stage_i, T in enumerate(temps):
        stage_iters = max(int(iters_per_stage * (temps[0] / max(T, 0.01)) ** 0.4), 1500)

        for it in range(stage_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_max

            t += 1
            grad, w, ac = _lse_grad_and_weights(a, T)
            # Clip gradient to prevent explosion at low T
            gnorm = np.linalg.norm(grad)
            if gnorm > 100.0:
                grad = grad * 100.0 / gnorm

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = lr_init * max(T, 0.05)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = _project_nonneg_sum(a_new, norm)
            a = a_new

            if it % 300 == 0:
                cur_max = _fft_autocorr(a).max()
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()

        cur_max = _fft_autocorr(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam stage={stage_i} T={T:.4f} ({elapsed:.0f}s): max_ac={best_max:.8f}")

    cur_max = _fft_autocorr(a).max()
    if cur_max < best_max:
        best_max = cur_max
        best_a = a.copy()

    return best_a, best_max


def _lp_best_response(a, M):
    """Find b >= 0 maximizing sum(b) subject to (a*b)[k] <= M for all k."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build Toeplitz-like constraint matrix
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        T_mat[k_start:k_end, j] = a[:k_end - k_start]

    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    for solver in [cp.CLARABEL, cp.SCS]:
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, verbose=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_max = _fft_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fft_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = _lp_best_response(y, M_curr)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fft_autocorr(y).max()
        if score_y < best_max:
            best_max = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fft_autocorr(avg_check).max()
                if score_avg < best_max:
                    best_max = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_max:.8f}")

    return best_a, best_max


def _interpolate_to_length(a, new_n):
    """Linearly interpolate sequence to a new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(x_new, x_old, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def _spectral_flatten(a, time_limit, lr=0.005, verbose=False):
    """
    Iteratively reduce spectral peaks by adjusting a in the time domain.
    Uses the fact that flat autocorrelation <=> concentrated spectrum.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_max = _fft_autocorr(a).max()
    t0 = time.monotonic()
    momentum = np.zeros(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fft_autocorr(a)
        ac_mean = ac.mean()
        # Gradient pushing toward flat autocorrelation
        # We want to reduce peaks above mean and increase valleys below
        # This is equivalent to minimizing variance of ac
        # grad_i = 2 * sum_k (ac[k] - ac_mean) * w_k * (a[k-i] + a[i-k])
        # where w_k = 1 for all k (uniform weighting)
        # This simplifies to: grad = 2 * (a * (ac - ac_mean)) [cross-correlation]

        w = (ac - ac_mean) / (np.std(ac) + 1e-10)
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        cross = irfft(np.conj(A) * W, fft_size)
        grad = 2.0 * cross[n - 1: n - 1 + n]

        # Gradient descent with momentum
        momentum = 0.9 * momentum + 0.1 * grad
        a_new = a - lr * momentum
        a_new = _project_nonneg_sum(a_new, norm)
        a = a_new

        if it % 200 == 0:
            cur_max = _fft_autocorr(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

    return best_a, best_max
