import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-12:
        return np.maximum(v, 0)
    v = np.maximum(v, 0)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def build_T_rows(a, indices):
    """Build rows of the convolution matrix T for given constraint indices."""
    n = len(a)
    num_rows = len(indices)
    T = np.zeros((num_rows, n))
    for i, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[i, j] = a[k - j]
    return T


def solve_lp_fast(a, peak_indices, solver=None):
    """
    Solve: min M s.t. (T[b] <= M for k in peak_indices), sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    num_constr = len(peak_indices)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_T_rows(a, peak_indices)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver is not None:
        try:
            prob.solve(solver=solver)
        except Exception:
            prob.solve()
    else:
        prob.solve()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def linearized_local_opt(a0, time_limit, top_k=20, max_steps=500, verbose=True):
    """
    Linearized peak reduction: at each step, solve a small LP to reduce
    the top peaks using a linearized model.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_sorted = np.argsort(b)[::-1]
        peak_idx = k_sorted[:top_k]
        peak_vals = b[peak_idx]
        
        # Build linearized LP:
        # min t
        # s.t. t >= b_k + 2 * sum_i delta_i * a[k-i] for k in peak_idx
        #      sum(delta) = 0
        #      -a_i <= delta_i <= a_i (to keep a+delta in [0, 2a])
        
        num_peek = len(peak_idx)
        # A matrix: (num_peek + 1) x (n + 1)
        # Variables: [delta_0, ..., delta_{n-1}, t]
        A_ub = np.zeros((num_peek + n, n + 1))
        b_ub = np.zeros(num_peek + n)
        
        for i, k in enumerate(peak_idx):
            # t - sum_j 2*a[k-j]*delta_j >= b_k  =>  -t + sum_j 2*a[k-j]*delta_j <= -b_k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[i, j] = 2.0 * a[k - j]
            A_ub[i, n] = -1.0  # coefficient of t
            b_ub[i] = -peak_vals[i]
        
        # -a_i <= delta_i => delta_i >= -a_i => -delta_i <= a_i
        # delta_i <= a_i
        for j in range(n):
            A_ub[num_peek + j, j] = 1.0
            b_ub[num_peek + j] = a[j]
            # We'll handle -a_i <= delta_i as another constraint or use bounds
        
        # Bounds on delta: -a_j <= delta_j <= a_j, and t >= 0
        bounds = [(-a[j], a[j]) for j in range(n)] + [(0, None)]
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Objective: minimize t => c = [0,...,0, 1]
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                delta = res.x[:n]
                a_new = a + delta
                a_new = np.maximum(a_new, 0)
                s2 = a_new.sum()
                if s2 > 1e-15:
                    a_new = a_new / s2
                    score = internal_score(a_new)
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                        a = a_new
                    elif score < best_score * 1.01:
                        # Accept even if not strictly better (to explore)
                        a = a_new
        except Exception:
            pass
        
        if verbose and step % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LinLocal step={step} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123, top_k=15):
    """Coordinate-descent peak shaving with random restarts on stalled moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_sorted = np.argsort(b)[::-1]
        top_peaks = k_sorted[:top_k]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Find destination: random index with small value
            candidates = np.where(a < np.median(a))[0]
            candidates = candidates[(candidates != src)]
            if len(candidates) == 0:
                candidates = [j for j in range(n) if j != src]
            dst = candidates[rng.integers(len(candidates))]
            
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score


def spectral_shaping(a, n_iter=200, alpha=0.3):
    """Iteratively flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS
