```python
def active_set_lp_best_response(a, M, solver, max_constraints=50):
    n = len(a)
    # Find initial active set: top K lags
    b_curr = fast_autocorr(a)
    active_set = set(np.argsort(b_curr)[::-1][:max_constraints].tolist())
    
    for iteration in range(20):
        # Build constraint matrix for active set
        k_values = sorted(active_set)
        T = np.zeros((len(k_values), n))
        for row, k in enumerate(k_values):
            for i in range(max(0, k-n+1), min(n, k+1)):
                T[row, i] = a[k-i]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        
        # Check for violated constraints
        new_b = fast_autocorr(b_val)
        violated = np.where(new_b > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            return b_val
        
        # Add violated constraints to active set
        for v in violated[:10]:  # Add at most 10 new constraints
            active_set.add(int(v))
        
        if len(active_set) > 200:  # Cap total constraints
            # Keep only the most violated
            viol_vals = [(int(v), new_b[v]) for v in violated]
            viol_vals.sort(key=lambda x: -x[1])
            active_set = set(np.argsort(new_b)[::-1][:200].tolist())
    
    # Fallback: use full constraint
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0) if b_var.value is not None else None
    if b_val is not None:
        return b_val
    return None
```

So Part1[i] = conv_full(a, w_rev)[2n-2-i]!

Let me verify: conv_full(a, w_rev)[m] = sum_j a[j] * w_rev[m-j] = sum_j a[j] * w[2n-2-m+j]
Setting m = 2n-2-i: = sum_j a[j] * w[2n-2-2n+2+i+j] = sum_j a[j] * w[i+j] ✓

So:
Part1[i] = conv_full(a, w_rev)[2n-2-i]
Part2[i] = conv_full(a, w)[i]

And grad[i] = Part1[i] + Part2[i] = conv_full(a, w_rev)[2n-2-i] + conv_full(a, w)[i]

I can compute both convolutions with FFT in O(n log n) time.

Great, so the gradient computation is:
1. b = autocorr(a) (FFT, O(n log n))
2. w = softmax(T * (b - max(b))) (O(n))
3. w_rev = w[::-1] (O(n))
4. conv1 = fft_convolve(a, w) (O(n log n))
5. conv2 = fft_convolve(a, w_rev) (O(n log n))
6. grad[i] = conv1[i] + conv2[2n-2-i] for i=0,...,n-1 (O(n))
7. Complete gradient of objective: grad_obj = 4*grad / (sum(a))^2 - 2*score*a/sum(a) (O(n))

This is all very efficient!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LSE (log-sum-exp) smooth surrogate with L-BFGS-B and analytical gradients
2. Active-set LP Nesterov acceleration
3. Multi-scale refinement (coarsen -> optimize -> upsample -> fine-tune)
4. Adaptive coordinate descent polish
5. Multi-length exploration
Key improvements over previous: O(n log n) exact gradient for LSE, active-set LP for
faster iterations, multi-scale strategy to escape local minima, better temperature
scheduling, and improved time allocation favoring the fast gradient method.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_convolve(a, b):
    """Compute full convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def lse_objective_and_grad(a_flat, n, T):
    """Compute LSE-smoothed objective and its gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # Compute weights w_k = exp(T*(b_k - max_b)) / sum(exp(T*(b_j - max_b)))
    scaled = T * (b - max_b)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 0)  # All <= 0 since b_k <= max_b
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    lse_val = max_b + np.log(exp_vals.sum()) / T
    obj = 2.0 * n * lse_val / (s * s)
    
    # Gradient computation
    # grad[i] = (4/s^2) * sum_k w_k * (a_{k-i} + a_{i-k}) - (2*obj/s) * a_i
    # Part 2: sum_k w_k * a_{i-k} = conv_full(a, w)[i]
    # Part 1: sum_k w_k * a_{k-i} = conv_full(a, w_rev)[2n-2-i]
    
    w_rev = w[::-1]
    conv_aw = fft_convolve(a, w)  # length 2n-1
    conv_awr = fft_convolve(a, w_rev)  # length 2n-1
    
    # For i = 0, ..., n-1:
    # Part2[i] = conv_aw[i]
    # Part1[i] = conv_awr[2*n-2-i]
    grad_conv = conv_aw[:n] + conv_awr[n-1::-1]  # conv_awr[n-1::-1] gives [n-1], [n-2], ..., [0] reversed
    # Wait: conv_awr[2n-2-i] for i=0,...,n-1 gives indices 2n-2, 2n-3, ..., n-1
    # That's conv_awr[n-1:2n-1][::-1]
    grad_conv = conv_aw[:n] + conv_awr[n-1:2*n-1][::-1]
    
    grad = (4.0 / (s * s)) * grad_conv - (2.0 * obj / s) * a
    return obj, grad.ravel()


def lse_lbfgs_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8):
    """Optimize using LSE smooth surrogate with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max() / (a.sum() ** 2)  # = max_conv since sum=1
    # Actually score = 2*n*max_conv/sum^2, and with sum=1, score = 2*n*max_conv
    # But for comparison we should use the actual score formula
    best_score_actual = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)
        
        x0 = a.copy()
        s0v = x0.sum()
        if s0v < 1e-10:
            x0 = np.ones(n) / n
        
        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val
        
        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'time_limit': stage_time, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate / s_c
                    score_c = 2.0 * n * fast_autocorr(a_candidate).max() / (a_candidate.sum() ** 2)
                    if score_c < best_score_actual:
                        best_score_actual = score_c
                        best_a = a_candidate.copy()
                    a = a_candidate
        except Exception:
            pass
    
    return best_a, best_score_actual


def lse_gradient_descent(a0, time_limit, T_start=100.0, T_end=2.0, n_iters=50000):
    """Custom gradient descent with momentum for LSE objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    T = T_start
    T_decay = (T_end / T_start) ** (1.0 / n_iters)
    
    for t in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        T *= T_decay
        
        obj, grad = lse_objective_and_grad(a, n, T)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        lr = min(lr * 1.001, 1.0 / (grad_norm + 1e-8))
        
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_autocorr(a_new).max() / (a_new.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD t={t} ({elapsed:.0f}s): T={T:.1f}, score={best_score:.8f}")
    
    return best_a, best_score


def active_set_lp_best_response(a, M, solver, top_k=30, max_rounds=10):
    """LP best-response using only the top K active constraints."""
    n = len(a)
    b_curr = fast_autocorr(a)
    # Start with top K lags
    active_indices = np.argsort(b_curr)[::-1][:top_k].astype(int)
    
    for round_idx in range(max_rounds):
        # Build constraint matrix
        K = len(active_indices)
        T_mat = np.zeros((K, n))
        for row, k in enumerate(active_indices):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_mat[row, i] = a[k_int - i]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            continue
        b_val = np.maximum(b_val, 0.0)
        if b_val.sum() < 1e-15:
            continue
        
        # Check for violated constraints
        b_new = fast_autocorr(b_val)
        violated = np.where(b_new > M * 1.001)[0]
        
        if len(violated) == 0:
            return b_val
        
        # Add most violated to active set
        violated_sorted = violated[np.argsort(b_new[violated])[::-1]][:10]
        active_indices = np.unique(np.concatenate([active_indices, violated_sorted]))
        active_indices = active_indices[:100]  # Cap
    
    # Fallback: return what we have
    if b_val is not None and b_val.sum() > 1e-15:
        return b_val
    return None


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, use_adaptive=True, top_k=30):
    """Nesterov-accelerated LP best-response with active set."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = active_set_lp_best_response(y, M_curr, solver, top_k=top_k)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        if use_adaptive:
            if best_score > 0:
                ratio = M_curr / best_score
                alpha = 2.0 / (t + 2.0)
                alpha = min(alpha * min(ratio, 2.0), 1.0)
            else:
                alpha = 2.0 / (t + 2.0)
        else:
            alpha = 2.0 / (t + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum_coeff = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum_coeff * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
        
        if time.monotonic() - last_improve > 20.0:
            stall_count += 1
            if stall_count > 20:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent that moves mass from peak-contributing to non-peak positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        
        improved = False
        for k_peak in top_peaks:
            k_peak = int(k_peak)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]
            
            # Find source to move mass from
            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                # Find destination: positions that contribute least to peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_peaks:
                    k2 = int(k2)
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Pick candidates with low values (to minimize impact)
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
                
                for dst in candidates_sorted[:3]:
                    for frac in [0.01, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(30):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break
            
            if stall > 300:
                break
    
    return best_a, best_score


def multi_scale_refine(a0, time_limit, solver):
    """Multi-scale refinement: coarsen, optimize, upsample, fine-tune."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, 2.0 * n * fast_autocorr(a).max() / (a.sum()**2)
    a = a / s0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Coarsen by factor 2
    if n % 2 == 0:
        n_coarse = n // 2
        a_coarse = (a[0::2] + a[1::2]) / 2.0
        s_c = a_coarse.sum()
        if s_c < 1e-10:
            a_coarse = np.ones(n_coarse) / n_coarse
        else:
            a_coarse = a_coarse / s_c
        
        # Optimize at coarse level
        coarse_time = min((time_limit - (time.monotonic() - t0)) * 0.4, 120.0)
        if coarse_time > 10:

            a_out, _ = lse_lbfgs_optimize(a_coarse_norm, coarse_time, T_start=20.0, T_end=2.0, n_stages=4)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def lse_obj_grad(a_flat, n, T):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    scaled = T * (b - max_b)
    scaled = np.clip(scaled, -700.0, 0.0)
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    lse_val = max_b + np.log(exp_vals.sum()) / T
    obj = 2.0 * n * lse_val / (s * s)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    h = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    grad = (4.0 * n / (T * s * s)) * h - (4.0 * n * lse_val / (s ** 3)) * a
    return obj, grad.ravel()


def lse_lbfgs_optimize(a0, time_limit, T_start=80.0, T_end=0.5, n_stages=10, perturb=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    rng = np.random.default_rng(12345)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 100.0)

        x0 = a.copy()
        if perturb and si > 0:
            perturbation = rng.normal(0, 0.001, n)
            x0 = np.maximum(x0 + perturbation, 0)
            ss = x0.sum()
            if ss > 1e-10:
                x0 = x0 / ss

        def obj_func(x):
            val, _ = lse_obj_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_obj_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 400, 'maxfun': 1500,
                                          'time_limit': stage_time, 'gtol': 1e-9})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate / s_c
                    score_c = 2.0 * n * fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                    a = a_candidate
        except Exception:
            pass
    return best_a, best_score


def lse_momentum_descent(a0, time_limit, T_start=200.0, T_end=5.0, n_samples=3000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    velocity = np.zeros(n)
    beta = 0.9
    T = T_start
    T_decay = (T_end / T_start) ** (1.0 / n_samples)
    lr = 0.5
    for t in range(n_samples):
        if time.monotonic() - t0 > time_limit:
            break
        T *= T_decay
        obj, grad = lse_obj_grad(a, n, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        velocity = beta * velocity + (1.0 - beta) * grad
        step = lr / (grad_norm + 1e-6)
        a_new = a - step * velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def active_set_lp_bresp(a, M, solver, top_k=30, max_rounds=8):
    n = len(a)
    b_curr = fast_autocorr(a)
    active_indices = np.argsort(b_curr)[::-1][:top_k].astype(int)
    b_val = None
    for round_idx in range(max_rounds):
        K = len(active_indices)
        T_mat = np.zeros((K, n))
        for row, k in enumerate(active_indices):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_mat[row, i] = a[k_int - i]
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_mat @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            continue
        b_val = np.maximum(b_val, 0.0)
        if b_val.sum() < 1e-15:
            continue
        b_new = fast_autocorr(b_val)
        violated = np.where(b_new > M * 1.002)[0]
        if len(violated) == 0:
            return b_val
        violated_sorted = violated[np.argsort(b_new[violated])[::-1]][:10]
        active_indices = np.unique(np.concatenate([active_indices, violated_sorted]))
        if len(active_indices) > 120:
            active_indices = np.argsort(b_new)[::-1][:120].astype(int)
    if b_val is not None and b_val.sum() > 1e-15:
        return b_val
    return None
