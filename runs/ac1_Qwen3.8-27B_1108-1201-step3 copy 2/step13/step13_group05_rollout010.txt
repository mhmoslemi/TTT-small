def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def obj_from_autocorr(a, ac):
    """Compute objective from sequence and its autocorrelation."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    return 2.0 * n * ac.max() / (s * s)


def compute_subgrad_full(a, ac, beta=2.0, threshold_frac=0.7):
    """Compute subgradient using all lags above threshold_frac * max."""
    n = len(a)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n)
    threshold = M * threshold_frac
    active = np.where(ac >= threshold)[0]
    weights = ac[active] ** beta
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / w_sum

    grad = np.zeros(n)
    for wi, k in zip(weights, active):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad


def compute_subgrad_toppk(a, ac, top_k=5, beta=3.0):
    """Compute subgradient using top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx].copy(), 0.0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad


def mirror_descent(a0, time_limit, top_k=5, beta=3.0, use_full=False, threshold_frac=0.7, verbose=True):
    """Mirror descent with exponential weights on simplex."""
    n = len(a0)

            break

        step = 1.0 / (np.sqrt(t) + 1.0)

        else:
            g = compute_subgrad_toppk(a, ac, top_k=top_k, beta=beta)

        log_a = np.log(a)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new

        ac = fast_autocorr(a)
        obj = obj_from_autocorr(a, ac)

            best_a = a.copy()

        n_avg += 1

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)


def prox_gradient(a0, time_limit, top_k=3, beta=2.0, verbose=True):
    """Proximal gradient with simplex projection and Nesterov momentum."""
    n = len(a0)

        step = 1.0 / L

            v_new = proj_simplex(v - step * g, c)
            ac_new = fast_autocorr(v_new)
            M_new = ac_new.max()
            if M_new <= M - 0.5 * step * np.dot(g, v - v_new) + 1e-10:

            if obj_new < best_obj:
                best_obj = obj_new

            x_new = proj_simplex(x_new, c)
            x_prev = v_new
            v = x_new
            obj_v = obj_from_autocorr(v, fast_autocorr(v))

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving shape (normalized)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Sine
    x = np.linspace(0, 1, n)
    s = np.sin(np.pi * x) ** 2
    inits.append(s / s.sum())
    # Gaussian bump
    mu = n // 2
    sigma = n / 6
    g = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    p1 = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n / 10)) ** 2)
    p2 = np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n / 10)) ** 2)
    two = p1 + p2
    inits.append(two / two.sum())
    # Random
    r = rng.random(n)
    inits.append(r / r.sum())
    # Triangle
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    inits.append(tri / tri.sum())
    return inits

        a_out, _ = mirror_descent(a_start, t_alloc, top_k=5, beta=2.0, use_full=True, threshold_frac=0.5, verbose=True)
        try_update(a_out, f"md_full_n{n_cur}")

    # Phase 3: Proximal gradient
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: ProxGrad n={len(best_seq)} ===")
    if remaining > 40:
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.10, 120.0)
        a_out, _ = prox_gradient(a_start, t_alloc, top_k=3, beta=2.0, verbose=True)
        try_update(a_out, f"pg_n{n_cur}")

    # Phase 4: LP Nesterov on current best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={len(best_seq)} ===")
    if remaining > 60:
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.15, 250.0)
        a_out, _ = nesterov_lp(a_start, t_alloc, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # Phase 5: Scale up to longer sequence, MD
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = max(2400, len(best_seq) + 800)
    print(f"\n=== Phase 5: MD at n={n_long} ===")
    if remaining > 60:
        a_interp = interpolate_to_length(best_seq, n_long)
        c_long = np.sqrt(2.0 * n_long)
        a_start = a_interp * c_long / a_interp.sum()
        t_alloc = min(remaining * 0.15, 200.0)
        a_out, _ = mirror_descent(a_start, t_alloc, top_k=5, beta=2.5, verbose=True)
        try_update(a_out, f"md_long_n{n_long}")

    # Phase 6: MD at n_long with full subgrad
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: MD full at n={len(best_seq)} ===")
    if remaining > 60:
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.12, 180.0)
        a_out, _ = mirror_descent(a_start, t_alloc, top_k=3, beta=1.5, use_full=True, threshold_frac=0.3, verbose=True)
        try_update(a_out, f"md_full2_n{n_cur}")

    # Phase 7: LP on longest
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LP at n={len(best_seq)} ===")
    if remaining > 60:
        n_cur = len(best_seq)
        if n_cur <= 3000:  # LP gets slow for large n
            c_cur = np.sqrt(2.0 * n_cur)
            a_start = best_seq * c_cur / best_seq.sum()
            t_alloc = min(remaining * 0.12, 180.0)
            a_out, _ = nesterov_lp(a_start, t_alloc, solver, verbose=True)
            try_update(a_out, f"lp2_n{n_cur}")
        else:
            # Skip LP for very long sequences, use MD instead
            c_cur = np.sqrt(2.0 * n_cur)
            a_start = best_seq * c_cur / best_seq.sum()
            t_alloc = min(remaining * 0.12, 180.0)
            a_out, _ = mirror_descent(a_start, t_alloc, top_k=7, beta=2.0, verbose=True)
            try_update(a_out, f"md7_n{n_cur}")

    # Phase 8: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Coordinate Descent n={len(best_seq)} ===")
    if remaining > 30:
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.15, 200.0)
        a_out, _ = coordinate_descent(a_start, t_alloc, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # Phase 9: Final MD with aggressive params
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final MD n={len(best_seq)} ===")
    if remaining > 30:
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.20, 200.0)
        a_out, _ = mirror_descent(a_start, t_alloc, top_k=3, beta=1.5, verbose=True)
        try_update(a_out, f"md_final_n{n_cur}")

    # Phase 10: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 10: Final CD polish ===")
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        t_alloc = min(remaining * 0.5, 100.0)
        a_out, _ = coordinate_descent(a_start, t_alloc, verbose=True)
        try_update(a_out, f"cd_final_n{n_cur}")

    # Final: one more MD with whatever time is left
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Final MD (remaining {remaining:.0f}s) ===")
        n_cur = len(best_seq)
        c_cur = np.sqrt(2.0 * n_cur)
        a_start = best_seq * c_cur / best_seq.sum()
        a_out, _ = mirror_descent(a_start, remaining - 5.0, top_k=5, beta=2.0, verbose=False)
        try_update(a_out, "md_last")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Returning uniform.")
        best_seq = np.ones(1600) / 1600
        best_score = evaluate_safe(best_seq)
        print(f"Uniform fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
