"""

1. LSE gradient descent with Nesterov acceleration (warm-up)
2. Fast 2-swap local search with vectorized L,Q computation and ternary search

4. Multi-length exploration with interpolation transfer
5. Adaptive candidate generation and batch processing

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Compute gradient of LSE surrogate T*log(sum exp(b_k/T))."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2


def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(50, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if it == 0:
                y_new = y - (1.0 / (it + 3.0)) * lse_gradient(y, T)
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrapolated = y + momentum * (y - y_prev)
                y_new = y_extrapolated - (1.0 / (it + 3.0)) * lse_gradient(y_extrapolated, T)
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            if it_count % 300 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_LQ(a, i, j):
    """
    Compute L and Q arrays such that after swap a[i]-=delta, a[j]+=delta:
    conv_new[k] = conv_old[k] + L[k]*delta + Q[k]*delta^2
    """
    n = len(a)
    size = 2 * n - 1
    k = np.arange(size)
    L = np.zeros(size)
    Q = np.zeros(size)

    # m = i: valid for k in [i, i+n-1]
    mask_i = (k >= i) & (k <= i + n - 1)

    # Case k == 2i: single term a[i]^2 -> (a[i]-delta)^2
    m_ii = mask_i & (k == 2 * i)
    if np.any(m_ii):
        L[m_ii] += -2.0 * a[i]
        Q[m_ii] += 1.0

    # Case k == i+j: term a[i]*a[j] -> (a[i]-delta)(a[j]+delta)
    m_ij = mask_i & (k == i + j)
    if np.any(m_ij):
        L[m_ij] += (a[i] - a[j])
        Q[m_ij] += -1.0

    # Case partner != i, j: term a[i]*a[k-i] -> (a[i]-delta)*a[k-i]
    m_ip = mask_i & (k != 2 * i) & (k != i + j)
    if np.any(m_ip):
        L[m_ip] += -a[k[m_ip] - i]

    # m = k-i (partner = i): same valid range, exclude k=2i and k=i+j
    mask_kmi = mask_i & (k != 2 * i) & (k != i + j)
    if np.any(mask_kmi):
        L[mask_kmi] += -a[k[mask_kmi] - i]

    # m = j: valid for k in [j, j+n-1]
    mask_j = (k >= j) & (k <= j + n - 1)

    # Case k == 2j: single term a[j]^2 -> (a[j]+delta)^2
    m_jj = mask_j & (k == 2 * j)
    if np.any(m_jj):
        L[m_jj] += 2.0 * a[j]
        Q[m_jj] += 1.0

    # Case k == i+j: term a[j]*a[i] -> (a[j]+delta)(a[i]-delta)
    m_ji = mask_j & (k == i + j)
    if np.any(m_ji):
        L[m_ji] += (a[i] - a[j])
        Q[m_ji] += -1.0

    # Case partner != i, j: term a[j]*a[k-j] -> (a[j]+delta)*a[k-j]
    m_jp = mask_j & (k != 2 * j) & (k != i + j)
    if np.any(m_jp):
        L[m_jp] += a[k[m_jp] - j]

    # m = k-j (partner = j): same valid range, exclude k=2j and k=i+j
    mask_kmj = mask_j & (k != 2 * j) & (k != i + j)
    if np.any(mask_kmj):
        L[mask_kmj] += a[k[mask_kmj] - j]

    return L, Q


def ternary_search_delta(conv, L, Q, delta_lo, delta_hi, iterations=40):
    """Find delta minimizing max_k(conv[k] + L[k]*delta + Q[k]*delta^2)."""
    lo = delta_lo
    hi = delta_hi
    for _ in range(iterations):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(conv + L * m1 + Q * m1 * m1)
        f2 = np.max(conv + L * m2 + Q * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining LSE gradient descent, 2-swap local search,
    and multi-length exploration to minimize the objective.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LSE Nesterov warm-up ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=15.0, T_end=0.3, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 2: 2-Swap main optimization ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: 2-Swap n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 600.0)
    if t2 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        conv_work = fast_autocorr(a_work)
        a_out, sc_out, _ = two_swap_optimize(a_work, conv_work, t2, seed=12345, verbose=True)
        do_try_update(a_out, f"swap_n{n_best}")

    # ============ Phase 3: Mirror descent at current best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: MirrorDescent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mirror_descent_lse(a_work, t3, T_start=8.0, T_end=0.2, n_stages=8, verbose=True)
        do_try_update(a_out, f"mirror_n{n_best}")

    # ============ Phase 4: 2-Swap after mirror descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: 2-Swap n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        conv_work = fast_autocorr(a_work)
        a_out, sc_out, _ = two_swap_optimize(a_work, conv_work, t4, seed=99999, verbose=True)
        do_try_update(a_out, f"swap2_n{n_best}")

    # ============ Phase 5: Try longer length ============
    n2 = n_best + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Longer length n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        # Quick LSE warm-up
        t5a = min(t5 * 0.3, 60.0)
        a_out, _ = nesterov_lse(a_work, t5a, T_start=10.0, T_end=0.5, n_stages=6, verbose=False)
        do_try_update(a_out, f"lse_long_n{n2}")
        # 2-swap on longer
        remaining5 = t5 - (time.monotonic() - t_global + (time_budget - remaining))
        t5b = min(remaining * 0.15, 180.0)
        if t5b > 20:
            n_cur = len(best_seq)
            a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            conv_work = fast_autocorr(a_work)
            a_out, sc_out, _ = two_swap_optimize(a_work, conv_work, t5b, seed=55555, verbose=True)
            do_try_update(a_out, f"swap_long_n{n_cur}")

    # ============ Phase 6: Final LSE polish at very low T ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish T low n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t6, T_start=3.0, T_end=0.1, n_stages=6, verbose=True)
        do_try_update(a_out, "lse_polish")

    # ============ Phase 7: Final 2-swap polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.50, 200.0)
        print(f"\n=== Phase 7: Final 2-Swap n={n_best} (budget={t7:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        conv_work = fast_autocorr(a_work)
        a_out, sc_out, _ = two_swap_optimize(a_work, conv_work, t7, seed=77777, verbose=True)
        do_try_update(a_out, "swap_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
