def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_surrogate(b, p, T=1.0):
    """
    Compute the p-norm surrogate: (sum |b_k|^p)^(1/p) approximated via log-sum-exp.
    Returns (value, gradient weights).
    """
    b = np.maximum(b, 0)  # We know b >= 0 for autocorrelation of non-negative a
    max_b = b.max()
    if max_b < 1e-300:
        return 0.0, np.ones_like(b) / len(b)
    # (sum b_k^p)^(1/p) = max_b * (sum (b_k/max_b)^p)^(1/p)
    # log of this = log(max_b) + (1/p) * log(sum (b_k/max_b)^p)
    # For gradient, we use softmax-like weights
    w = np.power(b / max_b, p)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    value = max_b * np.power(w_sum, 1.0 / p)
    return value, w

def p_norm_value(a, p):
    """Compute p-norm surrogate value of conv(a,a)."""
    b = fast_autocorr(a)
    b = np.maximum(b, 0)
    max_b = b.max()
    if max_b < 1e-300:
        return 0.0
    w_sum = np.sum(np.power(b / max_b, p))
    return max_b * np.power(w_sum, 1.0 / p)


def nesterov_pnorm(a0, time_limit, p_start=8.0, p_end=64.0, n_stages=15, verbose=True):
    """Nesterov-accelerated gradient descent on p-norm surrogate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    ps = np.geomspace(p_start, p_end, n_stages)
    it_count = 0

    for si, p in enumerate(ps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                y_extrap = y
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e < 1e-15:
                    y_extrap = y.copy()
                else:
                    y_extrap = y_extrap * norm / s_e

            grad = p_norm_gradient(y_extrap, p)
            
            # Step size: use backtracking or fixed
            eta = 1.0 / (it + 10.0)
            
            y_new = y_extrap - eta * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            # Track best
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if verbose and it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PNorm-Nesterov stage={si}/{n_stages} p={p:.1f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient(a, T):
    """LSE gradient (same as before but cleaner)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:

        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)

def lp_best_response_fast(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_j <= M for all j, b >= 0. Using sparse matrix."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build sparse constraint matrix
    # (a*b)_j = sum_i a[i] * b[j-i] for valid i, j-i
    # For constraint j (0 to 2n-2):
    #   sum over k where 0 <= k < n and 0 <= j-k < n of a[j-k] * b[k]
    #   Let me index: for b[k], the coefficient in constraint j is a[j-k] if 0 <= j-k < n
    
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for k in range(n):
        # b[k] appears in constraints j where 0 <= j-k < n, i.e., k <= j < k+n
        js = k
        je = min(k + n - 1, 2 * n - 2)
        for j in range(js, je + 1):
            T[j, k] = a[j - k]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov-style averaging."""
    n = len(a0)

def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: try modifying individual elements to reduce max conv."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        
        # Try moving mass from positions that contribute to the peak
        # to positions that reduce the peak
        improved = False
        n_tries = 0
        
        # Positions that contribute to b[peak_idx]
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contributing = np.arange(js, je + 1)
        
        # Find which contributing positions have the largest a values
        contrib_vals = a[contributing]
        top_contrib = contributing[np.argsort(contrib_vals)[::-1][:5]]
        
        for src in top_contrib:
            if a[src] < norm * 1e-6:
                continue
            
            # Find positions where moving mass would reduce the peak
            # Try positions with low "interaction" with the peak
            for _ in range(20):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-16 or a[src] - amt < -1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    n_tries += 1
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
        
        if verbose and (no_improve % 100 == 0 or best_score < 1.503):
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def adversarial_gradient_step(a, T, step_size, rng):
    """
    A novel approach: instead of using the exact gradient, we use a 
    "perturbed" gradient that focuses on the top few peaks simultaneously.
    This helps escape local minima where only one peak dominates.
    """
    n = len(a)
    b = fast_autocorr(a)
    # Focus on top k peaks with equal weight
    k = min(5, len(b))
    top_ks = np.argsort(b)[::-1][:k]
    w = np.zeros_like(b)
    for ik in top_ks:
        w[ik] = 1.0 / k
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(W * A, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    
    return r1 + r2

_global_solver = None


def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolution_gradient(a, w):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    Returns array of length n.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(W * A, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def lse_weights(b, T):
    """Compute LSE softmax weights."""
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


def pnorm_weights(b, p):
    """Compute p-norm weights."""
    b_pos = np.maximum(b, 0)
    max_b = b_pos.max()
    if max_b < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    w = np.power(b_pos / max_b, p)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


def topk_weights(b, k):
    """Give equal weight to top-k peaks."""
    w = np.zeros_like(b)
    top_ks = np.argsort(b)[::-1][:k]
    w[top_ks] = 1.0 / k
    return w


def nesterov_lse(a0, time_limit, T_start=10.0, T_end=0.1, n_stages=15, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(50, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it < 3:
                y_extrap = y
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e < 1e-15:
                    y_extrap = y.copy()
                else:
                    y_extrap = y_extrap * norm / s_e

            b = fast_autocorr(y_extrap)
            w = lse_weights(b, T)
            grad = convolution_gradient(y_extrap, w)

            eta = 1.0 / (it + 10.0)
            y_new = y_extrap - eta * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if verbose and it_count % 300 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE s={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): {best_score:.8f}")

    return best_a, best_score

def lp_best_response_fast(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for k in range(n):
        js = k
        je = min(k + n - 1, 2 * n - 2)
        coeffs = a[0:je - js + 1]
        T[js:je + 1, k] = coeffs
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:4]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]

                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
