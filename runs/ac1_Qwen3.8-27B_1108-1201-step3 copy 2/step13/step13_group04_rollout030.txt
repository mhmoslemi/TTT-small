    a /= a.sum()

    # Gradient w.r.t. a
    fft_size = 1 << (3 * n - 2).bit_length()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    A_fwd = rfft(a, fft_size)
    W = rfft(b_p, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n-1:n-1+n]
    r2 = conv2[:n]
    
    return prefactor * (r1 + r2)

def power_mean_z_obj_grad(z, p):
    """Objective and gradient for power-mean in z-space (a = softmax(z))."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    b = fast_autocorr(a)
    
    K = len(b)
    b_p = np.sum(b**p)
    S = (b_p / K) ** (1.0/p)
    obj = S
    
    # Gradient
    grad_a = compute_gradient_power(a, b, p)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def lse_z_obj_grad(z, T):
    """LSE surrogate objective and gradient in z-space."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    b = fast_autocorr(a)
    
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w_sum = exp_b.sum()
    w = exp_b / w_sum
    obj = T * np.log(w_sum) + max_b
    
    # Gradient
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n-1:n-1+n]
    r2 = conv2[:n]
    
    grad_a = r1 + r2
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

def run_lse_bfgs(z0, T, time_limit, maxiter=300):
    """Run L-BFGS on LSE surrogate."""
    t_start = time.monotonic()
    
    def obj_func(z):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError("Time limit exceeded")
        return lse_z_obj_grad(z, T)[0]
    
    def grad_func(z):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError("Time limit exceeded")
        return lse_z_obj_grad(z, T)[1]
    
    try:
        result = minimize(obj_func, z0, jac=grad_func, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'maxfun': 500, 'gtol': 1e-7})
        return result.x
    except Exception:
        return z0

def run_power_bfgs(z0, p, time_limit, maxiter=300):
    """Run L-BFGS on power-mean surrogate."""
    t_start = time.monotonic()
    
    def obj_func(z):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError("Time limit exceeded")
        return power_mean_z_obj_grad(z, p)[0]
    
    def grad_func(z):
        if time.monotonic() - t_start > time_limit:
            raise TimeoutError("Time limit exceeded")
        return power_mean_z_obj_grad(z, p)[1]
    
    try:
        result = minimize(obj_func, z0, jac=grad_func, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'maxfun': 500, 'gtol': 1e-7})

    except Exception:
        return z0

def multi_power_bfgs(z0, time_limit, verbose=True):
    """Run L-BFGS with increasing power parameter."""
    powers = [2, 3, 4, 6, 8, 12, 16, 24, 32]
    z = z0.copy()
    t_start = time.monotonic()
    
    for p in powers:
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 5:
            break
        # Allocate time proportionally to remaining powers
        n_remaining = len(powers) - powers.index(p)
        t_alloc = min(remaining * 0.7 / n_remaining, remaining * 0.4)
        if t_alloc < 3:
            t_alloc = remaining * 0.3
        
        if verbose:
            a_curr = softmax_safe(z)
            score_curr = fast_autocorr(a_curr).max()
            print(f"    Power={p}: current peak={score_curr:.8f}, time={time.monotonic()-t_start:.0f}s")
        
        z = run_power_bfgs(z, p, t_alloc, maxiter=200)
        
        if verbose:
            a_curr = softmax_safe(z)
            score_curr = fast_autocorr(a_curr).max()
            print(f"    Power={p} done: peak={score_curr:.8f}, time={time.monotonic()-t_start:.0f}s")
    
    return z

def softmax_safe(z):
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s

def a_to_z(a):
    """Convert non-negative sequence (sum~1) to z parameterization."""
    a = np.maximum(a, 1e-15)
    return np.log(a)

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, seed):
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Frequency-domain init
        phases = rng.uniform(0, 2*np.pi, n)
        A = np.exp(1j * phases)
        a = np.abs(ifft(A).real)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Improved peak-targeting local search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        sorted_idx = np.argsort(b)[::-1]
        top_ks = sorted_idx[:6]
        peak_val = b[sorted_idx[0]]
        
        improved = False
        for k_star in top_ks:
            if b[k_star] < 0.9 * peak_val:
                break
            
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_star - idx]
            top_contrib = np.argsort(contrib)[::-1][:5]
            
            for ci in range(len(top_contrib)):
                src = idx[top_contrib[ci]]
                if a[src] < norm * 1e-10:
                    continue
                
                # Compute danger score for each position
                danger = np.zeros(n)
                for k2 in top_ks[:3]:
                    partner = k2 - np.arange(n)
                    valid = (partner >= 0) & (partner < n)
                    danger[valid] += a[np.arange(n)[valid]] * a[partner[valid]]
                
                candidates = np.where((danger < np.percentile(danger, 30)) & (np.arange(n) != src))[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                
                rng.shuffle(candidates)
                
                for dst in candidates[:8]:
                    for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-16:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, 4)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.001, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 200:
                break

        if verbose and it_count % 1000 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score
