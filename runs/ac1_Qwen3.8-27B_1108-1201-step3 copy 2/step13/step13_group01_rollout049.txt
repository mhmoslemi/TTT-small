def build_conv_matrix(n, a):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:

    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()

def lse_objective_and_grad(a_flat, n, temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    shifted = b - max_b
    exp_vals = np.exp(temp * shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / temp
    obj = 2.0 * n * lse / (s * s)

    # Gradient computation via FFT cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s

    return obj, grad.ravel()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:

        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * 2 * np.pi * x) + 0.15 * np.cos(5 * 2 * np.pi * x) + 0.05 * np.cos(9 * 2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        a = np.exp(-4 * (x - 0.25) ** 2) + 0.6 * np.exp(-4 * (x - 0.75) ** 2)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + 0.1 * np.sin(2 * np.pi * 3 * x) + 0.05 * np.sin(2 * np.pi * 7 * x))
    else:
        a = np.ones(n) * (1.0 + 0.1 * np.random.default_rng(0).normal(size=n))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_peak_lp_direction(a, M, solver, n_peaks=3, T=None):
    """Solve LP targeting the top n_peaks specifically."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        size = 2 * n - 1
        T = np.zeros((size, n))
        for i in range(n):
            start = i
            end = min(i + n, size)
            idx = np.arange(start, end)
            T[idx, i] = a[idx - i]

    b_curr = fast_autocorr(a)
    top_peaks = np.argsort(b_curr)[::-1][:n_peaks]

    # Use only top peak constraints (tighter problem)
    constraints = [T[top_peaks, :] @ b_var <= M]
    constraints.append(T @ b_var <= 1.5 * M)  # Loose constraint for others
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1400, 1600, 1800, 2000]
    scan_results = {}
    phase0_budget = time_budget * 0.10

    for i, n_scan in enumerate(scan_lengths):
        if time.monotonic() - t_global > t_global + phase0_budget:
            break
        remaining_scan = (t_global + phase0_budget) - time.monotonic()
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / (len(scan_lengths) - i), 60.0)

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (total={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"\nAfter Phase 0: best_score={best_score:.6f}, n={n_base}")

    # Phase 1: Deep LP-Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                     momentum_reset=True, polyak_avg=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Also try with a slightly perturbed initialization
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        t1b = min(remaining * 0.10, 150.0)
        a_pert = best_seq * (1.0 + 0.05 * rng.normal(size=n_base))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        a_norm_p = a_pert * np.sqrt(2.0 * n_base) / a_pert.sum()
        a_out_p, _ = nesterov_lp_optimize(a_norm_p, t1b, solver, verbose=False,
                                           momentum_reset=True, polyak_avg=True)
        do_try_update(a_out_p, f"lp_pert_n{n_base}")

    # Phase 2: LSE + scipy at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=30.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_scipy_n{n_best}")

    # Phase 3: Advanced coordinate descent with SA
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent SA n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = advanced_coord_descent(a_norm, t3, seed=12345, sa_temp=0.005, decay=0.97)
        do_try_update(a_out, "coord_sa")

    # Phase 4: Second LP pass from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Second LP n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True,
                                         momentum_reset=True, polyak_avg=True)
        do_try_update(a_out, f"lp_pass2_n{n_best}")

    # Phase 5: LSE polish with lower temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t5, T_start=8.0, T_end=0.5, n_stages=4)
        do_try_update(a_out, "lse_polish")

    # Phase 6: Final coordinate descent (more aggressive)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 350.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = advanced_coord_descent(a_norm, t6, seed=77777, sa_temp=0.002, decay=0.99)
        do_try_update(a_out, "coord_final")

    # Phase 7: Try a different length if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_best + 200
    print(f"\n=== Phase 7: LP n={n_alt} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.40, 300.0)
    if t7 > 30:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp_optimize(a_norm_alt, t7, solver, verbose=True,
                                         momentum_reset=True, polyak_avg=True)
        do_try_update(a_out, f"lp_alt_n{n_alt}")
        # Polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 200.0)
            if t7b > 10:
                a_norm_p = a_out * np.sqrt(2.0 * n_alt) / a_out.sum()
                a_out2, _ = advanced_coord_descent(a_norm_p, t7b, seed=99999, sa_temp=0.003, decay=0.98)
                do_try_update(a_out2, f"coord_alt_n{n_alt}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
