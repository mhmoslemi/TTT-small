    """L-BFGS-B optimization on LSE surrogate with temperature schedule."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    for T in T_schedule:
        if _elapsed() > _TOTAL_TIME:
            break
        remaining = _TOTAL_TIME - _elapsed()
        if remaining < 10:
            break

        def obj(x):
            x = np.maximum(x, 0.0)
            b = fast_autocorr(x)
            return lse_value(b, T)

        def grad(x):
            x = np.maximum(x, 0.0)
            return compute_grad(x, T)

        try:
            res = scipy_minimize(obj, a.copy(), method='L-BFGS-B',
                                  jac=grad, bounds=[(0, None)] * n,
                                  options={'maxiter': max_iter_per_stage,
                                           'maxls': 50, 'ftol': 1e-14,
                                           'gtol': 1e-10, 'maxfun': 1000})
            if res.x is not None:
                a_c = np.maximum(res.x, 0.0)
                sc = a_c.sum()
                if sc > 1e-10:
                    a_c = a_c * norm / sc
                    score = fast_autocorr(a_c).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_c.copy()
                        a = a_c
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent: move mass from high-contributing peaks to low areas."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:6]

        improved = False
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:6]]

            # Find low-impact destinations
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(a < np.median(a))[0]
            if len(candidates) == 0:
                candidates = np.argsort(a)[:n // 4]

            for src in top_src:
                if a[src] < 1e-10:
                    continue
                for dst in candidates[np.argsort(a[candidates])[:4]]:
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-10:
                            continue
                        a_trial = a_trial * norm / st
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stalling
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial * norm / st
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def structured_init(n, rng, method='cosine'):
    """Generate diverse structured initializations."""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 8)

    if method == 'cosine':
        m = int(rng.integers(1, max(2, n // 100)))
        alpha = rng.uniform(0.1, 0.8)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
    elif choice == 0:
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * x) + 0.15 * np.cos(10.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = 1.0 + 0.5 * ((2.0 * x - 1.0) ** 2)  # U-shaped
    elif choice == 5:
        # Pseudo-random via golden ratio
        phi = 0.6180339887498949
        a = 0.5 + 0.5 * np.abs(np.sin(2.0 * np.pi * phi * np.arange(n)))
    elif choice == 6:
        # Multi-frequency
        a = 1.0
        for _ in range(int(rng.integers(2, 5))):
            m = int(rng.integers(1, 10))
            a += 0.2 * np.cos(2.0 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
    else:
        # Gaussian mixture
        centers = rng.uniform(0.1, 0.9, 3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-10.0 * (x - c) ** 2)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    global _T0
    _T0 = time.monotonic()
    _TOTAL_TIME_LOCAL = min(float(budget_s) - 60.0, 1740.0)

    rng = np.random.default_rng(seed)
    print(f"Time budget: {_TOTAL_TIME_LOCAL:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={_elapsed():.0f}s)")

    def _evaluate_safe(a):
        try:
            a_list = [float(x) for x in a]
            return float(evaluate_sequence(a_list))
        except Exception:
            return float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Phase 1: Fast multi-start with multiplicative weights
    print("\n=== Phase 1: Multi-start multiplicative weights ===")
    n_base = len(best_seq) if best_seq is not None else 1600
    scan_lengths = [n_base]
    if n_base != 2000:
        scan_lengths.append(2000)
    if n_base != 1200:
        scan_lengths.append(1200)

    phase1_time = _TOTAL_TIME_LOCAL * 0.30
    per_start_time = phase1_time / (len(scan_lengths) * 6)

    for n_scan in scan_lengths:
        if _elapsed() > _TOTAL_TIME_LOCAL * 0.35:
            break
        for init_idx in range(6):
            if _elapsed() > _TOTAL_TIME_LOCAL * 0.35:
                break
            if best_seq is not None and init_idx == 0:
                init_seq = interpolate_sequence(best_seq, n_scan)
            else:
                init_seq = structured_init(n_scan, rng, method='cosine')

            a_out, sc_out = multiplicative_weights(
                init_seq,
                T_schedule=[1.0, 3.0, 10.0, 30.0, 100.0],
                steps_per_stage=500,
                eta=0.5
            )
            try_update(a_out, f"mw_n{n_scan}_i{init_idx}")

    # Phase 2: Adam refinement on best candidates
    print(f"\n=== Phase 2: Adam refinement (best_score={best_score:.6f}) ===")
    n_best = len(best_seq)
    phase2_time = _TOTAL_TIME_LOCAL * 0.25
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()

    a_out, _ = adam_optimize(
        a_norm,
        T_schedule=[5.0, 15.0, 50.0, 150.0, 400.0],
        steps_per_stage=2000,
        lr_init=0.05
    )
    try_update(a_out, "adam_refine")

    # Phase 3: L-BFGS-B with high temperature
    print(f"\n=== Phase 3: L-BFGS-B (best_score={best_score:.6f}) ===")
    n_best = len(best_seq)
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = lse_scipy_optimize(
        a_norm,
        T_schedule=[20.0, 60.0, 150.0, 400.0],
        max_iter_per_stage=200
    )
    try_update(a_out, "lbfgsb_deep")

    # Phase 4: Try longer length
    print(f"\n=== Phase 4: Longer length (best_score={best_score:.6f}) ===")
    n_long = n_best + 400
    if _TOTAL_TIME_LOCAL - _elapsed() > 120:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = multiplicative_weights(
            init_long,
            T_schedule=[1.0, 5.0, 15.0, 50.0, 150.0],
            steps_per_stage=400,
            eta=0.4
        )
        try_update(a_out, f"mw_long_n{n_long}")

        if _TOTAL_TIME_LOCAL - _elapsed() > 60:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = adam_optimize(
                a_norm,
                T_schedule=[10.0, 30.0, 100.0, 300.0],
                steps_per_stage=1500,
                lr_init=0.03
            )
            try_update(a_out, "adam_long_refine")

    # Phase 5: Coordinate descent polish
    print(f"\n=== Phase 5: Coord descent polish (best_score={best_score:.6f}) ===")
    n_best = len(best_seq)
    remaining = _TOTAL_TIME_LOCAL - _elapsed()
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        try_update(a_out, "coord_polish")

    # Phase 6: Final Adam with very high T
    print(f"\n=== Phase 6: Final high-T Adam (best_score={best_score:.6f}) ===")
    remaining = _TOTAL_TIME_LOCAL - _elapsed()
    n_best = len(best_seq)
    if remaining > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_optimize(
            a_norm,
            T_schedule=[100.0, 300.0, 800.0, 2000.0],
            steps_per_stage=3000,
            lr_init=0.01
        )
        try_update(a_out, "adam_hight_final")

    # Phase 7: Final coordinate descent
    remaining = _TOTAL_TIME_LOCAL - _elapsed()
    if remaining > 30:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, min(remaining * 0.6, 200.0), seed=99999)
        try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={_elapsed():.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
