def _fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def _softmax_subgrad(a, ac, temp=0.5, floor_frac=0.5):
    """Compute subgradient using softmax weighting over all lags above a threshold."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = floor_frac * M
    # Weights: softmax over lags above threshold
    above = ac >= threshold
    if not np.any(above):
        above = np.argsort(ac)[::-1] >= 0  # all
        above = ac >= max(threshold * 0.5, ac.max() - 1e-10)
    log_w = np.where(above, ac / max(temp, 1e-10), -np.inf)
    log_w -= log_w.max()  # for numerical stability
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w_idx = np.argmax(ac)
        w = np.zeros(len(ac))
        w[w_idx] = 1.0
    else:
        w /= w_sum
    # Compute subgradient
    grad = np.zeros(n)
    active_k = np.where(w > 1e-12)[0]
    for k in active_k:
        k_int = int(k)
        wi = w[k_int]
        # ∂c[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def _topk_subgrad(a, ac, top_k=10, beta=3.0):
    """Compute weighted subgradient using top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def _mirror_descent(a0, time_limit, top_k=10, beta=3.0, temp=0.5, verbose=True):

    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    n_avg = 1

    for t in range(1, 50000000):

        ac = _fast_autocorr(a)

        # Use adaptive method: top-k for speed, softmax for smoothness

        else:
            g = _softmax_subgrad(a, ac, temp=temp, floor_frac=0.7)
        
        # Mirror descent step
        step = gamma / (np.sqrt(t) + 1.0)
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        a = a_new
        
        M_new = _fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
