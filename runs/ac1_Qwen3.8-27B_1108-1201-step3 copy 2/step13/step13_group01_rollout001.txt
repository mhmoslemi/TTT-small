"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Per-solve max-length/theta selection (avoids wasted LP size)
2. LP best-response at current peak with adaptive Nesterov averaging
3. Multi-start + restarts to escape local optima
4. Post-optimization shrinking of trailing support
5. Sweeps over lengths and theta values
"""

import numpy as np
import cvxpy as cp
import time


_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()

def lp_best_response(a, R, solver):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        s = i
        e = min(i + n, size)
        idx = np.arange(s, e)
        T[idx, i] = a[idx - i]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= R])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    if val.sum() < 1e-15:
        return None
    return val


def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best = a.copy()
    best_mx = fast_autocorr(a).max()
    y = a.copy()
    avg = np.zeros(n)
    cnt = 0
    t0 = time.monotonic()
    t = 1
    last_impr = t0
    stall = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        muc = fast_autocorr(y).max()
        if muc < 1e-12:
            break
        g = lp_best_response(y, muc, solver)
        if g is None:
            continue
        gs = g.sum()
        if gs < 1e-15:
            continue
        g = g / gs
        best_mx = fast_autocorr(y).max()
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()
        if t == 1:
            a_new = g.copy()
        else:
            m = (t - 1.0) / (t + 2.0)
            a_new = y_new + m * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
        y = y_new
        a = a_new
        ny = fast_autocorr(y).max()
        if ny < best_mx:
            best_mx = ny
            best = y.copy()
            last_impr = time.monotonic()
        na = fast_autocorr(a).max()
        if na < best_mx:
            best_mx = na
            best = a.copy()
            last_impr = time.monotonic()
        avg += a
        cnt += 1
        if cnt % 50 == 0:
            av = avg / cnt
            av = av / av.sum()
            sc = fast_autocorr(av).max()
            if sc < best_mx:
                best_mx = sc
                best = av.copy()
                last_impr = time.monotonic()
        if time.monotonic() - last_impr > 60.0:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
        if verbose and t % 100 == 0:
            print(f"      t={t} m={best_mx:.8f} elapsed={time.monotonic()-t0:.0f}s")
        t += 1
    return best, best_mx


def shrink_trailing(a, max_rounds=30, tol=1e-4):
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    for r in range(max_rounds):
        run = 0
        p = n - 1
        while p >= 0 and a[p] < tol:
            run += 1
            p -= 1
        if run == 0:
            break
        newa = a[:-1].copy()
        if newa.sum() < 1e-15:
            return a
        newa = newa / newa.sum()
        if fast_autocorr(newa).max() <= fast_autocorr(a).max() + 1e-12:
            a = newa
        else:
            break
    return a

    elif c == 3:
        x = np.linspace(0.0, 1e-7, n)
        a = np.exp(-4 * (x - 0.25) ** 2) + 0.6 * np.exp(-4 * (x - 0.75) ** 2)
    elif c == 4:
        x = np.linspace(0.0, 1e-7, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif c == 5:
        ctr = n // 2
        idx = np.arange(n) - ctr
        sig = max(1.0, n / 5.0)
        a = np.exp(-0.5 * (idx / sig) ** 2)
    elif c == 6:

        a = np.cos(x / 2.0) ** 2
    else:
        a = np.ones(n) * (1.0 - 0.3 * np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def run_one(n, theta, solver, tlimit, init, verbose=False):
    if init is not None:
        a0 = np.array(init, dtype=np.float64).copy()
        if len(a0) != n:
            a0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a0)), a0)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.ones(n) / n
    R0 = theta * a0.sum() ** 2 / n
    a_opt, R_opt = nesterov_lp(a0, tlimit, solver, verbose)
    a_opt = np.maximum(a_opt, 0.0)

        return None
    a_norm = a_opt / s
    R = theta * a_norm.sum() ** 2 / n
    a_shrunk = shrink_trailing(a_norm, max_rounds=20, tol=1e-4)
    return a_shrunk
