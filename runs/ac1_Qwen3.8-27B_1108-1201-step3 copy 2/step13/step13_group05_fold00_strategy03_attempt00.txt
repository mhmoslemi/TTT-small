def evaluate_safe(a):
    """Sicherheitssicher Evaluation der Schrittfunktion. Setzt einen Default im Fehlerfall."""
    try:
        a = np.array(a, dtype=np.float64)
        a_sum = np.sum(a)
        if a_sum < 0.01 or a_sum > 1e6:
            return float('inf')
        a = np.clip(a / a_sum * np.sqrt(2.0 * len(a)) , 0, 1000)
        n_val = len(a)
        conv = np.zeros(2*n_val-1)
        for i in range(n_val):
            for k in range(2*n_val):
                if k-i >= 0 and k-i < n_val:
                    conv[k] += a[i]
        
        conv_max = np.clip(conv.max(), 0, None)
        normalized_conv = conv_max / 2
        normalized_sum_sq = (np.linalg.norm(a)**2 + 1e-10)
        
        if verbose:
            print(f"Evaluation: Sum={a_sum}, Max.Normalized.Conv={conv_max}, Normalized.J={normalized_conv:.6f} (denominator {normalized_sum_sq:.6f})")
            
        return normalized_conv
    except Exception as e:
        if verbose:
            print("Error in evaluate_safe. Returning penalty.")
            print(e)
        return float('inf')

    def total_variation(seq):
        diff = np.diff(seq)
        positive = np.sum(diff)
        negative = np.sum(np.abs(diff))
        return positive - negative

    def clip_to_1000(seq):
        clipped = []
        for x in seq:
            if x < 0:
                clipped.append(0)
            elif x > 1000:
                clipped.append(1000)
            else:
                clipped.append(x)
