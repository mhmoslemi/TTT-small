import numpy as np

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def mirror_descent(a0, time_limit, n_top=5, beta=2.0, rho=0.7, eta0=5.0, tau=100):
    """Mirror descent with exponential weights on simplex."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()  # normalize to sum=1 for stability
    
    best_a = a.copy()
    best_ac = fast_autocorr(a)
    best_m = best_ac.max()
    t0 = time.monotonic()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute subgradient using top-n_top lags
        top_idx = np.argsort(ac)[::-1][:n_top]
        w = ac[top_idx].copy()
        w = np.maximum(w, 0)
        w = w ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            j = k - idx
            grad[idx] += 2.0 * wi * a[j]
        
        # Mirror descent update
        step = eta0 / ((t / tau) + 1.0) ** rho
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, n_top=5):
    """Coordinate descent polish."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            j = k - idx
            contrib = a[idx] * a[j]
            src = idx[np.argmax(contrib)]
            
            # Find a good destination
            # We want a position that doesn't contribute to top lags
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.setdiff1d(np.arange(n), [src])
            
            if len(candidates) == 0:
                continue
            
            # Pick lowest-value candidate
            cands_sorted = candidates[np.argsort(a[candidates])]
            dst = cands_sorted[len(cands_sorted) // 4]
            
            if dst == src:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = min(a[i] * frac, a[i] * 0.5)
                a[i] -= amt
                a[j] += amt
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

def make_u_shaped(n, sharpness=2.0):
    """Create U-shaped initialization: high at edges, low in middle."""

    profile = (x - 0.5)**2 * sharpness + 0.1
    return profile / profile.sum()

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    result = result / result.sum()
    return result
