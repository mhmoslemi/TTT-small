def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz_vec(y):
    """
    For the LP: we need T @ b <= M where T is (2n-1) x n
    T[i, j] = y[i - j] if 0 <= i-j < n, else 0
    i.e., T[i,j] = y[i-j] for max(0, i-n+1) <= j <= min(i, n-1)
    """
    n = len(y)
    # We won't build the full matrix (too large for n=1600: 3199*1600 = 5M entries)
    # Instead, we'll use a sparse representation or direct convolution for the LP
    return n


def solve_lp_best_response_hi(y, warm_start=None):
    """
    Solve LP: min M s.t. conv(y,b)(t) <= M for all t, sum(b)=1, b>=0
    Using scipy HiGHS.
    
    Variables: x = [b_0, ..., b_{n-1}, M] (length n+1)
    c = [0,...,0, 1]
    A_ub: (2n-1) x (n+1) matrix where A_ub[i, :n] gives the i-th row of T, A_ub[i, n] = -1
    b_ub = ones(2n-1)
    A_eq: 1 x (n+1), A_eq[0, :n] = 1, A_eq[0, n] = 0
    b_eq = [1]
    bounds: b_i in [0, inf], M in [0, inf]
    """
    n = len(y)
    # Build A_ub efficiently
    # A_ub[i, j] = y[i-j] if 0 <= i-j < n, else 0
    # A_ub[i, n] = -1
    
    # For memory efficiency with large n, we'll build the matrix in chunks
    # But for n=1600, the matrix is 3199 x 1601 = ~5.1M floats = ~41 MB
    # That's manageable.
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, n] = -1.0
    
    for j in range(n):
        # Column j of T: T[i, j] = y[i-j] for i = j, j+1, ..., j+n-1
        A_ub[j:j+n, j] = y
    
    c = np.zeros(n+1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_ub = np.ones(2*n-1)
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    x0 = None
    if warm_start is not None:
        x0 = np.append(warm_start, 1.0/n)  # initial M guess
    
    result = linprog(
        c, 
        A_ub=A_ub, b_ub=b_ub,
        A_eq=A_eq, b_eq=b_eq,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'dual_feasibility_tolerance': 1e-8, 'primal_feasibility_tolerance': 1e-8}
    )
    
    if result.status != 0:
        return None, None
    
    b_val = result.x[:n]
    M_val = result.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, max_iters=200000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    Uses HiGHS solver.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    warm_b = None
    stall_count = 0
    last_print = 0
    
    # Exponential moving average tracker
    ema_a = a.copy()
    ema_alpha = 0.01

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b_star, M_val = solve_lp_best_response_hi(y, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a

        # Check scores
        if t % 10 == 0 or t < 100:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0

            if t % 100 == 0:
                score_ema = 2.0 * n * fast_autocorr(ema_a).max()
                if score_ema < best_score:
                    best_score = score_ema
                    best_a = ema_a.copy()
                    stall_count = 0

        # Adaptive restart
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            ema_a = best_a.copy()
            stall_count = 0
            warm_b = None

        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}")
            last_print = t

    # Final check
    score_final = 2.0 * n * fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()

    return best_a, best_score


def lse_objective_and_grad(z, T_smoothing, topk):
    """
    Compute LSE objective and gradient in z-space (softmax parameterization).
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    
    k_use = min(topk, len(b))
    top_idx = np.argsort(b)[::-1][:k_use]
    top_vals = b[top_idx]
    
    max_v = top_vals.max()
    exp_v = np.exp(T_smoothing * (top_vals - max_v))
    lse_val = (max_v + np.log(exp_v.sum()) / T_smoothing)
    w = exp_v / exp_v.sum()
    
    # Gradient of b(t) w.r.t. a: grad_b_t[j] = 2*a[t-j] (with bounds)
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z


def lbfgs_lse_polish(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=10, topk=20):
    """L-BFGS on LSE smoothed objective with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        z = np.log(np.maximum(best_a, 1e-30))
        
        def obj_grad(z):
            obj, grad = lse_objective_and_grad(z, stage_T, topk)
            return obj, grad
        
        try:
            res = minimize(
                obj_grad, z, jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'maxfun': 2000, 'gtol': 1e-8}
            )
            if res.success or res.fun < 1e10:
                a_new = softmax_fn(res.x)
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
        except Exception:
            pass
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-BFGS stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Systematic peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(10, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:12]:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(6 * np.pi * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.7 * np.exp(-6 * (x - 0.7) ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.8 + 0.2 * np.cos(4 * np.pi * x))
    elif choice == 6:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 2) ** 0.5
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a * (1 + noise)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn

Wait, I need to double-check a few things:

1. The `linprog` call: I need to make sure the constraints are set up correctly.
   - We want: min M subject to T@b <= M, sum(b) = 1, b >= 0
   - In scipy format: min c^T x s.t. A_ub @ x <= b_ub, A_eq @ x == b_eq, bounds
   - x = [b_0, ..., b_{n-1}, M]
   - c = [0, ..., 0, 1]
   - A_ub @ x <= b_ub means: T@b - M <= 0, so A_ub = [T, -1] (where -1 is a column of -1s)
   - A_eq @ x == b_eq means: sum(b) == 1, so A_eq = [1, 1, ..., 1, 0], b_eq = [1]
   - bounds: b_i in [0, inf], M in [0, inf] (M >= 0 since all entries of T are non-negative when y >= 0)

   This looks correct.

2. Memory: For n=1600, A_ub is 3199 x 1601 ≈ 5.1M entries × 8 bytes ≈ 41 MB. That should be fine.

3. The LP solve time: With HiGHS, an LP with 1601 variables and 3199+1 constraints should take about 10-30ms. In 800 seconds, that's about 27000-80000 iterations. Good.

4. One concern: building A_ub each iteration is O(n^2) which is 2.56M operations for n=1600. At ~1ns per operation, that's ~2.5ms. Combined with the LP solve, each iteration is maybe 15-35ms total. In 800s, that's 23000-53000 iterations. Should be enough.
