        s = b_val.sum()
        if s > 1e-15:

        return b_val
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP zero-sum game with HiGHS solver and pre-allocated matrices
2. Exact-subgradient Top-k LogSumExp Adam optimization with multi-stage annealing

4. Systematic multi-length exploration with interpolation
5. Iterative scaling refinement for final polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Score = 2*n*max(conv(a,a))/(sum(a))^2, scale invariant."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_toy(y, A_ub_buf, c, b_ub, A_eq, b_eq, bounds):
    """Solve best-response LP using pre-allocated buffers. Returns b or None."""
    from scipy.optimize import linprog
    n = len(y)
    # Fill the Toeplitz part of A_ub
    A_ub2 = A_ub_buf
    for j in range(n):
        end = min(A_ub2.shape[0], j + n)
        A_ub2[j:end, j] = y[:end - j]
    res = linprog(c, A_ub=A_ub2, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b_val = np.maximum(res.x[:n], 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val /= s
        return b_val
    return None


def nesterov_lp_game(a0, time_limit, max_iters=100000):
    """Nesterov-accelerated LP zero-sum game with pre-allocated matrices."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Pre-allocate LP matrices
    A_ub_buf = np.zeros((2 * n - 1, n + 1))
    A_ub_buf[:, n] = -1.0
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    # Tracking averages
    avg_all = np.zeros(n)
    n_all = 0
    WINDOW = 80
    window_buf = [None] * WINDOW
    window_sum = np.zeros(n)
    window_idx = 0
    window_count = 0

    stall = 0
    last_eval = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star = solve_lp_toy(y, A_ub_buf, c_vec, b_ub, A_eq, b_eq, bounds)
        if b_star is None:
            continue

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_star
        y = normalize(y)

        if t <= 2:
            a = y.copy()
        else:
            theta = (t - 1.0) / (t + 2.0)
            a = y + theta * (y - y_old)
            a = normalize(a)

        # Update averages
        avg_all += a
        n_all += 1
        if window_buf[window_idx] is None:
            window_buf[window_idx] = a.copy()
            window_count += 1
        else:
            window_sum -= window_buf[window_idx]
        window_buf[window_idx] = a.copy()
        window_sum += a
        window_idx = (window_idx + 1) % WINDOW

        # Periodic evaluation
        if t - last_eval >= 10:
            last_eval = t
            sc_a = 2.0 * n * fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if n_all >= 50:
                avg_a = avg_all / n_all
                sc_avg = 2.0 * n * fast_autocorr(avg_a).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_a.copy()
                    stall = 0

            if window_count >= WINDOW and t % 20 == 0:
                win_avg = window_sum / WINDOW
                sc_win = 2.0 * n * fast_autocorr(win_avg).max()
                if sc_win < best_score:
                    best_score = sc_win
                    best_a = win_avg.copy()
                    stall = 0

            # Restart if stalled
            if stall > 300 and t > 200:
                a = best_a.copy()
                y = best_a.copy()
                avg_all = np.zeros(n)
                n_all = 0
                window_buf = [None] * WINDOW
                window_sum = np.zeros(n)
                window_idx = 0
                window_count = 0
                stall = 0

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLG t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final evaluation of all tracked solutions
    sc_a = 2.0 * n * fast_autocorr(a).max()
    if sc_a < best_score:
        best_score = sc_a
        best_a = a.copy()
    if n_all >= 20:
        sc_avg = 2.0 * n * fast_autocorr(avg_all / n_all).max()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = (avg_all / n_all).copy()

    return best_a, best_score


def topk_lse_obj_grad(a, T_temp, topk):
    """Compute top-k LSE objective and gradient w.r.t. a (normalized, sum=1)."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx] / T_temp

    max_val = top_vals.max()
    exp_vals = np.exp(top_vals - max_val)
    lse = max_val + np.log(exp_vals.sum())
    w = exp_vals / exp_vals.sum()

    # Gradient of LSE w.r.t. a
    grad = np.zeros(n)
    for i in range(k_use):
        K = top_idx[i]
        wi = w[i] / T_temp
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wi * a[K - idx]

    return lse, grad


def adam_lse_optimize(a0, time_limit, T_start=80.0, T_end=0.1, n_stages=40, topk=30):
    """Adam-optimized top-k LSE with multi-stage temperature annealing."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 5e-3 * min(1.0, 200.0 / n)
        lr = base_lr * (0.92 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(10, int(min(remaining, 30.0) * 8))
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break

            a_curr = softmax_fn(z)
            obj, grad_a = topk_lse_obj_grad(a_curr, stage_T, topk)
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a_check = softmax_fn(z)
                sc = 2.0 * n * fast_autocorr(a_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    if (it + 1) % 40 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        sc_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if sc_stage < best_score:
            best_score = sc_stage
            best_a = a_stage.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):

    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.98
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Compute importance: contribution to peaks
        importance = np.zeros(n)
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[idx] * a[k - idx]

        improved = False

        # Try single moves: reduce important, add to unimportant
        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (low importance)
            safe = np.where(importance < importance.max() * 0.3)[0]
            safe = safe[safe != src]
            if len(safe) == 0:

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]

                        stall = 0
                        found = True
                        break

            if found:

                        continue
                    for d1 in dsts:

                            if d2 <= d1:
                                continue
                            if d1 == s1 or d1 == s2 or d2 == s1 or d2 == s2:
                                continue
                            for frac in [0.01, 0.03]:
                                amt1 = a[s1] * frac
                                amt2 = a[s2] * frac
                                if amt1 < 1e-15 or amt2 < 1e-15:
                                    continue
                                a_trial = a.copy()
                                a_trial[s1] -= amt1
                                a_trial[s2] -= amt2
                                a_trial[d1] += amt1
                                a_trial[d2] += amt2
                                a_trial = np.maximum(a_trial, 0)
                                st = a_trial.sum()
                                if st < 1e-12:
                                    continue
                                a_trial /= st
                                new_max = fast_autocorr(a_trial).max()
                                if new_max < bmax * (1 - 1e-9):
                                    a = a_trial
                                    new_score = 2.0 * n * new_max
                                    if new_score < best_score:
                                        best_score = new_score
                                        best_a = a.copy()
                                    improved = True
                                    stall = 0
                                    break
                            if improved:
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 6)

                    j = rng.integers(n)

                        frac = rng.uniform(0.002, 0.04)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial /= st
                    sc = 2.0 * n * fast_autocorr(a_trial).max()

                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 300:

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_iters=None):
    """Exact subgradient method for the max autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    if n_iters is None:
        n_iters = 5000

    for t in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        bmax = b.max()
        # Find all positions at or near max
        near_max = np.where(b >= bmax * 0.999)[0]

        # Subgradient of max(b) w.r.t. a
        grad = np.zeros(n)
        for k in near_max:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step size: use known bounds
        # Lower bound: 2/n (from uniform)
        lb = 2.0 / n
        gap = best_score - 2.0 * n * lb
        if gap < 1e-10:
            gap = 1e-10
        eta = gap / (t + 1) / (gnorm ** 2)
        eta = min(eta, 0.1)  # Cap step size

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc_new = 2.0 * n * fast_autocorr(a_new).max()
        if sc_new < best_score:
            a = a_new
            best_score = sc_new
            best_a = a_new.copy()

    return best_a, best_score

    t0 = time.monotonic()

    cur_eta = eta

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        importance = np.zeros(n)
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[idx] * a[k - idx]

        imax = importance.max()
        if imax < 1e-15:
            break
        importance = importance / imax

        updates = 1.0 - cur_eta * importance
        updates = np.maximum(updates, 0.005)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc_new = 2.0 * n * fast_autocorr(a_new).max()
        if sc_new < best_score:
            a = a_new
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                cur_eta *= 0.5
                if cur_eta < 1e-7:
                    cur_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn2 = a.sum()
                    if sn2 > 1e-10:
                        a = a / sn2
            if stall > 400:
                break

    return best_a, best_score
