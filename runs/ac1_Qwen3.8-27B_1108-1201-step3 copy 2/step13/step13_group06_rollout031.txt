from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def evaluate_safe(a):

        return float(sc)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgrad_conv(a, peaks, weights=None):
    """
    Compute subgradient of max_{k in peaks} conv(a,a)[k] at a.
    If weights given, compute weighted combination.
    """
    n = len(a)
    g = np.zeros(n)
    if weights is None:
        weights = np.ones(len(peaks)) / len(peaks)
    for idx, k in enumerate(peaks):
        w = weights[idx]
        if w < 1e-15:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        for j in range(js, je + 1):

            term1 = a[k - j] if (0 <= k - j <= n - 1) else 0.0
            term2 = a[j - k] if (0 <= j - k <= n - 1) else 0.0
            g[j] += w * (term1 + term2)

def projected_subgrad(a0, time_limit, T_init=1.0, T_min=0.01, n_peaks=10, seed=42):
    """
    Projected subgradient method with:
    - Softmax-weighted subgradient over top peaks
    - Adaptive step size
    - Running average
    - Periodic annealing
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    a_avg = a.copy()
    n_avg = 1
    
    # Parameters
    alpha = 1.0  # step size scale
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        f_val = bmax  # f(a) = max_k conv(a,a)[k]
        
        # Find top peaks with softmax weighting
        # Temperature-based weighting
        T = max(T_min, T_init * (0.999 ** total_iters))
        # Use top n_peaks
        top_idx = np.argsort(b)[::-1][:n_peaks]
        top_vals = b[top_idx]
        # Softmax weights
        logits = (top_vals - bmax) / T  # negative or zero
        exp_logits = np.exp(logits)
        weights = exp_logits / exp_logits.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            k = top_idx[idx]
            w = weights[idx]
            if w < 1e-15:
                continue
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            for j in range(js, je + 1):
                t1 = a[k - j] if (0 <= k - j <= n - 1) else 0.0
                t2 = a[j - k] if (0 <= j - k <= n - 1) else 0.0
                g[j] += w * (t1 + t2)
        
        # Adaptive step size
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            eta = 0.0
        else:
            eta = alpha * f_val / (g_norm_sq + 1e-10)
            eta = min(eta, 0.5)  # cap step size
        
        # Subgradient step
        a_new = a - eta * g
        a_new = project_simplex(a_new)
        
        # Update
        a = a_new
        total_iters += 1
        
        # Update running average
        a_avg = a_avg + (a - a_avg) / n_avg
        n_avg += 1
        
        # Check score periodically
        if total_iters % 10 == 0 or total_iters < 20:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Check running average
        if total_iters % 50 == 0:
            score_avg = 2.0 * n * fast_autocorr(a_avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()
        
        # Periodic restart from average
        if total_iters % 500 == 0:
            a = best_a.copy()
            a_avg = a.copy()
            n_avg = 1
    
    return best_a, best_score

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad(a, peaks, weights):
    """
    Compute weighted subgradient of max_{k in peaks} conv(a,a)[k] at a.
    g[i] = 2 * sum_k w_k * a[k-i] for valid (k,i).
    """
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peaks)):
        k = peaks[idx]
        w = weights[idx]
        if w < 1e-15:
            continue
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        j_idx = np.arange(js, je + 1)
        # g[j] += 2 * w * a[k-j]
        g[j_idx] += 2.0 * w * a[k - j_idx]
    return g


def projected_subgrad_method(a0, time_limit, T_init=0.5, T_min=0.01,
                              n_peaks=12, alpha_init=1.0, seed=42):
    """
    Projected subgradient with momentum, adaptive step size, and running average.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Momentum state
    y = a.copy()  # virtual position
    momentum = 0.0

    # Running average
    a_avg = a.copy()
    n_avg = 1

    # Parameters
    alpha = alpha_init
    total_iters = 0
    check_interval = max(5, n // 100)

    while time.monotonic() - t0 < time_limit - 1:
        b = fast_autocorr(y)
        bmax = b.max()
        f_val = bmax

        # Adaptive temperature
        T = max(T_min, T_init * (0.9995 ** total_iters))

        # Find top peaks
        top_idx = np.argsort(b)[::-1][:n_peaks]
        top_vals = b[top_idx]

        # Softmax weights
        logits = (top_vals - bmax) / T
        exp_logits = np.exp(logits)
        w_sum = exp_logits.sum()
        if w_sum < 1e-300:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = exp_logits / w_sum

        # Compute subgradient at y
        g = compute_subgrad(y, top_idx, weights)

        # Adaptive step size
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            eta = 0.0
        else:
            eta = alpha * f_val / (g_norm_sq + 1e-10)
            eta = min(eta, 0.3)

        # Nesterov momentum update
        momentum = (total_iters + 1.0) / (total_iters + 1.0 + 1.5)
        y_new = project_simplex(y - eta * g)
        a_new = y_new + momentum * (y_new - y)
        a_new = project_simplex(a_new)

        y = y_new
        a = a_new
        total_iters += 1

        # Update running average
        a_avg = a_avg + (a - a_avg) / n_avg
        n_avg += 1

        # Check score
        if total_iters % check_interval == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_avg = 2.0 * n * fast_autocorr(a_avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()

        # Periodic: refresh from best
        if total_iters % 2000 == 0:
            a = best_a.copy()
            y = a.copy()
            a_avg = a.copy()
            n_avg = 1
            alpha = max(0.1, alpha * 0.5)

    # Final check
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    score_avg = 2.0 * n * fast_autocorr(a_avg).max()
    if score_avg < best_score:
        best_score = score_avg
        best_a = a_avg.copy()

    return best_a, best_score


def fourier_flatten_method(a0, time_limit, clip_factor=0.8, n_iters_target=5000, seed=42):
    """
    Gerchberg-Saxton-like iterative method in Fourier domain.
    Alternates between:
    1. Projecting |A|^2 towards flatness (in frequency domain)
    2. Projecting a to be non-negative (in time domain)
    """

    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length() - 1
    # Actually use next power of 2
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    iter_count = 0
    while time.monotonic() - t0 < time_limit - 1:
        # Forward FFT
        A = fft(a, fft_size)

        # Power spectrum
        power = np.abs(A) ** 2

        # Flattening: reduce peaks in power spectrum
        # Target: make power as uniform as possible
        # Use soft clipping
        p_mean = power.mean()
        # Scale down peaks, boost valleys slightly
        # Method: power_new = power * (p_mean/power + epsilon)^(beta * (1 - power/p_mean))
        # Simpler: iterative water-filling on power
        ratio = power / (p_mean + 1e-30)
        # Clip high values, boost low values
        beta = 0.3  # aggressiveness
        # Soft clip: new_power = power + beta * (p_mean - power) * 0.5
        # This is a relaxation towards uniform
        new_power = power * (1 - beta * 0.5) + p_mean * (beta * 0.5)
        # But also do hard clipping on the worst peaks
        p_max = new_power.max()
        p_new = new_power * clip_factor + p_mean * (1 - clip_factor)

        # Ensure non-negativity and normalize
        p_new = np.maximum(p_new, 1e-20)

        # Update amplitude, keep phase
        new_amplitude = np.sqrt(p_new)
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_amplitude * phase

        # Inverse FFT
        a_new = np.real(ifft(A_new, fft_size))[:n]

        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        a = a_new
        iter_count += 1

        # Check score periodically
        if iter_count % 20 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        # Adapt clip_factor: start aggressive, become conservative
        if iter_count % 500 == 0:
            clip_factor = max(0.5, clip_factor * 0.95)

    return best_a, best_score
