def lse_gradient(a, T, top_k=None):
    """

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        js = max(0, j - n + 1)
        je = min(n - 1, j)
        for k in range(js, je + 1):
            T_mat[j, k] = a[j - k]
    T_mat = T_mat.tocsr()
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    mu = 100.0  # penalty for sum != 1

    def objective(x):
        if time.monotonic() - t0 > time_limit:
            return float('inf')
        b = fast_autocorr(x)
        mb = b.max()
        lse = T * (mb / T + np.log(np.sum(np.exp((b - mb) / T))))
        pen = 0.5 * mu * (x.sum() - 1.0) ** 2
        return lse + pen

    def gradient(x):
        grad = lse_gradient(x, T) / T  # d(lse)/dx = w * (corr terms)
        grad += mu * (x.sum() - 1.0)
        return grad

    res = scipy_minimize(objective, a, method='L-BFGS-B', jac=gradient,
                         bounds=[(0, None)] * n,
                         options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 30})
    if res.success or res.fun < 1e10:
        a_out = normalize_seq(res.x, 1.0)
        sc = fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS-B ({elapsed:.0f}s): score={sc:.8f}, best={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate():
    """
    Main search: multi-phase optimization starting from parent sequence.
    Uses adaptive LSE-Nesterov, LP best-response, simulated annealing local
    search, L-BFGS-B, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1720.0
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: Adaptive LSE-Nesterov (600s) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adaptive LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_adaptive(a_work, t1, T_start=None, T_end=0.15, n_stages=16, verbose=True)
        try_update(a_out, f"nesterov_n{n_cur}")

    # ============ Phase 2: LP Best-Response (350s) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Best-Response n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # ============ Phase 3: L-BFGS-B refinement (150s) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: L-BFGS-B n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.09, 150.0)
    if t3 > 15:
        n_cur = len(best_seq)
        a_work = best_seq.copy()  # already sum=1
        b_init = fast_autocorr(a_work)
        T_lfkgb = max(0.5, (b_init.max() - b_init.min()) * 0.5)
        a_out, _ = lse_lfkgb(a_work, t3, T=T_lfkgb, verbose=True)
        try_update(a_out, "lfkgb")

    # ============ Phase 4: Simulated Annealing Local Search (300s) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LS-Annealing n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 15:
        n_cur = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_search_annealing(a_work, t4, seed=7777, verbose=True)
        try_update(a_out, f"lsa_n{n_cur}")

    # ============ Phase 5: Try longer length (250s) ============
    n2 = n_base + 300
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE-Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse_adaptive(a_work, t5, T_start=None, T_end=0.2, n_stages=12, verbose=True)
        try_update(a_out, f"nesterov_n{n2}")

    # ============ Phase 6: Shorter length (150s) ============
    n3 = max(800, n_base - 200)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 6: LSE-Nesterov n={n3} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.09, 150.0)
        if t6 > 15:
            init_short = interpolate_sequence(best_seq, n3)
            a_work = init_short * np.sqrt(2.0 * n3) / init_short.sum()
            a_out, _ = nesterov_lse_adaptive(a_work, t6, T_start=None, T_end=0.3, n_stages=10, verbose=True)
            try_update(a_out, f"nesterov_n{n3}")

    # ============ Phase 7: Final polish - Nesterov + Local Search (remaining) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7a = min(remaining * 0.5, 200.0)
        t7b = remaining - t7a
        print(f"\n=== Phase 7a: Final Nesterov n={n_best} (budget={t7a:.0f}s) ===")
        if t7a > 20:
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lse_adaptive(a_work, t7a, T_start=2.0, T_end=0.05, n_stages=8, verbose=True)
            try_update(a_out, "nesterov_polish")

        n_best = len(best_seq)
        print(f"\n=== Phase 7b: Final Local Search n={n_best} (budget={t7b:.0f}s) ===")
        if t7b > 10:
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = local_search_annealing(a_work, t7b, seed=12345, verbose=True)
            try_update(a_out, "ls_polish")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Returning default.")
        return [1.0] / [1.0]

    return [float(x) for x in best_seq]
