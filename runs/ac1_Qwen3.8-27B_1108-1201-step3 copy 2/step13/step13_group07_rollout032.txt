def internal_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x>=0, sum(x)=target_sum} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v.copy()
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.maximum(v - theta, 0.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        positive_mask = result > 0
        if positive_mask.any():
            result[positive_mask] += diff / positive_mask.sum()
            result = np.maximum(result, 0.0)
        else:
            result = np.ones(n) / n * target_sum
    ss = result.sum()
    if ss > 0:
        result = result / ss * target_sum
    return result


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = target / (non_dc + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

            continue
    return cp.ECOS


def solve_lp_best_response(b_var, M_var, T_sparse, solver, max_iter=300):
    """Solve: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns b, M."""
    n = b_var.shape[0]
    constraints = [T_sparse @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iter)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=max_iter)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64).flatten()
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, rows, cols, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 3000
    last_best_t = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        T_curr = make_conv_matrix(rows, cols, y, n)
        b_star, M_val = solve_lp_best_response(b_var, M_var, T_curr, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_best_t = t

        score_y = internal_score(y)
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_best_t = t

        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score - 1e-10:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
                last_best_t = t

        stall_count += 1
        if stall_count > stall_threshold:
            if verbose:
                print(f"      LP stalled at t={t}, best={best_score:.6f}")
            break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def compute_lse_gradient(a, T_smooth):
    """Compute gradient of (1/T)*log(sum exp(T*conv(a,a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_smooth * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    ar = a[::-1]
    h = np.convolve(ar, w, mode='full')
    grad = 2.0 * h[n - 1:n - 1 + n]
    return grad, w


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=20, lr=0.005):
    """LSE-smoothed optimization with Adam on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        iters = max(20, min(300, int(remaining * 40 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad, w = compute_lse_gradient(a, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / (1.0 + 0.005 * it)
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new)

            if it % 10 == 0:
                score = internal_score(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
            a = a_new

        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, verbose=True):
    """Subgradient method with Nesterov momentum and LSE smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta_mom = 0.95
    base_lr = 0.05
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(y)
        T_smooth = max(1.0, 100.0 * (1.0 - t / max(1, int(time_limit * 30))))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        ar = y[::-1]
        h = np.convolve(ar, w, mode='full')
        grad = 2.0 * h[n - 1:n - 1 + n]

        lr_t = base_lr / (1.0 + 0.002 * t)
        v = beta_mom * v + (1.0 - beta_mom) * grad
        a_new = y - lr_t * v
        a_new = project_simplex(a_new)

        alpha_t = 2.0 / (t + 2.0)
        y = (1.0 - alpha_t) * y + alpha_t * a_new
        y = np.maximum(y, 0)
        s = y.sum()
        if s > 0:
            y = y / s

        score_a = internal_score(a_new)
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a_new.copy()
            stall = 0

        score_y = internal_score(y)
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        stall += 1
        if stall > 5000:
            break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving for final refinement."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue

            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[int(rng.integers(len(dst_pool)))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        if not a_list:
            return float('inf')
        if sum(a_list) < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Load parent / spectral init
    2. Nesterov LP (primary, n=1600)
    3. LSE Adam polish
    4. Subgradient Nesterov
    5. Try longer n=2000
    6. Final LP round
    7. Peak shaving
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, internal_score={best_score:.6f}")
            # Verify with actual evaluator
            actual = evaluate_safe(best_seq)
            if np.isfinite(actual):
                print(f"Parent actual score: {actual:.6f}")
                best_score = actual
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = internal_score(best_seq)
        print(f"Default init (uniform): n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Precompute convolution indices for base length
    rows_base, cols_base = precompute_conv_indices(n_base)
    print(f"Conv matrix indices precomputed: {len(rows_base)} non-zeros")

    # Phase 1: Spectral shaping (fast, multiple alphas)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.2, 0.1]:
            if time.monotonic() - t_global_start > time_budget - t1:
                break
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 700.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver, rows_base, cols_base, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 3: LSE Adam polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE Adam n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_out = lse_optimize(best_seq, t3, T_start=50.0, T_end=0.05, n_stages=15, lr=0.004)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LSE_Adam]: {best_score:.6f}")

        # Second round with lower temperature
        t3b = min(remaining * 0.05, 80.0)
        if t3b > 5 and time.monotonic() - t_global_start < time_budget:
            a_out = lse_optimize(best_seq, t3b, T_start=10.0, T_end=0.02, n_stages=10, lr=0.002)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [LSE_low]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 4: Subgradient Nesterov (different exploration direction)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Subgradient Nesterov n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_out, _ = subgradient_nesterov(best_seq, t4, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 4: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 5: Try longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 60:
        rows_long, cols_long = precompute_conv_indices(n_long)
        init_long = interpolate_sequence(best_seq, n_long)

        # Quick spectral at longer length
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.25)
        sc = evaluate_safe(a_spec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_spec, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [spectral_n{n_long}]: {best_score:.6f}")

        # LP at longer length
        t5b = t5 * 0.70
        if time.monotonic() - t_global_start < time_budget - t5b:
            init_lp = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_lp, t5b, solver, rows_long, cols_long, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [LP_n{n_long}]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 6: Second LP round at best length found
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: More LP n={n_best} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 30:
        if n_best == n_base:
            rows_cur, cols_cur = rows_base, cols_base
        else:
            rows_cur, cols_cur = precompute_conv_indices(n_best)
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, rows_cur, cols_cur, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_round2]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 100.0)
    if t7 > 5:
        a_out, _ = peak_shaving(best_seq, t7, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f}")

    # Final verification and cleanup
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
