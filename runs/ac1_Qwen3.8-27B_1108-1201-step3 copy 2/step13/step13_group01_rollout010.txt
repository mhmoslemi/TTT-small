"""

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)

def get_solver():
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lse_objective_and_grad(a, T):
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 0.0, np.zeros(n)
    shifted = T * (b - max_b)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    cross2 = irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    dL_da = 2.0 * (cross + cross2)
    obj = 2.0 * n * lse / (s * s)
    grad = 2.0 * n * dL_da / (s * s) - 4.0 * n * lse * a / (s * s * s)
    return obj, grad

def lse_optimize(a0, time_limit, T_stages=None):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = score_fast(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0]

    for si, T in enumerate(T_stages):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        stage_budget = min((time_limit - elapsed) / max(1, len(T_stages) - si), 180.0)
        if stage_budget < 10:
            break

        x0 = a.copy().astype(np.float64)

        def obj_fn(x):
            o, _ = lse_objective_and_grad(x, T)
            return o

        def grad_fn(x):
            _, g = lse_objective_and_grad(x, T)
            return g

        try:
            res = scipy_minimize(obj_fn, x0, method='L-BFGS-B', jac=grad_fn,
                                 bounds=[(0.0, 1.0)] * n,
                                 options={'maxiter': 300, 'maxfun': 1000,
                                          'ftol': 1e-12, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate / s_c
                    sc = score_fast(a_candidate)
                    if sc < best_obj:
                        best_obj = sc
                        best_a = a_candidate.copy()
                    a = a_candidate
        except Exception:
            pass

    return best_a, best_obj

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T_mat[idx, i] = a[idx - i]
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_val

def nesterov_lp(a0, time_limit, solver, max_iters=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0)

    else:
        a = a / s0

    best_a = a.copy()
    best_score = score_fast(a)
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    t = 0
    last_improve = t0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b, dual = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        alpha = 2.0 / (t + 2.0)
        ratio = M_curr / max(best_score, 1e-15)
        alpha = min(alpha * min(ratio + 0.5, 2.0), 1.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        y = y_new
        a = a_new
        sc = score_fast(y)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
            last_improve = time.monotonic()
        sc2 = score_fast(a)
        if sc2 < best_score:
            best_score = sc2
            best_a = a.copy()
            last_improve = time.monotonic()
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:

                    last_improve = time.monotonic()
        if time.monotonic() - last_improve > 60.0:

        else:

def gradient_flow_optimize(a0, time_limit, K=10, n_restarts=0):

    lr = 0.01
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k_indices = np.argsort(b)[::-1][:K]
        T_adapt = 5.0 + 10.0 * (1.0 - min(1.0, (time.monotonic() - t0) / time_limit))
        shifted = T_adapt * (b - b.max())
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        fft_size = 1 << (3 * n - 2).bit_length()
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        w_padded = np.zeros(fft_size)
        w_padded[:2 * n - 1] = w
        A_fft = rfft(a_padded, fft_size)
        W_fft = rfft(w_padded, fft_size)

        cross2 = irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
        grad = 2.0 * (cross1 + cross2)
        s = a.sum()
        grad = grad / (s * s) - 2.0 * a / s
        grad = grad - grad.mean()
        velocity = beta * velocity - lr * grad
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        sc = score_fast(a_new)
        if sc < best_score:

        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.7
                if lr < 1e-5:
                    if no_improve > 500:
                        break
                velocity *= 0.5
        if time.monotonic() - t0 > time_limit:
            break
    return best_a, best_score

def peak_splitting(a0, time_limit):
    """Iteratively identify peaks and redistribute mass to reduce them."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_max = int(np.argmax(b))
        b_max = b[k_max]
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        top_src = idx[np.argsort(contrib)[::-1][:3]]
        improved = False
        for src in top_src:
            if a[src] < 1e-12:
                continue
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in np.argsort(b)[::-1][:min(20, 2 * n - 1)]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            for dst in candidates[:5]:
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    sc = score_fast(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            rng_step = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for _ in range(50):
                i = rng_step.integers(n)
                j = rng_step.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                amt = a[i] * rng_step.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                sc = score_fast(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    break
            else:
                break
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 / (1.0 + np.exp(-10.0 * (x - 0.5)))
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multi_start_lp(a0, time_limit, solver, n_starts=3):

            a_out, sc_out = nesterov_lp(init_seq, per_scan, solv)
        else:
            a_out, sc_out = gradient_flow_optimize(init_seq, per_scan)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        update_best(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LSE optimization at base length
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 1: Deep LSE n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        a_out, _ = lse_optimize(best_seq, t1)
        update_best(a_out, f"lse_deep_n{n_base}")

    # Phase 2: LP-Nesterov at best length
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-Nesterov n={n_best} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30 and solv is not None:
        a_out, _ = nesterov_lp(best_seq, t2, solv)
        update_best(a_out, f"lp_deep_n{n_best}")
    elif t2 > 30:
        a_out, _ = gradient_flow_optimize(best_seq, t2)
        update_best(a_out, f"gf_deep_n{n_best}")

    # Phase 3: Try longer sequence
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_long = n_best + 400
    print(f"\n=== Phase 3: Longer seq n={n_long} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        if solv is not None:
            a_out, _ = nesterov_lp(init_long, t3, solv)
        else:
            a_out, _ = lse_optimize(init_long, t3)
        update_best(a_out, f"long_n{n_long}")

    # Phase 4: Gradient flow (different trajectory)
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Gradient flow n={n_best} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = gradient_flow_optimize(best_seq, t4, K=15)
        update_best(a_out, f"gf_n{n_best}")

    # Phase 5: Peak splitting / coordinate descent
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak splitting n={n_best} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 15:
        a_out, _ = peak_splitting(best_seq, t5)
        update_best(a_out, f"peak_split_n{n_best}")

    # Phase 6: LSE polish with low temperature
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 15:
        T_low = [4.0, 3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6]
        a_out, _ = lse_optimize(best_seq, t6, T_stages=T_low)
        update_best(a_out, f"lse_polish_n{n_best}")

    # Phase 7: Final peak splitting
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final peak split n={n_best} (remaining={remaining:.0f}s) ===")
    t7 = max(0.0, remaining - 30.0)
    if t7 > 10:
        a_out, _ = peak_splitting(best_seq, t7)
        update_best(a_out, f"final_split_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
