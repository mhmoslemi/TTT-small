    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        rho = n - 1
    else:
        rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 1e-30:
            result = result / ss * target_sum
        else:
            result = np.full(n, target_sum / n)
    return result


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    """Find a working LP solver, prefer CLARABEL for speed."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, T, solver, verbose=False):
    """Solve: min M s.t. T @ b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=500)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=1000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix where T[k,j] = a_fixed[k-j] if valid."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a_fixed
    return T


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with adaptive schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 30000
    last_improve_t = 0

    # Adaptive Nesterov parameter
    base_alpha_mult = 2.0
    alpha_adapt = 2.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build T and solve LP
        T = build_conv_matrix(y)
        b_star, M_val = solve_lp_best_response(y, T, solver)
        if b_star is None:
            continue

        # Adaptive Nesterov step size
        # Use standard 2/(t+2) but adapt based on progress
        alpha_t = alpha_adapt / (t + alpha_adapt)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + alpha_adapt)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Check scores for both iterates
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
                last_improve_t = t
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall_count = 0
                last_improve_t = t
            best_score = score_y
            best_a = y.copy()

        # Time-averaged iterate (use running average, reset periodically)
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
                last_improve_t = t
        if n_avg >= 5000:
            # Reset average to recent window
            avg_a = avg_a * 0.5
            n_avg = n_avg // 2

        # Adapt alpha based on convergence
        if t > 100 and t % 100 == 0:
            if t - last_improve_t > 5000:
                alpha_adapt = max(1.0, alpha_adapt * 0.9)
            elif t - last_improve_t < 100:
                alpha_adapt = min(4.0, alpha_adapt * 1.1)

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}, alpha={alpha_adapt:.2f}")

    return best_a, best_score


def kkt_refine(a0, time_limit, n_peaks=15):
    """
    KKT-based active set refinement.
    At optimum: for active peaks A and support S of a:
      sum_{k in A} lambda_k * 2*a[k-m] = mu for all m in S
    We iteratively identify active set and solve for optimal a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M = b.max()
        # Active set: peaks within 1% of max
        threshold = M * 0.995
        active_mask = b >= threshold
        active_indices = np.where(active_mask)[0]

        if len(active_indices) == 0:
            break

        # Compute gradient contributions from active peaks
        # grad[m] = sum_{k in active} 2*a[k-m] (where valid)
        grad = np.zeros(n)
        for k in active_indices:
            # d(result[k])/d(a[m]) = 2*a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            m_range = np.arange(lo, hi + 1)
            km_range = k - m_range
            grad[m_range] += 2.0 * a[km_range]

        # At optimum, grad should be constant on the support of a
        # The "violation" is the deviation from constancy
        support = a > 1e-10
        if support.sum() < 2:
            break

        grad_support = grad[support]
        grad_mean = grad_support.mean()
        grad_std = grad_support.std()

        if grad_std < 1e-10:
            # Already at KKT optimum
            break

        # Direction: reduce grad where it's high, increase where it's low
        # Move a in direction that equalizes grad on support
        # Normalize grad to have zero mean on support
        grad_centered = grad[support] - grad_mean
        # The update direction for a[m] is -grad_centered[m] / |grad_centered|
        direction = np.zeros(n)
        direction[support] = -grad_centered
        direction_norm = np.linalg.norm(direction)
        if direction_norm < 1e-15:
            break
        direction = direction / direction_norm

        # Try multiple step sizes
        best_local = a.copy()
        best_local_score = internal_score(a)
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a + step * direction
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = internal_score(a_trial)
            if score < best_local_score:
                best_local_score = score
                best_local = a_trial.copy()

        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local.copy()
            a = best_local
        else:
            # No improvement - try a different approach: 
            # directly reduce the contribution to the worst peak
            worst_k = np.argmax(b)
            # Reduce a at positions that contribute most to the worst peak
            contrib = np.zeros(n)
            lo = max(0, worst_k - n + 1)
            hi = min(n - 1, worst_k)
            m_range = np.arange(lo, hi + 1)
            contrib[m_range] = a[m_range] * a[worst_k - m_range]
            
            # Move mass from high-contrib to low-contrib positions
            top_contrib = np.argsort(contrib)[::-1][:max(1, n // 100)]
            all_positions = np.arange(n)
            low_contrib_pool = [p for p in all_positions if p not in set(top_contrib)]
            if low_contrib_pool:
                src = top_contrib[0]
                dst = low_contrib_pool[np.random.randint(len(low_contrib_pool))]
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    ss = a_trial.sum()
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        break

    return best_a, best_score


def peak_shaving_a2(a0, time_limit, seed=42):
    """Aggressive peak shaving with multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M = b.max()
        # Get top peaks
        top_k = np.argsort(b)[::-1][:20]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find good destination: low contribution to all top peaks
            peak_areas = set()
            for k2 in top_k[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue

            # Sort by how "safe" the destination is
            safety = np.array([a[j] for j in candidates])
            # Prefer low values (less likely to create new peaks)
            safety_indices = np.argsort(safety)[:max(1, len(candidates) // 4)]
            dst_pool = [candidates[i] for i in safety_indices]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a Trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 2000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_refine(a0, time_limit):
    """Subgradient method with adaptive smoothing for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    # Use top-K weighted subgradient
    K = 5
    base_lr = 0.05
    t = 0
    momentum = 0.0
    v = np.zeros(n)
    beta = 0.85

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Weight on top-K peaks with softmax
        top_idx = np.argsort(b)[::-1][:K]
        top_vals = b[top_idx]
        # Temperature annealing
        temp = max(0.01, 1.0 * (1.0 - t / 5000.0))
        w = np.exp(top_vals / temp)
        w = w / w.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for hi, ki in enumerate(top_idx):
            lo = max(0, ki - n + 1)
            hidx = min(n - 1, ki)
            m_range = np.arange(lo, hidx + 1)
            grad[m_range] += w[hi] * 2.0 * a[ki - m_range]

        # Adaptive step size
        lr_t = base_lr / (1.0 + 0.005 * t)

        # Momentum update
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping initialization
    Phase 2: Nesterov LP (primary, bulk of time)
    Phase 3: Subgradient refinement
    Phase 4: KKT active-set polish
    Phase 5: Multi-n exploration
    Phase 6: Peak shaving
    Phase 7: Final LP round
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 3:
        t1_start = time.monotonic()
        for alpha_val in [0.4, 0.25, 0.1]:
            if time.monotonic() - t1_start > t1 * 0.8:
                break
            a_out = spectral_shaping(best_seq, n_iter=300, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.50, 900.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.8f}")
        else:
            print(f"  LP best={sc_out:.8f} (not better than {best_score:.8f})")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 3: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 10:
        a_out, sc_out = subgradient_refine(best_seq, t3)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [SubGrad]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 4: KKT refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: KKT refine n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.05, 60.0)
    if t4 > 5:
        a_out, sc_out = kkt_refine(best_seq, t4)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [KKT]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 4: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_candidates = []
    n_cur = len(best_seq)
    for factor in [0.85, 1.15, 1.3]:
        n_trial = int(round(n_cur * factor))
        if 800 <= n_trial <= 4000 and n_trial != n_cur:
            n_candidates.append(n_trial)

    print(f"\n=== Phase 5: Multi-n exploration (remaining={remaining:.0f}s) ===")
    t5_total = min(remaining * 0.20, 250.0)
    if t5_total > 30 and n_candidates:

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        rho = n - 1
    else:
        rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()

        result[idx] += diff
