_global_best = None

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def eval_score(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    R = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * R.max() / (s * s)

def compute_smoothed_grad(a, beta):
    """
    Compute gradient of f_beta(a) = (1/beta)*log(sum_k exp(beta*R(k)))
    where R = conv(a,a).
    
    Returns (grad, R) where grad has same shape as a.
    """
    n = len(a)
    R = fast_autocorr(a)
    
    # Softmax weights (numerically stable)
    R_shifted = R - R.max()
    log_p = beta * R_shifted
    log_p_shift = log_p - log_p.max()
    p = np.exp(log_p_shift)
    p /= p.sum()
    
    # Gradient: grad[m] = 2 * sum_k p[k] * a[k-m]
    # = 2 * cross_corr(p, a)[m] where cross_corr[p,a][m] = sum_j a[j]*p[j+m]
    # Using FFT: cross_corr[m] = ifft(conj(FFT(a)) * FFT(p))[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(np.conj(A) * P, fft_size)
    grad = 2.0 * corr[:n]
    
    return grad, R

def mirror_descent_smoothed(a_init, time_limit, beta_schedule=None, verbose=True):
    """

        a = a * c / s

    best_score = 2.0 * n * best_M / c**2
    
    t0 = time.monotonic()
    gamma = 1.0  # learning rate
    
    # Track running average for better convergence
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Beta schedule
        if beta_schedule is not None:
            beta = beta_schedule(t)
        else:
            beta = 10.0 + 40.0 * min(t / 10000.0, 1.0)
        
        # Compute gradient
        grad, R = compute_smoothed_grad(a, beta)
        
        # Step size
        step = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror descent update: a_new[m] = a[m] * exp(-step * grad[m])
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        
        a = a_new
        
        # Running average
        avg_a += a
        n_avg += 1
        
        # Check best every 50 iterations
        if t % 50 == 0 or t < 10:
            M = fast_autocorr(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                best_score = 2.0 * n * best_M / c**2
            
            # Also check the average
            if n_avg >= 100:
                avg_check = avg_a / n_avg
                avg_check = np.maximum(avg_check, 0.0)
                s_avg = avg_check.sum()
                if s_avg > 1e-15:

                        best_M = M_avg
                        best_a = avg_check.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} beta={beta:.1f} J={best_score:.6f} ({elapsed:.1f}s)")
    
    return best_a, best_score

def nesterov_smoothed(a_init, time_limit, verbose=True):
    """
    Nesterov-accelerated projected gradient with smoothed subgradient.
    """
    n = len(a_init)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()

        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_score = 2.0 * n * best_M / c**2
    
    t0 = time.monotonic()
    y = a.copy()  # Nesterov lookahead
    x_prev = a.copy()
    rr = 1.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = 5.0 + 45.0 * min(t / 5000.0, 1.0)
        
        # Compute gradient at y
        grad, R = compute_smoothed_grad(y, beta)
        
        # Step size (diminishing)
        step = 1.0 / (np.sqrt(t) + 1.0)
        
        # Gradient step
        y_new = y - step * grad
        y_new = proj_simplex(y_new, c)
        
        # Nesterov update
        rr_new = (1.0 + np.sqrt(1.0 + 4.0 * rr * rr)) / 2.0
        if t == 1:
            x_new = y_new
        else:
            theta = (rr - 1.0) / rr_new
            x_new = y_new + theta * (y_new - x_prev)
            x_new = proj_simplex(x_new, c)
        
        x_prev = y_new
        y = x_new
        rr = rr_new
        
        # Check best
        if t % 20 == 0:
            M = fast_autocorr(y).max()
            if M < best_M:
                best_M = M
                best_a = y.copy()
                best_score = 2.0 * n * best_M / c**2
            
            M_x = fast_autocorr(x_new).max()
            if M_x < best_M:
                best_M = M_x
                best_a = x_new.copy()
                best_score = 2.0 * n * best_M / c**2
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS t={t} beta={beta:.1f} J={best_score:.6f} ({elapsed:.1f}s)")
    
    return best_a, best_score

def spectral_smooth(a, sigma_frac=0.02):
    """Apply Gaussian smoothing in frequency domain."""
    n = len(a)
    A = rfft(a)
    freqs = np.fft.rfftfreq(n)
    sigma = sigma_frac * n
    A_smooth = A * np.exp(-2 * np.pi**2 * sigma**2 * freqs**2)
    result = irfft(A_smooth, n)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-15:
        return a.copy()
    return result / s * a.sum()

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def stochastic_smoothed_grad(a, beta, n_samples=50):
    """
    Stochastic version: sample a subset of lags for the gradient.
    This introduces randomness to help escape local minima.
    """
    n = len(a)
    R = fast_autocorr(a)
    
    # Softmax distribution over lags
    R_shifted = R - R.max()
    log_p = beta * R_shifted
    log_p_shift = log_p - log_p.max()
    p = np.exp(log_p_shift)
    p /= p.sum()
    
    # Sample lags
    n_lags = len(R)
    n_samples = min(n_samples, n_lags)
    samples = np.random.choice(n_lags, size=n_samples, replace=False, p=p)
    
    # Compute gradient from samples
    grad = np.zeros(n)
    for k in samples:
        k = int(k)
        # dR(k)/da[m] = 2*a[k-m] for valid m
        m_start = max(0, k - n + 1)
        m_end = min(n, k + 1)
        m_idx = np.arange(m_start, m_end)
        a_idx = k - m_idx
        grad[m_idx] += 2.0 * p[k] * n_lags * a[a_idx]
    
    return grad, R

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed minimax
with mirror descent, Nesterov acceleration, stochastic perturbation,
spectral smoothing, and multi-length search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def eval_score(a):
    """Compute objective 2*n*max(conv)/(sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    R = fast_autocorr(a)
    return 2.0 * n * R.max() / (s * s)

def compute_smoothed_grad(a, beta):
    """
    Compute gradient of f_beta(a) = (1/beta)*log(sum_k exp(beta*R(k))).
    Returns (grad, R) where grad has shape (n,).
    grad[m] = 2 * sum_k p[k] * a[k-m] (cross-correlation).
    """
    n = len(a)
    R = fast_autocorr(a)
    R_max = R.max()
    log_p = beta * (R - R_max)
    log_p -= log_p.max()
    p = np.exp(log_p)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(R)) / len(R)
    else:
        p /= p_sum
    # cross-correlation: grad[m] = 2 * sum_j a[j] * p[j+m]
    # = 2 * ifft(conj(FFT(a)) * FFT(p))[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(np.conj(A) * P, fft_size)
    grad = 2.0 * corr[:n]
    return grad, R

def mirror_descent_smoothed(a_init, time_limit, beta_start=5.0, beta_end=80.0,
                            gamma=1.0, verbose=True):
    """Mirror descent with smoothed subgradient on simplex."""
    n = len(a_init)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_score = 2.0 * n * best_M / c**2

    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(t / 8000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        grad, R = compute_smoothed_grad(a, beta)
        step = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        avg_a += a
        n_avg += 1
        if t % 30 == 0 or t < 5:
            M = fast_autocorr(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                best_score = 2.0 * n * best_M / c**2
            if n_avg >= 50 and t % 100 == 0:
                avg_check = avg_a / n_avg
                avg_check = np.maximum(avg_check, 0.0)
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * c / s_avg
                    M_avg = fast_autocorr(avg_check).max()
                    if M_avg < best_M:
                        best_M = M_avg
                        best_a = avg_check.copy()
                        best_score = 2.0 * n * best_M / c**2
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} beta={beta:.1f} J={best_score:.6f} ({elapsed:.1f}s)")
    return best_a, best_score

def nesterov_smoothed(a_init, time_limit, beta_start=5.0, beta_end=100.0,
                      verbose=True):
    """Nesterov-accelerated projected gradient with smoothed subgradient."""
    n = len(a_init)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_score = 2.0 * n * best_M / c**2

    t0 = time.monotonic()
    y = a.copy()
    x_prev = a.copy()
    rr = 1.0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(t / 5000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        grad, R = compute_smoothed_grad(y, beta)
        step = 1.0 / (np.sqrt(t) + 1.0)
        y_new = y - step * grad
        y_new = proj_simplex(y_new, c)
        rr_new = (1.0 + np.sqrt(1.0 + 4.0 * rr * rr)) / 2.0
        if t == 1:
            x_new = y_new.copy()
        else:
            theta = (rr - 1.0) / rr_new
            x_new = y_new + theta * (y_new - x_prev)
            x_new = proj_simplex(x_new, c)
        x_prev = y_new
        y = x_new
        rr = rr_new
        if t % 15 == 0:
            M = fast_autocorr(y).max()
            if M < best_M:
                best_M = M
                best_a = y.copy()
                best_score = 2.0 * n * best_M / c**2
            M_x = fast_autocorr(x_new).max()
            if M_x < best_M:
                best_M = M_x
                best_a = x_new.copy()
                best_score = 2.0 * n * best_M / c**2
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS t={t} beta={beta:.1f} J={best_score:.6f} ({elapsed:.1f}s)")
    return best_a, best_score

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    gamma = 1.5

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta = 10.0 + 60.0 * min(t / 3000.0, 1.0)
        R = fast_autocorr(a)
        R_max = R.max()
        log_p = beta * (R - R_max)
        log_p -= log_p.max()
        p = np.exp(log_p)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(R)) / len(R)
        else:
            p /= p_sum
        n_lags = len(R)
        ns = min(n_samples, n_lags)
        samples = rng.choice(n_lags, size=ns, replace=False, p=p)
        grad = np.zeros(n)
        for k in samples:
            k = int(k)

            m_end = min(n, k + 1)
            m_idx = np.arange(m_start, m_end)
            a_idx = k - m_idx
            grad[m_idx] += 2.0 * p[k] * n_lags * a[a_idx]
        step = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        if t % 50 == 0:

            if M < best_M:
                best_M = M
                best_a = a.copy()
                best_score = 2.0 * n * best_M / c**2
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SP t={t} J={best_score:.6f} ({elapsed:.1f}s)")

def coordinate_polish(a_init, time_limit, verbose=False):

    n = len(a_init)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_score = 2.0 * n * best_M / c**2
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        R = fast_autocorr(a)
        top_k = np.argsort(R)[::-1][:6]
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:

            if len(candidates) == 0:
                continue
            cands_sorted = candidates[np.argsort(a[candidates])]
            n_try = min(3, len(cands_sorted))
            for di in range(n_try):
                dst = cands_sorted[di]
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        best_score = 2.0 * n * best_M / c**2
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()

                            best_M = M_new
                            best_a = a.copy()
                            best_score = 2.0 * n * best_M / c**2
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a * c / a.sum()
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): J={best_score:.6f}")

def spectral_smooth(a, sigma_frac=0.01):
    """Gaussian smoothing in frequency domain."""
    n = len(a)
    A = rfft(a)
    freqs = np.fft.rfftfreq(n)
    sigma = sigma_frac * n
    A_s = A * np.exp(-2 * np.pi**2 * sigma**2 * freqs**2)
    result = irfft(A_s, n)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-15:
        return a.copy()
    return result / s * a.sum()

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def multi_phase_optimize(a_init, time_limit, verbose=True):
    """Run multiple optimization phases and return best."""
    n = len(a_init)
    c = np.sqrt(2.0 * n)
    best_a = a_init.copy()
    best_score = eval_score(a_init)
    t0 = time.monotonic()

    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        t_a = min(rem * 0.35, time_limit * 0.4)
        a_out, sc = mirror_descent_smoothed(a_init, t_a, beta_start=5.0, beta_end=60.0, gamma=1.2, verbose=verbose)
        if sc < best_score:
            best_a, best_score = a_out, sc
            if verbose:
                print(f"  MD improved to {sc:.6f}")

        t_b = min(rem * 0.35, time_limit * 0.4)

        if sc < best_score:
            best_a, best_score = a_out, sc
            if verbose:
                print(f"  NS improved to {sc:.6f}")

    if rem > 15:
        a_smooth = spectral_smooth(best_a, sigma_frac=0.008)

        a_smooth = a_smooth * c_n / a_smooth.sum()
        t_c = min(rem * 0.25, 60.0)
        if t_c > 5:

            if sc < best_score:

    # Phase C: Coordinate polish

    if rem > 10:
        t_d = min(rem * 0.3, 120.0)
        a_out, sc = coordinate_polish(best_a, t_d, verbose=verbose)
        if sc < best_score:
            best_a, best_score = a_out, sc
            if verbose:
                print(f"  Coord improved to {sc:.6f}")

    # Phase D: Stochastic polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
