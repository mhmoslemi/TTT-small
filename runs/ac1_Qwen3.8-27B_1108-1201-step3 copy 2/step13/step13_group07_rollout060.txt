def spectral_shaping(a, n_iter=300, alpha=0.3):

def refine_peak_shaving(a0, time_limit, seed=123):
    """Aggressive peak shaving with multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        # Strategy 1: Move mass from peak-contributing to non-peak positions
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
                
            src = idx[np.argmax(contrib)]
            
            # Find good destination: low value, not contributing to any peak
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and a[j] < 0.5 * a.max()]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Try moving to the best candidate
            best_candidate = None
            best_candidate_score = best_score
            for j in rng.choice(candidates, size=min(10, len(candidates)), replace=False):
                for frac in [0.001, 0.003, 0.005, 0.01]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[j] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_candidate_score:
                        best_candidate_score = score
                        best_candidate = (a_trial, j, frac)
                if best_candidate is not None:
                    break
            
            if best_candidate is not None:
                a = best_candidate[0]
                best_score = best_candidate[1]
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Strategy 2: Random small perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 2000:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Let me also think about a different approach entirely: **mirror descent on the game**.

Instead of solving the LP exactly, we can use mirror descent (exponential weights) to approximately solve the best-response problem:

Given `a_fixed`, we want to find `b >= 0, sum(b)=1` that minimizes `max_k (a_fixed * b)_k`.

Mirror descent:
- Initialize `b = 1/n`.
- At each step: compute `c = a_fixed * b` (the convolution).
- Find the "worst" lag `k*` that maximizes `c[k]`.
- Update: `b_new[j] proportional to b[j] * exp(-eta * a_fixed[k*-j])` (for valid indices).

This is a first-order method that requires only FFT computations (no LP solve). It's much faster per iteration but requires more iterations to converge.

Let me implement this as an alternative to the LP:

The subgradient of `max_k (a*b)_k` at any point is in the convex hull of the Lagrangian gradients. The "umbrella" subgradient uses the softmax weights over the peaks.

So the mirror descent update should be:

where `w_k` are the softmax weights on the peaks.

`b_new[j] proportional to b[j] * exp(-eta * (w * a)[j])`

where `(w * a)[j] = sum_k w_k * a[k-j]` is a weighted sum of shifted copies of `a`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (frequency-domain initialization)
2. Nesterov-accelerated zero-sum game with mirror-descent inner loop
3. Peak equalization (active-set LP)
4. Refined peak shaving (coordinate descent)
5. Multi-length exploration
Key improvement over parent: faster inner best-response via mirror descent,
enabling more iterations; adaptive Nesterov restart; active-set peak equalization.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def internal_score(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_autocorr(a).max())


def project_simplex(v, target_sum=1.0):
    """Project onto {x: sum(x)=target, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.nonzero(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.ones(n) * (target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(n) * (target_sum / n)
    return result / ss * target_sum


def get_lp_solver():
    for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS


def mirror_best_response(a_fixed, n_inner=200, eta=None, top_k=8):
    """Approximate LP best response via mirror descent.
    Solves: min_b max_k (a_fixed * b)_k s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    if eta is None:
        eta = 1.0 / n * 5.0
    b = np.ones(n) / n
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a_fixed, fft_size)
    
    for t in range(n_inner):
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:2*n-1]
        
        # Top-k weights
        kk = min(top_k, len(c))
        top_idx = np.argpartition(c, -kk)[-kk:]
        c_top = c[top_idx]
        max_c = c_top.max()
        w = np.exp(50.0 * (c_top - max_c))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(kk) / kk
        else:
            w = w / ws
        
        # Gradient: grad[j] = sum_i w_i * a_fixed[top_idx[i] - j]
        grad = np.zeros(n)
        for ii in range(kk):
            k = top_idx[ii]
            wi = w[ii]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                # a_fixed[k-j] for j in [js, je]
                # indices: k-je, k-je+1, ..., k-js (reversed)
                grad[js:je+1] += wi * a_fixed[k-je:k-js+1][::-1]
        
        # Update
        b = b * np.exp(-eta * grad)
        bs = b.sum()
        if bs < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / bs
    
    return b


def nesterov_game_optimize(a0, time_limit, use_lp=True, solver=None, verbose=True):
    """Nesterov-accelerated zero-sum game optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    momentum = 0.0
    stall = 0
    last_improve = 0
    
    # Inner iterations for mirror descent
    n_inner = 100 if not use_lp else 0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        if use_lp:
            # Solve LP
            b_star = _solve_lp_br(y, solver, n)
        else:
            # Mirror descent
            b_star = mirror_best_response(y, n_inner=n_inner, top_k=8)
        
        if b_star is None:
            continue
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        if t > 1:
            momentum = 0.99 * momentum + 0.01 * min(1.0, 5.0 / (t + 1))
        
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = project_simplex(a_new)
        
        y_prev = y
        y = y_new
        a = a_new
        
        avg_a += a
        n_avg += 1
        
        # Score check (every 5 iterations for speed)
        if t % 5 == 0:
            sc_a = internal_score(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall = 0
                last_improve = t
            else:
                stall += 5
            
            sc_y = internal_score(y)
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                stall = 0
                last_improve = t
        
        # Time average
        if n_avg >= 50 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            sc_avg = internal_score(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
                stall = 0
            
        # Restart
        if stall > 3000:
            momentum = 0.0
            stall = 0
            if verbose:
                print(f"      [restart t={t}, best={best_score:.6f}]")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Game t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def _solve_lp_br(a_fixed, solver, n):
    """Solve LP best response."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build T @ b <= M
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    prob = cp.Problem(cp.Minimize(M_var), [T @ b_var <= M_var, cp.sum(b_var) == 1])
    try:
        prob.solve(solver=solver, max_iter=300)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iter=300)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    ss = b_val.sum()
    if ss < 1e-15:
        return None
    return b_val / ss
