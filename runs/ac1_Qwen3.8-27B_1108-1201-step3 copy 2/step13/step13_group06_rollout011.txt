    """Compute 2*n*max_b/sum_a^2 for a full sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def lse_objective_grad_sym(half_a, T, n):
    """
    Compute LSE-smoothed objective and gradient w.r.t. half_a.
    The full sequence is expand_symmetric(half_a, n).
    Returns (objective_value, gradient_wrt_half_a).
    """
    full_a = expand_symmetric(half_a, n)
    c = fast_autocorr(full_a)
    
    # LSE smooth max
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    log_sum = np.log(exp_c.sum())
    obj = T * (c_max + log_sum)
    
    # Weights
    w = exp_c / exp_c.sum()
    
    # Gradient of (a*a) at lag k w.r.t. a[m]: 2*a[k-m] if 0<=k-m<n else 0
    # So grad_full[m] = 2 * sum_{k: 0<=k-m<=2n-2} w[k] * a[k-m]
    #                  = 2 * sum_{t=0}^{n-1} w[m+t] * a[t]
    
    # Compute using the FFT trick:
    # out[m] = sum_{t=0}^{n-1} a[t] * w[m+t]
    # flip_w[k] = w[2n-2-k]
    # out[m] = conv(a, flip_w)[2n-2-m]
    # We need out[0..n-1], so we need conv(a, flip_w)[n-1..2n-2]
    
    flip_w = w[::-1]
    # conv(a, flip_w) has length n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(full_a, fft_size)
    W = rfft(flip_w, fft_size)
    conv_full = irfft(A * W, fft_size)[:3*n-2]
    
    # out[m] = conv_full[2n-2-m] for m = 0..n-1
    # So out = conv_full[n-1:2n-1][::-1]
    grad_full = 2.0 * conv_full[n-1:2n-1][::-1]
    
    # Now project gradient to half_a space
    # full_a = expand_symmetric(half_a, n)
    # If n is even: full_a[i] = half_a[i] for i < n//2, full_a[i] = half_a[n-1-i] for i >= n//2
    # d(full_a[i])/d(half_a[j]) = 1 if i == j (i < n//2), 1 if i == n-1-j (i >= n//2)
    # So grad_half[j] = grad_full[j] + grad_full[n-1-j] (for even n)
    # For odd n: grad_half[j] = grad_full[j] + grad_full[n-1-j] for j < (n-1)//2
    #            grad_half[mid] = grad_full[mid] where mid = (n-1)//2
    
    half_len = (n + 1) // 2
    grad_half = np.zeros(half_len)
    for j in range(half_len):
        grad_half[j] = grad_full[j]
        if n - 1 - j != j:
            grad_half[j] += grad_full[n - 1 - j]
    
    return obj, grad_half
