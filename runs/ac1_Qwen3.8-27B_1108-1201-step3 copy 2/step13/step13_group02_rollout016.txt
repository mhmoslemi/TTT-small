def fast_autocorr(a):

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that (T @ b) = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Temperature schedule
    initial_score = best_score
    T_start = max(0.01, initial_score * 0.05)
    T_end = max(0.001, initial_score * 0.001)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Softmax weights
        b_shifted = (b - max_b) / T
        b_shifted = np.clip(b_shifted, -50, 0)  # numerical stability
        w = np.exp(b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient
        grad = compute_lse_gradient(a, w)
        
        # Adam update
        steps += 1
        lr = 1.0 / np.sqrt(steps + 1)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = a - lr * update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        score = fast_autocorr(a).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, verbose=True):
    """Direct subgradient method on max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find all argmax positions (within tolerance)
        tol = 1e-10 * max_b
        argmax_set = np.where(b >= max_b - tol)[0]
        
        # Compute subgradient: average of gradients at argmax positions
        grad = np.zeros(n)
        for k in argmax_set:
            # d(a*a)_k / da_i = a[k-i] + a[i-k] (with boundary handling)
            # For a fixed k: contrib_i = a[max(0,k-i):min(n,k+1)] flipped and added appropriately
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[k - idx]
            grad[idx] += contrib
            # The other term: a[i-k] for i in range
            idx2 = np.arange(max(0, k), min(n, k + n))  # i such that i-k in [0,n-1]
            # Wait, this is the same as the first term by symmetry of the convolution
            # Actually for (a*a)_k = sum_j a[j]*a[k-j], d/da_i:
            # = a[k-i] (from the first factor) + a[i-k] (from the second factor)
            # The first term: a[k-i] for i in [max(0,k-n+1), min(n-1,k)]
            # The second term: a[i-k] for i in [max(0,k), min(n-1,k+n-1)]
            # Since k is in [0, 2n-2]:
            # First term valid for i in [max(0,k-n+1), min(n-1,k)]
            # Second term valid for i in [max(0,k), min(n-1,k+n-1)] = [k, min(n-1, k+n-1)]
            if k > 0:
                idx2 = np.arange(k, min(n, k + n))
                # a[i-k] for i in idx2: a[0], a[1], ..., a[min(n-1, k+n-1)-k]
                grad[idx2] += a[idx2 - k]
        
        # Normalize subgradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step size: O(1/sqrt(t))
        lr = norm / (grad_norm * np.sqrt(steps + 1) * 10)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        steps += 1
        score = fast_autocorr(a).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

                non_peak = np.where(np.arange(n) != src)[0]

            sorted_dst = non_peak[np.argsort(a[non_peak])][:n_dest * 2]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]
            
            frac = rng.choice([0.05, 0.1, 0.15, 0.2])
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            
            dst_vals = a[selected_dst] + 1e-10
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()
            amounts = amt * weights
            
            a_trial = a.copy()
            a_trial[src] -= amt
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amounts[di]
            
            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            n_stall += 1
            if n_stall > 100:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.1, 0.3)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                            a = a_trial
                n_stall = 0
        
        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
