from scipy.optimize import minimize as scipy_minimize

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_seq(a, target_sum=None):
    """Normalize sequence to non-negative with given sum."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    return a * target_sum / s


def gradient_at_peak(a, k_peak):
    """
    Compute gradient of b[k_peak] with respect to a.
    db[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    valid = np.zeros(n, dtype=bool)
    idx = k_peak - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad


def full_objective_gradient(a, k_peak):
    """
    Gradient of 2*n*max_b/s^2 at the current peak.
    = (2n/s^2) * grad_b - (4n*max_b/s^3) * a
    """
    n = len(a)
    s = a.sum()
    b = fast_autocorr(a)
    max_b = b[k_peak]
    grad_b = gradient_at_peak(a, k_peak)
    grad = (2.0 * n / (s * s)) * grad_b - (4.0 * n * max_b / (s ** 3)) * a
    return grad


def objective_value(a):
    """Compute 2*n*max_b/s^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build constraint matrix efficiently
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M, cp.sum(b_var) == np.sqrt(2.0 * n)]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_unconstrained(a, M, solver):
    """LP without sum constraint (for Nesterov averaging)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, adapt_step=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    last_improve = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_unconstrained(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        if adapt_step:
            ratio = M_curr / max(best_score, 1e-15)
            alpha = 2.0 / (t + 2.0)
            alpha = min(alpha * min(ratio, 3.0), 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    NLP stall at iter={t}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mwu_optimize(a0, time_limit, eta_init=0.1, verbose=False):
    """Multiplicative Weights Update optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_init
    last_improve = time.monotonic()
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        grad = gradient_at_peak(a, k_star)

        # MWU update: a_new[j] = a[j] * exp(-eta * grad[j] / (2*max_grad))
        max_grad = grad.max()
        if max_grad < 1e-15:
            break
        a_new = a * np.exp(-eta * grad / max_grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = normalize_seq(a_new, norm)

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            last_improve = time.monotonic()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                eta *= 0.9

        a = a_new

        if time.monotonic() - last_improve > 30.0 and stall > 100:
            if verbose:
                print(f"    MWU stall at iter={t}, best={best_score:.8f}")
            break

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU iter={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    """Log-sum-exp smooth approximation of max autocorrelation."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    return 2.0 * n * lse / (s * s)


def lse_gradient(a_flat, n, T_temp):
    """Gradient of LSE objective."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    # Gradient of b[k] w.r.t. a[j]: sum over k of w[k] * 2*a[k-j] (if valid)
    # Use FFT trick: grad[j] = 2 * (w * a)[j] where * is circular correlation
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    W_fft = rfft(w_padded, fft_size)
    A_fft = rfft(a_padded, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    """Optimize using LSE smooth surrogate with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 100.0)

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'ftol': 1e-10, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def gradient_coord_descent(a0, time_limit, seed=42):
    """
    Coordinate descent using analytical gradients.
    For each coordinate, try moving mass to reduce the peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        max_b = b[k_peak]
        s = a.sum()

        # Compute gradient at peak
        grad = gradient_at_peak(a, k_peak)

        # Find coordinates with highest gradient (most impactful on peak)
        # and coordinates with lowest values (good targets for receiving mass)
        grad_sorted = np.argsort(grad)[::-1]
        a_sorted = np.argsort(a)[::-1]

        improved = False

        # Try moving mass from high-gradient coords to low-value coords
        for src_pos in range(min(10, n)):
            src = int(grad_sorted[src_pos])
            if a[src] < 1e-10:
                continue

            # Targets: coordinates not contributing to the peak much
            # and with low current values
            for dst_pos in range(min(10, n)):
                dst = int(a_sorted[dst_pos])
                if dst == src:
                    continue
                # Check if dst contributes to peak
                if 0 <= k_peak - dst < n or 0 <= dst - k_peak < n:
                    grad_dst = grad[dst]
                    if grad_dst >= grad[src] * 0.5:
                        continue  # dst also contributes significantly

                for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbations
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.01, 0.15))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved:
                stall += 1
                if stall > 100:
                    break
        else:
            stall = 0

    return best_a, best_score


def peak_targeted_polish(a0, time_limit, seed=123):
    """
    Aggressive polish targeting specific peaks.
    Identify all significant peaks and try to reduce them.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        # Find all peaks above 90% of max
        threshold = 0.9 * max_b
        peak_lags = np.where(b >= threshold)[0]

        improved = False
        for k_peak in peak_lags:
            # For this peak, find the top contributors
            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[int(k_peak) - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:8]]

            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                # Find good destinations (low contribution to this peak)
                # and to other peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in peak_lags:
                    js2 = max(0, int(k2) - n + 1)
                    je2 = min(n - 1, int(k2))
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by a value (prefer moving to low-value positions)
                candidates = candidates[np.argsort(a[candidates])][:10]

                for dst in candidates:
                    for frac in [0.01, 0.03, 0.08, 0.15, 0.25]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random restart within neighborhood
                for _ in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = float(rng.uniform(0.005, 0.1))
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 200:
                break
        else:
            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    f_padded = np.append(f_norm, f_norm[0])  # circular interp
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_padded)
    f_interp = f_interp[:new_n]
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Ramp + oscillation
        x = np.linspace(0.0, 1.0, n)
        a = x * (1 - x) * (1.0 + 0.3 * np.sin(10 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        # Asymmetric (key for breaking symmetry)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.0 * (x - 0.2) ** 2 / (0.1 + x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def pair_swap_optimize(a0, time_limit, seed=999):
    """
    Optimized pair-swap: move mass from high-impact to low-impact coordinates.
    Uses the gradient structure to guide moves efficiently.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    # Precompute: for each lag k, which pairs (i, j) contribute a[i]*a[j]
    # For peak at k: contrib(i) = a[i] * a[k-i] (if valid)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        max_b = b[k_peak]

        # Gradient: grad[j] = 2*a[k_peak - j] if valid
        grad = gradient_at_peak(a, k_peak)

        # Source: high gradient (reducing it helps most)
        # Destination: low gradient AND low a value
        src_candidates = np.argsort(grad)[::-1][:20]
        dst_candidates = np.argsort(grad)[:20]

        improved = False
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                # Only move if dst has much lower gradient
                if grad[dst] > grad[src] * 0.3:
                    continue
                # Try different fractions
                for frac in [0.05, 0.1, 0.2, 0.35]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Broader search: random pairs with gradient-based scoring
            found = False
            for _ in range(100):
                src = int(rng.integers(n))
                dst = int(rng.integers(n))
                if src == dst or a[src] < 1e-12:
                    continue
                if grad[dst] >= grad[src] * 0.5:
                    continue
                frac = float(rng.uniform(0.02, 0.3))
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found:
                stall += 1
                if stall > 150:
                    break
        else:
            stall = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load previous best
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Multi-length scan ===
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1600, 2000, 2400]
    scan_results = {}

    for n_scan in scan_lengths:
        elapsed_check = time.monotonic() - t_global
        if elapsed_check > time_budget * 0.12:
            break
        remaining_for_scan = time_budget * 0.12 - elapsed_check
        scan_time = min(remaining_for_scan / max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 60.0)
        if scan_time < 15:
            break

        if best_seq is not None and len(best_seq) != n_scan:
            init_seq = interpolate_sequence(best_seq, n_scan)
        elif best_seq is not None:
            init_seq = best_seq.copy()
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = normalize_seq(init_seq, np.sqrt(2.0 * n_scan))
        a_out, sc_out = nesterov_lp(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: max_b={sc_out:.8f} (eval={2*n_scan*sc_out/(np.sqrt(2*n_scan)**2):.8f}) ({time.monotonic()-t_global:.0f}s)")

    # Pick best from scan
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1] / k)
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Deep LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_base))
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, adapt_step=True)
        do_try_update(a_out, f"lp_deep_n{n_base}")

    # === Phase 2: Pair swap optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Pair swap n={n_best} (budget={t2:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = pair_swap_optimize(a_norm, t2, seed=111)
        do_try_update(a_out, f"pair_swap_n{n_best}")

    # === Phase 3: LSE + scipy gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={t3:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=40.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # === Phase 4: Try longer length ===
    n_long = min(n_base + 400, 2800)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30 and n_long != n_base:
        print(f"\n=== Phase 4: LP n={n_long} (budget={t4:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = normalize_seq(init_long, np.sqrt(2.0 * n_long))
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=True, adapt_step=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 5: MWU for diversity ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: MWU n={n_best} (budget={t5:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = mwu_optimize(a_norm, t5, eta_init=0.05, verbose=False)
        do_try_update(a_out, f"mwu_n{n_best}")

    # === Phase 6: Peak-targeted polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Peak polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = peak_targeted_polish(a_norm, t6, seed=555)
        do_try_update(a_out, f"peak_polish_n{n_best}")

    # === Phase 7: Gradient coord descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Grad CD n={n_best} (budget={t7:.0f}s) ===")
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = gradient_coord_descent(a_norm, t7, seed=777)
        do_try_update(a_out, f"grad_cd_n{n_best}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_best} (remaining={remaining:.0f}s) ===")

    # Alternate between LP (short bursts) and local search
    phases = [
        ("lp_short", lambda b: nesterov_lp(normalize_seq(b, np.sqrt(2.0*len(b))), min(remaining*0.15, 80), solver, verbose=False)[0]),
        ("polish1", lambda b: peak_targeted_polish(normalize_seq(b, np.sqrt(2.0*len(b))), min(remaining*0.15, 80), seed=888)[0]),
        ("lp_short2", lambda b: nesterov_lp(normalize_seq(b, np.sqrt(2.0*len(b))), min(remaining*0.10, 50), solver, verbose=False)[0]),
        ("polish2", lambda b: peak_targeted_polish(normalize_seq(b, np.sqrt(2.0*len(b))), min(remaining*0.20, 100), seed=999)[0]),
    ]

    for i, (name, func) in enumerate(phases):
        if time.monotonic() - t_global > time_budget - 5:
            break
        a_out = func(best_seq)
        do_try_update(a_out, f"final_{name}")

    # Final pair swap polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_norm = normalize_seq(best_seq, np.sqrt(2.0 * len(best_seq)))
        a_out, _ = pair_swap_optimize(a_norm, remaining * 0.5, seed=2024)
        do_try_update(a_out, "final_pair")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
