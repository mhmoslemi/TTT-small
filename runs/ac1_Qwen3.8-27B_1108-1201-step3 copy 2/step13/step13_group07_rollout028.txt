    # Bisection on threshold
    lo = v.min()
    hi = v.max()
    for _ in range(60):
        theta = 0.5 * (lo + hi)
        x = np.maximum(v - theta, 0)
        s = x.sum()
        if s > target_sum:
            lo = theta
        else:
            hi = theta
    theta = 0.5 * (lo + hi)
    x = np.maximum(v - theta, 0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-13:
        idx = np.argmax(x)
        x[idx] = max(0, x[idx] + diff)
        s = x.sum()
        if s > 0:
            x = x / s * target_sum
    return x


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def cross_corr_grad(a_fixed, w, n):
    """
    Compute grad[i] = sum_j a_fixed[j] * w[i+j] for i=0..n-1.
    This is the gradient of sum_k w[k]*(a*b)[k] w.r.t. b[i].
    Uses FFT: cross_corr(a, w)[i] = conv(a_rev, w)[n-1+i]
    """
    a_rev = a_fixed[::-1].copy()
    conv = fast_cross_conv(a_rev, w)
    return conv[n - 1: 2 * n - 1]


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def approximate_best_response(a_fixed, n_inner=30, T=50.0):
    """
    Approximate argmin_b LSE(T * conv(a_fixed, b)) s.t. b in simplex
    using projected gradient descent.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    fft_size_bc = 1 << (2 * n - 1).bit_length()

    for it in range(n_inner):
        # Compute conv(a_fixed, b)
        conv = fast_cross_conv(a_fixed, b)
        # LSE gradient: softmax(T * conv)
        Tb = T * conv
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Gradient w.r.t. b
        grad_b = cross_corr_grad(a_fixed, w, n)
        # Backtracking line search
        obj_cur = (max_Tb + np.log(exp_Tb.sum())) / T
        lr = 1.0 / (T * 2.0 * a_fixed.max() + 1e-10)
        for _ in range(15):
            b_new = project_simplex(b - lr * grad_b)
            conv_new = fast_cross_conv(a_fixed, b_new)
            obj_new = (T * conv_new).max() / T  # use max as upper bound
            if obj_new <= obj_cur + 1e-12:
                break
            lr *= 0.5
        b = b_new

    return b


def get_lp_solver():
    """Find a working LP solver."""
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a_fixed[k-j] if valid."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve exact LP: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=500)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=500)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_game_fast(a0, time_limit, use_lp_every=2000):
    """
    Nesterov-accelerated zero-sum game with fast LSE-approximated best responses.
    Periodically uses exact LP for precision.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)  # momentum
    beta = 0.9

    T_smooth = 80.0  # initial temperature for LSE
    n_inner = 25

    solver = get_lp_solver()

    stall = 0
    stall_threshold = 8000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Periodically decrease temperature for more precise best response
        T_smooth = max(10.0, 80.0 * math.exp(-t / 50000.0))
        n_inner_cur = min(40, 15 + t // 5000)

        # Get best response
        if use_lp_every > 0 and t % use_lp_every == 0 and solver is not None:
            b_star, M_val = solve_lp_best_response(y, solver)
            if b_star is None:
                b_star = approximate_best_response(y, n_inner=n_inner_cur, T=T_smooth)
        else:
            b_star = approximate_best_response(y, n_inner=n_inner_cur, T=T_smooth)

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = project_simplex(y_new)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Track best
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall = 0
            best_score = score_y
            best_a = y.copy()

        stall += 1
        if stall > stall_threshold:
            # Try perturbation
            perturbation = np.random.default_rng(t).normal(0, 0.01, n)
            a_pert = project_simplex(a + perturbation)
            score_pert = internal_score(a_pert)
            if score_pert < best_score * 1.01:
                y = a_pert.copy()
                a = a_pert.copy()
                stall = 0
            else:
                break

        if t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NG t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T_smooth:.1f}")

    return best_a, best_score

    n = len(a0)

    a = np.maximum(a0.astype(np.float64), 0)

    for t in range(1, 1000000):

            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = project_simplex(y_new)

        if t == 1:

        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:

        if t % 500 == 0:

    return best_a, best_score


def lse_smoothed_grad_descent(a0, time_limit, T_start=50.0, T_end=1.0):
    """
    Smoothed gradient descent on LSE approximation of max autocorrelation.
    Uses Adam optimizer with adaptive temperature annealing.
    """
    n = len(a0)
    a_init = np.maximum(a0.astype(np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    best_a = a_init.copy()
    best_score = internal_score(a_init)
    t0 = time.monotonic()

    z = np.log(np.maximum(a_init, 1e-15))
    fft_size = 1 << (2 * n - 1).bit_length()
    fft_size_grad = 1 << (3 * n - 1).bit_length()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0
    base_lr = 0.02

    n_total_steps = max(1, int(time_limit * 15))
    # Geometric temperature schedule
    for step in range(1, n_total_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        frac = step / n_total_steps
        T_cur = T_start * (T_end / T_start) ** frac

        a = softmax_fn(z)
        b = fast_autocorr(a)
        Tb = T_cur * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient w.r.t. a: 2 * sum_k w_k * a[k-i] (valid)
        a_rev = a[::-1].copy()
        conv = fast_cross_conv(a_rev, w)
        grad_a = 2.0 * conv[n - 1: 2 * n - 1]

        # Adam update in z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr_t = base_lr / (1 + 0.005 * step)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Check score periodically
        if step % 50 == 0:
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    a_final = softmax_fn(z)
    score = internal_score(a_final)
    if score < best_score:
        best_score = score
        best_a = a_final.copy()

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with structured moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: coordinate with minimum contribution to any top peak
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if len(candidates) == 0:
                continue

            # Score candidates by their contribution to peaks
            min_contrib = np.inf
            dst = candidates[0]
            for j in candidates[:min(len(candidates), 200)]:
                c = 0.0
                for k2 in top_k:
                    if 0 <= k2 - j < n:
                        c += b[k2] * a[k2 - j]
                if c < min_contrib:
                    min_contrib = c
                    dst = j

            for frac in [0.003, 0.01, 0.03, 0.08, 0.15]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def evaluate_safe(arr):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

            a_out, _ = peak_shaving(best_seq, t7, seed=999)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [Shave2]: {best_score:.6f}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)}, t={time.monotonic()-t_global_start:.0f}s)")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
