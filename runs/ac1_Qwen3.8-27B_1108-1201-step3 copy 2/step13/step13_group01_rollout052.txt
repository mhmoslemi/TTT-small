from scipy.fft import rfft, irfft
import cvxpy as cp

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def _fast_conv_self(a):
    """Full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _fast_conv(a, b):
    """Full linear convolution of a and b using FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def _lse_value_and_grad(a, T):
    """
    Compute LSE value and gradient.
    f(a) = (1/T) * log(sum_k exp(T * c[k])) where c = conv(a,a)
    grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    """
    n = len(a)
    c = _fast_conv_self(a)
    max_c = c.max()
    exp_c = np.exp(T * (c - max_c))
    w = exp_c / exp_c.sum()
    lse_val = max_c + np.log(exp_c.sum()) / T
    
    # Compute gradient
    # term2[i] = sum_k w[k] * a[i-k] = conv(w, a)[i]
    conv_wa = _fast_conv(w, a)  # length (2n-1) + n - 1 = 3n-2
    term2 = conv_wa[:n]
    
    # term1[i] = sum_k w[k] * a[k-i]
    w_rev = w[::-1]  # length 2n-1
    conv_wrev_a = _fast_conv(w_rev, a)  # length 3n-2
    term1 = conv_wrev_a[2*n-2:3*n-2]
    
    grad = term1 + term2
    return lse_val, grad


def _normalize(a, target_sum=1.0):
    """Normalize a to have the target sum, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a) * target_sum
    return a * target_sum / s


def _evaluate_local(a):
    """Quick local evaluation of the score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = _fast_conv_self(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _fista_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8, verbose=True):
    """
    FISTA with LSE smooth surrogate and temperature annealing.
    Uses adaptive restart for robustness.
    """
    n = len(a0)
    a = _normalize(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_score = _evaluate_local(a)
    t0 = time.monotonic()
    
    # FISTA state
    y = a.copy()
    t_param = 1.0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(1, n_stages - 1)
        T = T_start * (T_end / T_start) ** frac
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / (n_stages - stage)
        
        # Step size for this stage
        # L (Lipschitz constant) ~ T * 2 (bound on Hessian of LSE)
        # Use slightly smaller for safety
        eta_base = 0.8 / (T + 0.1)
        
        if verbose:
            print(f"    FISTA stage {stage}/{n_stages}: T={T:.1f}, eta={eta_base:.4f}")
        
        stage_start = time.monotonic()
        iters_in_stage = 0
        last_restart_dist = 0.0
        
        while time.monotonic() - stage_start < stage_time:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at y
            _, grad_y = _lse_value_and_grad(y, T)
            
            # FISTA update
            t_old = t_param
            t_param = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
            
            # Adaptive restart: if distance is increasing, reset
            # (simplified: reset every K iterations or based on objective)
            
            y_new = y - eta_base * grad_y
            y_new = np.maximum(y_new, 0.0)
            y_new = _normalize(y_new, 1.0)
            
            # Nesterov extrapolation
            momentum = (t_old - 1.0) / t_param
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, 1.0)
            
            y = y_new
            a = a_new
            iters_in_stage += 1
            
            # Check improvement
            score = _evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            if iters_in_stage % 100 == 0:
                score = _evaluate_local(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score


def _multiplicative_weights(a0, time_limit, T=10.0, eta=0.5, verbose=True):
    """
    Multiplicative weights (exponential gradient) method.
    Update: a[i] <- a[i] * exp(-eta * g[i]), then normalize.
    """
    n = len(a0)
    a = _normalize(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_score = _evaluate_local(a)
    t0 = time.monotonic()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        _, grad = _lse_value_and_grad(a, T)
        
        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * grad
        # Clip to avoid overflow
        log_a = np.clip(log_a, -50, 50)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, 1.0)
        
        a = a_new
        it += 1
        
        if it % 50 == 0:
            score = _evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    MW iter={it}: score={score:.8f}")
    
    score = _evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T_mat[idx, i] = a[idx - i]
    
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with adaptive Nesterov acceleration."""
    n = len(a0)
    a = _normalize(np.asarray(a0, dtype=np.float64), 1.0)
    y = a.copy()
    best_a = a.copy()
    best_score = _evaluate_local(a)
    t0 = time.monotonic()
    t_param = 1.0
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    prev_y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c_y = _fast_conv_self(y)
        M_curr = c_y.max()
        if M_curr < 1e-12:
            break
        
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b = _normalize(b, 1.0)
        n_iters += 1
        
        # Adaptive step size
        alpha = 2.0 / (t_param + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        y_new = _normalize(y_new, 1.0)
        
        # FISTA-style extrapolation with restart
        t_old = t_param
        t_param = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        momentum = (t_old - 1.0) / t_param
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, 1.0)
        
        # Adaptive restart: if function value at a_new > function value at y_new, reset
        # (simplified: check every 100 iters)
        if n_iters % 100 == 0:
            s_a = _evaluate_local(a_new)
            s_y = _evaluate_local(y_new)
            if s_a > s_y * 1.01:
                # Restart
                a_new = y_new.copy()
                t_param = 1.0
            if s_y < best_score:
                best_score = s_y
                best_a = y_new.copy()
                last_improve = time.monotonic()
            if s_a < best_score:
                best_score = s_a
                best_a = a_new.copy()
                last_improve = time.monotonic()
        
        y = y_new
        a = a_new
        
        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 20:
                if verbose:
                    print(f"    LP-Nesterov stalled at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 200 == 0:
            score = _evaluate_local(a)
            print(f"    LP-Nest iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def _gilbert_proak(a0, time_limit, n_freq_check=50, verbose=True):
    """
    Gilbert-Proak style peak reduction.
    Iteratively find the peak and push it down.
    """
    n = len(a0)
    a = _normalize(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_score = _evaluate_local(a)
    t0 = time.monotonic()
    
    it = 0
    eta = 0.1
    while time.monotonic() - t0 < time_limit:
        c = _fast_conv_self(a)
        k_star = int(np.argmax(c))
        M_curr = c[k_star]
        
        # Subgradient direction: d[i] = a[k_star - i] + a[i - k_star]
        d = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                d[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                d[i] += a[j2]
        
        # Update
        a_new = a - eta * d
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, 1.0)
        
        a = a_new
        it += 1
        
        if it % n_freq_check == 0:
            score = _evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            # Decay step size
            eta *= 0.995
            if eta < 0.001:
                eta = 0.001
    
    score = _evaluate_local(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def _coord_descent(a0, time_limit, seed=42):
    """Coordinate descent with guided moves based on peak analysis."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = _normalize(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_score = _evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = _fast_conv_self(a)
        top_peaks = np.argsort(c)[::-1][:5]
        
        improved = False
        for k_peak in top_peaks:
            # Find positions contributing to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:3]]
            
            # Try moving mass from top contributors to low-contribution positions
            for src in top_idx:
                if a[src] < 1e-10:
                    continue
                # Find target: positions NOT contributing to top peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                
                candidates = np.where(peak_mask)[0]  # positions contributing to peaks
                candidates = candidates[candidates != src]
                non_candidates = np.where(~peak_mask)[0]
                
                for dst_list in [non_candidates[:5], candidates[:3]]:
                    for dst in dst_list:
                        if dst == src or a[src] < 1e-10:
                            continue
                        for frac in [0.02, 0.05, 0.1, 0.2]:
                            amt = a[src] * frac
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            a_trial = _normalize(a_trial, 1.0)
                            score = _evaluate_local(a_trial)
                            if score < best_score - 1e-10:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = _normalize(a_trial, 1.0)
                    score = _evaluate_local(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 100:
                break
    
    return best_a, best_score


def _interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length (preserving normalized shape)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _structured_init(n, rng):
    """Generate various structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 3:
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.7 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 4:
        # Chirp-like
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (3 * x ** 2 - 2 * x))
    elif choice == 5:
        # Random smooth
        a = np.exp(-np.abs(x - 0.5) / 0.2)
    elif choice == 6:
        # Multi-peak
        a = np.exp(-8*(x-0.2)**2) + 0.8*np.exp(-8*(x-0.5)**2) + 0.6*np.exp(-8*(x-0.8)**2)
    else:
        # Near-uniform with small perturbation
        a = 1.0 + 0.1 * np.sin(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main function: search for the best sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    print(f"Starting optimization. Time budget: {time_budget:.0f}s")
    
    solver = _get_solver()
    
    best_seq = None
    best_score = float('inf')
    
    # Load initial solution if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = _normalize(p, 1.0)
            p_score = _evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
    
    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Phase 0: Multi-length quick scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1200, 1400, 1600, 1800]
    
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 1600
    
    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.12:
            break
        remaining_scan = time_budget * 0.12 - (time.monotonic() - t_global)
        if remaining_scan < 15:
            break
        
        if best_seq is not None:
            init_seq = _interpolate_seq(best_seq, n_scan)
        else:
            init_seq = _structured_init(n_scan, rng)
        
        a_out, sc_out = _fista_lse(init_seq, min(remaining_scan, 60.0), 
                                     T_start=30.0, T_end=5.0, n_stages=3, verbose=False)
        _try_update(a_out, f"scan_n{n_scan}")
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

        best_seq = _structured_init(n_base, rng)

        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    # Phase 1: Deep FISTA-LSE
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA-LSE n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_out, _ = _fista_lse(best_seq, t1, T_start=50.0, T_end=1.0, n_stages=8, verbose=True)
    _try_update(a_out, f"fista_n{n_base}")
    
    # Phase 2: Multiplicative weights
    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 2: Mult. Weights n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20
