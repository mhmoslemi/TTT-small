import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):
    """
    Compute gradient of T*log(sum exp(b_k/T)) where b = autocorr(a).
    Returns dL/da[m] = r1[m] + r2[m].
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    
    return r1 + r2

def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive step size."""

    s0 = a.sum()

    T = 5.0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        grad = lse_gradient(a, T)

        eta = 1.0 / (T * np.sqrt(it_count + 1))
        
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()

        it_count += 1

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)  # Include endpoints
    # Actually, let's use a different approach: treat the sequence as samples
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0.0, old_n - 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

- `h[m] = sum_k w_k * (a[m-k] + a[k-m])` where the sum is over valid indices.
- First term: `h1[m] = sum_k w_k * a[m-k]` (valid when 0 ≤ m-k < n)

Actually, let me be more careful. Let's define:
- `a` has indices 0 to n-1
- `w_full` has indices 0 to 2n-2 (it's the weight array for the autocorrelation peaks)

`h1[m] = sum_k w_full[k] * a[m-k]` where both `0 ≤ k ≤ 2n-2` and `0 ≤ m-k ≤ n-1`.

This is the cross-correlation of `a` and `w_full` at lag m:
`h1[m] = sum_j a[j] * w_full[m+j]` ... no wait.

Let me use the standard definition. The cross-correlation of sequences x and y at lag m is:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive-temperature LSE gradient descent with Nesterov acceleration
2. Gradient-guided 2-swap local search (uses autocorrelation gradient to select swap pairs)
3. Mirror descent (multiplicative weights) as complementary optimizer
4. Multi-length exploration with interpolation transfer
5. Direct peak-shaving via subgradient descent on active peaks

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """
    Compute gradient of T*log(sum exp(b_k/T)) where b = autocorr(a).
    dL/da[m] = sum_k w_k * (a[m-k] + a[k-m])
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def compute_peak_gradient(a, n_peaks=10):
    """
    Compute the subgradient of max_k b_k using top peaks.
    Returns h[m] = sum_t w_t * (a[m-k_t] + a[k_t-m]) for top peaks k_t.
    This is O(n * n_peaks) using direct computation (fast for small n_peaks).
    """
    n = len(a)
    b = fast_autocorr(a)
    top_ks = np.argsort(b)[::-1][:n_peaks]
    weights = b[top_ks]
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones(n_peaks) / n_peaks
    else:
        weights = weights / w_sum

    h = np.zeros(n)
    for t in range(n_peaks):
        k = int(top_ks[t])
        w = weights[t]
        if w < 1e-15:
            continue
        # Term 1: w * a[m-k] for valid m (0 <= m-k < n)
        m_start1 = max(0, k)
        m_end1 = min(n - 1, k + n - 1)
        if m_start1 <= m_end1:
            h[m_start1:m_end1 + 1] += w * a[np.arange(m_start1, m_end1 + 1) - k]
        # Term 2: w * a[k-m] for valid m (0 <= k-m < n)
        m_start2 = max(0, k - n + 1)
        m_end2 = min(n - 1, k)
        if m_start2 <= m_end2:
            h[m_start2:m_end2 + 1] += w * a[k - np.arange(m_start2, m_end2 + 1)]

    return h, b, top_ks


def nesterov_lse_adaptive(a0, time_limit, T_start=15.0, T_end=0.1, verbose=True):
    """
    Nesterov-accelerated LSE with adaptive temperature based on peak concentration.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    T = T_start
    it_count = 0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        if it_count == 0:
            grad = lse_gradient(y, T)
            step = 1.0 / 3.0
            y_new = y - step * grad
        else:
            momentum = (it_count - 1.0 + np.sqrt(1.0 + 4.0 * (it_count - 1.0) ** 2)) / (2.0 * (it_count + 2.0))
            y_extrap = y + momentum * (y - y_prev)
            y_extrap = np.maximum(y_extrap, 0.0)
            s_ye = y_extrap.sum()
            if s_ye > 1e-15:
                y_extrap = y_extrap * norm / s_ye
            grad = lse_gradient(y_extrap, T)
            step = 1.0 / (it_count + 3.0)
            y_new = y_extrap - step * grad

        y_prev = y.copy()
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new
        it_count += 1

        score = fast_autocorr(y).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive temperature
        if it_count % 30 == 0:
            b = fast_autocorr(y)
            max_b = b.max()
            w = np.exp((b - max_b) / T)
            w /= w.sum()
            entropy = np.exp(-np.sum(w * np.log(w + 1e-300)))
            if entropy < 3.0:
                T = max(T * 0.97, T_end)
            elif entropy > 200.0:
                T = min(T * 1.03, T_start)
            else:
                T = max(T * 0.99, T_end)

        if verbose and it_count % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE it={it_count} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

        if patience > 3000:
            break

    return best_a, best_score


def gradient_guided_search(a0, time_limit, seed=42, verbose=True):
    """
    Novel local search: uses autocorrelation gradient to select the most
    promising 2-swap moves (source = high gradient, dest = low gradient).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        h, b, top_ks = compute_peak_gradient(a, n_peaks=8)

        positive_mask = a > norm * 1e-9
        if positive_mask.sum() < 2:
            break

        # Source: elements with highest h (decreasing them reduces peaks most)
        h_src = h.copy()
        h_src[~positive_mask] = -1e18
        src_rank = np.argsort(h_src)[::-1]

        # Dest: elements with lowest h (increasing them increases peaks least)
        dst_rank = np.argsort(h)[:20]

        improved = False
        for src in src_rank[:8]:
            if not positive_mask[src]:
                continue
            for dst in dst_rank:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-16:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        it_count += 1
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape
            if rng.integers(3) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > norm * 1e-8:
                    frac = float(rng.uniform(0.01, 0.08))
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            it_count += 1

            if stall > 500:
                if verbose:
                    print(f"  GradSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"  GradSearch it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, verbose=True):
    """
    Direct subgradient descent on the max: move against the subgradient
    of the most active peak. Uses Polyak step size.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Lower bound on max autocorrelation for sum=1 sequence: 1/(2n-1)
    # Score lower bound: 2n/(2n-1) ≈ 1
    lb = 2.0 * n / (2.0 * n - 1.0)
    
    y = a.copy()
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        max_b = b[k_star]

        # Subgradient of max_b w.r.t. a[m]: a[m-k*] + a[k*-m]
        sg = np.zeros(n)
        m_start1 = max(0, k_star)
        m_end1 = min(n - 1, k_star + n - 1)
        if m_start1 <= m_end1:
            sg[m_start1:m_end1 + 1] += a[np.arange(m_start1, m_end1 + 1) - k_star]
        m_start2 = max(0, k_star - n + 1)
        m_end2 = min(n - 1, k_star)
        if m_start2 <= m_end2:
            sg[m_start2:m_end2 + 1] += a[k_star - np.arange(m_start2, m_end2 + 1)]

        # Polyak step: eta = (f(y) - f*) / ||sg||^2
        f_curr = 2.0 * n * max_b / (y.sum() ** 2)
        f_star = lb
        sg_norm_sq = np.sum(sg ** 2)
        if sg_norm_sq < 1e-30:
            break
        eta = (f_curr - f_star) / (4.0 * n * sg_norm_sq / (y.sum() ** 2) + 1e-15)
        eta = min(eta, 0.1)  # Cap step size

        y_new = y - eta * sg
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new
        it_count += 1

        score = fast_autocorr(y).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = y.copy()

        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SubGrad it={it_count} k*={k_star} ({elapsed:.0f}s): best={best_score:.8f}")

        if it_count > 100000:
            break

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()

        a = a * norm / s0

    best_a = a.copy()

    t0 = time.monotonic()

    T = 5.0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        grad = lse_gradient(a, T)
        eta = 1.0 / (T * np.sqrt(it_count + 1.0))
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0.0, old_n - 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x - np.pi))
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-((x - 0.5) / 0.2) ** 8)
    else:
        a = np.ones(n) * 0.5
        for _ in range(int(rng.integers(3, 8))):
            center = float(rng.uniform(0, n))
            width = float(rng.uniform(n * 0.05, n * 0.3))
            amp = float(rng.uniform(-0.3, 0.5))
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LSE, mirror descent,
    gradient-guided local search, and subgradient peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, int(rng.integers(0, 1000)))
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # ============ Phase 1: Adaptive LSE Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Adaptive n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse_adaptive(a_work, t1, T_start=15.0, T_end=0.1, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 2: Gradient-Guided Local Search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: GradSearch n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = gradient_guided_search(a_work, t2, seed=42, verbose=True)
        do_try_update(a_out, f"grad_n{n_cur}")

    # ============ Phase 3: Subgradient Peak Shaving ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: SubGradShave n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 160.0)
    if t3 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_peak_shave(a_work, t3, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # ============ Phase 4: Mirror Descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: MirrorDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent(a_work, t4, verbose=True)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # ============ Phase 5: Longer length exploration ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse_adaptive(a_work, t5, T_start=15.0, T_end=0.2, verbose=True)
        do_try_update(a_out, f"lse_n{n2}")
        # Gradient search on longer
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.08, 120.0)
        if t5b > 10 and best_seq is not None and len(best_seq) == n2:
            a_work = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
            a_out, _ = gradient_guided_search(a_work, t5b, seed=777, verbose=True)
            do_try_update(a_out, f"grad_n{n2}")

    # ============ Phase 6: Final LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LSE Polish n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_adaptive(a_work, t6, T_start=2.0, T_end=0.01, verbose=True)
        do_try_update(a_out, "lse_polish")

    # ============ Phase 7: Final Gradient-Guided Search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final GradSearch n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 350.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = gradient_guided_search(a_work, t7, seed=99999, verbose=True)
        do_try_update(a_out, "grad_final")

    # ============ Phase 8: Final subgradient polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 10:
        t8 = min(remaining * 0.30, 200.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final SubGrad n={n_cur} (budget={t8:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_peak_shave(a_work, t8, verbose=True)
            do_try_update(a_out, "subgrad_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
