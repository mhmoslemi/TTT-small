        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        best_ema = ema_decay * best_ema + (1 - ema_decay) * a
        n_ema += 1
        if n_ema % 50 == 0:
            s_ema = best_ema.sum()
            if s_ema > 1e-15:
                ema_check = best_ema * norm / s_ema
                score_ema = fast_autocorr(ema_check).max()
                if score_ema < best_score:
                    best_score = score_ema
                    best_a = ema_check.copy()
                    stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y * norm / s_y
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = y.copy()
            else:
                a = a * norm / s_a
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def compute_subgradient(a, weights):
    """Compute subgradient of weighted sum of autocorrelation terms using FFT.
    weights has length 2n-1. Returns gradient of length n.
    grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) * 1_{valid}
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    # s1[i] = sum_j a[j] * weights[i+j] = cross-correlation
    # s1 = convolve(a_reversed, weights)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv1 = irfft(rfft(a_rev, fft_size) * rfft(weights, fft_size), fft_size)[:size + n - 1]
    s1 = conv1[n - 1:2 * n - 1]

    # s2[i] = sum_j a[j] * weights[i-j] = convolve(weights, a)[0:n]
    conv2 = irfft(rfft(weights, fft_size) * rfft(a, fft_size), fft_size)[:size + n - 1]
    s2 = conv2[0:n]

    return s1 + s2


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient descent with line search for minimizing max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    eta = 1.0
    patience = 800
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Use top-k peaks for subgradient (softening)
        top_k = min(16, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        weights = np.zeros(2 * n - 1)
        # Exponential weighting on top peaks
        for rank, idx in enumerate(sorted_idx[:top_k]):
            weights[idx] = np.exp(-0.3 * rank)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights = weights / w_sum

        grad = compute_subgradient(a, weights)

        # Adaptive step size with backtracking
        for _ in range(20):
            a_trial = a - eta * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            score_trial = fast_autocorr(a_trial).max()
            if score_trial < max_b * (1 - 0.01 * eta):
                a = a_trial
                if score_trial < best_score:
                    best_score = score_trial
                    best_a = a.copy()
                    stall = 0
                break
            else:
                eta *= 0.7
        else:
            eta = max(eta * 0.7, 1e-10)

        steps += 1

        if steps % 100 == 0:
            #偶尔尝试小随机扰动
            if rng.random() < 0.1:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * norm / s_t

            score_now = fast_autocorr(a).max()
            if score_now < best_score:
                best_score = score_now
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step={steps} ({elapsed:.0f}s): eta={eta:.4f}, best={best_score:.8f}")

        if stall > patience:
            # Structured restart
            rng2 = np.random.default_rng(steps * 31)
            pert = rng2.normal(0, 0.02 * a.mean(), n)
            a = np.maximum(a + pert, 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            eta = eta * 2.0
            stall = 0

    return best_a, best_score


def variance_smooth(a0, time_limit, verbose=True):
    """Reduce variance of autocorrelation (L2 polish toward flat spectrum)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    lr = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        mean_b = b.mean()
        residual = b - mean_b

        # Gradient of sum(residual^2) w.r.t. a
        grad = compute_subgradient(a, residual)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new <= best_score * 1.001:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        else:
            lr *= 0.8

        steps += 1
        if steps % 500 == 0:
            lr = min(lr * 1.1, 0.05)

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    VarSmooth step={steps} ({elapsed:.0f}s): lr={lr:.5f}, best={best_score:.8f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, seed=42, verbose=True):
    """Targeted local search: reduce the highest peak by redistributing mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find which pairs (i, j) with i+j=peak_k contribute most
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-18:
            break

        # Try to reduce the peak by moving mass from high-contribution positions
        src = idx[np.argmax(contrib)]

        # Find target: position that minimizes impact on other peaks
        # Heuristic: position with low value and not contributing to other top peaks
        top_peaks = np.argsort(b)[::-1][:5]
        dangerous = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            js2 = max(0, kp - n + 1)
            je2 = min(n - 1, kp)
            dangerous[js2:je2 + 1] = True

        safe_positions = np.where(~dangerous)[0]
        if len(safe_positions) == 0:
            safe_positions = np.where(np.arange(n) != src)[0]

        # Pick target from safe positions with low current value
        candidates = safe_positions[np.argsort(a[safe_positions])][:20]
        if len(candidates) == 0:
            candidates = np.array([0])

        dst = candidates[rng.integers(len(candidates))]

        # Try different fractions
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                break
            elif score < max_b:
                # Accept with some probability (simulated annealing style)
                if rng.random() < 0.1:
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            if n_stall > 100:
                # Larger perturbation
                n_pert = max(3, n // 100)
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        amt = a[i] * rng.uniform(0.1, 0.3)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def geometric_init(n, r):
    """Generate geometric sequence a[i] = r^i, normalized."""
    a = r ** np.arange(n, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def polynomial_init(n, p):
    """Generate polynomial-like sequence a[i] = (i/n)^p, normalized."""
    x = np.linspace(0, 1, n)
    a = x ** p
    a[0] = max(a[0], 1e-6)  # avoid zero
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def beta_like_init(n, alpha, beta):
    """Generate beta-like distribution on discrete grid."""
    x = np.linspace(0.01, 0.99, n)
    a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = geometric_init(n_base, 1.005)
        best_score = evaluate_safe(best_seq)
        print(f"Default init (geometric): n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # ============ Phase 1: LP Nesterov on parent (main optimization) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={min(remaining*0.3, 480):.0f}s) ===")
    t1 = min(remaining * 0.3, 480.0)
    if t1 > 60:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp1_n{n_base}")

    # ============ Phase 2: Subgradient descent (new method) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Subgradient Descent n={n_cur} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"sgd_n{n_cur}")

    # ============ Phase 3: Variance smoothing ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Variance Smooth n={n_cur} (budget={min(remaining*0.1, 120):.0f}s) ===")
    t3 = min(remaining * 0.1, 120.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = variance_smooth(a_norm, t3, verbose=True)
        do_try_update(a_out, f"var_n{n_cur}")

    # ============ Phase 4: Peak shaving ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Peak Shave n={n_cur} (budget={min(remaining*0.1, 120):.0f}s) ===")
    t4 = min(remaining * 0.1, 120.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_local(a_norm, t4, seed=99, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # ============ Phase 5: Try longer length ============
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n2} (budget={min(remaining*0.15, 200):.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

        # Quick subgradient on new length
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.08, 80.0)
        if t5b > 20:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = subgradient_optimize(a_norm, t5b, verbose=False)
            do_try_update(a_out, f"sgd_n{n_cur}")

    # ============ Phase 6: Try different initialization at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        # Try a geometric init
        t6 = min(remaining * 0.12, 150.0)
        print(f"\n=== Phase 6: Structured init + LP n={n_base} (budget={t6:.0f}s) ===")
        if t6 > 30:
            # Geometric with slightly different ratio
            r_val = n_base ** (1.0 / (n_base - 1)) * 1.1
            init_geom = geometric_init(n_base, r_val)
            a_norm = init_geom * np.sqrt(2.0 * n_base) / init_geom.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6 * 0.5, solver, verbose=False)
            do_try_update(a_out, f"lp_geom_n{n_base}")

            remaining = time_budget - (time.monotonic() - t_global)
            t6b = min(remaining * 0.06, 60.0)
            if t6b > 15:
                # Beta-like init
                init_beta = beta_like_init(n_base, 0.5, 2.0)
                a_norm = init_beta * np.sqrt(2.0 * n_base) / init_beta.sum()
                a_out, _ = nesterov_lp_optimize(a_norm, t6b, solver, verbose=False)
                do_try_update(a_out, f"lp_beta_n{n_base}")

    # ============ Phase 7: Final LP polish on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} (budget={min(remaining*0.15, 150):.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # ============ Phase 8: Final subgradient + peak shave ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final SGD + PeakShave n={n_best} (budget={min(remaining*0.1, 100):.0f}s) ===")
    t8 = min(remaining * 0.06, 60.0)
    if t8 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t8, verbose=False)
        do_try_update(a_out, f"sgd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8b = min(remaining * 0.08, 80.0)
    if t8b > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave_local(a_norm, t8b, seed=555, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_best}")

    # ============ Phase 9: Final variance smooth ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.08, 60.0)
    if t9 > 10:
        print(f"\n=== Phase 9: Final VarSmooth n={n_best} (budget={t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = variance_smooth(a_norm, t9, verbose=False)
        do_try_update(a_out, f"var_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
