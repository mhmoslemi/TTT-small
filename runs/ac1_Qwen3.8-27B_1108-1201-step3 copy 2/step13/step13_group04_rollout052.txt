    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_bmax = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    global_it = 0
    stall_count = 0
    stall_threshold = best_bmax

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(80, int(remaining * 4.0 / max(1, n_stages - si)))

        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if global_it == 0:
                y_ext = y.copy()
            else:
                mu = (1.0 + np.sqrt(1.0 + 4.0 * global_it * global_it)) / 2.0
                beta = (mu - 1.0) / mu
                y_ext = y + beta * (y - y_prev)
                y_ext = np.maximum(y_ext, 0.0)
                s_ext = y_ext.sum()
                if s_ext > 1e-15:
                    y_ext = y_ext * norm / s_ext

            # Compute corrected gradient at extrapolated point
            grad = lse_gradient_corrected(y_ext, T)

            # Step with decreasing step size
            step_size = 1.5 / (global_it + 2.0)
            y_new = y - step_size * grad

            # Project: clip to non-negative, rescale to norm
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            global_it += 1

            # Check actual objective periodically
            if global_it % 30 == 0 or it == n_iters_stage - 1:
                bmax = fast_autocorr(y).max()
                if bmax < best_bmax:
                    best_bmax = bmax
                    best_a = y.copy()
                    stall_count = 0
                    stall_threshold = best_bmax
                else:
                    stall_count += 1

            # Adaptive: if stalled, increase temperature effect (already handled by stages)
            if verbose and (global_it % 300 == 0 or it == n_iters_stage - 1):
                elapsed = time.monotonic() - t0
                bmax = fast_autocorr(y).max()
                print(f"    LSE-N stage={si}/{n_stages} T={T:.3f} it={global_it} ({elapsed:.0f}s): "
                      f"cur={bmax:.8f} best={best_bmax:.8f} stall={stall_count}")

    return best_a, best_bmax


def mirror_descend_opt(a0, time_limit, T_start=10.0, T_end=0.3, n_stages=12, verbose=True):
    """Mirror descent (multiplicative weights) with corrected LSE gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_bmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    global_it = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(60, int(remaining * 3.0 / max(1, n_stages - si)))

        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient_corrected(a, T)

            # Multiplicative update: a <- a * exp(-eta * grad), then normalize
            eta = 2.0 / (T * (1.0 + 0.1 * np.log1p(global_it)))
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            global_it += 1

            if global_it % 40 == 0 or it == n_iters_stage - 1:
                bmax = fast_autocorr(a).max()
                if bmax < best_bmax:
                    best_bmax = bmax
                    best_a = a.copy()

            if verbose and global_it % 400 == 0:
                elapsed = time.monotonic() - t0
                bmax = fast_autocorr(a).max()
                print(f"    MirrorD stage={si}/{n_stages} T={T:.3f} ({elapsed:.0f}s): "
                      f"cur={bmax:.8f} best={best_bmax:.8f}")

    return best_a, best_bmax


def peak_shaving(a0, time_limit, verbose=True, seed=42):
    """
    Peak shaving: iteratively reduce the maximum autocorrelation peak
    by transferring mass from contributing positions to non-contributing ones.
    Uses exact subgradient at top-k peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_bmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top peaks
        top_k = 3
        peak_indices = np.argsort(b)[::-1][:top_k]
        max_b = b[peak_indices[0]]

        # Build mask of positions contributing to top peaks
        contrib_mask = np.zeros(n, dtype=bool)
        for k_star in peak_indices:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            contrib_mask[js:je + 1] = True

        improved = False

        # Strategy 1: Direct subgradient step with small step size
        grad = peak_subgradient(a, top_k=top_k)
        for step_frac in [0.005, 0.002, 0.001]:
            step = step_frac * norm
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_autocorr(a_trial).max()
            if score < best_bmax - 1e-10:
                a = a_trial
                best_bmax = score
                best_a = a.copy()
                improved = True
                it_count += 1
                break

        # Strategy 2: Targeted mass transfer from high-contributing to low-absence
        if not improved:
            # Find the most contributing pair to the main peak
            k_main = peak_indices[0]
            js = max(0, k_main - n + 1)
            je = min(n - 1, k_main)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_main - idx]
            if contrib.max() > 1e-20:
                src = idx[np.argmax(contrib)]
                # Find destination: positions not in any top-peak contribution
                non_contrib = np.where(~contrib_mask)[0]
                if len(non_contrib) > 0:
                    # Pick from among the smallest values at non-contributing positions
                    non_contrib = non_contrib[np.argsort(a[non_contrib])[:10]]
                    for dst in non_contrib[:5]:
                        for frac in [0.01, 0.005, 0.002, 0.001]:
                            amt = a[src] * frac
                            if amt < 1e-16:
                                continue
                            a_trial = a.copy()
                            a_trial[src] -= amt
                            a_trial[dst] += amt
                            a_trial = np.maximum(a_trial, 0.0)
                            s_t = a_trial.sum()
                            if s_t < 1e-15:
                                continue
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_bmax - 1e-10:
                                a = a_trial
                                best_bmax = score
                                best_a = a.copy()
                                improved = True
                                it_count += 1
                                break
                        if improved:
                            break

        # Strategy 3: Random perturbation when stuck
        if not improved:
            stall += 1
            if stall % 50 == 0 and stall < 500:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-6:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_bmax:
                            a = a_trial
                            best_bmax = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                if verbose:
                    print(f"    PeakShave: stalled at {best_bmax:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 200 == 0 and it_count > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={it_count} ({elapsed:.0f}s): best={best_bmax:.8f}")

    return best_a, best_bmax


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Asymmetric peak
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.8) ** 2)
    elif choice == 6:
        # Multi-frequency
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.4 + 0.3 * np.cos(3.0 * x) ** 2 + 0.2 * np.cos(5.0 * x + 1.0) ** 2
    else:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 10)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def nesterov_multistart(a0, time_limit, n_restarts=3, verbose=True):
    """
    Multi-restart Nesterov: try different temperature schedules and restarts.
    Returns the best found.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a * norm / s
    best_bmax = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Perturb the current best for diversity
        a_init = best_a.copy()
        if r > 0:
            # Random smooth perturbation
            noise = rng.normal(0, 0.1 * norm / n, n)
            a_init = best_a + noise
            a_init = np.maximum(a_init, 0.0)
            s_init = a_init.sum()
            if s_init > 1e-15:
                a_init = a_init * norm / s_init
            else:
                a_init = best_a.copy()

        t_per_restart = remaining * 0.8 / (n_restarts - r)
        a_out, bmax_out = nesterov_lse_opt(
            a_init, t_per_restart,
            T_start=15.0 + 5.0 * r,
            T_end=0.1,
            n_stages=12,
            verbose=verbose and r == 0
        )
        if bmax_out < best_bmax:
            best_bmax = bmax_out
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiRestart [{r}]: new best {best_bmax:.8f}")

    return best_a, best_bmax


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses corrected LSE gradient with Nesterov acceleration,
    mirror descent, peak shaving, and multi-length exploration.
    
    Key improvement over previous: CORRECTED gradient computation.
    The previous code used grad[m] = (a_rev*w)[m+n-1] + (a*w)[m], which is only
    correct for symmetric a. The correct gradient is 2*(a_rev*w)[m+n-1].
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: CORRECTED LSE Nesterov at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Corrected LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse_opt(a_work, t1, T_start=20.0, T_end=0.2,
                                     n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_corrected_n{n_base}")

    # ============ Phase 2: Mirror descent at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 2: MirrorDescent n={n_curr} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = mirror_descend_opt(a_work, t2, T_start=10.0, T_end=0.3,
                                       n_stages=10, verbose=True)
        do_try_update(a_out, f"mirror_n{n_curr}")

    # ============ Phase 3: Peak shaving at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: PeakShaving n={n_curr} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 240.0)
    if t3 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = peak_shaving(a_work, t3, verbose=True, seed=777)
        do_try_update(a_out, f"peak_n{n_curr}")

    # ============ Phase 4: Multi-restart LSE at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 4: MultiRestart LSE n={n_curr} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = nesterov_multistart(a_work, t4, n_restarts=3, verbose=True)
        do_try_update(a_out, f"multirestart_n{n_curr}")

    # ============ Phase 5: LSE at longer length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE-Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse_opt(a_work, t5, T_start=20.0, T_end=0.3,
                                     n_stages=14, verbose=True)
        do_try_update(a_out, f"lse_n{n2}")

    # ============ Phase 6: LSE at even longer length ============
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LSE-Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_work = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lse_opt(a_work, t6, T_start=25.0, T_end=0.5,
                                     n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_n{n3}")

    # ============ Phase 7: Peak shaving on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: PeakShaving final n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.18, 200.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving(a_work, t7, verbose=True, seed=99999)
        do_try_update(a_out, f"peak_final_n{n_best}")

    # ============ Phase 8: Final LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 15:
            print(f"\n=== Phase 8: LSE polish n={n_best} (budget={t8:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lse_opt(a_work, t8, T_start=3.0, T_end=0.05,
                                         n_stages=10, verbose=True)
            do_try_update(a_out, "lse_polish")

    # ============ Phase 9: Final peak shaving ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t9 = min(remaining * 0.7, 100.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final peak shaving n={n_best} (budget={t9:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_shaving(a_work, t9, verbose=True, seed=31415)
            do_try_update(a_out, "peak_ultimate")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
