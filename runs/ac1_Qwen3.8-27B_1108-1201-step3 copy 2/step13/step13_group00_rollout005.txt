            A_mat = T[active_set, :]
        constraints = [A_mat @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_full(a, M, solver):
    return lp_best_response(a, M, solver, screened=False)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    no_improve_count = 0
    adaptive_eps = 0.0

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive M: if stuck, push M lower
        M_target = M_curr * (1.0 - adaptive_eps)
        b = lp_best_response(y, M_target, solver, screened=True, tolerance=0.02)

        if b is None or b.sum() < 1e-15:
            b = lp_best_response_full(y, M_curr, solver)
            if b is None:
                adaptive_eps = min(adaptive_eps + 0.005, 0.05)
                continue

        b_sum = b.sum()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterow accelerated step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Check improvement
        score_y = fast_autocorr(y).max()
        improved = False
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            improved = True

        score_a = fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            improved = True

        if improved:
            no_improve_count = 0
            adaptive_eps = max(0.0, adaptive_eps - 0.002)
        else:
            no_improve_count += 1
            if no_improve_count % 10 == 0:
                adaptive_eps = min(adaptive_eps + 0.003, 0.05)

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score - 1e-10:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, eps={adaptive_eps:.4f}")

    return best_a, best_score


def fast_pair_cd(a, time_limit, rng, verbose=True):
    """
    Fast pair coordinate descent.
    For each i, try K random j's. Compute gain vector in O(n),
    find optimal delta via a few evaluations.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    K = 8  # random j's per i per pass

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            ai = a[i]
            # Compute gain vector: g[p] = 2*(a[p-j] - a[p-i]) for valid indices
            # Precompute the "shifted a" arrays for position i
            # shifted_i[p] = a[p-i] if 0<=p-i<n else 0
            # We'll compute it on the fly

            for attempt in range(K):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]

                # Gain vector: g[p] = 2*(a[p-j] - a[p-i]) for p in [0, size-1]
                # a[p-i] is valid for p in [i, i+n-1]
                # a[p-j] is valid for p in [j, j+n-1]
                g = np.zeros(size)
                lo_i = i
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    g[lo_i:hi_i] -= a[:hi_i - lo_i]  # a[p-i] for p in [i, i+n-1]
                lo_j = j
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    g[lo_j:hi_j] += a[:hi_j - lo_j]  # a[p-j] for p in [j, j+n-1]
                g *= 2.0

                # Also need quadratic corrections at p=2i and p=2j
                # At p=2i: (a[i]-d)^2 vs a[i]^2 -> change = -2*a[i]*d + d^2
                # At p=2j: (a[j]+d)^2 vs a[j]^2 -> change = 2*a[j]*d + d^2

                # Find best delta in [0, ai]
                # Try a few values
                candidates = np.array([0.0, ai * 0.1, ai * 0.25, ai * 0.5, ai * 0.75, ai * 0.9, ai])
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0:
                        continue
                    new_conv_est = conv + d * g
                    # Quadratic corrections
                    if 2 * i < size:
                        new_conv_est[2 * i] = conv[2 * i] - 2.0 * ai * d + d * d
                    if 2 * j < size:
                        new_conv_est[2 * j] = conv[2 * j] + 2.0 * aj * d + d * d
                    # Also: the linear term already includes the 2*a[p-i]*d part
                    # But at p=2i, linear gives 2*a[i]*d and quadratic adds d^2
                    # So total at 2i: conv[2i] + 2*a[i]*d + (-2*a[i]*d + d^2) = conv[2i] + d^2
                    # Wait that's wrong. Let me recheck.
                    # Original conv[2i] includes a[i]^2 (as part of the sum)
                    # After: (a[i]-d)^2 at position 2i from the "self" term
                    # Change in the self term: -2*a[i]*d + d^2
                    # Change from cross terms: 2*(a[2i-i-j]... this gets complex
                    # The gain vector already accounts for the linear change in cross terms.
                    # The only additional term is the change in the self-convolution term.
                    # At p=2i: the self term changes from a[i]^2 to (a[i]-d)^2, delta = -2*a[i]*d + d^2
                    # But the gain vector at p=2i is 2*(a[2i-j] - a[2i-i]) = 2*(a[2i-j] - a[i])
                    # The linear part of the change at p=2i from the gain is d*2*(a[2i-j]-a[i])
                    # The actual change is: -d*2*a[i] (from reducing a[i] in cross terms with a[2i-i]=a[i])
                    #   + d*2*a[2i-j] (from increasing a[j] ... wait no)
                    
                    # Let me just use the full recomputation for accuracy:
                    # new_a = a.copy(); new_a[i] -= d; new_a[j] += d
                    # But that's O(n) per evaluation which defeats the purpose.
                    # For now, trust the first-order + diagonal correction:
                    
                    new_max = new_conv_est.max()
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        s = a.sum()
                        if s > 1e-15:
                            a = a * norm / s
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a, time_limit, verbose=True):
    """
    Direct peak shaving: identify top peaks in autocorrelation,
    compute subgradient, and take a projected step to reduce them.
    """
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_peaks = 20  # number of peaks to target

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        max_conv = conv.max()

        # Find top peaks
        peak_threshold = max_conv * 0.95
        peak_indices = np.where(conv >= peak_threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        peak_indices = peak_indices[:n_peaks]

        # Compute combined subgradient
        # For peak at position p: subgrad_i = 2*a[p-i] if 0<=p-i<n else 0
        grad = np.zeros(n)
        for p in peak_indices:
            # a[p-i] valid for i in [max(0, p-n+1), min(n-1, p)]
            i_lo = max(0, p - n + 1)
            i_hi = min(n, p + 1)
            if i_hi > i_lo:
                # a[p-i] for i in [i_lo, i_hi-1]
                indices = np.arange(i_lo, i_hi)
                vals = a[p - indices]
                valid = (p - indices >= 0) & (p - indices < n)
                grad[i_lo:i_hi] += 2.0 * np.where(valid, vals, 0.0)

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Projected gradient descent step
        # Learn rate: adaptive
        step_size = 0.1 * norm / np.sqrt(n)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * norm / s

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            step_size *= 0.5
            if step_size < norm * 1e-8:
                break
            a_new = a - step_size * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new * norm / s
            score_new = fast_autocorr(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
            else:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_smooth_optimize(a0, time_limit, verbose=True):
    """LSE gradient descent with FFT-accelerated gradient computation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    temps = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(5, min(50, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Gradient via FFT: grad_i = 2 * sum_p w_p * a[p-i] (valid)
            # = 2 * (w * a_rev)[i] projected... 
            # Actually: sum_p w_p * a[p-i] for valid p,i
            # This is the cross-correlation of w and a at lag i
            # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives circular cross-correlation
            W_fft = rfft(w, fft_size)
            A_fft = rfft(a, fft_size)
            # cross_corr[m] = sum_k w[k] * a[k+m] (circular)
            cross = irfft(W_fft * np.conj(A_fft), fft_size)
            # We want sum_p w[p] * a[p-i] = sum_p w[p] * a[p-i]
            # Let q = p-i: sum_q w[q+i] * a[q] = cross[i] (approximately, for linear)
            grad = 2.0 * cross[:n].copy()
            # Also need the "other" part: sum_p w[p] * a[i-p]
            # = sum_q w[i-q] * a[q] = cross at negative lag
            cross2 = irfft(A_fft * np.conj(W_fft), fft_size)
            # cross2[m] = sum_k a[k]*w[k+m], so cross2[fft_size-m] = sum_k a[k]*w[k-m+fft_size]
            # We want sum_p w[p]*a[i-p] = sum_q a[q]*w[i-q]
            # Let m = i-q: sum_q a[q]*w[m] where m = i-q... this is conv(a,w)[i]
            # Actually simpler: sum_p w[p]*a[i-p] = (w*a)[i] but with flipped a
            # = conv(w, a_flip)[n-1-i]... 
            # Let's just use: the autocorrelation gradient is symmetric
            # grad_i = 2 * (sum_p w_p * a_{p-i} + sum_p w_p * a_{i-p})
            # The second term: sum_p w_p * a_{i-p}. Let j = i-p: sum_j w_{i-j} * a_j
            # = (w_rev * a)[i] where w_rev is reversed w... 
            # For the FFT approach: this equals cross at negative lag
            # cross_neg[i] = sum_k w[k] * a[k-i + n] (circular)
            # For i < n and fft_size > 2n, cross[fft_size - i] should give us the negative lag
            if fft_size > n:
                grad += 2.0 * cross[fft_size - n + 1:fft_size + 1][:n] if fft_size >= n else np.zeros(n)

            # Projected gradient step
            step = norm / (T * np.sqrt(n)) * 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.sin(4 * np.pi * x) * np.sin(np.pi * x)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.6 * np.exp(-2.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_and_refine(a, rng, time_limit, solver, verbose=False):
    """Perturb the sequence and refine - helps escape local minima."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    # Add structured perturbation
    sigma = 0.05
    noise = rng.normal(0, sigma, n)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a, fast_autocorr(a).max()
    a_pert = a_pert * norm / s

    # Quick LP refinement
    t_lp = min(time_limit * 0.4, 60.0)
    a_out, score = nesterov_lp_optimize(a_pert, t_lp, solver, verbose=False)
    
    # Quick PCD polish
    t_pcd = min(time_limit * 0.6, 60.0)
    a_out2, score2 = fast_pair_cd(a_out, t_pcd, rng, verbose=False)
    if score2 < score:
        a_out = a_out2
        score = score2

    return a_out, score


def subgrad_fw(a0, time_limit, verbose=False):
    """Frank-Wolfe with subgradient - fast, no LP needed."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient at argmax
        grad = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        if i_hi > i_lo:
            indices = np.arange(i_lo, i_hi)
            vals = y[k_star - indices]
            valid = (k_star - indices >= 0) & (k_star - indices < n)
            grad[i_lo:i_hi] = 2.0 * np.where(valid, vals, 0.0)

        # FW: go to vertex with minimum gradient
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length (main workhorse, most time)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.38, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Fast PCD (key polish step)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Fast PCD n={n_best} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak Shaving n={n_best} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 80.0)
    if t3 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving(a_norm, t3, verbose=True)
        do_try_update(a_out, f"pkshave_n{n_best}")

    # Phase 4: Try longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

        # PCD on longer
        remaining = time_budget - (time.monotonic() - t_global)
        n_best = len(best_seq)
        t4b = min(remaining * 0.10, 100.0)
        if t4b > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = fast_pair_cd(a_norm, t4b, rng, verbose=True)
            do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: LSE smooth optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE n={n_best} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_smooth_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 6: Perturb and refine (escape local minima)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Perturb+Refine n={n_best} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = perturb_and_refine(a_norm, rng, t6, solver, verbose=False)
        do_try_update(a_out, f"perturb_n{n_best}")

    # Phase 7: FW subgradient (fast, many iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: FW subgrad n={n_best} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_fw(a_norm, t7, verbose=False)
        do_try_update(a_out, f"fw_n{n_best}")

    # Phase 8: Final PCD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final PCD n={n_best} (remaining={remaining:.0f}s) ===")
    t8 = max(remaining * 0.20, 20.0)
    if t8 > 10 and remaining > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_norm, t8, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final: one more peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_final = max(remaining * 0.3, 5.0)
    if t_final > 3 and remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shaving(a_norm, t_final, verbose=False)
        do_try_update(a_out, "pkshave_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
