2. **The LP Nesterov approach** is the most powerful but slow per iteration. I'll improve it with better warm-starting and constraint reduction.
3. **Key insight**: After the main optimization, the peak is concentrated at a few positions. A targeted coordinate descent that specifically attacks these peaks can squeeze out improvements.
4. **Subgradient methods** with Polyak step sizes provide a good complementary approach.

Let me implement an improved version with:
- Warm-started LP with constraint reduction
- Subgradient polishing with Polyak step
- Peak-targeted coordinate descent (more aggressive than before)
- Adam-based LSE gradient descent
- Better multi-length exploration with interpolation

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Warm-started LP Nesterov with constraint reduction and adaptive restarts
2. Subgradient method with Polyak step size for polishing
3. Peak-targeted coordinate descent (aggressive multi-peak reduction)
4. Adam-based LSE smooth gradient descent

import numpy as np

import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response_warm(a, M, solver, warm_start=None):
    """LP best response with warm-starting and constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    # Phase 1: Solve with all constraints
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        if warm_start is not None:
            b_var.value = warm_start
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None

    # Phase 2: Iteratively reduce to active constraints
    for _ in range(5):
        circ = T @ b_val
        tol = 1e-7 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 10:
            break
        if n_active < 10:
            break

        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            b_var.value = b_val
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break

    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_iter=500000):
    """LP Nesterov acceleration with warm-starting and adaptive restart."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    ema_a = np.zeros(n)
    n_ema = 0
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    warm_b = None
    best_ever = best_score
    restart_count = 0
    max_restarts = 10

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Use warm-started LP
        b = lp_best_response_warm(y, M_curr, solver, warm_start=warm_b)
        warm_b = b  # save for next iteration's warm start

        if b is None:
            stall += 1
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            if score_y < best_ever:
                best_ever = score_y

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            if score_a < best_ever:
                best_ever = score_a

        # EMA tracking
        decay = 0.99
        ema_a = decay * ema_a + (1 - decay) * a
        n_ema += 1
        if n_ema % 50 == 0:
            s_ema = ema_a.sum()
            if s_ema > 1e-15:
                ema_check = ema_a * norm / s_ema
                score_ema = fast_autocorr(ema_check).max()
                if score_ema < best_score:
                    best_score = score_ema
                    best_a = ema_check.copy()
                    stall = 0
                    if score_ema < best_ever:
                        best_ever = score_ema

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}, restarts={restart_count}")

        # Adaptive restart
        if stall > 300 and restart_count < max_restarts:
            rng = np.random.default_rng(n_iters * 13 + 7 + restart_count)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            warm_b = None
            stall = 0
            restart_count += 1
            if verbose:
                print(f"    [Restart #{restart_count} at iter {n_iters}]")

        if stall > 300 and restart_count >= max_restarts:
            # Try a different perturbation scale
            rng = np.random.default_rng(n_iters * 7)
            idx = rng.choice(n, size=max(3, n // 20), replace=False)
            y_temp = y.copy()
            for i in idx:
                y_temp[i] *= rng.uniform(0.5, 1.5)
            y_temp = np.maximum(y_temp, 0)
            s_t = y_temp.sum()
            if s_t > 1e-15:
                y = y_temp * norm / s_t
            a_temp = a.copy()
            for i in idx:
                a_temp[i] *= rng.uniform(0.5, 1.5)
            a_temp = np.maximum(a_temp, 0)
            s_t = a_temp.sum()
            if s_t > 1e-15:
                a = a_temp * norm / s_t
            warm_b = None
            stall = 0

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True, max_iter=100000):
    """Subgradient method with Polyak step for polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    lower_bound = 2.0 * n / (2 * n - 1)  # theoretical lower bound

    t0 = time.monotonic()
    steps = 0
    momentum_vec = np.zeros(n)
    beta = 0.9

    for step in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        curr_score = b.max()

        if curr_score <= lower_bound * 1.001:
            break

        # Get top peaks for subgradient
        n_peaks = min(5, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:n_peaks]

        # Compute subgradient (average over top peaks)
        grad = np.zeros(n)
        for k in top_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j] / n_peaks

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        # Polyak step size with smoothing
        gap = curr_score - lower_bound
        eta = gap / (grad_norm_sq + 1e-10)
        # Cap the step to avoid instability
        eta = min(eta, 2.0 / (np.linalg.norm(a) + 1e-10))

        # Momentum update
        momentum_vec = beta * momentum_vec + (1 - beta) * grad
        a_new = a - eta * momentum_vec
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Aggressive coordinate descent targeting the peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_trials = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        curr_max = b.max()

        # Identify top peak positions
        n_top = min(10, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:n_top]

        improved = False
        for k in top_k:
            # Determine which indices contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            # Contribution of each pair
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue

            # Identify the element with max contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src

            # Try reducing src and adding to a low-contribution position
            # Find positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]

            # Try multiple move sizes
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                # Try moving to different destinations
                # First: to the lowest-valued non-peak position
                sorted_nonpeak = non_peak[np.argsort(a[non_peak])]
                for dst in sorted_nonpeak[:min(5, len(sorted_nonpeak))]:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t

                    score = fast_autocorr(a_trial).max()
                    n_trials += 1

                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                        improved = True
                        break
                if improved:
                    break

            # Also try: split src between multiple destinations
            if not improved:
                n_dst = min(3, len(non_peak))
                if n_dst >= 2:
                    sorted_np = non_peak[np.argsort(a[non_peak])]
                    dsts = sorted_np[:n_dst]
                    frac = 0.1
                    amt = a[src] * frac
                    weights = 1.0 / (a[dsts] + 1e-10)
                    weights /= weights.sum()
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    for di, dst in enumerate(dsts):
                        a_trial[dst] += amt * weights[di]
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        n_trials += 1
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                            improved = True

        if not improved and n_trials % 100 == 0:
            # Random perturbation to escape local minima
            n_perturb = max(2, n // 100)
            idx_pert = rng.choice(n, size=n_perturb, replace=False)
            a_trial = a.copy()
            for idx in idx_perturb:
                if a[idx] > norm * 1e-8:
                    j = rng.integers(n)
                    if j != idx:
                        amt = a[idx] * rng.uniform(0.05, 0.2)
                        a_trial[idx] -= amt
                        a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                # Accept with simulated annealing criterion
                T = max(0.01, (best_score - 1.0) * 0.1 * (1.0 - (time.monotonic() - t0) / time_limit))
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_adam_gd(a0, time_limit, verbose=True):
    """LSE gradient descent with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    # Learning rate schedule
    lr_start = 0.5
    lr_end = 0.01

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Temperature annealing
        T = max(max_b * 0.01, 0.001) * (1.0 + max_b * 0.5 * (1.0 - progress))

        # Softmax weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute gradient using FFT
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def dual_polish(a0, time_limit, verbose=True):
    """Polish using the dual problem interpretation.
    The dual of the LP best response gives a distribution y over peak positions.
    We use this as a "guiding" distribution for coordinate descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        # Top peaks with weights
        n_top = min(15, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:n_top]
        # Weight by how close to max
        weights = np.exp(-(b[top_k] - max_b) / max(max_b * 0.1, 1e-10))
        weights /= weights.sum()

        # For each peak, try to reduce it by adjusting contributing elements
        for k, w_k in zip(top_k, weights):
            if w_k < 0.01:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue

            # Find the element whose reduction most helps
            # Reducing a[i] by delta reduces b_k by 2*delta*a[k-i] (approx)
            # But it also affects other b_k values
            src = idx[np.argmax(contrib)]

            # Try small reductions
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                # Distribute to positions that minimize impact on other peaks
                # Score each candidate by how much it increases other peaks
                candidates = np.where(np.arange(n) != src)[0]
                # Prefer low-value positions far from current peaks
                candidate_scores = np.zeros(len(candidates))
                for ci, c in enumerate(candidates):
                    # How much would adding to position c increase the max?
                    # Approximate: sum over top peaks of a[c] * a[k-c]
                    inc = 0.0
                    for k2 in top_k[:5]:
                        j2 = k2 - c
                        if 0 <= j2 < n:
                            inc += a[j2]
                    candidate_scores[ci] = a[c] + inc * 0.1

                best_cand = candidates[np.argmin(candidate_scores)]
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_cand] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()
                n_iter += 1

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break

        if verbose and n_iter % 2000 == 0 and n_iter > 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish iter={n_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Parent load error: {e}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return False
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (most time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Subgradient polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SubGradient n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.08, 100.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak CoordDescent n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t3, seed=123, verbose=True)
        do_try_update(a_out, f"coord_n{n_cur}")

    # === Phase 4: LSE Adam GD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-Adam n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam_gd(a_norm, t4, verbose=True)
        do_try_update(a_out, f"lseadam_n{n_cur}")

    # === Phase 5: Dual polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: DualPolish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = dual_polish(a_norm, t5, verbose=True)
        do_try_update(a_out, f"dual_n{n_cur}")

    # === Phase 6: Try longer length ===
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.22, 300.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

        # Quick polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.06, 60.0)
        if t6b > 10:
            n_best_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best_cur) / best_seq.sum()
            a_out, _ = peak_coordinate_descent(a_norm, t6b, seed=456, verbose=False)
            do_try_update(a_out, f"coord2_n{n_best_cur}")

    # === Phase 7: Try even longer ===
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.18, 220.0)
    if t7 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # === Phase 8: Final combined polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Polish n={n_best} (budget={remaining:.0f}s) ===")

    # 8a: LP Nesterov
    t8a = min(remaining * 0.35, 150.0)
    if t8a > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8a, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # 8b: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8b = min(remaining * 0.35, 120.0)
    if t8b > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t8b, seed=789, verbose=True)
        do_try_update(a_out, f"coord_final_n{n_best}")

    # 8c: Subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8c = min(remaining * 0.30, 80.0)
    if t8c > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_optimize(a_norm, t8c, verbose=False)
        do_try_update(a_out, f"subgrad_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
