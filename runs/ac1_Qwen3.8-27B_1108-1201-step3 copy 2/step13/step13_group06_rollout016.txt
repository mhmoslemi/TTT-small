    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute FFT size for cross-correlation
    cross_fft_size = 1 << (3 * n - 2).bit_length()
    
    def fun_and_grad(z):
        a_local = np.empty(n)
        a_local[:n-1] = z
        a_local[n-1] = 1.0 - z.sum()
        a_local = np.maximum(a_local, 1e-15)
        a_local = a_local / a_local.sum()
        
        c = fast_autocorr(a_local)
        max_c = c.max()
        shifted = c / T - max_c / T
        exp_vals = np.exp(shifted)
        Z = exp_vals.sum()
        obj = 2.0 * n * (max_c + T * np.log(Z))
        
        w = exp_vals / Z
        W = rfft(w, cross_fft_size)
        A = rfft(a_local, cross_fft_size)
        cross = irfft(W * np.conj(A), cross_fft_size)
        grad_a = (2.0 / T) * cross[:n]
        
        grad_z = grad_a[:n-1] - grad_a[n-1]
        return obj, grad_z
    
    current_a = a.copy()
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Initial point: current solution
        z0 = current_a[:n-1].copy()
        
        result = minimize(
            fun_and_grad, z0, method='L-BFGS-B',
            jac=True,
            options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 1000}
        )
        
        if result.success or result.fun < best_score / (2.0 * n):
            z_opt = result.x
            a_new = np.empty(n)
            a_new[:n-1] = z_opt
            a_new[n-1] = 1.0 - z_opt.sum()
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    current_a = a_new.copy()
        
        elapsed = time.monotonic() - t0
        print(f"      L-BFGS T={T}: best={best_score:.6f} ({elapsed:.0f}s)")
    
    return best_a


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Build Toeplitz matrix
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j+n, j] = np.ones(n)  # placeholder, will rebuild
    # Actually, T_mat depends on y, so I need to rebuild each iteration
    # This is O(n^2) per iteration which is 2.5M for n=1600 - too slow for many iterations
    
    # Alternative: use a different approach
    # The LP is: min M s.t. (y*b)[k] <= M for all k, sum(b)=1, b>=0
    # Where y is fixed. This can be written as:
    # min M s.t. T_y @ b <= M * 1, sum(b)=1, b>=0
    # where T_y is the Toeplitz matrix defined by y
    
    avg_long = np.zeros(n)
    n_long = 0
    stall_count = 0
    last_print = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Build Toeplitz for current y
        T_y = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_y[j:j+n, j] = y
        
        # Solve LP: min M s.t. T_y @ b <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_y @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_star = np.asarray(b_var.value, dtype=np.float64)
        b_star = np.maximum(b_star, 0)
        b_star = b_star / b_star.sum()
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        
        y = y_new
        a = a_new
        
        avg_long += a
        n_long += 1
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if n_long % 20 == 0 and n_long >= 40:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            stall_count = 0
        
        if t - last_print > 500:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
            last_print = t
    
    return best_a, best_score


def peak_shaving_refine(a0, time_limit, seed=42):
    """Fine-grained peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]
            
            for src in top_contrib_idx:
                # Find safe destinations (away from all top peaks)
                danger = np.zeros(n, dtype=bool)
                for pk in top_peaks[:10]:
                    djs = max(0, pk - n + 1)
                    dje = min(n - 1, pk)
                    danger[djs:dje+1] = True
                danger[src] = True
                
                safe = np.where(~danger)[0]
                if len(safe) == 0:
                    safe = np.where(np.arange(n) != src)[0]
                safe_sorted = safe[np.argsort(a[safe])]
                
                for dst in safe_sorted[:5]:
                    for frac in [0.003, 0.008, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-10):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 100:
                break
    
    return best_a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
