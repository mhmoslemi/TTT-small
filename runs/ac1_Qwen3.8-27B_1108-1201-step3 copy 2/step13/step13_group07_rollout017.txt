from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    result = result / result.sum() * target_sum
    return result

def lse_grad(a, T):
    """Compute gradient of LSE-smoothed max of autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i} for valid k-i
    # This is 2 * (w * a_reversed) truncated

    A_rev = rfft(a[::-1], fft_size)

    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n-1:n-1+n]
    return grad, b.max()

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return float('inf')
    return 2.0 * n * float(fast_autocorr(a).max())


def evaluate_safe(arr):
    """Safely evaluate a sequence using the external evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using exact algorithm."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result * (target_sum / s)
    else:
        result = np.full(n, target_sum / n)
    return result


def compute_lse_grad(a, T):
    """Compute LSE-smoothed gradient of max_k (a*a)_k w.r.t. a.
    Returns (gradient, weights, autocorrelation)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i} for valid (k-i) in [0,n-1]
    # Computed as 2*(w * flip(a))_{n-1+i}
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad, w, b


def compute_topk_grad(a, K=10):

    n = len(a)
    b = fast_autocorr(a)

def spectral_shaping(a, n_iter=200, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        if non_dc.max() < 1e-30:
            break
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def coord_descent_search(a0, time_limit, seed=42, verbose=True):
    """Systematic coordinate descent: transfer mass from high-gradient to low-gradient coords."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        grad, w, b = compute_lse_grad(a, T=50.0)

        # Find source (high grad) and target (low grad) coordinates
        # Exclude coordinates that are already near zero as targets
        grad_sorted = np.sort(grad)[::-1]
        # Top 20% as potential sources, bottom 50% as potential targets
        src_threshold = grad_sorted[max(0, n // 5 - 1)]
        dst_threshold = grad_sorted[min(n - 1, 3 * n // 4)]

        src_candidates = np.where((grad > src_threshold) & (a > 1e-6))[0]
        dst_candidates = np.where((grad < dst_threshold) & (a < np.median(a)))[0]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            # Fallback: use global max/min
            src = int(np.argmax(grad * a))  # weighted
            dst = int(np.argmin(grad))
            src_candidates = np.array([src])
            dst_candidates = np.array([dst])

        improved = False
        for _ in range(min(20, len(src_candidates))):
            if time.monotonic() - t0 > time_limit:
                break
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            if src == dst:
                continue

            # Try different transfer fractions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Try random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 300:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 50 < 1:
                print(f"          CoordDescent ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score

        s = p.sum()
        if s > 1e-10:
            current = p / s
        else:
            current = None
    else:
        current = None

    for size in sizes:
        if current is not None and len(current) != size:
            # Interpolate
            x_old = np.linspace(0, 1, len(current))
            x_new = np.linspace(0, 1, size)
            current = np.interp(x_new, x_old, current)
            current = np.maximum(current, 0)
            s = current.sum()
            if s > 1e-15:
                current = current / s

        if current is None:
            current = np.ones(size) / size

        # Quick spectral shaping
        current = spectral_shaping(current, n_iter=100, alpha=0.3)

        # Quick Adam optimization
        remaining_per_stage = 15.0
        current, _ = adam_lse_search(current, remaining_per_stage,
                                      T_start=50.0, T_end=0.1,
                                      n_stages=10, base_lr=0.005, verbose=False)

    # Final interpolation to target_n
    if len(current) != target_n:
        x_old = np.linspace(0, 1, len(current))
        x_new = np.linspace(0, 1, target_n)
        current = np.interp(x_new, x_old, current)
        current = np.maximum(current, 0)
        s = current.sum()
        if s > 1e-15:
            current = current / s

    return current


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def frequency_domain_optimize(a0, time_limit, n_iter=200, verbose=True):
    """Optimize in frequency domain: adjust magnitudes to flatten spectrum while keeping time-domain non-negative."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    for i in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        # Compute current autocorrelation
        b = fast_autocorr(a)
        # Find peaks
        peak_vals = b[b > b.mean()]
        if len(peak_vals) == 0:
            break

        # Target: reduce peaks by scaling down frequency bins that contribute most
        # The contribution of bin k to peak m is |A_k|^2 * exp(2*pi*i*k*m/fft_size)
        # Simplified: scale bins proportionally to their contribution to the max peak
        max_k = b.argmax()
        A_abs = np.abs(A)
        # Phase contribution to peak
        phases = np.angle(A)
        contributions = A_abs * np.cos(2 * np.pi * max_k * np.arange(len(A)) / fft_size)
        contributions = np.maximum(contributions, 0)
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        # Scale: reduce bins with high contribution, increase those with low
        target_ratio = np.ones(len(A))
        for k in range(1, len(A)):
            if contributions[k] > total_contrib / len(A) * 2:
                target_ratio[k] = 0.7
            elif contributions[k] < total_contrib / len(A) * 0.5:
                target_ratio[k] = 1.1

        # Apply smooth scaling
        A_new = A.copy()
        for k in range(1, len(A_new)):
            A_new[k] = A_new[k] * (1 + 0.1 * (target_ratio[k] - 1))

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and i % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"          FreqOpt {i}/{n_iter} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Phase 0: Coarse-to-fine multi-resolution initialization
    Phase 1: Spectral shaping
    Phase 2: Adam LSE with cosine annealing (main optimizer, ~50% time)
    Phase 3: Coordinate descent peak targeting (~15% time)
    Phase 4: Frequency-domain optimization (~10% time)
    Phase 5: Try longer n with interpolation (~15% time)
    Phase 6: Final polish (~10% time)
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"Time budget: {time_budget:.0f}s")

    # Phase 0: Multi-resolution initialization (try to find a better starting point)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 0: Multi-res init n={n_base} (remaining={remaining:.0f}s) ===")
    t0_phase = min(remaining * 0.05, 60.0)
    if t0_phase > 10:
        init_mr = multi_res_init(best_seq, n_base, seed=seed)
        sc = internal_score(init_mr)
        if sc < best_score:
            best_score = sc
            best_seq = init_mr.copy()
            print(f"  Multi-res init improved: {best_score:.6f}")
        else:
            print(f"  Multi-res init: {sc:.6f} (no improvement over {best_score:.6f})")

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.2, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After Phase 1: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 2: Adam LSE (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Adam LSE n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 30:
        a_out, sc_out = adam_lse_search(
            best_seq, t2,
            T_start=300.0, T_end=0.01,
            n_stages=35, base_lr=0.012,
            verbose=True
        )
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Adam_LSE]: {best_score:.6f}")
        else:
            print(f"  Adam LSE: {sc_out:.6f} (best overall still {best_score:.6f})")
    print(f"  After Phase 2: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Coord descent n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_out, sc_out = coord_descent_search(best_seq, t3, seed=seed + 1)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [CoordDesc]: {best_score:.6f}")
        else:

    # Phase 4: Frequency domain optimization
    remaining = time_budget - (time.monotonic() - t_global_start)

    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.25)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  *** NEW BEST [Spectral_n{n_long}]: {best_score:.6f}")

        # Adam LSE at longer length
        t5_adam = t5 * 0.7
        a_out, sc_out = adam_lse_search(
            interpolate_sequence(best_seq, n_long), t5_adam,

        sc = evaluate_safe(a_out)

    print(f"  After Phase 5: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 6: Final polish at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:

        t6a = t6 * 0.5
        a_out, sc_out = adam_lse_search(

            T_start=20.0, T_end=0.005,

        if np.isfinite(sc) and sc < best_score:

            best_seq = np.maximum(a_out, 0)

            print(f"  *** NEW BEST [Polish_LSE]: {best_score:.6f}")

        # Coordinate descent final
        t6b = t6 * 0.5
        a_out, sc_out = coord_descent_search(best_seq, t6b, seed=seed + 99)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Polish_Coord]: {best_score:.6f}")

    # Phase 7: Try a completely different initialization (random)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"\n=== Phase 7: Random restart n={n_base} (remaining={remaining:.0f}s) ===")
        t7 = min(remaining * 0.4, 200.0)
        # Random initial sequence
        a_rand = rng.dirichlet(np.ones(n_base))
        a_rand = spectral_shaping(a_rand, n_iter=100, alpha=0.3)

        print(f"  Random init score: {sc:.6f}")
        if sc < best_score * 1.05:  # Only worth pursuing if competitive

                a_rand, t7,
