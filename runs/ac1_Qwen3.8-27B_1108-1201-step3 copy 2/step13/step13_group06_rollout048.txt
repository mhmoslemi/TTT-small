    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # First, try simple clipping
    v_pos = np.maximum(v, 0)
    s = v_pos.sum()
    if s < 1e-15:

    # Scale
    v_scaled = v_pos * (target_sum / s)

def spectral_shaping_step(a, alpha=0.1, p=0.5):
    """
    Apply spectral shaping: compress the power spectrum to reduce autocorrelation peaks.
    
    a: input sequence (will be normalized)
    alpha: mixing parameter
    p: power for compression (p < 1 compresses, p > 1 expands)
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a = a / s
    
    # Pad to 2n for linear convolution
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    P = A.real ** 2 + A.imag ** 2  # |A|^2
    
    # Compress the spectrum: P_new = P^(p) / sum(P^(p)) * sum(P)
    # This flattens the spectrum
    P_compressed = P ** p
    P_compressed = P_compressed / P_compressed.sum() * P.sum()
    
    # Need to construct new A with the same phase but compressed magnitude
    mag_old = np.sqrt(P)
    mag_new = np.sqrt(P_compressed)
    if (mag_old > 1e-15).any():

    else:
        phase = 1.0

    a_new = irfft(A_new, fft_size)[:n]
    
    # Mix with original
    a_mix = (1 - alpha) * a + alpha * a_new
    
    # Project to simplex
    a_mix = project_simplex(a_mix)
    return a_mix

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cosine-basis parameterized optimization (dimensionality reduction)
2. Ultra-fine pair-based peak shaving with adaptive steps
3. Short active-set LP refinement (Nesterov with exact best responses)
4. Multi-length exploration via interpolation
5. Random restarts with simulated annealing
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_fast(a):
    """Fast local evaluation (no security checks)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))

def norm_simplex(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interp_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm_simplex(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    return norm_simplex(a_new)
