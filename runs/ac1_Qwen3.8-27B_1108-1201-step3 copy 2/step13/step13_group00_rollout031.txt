        return float('inf')


def norm_sequence(a, target_sum):
    """Normalize sequence to have the given sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def lp_best_response(a, M, max_constraints=300, solver_method='highs'):
    """
    Solve LP: max sum(b) s.t. cross_conv(a,b) <= M, b >= 0.
    Uses constraint screening to limit the number of constraints.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Compute current cross-convolutions to determine active set
    ones_b = np.ones(n)
    conv_ones = cross_conv(a, ones_b)
    
    # Select top constraints (most restrictive)
    if size > max_constraints:
        # Use the constraints where conv is largest relative to M
        ratios = conv_ones / max(M, 1e-15)
        # Include constraints with ratio > 0.5, up to max_constraints
        active_full = np.where(ratios > 0.3)[0]
        if len(active_full) > max_constraints:
            # Take the top max_constraints
            top_idx = np.argsort(ratios[active_full])[::-1][:max_constraints]
            active_set = active_full[top_idx]
        else:
            active_set = active_full
        if len(active_set) == 0:
            active_set = np.where(ratios > 0.1)[0]
        if len(active_set) == 0:
            active_set = np.arange(size)
    else:
        active_set = np.arange(size)
    
    # Build constraint matrix T[active_set, :]
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        # T[k, j] = a[k-j] for j in [max(0,k-n+1), min(k, n-1)]
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            T[idx, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi + 1)]
    
    # LP: min -sum(b) s.t. T@b <= M, b >= 0
    c = -np.ones(n)
    A_ub = T
    b_ub = np.full(len(active_set), M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method=solver_method)
        if res.success and res.x is not None:
            b_val = np.maximum(res.x, 0.0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val
    except Exception:
        pass
    
    # Fallback: proportional allocation
    conv_max = cross_conv(a, ones_b).max()
    if conv_max > 1e-15:
        return ones_b * (M / conv_max) * 0.5
    return ones_b


def lp_nesterov(a0, time_limit, max_constraints=300, verbose=True):
    """
    Nesterov-accelerated LP best-response iterations.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    y = a.copy()
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, max_constraints=max_constraints)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        y = y_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best_max={best_score:.8f}")
    
    return best_a, best_score


def targeted_cd(a0, time_limit, rng, verbose=True):
    """
    Targeted coordinate descent: focus on positions contributing to the peak.
    Tries both single-coordinate moves and pair transfers.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))
        
        # Identify positions contributing most to the peak
        # (a*a)[k_star] = sum_i a[i]*a[k_star-i]
        # Contribution of position j: a[j] * a[k_star-j] (if valid)
        contrib = np.zeros(n)
        for j in range(n):
            j2 = k_star - j
            if 0 <= j2 < n:
                contrib[j] = a[j] * a[j2]
        
        # Sort by contribution (descending)
        top_positions = np.argsort(contrib)[::-1][:min(60, n)]
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            i = perm[idx]
            
            if a[i] < norm * 1e-10:
                continue
            
            # Try transferring from i to several candidate j's
            candidates = []
            # Top contributing positions (excluding i)
            for j in top_positions:
                if j != i:
                    candidates.append(int(j))
            # Plus a few random
            for _ in range(4):
                candidates.append(int(rng.integers(n)))
            
            best_d = 0.0
            best_j = -1
            best_val = conv.max()
            
            for j in candidates:
                if j == i or j < 0 or j >= n:
                    continue
                if a[i] < 1e-15:
                    continue
                
                # Compute the effect of transferring delta from i to j
                # Change in conv_k for small delta:
                # d(conv_k)/d(delta) = a[k-j] + a[j-k] - a[k-i] - a[i-k] (first order)
                # Plus second order at k=2i: -2*a[i]*delta + delta^2
                # Plus second order at k=2j: +2*a[j]*delta + delta^2
                
                # Evaluate at several delta values
                n_samples = 15
                deltas = np.linspace(0.0, a[i] * 0.8, n_samples)
                for d in deltas:
                    if d < 1e-15:
                        continue
                    # First-order change
                    g = np.zeros(size)
                    # Positive: adding to j
                    lo = max(0, j)
                    hi = min(size, j + n)
                    if hi > lo:
                        g[lo:hi] += a[max(0, lo - j):max(0, hi - j)]
                    lo2 = max(0, n - 1 - j)
                    hi2 = min(size, n + i)
                    if hi2 > lo2:
                        # a[j + (n-1-k)] for k in [n-1-j, ...]
                        # Actually: conv[j+m] gets a[m] added for m=0..n-1
                        # And conv[n-1-(j+m)] gets a[m] added
                        pass
                    # Simpler: just compute the change directly
                    # conv_new_k = conv_k + d*(a[k-j]+a[j-k] - a[k-i]-a[i-k]) + O(d^2)
                    # Let's just evaluate a few points directly
                    
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial * norm / s
                    conv_trial = fast_autocorr(a_trial)
                    # Add second-order correction for the diagonal terms
                    if 2 * i < size:
                        conv_trial[2*i] = a_trial[i]**2 * (2*(n-1-2*i) + 1 - (n-1-2*i))  # complicated
                    # Just use the FFT result
                    trial_max = conv_trial.max()
                    if trial_max < best_val - 1e-12:
                        best_val = trial_max
                        best_d = d
                        best_j = j
            
            if best_j >= 0 and best_d > 1e-15:
                a[i] -= best_d
                a[best_j] += best_d
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[best_j] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[best_j] -= best_d
                    a[i] += best_d
                    a = np.maximum(a, 0.0)
                    a = a * norm / max(a.sum(), 1e-15)
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_gradient(a0, time_limit, verbose=True):
    """
    LSE (log-sum-exp) smooth approximation with projected gradient.
    Uses FFT-based gradient computation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Multi-stage temperature annealing
    temps = np.geomspace(30.0, 0.3, 12)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(5, min(200, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            max_conv = conv.max()
            # LSE weights (important lags get more weight)
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            
            # Gradient of sum_k w_k * (a*a)_k w.r.t. a_j
            # = sum_k w_k * (a[k-j] + a[j-k]) for valid indices
            # = 2 * cross_corr(w, a)[j] (approximately, for symmetric case)
            # More precisely: grad_j = sum_k w_k * a[k-j] + sum_k w_k * a[j-k]
            # First term: sum_k w_k * a[k-j] = cross_corr(w, a) at lag j
            # Second term: sum_k w_k * a[j-k] = cross_corr(a, w) at lag j = same (by symmetry of real cross-corr)
            # So grad_j = 2 * cross_corr(w, a)[j]
            
            # Compute cross-correlation via FFT
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # cross_corr(w, a)[j] = sum_k w[k]*a[k+j]
            cc = irfft(W * np.conj(A), fft_size)
            # For j=0: cc[0] = sum w[k]*a[k] = dot(w, a)
            # For j>0: we need cc at the right position
            # cross_corr[j] = sum_k w[k] * a[k+j] (circular)
            # We want: sum_k w[k] * a[k-j] for j=0,...,n-1
            # = cross_corr at negative lag = cc[fft_size - j] for j>0
            grad = np.zeros(n)
            grad[0] = 2.0 * cc[0]
            if n > 1:
                # cc[fft_size - j] for j = 1, ..., n-1
                neg_lags = cc[fft_size - 1 : fft_size - n : -1]
                grad[1:] = 2.0 * neg_lags
            
            # Projected gradient step on {a >= 0, sum(a) = norm}
            step = norm / (T * n * (1.0 + 0.1 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def fw_subgrad(a0, time_limit, n_peak=5, verbose=False):
    """
    Frank-Wolfe with subgradient using top-n peak averaging.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        
        # Use top-n peaks for a smoothed subgradient
        top_k = np.argsort(conv_y)[::-1][:n_peak]
        weights = np.zeros(len(conv_y))
        for k in top_k:
            weights[k] = 1.0 / n_peak
        
        # Subgradient: grad_j = 2 * sum_k weights[k] * (y[k-j] + y[j-k])
        # = 2 * cross_corr(weights, y)[j]
        W = rfft(weights, 1 << ((2*n-2).bit_length()))
        Y = rfft(y, 1 << ((2*n-2).bit_length()))
        fft_sz = 1 << ((2*n-2).bit_length())
        cc = irfft(W * Y, fft_sz)
        # Actually use cross_corr properly
        Wf = rfft(weights, fft_sz)
        Yf = rfft(y, fft_sz)
        cc = irfft(Wf * np.conj(Yf), fft_sz)
        grad = np.zeros(n)
        grad[0] = 2.0 * cc[0]
        if n > 1:
            grad[1:] = 2.0 * cc[fft_sz - 1 : fft_sz - n : -1]
        
        # FW step: vertex = e_{argmin(grad)} * norm
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new
        
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.7 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.2 + 0.8 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        # Flattened top-hat
        a = np.ones(n)
        fade = 0.15 * n
        for i in range(int(fade)):
            fract = (i + 1) / fade
            a[i] = fract
            a[n - 1 - i] = fract
    else:
        # Slightly tapered uniform
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * x - 1) ** 4
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def single_coord_descent(a0, time_limit, rng, verbose=True):
    """
    Single-coordinate descent: for each coordinate, find optimal value
    that minimizes the max autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = norm_sequence(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-10 and rng.random() > 0.1:
                continue
            
            # Current convolution
            conv = fast_autocorr(a)
            cur_max = conv.max()
            
            # Try a few values for a[i]
            # The "natural" range is [0, min(a[i]*2, norm*0.1)]
            old_val = a[i]
            trial_range = np.array([
                0.0,
                old_val * 0.5,
                old_val * 0.8,
                old_val * 1.2,
                old_val * 1.5,
                min(old_val * 2.0, norm * 0.05),
            ])
            # Add some random
            trial_range = np.append(trial_range, rng.random() * norm * 0.02)
            
            best_trial = old_val
            best_trial_score = cur_max
            
            for new_val in trial_range:
                new_val = max(0.0, new_val)
                if abs(new_val - old_val) < 1e-12:
                    continue
                
                a[i] = new_val
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a * norm / s
                sc = fast_autocorr(a_norm).max()
                if sc < best_trial_score - 1e-12:
                    best_trial_score = sc
                    best_trial = new_val
            
            if best_trial != old_val:
                a = a * norm / max(a.sum(), 1e-15)
                a[i] = best_trial
                a = np.maximum(a, 0.0)
                a = a * norm / max(a.sum(), 1e-15)
                sc = fast_autocorr(a).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = old_val
        
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and returns the best sequence.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    parent = None
    
    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = min(remaining * 0.30, 480.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t1, max_constraints=250, verbose=True)
        do_update(a_out, f"lpn_n{n1}")
    
    # === Phase 2: Targeted CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Targeted CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_cd(a_norm, t2, rng, verbose=True)
        do_update(a_out, f"tcd_n{n_best}")
    
    # === Phase 3: LSE gradient ===
