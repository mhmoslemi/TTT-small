import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        dual_val = M_val

        a = a_new

            stall_count = 0

            if score_avg < best_score:

def lse_objective(z, T, n):
    """Compute LSE-smoothed objective for L-BFGS."""
    a = softmax_fn(z)
    c = fast_autocorr(a)
    Tc = T * (c - c.max())
    return 2.0 * n * (c.max() + np.log(np.exp(Tc).sum()) / T)


def lse_gradient(z, T, n):
    """Compute gradient of LSE-smoothed objective w.r.t. z."""
    a = softmax_fn(z)
    c = fast_autocorr(a)
    Tc = T * (c - c.max())
    exp_Tc = np.exp(Tc)
    w = exp_Tc / exp_Tc.sum()

    # g_a[i] = sum_k w[k] * a[k-i] = convolve(a[::-1], w, 'full')[n-1 : 2n-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A_rev * W, fft_size)
    g_a = conv_full[n - 1:2 * n - 1]
    g_a = 2.0 * g_a

    S = np.dot(g_a, a)
    grad_z = 4.0 * n * a * (g_a - S)
    return grad_z

    # Multi-stage temperature schedule

    for si, (stage_T, max_iter) in enumerate(stages):

        # Check if we should reduce max_iter based on time

        try:

def coord_descent_refine(a0, time_limit, rng):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    bmax = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        # Find top peaks
        threshold = bmax * 0.98
        top_peaks = np.where(conv >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(conv)[::-1][:10]
        top_peaks = top_peaks[:20]

        resp = np.zeros(n)
        for k in top_peaks:

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            resp[idx] += a[k - idx]

        if resp.max() < 1e-15:
            break

        # Find source (highest responsibility)
        src = int(np.argmax(resp))
        if a[src] < 1e-10:
            # Find next highest
            resp[src] = 0
            src = int(np.argmax(resp))
            if a[src] < 1e-10:
                stall += 50
                if stall > 200:
                    break
                # Random perturbation
                for _ in range(3):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * 0.02
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_autocorr(a)
                bmax = conv.max()
                continue

        # Find safe destination: minimize increase in top peaks
        # Estimate damage at each position j: how much would conv increase?
        damage = np.zeros(n)
        for k in top_peaks[:10]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            damage[idx] += a[k - idx]

        # Exclude source and positions near source
        damage[src] = float('inf')
        safe_candidates = np.where(damage < float('inf'))[0]
        if len(safe_candidates) == 0:
            stall += 100
            break

        # Pick the safest destination (lowest damage, and low current value)
        safe_sorted = safe_candidates[np.argsort(damage[safe_candidates] + 0.1 * a[safe_candidates])]
        dst_pool = safe_sorted[:min(20, len(safe_sorted))]

        improved = False
        for dst in dst_pool[:10]:
            # Binary search on step size
            lo, hi = 1e-6, a[src] * 0.5
            best_frac = 0
            best_new_max = bmax
            for _ in range(8):
                mid = (lo + hi) / 2
                delta = a[src] * mid
                # Fast conv update: conv_new[k] = conv[k] + 2*delta*(a[k-src] - a[k-dst])
                # Check if max decreases
                k_src = np.arange(n)
                val_src = np.zeros(2 * n - 1)
                val_dst = np.zeros(2 * n - 1)
                # conv[k] += 2*delta*a[k-src] for valid k
                for kk in range(2 * n - 1):
                    if 0 <= kk - src < n:
                        val_src[kk] = a[kk - src]
                    if 0 <= kk - dst < n:
                        val_dst[kk] = a[kk - dst]
                delta_conv = 2.0 * delta * (val_src - val_dst)
                new_conv = conv + delta_conv
                new_max = new_conv.max()
                if new_max < best_new_max - 1e-10:
                    best_new_max = new_max
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid

            if best_frac > 0 and best_new_max < bmax * (1 - 1e-8):
                delta = a[src] * best_frac
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_autocorr(a)
                bmax = conv.max()
                new_score = 2.0 * n * bmax
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1

            if stall % 20 == 0:

                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
