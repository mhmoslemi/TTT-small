Let me estimate: for n=1600, the LP has 1600 vars and 3199 constraints. With CLARABEL (interior point), this should be fast, maybe 10-50ms. So we could do 10,000-50,000 iterations.

Actually, I think the main bottleneck in the previous code was the PCD, not the LP. Let me focus on:
1. Making the LP Nesterov loop faster and more effective
2. Adding a much faster and more effective polishing step

**New Polishing Step: Vectorized Coordinate Descent**

Instead of pair swaps, do single-coordinate updates:
- For each coordinate i, find the optimal value a_i given all other coordinates fixed.
- The objective is max_k conv(a,a)[k]. For a fixed a_i change, the change in conv is:
  delta_conv[k] = 2 * (a_i_new - a_i) * a[k-i] + (a_i_new^2 - a_i^2) * [k == 2i] (approximately)
  
  Wait, it's more complex because changing a_i affects multiple convolution entries.
  
  conv[k] = sum_j a[j] * a[k-j]
  d(conv[k])/d(a_i) = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
  
  At k=2i: d(conv[2i])/d(a_i) = a[i] + a[i] = 2*a[i] (both terms contribute)
  
  So the gradient w.r.t. a_i is: sum_k w_k * (a[k-i] + a[i-k]) where w_k are the LSE weights.

This is the same gradient we computed above! So the LSE gradient descent IS coordinate descent in a sense.

For exact coordinate descent (minimizing the true max, not LSE), we'd need to solve a 1D problem for each coordinate, which is more complex.

Let me think about what's the most impactful improvement I can make:

1. **The LSE gradient with FFT is the key** - it's O(n log n) per iteration vs O(n^2) for the previous direct computation. This allows thousands of iterations.

2. **Temperature annealing schedule** - start with high T (smooth, can escape local optima) and gradually decrease T (sharper, converges to true max).

3. **Adaptive step size with backtracking** - instead of fixed step, use backtracking to ensure decrease.

4. **Combination with LP best response** - use LSE gradient for most of the time, and occasionally use LP best response for a big jump.

5. **Multi-scale** - optimize at length n/4, then n/2, then n. Each provides a better initialization.

Let me now write the code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient descent with temperature annealing (fast, main workhorse)
2. LP best-response with Nesterov acceleration (periodic big jumps)
3. Vectorized coordinate descent polish
4. Multi-scale initialization and length exploration
Key innovation: O(n log n) gradient computation via FFT enables 100x more iterations
than the previous O(n^2) approach.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

# Global solver cache
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_gradient(a, w, fft_size):
    """
    Compute gradient of LSE_T(conv(a,a)) w.r.t. a.
    grad_i = 2 * (sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k})
    Using FFT: grad_i = 2 * (conv(w_rev, a)[2n-2-i] + conv(w, a)[i])
    where w_rev = w[::-1]
    """
    n = len(a)
    w_rev = w[::-1]
    W = rfft(w, fft_size)
    Wr = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    
    # conv(w, a) has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    # We need elements [0:n] of conv(w, a)
    conv_wa = irfft(W * A, fft_size)
    # conv(w_rev, a) has same length
    conv_wra = irfft(Wr * A, fft_size)
    
    # grad_i = 2 * (conv_wra[2n-2-i] + conv_wa[i]) for i = 0, ..., n-1
    # conv_wra has indices 0 to 3n-3 (length 3n-2)
    # We need conv_wra[2n-2-i] for i=0..n-1, so indices 2n-2 down to n-1
    indices = np.arange(2*n-2, n-2, -1)  # 2n-2, 2n-3, ..., n-1
    # Wait: for i=0, index = 2n-2; for i=n-1, index = 2n-2-(n-1) = n-1
    # So indices = [2n-2, 2n-3, ..., n-1]
    
    grad = 2.0 * (conv_wra[n-1:2*n-1][::-1] + conv_wa[0:n])
    # Let me verify: conv_wra[n-1:2*n-1] gives indices n-1, n, ..., 2n-2
    # Reversed: 2n-2, 2n-3, ..., n-1
    # conv_wa[0:n] gives indices 0, 1, ..., n-1
    # So grad[i] = 2 * (conv_wra[2n-2-i] + conv_wa[i]) ✓
    
    return grad

def lse_gradient_optimize(a0, time_limit, verbose=True, T_start=50.0, T_end=0.1, n_stages=10):
    """
    LSE-smoothed gradient descent with temperature annealing and FFT-accelerated gradients.
    Minimizes LSE_T(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    # Need fft_size >= 3n-2 for the gradient computation
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        stage_start = time.monotonic()
        stage_budget = min((time_limit - (stage_start - t0)), time_limit / n_stages * 2)
        if stage_budget < 1:
            break
        
        # Adaptive step size: start large, decrease
        base_step = norm / (T * np.sqrt(n)) * 0.1
        
        while time.monotonic() - t0 < time_limit - 0.5:
            if time.monotonic() - stage_start > stage_budget:
                break
            
            conv = fast_autocorr(a)
            max_conv = conv.max()
            
            # LSE weights
            scaled = (conv - max_conv) / T
            exp_vals = np.exp(scaled)
            w = exp_vals / exp_vals.sum()
            
            # Compute gradient via FFT
            grad = compute_lse_gradient(a, w, fft_size)
            
            # Backtracking line search
            step = base_step / (1.0 + 0.01 * si)
            improved = False
            for backtrack in range(10):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new * norm / s
                score = fast_autocorr(a_new).max()
                if score < max_conv:
                    a = a_new
                    improved = True
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    break
                step *= 0.5
            
            if not improved:
                break
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint: T @ b <= M where T[k,i] = a[k-i] (valid)
    # Instead of building the full matrix, use the structure
    # Actually for cvxpy, we need to build it or use the convolution structure
    # For n <= 3000, building the matrix is fine (3n-2 entries per row, total ~n*(2n-1))
    
    # Build the convolution matrix efficiently
    # T has shape (2n-1, n), T[k,i] = a[k-i] if 0 <= k-i < n else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit - 1.0:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            break
        
        b_sum = b.sum()
        g = b * norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """
    Fast coordinate descent: for each coordinate, find optimal value
    using the LSE gradient direction (1D line search).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()
    
    # Use a fixed moderate temperature for the gradient
    T = 1.0
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 1.0:
        n_passes += 1
        pass_improved = False
        
        conv = fast_autocorr(a)
        max_conv = conv.max()
        scaled = (conv - max_conv) / T
        exp_vals = np.exp(scaled)
        w = exp_vals / exp_vals.sum()
        grad = compute_lse_gradient(a, w, fft_size)
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            if grad[i] <= 0:
                continue
            
            # Try decreasing a[i] (since grad[i] > 0 means increasing a[i] increases objective)
            # Actually, we want to MINIMIZE the objective, so we should DECREASE a[i] if grad[i] > 0
            # and INCREASE a[i] if grad[i] < 0
            # But we need to maintain sum(a) = norm and a[i] >= 0
            # So we transfer from i to some j where grad[j] is most negative
            
            # Find best j to transfer to
            if grad[i] > 0:
                # Transfer from i to argmin(grad)
                j = int(np.argmin(grad))
                if j == i or grad[j] >= grad[i]:
                    continue
                # Find optimal delta
                ai = a[i]
                best_d = ai * 0.1  # start with small transfer
                best_val = best_score
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    d = ai * frac
                    a[i] -= d
                    a[j] += d
                    s = a.sum()
                    if s > 1e-15:
                        a = a * norm / s
                        score = fast_autocorr(a).max()
                        if score < best_val:
                            best_val = score
                            best_d = d
                    a[j] -= d
                    a[i] += d
                    a = a * norm / a.sum()
                
                if best_d > 0 and best_val < best_score - 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    best_score = best_val
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        # Adjust temperature
        T *= 0.8
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Symmetric bump
        x = np.linspace(-1, 1, n)
        a = np.exp(-4*x**2)
    else:
        # Broad symmetric
        x = np.linspace(-1, 1, n)
        a = (1 - x**2)**2
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def multi_scale_optimize(a_best, n_target, time_limit, solver, verbose=True):
    """
    Multi-scale optimization: optimize at increasing lengths.
    Start from a_short, optimize, upsample, optimize at a_medium, etc.
    """
    n = n_target
    # Create a schedule of lengths
    lengths = []
    cur = max(64, n // 8)
    while cur < n:
        lengths.append(cur)
        cur = min(cur * 2, n)
    lengths.append(n)
    
    a = a_best.copy()
    t0 = time.monotonic()
    
    for li, n_li in enumerate(lengths):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        if len(a) != n_li:
            a = interpolate_sequence(a, n_li)
        
        stage_budget = remaining * (0.3 if li < len(lengths)-1 else 0.7)
        
        if verbose:
            print(f"    Multi-scale: n={n_li}, budget={stage_budget:.0f}s")
        
        # LSE gradient at this scale
        a_out, _ = lse_gradient_optimize(a, stage_budget * 0.6, verbose=False)
        a = a_out
        
        # LP Nesterov at this scale (if enough time)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 10 and n_li >= 256:
            a_out, _ = nesterov_lp_optimize(a, remaining * 0.5, solver, verbose=False)
            # Check if it's better
            s_new = a_out.sum()
            a_out_norm = a_out * np.sqrt(2*n_li) / s_new
            sc_new = fast_autocorr(a_out_norm).max()
            s_cur = a.sum()
            a_cur_norm = a * np.sqrt(2*n_li) / s_cur
            sc_cur = fast_autocorr(a_cur_norm).max()
            if sc_new < sc_cur:
                a = a_out
    
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP best-response with Nesterov acceleration (periodic big jumps)

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)

        except Exception:

    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, w, fft_size):
    """
    Compute gradient of LSE_T(conv(a,a)) w.r.t. a using FFT.
    grad_i = 2 * (sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k})
    = 2 * (conv(w_rev, a)[2n-2-i] + conv(w, a)[i])
    """
    n = len(a)
    w_rev = w[::-1]
    W = rfft(w, fft_size)
    Wr = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)

    conv_wa = irfft(W * A, fft_size)
    conv_wra = irfft(Wr * A, fft_size)

    # grad[i] = 2 * (conv_wra[2n-2-i] + conv_wa[i])
    # conv_wra[n-1:2n-1] reversed gives indices 2n-2, 2n-3, ..., n-1
    grad = 2.0 * (conv_wra[n - 1:2 * n - 1][::-1] + conv_wa[0:n])
    return grad


def lse_gradient_optimize(a0, time_limit, verbose=True, T_start=None, T_end=None, n_stages=12):
    """
    LSE-smoothed gradient descent with temperature annealing and FFT-accelerated gradients.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv_init = fast_autocorr(a)
    max_init = conv_init.max()
    if T_start is None:
        T_start = max_init / np.sqrt(n) * 2.0
    if T_end is None:
        T_end = max_init * 0.005

    best_a = a.copy()
    best_score = max_init
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()

    temps = np.geomspace(max(T_start, 0.01), max(T_end, 0.001), n_stages)

    total_iters = 0
    for si, T in enumerate(temps):
        stage_start = time.monotonic()
        elapsed_global = time.monotonic() - t0
        remaining_global = time_limit - elapsed_global
        if remaining_global < 1.0:
            break

        # Budget: equal split among remaining stages
        stages_left = n_stages - si
        stage_budget = remaining_global / stages_left * 1.2
        stage_budget = min(stage_budget, remaining_global)

        # Step size proportional to T (Lipschitz constant ~ 4/T)
        base_lr = T * 0.05

        iters_this_stage = 0
        while time.monotonic() - t0 < time_limit - 0.5:
            if time.monotonic() - stage_start > stage_budget:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()

            # LSE weights (shift to avoid overflow)
            scaled = (conv - max_conv) / T
            exp_vals = np.exp(scaled)
            w = exp_vals / exp_vals.sum()

            # FFT gradient
            grad = compute_lse_gradient(a, w, fft_size)

            # Backtracking line search
            lr = base_lr
            improved = False
            for bt in range(8):
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new * norm / s
                score = fast_autocorr(a_new).max()
                if score < max_conv - 1e-15:
                    a = a_new
                    improved = True
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    break
                lr *= 0.5

            iters_this_stage += 1
            total_iters += 1

            if not improved:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.4f} iters={iters_this_stage} ({elapsed:.0f}s): best={best_score:.8f}")

    if verbose:
        print(f"    LSE total iters: {total_iters}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    # Build convolution matrix T[k,i] = a[k-i] when 0<=k-i<n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t = 0

    while time.monotonic() - t0 < time_limit - 1.0:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            break

        b_sum = b.sum()
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def cd_polish(a0, time_limit, verbose=True):
    """
    Fast coordinate descent polish using FFT gradients.
    At each step, transfer mass from the coordinate with highest positive grad
    to the one with lowest (most negative) grad.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv_init = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv_init.max()
    t0 = time.monotonic()
    fft_size = 1 << (3 * n - 3).bit_length()

    T = max(conv_init.max() / n, 0.01)
    T_min = 0.005
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 1.0 and T > T_min:
        n_passes += 1
        pass_improved = False

        conv = fast_autocorr(a)
        max_conv = conv.max()
        scaled = (conv - max_conv) / T
        exp_vals = np.exp(scaled)
        w = exp_vals / exp_vals.sum()
        grad = compute_lse_gradient(a, w, fft_size)

        # Multiple transfer attempts per pass
        for _ in range(5):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst or grad[i_src] <= 0:
                break

            a_i = a[i_src]
            if a_i < norm * 1e-6:
                break

            # Try several fractions
            best_frac = 0.0
            best_val = best_score
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                d = a_i * frac
                a[i_src] -= d
                a[i_dst] += d
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    score = fast_autocorr(a).max()
                    if score < best_val:
                        best_val = score
                        best_frac = frac
                # Undo
                a[i_dst] -= d * norm / s
                a[i_src] += d * norm / s
                a = a * norm / a.sum()

            if best_frac > 0 and best_val < best_score - 1e-12:
                d = a[i_src] * best_frac
                a[i_src] -= d
                a[i_dst] += d
                a = a * norm / a.sum()
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
            else:
                break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        T = max(T * 0.7, T_min)

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(-1, 1, n)
        a = np.exp(-4.0 * x * x)
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = (1.0 - x * x) ** 2
    elif choice == 8:
        # Symmetric with dip in middle
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x * x) + 0.3 * np.exp(-8.0 * x * x)
    else:
        # Broad plateau
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) < 0.5, 1.0, np.exp(-4.0 * (np.abs(x) - 0.5) ** 2))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate():
    """
    Main search function combining FFT-accelerated LSE gradient, LP Nesterov,
    and coordinate descent polish with multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1740.0  # 60s safety buffer

    rng = np.random.default_rng(123)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0.0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ===== PHASE 1: LSE gradient at base length (primary workhorse) =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE gradient n={n_cur} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_optimize(a_norm, t1, verbose=True, n_stages=12)
        do_try_update(a_out, f"lse1_n{n_cur}")

    # ===== PHASE 2: LP Nesterov for structural improvement =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 20:
        print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # ===== PHASE 3: LSE gradient continued (lower T, more precise) =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 15:
        print(f"\n=== Phase 3: LSE low-T n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        conv_now = fast_autocorr(a_norm)
        max_now = conv_now.max()
        a_out, _ = lse_gradient_optimize(a_norm, t3, verbose=True,
                                         T_start=max_now / 5.0,
                                         T_end=max_now * 0.002, n_stages=10)
        do_try_update(a_out, f"lse2_n{n_cur}")

    # ===== PHASE 4: Try a longer sequence =====
    n_long = min(n_base + 400, 2200)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        print(f"\n=== Phase 4: LSE gradient n={n_long} (budget={t4:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = lse_gradient_optimize(a_norm, t4, verbose=True, n_stages=8)
        do_try_update(a_out, f"lse_n{n_long}")
        # LP on longer if time
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and len(best_seq) == n_long:
            t4b = min(remaining * 0.3, 80.0)
            if t4b > 10:
                a_norm = best_seq * np.sqrt(2.0 * n_long) / best_seq.sum()
                a_out, _ = nesterov_lp_optimize(a_norm, t4b, solver, verbose=False)
                do_try_update(a_out, f"lp_n{n_long}")

    # ===== PHASE 5: Coordinate descent polish =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        print(f"\n=== Phase 5: CD polish n={n_cur} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = cd_polish(a_norm, t5, verbose=True)
        do_try_update(a_out, f"cd_n{n_cur}")

    # ===== PHASE 6: Final LSE with very low temperature =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 10:
        print(f"\n=== Phase 6: LSE ultra-low-T n={n_cur} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        conv_now = fast_autocorr(a_norm)
        max_now = conv_now.max()
        a_out, _ = lse_gradient_optimize(a_norm, t6, verbose=True,
                                         T_start=max_now / 10.0,
                                         T_end=max_now * 0.001, n_stages=8)
        do_try_update(a_out, f"lse_lowT_n{n_cur}")

    # ===== PHASE 7: Final CD polish =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 5:
        print(f"\n=== Phase 7: Final CD n={n_cur} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = cd_polish(a_norm, t7, verbose=True)
        do_try_update(a_out, f"cd_final_n{n_cur}")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
