2. Compute f(a) = max_k f_k(a), find K = argmax.
3. Compute subgradient: if |K| = 1, use ∇f_k for the unique k in K. If |K| > 1, use average (or solve for the best combination).

5. Repeat.

def qp_polish_step(a, delta_frac=0.01, top_P=5, solver=None, max_retries=5):
    n = len(a)
    b = fast_autocorr(a)
    M_max = b.max()
    top_ks = np.argsort(b)[::-1][:top_P]
    
    # Gradients
    grads = np.zeros((top_P, n))
    for j, k in enumerate(top_ks):
        for m in range(n):
            km = k - m
            if 0 <= km < n:
                grads[j, m] = 2 * a[km]
    
    # Try different delta values
    for attempt in range(max_retries):
        delta = delta_frac * M_max / (2 ** attempt)
        if delta < 1e-10 * M_max:
            break
        
        rhs = np.array([grads[j] @ a - delta for j in range(top_P)])
        
        a_var = cp.Variable(n)
        obj = cp.Minimize(cp.sum_squares(a_var - a))
        constraints = [a_var >= 0, cp.sum(a_var) == 1]
        for j in range(top_P):
            constraints.append(grads[j] @ a_var <= rhs[j])
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status in ('optimal', 'optimal_inaccurate'):
            a_new = np.maximum(np.asarray(a_var.value), 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                # Verify: check if peaks actually reduced
                b_new = fast_autocorr(a_new)
                if b_new.max() < M_max - 0.5 * delta:
                    return a_new, True
        # If not successful, retry with smaller delta
    
    return a, False

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Corrected LSE gradient (single cross-correlation, fixing previous double-counting bug)
2. Local QP polish via linearized peak constraints (convex subproblem)
3. Nesterov-accelerated LSE descent with adaptive temperature
4. Multi-length exploration with interpolation transfer
5. Peak-targeting local search for final polish
Key fix: previous gradient was r1+r2 (two convolutions); correct gradient is 2*r1 (one cross-correlation).
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_value(a, T):
    """Compute LSE surrogate value: T * log(sum exp(f_k/T))."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def lse_grad_correct(a, T):
    """
    Correct LSE gradient: d/da[m] [T*log(sum_k exp(f_k/T))]
    = sum_k w_k * 2*a[k-m] for valid k-m
    = 2 * conv(a_rev, w)[n-1 : 2n-1]
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def lse_grad_old(a, T):
    """Old (incorrect but sometimes helpful) gradient: r1 + r2."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2

def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.3, n_stages=14, use_correct_grad=True, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0
    grad_fn = lse_grad_correct if use_correct_grad else lse_grad_old

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                y_new = y - (1.0 / (it + 3.0)) * grad_fn(y, T)
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_new = y_extrap - (1.0 / (it + 3.0)) * grad_fn(y_extrap, T)

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 300 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    # Also check the averaged iterate
    return best_a, best_score

def qp_polish_step(a, delta_frac=0.004, top_P=6, solver=None):
    """
    One QP polish step: minimize 0.5*||x-a||^2 subject to x>=0, sum(x)=1,
    and linearized peak constraints for top P peaks.
    Returns (new_a, success).
    """
    n = len(a)
    b = fast_autocorr(a)
    M_max = b.max()
    if M_max < 1e-15:
        return a, False

    # Find peaks above threshold
    threshold = 0.85 * M_max
    peak_mask = b >= threshold
    peak_indices = np.where(peak_mask)[0]
    # Sort by value descending, take top_P
    peak_indices = peak_indices[np.argsort(b[peak_indices])[::-1]][:top_P]

    if len(peak_indices) == 0:
        return a, False

    P = len(peak_indices)

    # Compute gradients for each peak: grad_k[m] = 2*a[k-m] if 0<=k-m<n
    grads = np.zeros((P, n))
    for j, k in enumerate(peak_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grads[j, idx] = 2.0 * a[k - idx]

    # Try decreasing delta
    for attempt in range(8):
        delta = delta_frac * M_max * (0.5 ** attempt)
        if delta < 1e-8 * M_max:
            break

        # Linearized constraint: grad_j @ x <= grad_j @ a - delta
        rhs = np.array([grads[j] @ a - delta for j in range(P)])

        try:
            x = cp.Variable(n)
            obj = cp.Minimize(0.5 * cp.sum_squares(x - a))
            constraints = [x >= 0, cp.sum(x) == 1]
            for j in range(P):
                constraints.append(grads[j] @ x <= rhs[j])

            prob = cp.Problem(obj, constraints)
            prob.solve(solver=solver, verbose=False)

            if prob.status in ('optimal', 'optimal_inaccurate'):
                x_val = np.maximum(np.asarray(x.value, dtype=np.float64), 0.0)
                s = x_val.sum()
                if s < 1e-10:
                    continue
                x_val /= s

                # Verify: check actual peak reduction
                b_new = fast_autocorr(x_val)
                M_new = b_new.max()
                if M_new < M_max - 0.3 * delta:
                    return x_val, True
                elif M_new < M_max - 1e-10:
                    return x_val, True
        except Exception:
            continue

    return a, False

def qp_polish_loop(a0, time_limit, delta_frac=0.004, top_P=6, solver=None, verbose=True):
    """Iterate QP polish steps."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        a_new, success = qp_polish_step(a, delta_frac=delta_frac, top_P=top_P, solver=solver)
        it_count += 1

        if success:
            s = a_new.sum()
            a_new = a_new / s
            score = fast_autocorr(a_new).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 30:
            if verbose:
                print(f"    QP polish: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
            break

        if verbose and it_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP polish it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    # Rescale to norm
    best_a = best_a * norm / best_a.sum()
    final_score = fast_autocorr(best_a).max()
    return best_a, final_score

def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search: move mass from peak-contributing positions to safe positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            # Build peak mask: positions that contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_ks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < norm * 1e-8:
                    continue

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(a < a.mean())[0]
                if len(candidates) == 0:
                    continue

                candidates = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 4)]]

                for dst in candidates[:5]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-16:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation every few stalls
            if stall % 5 == 0:
                for _ in range(3):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < norm * 1e-8:
                        continue
                    frac = rng.uniform(0.003, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 1000 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    best_a = best_a * norm / best_a.sum()
    final_score = fast_autocorr(best_a).max()
    return best_a, final_score

def lse_qp_alternating(a0, time_limit, solver=None, verbose=True):
    """Alternate between LSE descent and QP polish for refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n)
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_cycles = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        # LSE phase: 60% of remaining (but at least 10s, at most 60s)
        t_lse = max(10.0, min(remaining * 0.6, 60.0))
        a_lse, _ = nesterov_lse(a * norm, t_lse, T_start=5.0, T_end=0.1,
                                  n_stages=8, use_correct_grad=True, verbose=False)
        s = a_lse.sum()
        if s > 1e-15:
            a_lse = a_lse / s
            score = fast_autocorr(a_lse).max()
            if score < best_score:
                best_score = score
                best_a = a_lse.copy()
            a = a_lse

        # QP phase: 40% of remaining (but at least 5s, at most 40s)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_qp = max(5.0, min(remaining * 0.4, 40.0))
        a_qp, _ = qp_polish_loop(a, t_qp, delta_frac=0.003, top_P=6, solver=solver, verbose=False)
        score = fast_autocorr(a_qp).max()
        if score < best_score:
            best_score = score
            best_a = a_qp.copy()
        a = a_qp / a_qp.sum()

        n_cycles += 1
        if verbose and n_cycles % 3 == 0:
            print(f"    LSE+QP cycle={n_cycles} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    best_a = best_a * norm / best_a.sum()
    final_score = fast_autocorr(best_a).max()
    return best_a, final_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(11.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif choice == 6:
        # Slight asymmetry
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2 * np.exp(-0.5 * (x - 0.4) ** 2)
    else:
        # Alternating structure
        a = np.ones(n) * 0.6
        a[::2] += 0.3 * np.cos(np.linspace(0, 4*np.pi, n)[::2])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses corrected LSE gradient, QP polish, and peak targeting.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LSE Nesterov (correct gradient) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov (correct grad) n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 420.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=20.0, T_end=0.2, n_stages=16,
                                use_correct_grad=True, verbose=True)
        do_try_update(a_out, f"lse_corr_n{n_base}")

    # ============ Phase 1b: LSE Nesterov (old gradient for comparison) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1b: LSE-Nesterov (old grad) n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t1b = min(remaining * 0.10, 150.0)
    if t1b > 20:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1b, T_start=10.0, T_end=0.2, n_stages=10,
                                use_correct_grad=False, verbose=True)
        do_try_update(a_out, f"lse_old_n{n_best}")

    # ============ Phase 2: QP Polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: QP Polish n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = qp_polish_loop(a_work, t2, delta_frac=0.004, top_P=6, solver=solver, verbose=True)
        do_try_update(a_out, f"qp_n{n_best}")

    # ============ Phase 3: LSE at longer length ============
    n2 = n_base + 600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-Nesterov (correct) n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 330.0)
    if t3 > 30:
