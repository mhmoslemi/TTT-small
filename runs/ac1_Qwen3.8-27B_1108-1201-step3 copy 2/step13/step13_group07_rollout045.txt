    """Solve LP best response using constraint generation."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    base_constraints = [cp.sum(b_var) == 1]
    gen_constraints = []
    
    b_current = np.ones(n) / n
    M_current = 1.0
    
    for it in range(max_iters):
        # Evaluate current solution
        conv = np.convolve(a_fixed, b_current)[:2*n-1]
        max_conv = conv.max()
        
        # Find active constraints (top violations)
        top_k = np.argsort(conv)[::-1][:max_constraints]
        
        # Build constraint matrix for active constraints
        T_sub = np.zeros((len(top_k), n))
        for i, k in enumerate(top_k):
            for j in range(n):

                if 0 <= idx < n:

        # Solve LP
        constraints = base_constraints + [T_sub @ b_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)

One more idea: **using the eigendecomposition of the convolution matrix**

The matrix T (where T @ b gives the convolution of a_fixed with b) is related to a Toeplitz matrix. For large n, the eigenvalues of the Toeplitz matrix are given by the Fourier transform of the generating function. Specifically, if a has Fourier transform A(ω), then the eigenvalues of the Toeplitz matrix are approximately A(ω) for ω in [0, 2π).

This means the spectral radius of T is approximately max |A(ω)| = max |FFT(a)|.

The LP: min M s.t. (T @ b)_k <= M, sum(b) = 1, b >= 0

The optimal M is related to the "minimum over b of max_k (T@b)_k" which is the value of the game.

By von Neumann's minimax theorem, this equals "max over w (probability distribution on k) of min over b of sum_k w_k (T@b)_k" = "max_w min_b sum_j b_j (T^T w)_j"

The inner min over b (with sum(b)=1, b>=0) gives: min_j (T^T w)_j

So the game value = max_w min_j (T^T w)_j where w is a probability distribution on {0,...,2n-2}.

This is a linear program in w! And it's the dual of the primal LP.

The dual LP:
- Variables: w_0, ..., w_{2n-2}
- Maximize: min_j sum_k w_k * T[k,j]
- Subject to: sum_k w_k = 1, w_k >= 0

But this has 2n-1 variables and n constraints (one for each j). It's not obviously easier.

However, we can use this dual to get a certificate: if we find w such that min_j (T^T w)_j = v, then v is a lower bound on the game value (and hence on the optimal max autocorrelation).

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT, returns length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a_fixed, b):
    """Compute convolve(a_fixed, b) using FFT. Both length n, result length 2n-1."""
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(v_sorted * idx > (cssv - target_sum))[0]
    if len(rho) == 0:
        rho = np.array([n - 1])
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        result = result / max(result.sum(), 1e-30) * target_sum
    return result


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_sparse_row(a_fixed, k, n):
    """Build row k of the convolution matrix: T[k,j] = a[k-j] if 0<=k-j<n."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    row = np.zeros(n)
    if js <= je:
        row[js:je + 1] = a_fixed[k - je:k - js + 1][::-1]
    return row


def solve_lp_constraint_gen(a_fixed, max_constraints=800, max_rounds=15, solver=cp.ECOS):
    """
    Solve LP best response using constraint generation.
    min M s.t. (a_fixed * b)_k <= M for all k, sum(b)=1, b>=0
    Uses iterative constraint addition for speed.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    base_constraints = [cp.sum(b_var) == 1]
    active_k = list(range(2 * n - 1))[:min(200, 2 * n - 1)]
    # Initialize with top peaks of current autocorrelation
    curr_a = a_fixed
    conv_init = fast_cross_corr(a_fixed, np.ones(n) / n)
    active_k = np.argsort(conv_init)[::-1][:300].tolist()

    b_val = np.ones(n) / n
    M_val = np.inf

    for round_idx in range(max_rounds):
        # Build constraint matrix for active constraints
        T_sub = np.zeros((len(active_k), n))
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                T_sub[i, js:je + 1] = a_fixed[k - je:k - js + 1][::-1]

        constraints = base_constraints + [T_sub @ b_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, max_iter=500)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=1000)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS, max_iter=1000)
                except Exception:
                    break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s < 1e-15:
            break
        b_new = b_new / s
        M_new = float(M_var.value)

        # Check for violated constraints
        conv = fast_cross_corr(a_fixed, b_new)
        violations = np.where(conv > M_new + 1e-8)[0]

        if len(violations) == 0:
            return b_new, M_new

        # Add violated constraints (prioritize worst violations)
        violations_sorted = violations[np.argsort(conv[violations])[::-1]]
        active_k.extend(violations_sorted[:500].tolist())
        active_k = list(set(active_k))[:max_constraints]

        b_val = b_new
        M_val = M_new

    # Return best found
    if M_val < np.inf:
        return b_val, M_val
    return None, None


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with constraint generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 3000
    gap_history = []

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_constraint_gen(y, max_constraints=1000, max_rounds=12, solver=solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Track scores
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall_count = 0
            best_score = score_y
            best_a = y.copy()

        # Time-averaged iterate
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def dual_guided_refine(a0, time_limit, solver=cp.ECOS, verbose=True):
    """
    Use LP dual variables to guide targeted updates.
    At each step, solve the LP, get the dual weights w_k,
    then do a gradient step using these weights.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (4 * n - 3).bit_length()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Solve LP to get best response and dual
        b_star, M_val = solve_lp_constraint_gen(a, max_constraints=500, max_rounds=8, solver=solver)
        if b_star is None:
            break

        # Move toward b_star with adaptive step
        alpha = 0.3
        a_new = (1 - alpha) * a + alpha * b_star
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            score = internal_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                alpha *= 0.5
                if alpha < 0.01:
                    break
        else:
            break

        if stall > 200:
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      DGR ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_refine(a0, time_limit, seed=42):
    """Aggressive coordinate descent using peak information."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:20]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src = idx[np.argmax(contrib)]

            # Find good destination: low contribution to all peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(a < a.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Pick from lowest values
            sorted_idx = candidates[np.argsort(a[candidates])]
            dst_pool = sorted_idx[:max(1, len(sorted_idx) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=20, lr=0.004):
    """LogSumExp-smoothed optimization with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    fft_size = 1 << (4 * n - 3).bit_length()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(20, min(150, int(remaining * 50 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient via FFT
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.05 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_check = softmax_fn(z)
        score = internal_score(a_check)
        if score < best_score:
            best_score = score
            best_a = a_check.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):

            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Adam_LSE]: {best_score:.6f}")

    # Phase 7: Another LP round with remaining time
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LP n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.35, 200.0)
    if t7 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_final]: {best_score:.6f}")

    # Phase 8: Final coordinate shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8 = min(remaining * 0.4, 60.0)
        if t8 > 5:
            a_out, _ = coordinate_refine(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [final_coord]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure we return within time
    elapsed = time.monotonic() - t_global_start
    print(f"Total elapsed: {elapsed:.0f}s")
    return [float(x) for x in best_seq]
