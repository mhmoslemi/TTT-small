def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _focused_lse_grad(y, T, active_lags):
    """Gradient of LSE-smoothed max over only active lags."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_active = ac[active_lags]
    ac_max = ac_active.max()
    shifted = (ac_active - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w_active = exp_s / exp_s.sum()

    # Build full weight vector (zeros for inactive)
    w = np.zeros(2 * n - 1)
    w[active_lags] = w_active

    # Cross-correlation via FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _reduced_lp_best_response(a, M, active_lags, solver):
    """LP: maximize sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix only for active lags
    num_constraints = len(active_lags)
    T_rows = []
    for k in active_lags:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        row[js:je + 1] = a[k - np.arange(js, je + 1)]
        T_rows.append(row)

    if num_constraints == 0:
        return np.ones(n) / n

    T_mat = np.array(T_rows)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_init = len(best_seq)
    print(f"Starting: n={n_init}, score={best_score:.8f}")

    # === Phase 1: Adaptive LSE-FISTA at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Adaptive LSE-FISTA n={n_init} (budget={t1:.0f}s) ===")
        a_out, _ = _adaptive_lse_fista(best_seq, t1, verbose=True)
        _try_update(a_out, f"adapt_lse_n{n_init}")

    # === Phase 2: LP-Nesterov with reduced constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Reduced Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_reduced(a_norm, t2, solver, active_tol=0.02, verbose=True)
        _try_update(a_out, f"lp_red_n{n_cur}")

    # === Phase 3: Peak-pair equality ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Peak-pair equality n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_pair_equality(a_norm, t3, seed=12345)
        _try_update(a_out, f"peakpair_n{n_cur}")

    # === Phase 4: LP at extended length ===
    n2 = n_cur + 200
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 40:
        print(f"\n=== Phase 4: LP-Reduced n={n2} (budget={t4:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _lp_nesterov_reduced(a_norm, t4, solver, active_tol=0.02, verbose=True)
        _try_update(a_out, f"lp_red_n{n2}")

    # === Phase 5: Best length LP with tighter tol ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 40:
        print(f"\n=== Phase 5: LP-Reduced tight tol n={n_cur} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_reduced(a_norm, t5, solver, active_tol=0.005, verbose=True)
        _try_update(a_out, f"lp_tight_n{n_cur}")

    # === Phase 6: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Spectral refinement n={n_cur} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _spectral_refinement(a_norm, t6, seed=999)
        _try_update(a_out, f"spectral_n{n_cur}")

    # === Phase 7: Final LSE with very low T ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30:
        print(f"\n=== Phase 7: Final LSE low-T n={n_cur} (budget={t7:.0f}s) ===")
        a_out, _ = _adaptive_lse_fista(best_seq, t7, verbose=True)
        _try_update(a_out, f"final_lse_n{n_cur}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 20:
        print(f"\n=== Phase 8: Final peak-equality n={n_cur} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_pair_equality(a_norm, t8, seed=77777)
        _try_update(a_out, "final_peakpair")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
