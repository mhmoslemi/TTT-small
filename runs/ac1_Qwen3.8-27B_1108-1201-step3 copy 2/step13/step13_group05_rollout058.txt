def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    ac_shifted = ac - ac.max()
    top_idx = np.argsort(ac_shifted)[::-1][:top_k]
    top_vals = ac_shifted[top_idx]
    w_top = np.exp(top_vals / tau)
    # Approximate the denominator using just top-k (good when tau is small)
    # Also include a uniform baseline
    w_sum = w_top.sum()
    w = w_top / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        # Contribution: 2 * w_k * (a[k-m] for m in [max(0,k-n+1), min(n-1,k)] 
        #                + a[m-k] for m in [max(0,k), min(n-1,k+n-1)] where m != k)
        # Simplified: for each valid m, grad[m] += 2*w_k*a[|k-m|] adjusted
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        valid = (j >= 0) & (j < n)
        grad[idx[valid]] += 2.0 * wi * a[j[valid]]
    
    return grad

def compute_full_softmax_grad(a, ac, tau, n_keep=50):
    """Full softmax gradient considering top n_keep peaks."""
    n = len(a)
    ac_shifted = ac - ac.max()
    # Get top peaks
    idx = np.argsort(ac_shifted)[::-1][:n_keep]
    vals = ac_shifted[idx]
    w = np.exp(vals / tau)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx_m = np.arange(i_start, i_end)
        j = k - idx_m
        valid = (j >= 0) & (j < n)
        grad[idx_m[valid]] += 2.0 * wi * a[j[valid]]
    
    return grad

def softmax_mirror_descent(a0, time_limit, tau_init=5.0, tau_decay=0.9995, 
                           top_k=30, nesterov_mu=0.9, verbose=True, label=""):
    """Mirror descent with softmax-smoothed gradient and Nesterov acceleration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    tau = tau_init
    v = a.copy()  # Nesterov lookahead
    x = a.copy()
    avg_a = a.copy()
    n_avg = 1
    gamma = 1.0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(v)
        
        # Compute gradient with softmax
        if t < 100:
            # Warm-up: use larger tau for stable initialization
            tau_eff = tau_init
        else:
            tau_eff = tau
            tau *= tau_decay
        
        g = compute_softmax_grad(v, ac, tau_eff, top_k=top_k)
        
        # Step size
        step = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror descent update
        log_v = np.log(v + 1e-30)
        log_v_new = log_v - step * g
        max_log = log_v_new.max()
        v_new = np.exp(log_v_new - max_log)
        v_new = np.maximum(v_new, 0.0)
        s_new = v_new.sum()
        if s_new < 1e-15:
            v_new = v.copy()
        else:
            v_new = v_new * c / s_new
        
        # Nesterov momentum
        if t == 1:
            x_new = v_new
        else:
            beta = (nesterov_mu * (1.0 + np.sqrt(1.0 + 4.0 * nesterov_mu * (1.0 - nesterov_mu)))) / 2.0
            x_new = v_new + (nesterov_mu - beta) * (v_new - x)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x > 1e-15:
                x_new = x_new * c / s_x
            else:
                x_new = v_new.copy()
        
        x = v_new
        v = x_new
        
        # Track best
        M_new = fast_autocorr(v).max()
        if M_new < best_m:
            best_m = M_new
            best_a = v.copy()
        
        # Running average
        avg_a += v
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD[{label}] t={t} ({elapsed:.1f}s): J={J:.6f} tau={tau_eff:.3f}")
    
    return best_a, best_m

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, label=""):
    """Nesterov-accelerated LP best-response optimization."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP[{label}] iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: T[k, j] = a[j] if 0 <= k-j < n, else 0
    # conv(a,b)[k] = sum_j a[j] * b[k-j] (where b is indexed 0..n-1)
    # = sum_j T[k,j] * b[j] where T[k,j] = a[j] if 0 <= k-j < n
    
    # For efficiency, use banded structure
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def coordinate_polish(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: move mass to reduce max convolution."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    ac = fast_autocorr(a)
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            j_vals = k - idx
            valid = (j_vals >= 0) & (j_vals < n)
            idx_v = idx[valid]
            jv_v = j_vals[valid]
            contrib = a[idx_v] * a[jv_v]
            
            if contrib.max() < 1e-15:
                continue
            
            # Find source (max contributor to this peak)
            src = idx_v[np.argmax(contrib)]
            src_j = jv_v[np.argmax(contrib)]
            
            # Find destination: minimize increase in all top peaks
            # Score each candidate destination
            n_cand = min(200, n)
            cand_idx = rng.choice(n, size=min(n_cand, n), replace=False)
            
            best_trial = None
            best_trial_m = best_m
            
            for dst in cand_idx:
                if dst == src or dst == src_j:
                    continue
                if a[src] < 1e-15:
                    break
                
                for frac in [0.01, 0.03, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_trial = a_trial
                        break
            
            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                ac = fast_autocorr(a)
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a * c / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a * c / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord[{label}] ({elapsed:.1f}s): best_J={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initializations(n, parent=None, rng=None):
    """Create diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng()
    
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Raised cosine (sin^2)
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x) ** 2)
    
    # 3. Gaussian centered
    x = np.linspace(-1, 1, n)
    g = np.exp(-2 * x ** 2)
    inits.append(g / g.sum())
    
    # 4. Flat top with tapered edges
    edge = max(3, n // 20)
    f = np.ones(n)
    for i in range(edge):
        f[i] *= (i + 1) / edge
        f[n-1-i] *= (i + 1) / edge
    inits.append(f / f.sum())
    
    # 5. Parent interpolated
    if parent is not None and len(parent) > 0:
        inits.append(interpolate_sequence(parent, n))
    
    # 6. Polynomial shape
    x = np.linspace(0, 1, n)
    p = 1 - (2*x - 1)**4
    p = np.maximum(p, 0.01)
    inits.append(p / p.sum())
    
    # Normalize all
    inits = [np.maximum(i, 0) for i in inits]
    inits = [i / i.sum() if i.sum() > 1e-15 else np.ones(n)/n for i in inits]
    
    return inits

def propose_candidate(budget_s=1800.0):

        pass
