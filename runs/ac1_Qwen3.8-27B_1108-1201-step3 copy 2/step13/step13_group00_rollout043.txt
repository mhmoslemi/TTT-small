def compute_lse_gradient(a, w, fft_size):
    """Compute gradient of LSE objective using FFT cross-correlation.
    grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_corr(w, a)[i]
    """
    n = len(a)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return 2.0 * corr[:n]


def lse_gradient_optimize(a0, time_limit, verbose=True):
    """LSE smooth approximation with FFT-based gradient (O(n log n) per step)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    corr_fft_size = 1 << (3 * n - 2 + 1).bit_length()

    n_stages = 15
    temps = np.geomspace(100.0, 0.05, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        n_steps = max(30, min(300, int(remaining * 3)))

        step_base = 2.0 * T / (n * 10.0)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()

            shifted = (conv - max_conv) / T
            max_shifted = shifted.max()
            exp_shifted = np.exp(shifted - max_shifted)
            w = exp_shifted / exp_shifted.sum()

            grad = compute_lse_gradient(a, w, corr_fft_size)

            step = step_base / (1.0 + 0.02 * it)
            step = min(step, 0.5 * norm / (np.max(np.abs(grad)) + 1e-15))

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s

            a = a_new
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            cur_score = fast_autocorr(a).max()
            print(f"    LSE stage T={T:.3f} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix_rows(a, row_indices):
    """Build only the specified rows of the convolution matrix T.
    T[k, j] = a[j] if 0 <= k-j < n, else 0.
    So row k of T has a[j] at positions j where 0 <= k-j < n, i.e., j in [max(0,k-n+1), min(k, n-1)].
    """
    n = len(a)
    T = np.zeros((len(row_indices), n))
    for i, k in enumerate(row_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k + 1, n)
        if j_end > j_start:
            T[i, j_start:j_end] = a[j_start:j_end]
    return T


def lp_best_response(a, M, solver, active_set=None, n_constraints_max=300):
    """Solve LP: max sum(b) s.t. T_active @ b <= M, b >= 0.
    If no active_set given, use all constraints near the peak.
    """
    n = len(a)
    size = 2 * n - 1
    conv_current = fast_autocorr(a)

    if active_set is None:
        active_set = np.where(conv_current >= M * 0.99)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_current >= M * 0.9)[0]
        if len(active_set) < 3:
            active_set = np.where(conv_current >= M * 0.8)[0]
        if len(active_set) > n_constraints_max:
            idx = np.argsort(conv_current[active_set])[::-1][:n_constraints_max]
            active_set = active_set[idx]

    T = build_conv_matrix_rows(a, active_set)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M, cp.sum(b_var) == 1.0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    b_val = b_val / s

    # Check for violated constraints
    full_conv_b = fast_autocorr(a * b_val)
    # Actually we need conv(a, b_val) not conv(a*b_val, a*b_val)
    # Let me compute it properly using FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b_val, fft_size)
    conv_ab = irfft(A * B, fft_size)[:2 * n - 1]

    violated = np.where(conv_ab > M * (1 + 1e-5))[0]
    if len(violated) > 0:
        new_active = np.union1d(active_set, violated)
        if len(new_active) <= n_constraints_max:
            T2 = build_conv_matrix_rows(a, new_active)
            b_var2 = cp.Variable(n, nonneg=True)
            constraints2 = [T2 @ b_var2 <= M, cp.sum(b_var2) == 1.0]
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                pass
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                s2 = b_val.sum()
                if s2 > 1e-15:
                    b_val = b_val / s2
                active_set = new_active
        elif len(new_active) > 0.8 * size:
            # Too many constraints, use full
            T_full = build_conv_matrix_rows(a, np.arange(size))
            b_var3 = cp.Variable(n, nonneg=True)
            constraints3 = [T_full @ b_var3 <= M, cp.sum(b_var3) == 1.0]
            prob3 = cp.Problem(cp.Maximize(cp.sum(b_var3)), constraints3)
            try:
                prob3.solve(solver=solver, verbose=False)
            except Exception:
                pass
            if prob3.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var3.value, dtype=np.float64), 0.0)
                s3 = b_val.sum()
                if s3 > 1e-15:
                    b_val = b_val / s3

    return b_val, active_set


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = None
    n_iters = 0
    stale_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            b, _ = lp_best_response(y, M_curr, solver, None)
            if b is None:
                stale_count += 1
                if stale_count > 10:
                    break
                continue
            stale_count = 0

        n_iters += 1
        alpha = 2.0 / (t + 2.0)

        # Nesterov step
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = b * norm
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * norm / a_new.sum()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_pair_cd(a0, time_limit, rng, verbose=True):
    """Gradient-guided pair coordinate descent.
    Identifies high-pressure and low-pressure elements, transfers mass from high to low.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    corr_fft_size = 1 << (3 * n - 2 + 1).bit_length()

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        conv = fast_autocorr(a)
        max_conv = conv.max()

        # Compute pressure (gradient-like quantity)
        T_smooth = max_conv / (n * 10.0)
        shifted = (conv - max_conv) / T_smooth
        max_sh = shifted.max()
        exp_sh = np.exp(shifted - max_sh)
        w = exp_sh / exp_sh.sum()
        pressure = compute_lse_gradient(a, w, corr_fft_size)

        # Sort by pressure: high pressure = reduce, low pressure = increase
        order = np.argsort(pressure)
        high_idx = order[-n // 4:]  # top 25% highest pressure
        low_idx = order[:n // 4]  # bottom 25% lowest pressure

        # For each high-pressure element, try transferring to a few low-pressure elements
        for i in high_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-10:
                continue

            # Try a few random low-pressure targets
            n_targets = min(5, len(low_idx))
            target_indices = rng.choice(low_idx, size=n_targets, replace=False)

            for j in target_indices:
                if j == i:
                    continue
                ai = a[i]
                if ai < 1e-15:
                    continue

                # Try several delta values
                best_d = 0.0
                best_val = best_score
                n_samples = 10
                deltas = np.linspace(0.1 * ai, 0.9 * ai, n_samples)

                for d in deltas:
                    # Approximate new max conv
                    # First order: change in conv[k] = d * (a[k-j] + a[k-i] - a[k-i] - a[k-j]) ... 
                    # Actually: removing d from a[i] and adding d to a[j]
                    # conv_k changes by: -d*(a[k-i] + a[i-k]) + d*(a[k-j] + a[j-k])
                    # = d * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
                    # But we already know pressure[i] and pressure[j] approximate this.
                    # For accuracy, let's just evaluate at the argmax position(s)
                    k_star = np.argmax(conv)
                    # Change at k_star:
                    delta_conv = 0.0
                    ji = k_star - j
                    jj = k_star - i
                    if 0 <= ji < n:
                        delta_conv += a[ji]
                    if 0 <= j - k_star < n:
                        delta_conv += a[k_star - j]
                    # Wait, d conv_k / d a_i = 2 * a[k-i]
                    # So change = -d * 2*a[k_star - i] + d * 2*a[k_star - j]
                    val_i = 2.0 * a[k_star - i] if 0 <= k_star - i < n else 0.0
                    val_j = 2.0 * a[k_star - j] if 0 <= k_star - j < n else 0.0
                    new_max_approx = conv[k_star] + d * (val_j - val_i)
                    # Also check other potentially tight positions
                    top_k = np.argsort(conv)[-5:]
                    for k2 in top_k:
                        v_i2 = 2.0 * a[k2 - i] if 0 <= k2 - i < n else 0.0
                        v_j2 = 2.0 * a[k2 - j] if 0 <= k2 - j < n else 0.0
                        new_max_approx = max(new_max_approx, conv[k2] + d * (v_j2 - v_i2))
                    # Second order at k=2i and k=2j
                    if 2 * i < size:
                        new_max_approx = max(new_max_approx, conv[2 * i] - 2 * ai * d + d * d)
                    if 2 * j < size:
                        new_max_approx = max(new_max_approx, conv[2 * j] + 2 * a[j] * d + d * d)

                    if new_max_approx < best_val - 1e-12:
                        best_val = new_max_approx
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s > 1e-15:
                        a = a * norm / s
                        conv_actual = fast_autocorr(a)
                        cur_max = conv_actual.max()
                        if cur_max < best_score - 1e-10:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                    else:
                        a[j] -= best_d
                        a[i] += best_d

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_pair_cd(a0, time_limit, rng, verbose=True):
    """Simple random pair coordinate descent for exploration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_attempts = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        n_attempts += 1
        if n_attempts % 1000 == 0:
            if stale > 5000:
                break

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < norm * 1e-10:
            continue

        ai = a[i]
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))

        # First-order improvement estimate
        val_i = 2.0 * a[k_star - i] if 0 <= k_star - i < n else 0.0
        val_j = 2.0 * a[k_star - j] if 0 <= k_star - j < n else 0.0
        if val_j >= val_i:
            continue  # No first-order improvement

        # Try a fraction of ai
        for frac in [0.3, 0.5, 0.7]:
            d = frac * ai
            a[i] -= d
            a[j] += d
            a = a * norm / a.sum()
            cur = fast_autocorr(a).max()
            if cur < best_score - 1e-10:
                best_score = cur
                best_a = a.copy()
                stale = 0
                break
            else:
                a = best_a * norm / best_a.sum()
                stale += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Try various parametric shapes."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(x) + 0.15 * np.cos(3.0 * x) + 0.05 * np.cos(5.0 * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - (2.0 * x - 1.0) ** 2) ** 2
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        # Raised cosine squared
        a = (0.5 + 0.5 * np.cos(2.0 * np.pi * x)) ** 2
    else:
        x = np.linspace(0.0, 1.0, n)
        # Flat top
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE gradient descent (fast, FFT-based)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LSE gradient n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = lse_gradient_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse_n{n1}")

    # Phase 2: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_optimize(a_norm, t2, solver, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Targeted PCD n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_pair_cd(a_norm, t3, rng, verbose=True)

    # Phase 4: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length exploration (budget={remaining:.0f}s) ===")
    n_opts = [n_base - 200, n_base, n_base + 200, n_base + 500]
    n_opts = [max(400, x) for x in n_opts]
    per_len = max(20.0, (remaining - 60.0) / len(n_opts))

    for n_try in n_opts:
        if n_try == n_base and best_seq is not None and len(best_seq) == n_base:
            continue  # Already optimized at this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < per_len + 30:

        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
