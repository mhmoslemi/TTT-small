import numpy as np

def fast_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * float(np.max(b)) / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)

def lbfgs_lse(a_init, time_limit, T_start=50.0, T_end=0.5, n_stages=10, topk=50, max_iter_per_stage=50):
    """
    L-BFGS optimization of top-k LSE smoothed objective.
    Works in the simplex using a softmax parameterization.
    """
    n = len(a_init)
    a = normalize(np.asarray(a_init, dtype=np.float64))
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Use log-barrier parameterization: a = exp(z) / sum(exp(z))
    z = np.log(np.maximum(a, 1e-30))
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        def obj_grad(z_vec):
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            # Clamp to avoid numerical issues
            a_vec = np.maximum(a_vec, 1e-30)
            a_vec = a_vec / a_vec.sum()
            
            b = fast_conv(a_vec)
            k = min(topk, len(b))
            top_idx = np.argsort(b)[::-1][:k]
            top_vals = b[top_idx]
            max_v = top_vals.max()
            exp_v = np.exp(T * (top_vals - max_v))
            obj_val = max_v + np.log(exp_v.sum()) / T
            w = exp_v / exp_v.sum()
            
            grad_a = np.zeros(n)
            for idx, wK in zip(top_idx, w):
                js = max(0, idx - n + 1)
                je = min(n - 1, idx)
                i_range = np.arange(js, je + 1)
                grad_a[i_range] += 2.0 * wK * a_vec[idx - i_range]
            
            # Chain rule: dz -> da
            # a_i = exp(z_i) / S where S = sum_j exp(z_j)
            # da_i/dz_j = a_i * (delta_ij - a_j)
            S = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - S)
            
            return obj_val, grad_z
        
        # L-BFGS
        result = minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter_per_stage, 'gtol': 1e-6})
        
        z = result.x
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            z = np.log(np.maximum(a, 1e-30))
        
        elapsed = time.monotonic() - t0
        if si % 3 == 0:
            print(f"    LSE stage {si}/{n_stages} T={T:.1f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
