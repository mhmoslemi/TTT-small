_global_solver = None

def get_solver():

        return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):

        a = a / s
    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    L_init = n / 50.0
    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 10.0), n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 2000)
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = proj_simplex(z)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:

                else:
                    y_new = a_new.copy()
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if it % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()

        if verbose:

def build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def subgradient_mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent with KL (multiplicative weight) on max autocorrelation subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / (4.0 * np.sqrt(n))
    # Use exponentially weighted average
    avg_a = a.copy()
    decay = 0.99
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        M_curr = ac[k_star]
        # Subgradient
        g = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                g[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                g[i] += a[j2]
        g *= 2.0
        # Mirror descent step (multiplicative weight)
        # a_new_i = a_i * exp(-eta * g_i)
        # To avoid underflow, shift by max
        g_shifted = g - g.max()
        log_a = np.log(a + 1e-300)
        log_a_new = log_a - eta * g_shifted
        # Normalize
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new
        # Exponential average
        avg_a = decay * avg_a + (1 - decay) * a
        s_avg = avg_a.sum()
        if s_avg > 1e-15:
            avg_check = avg_a * norm / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    # Precompute which lags each position affects
    # For position i, it affects autocorrelation lags from |i-j| for all j
    # When we change a[i] by delta, the change in (a*a)(k) is:
    #   2 * delta * sum_{j: |i-j|=k} a[j] + delta^2 * [k=0]  (approx)
    # Actually (a*a)(k) = sum_i a[i] * a[i+k] (for the full convolution)
    # d(a*a)(k)/da[i] = a[i+k] + a[i-k] (where defined)
    # So the max constraint is most binding at k*, and we want to decrease a[i]
    # where a[i+k*] + a[i-k*] is large.
    stall = 0
    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        M_curr = ac[k_star]
        # Find the position i that most reduces the peak
        # Decreasing a[i] reduces (a*a)(k_star) by 2*a[i]*(a[i+k_star] + a[i-k_star]) / a[i]
        # = 2*(a[i+k_star] + a[i-k_star])
        # So we want to decrease a[i] where (a[i+k_star] + a[i-k_star]) is large
        # and a[i] is also large (to have meaningful effect)
        contrib = np.zeros(n)
        for i in range(n):
            c = 0.0
            j1 = i + k_star
            if j1 < n:
                c += a[j1]
            j2 = i - k_star
            if j2 >= 0:
                c += a[j2]
            contrib[i] = c
        # Try reducing the top contributors
        top_indices = np.argsort(contrib)[::-1][:min(50, n)]
        improved = False
        for i in top_indices:
            if a[i] < 1e-15:
                continue
            # Try reducing a[i] by various amounts
            for frac in [0.1, 0.05, 0.02, 0.01]:
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                # Redistribute to other positions (those with low contribution)
                low_idx = np.argsort(contrib)[:max(1, n // 10)]
                j = low_idx[np.random.randint(len(low_idx))]
                if j == i:
                    continue
                a_trial[j] += amt
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 200:
                break
    return best_a, best_score

def pairwise_swap_polish(a0, time_limit, seed=42, verbose=False):
    """Try pairwise swaps of mass between positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:

        top_k = np.argsort(ac)[::-1][:5]
        # Find positions contributing to peaks

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Compute full linear cross-correlation (a * b) via FFT."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:2 * n - 1]


def proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0
    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 10.0), n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 2000)
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            grad = lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = proj_simplex(z)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if it % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0

    sc = fast_autocorr(a).max()

def lp_best_response_fast(a, M, solver, n_active=300):
    """LP best response using top-n_active constraints with verification."""
    n = len(a)
    ac = fast_autocorr(a)
    n_use = min(n_active, 2 * n - 1)
    active = np.argsort(ac)[::-1][:n_use]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        length = je - js + 1
        if length <= 0:
            continue
        coeffs = np.asarray(a[k - je: k - js + 1][::-1], dtype=np.float64)
        constraints.append(coeffs @ b_var[js:je + 1] <= M)
    
    if len(constraints) == 0:
        return a.copy()
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    # Verify all constraints via FFT
    cross = cross_corr(a, b_val)
    max_cross = cross.max()
    if max_cross > M * 1.01 and max_cross > 0:
        # Scale down to satisfy constraints
        scale = M / max_cross
        b_val = b_val * scale
    return b_val


def lp_nesterov_fast(a0, time_limit, solver, verbose=True, n_active=300):
    """LP-best-response mirror descent with Nesterov acceleration (fast LP)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    lp_time_total = 0.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        t_lp_start = time.monotonic()
        b = lp_best_response_fast(y, M_curr, solver, n_active)
        t_lp_end = time.monotonic()
        lp_time_total += (t_lp_end - t_lp_start)
        
        if b is None:
            b = y.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Polyak average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp_time = lp_time_total / n_iters * 1000
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s, avg_lp={avg_lp_time:.1f}ms): best={best_score:.8f}")
    
    return best_a, best_score
