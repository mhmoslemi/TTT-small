from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

    ac = _fast_autocorr(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    return ac_max + T * np.log(exp_s.sum() / len(ac))

    # f = w0*(a*a) + w1*(a*b + b*a) + w2*(a*c + b*b + c*a) + w3*(b*c + c*b) + w4*(c*c)
    #   = w0*a^2 + 2*w1*a*b + w2*(a*c + b^2 + c*a) + 2*w3*b*c + w4*c^2
    #   = w0*a^2 + 2*w1*a*b + 2*w2*a*c + w2*b^2 + 2*w3*b*c + w4*c^2
    # 
    # df/da = 2*w0*a + 2*w1*b + 2*w2*c
    # df/db = 2*w1*a + 2*w2*b + 2*w3*c
    # df/dc = 2*w2*a + 2*w3*b + 2*w4*c
    #
    # So grad[j] = 2 * sum_{k} w[k] * y[k-j] (with appropriate bounds)
    # For j=0 (a): grad[0] = 2*(w0*a + w1*b + w2*c) = 2*sum_k w[k]*y[k] (k from 0 to 2)
    # For j=1 (b): grad[1] = 2*(w1*a + w2*b + w3*c) = 2*sum_k w[k+1]*y[k] (k from 0 to 2)
    # For j=2 (c): grad[2] = 2*(w2*a + w3*b + w4*c) = 2*sum_k w[k+2]*y[k] (k from 0 to 2)
    #
    # So grad[j] = 2 * sum_{k=0}^{n-1} w[k+j] * y[k]
    # This is the cross-correlation of w and y at lag j!
    # (w ⋆ y)[j] = sum_k w[k+j] * y[k]
    
    # This is: ifft(conj(FT(y)) * FT(w)) shifted appropriately
    # Or simply: the cross-correlation of w (length 2n-1) with y (length n)
    
    # Using FFT:
    # Let's zero-pad both to length >= (2n-1) + n - 1 = 3n-2
    # cross = ifft(FT(w) * conj(FT(y)))  -- this gives (w ⋆ y) in circular sense
    # For linear: we need the appropriate slice
    
    w_full = np.zeros(fft_size)
    w_full[:len(w)] = w
    y_full = np.zeros(fft_size)
    y_full[:n] = y
    
    W_full = rfft(w_full, fft_size)
    Y_full = rfft(y_full, fft_size)
    
    # Cross-correlation (w ⋆ y)[j] = sum_k w[k+j] * y[k]
    # = ifft(conj(Y_full) * W_full)... let me verify
    # ifft(conj(Y) * W)[j] = sum_k y[k] * w[k+j] (circular)
    # For linear (with enough zero-padding): this is the linear cross-correlation
    cross = irfft(np.conj(Y_full) * W_full, fft_size)
    # cross[j] = sum_{k=0}^{n-1} y[k] * w[k+j] (for j in [0, 2n-2])
    # We want grad[j] = 2 * cross[j] for j in [0, n-1]
    grad = 2.0 * cross[:n]
    return grad

def _mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights update) for minimizing max autocorr."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive step size
    eta0 = 5.0
    decay = 0.999
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        
        # Sub-gradient: grad[j] = sum_k w[k] * y[k+j] where w is delta at k_star
        # For the exact max (not smoothed), the sub-gradient is:
        # grad[j] = a[k_star - j] * [0<=k_star-j<n] + a[j - k_star] * [0<=j-k_star<n]
        # Wait, from my derivation above:
        # grad[j] = 2 * a[k_star + j] ... no
        # Let me redo: for the exact max at k*, the sub-gradient of conv_{k*}(a) w.r.t. a[j] is:
        # d/d a[j] (conv_{k*}) = a[k*-j] * [0<=k*-j<n] + a[j-k*] * [0<=j-k*<n]
        # (the second term is 0 if j-k* = k*-j, i.e., j = k*/2, in which case it's 2*a[j])
        
        # Actually from my earlier derivation:
        # conv_k = sum_{i+j'=k} a[i]*a[j']
        # d conv_k / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # If k-j = j (i.e., j = k/2): both terms are a[k/2], so total = 2*a[k/2]
        # If k-j ≠ j: terms are a[k-j] and a[j-k] (different indices)
        
        k = k_star
        # First term: a[k-j] for j in [max(0, k-n+1), min(k, n-1)]
        # Second term: a[j-k] for j in [k, min(2k-n+1, n-1)]... wait
        # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        # Combined: j in [0, n-1]
        
        idx1 = k - np.arange(n)  # k - j for j = 0, 1, ..., n-1
        idx2 = np.arange(n) - k  # j - k for j = 0, 1, ..., n-1
        
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        grad = np.zeros(n)
        grad[valid1] += a[idx1[valid1]]
        grad[valid2] += a[idx2[valid2]]
        
        # Multiplicative update
        eta = eta0 / (1.0 + it * 0.01)
        # Clip gradient to avoid overflow
        grad = np.clip(grad, -100.0, 100.0)
        mult = np.exp(-eta * grad / max(a.max(), 1e-10))
        a_new = a * mult
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        score = _fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror-Descent iter={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _lse_fista(a0, T_schedule, time_limit, verbose=False):
    """FISTA on LSE-smoothed max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        L = n / (T * 2.0)
        L = max(L, n / 50.0)
        t_fista = 1
        staging = a.copy()
        max_iters = max(2000, int(5000 * (T_schedule[0] / T) ** 0.2))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if it % 200 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def _random_perturb(a, rng, strength=0.01):
    """Apply random perturbation to escape local minima."""
    n = len(a)
    a_pert = a.copy()
    # Random small perturbations
    noise = rng.normal(0, strength, n)
    a_pert = a_pert + noise
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def _simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing with random perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    current_score = best_score
    t0 = time.monotonic()
    
    T_start = 0.05
    T_end = 0.001
    total_iters = 50000
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Random perturbation
        strength = T * 0.5
        a_trial = _random_perturb(a, rng, strength)
        score_trial = _fast_autocorr(a_trial).max()
        
        delta = score_trial - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={it} T={T:.5f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
