import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except (NameError, Exception):
        return float('inf')

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def peak_descent_mirror(a0, time_limit, lr_init=0.1, lr_decay=0.999, seed=42):
    """
    Mirror descent (multiplicative weights) with subgradient on top peaks.
    At each step, compute subgradient from the active peak(s) and apply
    multiplicative update.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find active peaks (within 0.1% of max)
        threshold = bmax * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient: average gradient over active peaks
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        
        # Multiplicative update
        updates = np.exp(-lr * grad)
        a_new = a * updates
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            a = a_new
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            a = a_new  # still move (to explore)
            stall += 1
        
        if stall % 100 == 0:
            lr *= lr_decay
            if lr < 1e-6:
                # Restart from best with random perturbation
                a = best_a.copy()
                noise = rng.normal(0, 0.005, n)
                a = a * (1 + noise)
                a = normalize(a)
                lr = lr_init * 0.5
                stall = 0
        
        if stall > 2000:
            break
        
        if int(time.monotonic() - t0) % 30 == 0:
            print(f"    Mirror t={int(time.monotonic()-t0)}s: best={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score

def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=40, topk=10, seed=123):
    """
    Optimize the top-k LogSumExp surrogate using Adam.
    The surrogate is: T * log(sum_{top-k} exp(b_k/T))
    which smoothly approximates max_k b_k.
    """
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Learning rate: larger for larger T, smaller for smaller T
        base_lr = 0.01 * min(1.0, 200.0 / n)
        lr = base_lr * (0.9 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        
        # Number of iterations per stage
        iters_per_stage = max(20, int(min(remaining * 0.3, 120.0) * 8))
        
        stall = 0
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            b = fast_autocorr(a)
            
            # Top-k LogSumExp
            k_use = min(topk, len(b))
            top_idx = np.argsort(b)[::-1][:k_use]
            top_vals = b[top_idx]
            
            # LSE value and weights
            max_tv = top_vals.max()
            exp_tv = np.exp((top_vals - max_tv) / stage_T)
            w = exp_tv / exp_tv.sum()
            
            # Gradient of LSE w.r.t. a:
            # d/d a[i] of sum_k w_k * b_k = sum_k w_k * d b_k / d a[i]
            # d b_k / d a[i] = 2 * a[k-i] (if 0 <= k-i < n)
            grad_a = np.zeros(n)
            for idx_i, k in enumerate(top_idx):
                wK = w[idx_i]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[k - idx_range]
            
            # Gradient in z-space: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic evaluation
            if (it + 1) % 15 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
                    stall = 0
                else:
                    stall += 1
                
                if stall > 200:
                    # Reset from best
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall = 0
                    break
        
        # End of stage
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
            z = np.log(np.maximum(best_a, 1e-30))
        
        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
