    grad_i = sum_k w_k * (a_{k-i} + a_{i-k})

      term1 = conv(a_flip, w)[n-1:2n-1]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def normalized_autocorr_score(a):
    """Compute 2*n*max(conv)/(sum)^2"""
    conv = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * conv.max() / (s * s)

def fft_lse_gradient(a, w, fft_size):
    """Compute gradient of LSE_T(conv(a,a)) via FFT.
    Returns grad of shape (n,).
    """
    n = len(a)
    a_flip = a[::-1]
    A = rfft(a, fft_size)
    AF = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    term2 = irfft(A * W, fft_size)[:n]
    term1 = irfft(AF * W, fft_size)[n-1:2*n-1]
    return 2.0 * (term1 + term2)

def lse_mirror_descent(a0, time_limit, T_start=30.0, T_end=0.5, verbose=True):
    """LSE mirror descent with FFT gradients and geometric temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T = T_start
    eta = 1.0
    n_iters = 0
    avg_a = np.zeros(n)
    n_avg = 0
    lse_val_prev = float('inf')
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        max_conv = conv.max()
        
        # LSE weights (with numerical stability)
        exp_conv = np.exp((conv - max_conv) / T)
        w_sum = exp_conv.sum()
        if w_sum < 1e-30:
            w = np.zeros(size)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = exp_conv / w_sum
        
        # LSE value
        lse_val = T * np.log(w_sum) + max_conv
        
        # Gradient via FFT
        grad = fft_lse_gradient(a, w, fft_size)
        
        # Mirror descent step
        log_update = -eta * grad
        log_update -= log_update.max()  # stability
        update = np.exp(log_update)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        n_iters += 1
        
        # Running average (Polyak averaging)
        avg_a += a
        n_avg += 1
        
        # Check score every 20 iterations
        if n_iters % 20 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Also check the average
            if n_avg % 100 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
        
        # Anneal temperature every 50 iterations
        if n_iters % 50 == 0:
            T = max(T_end, T * 0.98)
        
        # Decay eta
        if n_iters % 200 == 0:
            eta = max(0.1, eta * 0.99)
        
        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-MD iter={n_iters} ({elapsed:.0f}s): T={T:.2f}, eta={eta:.3f}, best={best_score:.8f}")
    
    return best_a, best_score

def build_conv_matrix(a):
    """Build the convolution matrix T where T[k,i] gives contribution of a[i] to conv[k].
    T @ b = conv(a, b) for any b of the same length.
    """
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
