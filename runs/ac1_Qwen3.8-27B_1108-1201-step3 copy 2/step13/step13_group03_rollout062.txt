    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def _max_conv(a):
    """Compute max of autocorrelation."""
    return _fast_conv(a).max()


def _proj_simplex(v, target_sum=1.0):
    """Project onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out *= target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out


def _lse_smooth(a, T):
    """Log-sum-exp smoothed max of autocorrelation."""
    ac = _fast_conv(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    return ac_max + T * np.log(np.exp(shifted).sum())

def _lp_best_response(a, M, solver):
