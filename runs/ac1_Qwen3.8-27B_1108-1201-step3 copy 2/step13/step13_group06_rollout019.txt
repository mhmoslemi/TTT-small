import numpy as np

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""

        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def nesterov_lp_game(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP zero-sum game.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_maxb = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    # Nesterov average: x_bar = (1/t) * sum_{s=1}^{t} x_s (of extrapolated iterates)
    nesterov_avg = a.copy()

    stall_count = 0
    last_print = 0
    iter_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_response(y, solver)
        if b_star is None:
            b_star = np.ones(n) / n
            M_val = float(fast_autocorr(y).max())

        iter_count += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        # Update Nesterov average: x_bar_t = ((t-1)/t) * x_bar_{t-1} + (1/t) * x_t
        nesterov_avg = ((t - 1.0) / t) * nesterov_avg + (1.0 / t) * a_new
        na = nesterov_avg.sum()
        if na > 1e-15:
            nesterov_avg = nesterov_avg / na

        y = y_new
        a = a_new

        # Check multiple candidates
        for candidate in [a, y, nesterov_avg]:
            cand = np.maximum(candidate, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            cand_maxb = float(fast_autocorr(cand).max())
            if cand_maxb < best_maxb:
                best_maxb = cand_maxb
                best_a = cand.copy()
                stall_count = 0
        else:
            stall_count += 1

        if iter_count - last_print > 2000:
            elapsed = time.monotonic() - t0
            y_mb = float(fast_autocorr(y).max())
            print(f"    LP iter={iter_count} ({elapsed:.0f}s): best={best_maxb:.8f}, y={y_mb:.8f}, M={M_val:.8f}")
            last_print = iter_count

        if stall_count > 3000 and t > 500:
            print(f"    LP restart at iter={iter_count}, best={best_maxb:.8f}")
            a = best_a.copy()
            y = best_a.copy()
            nesterov_avg = best_a.copy()
            warm_b = None
            stall_count = 0

    return best_a, best_maxb


def coordinate_descent_polish(a0, time_limit, n_inner=5):
    """
    Coordinate descent: for each coordinate j, optimize a_j given all others fixed.
    The objective max_k (a*a)_k as a function of a_j is the max of quadratics.
    We use a bisection/gradient approach to find the best a_j.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_autocorr(a).max())
    t0 = time.monotonic()
    n_improve = 0
    n_stagnate = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            a_j = a[j]
            if a_j < 1e-12:
                continue

            # Current autocorrelation
            b = fast_autocorr(a)
            bmax = b.max()

            # Effect of changing a_j: b_k changes by 2*a_j*delta*a_j_contribution + delta^2
            # Actually: if we change a_j to a_j + delta, then
            # b_k changes by 2*delta*(sum of a_{k-j} for valid indices) + delta^2 (if k=2j or appropriate)
            # More precisely: b_k = sum_i a_i * a_{k-i}
            # db_k/d(a_j) = sum_i [delta(i,j)*a_{k-i} + a_i*delta(k-i,j)]
            #             = a_{k-j} + a_{k-j} (if k-j valid in both) ... 
            # Actually: b_k = sum_{i=0}^{n-1} a_i * a_{k-i} (with a_x = 0 if x not in [0,n-1])
            # db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (if 0 <= k-j < n)
            # Wait, that's not right either. Let me be careful.
            # b_k = sum_i a_i * a_{k-i}
            # db_k/da_j: differentiate w.r.t. a_j
            # = sum_i [delta(i,j)*a_{k-i} + a_i*delta(k-i, j)]
            # = a_{k-j} + sum_i a_i * delta(k-i, j)
            # = a_{k-j} + a_{k-j} (when i = k-j)
            # = 2 * a_{k-j} (if 0 <= k-j <= n-1)
            # So db_k/da_j = 2 * a_{k-j} if 0 <= k-j < n, else 0.

            # The second derivative: d2b_k/da_j^2 = 2 if k = 2j (since a_j appears twice in the product)
            # Actually d2b_k/da_j^2 = 0 for all k (since b_k is linear in each a_j when others are fixed)
            # Wait no: b_k = sum_i a_i * a_{k-i}. If we write this out:
            # b_k includes the term a_j * a_{k-j} (when i=j) and a_{k-j} * a_j (when i=k-j, if k-j != j)
            # If k-j == j (i.e., k = 2j), then b_k includes a_j^2, so d2b_k/da_j^2 = 2.
            # Otherwise, b_k is linear in a_j (appears only once), so d2b_k/da_j^2 = 0.

            # So the objective max_k b_k as a function of a_j (with others fixed) is:
            # max_k [b_k + 2*a_{k-j}*delta + (2 if k==2j else 0)*delta^2/2]
            # = max_k [b_k + 2*a_{k-j}*delta + (1 if k==2j else 0)*delta^2]

            # This is the max of linear/quadratic functions of delta. We want to minimize it
            # subject to a_j + delta >= 0 and sum constraint.

            # Simple approach: try a few values of delta and pick the best.
            # The optimal delta is where the max is minimized. For the max of lines,
            # the optimum is where the highest lines cross.

            # For now, use a simple 1D search
            best_delta = 0.0
            best_local = bmax
            delta_scale = a_j * 0.5
            for frac in [-0.4, -0.3, -0.2, -0.15, -0.1, -0.05, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
                delta = delta_scale * frac
                if a_j + delta < -1e-12:
                    continue
                a_trial = a.copy()
                a_trial[j] = max(0.0, a_j + delta)
                # Renormalize
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                nb = fast_autocorr(a_trial)
                nbmax = float(nb.max())
                if nbmax < best_local:
                    best_local = nbmax
                    best_delta = delta

            if best_delta != 0.0:
                a[j] = max(0.0, a[j] + best_delta)
                st = a.sum()
                if st > 1e-12:
                    a = a / st
                score = 2.0 * n * float(fast_autocorr(a).max())
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            n_stagnate += 1
            if n_stagnate > 3:
                break
        else:
            n_stagnate = 0

    return best_a, best_score


def pairwise_transfer_polish(a0, time_limit, seed=42):
    """
    Try pairwise mass transfers: move small amounts from position i to position j.
    Only accept if global peak decreases.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_maxb = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        top_frac = 0.98
        threshold = bmax * top_frac
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) < 5:
            top_peaks = np.argsort(b)[::-1][:20]

        # Compute importance map
        importance = np.zeros(n)
        for k in top_peaks[:30]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[idx] * a[k - idx]

        improved = False
        # Try random pairs weighted by importance
        n_attempts = 200
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break

            # Source: high importance
            src_choices = np.argsort(importance)[::-1][:max(20, n // 10)]
            src = src_choices[rng.integers(len(src_choices))]
            if a[src] < 1e-10:
                continue

            # Destination: low importance
            low_mask = importance < importance.mean() * 0.5
            if low_mask.sum() < 10:
                low_mask = importance < np.percentile(importance, 50)
            dst_pool = np.where(low_mask)[0]
            dst_pool = dst_pool[dst_pool != src]
            if len(dst_pool) == 0:
                dst = rng.integers(n)
                if dst == src:
                    continue
            else:
                dst = dst_pool[rng.integers(len(dst_pool))]

            # Try multiple fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] = a[src] - amt
                a_trial[dst] = a[dst] + amt
                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                nbmax = float(fast_autocorr(a_trial).max())
                if nbmax < bmax * (1 - 1e-9):
                    a = a_trial
                    if nbmax < best_maxb:
                        best_maxb = nbmax
                        best_a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0

    return best_a, best_maxb


def freq_smooth_polish(a0, time_limit):
    """
    Frequency-domain smoothing: flatten the power spectrum slightly.
    a -> IFFT(|FFT(a)|^(1-alpha) * phase) then project to simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_maxb = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    for alpha in [0.05, 0.1, 0.15, 0.2, 0.03, 0.07, 0.12]:
        if time.monotonic() - t0 > time_limit:
            break

        fft_size = 1 << (2 * n - 2).bit_length()
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Smooth: reduce peaks in magnitude
        mag_smooth = mag ** (1 - alpha)
        # Add a small floor to avoid zeroing
        mag_floor = mag_smooth.max() * 0.01
        mag_smooth = np.maximum(mag_smooth, mag_floor)

        A_new = mag_smooth * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn

        nbmax = float(fast_autocorr(a_new).max())
        if nbmax < best_maxb:
            best_maxb = nbmax
            best_a = a_new.copy()

    # Try combining with original
    for w in [0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - t0 > time_limit:
            break
        a_mix = (1 - w) * a + w * best_a
        a_mix = np.maximum(a_mix, 0)
        sm = a_mix.sum()
        if sm < 1e-15:
            continue
        a_mix = a_mix / sm
        nbmax = float(fast_autocorr(a_mix).max())
        if nbmax < best_maxb:
            best_maxb = nbmax
            best_a = a_mix.copy()

    return best_a, best_maxb


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a0, rng, scale=0.05):
    """Create a perturbed version."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a * (1 + noise)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def lse_gradient_step(a, T_temp, topk, lr):
    """One step of LSE-smoothed gradient descent in log space."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    max_Tc = T_temp * top_vals.max()
    exp_Tc = np.exp(T_temp * top_vals - max_Tc)
    w = exp_Tc / exp_Tc.sum()

    # Gradient of (1/T)log(sum exp(T*c_k)) w.r.t. a_j
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    # Project gradient to simplex-tangent (remove component along (1,...,1))
    S = np.dot(grad_a, a)
    grad_proj = grad_a - S  # tangent to simplex at a

    # Log-space update
    z = np.log(np.maximum(a, 1e-30))
    z -= lr * grad_proj / np.maximum(a, 1e-10)
    a_new = np.exp(z - z.max())
    sn = a_new.sum()
    if sn < 1e-15:
        return a
    return a_new / sn


def lse_refine(a0, time_limit, T_start=5.0, T_end=0.1, n_stages=15, topk=15):
    """LSE-smoothed gradient descent refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_maxb = float(fast_autocorr(a).max())
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        lr = 0.01 * min(1.0, 50.0 / n)
        max_steps = int(min(remaining, 30.0) * 3)

        for step in range(max_steps):
            if time.monotonic() - t0 > time_limit - 1:
                break
            a_new = lse_gradient_step(a, stage_T, topk, lr)
            nbmax = float(fast_autocorr(a_new).max())
            if nbmax < best_maxb:
                best_maxb = nbmax
                best_a = a_new.copy()
            a = a_new

    return best_a, best_maxb


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: LP Nesterov game with constraint generation (main)
    Phase 2: LSE gradient refinement
    Phase 3: Coordinate descent + pairwise transfer polishing
    Phase 4: Frequency smoothing
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    t_global = time.monotonic()
    time_budget = 1700.0

    solver = find_solver()
    print(f"Solver: {solver}")

    rng = np.random.default_rng(12345)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent.tolist())
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        best_seq = None

    n_base = len(best_seq) if best_seq is not None else 1600
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq.tolist())
        print(f"Initialized uniform: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP Nesterov Game ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.55, 850.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_game(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"=== Phase 2: LSE refine n={n_base}, budget={t2:.0f}s ===")
    if t2 > 15:
        a_out, _ = lse_refine(best_seq, t2, T_start=5.0, T_end=0.1, n_stages=12, topk=20)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Pairwise transfer ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 150.0)
    print(f"=== Phase 3: Pairwise transfer, budget={t3:.0f}s ===")
    if t3 > 10:
        a_out, _ = pairwise_transfer_polish(best_seq, t3, seed=777)
        try_update(a_out, "pairwise_1")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"=== Phase 4: Coord descent, budget={t4:.0f}s ===")
    if t4 > 10:
        a_out, _ = coordinate_descent_polish(best_seq, t4)
        try_update(a_out, "coord_1")

    # === Phase 5: Frequency smoothing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 60.0)
    print(f"=== Phase 5: Freq smooth, budget={t5:.0f}s ===")
    if t5 > 5:
        a_out, _ = freq_smooth_polish(best_seq, t5)
        try_update(a_out, "freq_smooth")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Multi-length, budget={remaining:.0f}s ===")
    if remaining > 60:
        for n_try in [n_base + 100, n_base + 200, n_base - 100]:
            if n_try < 500 or time.monotonic() - t_global > t_global + time_budget - 30:
                continue
            t_each = min(remaining * 0.25, 120.0)
            if t_each < 20:
                continue
            print(f"  Trying n={n_try}")
            init = interpolate_sequence(best_seq, n_try)
            a_out, _ = nesterov_lp_game(init, t_each, solver)
            try_update(a_out, f"lp_n{n_try}")
            # Quick LSE
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = lse_refine(a_out, min(remaining * 0.5, 40.0), T_start=3.0, T_end=0.2, n_stages=8, topk=15)
                try_update(a_out2, f"lse_n{n_try}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.3, 100.0)
    if t7 > 10:
        print(f"=== Phase 7: Final LP polish, budget={t7:.0f}s ===")
        a_out, _ = nesterov_lp_game(best_seq, t7, solver)
        try_update(a_out, "lp_final")

    # === Phase 8: Final pairwise ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.4, 60.0)
    if t8 > 5:
        print(f"=== Phase 8: Final pairwise, budget={t8:.0f}s ===")
        a_out, _ = pairwise_transfer_polish(best_seq, t8, seed=999)
        try_update(a_out, "pairwise_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq.tolist())
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
