import numpy as np
import scipy.fft as spfft
import time

def fast_autocorr(a):
    """Compute full convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = spfft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return spfft.irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def lse_grad(y, T):
    """Gradient of log-sum-exp smoothed max autocorrelation."""
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = spfft.rfft(y, fft_size)
    W = spfft.rfft(w, fft_size)
    cross = spfft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * cross[n-1: n-1+n]

            t_fista += 1
            if it % 500 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            print(f"    LSE stage {stage_i} T={T:.2f}: best={best_score:.8f}")
    
    return best_a, best_score

def build_conv_matrix(a, n):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Adaptive Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 50 == 0:
            print(f"    LP-Nest iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def aggressive_peak_shave(a, time_limit, seed=42):
    """Aggressive peak reduction with binary search on step sizes."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find all indices that contribute to this peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Try both low-mass destinations (to spread mass) and 
            # strategically chosen destinations
            lows = candidates[np.argsort(a[candidates])][:max(5, len(candidates)//8)]
            highs = candidates[np.argsort(a[candidates])[::-1]][:max(5, len(candidates)//8)]
            targets = list(set(list(lows) + list(highs)))
            
            for dst in targets[:20]:
                # Binary search for the maximum step that reduces the score
                lo, hi = 0.0, a[src]
                best_step = 0
                best_trial_score = best_score
                
                for _ in range(8):
                    mid = (lo + hi) / 2
                    if mid < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    if a_trial[src] < -1e-15:
                        hi = mid
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_step = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_step > 0 and best_trial_score < best_score - 1e-15:
                    a = a.copy()
                    a[src] -= best_step
                    a[dst] += best_step
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    a = a * norm / s_t
                    best_score = fast_autocorr(a).max()
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial * norm / a_trial.sum()
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            
            if stall > 200:
                break
    
    return best_a, best_score

def coordinate_descent_max(a, time_limit, verbose=False):
    """Coordinate descent focusing on the maximum lag."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        # Gradient of c[k_star] w.r.t. a
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        for i in range(js, je + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        # Find the best single-coordinate move
        # Moving mass from i to j changes c[k] by:
        # -2*amt*a[k-i] + 2*amt*a[k-j] (approximately, for small amt)
        # Plus second-order terms
        
        best_gain = 0
        best_move = None
        
        # For each source i (where grad is high) and destination j (where grad is low)
        top_src = np.argsort(grad)[::-1][:20]
        top_dst = np.argsort(grad)[:20]
        
        for i in top_src:
            if a[i] < 1e-15:
                continue
            for j in top_dst:
                if i == j:
                    continue
                # Linear approximation of the change in c[k_star]
                # Moving amt from i to j: delta c ≈ amt * (grad[j] - grad[i])
                delta = grad[j] - grad[i]
                if delta < 0 and abs(delta) > best_gain:
                    best_gain = abs(delta)
                    best_move = (i, j, delta)
        
        if best_move is None:
            break
        
        i, j, delta = best_move
        # Binary search for optimal step
        lo, hi = 0.0, a[i]
        best_amt = 0
        best_sc = best_score
        
        for _ in range(10):
            mid = (lo + hi) / 2
            if mid < 1e-15:
                break
            a_trial = a.copy()
            a_trial[i] -= mid
            a_trial[j] += mid
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_sc:
                best_sc = sc
                best_amt = mid
                lo = mid
            else:
                hi = mid
        
        if best_amt > 0 and best_sc < best_score:
            a[i] -= best_amt
            a[j] += best_amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            best_score = fast_autocorr(a).max()
            best_a = a.copy()
        else:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.CLARABEL
