    seq = np.asarray(seq, dtype=np.float64)

    gamma = t / (np.sqrt(t) + 1.0)  # scaling from the problem
    
    # Create new distribution on simplex
    log_a = np.log(a_norm + 1e-20)
    log_a_new = log_a - gamma * np.asarray(compute_soft_subgrad(a_norm, score, top_k=top_k, beta=beta))
    new_a = np.exp(log_a_new - np.max(log_a_new))
    new_a = new_a * np.sqrt(2.0*n) / new_a.sum()  # normalization required for evaluation
    
    # Compare the new distribution
    new_score = 2.0 * n * max(np.convolve(new_a, new_a)) / (new_a.sum())**2
    
    if new_score < best_score:
        best_score = new_score
        best_candidate = new_a
    else:
        # System may not converge, so keep original
        pass
    
    # Track time
    a_norm = new_a
    gamma increases, so break after steps
    
    # Print progress every 10000 updates, and also if time budget about to expire
    if t % 10000 == 0:
        print(f"Mirror descent iteration {t}: J={best_score:.6f}")
    
    if time_budget is not None and (max_iter - t) / 10000 == 0:
        print("Time used up, wrapping up optimization")
        break

return best_candidate if best_score < float('inf') else []
