```python
def subgradient_direction(a, b, n):
    """Compute subgradient of max((a*a)_k) with respect to a."""
    max_b = b.max()
    # Find active set (indices where b_k is close to max)
    tol = 1e-6
    active = np.where(b >= max_b - tol)[0]
    
    g = np.zeros(n)
    for k in active:
        # d(a*a)_k / da_m = a[k-m] + a[m-k] (for valid indices)
        # For m from 0 to n-1:
        #   a[k-m] is valid if 0 <= k-m < n, i.e., k-n+1 <= m <= k
        #   a[m-k] is valid if 0 <= m-k < n, i.e., k <= m <= k+n-1
        for m in range(n):
            if 0 <= k - m < n:
                g[m] += a[k - m]
            if 0 <= m - k < n:
                g[m] += a[m - k]
    g /= len(active)
    return g
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using correct convolution-based
LSE gradient descent with Nesterov acceleration, combined with
peak-targeting local search using the correct subgradient.

Key fix: The previous code computed the gradient of the autocorrelation
(a * a_rev) instead of the convolution (a * a). This was a fundamental
error that limited the achievable score.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_lse_value(a, T):
    """Value of LSE surrogate for convolution."""
    b = fast_conv(a)
    max_b = b.max()
    if max_b < 1e-300:
        return 0.0
    return T * np.log(np.sum(np.exp((b - max_b) / T))) + max_b

def conv_lse_gradient(a, T):
    """
    Correct gradient of LSE of convolution (a*a).
    d(T*log(sum exp(b_k/T)))/da_m = 2 * sum_j a[j] * w[m+j]
    where w = softmax(b/T) and b = a*a.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    exponents = (b - max_b) / T
    w = np.exp(exponents)
    W = w.sum()
    if W < 1e-300:
        # All weights are effectively zero, use uniform
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= W
    
    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # = 2 * (a * w_rev)[2n-2-m]
    w_rev = w[::-1]
    fft_size = 1 << (3*n).bit_length()
    A = rfft(a, fft_size)
    WR = rfft(w_rev, fft_size)
    conv_result = irfft(A * WR, fft_size)
    
    # Need conv_result[2n-2-m] for m=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    
    return grad

def conv_subgradient(a, b, tol=1e-8):
    """
    Compute a subgradient of max_k (a*a)_k.
    Returns the average gradient over the active set.
    """
    n = len(a)
    max_b = b.max()
    active = np.where(b >= max_b - tol * max(max_b, 1.0))[0]
    
    g = np.zeros(n)
    for k in active:
        # d(a*a)_k / da_m = 2 * a[k-m] if 0 <= m < n and 0 <= k-m < n
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            # g[m] += 2 * a[k-m] for m in [m_start, m_end]
            # a[k-m] for m = m_start,...,m_end gives a[k-m_start], ..., a[k-m_end]
            # = a[k-m_start : k-m_end-1 : -1]  (reversed)
            # k-m_start down to k-m_end
            idx = k - np.arange(m_start, m_end + 1)
            g[m_start:m_end+1] += 2.0 * a[idx]
    
    g /= len(active)
    return g

def nesterov_lse_conv(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=15, verbose=True):
    """Nesterov-accelerated LSE gradient descent on the correct convolution objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    it_count = 0
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters_stage = max(30, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov extrapolation
            if it == 0:
                y_extrap = y
            else:
                beta = (1.0 - np.sqrt(1.0 + 4.0 * (it-1)**2)) / 2.0 + 1.0  # standard Nesterov beta
                # Actually, standard Nesterov: y_{k+1} = x_{k+1} + (t_{k+1}-1)/t_{k+2} * (x_{k+1} - x_k)
                # where t_{k+1} = (1+sqrt(1+4k^2))/2
                # Let's use the simpler formulation:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0)**2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s = y_extrap.sum()
                if s > 1e-15:
                    y_extrap = y_extrap * norm / s
            
            grad = conv_lse_gradient(y_extrap, T)
            lr = 1.0 / (it + 3.0)
            
            y_new = y - lr * grad
            y_prev = y.copy()
            
            # Project: clip to non-negative, rescale to norm
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            y = y_new
            it_count += 1
            
            # Track best (every few iterations to save time)
            if it_count % 5 == 0:
                score = fast_conv(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()
                
                if verbose and it_count % 200 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    ConvLSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    # Final check
    score = fast_conv(y).max()
    if score < best_score:
        best_score = score
        best_a = y.copy()
    
    return best_a, best_score

def peak_targeting_conv(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search using correct convolution subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_b = b.max()
        tol = max(max_b * 1e-6, 1e-10)
        active = np.where(b >= max_b - tol)[0]
        
        improved = False
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                idx = k - np.arange(m_start, m_end + 1)
                g[m_start:m_end+1] += 2.0 * a[idx]
        g /= len(active)
        
        # Move mass from high-gradient positions to low-gradient positions
        # Find top contributors (high g) and low contributors (low g)
        g_sorted = np.argsort(g)
        n_move = max(1, n // 50)
        src_candidates = g_sorted[-n_move:]
        dst_candidates = g_sorted[:n_move]
        
        random_shuffle = rng.permutation(len(src_candidates))
        src_candidates = src_candidates[random_shuffle]
        dst_candidates = dst_candidates[rng.permutation(len(dst_candidates))]
        
        found_improvement = False
        for src in src_candidates:
            if found_improvement:
                break
            if a[src] < norm * 1e-10:
                continue
            for dst in dst_candidates:
                if found_improvement:
                    break
                if src == dst:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-16:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        found_improvement = True
                        it_count += 1
                        break
        
        # Also try random perturbations
        if not improved and rng.integers(5) == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it_count += 1
        
        if not improved:
            stall += 1
            if stall > 100:
                if verbose:
                    print(f"    PeakConv: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break
        else:
            stall = 0
        
        if verbose and it_count % 200 == 0:
            print(f"    PeakConv it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
```

    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    exponents = (b - max_b) / T
    w = np.exp(exponents)
    W = w.sum()
    if W < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / W

    w_rev = w[::-1]
    fft_size = 1 << (3 * n + 2).bit_length()
    A = rfft(a, fft_size)
    WR = rfft(w_rev, fft_size)
    conv_result = irfft(A * WR, fft_size)

    # grad[m] = 2 * conv_result[2n-2-m] for m=0,...,n-1
    # indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad


def conv_subgradient(a, b, rel_tol=1e-6):
    """
    Compute a subgradient of max_k (a*a)_k.
    Average over the active set (indices where b_k is near max).
    """
    n = len(a)
    max_b = b.max()
    threshold = max_b - rel_tol * max(max_b, 1e-10)
    active = np.where(b >= threshold)[0]

    g = np.zeros(n)
    for k in active:
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            g[m_start:m_end + 1] += 2.0 * a[idx]

    g /= len(active)
    return g
