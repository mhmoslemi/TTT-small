def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_reduced(a, M, solver, max_active=200):
    """LP best response with aggressive constraint reduction via active set iteration."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    # Start with all constraints
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    # Iteratively reduce to active constraints
    for iteration in range(5):
        circ = T @ b_val
        tol = 1e-7 * M
        active = circ >= M - tol
        n_active = int(active.sum())

        # If too many or too few active, stop
        if n_active >= min(size, max_active + 50) or n_active < 10:
            break

        # Keep top max_active active constraints
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top_idx] = True

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_active=200):
    """Fessler-Parks accelerated LP Nesterov with adaptive restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    M_hist = []
    gamma = 1.8  # Fessler-Parks parameter (2.0 is standard Nesterov)

    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1

        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        # Solve LP
        b = lp_solve_reduced(y, M_curr, solver, max_active=max_active)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        if alpha > 0.5:
            alpha = 0.5

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum step
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            if momentum > 1.0:
                momentum = 1.0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma: reduce if oscillating
        if len(M_hist) > 100:
            recent = M_hist[-50:]
            osc = np.sum(np.abs(np.diff(recent))) / (50 * max(recent[0], 1e-15))
            if osc > 0.05:
                gamma = max(1.0, gamma - 0.1)
            elif osc < 0.01:
                gamma = min(2.0, gamma + 0.05)

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}, stall={stall}")

        # Adaptive restart
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.8
            M_hist = []
            if verbose:
                print(f"    [Restart at iter {n_iters}, gamma reset]")

    return best_a, best_score


def structured_subgradient(a0, time_limit, verbose=True, n_wn=5000):
    """
    Structured subgradient method exploiting the geometry of autocorrelation gradients.
    Uses the fact that d/d a_j [c_k] = 2 * a_{k-j} (shifted copy of a).
    Employs polyak-type step sizes and multi-peak averaging.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    M_lower = 2 * n * best_score / (norm ** 2)  # Theoretical lower bound in M-space
    t0 = time.monotonic()
    steps = 0

    # Initialize step size estimate
    step_size = 0.1 / n
    M_min = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)

        b = fast_autocorr(a)
        max_b = b.max()

        # Find top K peaks for multi-peak subgradient
        K = min(5, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:K]
        peak_vals = b[top_peaks]

        # Softmax weighting (temperature annealed)
        T_speed = max(0.01, 0.1 * (max_b - peak_vals.min() + 1e-10) * (1.0 - elapsed_frac))
        w = np.exp((peak_vals - max_b) / T_speed)
        w = w / w.sum()

        # Compute combined subgradient
        # grad_j = sum_m w_m * 2 * a_{p_m - j} for valid indices
        grad = np.zeros(n)
        for m, p in enumerate(top_peaks):
            if w[m] < 1e-6:
                continue
            # grad contribution: 2 * a_{p-j} for j where 0 <= p-j < n
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js > je:
                continue
            grad[js:je + 1] += 2.0 * w[m] * a[p - np.arange(js, je + 1)]

        # Project gradient onto tangent space of {sum a = norm}
        grad -= np.mean(grad)

        # Polyak-type step: step = (M_curr - M_lower) / ||grad||^2
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        polyak_step = (max_b - best_score * 0.999) / grad_norm_sq
        step = min(step_size, polyak_step) if polyak_step > 0 else step_size
        step = max(step, 1e-6 / n)

        # Take step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()

        # Accept if improved, or with annealing
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            M_min = min(M_min, score_new)
            step_size = min(step_size * 1.05, 1.0 / n)
        elif score_new < max_b * 1.001:
            # Accept small worsening (annealing)
            accept_prob = np.exp(-(score_new - best_score) / max(0.001 * best_score, 1e-10))
            if np.random.random() < accept_prob * (1.0 - elapsed_frac):
                a = a_new
                step_size *= 0.7
            else:
                step_size *= 0.8
        else:
            step_size *= 0.5

        steps += 1

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, M={max_b:.6f}, step={step:.2e}")

        # Escape stagnant regions
        if steps % 2000 == 0 and step_size < 1e-10 / n:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
            a = a * norm / a.sum()
            step_size = 0.1 / n

    return best_a, best_score

        sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        # Line search: try different fractions
        best_trial_score = max_b
        best_frac = 0
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = frac

        if best_frac > 0:
            a_trial = a.copy()
            a_trial[src] -= src_val * best_frac
            a_trial[dst] += src_val * best_frac
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()

            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                n_improve += 1
                n_stall = 0
            else:
                # Annealing acceptance
                T = best_score * 0.001 * (1.0 - elapsed_frac)
                if rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
                    a = a_trial
                    n_stall += 1
                else:
                    n_stall += 1
        else:
            n_stall += 1

        # Periodic random exploration
        if n_stall > 50:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 0:
                amt = a[i] * rng.uniform(0.05, 0.2)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
                        n_stall = 0
                    else:
                        n_stall += 1

        if n_stall > 300:
            # Larger perturbation
            n_pert = max(5, n // 50)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.5)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.02:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            n_stall = 0

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LogSumExp-smoothed gradient descent via FFT-based gradients."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.001, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient via FFT
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]

        step = 1.0 / (T * n * 0.3)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_active=150)
        do_try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Structured subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Structured SubGrad n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = structured_subgradient(a_norm, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, max_active=100)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Try longer length ===
    n2 = n_base + 100
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, max_active=120)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 5: Structured subgradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: SubGrad n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = structured_subgradient(a_norm, t5, verbose=True)
        do_try_update(a_out, f"subgrad2_n{n_cur}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t6, seed=999, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # === Phase 7: LSE gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: LSE-GD n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.08, 90.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t7, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, max_active=80)
        do_try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 9: Final PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.15, 100.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t9, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
