- This is a "bottleneck" problem: find delta such that the maximum of the lines conv[k] + delta*g[k] is minimized
- Can be solved by binary search on the value: for a given target value v, check if there exists delta such that conv[k] + delta*g[k] <= v for all k. This gives delta <= (v - conv[k])/g[k] for g[k] > 0 and delta >= (v - conv[k])/g[k] for g[k] < 0. So we need max_{k: g[k]<0} (v-conv[k])/g[k] <= min_{k: g[k]>0} (v-conv[k])/g[k].

This is solvable in O(n) per pair (compute the max of ratios and min of ratios, then binary search or solve directly).

But actually, for the practical implementation, a simple approach works:
- The current max is M = conv[k*] for some k*.
- If g[k*] > 0, then increasing delta makes the peak worse, so we should decrease delta (or move in the other direction).
- If g[k*] < 0, then increasing delta helps at the peak.
- The optimal delta (first-order) is: delta* = (M - conv[k2]) / (g[k*] - g[k2]) where k2 is the next highest conv value. Or more simply, delta* = min over k of (M - conv[k]) / g[k] for g[k] > 0 (the delta at which some constraint becomes binding).

For simplicity, I'll just use a small grid of delta values (e.g., 10 points in [0, a_i/10] for small perturbations, plus the boundary a_i).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE (log-sum-exp) gradient descent with Nesterov momentum
2. Systematic windowed pair coordinate descent (full sweep, not random)
3. LP best-response for global structure moves
4. Adaptive temperature scheduling and perturbation restarts
Key improvements over previous: O(n log n) gradient via FFT, systematic CD sweep,
better momentum scheduling, and perturbation-based escape from local optima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        for x in a_list:
            if x < 0 or not np.isfinite(x):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_gradient_fft(a, conv, T):
    """
    Compute gradient of LSE_T(conv(a,a)) w.r.t. a using FFT.
    grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k])
    where w = softmax(conv / T)
    Returns the gradient vector of length n.
    """
    n = len(a)
    size = 2 * n - 1
    # Softmax weights
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum

    # We need:
    # B[i] = (w * a)[i] = sum_j w[j] * a[i-j]  (full conv of w and a, first n elements)
    # A[i] = sum_k w[k] * a[k-i]  (cross-correlation at lag -i)
    #
    # B_full = IFFT(FFT(w) * FFT(a)) -- full convolution
    # C = IFFT(FFT(w) * conj(FFT(a))) -- circular cross-correlation
    #   C[m] = sum_k w[k] * a[(k+m) mod N]
    #   A[i] = C[(N-i) mod N] for linear cross-correlation (when N is large enough)

    N = 1 << (size + n - 1).bit_length()  # >= 3n-2

    W = rfft(w, N)
    A_fft = rfft(a, N)

    # Full convolution B
    B_full = irfft(W * A_fft, N)[:n]

    # Cross-correlation C
    C = irfft(W * np.conj(A_fft), N)

    # A[i] = C[(N - i) % N]
    idx = (N - np.arange(n)) % N
    A_vals = C[idx]

    grad = 2.0 * (A_vals + B_full)
    return grad


def lse_nesterov_optimize(a0, time_limit, verbose=True):
    """
    LSE gradient descent with Nesterov momentum and adaptive temperature.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    v = np.zeros(n)  # momentum
    avg_a = np.zeros(n)
    n_avg = 0

    # Temperature schedule: start high, anneal down
    T_start = 5.0
    T_end = 0.3
    n_total_iters = 20000
    iter_count = 0

    # Base step size (will be adapted)
    base_lr = 0.5 / n

    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count = it + 1

        # Anneal temperature
        frac = it / max(1, n_total_iters - 1)
        T = T_start * (T_end / T_start) ** frac

        # Compute LSE gradient
        conv_y = fast_autocorr(y)
        grad = lse_gradient_fft(y, conv_y, T)

        # Nesterov momentum
        mu = 0.9 + 0.05 * (1 - frac)  # start with 0.95, decrease to 0.9
        lr = base_lr * (1.0 + 0.5 * (1 - frac))

        v = mu * v + lr * grad
        y_new = y - v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Evaluate and track best
        score_y = fast_autocorr(y_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()

        # Averaged iterate (Polyak averaging)
        avg_a += y_new
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        y = y_new

        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter={it+1} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def systematic_pair_cd(a0, time_limit, rng, verbose=True):
    """
    Systematic windowed pair coordinate descent.
    For each position i, try transferring mass to nearby positions j.
    Uses exact 1D optimization (grid + refine).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Window size for CD (how far from i to look for j)
    window = min(200, n // 4)
    n_deltas = 16

    n_passes = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Sweep order: random start point, wrap around
        start = int(rng.integers(n))
        order = [(start + k) % n for k in range(n)]

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            i = idx
            if a[i] < norm * 1e-10:
                continue

            ai = a[i]
            # Precompute gain vector g[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k] for each j
            # For efficiency, only try a subset of j values per pass
            n_try = min(30, 2 * window + 1)
            j_candidates = [(i + 1 + d) % n for d in range(window)]
            # Shuffle for diversity
            rng.shuffle(j_candidates)
            j_candidates = j_candidates[:n_try]

            for j in j_candidates:
                if j == i or a[j] >= norm * 0.99:
                    continue

                # Compute g[k] = change in conv[k] per unit delta transferred from i to j
                # g[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k] (with zero padding)
                g = np.zeros(size)

                # a[k-j] for k in [j, j+n-1]
                lo1 = j
                hi1 = min(size, j + n)
                if hi1 > lo1:
                    g[lo1:hi1] += a[:hi1 - lo1]

                # a[j-k] for k in [max(0, j-n+1), j]
                lo2 = max(0, j - n + 1)
                hi2 = j + 1
                if hi2 > lo2:
                    src = j - np.arange(lo2, hi2)
                    valid = (src >= 0) & (src < n)
                    g[lo2:hi2][valid] += a[src[valid]]

                # -a[k-i] for k in [i, i+n-1]
                lo3 = i
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[:hi3 - lo3]

                # -a[i-k] for k in [max(0, i-n+1), i]
                lo4 = max(0, i - n + 1)
                hi4 = i + 1
                if hi4 > lo4:
                    src4 = i - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    g[lo4:hi4][valid4] -= a[src4[valid4]]

                # Find best delta in [0, ai] using grid
                # conv_new[k] ≈ conv[k] + delta * g[k] + delta^2 * h[k]
                # h has nonzero entries only at k=2i, k=2j, k=i+j
                # For simplicity, use first-order + check at a few delta values

                cur_max = conv.max()
                best_d = 0.0
                best_val = cur_max

                # Try geometric spacing of deltas
                deltas = ai * np.array([0.01, 0.03, 0.07, 0.15, 0.3, 0.5, 0.7, 1.0])
                deltas = deltas[deltas < ai + 1e-15]

                for d in deltas:
                    if d <= 0:
                        continue
                    new_conv_max = (conv + d * g).max()
                    # Second-order correction
                    if 2 * i < size:
                        val_2i = conv[2*i] - 2*ai*d + d*d
                        new_conv_max = max(new_conv_max, val_2i)
                    if 2 * j < size:
                        val_2j = conv[2*j] + 2*a[j]*d + d*d
                        new_conv_max = max(new_conv_max, val_2j)
                    if i + j < size:
                        val_ij = conv[i+j] + d*g[i+j] - 2*d*d
                        new_conv_max = max(new_conv_max, val_ij)

                    if new_conv_max < best_val - 1e-12:
                        best_val = new_conv_max
                        best_d = d

                if best_d > 0:
                    # Apply
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
        else:
            no_improve_count = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_best_response(a, M, solver_name):
    """Solve LP: max sum(b) s.t. T(b) <= M, b >= 0. T is the convolution matrix."""
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1

    # Build constraint matrix T where T[k, i] = a[k-i] + a[i-k]
    # (the Jacobian of the k-th autocorrelation wrt a_i)
    T = np.zeros((size, n))
    for i in range(n):
        # a[k-i] nonzero for k in [i, i+n-1]
        lo = i
        hi = min(size, i + n)
        if hi > lo:
            T[lo:hi, i] += a[:hi - lo]
        # a[i-k] nonzero for k in [max(0,i-n+1), i]
        lo2 = max(0, i - n + 1)
        hi2 = i + 1
        if hi2 > lo2:
            src = i - np.arange(lo2, hi2)
            valid = (src >= 0) & (src < n)
            T[lo2:hi2, i][valid] += a[src[valid]]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_nesterov_few(a0, time_limit, solver_name, max_iters=50, verbose=True):

    t0 = time.monotonic()

    avg_a = np.zeros(n)

            break

        b = lp_best_response(y, M_curr, solver_name)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:

        if t == 1:

        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

def get_solver():

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            if time.monotonic() - t_global > time_budget - 30:

            # Perturb: add small noise and re-project
            noise_scale = 0.05
            perturbed = best_seq + noise_scale * rng.standard_normal(n_best)
            perturbed = np.maximum(perturbed, 0.0)
            sp = perturbed.sum()
            if sp < 1e-10:
                continue
            perturbed = perturbed / sp

            # Quick LSE optimization on perturbed
            t_sub = min((time_budget - (time.monotonic() - t_global) - 30) * 0.5, 80.0)
            if t_sub < 10:
                break
            a_norm = perturbed * np.sqrt(2.0 * n_best) / perturbed.sum()
            a_out, _ = lse_nesterov_optimize(a_norm, t_sub, verbose=False)
            improved = do_try_update(a_out, f"perturb_{attempt}")

            # Also try CD on perturbed
            remaining_sub = time_budget - (time.monotonic() - t_global)
            t_cd = min(remaining_sub * 0.3, 50.0)
            if t_cd > 10:
                a_norm2 = (a_out if a_out is not None else perturbed)
                a_norm2 = np.maximum(a_norm2, 0.0)
                s2 = a_norm2.sum()
                if s2 > 1e-10:
                    a_norm2 = a_norm2 * np.sqrt(2.0 * n_best) / s2
                    a_out2, _ = systematic_pair_cd(a_norm2, t_cd, rng, verbose=False)
                    do_try_update(a_out2, f"perturb_cd_{attempt}")

    # === Phase 7: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n_base + 200
        print(f"\n=== Phase 7: Alternate length n={n_alt} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out, _ = lse_nesterov_optimize(a_norm, t7, verbose=False)
            do_try_update(a_out, f"lse_n{n_alt}")

            remaining_sub = time_budget - (time.monotonic() - t_global)
            t_cd2 = min(remaining_sub * 0.4, 60.0)
            if t_cd2 > 10:
                a_norm2 = a_out * np.sqrt(2.0 * n_alt) / a_out.sum()
                a_out2, _ = systematic_pair_cd(a_norm2, t_cd2, rng, verbose=False)
                do_try_update(a_out2, f"cd_n{n_alt}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
