def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project vector v onto the probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new
            a = a_new
            y = y_new
            t_fista += 1

            if it % 1000 == 0:
                sc = _autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _gradient_peak_shave(a, time_limit, seed=42):
    """Gradient-informed peak shaving."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _autocorr(a)
        top_k_idx = np.argsort(b)[::-1][:12]
        improved = False

        for k in top_k_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            valid = (idx >= 0) & (k - idx >= 0) & (k - idx < n)
            idx = idx[valid]
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Build peak mask: positions that contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k_idx:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Prefer low-value positions
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_cand = max(3, len(candidates_sorted) // 8)
            candidate_pool = candidates_sorted[:n_cand]

            dst = candidate_pool[rng.integers(len(candidate_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < norm * 1e-8:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = _autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def _spectral_shave(a, time_limit, seed=0):
    """Spectral domain peak shaving: adjust power spectrum to flatten autocorrelation."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 200:
        b = _autocorr(a)
        k_peak = int(np.argmax(b))

        # Compute spectrum
        A = rfft(a, fft_size)
        phase = np.angle(A)
        magnitude = np.abs(A)

        # In frequency domain, reducing peak at lag k_peak corresponds to
        # reducing the correlation at
        # Try small adjustments to the spectrum
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03]:
            # The peak b[k_peak] = sum_i a[i]*a[k_peak-i]
            # In frequency: b[k] = IFFT(|A|^2)[k]
            # To reduce b[k_peak], we can reduce |A|^2 in a targeted way
            # Simple approach: reduce magnitude at frequencies that contribute most to the peak

            A_mod = A.copy()
            # Reduce overall magnitude slightly and re-add to maintain sum
            A_mod *= (1.0 - lr * 0.1)
            a_new = irfft(A_mod, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
                score = _autocorr(a_new).max()
                if score < best_score - 1e-14:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Try random spectral perturbations
            for _ in range(10):
                A_mod = A.copy().astype(complex)
                # Small random phase perturbation
                dphase = rng.normal(0, 0.01, size=A_mod.shape)
                A_mod *= np.exp(1j * dphase)
                a_new = irfft(A_mod, fft_size)[:n]
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    score = _autocorr(a_new).max()
                    if score < best_score - 1e-14:
                        a = a_new
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            if not improved:
                stall += 1

    return best_a, best_score


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
