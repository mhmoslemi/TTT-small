_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the matrix T such that (T @ b)_k = (a * b)_k for full convolution."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, warm_start=None):
    """Solve the LP: max sum(b) s.t. a*b <= M, b >= 0. With optional warm start."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    last_print = 0
    
    # Use a slightly more aggressive momentum schedule
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
            
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check multiple candidates
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters - last_print >= 100:
            last_print = n_iters
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best_max={best_score:.8f}, M_y={M_curr:.8f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Greedy coordinate descent: for each i, try transferring to all j's."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue
            
            # Find the gain vector for transferring from i
            # g_k = a_{k-i} + a_{i-k} (terms to remove)
            # h_k = a_{k-j} + a_{j-k} (terms to add for each j)
            # Net change at k: -delta * (a_{k-i} + a_{i-k}) + delta * (a_{k-j} + a_{j-k})
            
            # Precompute the "remove" vector
            remove_vec = np.zeros(size)
            # a_{k-i} for valid k: k in [i, i+n-1], value = a[k-i]
            k_lo = i
            k_hi = min(size, i + n)
            if k_hi > k_lo:
                remove_vec[k_lo:k_hi] = a[:k_hi - k_lo]
            # a_{i-k} for valid k: k in [max(0, i-n+1), min(n-1, i)], value = a[i-k]
            k_lo2 = max(0, i - n + 1)
            k_hi2 = min(n, i + 1)
            if k_hi2 > k_lo2:
                remove_vec[k_lo2:k_hi2] += a[i - np.arange(k_lo2, k_hi2)]
            
            # For each j, the "add" vector is similar
            # Try a subset of j's to save time
            # Use a smarter selection: try j's near i and j's far from i
            n_tries = min(50, n)
            j_candidates = np.concatenate([
                rng.integers(0, n, min(n_tries // 2, n)),
                # Also try positions near the peak
            ])
            
            best_delta = 0.0
            best_j = -1
            best_new_max = best_score
            
            for j in j_candidates:
                if j == i or a[j] >= norm:  # can't add too much
                    continue
                
                # Compute add vector for this j
                add_vec = np.zeros(size)
                k_lo = j
                k_hi = min(size, j + n)
                if k_hi > k_lo:
                    add_vec[k_lo:k_hi] = a[:k_hi - k_lo]
                k_lo2 = max(0, j - n + 1)
                k_hi2 = min(n, j + 1)
                if k_hi2 > k_lo2:
                    add_vec[k_lo2:k_hi2] += a[j - np.arange(k_lo2, k_hi2)]
                
                gain = add_vec - remove_vec  # change per unit delta
                
                # Find best delta in (0, ai)
                # We want to minimize max_k(conv_k + delta * gain_k)
                # Also second order at k=2i: (ai-d)^2 vs ai^2, change = -2*ai*d + d^2
                # And at k=2j: (aj+d)^2 vs aj^2, change = 2*aj*d + d^2
                
                # Binary search or sampling
                lo, hi = 0.0, ai
                # Try a few points
                for d in [ai * f for f in [0.01, 0.05, 0.1, 0.2, 0.5, 0.8]]:
                    if d <= 0 or d >= ai:
                        continue
                    new_conv_max = (conv + d * gain).max()
                    # Second order corrections
                    if 2 * i < size:
                        val_2i = conv[2*i] - 2*ai*d + d*d
                        new_conv_max = max(new_conv_max, val_2i)
                    if 2 * j < size:
                        val_2j = conv[2*j] + 2*a[j]*d + d*d
                        new_conv_max = max(new_conv_max, val_2j)
                    if new_conv_max < best_new_max:
                        best_new_max = new_conv_max
                        best_delta = d
                        best_j = j
            
            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best_max={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def build_conv_matrix(n):
    """Build the structure matrix for convolution. T[k,j] gives the coefficient
    of a[j] in (a*b)[k] when b is the variable."""
    size = 2 * n - 1
    # T is size x n, T[k,j] = 1 if a_{k-j} is a valid index, else 0
    # Actually we need T such that (T @ b)_k = sum_j a_{k-j} * b_j
    # where a_{k-j} means a[k-j] if 0 <= k-j < n else 0
    # But we also need a_{j-k} * b_j for the symmetric part
    # conv(a,b)[k] = sum_j a_{k-j} * b_j (with a indexed 0..n-1)
    # = sum_{j: 0<=k-j<n} a[k-j] * b_j
    T = np.zeros((size, n))
    for j in range(n):
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        lo = j
        hi = min(size, j + n)
        T[lo:hi, j] = a[:(hi - lo)] if False else T[lo:hi, j]
    return T


def build_conv_matrix_fast(a):
    """Build T such that (T @ b)[k] = (a * b)[k] for full convolution."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = min(size, j + n)
        T[lo:hi, j] = a[:hi - lo]
    return T


def lp_best_response(a, M, solver):
    """Solve LP: max 1^T b s.t. T(a) @ b <= M*1, b >= 0."""
    n = len(a)
    T = build_conv_matrix_fast(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        a = a_new

            stale_count = 0

def compute_remove_vec(i, a, size):
    """Compute the vector r[k] = a_{k-i} + a_{i-k} (indices valid only)."""
    n = len(a)
    r = np.zeros(size)
    # a[k-i]: valid when 0 <= k-i < n => i <= k <= i+n-1
    lo = i
    hi = min(size, i + n)
    if hi > lo:
        r[lo:hi] += a[:hi - lo]
    # a[i-k]: valid when 0 <= i-k < n => i-n+1 <= k <= i
    lo2 = max(0, i - n + 1)
    hi2 = min(size, i + 1)
    if hi2 > lo2:
        r[lo2:hi2] += a[i - np.arange(lo2, hi2)]
    return r


def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Systematic greedy coordinate descent with structured j-selection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    # Precompute all remove vectors for speed
    remove_vecs = np.zeros((n, size))
    for i in range(n):
        remove_vecs[i] = compute_remove_vec(i, a, size)

    delta_fractions = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            r = remove_vecs[i]

            # Try a mix of random and structured j candidates
            n_random = 20
            j_candidates = rng.integers(0, n, n_random)
            # Also add positions near the conv peaks
            peak_positions = np.argsort(conv)[-10:]
            for pp in peak_positions:
                for offset in range(-5, 6):
                    j2 = int(pp - offset)
                    if 0 <= j2 < n:
                        j_candidates = np.append(j_candidates, j2)
            j_candidates = np.unique(j_candidates)

            best_delta = 0.0
            best_j = -1
            best_new_max = best_score

            for j in j_candidates:
                if j == i:
                    continue
                aj = a[j]

                # Compute add vector for j
                add_vec = compute_remove_vec(j, a, size)
                gain = add_vec - r

                # Find best delta
                for frac in delta_fractions:
                    d = ai * frac
                    if d <= 0:
                        continue
                    new_conv_max = (conv + d * gain).max()
                    # Second order corrections
                    if 2 * i < size:
                        val_2i = conv[2 * i] - 2 * ai * d + d * d
                        if val_2i > new_conv_max:
                            new_conv_max = val_2i
                    if 2 * j < size:
                        val_2j = conv[2 * j] + 2 * aj * d + d * d
                        if val_2j > new_conv_max:
                            new_conv_max = val_2j
                    if new_conv_max < best_new_max - 1e-14:
                        best_new_max = new_conv_max
                        best_delta = d
                        best_j = int(j)

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                # Update remove vectors (change at i and best_j)
                remove_vecs[i] = compute_remove_vec(i, a, size)
                remove_vecs[best_j] = compute_remove_vec(best_j, a, size)
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[best_j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    remove_vecs[i] = compute_remove_vec(i, a, size)
                    remove_vecs[best_j] = compute_remove_vec(best_j, a, size)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def lse_smooth_minimize(a0, time_limit, verbose=True):
    """Log-sum-exp smooth minimization with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Anneal from high to low temperature
    temps = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2]
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(5, int(remaining * 2))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_c = conv.max()
            exp_c = np.exp((conv - max_c) / T)
            w = exp_c / exp_c.sum()

            # Compute gradient via FFT cross-correlation
            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)
            # grad[i] = 2 * (sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k])
            # sum_k w[k]*a[k-i] for all i = circular cross-correlation
            cc1 = irfft(W_fft * np.conj(A_fft), fft_size)
            g1 = cc1[:n]
            cc2 = irfft(A_fft * np.conj(W_fft), fft_size)
            g2 = cc2[fft_size - n + 1:fft_size + 1]
            grad = 2.0 * (g1 + g2)

            # Projected gradient step
            lr = norm * 0.1 / (1.0 + 0.1 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a_arr, label):
        nonlocal best_seq, best_score
        if a_arr is None:
            return
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_norm = a_arr / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_arr)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # === Phase 3: Try longer length n=2000 ===
    n2 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 4: CD on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"cd2_n{n_best}")

    # === Phase 5: LSE smooth refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_smooth_minimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"cd_final_n{n_best}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
