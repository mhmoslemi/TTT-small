def choose_active_lags(C, max_K=10, tol_frac=1e-3):
    L=len(C); M=C.max()
    if M <= 0: return np.array([L//2], dtype=int)
    tol = max(1e-18, tol_frac*M)
    mask = C >= M - tol
    idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(C))])
    if len(idx) > max_K:
        order = np.argsort(C[idx])[::-1][:max_K]
        idx = idx[order]
    return idx

"""
Search for low-peak self-convolution sequences.

Main ideas implemented here:
1. Corrected exact subgradients for max(self-convolution).
   For C = conv(a,a), dC_k/da_m = 2 a_{k-m} when valid.
2. Gradient/mirror descent with multiple active-peak weightings.
3. LP best-response on the active peaks:
   minimize max_k (a*b)_k over b >= 0, sum(b)=1.
4. Exact pairwise local search:
   moving mass from i to j changes C by a sparse quadratic update.
   We target both gradient-driven pairs and pairs that directly reduce
   the largest contributing terms of the active peaks.
5. Multi-length interpolation followed by short optimization.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft, next_fast_len
except Exception:
    from numpy.fft import rfft, irfft

    def next_fast_len(n):
        n = max(1, int(n))
        return 1 << (n - 1).bit_length()

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def normalize_simplex(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 1e-15:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def full_conv(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    A = rfft(a, size)
    c = irfft(A * A, size)[:L]
    c = np.asarray(np.real(c), dtype=np.float64)
    maxabs = float(np.max(np.abs(c))) if c.size else 0.0
    if maxabs > 0.0:
        c[np.abs(c) < 1e-15 * maxabs] = 0.0
    c = np.maximum(c, 0.0)
    return c


def ratio_from_conv(a, c):
    n = a.size
    s = a.sum()
    if n == 0 or s <= 0.0:
        return float("inf")
    return float(2.0 * n * c.max() / (s * s))


def internal_ratio(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if a.size == 0 or s <= 1e-15:
        return float("inf")
    c = full_conv(a / s)
    return float(2.0 * a.size * c.max())


def evaluate_safe(a):
    try:
        out = evaluate_sequence([float(x) for x in a])
        return float(out)
    except Exception:
        return float("inf")


def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    best = None
    best_m = float("inf")
    inits = []

    inits.append(np.ones(n, dtype=np.float64))

    x = np.linspace(0.0, np.pi, n)
    inits.append(np.sin(x) ** 2)

    x = np.linspace(0.0, 1.0, n)
    inits.append(np.exp(-6.0 * (x - 0.25) ** 2) + 0.9 * np.exp(-6.0 * (x - 0.75) ** 2))

    x = np.linspace(0.0, 2.0 * np.pi, n)
    inits.append(0.5 + 0.3 * np.cos(2.0 * x) + 0.2 * np.cos(5.0 * x) + 0.1 * np.cos(11.0 * x))

    inits.append(np.exp(-rng.exponential(0.8, size=n)))
    inits.append(np.maximum(rng.normal(size=n), 0.0) + np.ones(n) * 0.05)

    for v in inits:
        v = np.asarray(v, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            continue
        v = v / s
        m = full_conv(v).max()
        if m < best_m:
            best_m = m
            best = v

    if best is None:
        best = np.ones(n, dtype=np.float64) / n
    return best


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if old_n == 0 or new_n <= 0:
        return normalize_simplex(np.ones(max(1, int(new_n)), dtype=np.float64))
    if old_n == int(new_n):
        return normalize_simplex(f)

    s = f.sum()
    if s <= 1e-15:
        return np.ones(int(new_n), dtype=np.float64)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    y = np.interp(new_x, old_x, fn)
    y = np.maximum(y, 0.0)
    ss = y.sum()
    if ss <= 1e-15:
        return np.ones(int(new_n), dtype=np.float64)
    return y / ss


def make_weights(c, mode, k=20, t=None):
    L = c.size
    m = c.max()
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w
    if m <= 0.0:
        w[L // 2] = 1.0
        return w

    if mode == "exp":
        if t is None or t <= 0.0:
            t = m * 1e-2
        z = (c - m) / max(t, 1e-300)
        z = np.clip(z, -60.0, 0.0)
        w = np.exp(z)
        s = w.sum()
        if s <= 1e-300:
            w = np.zeros(L, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= s
    else:
        k = int(min(max(1, k), L))
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(L, dtype=np.float64)
        if mode == "topw":
            vals = c[idx]
            s = vals.sum()
            if s <= 0.0:
                w[idx] = 1.0 / idx.size
            else:
                w[idx] = vals / s
        else:
            w[idx] = 1.0 / idx.size
    return w


def weighted_gradient(x, w):
    """
    Gradient of sum_k w_k conv(x,x)_k.

    For C_k = sum_i x_i x_{k-i},
    dC_k/dx_m = 2 x_{k-m} if 0 <= k-m < n, else 0.
    Thus g_m = 2 sum_i x_i w_{m+i}.
    """
    n = x.size
    L = 2 * n - 1
    size = next_fast_len(n + L - 1)
    X = rfft(x, size)
    Wr = rfft(w[::-1].copy(), size)
    c = irfft(X * Wr, size)
    idx = L - 1 - np.arange(n)
    g = 2.0 * c[idx]
    g = np.asarray(np.real(g), dtype=np.float64)
    g[np.abs(g) < 1e-16] = 0.0
    return np.maximum(g, 0.0)


def mirror_step(x, g, eta):
    logx = np.log(np.maximum(x, 1e-300)) - eta * np.asarray(g, dtype=np.float64)
    mx = logx.max()
    if not np.isfinite(mx):
        return x.copy()
    y = np.exp(logx - mx)
    return normalize_simplex(y)


def line_search_blend(x, y, current_m, n, alphas):
    best_x = x
    best_m = current_m
    best_c = None
    for alpha in alphas:
        xp = (1.0 - alpha) * x + alpha * y
        xp = normalize_simplex(xp)
        cx = full_conv(xp)
        mx = cx.max()
        if mx < best_m:
            best_m = mx
            best_x = xp
            best_c = cx
    return best_x, best_m, best_c


def perturb_sequence(x, rng, alpha):
    n = x.size
    noise = rng.normal(size=n)
    noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    noise = np.maximum(noise, 0.0)
    s = noise.sum()
    if s <= 1e-15:
        return x.copy()
    noise /= s
    y = (1.0 - alpha) * x + alpha * noise
    return normalize_simplex(y)


def gradient_mirror_search(x, time_limit, seed=42):
    x = normalize_simplex(np.asarray(x, dtype=np.float64).copy())
    n = x.size
    c = full_conv(x)
    m = c.max()
    best_m = m
    best_x = x.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    configs = []
    for frac in (0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004):
        configs.append(("exp", frac, 0))
    for k in (4, 8, 16, 32, 64, 128):
        configs.append(("top", None, k))
    for k in (8, 16, 32, 64):
        configs.append(("topw", None, k))

    alphas = (1.0, 0.6, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01)

    for ci, (mode, frac, k) in enumerate(configs):
        if time.monotonic() - t0 > time_limit:
            break

        step = 0.35
        no_imp = 0

        for it in range(220):
            if time.monotonic() - t0 > time_limit:
                break

            if m < best_m - 1e-15:
                best_m = m
                best_x = x.copy()

            if m <= 1e-300:
                break

            if mode == "exp":
                t = max(1e-12, frac * m)
                w = make_weights(c, "exp", 0, t)
            elif mode == "topw":
                w = make_weights(c, "topw", k)
            else:
                w = make_weights(c, "top", k)

            g = weighted_gradient(x, w)
            grange = float(g.max() - g.min())
            if grange < 1e-17:
                break

            eta = step / grange
            eta = min(1e5, max(1e-8, eta))

            if it % 2 == 0:
                y = mirror_step(x, g, eta)
            else:
                gc = g - g.mean()
                scale = float(gc.std()) + 1e-18
                y = normalize_simplex(x - step * gc / scale)

            xp, mp, cp = line_search_blend(x, y, m, n, alphas)
            if mp < m - 1e-14:
                x = xp
                m = mp
                c = cp if cp is not None else full_conv(x)
                step = min(1.6, step * 1.18)
                no_imp = 0
            else:
                step *= 0.5
                no_imp += 1
                if step < 1e-5:
                    break
                if no_imp > 35:
                    x = perturb_sequence(best_x, rng, 0.08)
                    c = full_conv(x)
                    m = c.max()
                    step = 0.35
                    no_imp = 0

    return best_x, float(2.0 * n * best_m)


def choose_active_lags(c, max_k=10, tol_frac=1e-3):
    L = c.size
    m = c.max()
    if L == 0:
        return np.zeros(0, dtype=int)
    if m <= 0.0:
        return np.array([L // 2], dtype=int)
    tol = max(1e-18, tol_frac * m)
    mask = c >= m - tol
    idx = np.where(mask)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    if idx.size > max_k:
        order = np.argsort(c[idx])[::-1][:max_k]
        idx = idx[order]
    return idx


def deriv_profile_conv(x, m):
    n = x.size
    L = 2 * n - 1
    p = np.zeros(L, dtype=np.float64)
    p[m:m + n] = 2.0 * x
    return p


def max_after_move(c, d, eps, i, j, n):
    val = c + eps * d
    e2 = eps * eps
    val[2 * i] += e2
    val[2 * j] += e2
    val[i + j] -= 2.0 * e2
    return float(val.max())


def apply_move_with_d(x, c, d, i, j, eps, n):
    x[i] -= eps
    if x[i] < 0.0:
        x[i] = 0.0
    x[j] += eps
    c = c + eps * d
    e2 = eps * eps
    c[2 * i] += e2
    c[2 * j] += e2
    c[i + j] -= 2.0 * e2
    c = np.maximum(c, 0.0)
    return x, c


def pair_local_search(x, time_limit, seed=42):
    x = normalize_simplex(np.asarray(x, dtype=np.float64).copy())
    n = x.size
    c = full_conv(x)
    m = float(c.max())
    best_m = m
    best_x = x.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stall = 0
    accepts = 0

    eps_fracs = (0.002, 0.008, 0.03, 0.10, 0.30, 0.60, 0.90)
    grad_sources = 22
    grad_dests = 22

    while time.monotonic() - t0 < time_limit:
        if accepts > 0 and accepts % 25 == 0:
            x = normalize_simplex(x)
            c = full_conv(x)
            m = float(c.max())
            if m < best_m:
                best_m = m
                best_x = x.copy()

        if m <= 1e-300:
            break

        active = choose_active_lags(c, 10, 1e-3)
        w = np.zeros(c.size, dtype=np.float64)
        vals = c[active]
        s = vals.sum()
        if s > 0.0:
            w[active] = vals / s
        else:
            w[active] = 1.0 / max(1, active.size)

        g = weighted_gradient(x, w)
        pairs = []

        pos = np.where(x > 1e-14)[0]
        if pos.size > 0:
            src = pos[np.argsort(g[pos])[::-1]][:grad_sources]
            dst = np.argsort(g)[:grad_dests]
            for i in src:
                i = int(i)
                if x[i] <= 1e-14:
                    continue
                for j in dst:
                    j = int(j)
                    if j != i:
                        pairs.append((i, j))

        # Directly attack largest contributing products of active peaks.
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            jarr = k - idx
            mask = idx < jarr
            idxu = idx[mask]
            if idxu.size == 0:
                continue
            jarru = jarr[mask]
            contrib = x[idxu] * x[jarru]
            n_top = min(4, idxu.size)
            top = np.argsort(contrib)[::-1][:n_top]
            for a in idxu[top]:
                a = int(a)
                b = int(k - a)
                if b < 0 or b >= n or a == b:
                    continue
                if x[a] < x[b]:
                    i, j = a, b
                elif x[b] < x[a]:
                    i, j = b, a
                else:
                    i, j = a, b
                if x[i] > 1e-14 and x[j] > 0.0:
                    pairs.append((i, j))

        if not pairs:
            stall += 1
        else:
            seen = set()
            unique_pairs = []
            for i, j in pairs:
                if i == j:
                    continue
                code = int(i) * n + int(j)
                if code in seen:
                    continue
                seen.add(code)
                unique_pairs.append((int(i), int(j)))
            if len(unique_pairs) > 1800:
                unique_pairs = unique_pairs[:1800]

            all_i = []
            all_j = []
            for i, j in unique_pairs:
                all_i.append(i)
                all_j.append(j)
            all_idx = np.unique(np.concatenate((np.array(all_i, dtype=int), np.array(all_j, dtype=int))))
            profs = {int(i): deriv_profile_conv(x, int(i)) for i in all_idx}

            best_move_m = m
            best_tuple = None
            for i, j in unique_pairs:
                xi = x[i]
                if xi <= 1e-14:
                    continue
                d = profs[j] - profs[i]
                for frac in eps_fracs:
                    eps = frac * xi
                    if eps <= 1e-18:
                        continue
                    mn = max_after_move(c, d, eps, i, j, n)
                    if mn < best_move_m - 1e-14:
                        best_move_m = mn
                        best_tuple = (i, j, eps)

            if best_tuple is not None:
                i, j, eps = best_tuple
                d = profs[j] - profs[i]
                x, c = apply_move_with_d(x, c, d, i, j, eps, n)
                m = float(c.max())
                if m < best_m:
                    best_m = m
                    best_x = x.copy()
                accepts += 1
                stall = 0
            else:
                stall += 1

        if stall > 0:
            found = False
            for _ in range(35):
                i = int(rng.integers(n))
                if x[i] <= 1e-14:
                    continue
                j = int(rng.integers(n))
                if i == j:
                    continue
                d = deriv_profile_conv(x, j) - deriv_profile_conv(x, i)
                for frac in (0.005, 0.02, 0.08, 0.25):
                    eps = frac * x[i]
                    mn = max_after_move(c, d, eps, i, j, n)
                    if mn < m - 1e-14:
                        x, c = apply_move_with_d(x, c, d, i, j, eps, n)
                        m = float(c.max())
                        if m < best_m:
                            best_m = m
                            best_x = x.copy()
                        found = True
                        break
                if found:
                    break

            if found:
                stall = 0
            else:
                if stall > 60:
                    x = perturb_sequence(best_x, rng, 0.07)
                    c = full_conv(x)
                    m = float(c.max())
                    stall = 20

    return best_x, float(2.0 * n * best_m)


def make_lp_static(n):
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    eq = np.zeros((1, n + 1), dtype=np.float64)
    eq[0, :n] = 1.0
    upper = 1.0 + 1e-9
    bounds = [(0.0, upper)] * (n + 1)
    return c, eq, bounds


def lp_best_response(x, k_lags, c_obj, eq, bounds, b_eq):
    if linprog is None:
        return None, None

    n = x.size
    k_lags = np.unique(np.asarray(k_lags, dtype=int))
    if k_lags.size == 0:
        return None, None

    m = k_lags.size
    A = np.zeros((m, n + 1), dtype=np.float64)
    for r, k in enumerate(k_lags):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A[r, js:je + 1] = x[k - js:k - je - 1:-1]
        A[r, n] = -1.0

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=np.zeros(m, dtype=np.float64),
            A_eq=eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"disp": False, "presolve": True}
        )
    except Exception:
        return None, None

    if res.status != 0:
        return None, None
    xv = getattr(res, "x", None)
    if xv is None:
        return None, None

    b = np.asarray(xv[:n], dtype=np.float64)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 1e-15:
        return None, None
    b /= s
    t_opt = float(xv[n])
    if not np.isfinite(t_opt):
        return None, None
    return b, t_opt


def lp_descent(x, time_limit):
    n = len(np.asarray(x))
    if n <= 0:
        return normalize_simplex(np.array([1.0])), float("inf")

    base = normalize_simplex(np.asarray(x, dtype=np.float64).copy())
    if linprog is None:
        c = full_conv(base)
        return base, float(2.0 * n * c.max())

    c_obj, eq, bounds = make_lp_static(n)
    b_eq = np.array([1.0], dtype=np.float64)

    x = base.copy()
    c = full_conv(x)
    m = float(c.max())
    best_m = m
    best_x = x.copy()

    t0 = time.monotonic()
    alphas = (1.0, 0.6, 0.3, 0.15, 0.08, 0.03, 0.012)

    for max_k in (5, 10, 16):
        stall = 0
        while time.monotonic() - t0 < time_limit:
            if stall > 25:
                break
            if m < best_m - 1e-15:
                best_m = m
                best_x = x.copy()

            active = choose_active_lags(c, max_k, 0.01)
            b, t_opt = lp_best_response(x, active, c_obj, eq, bounds, b_eq)
            if b is None:
                stall += 1
                continue

            if t_opt >= m - 1e-12:
                stall += 1
                continue

            xp, mp, cp = line_search_blend(x, b, m, n, alphas)
            if mp < m - 1e-14:
                x = xp
                m = mp
                c = cp if cp is not None else full_conv(x)
                stall = 0
                if m < best_m:
                    best_m = m
                    best_x = x.copy()
            else:
                stall += 1

    return best_x, float(2.0 * n * best_m)


def optimize_length(parent, new_n, grad_time, pair_time, seed=0):
    x = interpolate_sequence(parent, int(new_n))
    if grad_time > 5.0:
        x, f = gradient_mirror_search(x, grad_time, seed=seed)
    else:
        c = full_conv(x)
        f = float(2.0 * x.size * c.max())
    if pair_time > 5.0:
        x2, f2 = pair_local_search(x, pair_time, seed=seed + 1000)
        if f2 < f:
            x = x2
            f = f2
    return x, f


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    budget = min(float(budget_s) - 90.0, 1700.0)
    if budget < 30.0:
        budget = max(1.0, float(budget_s) - 5.0)

    best_x = None
    best_f = float("inf")

    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.all(np.isfinite(arr)):
            arr = np.clip(arr, 0.0, None)
            if arr.sum() > 1e-10:
                if arr.size > 4000:
                    arr = interpolate_sequence(arr, 2400)
                arr = normalize_simplex(arr)
                cc = full_conv(arr)
                best_x = arr
                best_f = float(2.0 * arr.size * cc.max())
                print(f"Loaded parent: n={arr.size}, score={best_f:.9f}")
    except Exception:
        best_x = None

    if best_x is None:
        n0 = 1600
        best_x = structured_init(n0, seed=seed)
        cc = full_conv(best_x)
        best_f = float(2.0 * n0 * cc.max())
        print(f"Default start: n={n0}, score={best_f:.9f}")

    best_x = normalize_simplex(best_x)

    def remaining_time():
        return max(0.0, budget - (time.monotonic() - t_global))

    rem = remaining_time()
    t1 = min(620.0, rem * 0.32)
    if t1 > 10.0:
        print(f"Phase 1: gradient/mirror, budget={t1:.0f}s")
        x1, f1 = gradient_mirror_search(best_x, t1, seed=seed)
        if f1 < best_f:
            best_x = x1
            best_f = f1
            print(f"  phase1 best={best_f:.9f}")
    best_x = normalize_simplex(best_x)

    rem = remaining_time()
    t2 = min(520.0, rem * 0.30)
    if t2 > 10.0:
        print(f"Phase 2: exact pair/local, budget={t2:.0f}s")
        x2, f2 = pair_local_search(best_x, t2, seed=seed + 1)
        if f2 < best_f:
            best_x = x2
            best_f = f2
            print(f"  phase2 best={best_f:.9f}")
    best_x = normalize_simplex(best_x)

    rem = remaining_time()
    t3 = min(320.0, rem * 0.22)
    if t3 > 20.0:
        print(f"Phase 3: LP best-response, budget={t3:.0f}s")
        x3, f3 = lp_descent(best_x, t3)
        if f3 < best_f:
            best_x = x3
            best_f = f3
            print(f"  phase3 best={best_f:.9f}")
    best_x = normalize_simplex(best_x)

    rem = remaining_time()
    t4 = min(340.0, rem * 0.25)
    if t4 > 10.0:
        print(f"Phase 4: pair/local polish, budget={t4:.0f}s")
        x4, f4 = pair_local_search(best_x, t4, seed=seed + 2)
        if f4 < best_f:
            best_x = x4
            best_f = f4
            print(f"  phase4 best={best_f:.9f}")
    best_x = normalize_simplex(best_x)

    rem = remaining_time()
    if rem > 160.0:
        base_n = int(best_x.size)
        alt_budget = min(220.0, rem * 0.35)
        if alt_budget > 70.0:
            n_alt = max(500, base_n + 350)
            print(f"Phase 5: alternate length n={n_alt}, budget={alt_budget:.0f}s")
            x5, f5 = optimize_length(best_x, n_alt, alt_budget * 0.58, alt_budget * 0.42, seed=seed + 30)
            if f5 < best_f:
                best_x = x5
                best_f = f5
                print(f"  phase5 best={best_f:.9f} n={best_x.size}")
            best_x = normalize_simplex(best_x)

    rem = remaining_time()
    if rem > 30.0:
        t6 = rem * 0.78
        print(f"Phase 6: final pair polish, budget={t6:.0f}s")
        x6, f6 = pair_local_search(best_x, t6, seed=seed + 77)
        if f6 < best_f:
            best_x = x6
            best_f = f6
            print(f"  phase6 best={best_f:.9f}")

    best_x = normalize_simplex(np.asarray(best_x, dtype=np.float64))
    final_internal = internal_ratio(best_x)
    final_eval = evaluate_safe(best_x)
    print(f"\nFinal internal score={final_internal:.9f}")
    if np.isfinite(final_eval):
        print(f"Final evaluate_sequence score={final_eval:.9f}")
    print(f"Final length={best_x.size}, total time={time.monotonic() - t_global:.1f}s")

    return [float(v) for v in best_x]
