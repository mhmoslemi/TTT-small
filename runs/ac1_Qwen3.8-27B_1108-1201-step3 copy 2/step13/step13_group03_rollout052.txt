"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-FISTA with adaptive temperature (fast convergence)
2. LP best-response with line search (precise optimization)
3. Stochastic perturbation + restart (escape local minima)
4. Coordinate descent polish (fine-tuning)
Multi-length strategy to explore different n values.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _lp_line_search(a0, solver, time_limit, max_iters=5000, verbose=True):
    """LP best-response with line search for step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    prev_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = _fast_autocorr(a).max()
        if M_curr < 1e-12:
            break
        
        b = _lp_best_response(a, M_curr, solver)
        if b is None:
            continue
        b = b / b.sum()
        
        # Line search: try several step sizes
        best_t = 0.01
        best_trial_score = best_score
        for t in [0.01, 0.03, 0.07, 0.15, 0.3, 0.5, 0.8, 1.0]:
            trial = (1 - t) * a + t * b
            trial = np.maximum(trial, 0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            sc = _fast_autocorr(trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        prev_a = a
        a = a_new
        
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-linsearch iter={it} t={best_t:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _fast_autocorr(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-18:
        out = out / s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_solve(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

        b = _lp_solve(a, M_curr, solver)
        if b is None:
            stall += 1
            if stall > 100:
                break
            continue

        best_t = 0.05
        best_trial = best_score
        t_candidates = [0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]
        for t in t_candidates:
            trial = (1.0 - t) * a + t * b
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            sc = _fast_autocorr(trial).max()
            if sc < best_trial:
                best_trial = sc
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        it += 1

        avg_buf[avg_idx % 500] = a
        avg_idx += 1
        avg_count = min(avg_count + 1, 500)

        sc = _fast_autocorr(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            stall = 0
            last_improve_score = best_score
        else:
            stall += 1

        if avg_count >= 100 and it % 50 == 0:
            avg_vec = avg_buf[:avg_count].mean(axis=0)
            avg_vec = np.maximum(avg_vec, 0.0)
            s_avg = avg_vec.sum()
            if s_avg > 1e-15:
                avg_vec = avg_vec / s_avg
                sc_avg = _fast_autocorr(avg_vec).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_vec.copy()
                    a = avg_vec.copy()
                    stall = 0
                    last_improve_score = best_score

        if stall > 200:
            rng_local = np.random.default_rng(it)
            pert = rng_local.normal(0, 0.01, a.shape)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-15:
                a_pert = a_pert / s_pert
                sc_pert = _fast_autocorr(a_pert).max()
                if sc_pert < best_score:
                    a = a_pert
                    best_score = sc_pert
                    best_a = a.copy()
                    stall = 0
                    last_improve_score = best_score
                else:
                    a = a.copy()
                    if time.monotonic() - t0 > time_limit * 0.9:
                        break
                    stall = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-linsearch iter={it} t={best_t:.3f} stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _coord_descent(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            ac = _fast_autocorr(a)
            max_ac = ac.max()
            if max_ac <= best_score * (1 + 1e-10):
                continue

            peak_k = int(np.argmax(ac))

            # Compute how a[i] contributes to the peak
            # (a*a)[peak_k] = sum_j a[j] * a[peak_k - j]
            # a[i] appears in this sum when: a[i]*a[peak_k-i] + a[peak_k-i]*a[i] (if i != peak_k-i)
            # or a[i]^2 if i == peak_k-i (i.e., 2*i == peak_k)

            # Try reducing a[i] and distributing to other coordinates
            save_i = a[i]
            if save_i < 1e-12:
                continue

            # Compute gradient contribution at peak
            contrib = 0.0
            j = peak_k - i
            if 0 <= j < n and j != i:
                contrib = 2.0 * a[j]
            elif j == i:
                contrib = 2.0 * a[i]

            if contrib <= 0:
                continue

            # Try moving some mass from i to a random other coordinate
            for attempt in range(5):
                j_dst = rng.integers(n)
                if j_dst == i:
                    continue

                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = save_i * frac
                    if amt < 1e-15:
                        break
                    a[i] = save_i - amt
                    a[j_dst] += amt
                    s_t = a.sum()
                    a = a / s_t
                    sc = _fast_autocorr(a).max()
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                    a[i] = save_i
                    a[j_dst] -= amt

                if improved:
                    break

            if not improved:
                a[i] = save_i

        if not improved:
            break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _stochastic_refine(a0, solver, time_limit, verbose=False):
    """Stochastic perturbations combined with LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        # Apply stochastic perturbation
        noise_scale = 0.02 / (1.0 + 0.1 * epoch)
        noise = rng.normal(0, noise_scale, a.shape)
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0.0)
        s_pert = a_pert.sum()
        if s_pert > 1e-15:
            a_pert = a_pert / s_pert

        # Short LP refinement from perturbed point
        sub_time = min((time_limit - (time.monotonic() - t0)) * 0.3, 10.0)
        a_out, _ = _lp_with_linersearch(a_pert, solver, sub_time, verbose=False)
        sc = _fast_autocorr(a_out).max()
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            a = a_out.copy()
        else:
            # Try the perturbation directly
            sc_pert = _fast_autocorr(a_pert).max()
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert.copy()
                a = a_pert.copy()

        if verbose and epoch % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Stochastic epoch={epoch} ({elapsed:.0f}s): best={best_score:.8f}")

        if time.monotonic() - t0 > time_limit * 0.85:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with LP line search and stochastic restarts."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_curr = len(best_seq)

    # === Phase 1: LSE-FISTA (fast initial convergence) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE-FISTA n={n_curr} (budget={t1:.0f}s) ===")
        a_out, _ = _lse_fista(best_seq, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_curr}")

    # === Phase 2: LP with line search (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-linsearch n={n_curr} (budget={t2:.0f}s) ===")
        a_out, _ = _lp_with_linersearch(best_seq, solver, t2, verbose=True)
        _try_update(a_out, f"lp1_n{n_curr}")

    # === Phase 3: Stochastic refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Stochastic n={n_curr} (budget={t3:.0f}s) ===")
        a_out, _ = _stochastic_refine(best_seq, solver, t3, verbose=True)
        _try_update(a_out, f"stoch_n{n_curr}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_new = len(best_seq) + 200
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-linsearch n={n_new} (budget={t4:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n_new)
        a_out, _ = _lp_with_linersearch(init_long, solver, t4, verbose=True)
        _try_update(a_out, f"lp2_n{n_new}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Coord descent n={n_curr} (budget={t5:.0f}s) ===")
        a_out, _ = _coord_descent(best_seq, t5, verbose=True)
        _try_update(a_out, f"coord_n{n_curr}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    t6 = min(remaining * 0.35, 300.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Final LP n={n_curr} (budget={t6:.0f}s) ===")
        a_out, _ = _lp_with_linersearch(best_seq, solver, t6, verbose=True)
        _try_update(a_out, f"lp_final_n{n_curr}")

    # === Phase 7: Final LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.3, 60.0)
        if t7 > 10:
            n_curr = len(best_seq)
            print(f"\n=== Phase 7: LSE final n={n_curr} (budget={t7:.0f}s) ===")
            a_out, _ = _lse_fista(best_seq, t7, verbose=True)
            _try_update(a_out, f"lse_final_n{n_curr}")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
