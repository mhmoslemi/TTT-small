def lp_best_response_reduced(a, M, active_indices, solver):
    """LP with only a subset of constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Only include constraints at active_indices
    constraints = []
    for k in active_indices:
        # (a*b)[k] = sum_i a[k-i]*b[i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        row = a[k - idx]  # a[k-i] for i in idx
        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    
    # Get dual values for the active constraints
    dual_vals = []
    for c in constraints:
        try:
            d = c.dual_value
            if d is not None and d > 0:
                dual_vals.append(d)
        except:
            pass
    
    return b_val, dual_vals

def subgrad_at(a, k_star):
    """Compute subgradient of (a*a)[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j1 = k_star - i
        if 0 <= j1 < n:
            g[i] += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            g[i] += a[j2]
    return g


def project_nonneg_sum(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum} using water-filling."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    if abs(s - target_sum) < 1e-10:
        return a
    # Need to scale: if s > target_sum, we need to reduce (water-filling down)
    # if s < target_sum, we need to increase (just scale up, but keep non-neg)
    if s > target_sum:
        # Water fill: find tau such that sum(max(0, a - tau)) = target_sum
        a_sorted = np.sort(a)[::-1]
        cumsum = np.cumsum(a_sorted)
        # Find where cumsum - i*tau = target_sum => tau = (cumsum[i] - target_sum)/(i+1)
        # We need tau >= 0 and a_sorted[i] >= tau
        n_nonzero = len(a_sorted)
        tau = 0.0
        for i in range(n_nonzero):
            t_candidate = (cumsum[i] - target_sum) / (i + 1)
            if t_candidate >= 0 and (i + 1 == n_nonzero or a_sorted[i + 1] <= t_candidate):
                tau = t_candidate
                break
            elif t_candidate >= 0:
                tau = t_candidate
        if tau < 0:
            tau = 0.0
        result = np.maximum(a - tau, 0.0)
        rs = result.sum()
        if rs > 1e-15 and abs(rs - target_sum) > 1e-8:
            result = result * target_sum / rs
        return result
    else:
        # s < target_sum: just scale up
        if s < 1e-15:
            return np.full(n, target_sum / n)
        return a * (target_sum / s)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, K_reduced=100,
                          full_lp_every=20):
    """LP-Nesterov with periodic full LP and reduced LP in between."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    active_k = np.arange(2 * n - 1)  # Start with all constraints

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Every full_lp_every iterations, solve full LP and update active set
        if t % full_lp_every == 1 or t == 1:
            b, duals = lp_best_response_full(y, M_curr, solver)
            if b is not None and duals is not None:
                # Update active set based on dual values
                top_dual = np.argsort(duals)[::-1][:K_reduced]
                active_k = top_dual[top_dual > 0]
                if len(active_k) < 10:
                    # Also include top autocorrelation peaks
                    b_full = fast_autocorr(y)
                    top_peaks = np.argsort(b_full)[::-1][:K_reduced]
                    active_k = np.unique(np.concatenate([active_k, top_peaks]))
        else:
            b, _ = lp_best_response_reduced(y, M_curr, active_k, solver)

        if b is None:
            # Fallback: use subgradient direction
            b_full = fast_autocorr(y)
            k_star = int(np.argmax(b_full))
            g = subgrad_at(y, k_star)
            b = np.maximum(norm / (n * g.max() + 1e-15) - g, 0)
            b_sum = b.sum()
            if b_sum < 1e-15:
                b = np.ones(n) / n
            else:
                b = b / b_sum
            b = b * norm  # Rescale

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b * norm / b_sum
        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        # Cap the step to avoid overshooting
        step_norm = np.linalg.norm(g_dir - y)
        if step_norm > norm:
            alpha = alpha * norm / step_norm

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g_dir.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def stochastic_subgrad_nesterov(a0, time_limit, n_max_iters=None, verbose=False):
    """Fast stochastic subgradient with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    n_iters = 0
    T_smoothing = 10.0  # Temperature for soft-max selection

    for t in range(1, n_max_iters or 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        b_max = b.max()
        if b_max < 1e-12:
            break

        # Soft-max selection of peak (smooths the subgradient)
        exp_b = np.exp(T_smoothing * (b - b_max))
        probs = exp_b / exp_b.sum()
        k_star = rng.choice(len(b), p=probs)

        g = subgrad_at(y, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        alpha = 2.0 / (t + 2.0)
        # Scale gradient to unit step
        step = alpha * norm * g / g_norm

        y_new = y - step
        y_new = project_nonneg_sum(y_new, norm)

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_nonneg_sum(a_new, norm)

        y = y_new
        a = a_new
        n_iters += 1

        if n_iters % 1000 == 0:
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    StochSubGrad iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        avg_a += a
        n_avg += 1
        if n_avg % 5000 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

    # Final check
    score_final = fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()

    return best_a, best_score


def lse_objective(x, n, T_temp):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    return 2.0 * n * lse / (s * s)


def lse_gradient(x, n, T_temp):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj_val = 2.0 * n * lse / (s * s)

    # Gradient of LSE w.r.t. a: 4 * (a * w_convolved) where w is padded
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    a_p = np.zeros(fft_size)
    a_p[:n] = a
    w_p = np.zeros(fft_size)
    w_p[:2 * n - 1] = w
    A_fft = rfft(a_p, fft_size)
    W_fft = rfft(w_p, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)
    # The gradient of (a*a) w.r.t. a[i] is 2*(aconv[a][i]) where aconv is the
    # linear convolution with w. We need the "cross-correlation" part.
    # Actually: d/da[i] sum_k w[k]*(a*a)[k] = 2 * sum_k w[k] * (a[k-i] + a[i-k])
    # This equals 2 * (w * a)[i] in the appropriate convolution sense.
    # Let me compute it directly using FFT.
    # (w * a)[i] where * denotes correlation: sum_k w[k] * a[k-i]
    # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives the cross-correlation.
    
    # Simpler: just use the identity that the gradient of max_b (or LSE) w.r.t. a
    # is 2 * sum_k w[k] * (a convolved with delta_k) [i]
    # For each k, the gradient of (a*a)[k] w.r.t. a[i] is a[k-i] + a[i-k] (if valid)
    # So total gradient = 2 * sum_k w[k] * (a[k-i] + a[i-k])
    
    # This is 2 * (w_full * a) where w_full is w padded and reversed appropriately
    # Let me just compute it numerically for correctness
    grad_conv = np.zeros(n)
    for i in range(n):
        val = 0.0
        # sum_k w[k] * (a[k-i] + a[i-k])
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        for k in range(k_min, k_max + 1):
            j1 = k - i
            j2 = i - k
            v = 0.0
            if 0 <= j1 < n:
                v += a[j1]
            if 0 <= j2 < n:
                v += a[j2]
            grad_conv[i] += w[k] * v
    grad_conv *= 2.0

    # Total gradient of 2*n*lse/(s^2)
    grad = 2.0 * n * grad_conv / (s * s) - 2.0 * obj_val * a / s
    return grad


def lse_scipy_optimize(a0, time_limit, T_start=20.0, T_end=1.0, n_stages=5):
    """LSE smooth surrogate optimized with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm / s_c
                    score_c = fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception as e:
            if si == 0:
                print(f"    LSE stage error: {e}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=5, verbose=False):
    """Try to equalize the top peaks by adjusting a."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        peak_vals = b[top_peaks]
        
        if len(top_peaks) < 2:
            break
        
        # Try to reduce the max peak by moving mass from high-contributing
        # indices to low-contributing indices
        k_max = top_peaks[0]
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        
        # Find the top contributor
        top_contrib_local = np.argsort(contrib)[::-1][:10]
        
        improved = False
        for ci in top_contrib_local:
            src = idx[ci]
            if a[src] < 1e-10:
                continue
            # Find a destination that's away from all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for kp in top_peaks:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Sort candidates by current value (prefer moving to low-value spots)
            candidates_sorted = candidates[np.argsort(a[candidates])]
            
            for dst in candidates_sorted[:max(5, len(candidates)//10)]:
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape local minima
            for _ in range(50):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break
            else:
                # No improvement in 50 random tries, try a larger perturbation
                idxs = rng.choice(n, size=max(2, n//50), replace=False)
                a_pert = a.copy()
                for idx in idxs:
                    a_pert[idx] *= rng.uniform(0.8, 1.2)
                a_pert = np.maximum(a_pert, 0.0)
                s_p = a_pert.sum()
                if s_p > 1e-10:
                    a_pert = a_pert * norm / s_p
                    score_p = fast_autocorr(a_pert).max()
                    if score_p < best_score * 1.01:  # Accept slightly worse
                        a = a_pert  # Don't update best, just move
                        break
                break
                # Stuck
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent for final polishing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:8]

        improved = False
        for k_peak in top_peaks[:5]:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:8]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-10:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates_sorted = candidates[np.argsort(a[candidates])]

                for dst in candidates_sorted[:5]:
                    for frac in [0.005, 0.01, 0.03, 0.08, 0.15]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.06)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.15 * np.cos(10.0 * np.pi * x)
        a = np.maximum(a, 0.1)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        a = 1.0 / (1.0 + 10.0 * x)
    elif choice == 6:
        a = (1.0 - x) ** 0.5
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Tries multiple strategies to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = max(100.0, min(float(budget_s) - 60.0, 1720.0))

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 0: Quick scan at a couple other lengths (30s)
    print("\n=== Phase 0: Quick multi-length scan ===")
    scan_lengths = [n_base + 400, n_base - 200]
    scan_lengths = [s for s in scan_lengths if 800 <= s <= 4000]
    
    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.05:
            break
        scan_time = min(15.0, (time_budget * 0.05 - (time.monotonic() - t_global)) / 2)
        if scan_time < 3:
            break
        init_scan = interpolate_sequence(best_seq, n_scan)
        a_norm = init_scan * np.sqrt(2.0 * n_scan) / init_scan.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False,
                                              K_reduced=60, full_lp_every=30)
        do_try_update(a_out, f"scan_n{n_scan}")
        print(f"  Scan n={n_scan}: {sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 1: Deep LP-Nesterov at base length (45% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Deep LP-Nesterov n={n_best} (budget={t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, sc_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=True,
                                          K_reduced=80, full_lp_every=15)
    do_try_update(a_out, f"lp_deep_n{n_best}")

    # Phase 2: Stochastic subgradient Nesterov (25% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.25
    print(f"\n=== Phase 2: Stochastic SubGrad n={n_best} (budget={t2:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, sc_out = stochastic_subgrad_nesterov(a_norm, t2, verbose=True)
    do_try_update(a_out, f"stoch_subgrad_n{n_best}")

    # Phase 3: LSE + L-BFGS (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.10
    print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = lse_scipy_optimize(a_norm, t3, T_start=30.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Peak equalization / coordinate descent (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Peak equalization n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = peak_equalization(a_norm, t4, n_peaks=6, verbose=True)
        do_try_update(a_out, f"peak_eq_n{n_best}")

    # Phase 5: Final LP-Nesterov polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.10
    print(f"\n=== Phase 5: LP polish n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t5, solver, verbose=True,
                                              K_reduced=50, full_lp_every=10)
        do_try_update(a_out, f"lp_polish_n{n_best}")

    # Phase 6: Final coordinate descent (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = max(remaining * 0.6, 10.0)
    print(f"\n=== Phase 6: Final coord descent n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = coordinate_descent_polish(a_norm, t6, seed=77777)
        do_try_update(a_out, "coord_final")

    # Phase 7: One more LP round if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t7 = remaining * 0.7
        print(f"\n=== Phase 7: Extra LP n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t7, solver, verbose=True,
                                              K_reduced=60, full_lp_every=10)
        do_try_update(a_out, f"lp_extra_n{n_best}")

        # Final coord descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t8 = remaining * 0.8
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, sc_out = coordinate_descent_polish(a_norm, t8, seed=424242)
            do_try_update(a_out, "coord_extra")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
