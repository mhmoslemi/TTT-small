        # Actually wait: idx1[i] = k_star - i and idx2[i] = i - k_star
        # These are equal when k_star - i = i - k_star, i.e., 2*k_star = 2*i, i.e., i = k_star
        # But k_star can be up to 2n-2, so i = k_star is only valid if k_star < n
        # In that case, a[k_star - i] = a[0] and a[i - k_star] = a[0], so we add 2*a[0]
        # But the actual derivative should be a[0] (since it's the same term a_i * a_{k-i} = a_{k_star} * a_0)

        g = np.zeros(n)
        for i in range(n):
            # Term from j=i: a[k-i] if 0 <= k-i < n
            if 0 <= k_star - i < n:
                g[i] += a[k_star - i]
            # Term from j=k-i: a[i] if 0 <= k-i < n and k-i != i
            if 0 <= k_star - i < n and (k_star - i) != i:
                g[i] += a[i]
            # Wait, this is wrong. Let me think again.
            # conv(a,a)_k = sum_j a_j * a_{k-j}
            # d/d a_i = sum_j [delta(j,i) * a_{k-j} + a_j * delta(k-j, i)]
            #           = a_{k-i} (if 0<=k-i<n) + a_{k-i} (if 0<=k-i<n and k-i != i)
            #           = a_{k-i} + a_{k-i} * [k-i != i] (if 0<=k-i<n)
            #           = 2*a_{k-i} - a_{k-i}*[k=2i] (if 0<=k-i<n)
            #           = 2*a_{k-i} (if 0<=k-i<n and k != 2i)
            #           = a_{k-i} (if 0<=k-i<n and k == 2i)
            #           = 0 (if not (0<=k-i<n))
            pass
        
        # OK let me just do it the simple way:
        g = np.zeros(n)
        for i in range(n):
            if 0 <= k_star - i < n:
                if k_star == 2 * i:
                    g[i] = a[i]  # = a[k_star - i]
                else:
                    g[i] = 2.0 * a[k_star - i]
        
        # Mirror descent update
        # a_new_i ∝ a_i * exp(-eta * g_i / max(g))  (normalize gradient)
        g_max = g.max()
        if g_max > 1e-15:
            logits = np.log(a) - eta * g / g_max
            logits -= logits.max()  # for numerical stability
            a_new = np.exp(logits)
            a_new = a_new / a_new.sum()
        else:
            a_new = a.copy()
        
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adapt eta
        eta *= 0.9999
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): peak={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid strategy:
1. LP-Nesterov with adaptive M (primary optimizer)
2. LSE-smoothed FISTA with aggressive temperature annealing
3. Mirror descent (multiplicative weights) for fast fine-tuning
4. Multi-length strategy (base -> 2000 -> 2500)
5. Targeted peak shaving with gradient-informed moves
Key improvements over previous: adaptive M relaxation, mirror descent,

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
    HAS_CP = True
except ImportError:
    HAS_CP = False

_global_solver = None


def _get_solver():
    global _global_solver
    if not HAS_CP:
        return None
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def lse_grad(y, T):
    n = len(y)
    ac = fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), n / 100.0)
        step = 1.0 / L
        t_fista = 1
        staging = a.copy()
        max_iters = 50000

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = lse_grad(y, T)
            z = y - step * grad
            a_new = proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 200 == 0:
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov(a0, time_limit, solver, verbose=True, m_relax=1.0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_curr = fast_autocorr(y)
        M_curr = b_curr.max()
        if M_curr < 1e-12:
            break

        M_use = M_curr * m_relax
        b = lp_best_response(y, M_use, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={n_iters} ({elapsed:.0f}s): peak={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, eta_scale=1.0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-12)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_scale / n
    a_log = np.log(a / a.sum())

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        g = np.zeros(n)
        idx1 = k_star - np.arange(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        if k_star == 0:
            g[valid1] = a[idx1[valid1]]
        else:
            g[valid1] = 2.0 * a[idx1[valid1]]
            zero_idx = k_star // 2
            if k_star % 2 == 0 and 0 <= zero_idx < n:
                g[zero_idx] = a[zero_idx]

        g_max = g.max()
        if g_max < 1e-15:
            break

        eta_t = eta / (1.0 + 0.01 * t * eta / n)
        weight = a_log - eta_t * g / g_max
        weight -= weight.max()
        a_new = np.exp(weight)
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a = a_new / s_new * norm
        # Renormalize for log
        a_log = np.log(np.maximum(a, 1e-30)) - np.log(a.sum())

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): peak={best_score:.8f}")

    return best_a, best_score

def peak_shave(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Build peak mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Sort by current value (prefer moving to low-mass positions)
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 3)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape
            stall += 1
            n_rand = rng.integers(3, 10)
            srcs = rng.choice(n, size=n_rand, replace=False)
            dsts = rng.choice(n, size=n_rand, replace=False)
            a_trial = a.copy()
            for i, j in zip(srcs, dsts):
                if i != j and a_trial[i] > norm * 1e-8:
                    amt = a_trial[i] * rng.uniform(0.01, 0.05)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 300:
                break

        if verbose and (stall % 50 == 0):
            elapsed = time.monotonic() - t0
            print(f"    PeakShave stall={stall} ({elapsed:.0f}s): peak={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
