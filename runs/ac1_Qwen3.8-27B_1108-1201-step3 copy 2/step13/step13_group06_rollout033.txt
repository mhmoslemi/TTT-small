def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(z, T):
    """
    Compute LogSumExp(conv(softmax(z), softmax(z))/T) and gradient w.r.t. z.
    
    Returns (objective_value, gradient_vector).
    """

    n = len(a)
    C = fast_autocorr(a)  # length 2n-1

    C_scaled = C / T
    C_max = C_scaled.max()
    exp_C = np.exp(C_scaled - C_max)
    lse_val = C_max + np.log(exp_C.sum())

    w = exp_C / exp_C.sum()
    
    # Compute h[j] = sum_i a[i] * w[j+i] for j=0..n-1
    # = conv(flip(a), w)[n-1 : 2n-1]

    h = conv_aw[n-1 : 2*n-1]
    
    # sum(w * C)
    wc = np.dot(w, C)
    
    # Gradient
    grad_z = (2.0 / T) * a * (h - wc)
    
    return lse_val, grad_z

            best_a = a_new.copy()
            z = z_new  # warm start for next stage
        else:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization of LogSumExp(conv/T) with multi-temperature annealing in softmax space
2. LP Nesterov zero-sum game for refinement
3. Aggressive peak shaving with global peak verification
4. Multi-length exploration with interpolation
5. Multiplicative weights (mirror descent) for final polish

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


# === Core utility functions ===

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(z, T):
    """
    Compute LogSumExp(conv(softmax(z), softmax(z))/T) and gradient w.r.t. z.
    Returns (objective_value, gradient_vector).
    """
    a = softmax_fn(z)
    n = len(a)
    C = fast_autocorr(a)

    C_scaled = C / T
    C_max = C_scaled.max()
    exp_C = np.exp(C_scaled - C_max)
    lse_val = C_max + np.log(exp_C.sum())
    w = exp_C / exp_C.sum()

    # h[j] = sum_i a[i] * w[j+i] for j=0..n-1
    # Computed as conv(flip(a), w)[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (len(a_flip) + len(w) - 1).bit_length()
    A_flip = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    conv_aw = irfft(A_flip * W, fft_size)
    h = conv_aw[n - 1:2 * n - 1]

    wc = np.dot(w, C)
    grad_z = (2.0 / T) * a * (h - wc)

    return lse_val, grad_z


class LSEObjective:
    """Callable class for L-BFGS to avoid lambda/closure issues."""
    def __init__(self, T):
        self.T = T

    def __call__(self, z):
        return lse_obj_grad(z, self.T)

def lbfgs_lse_optimize(a0, time_limit, T_list=None, max_iter=400):

# === LP Nesterov method ===

def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    stall_count = 0
    warm_b = None

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if stall_count > 400 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            stall_count = 0
            warm_b = None

    return best_a, best_score

def peak_shaving(a0, time_limit, seed=42):
    """Systematic peak shaving with aggressive moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(8, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:12]:
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                n_pert = rng.integers(2, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score

def mirror_descent_refine(a0, time_limit, eta=0.008, seed=999):
    """Multiplicative weights refinement targeting top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    current_eta = eta

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        importance = np.zeros(n)
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[k - idx]

        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax

        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 40 == 0:
                current_eta *= 0.7
                if current_eta < 1e-7:
                    current_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn2 = a.sum()
                    if sn2 > 1e-10:
                        a = a / sn2

            if stall > 400:
                break

    return best_a, best_score


# === Sequence interpolation ===

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.sin(np.pi * x) ** 2
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2) + 0.4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


# === Main search function ===

def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. L-BFGS LSE with aggressive multi-temperature annealing
    2. LP Nesterov zero-sum game
    3. Peak shaving + mirror descent
    4. Multi-length exploration
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create one
    if best_seq is None:
        n_default = 1600
        best_seq = structured_init(n_default, rng, choice=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: L-BFGS LSE on parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS LSE n={n_base} ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        T_schedule_1 = [100.0, 50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.75, 0.4, 0.2,
                        0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
        a_out, _ = lbfgs_lse_optimize(best_seq, t1, T_list=T_schedule_1, max_iter=300)
        try_update(a_out, f"lbfgs_n{n_base}")

    # === Phase 2: L-BFGS from perturbed parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: L-BFGS perturbed n={n_base} ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 20:
        pert = perturbed_init(best_seq, rng, scale=0.04)
        T_schedule_2 = [30.0, 15.0, 7.5, 3.7, 1.8, 0.9, 0.45, 0.22, 0.11, 0.055,
                        0.028, 0.014, 0.007, 0.0035, 0.0018, 0.0009]
        a_out, _ = lbfgs_lse_optimize(pert, t2, T_list=T_schedule_2, max_iter=250)
        try_update(a_out, f"lbfgs_pert_n{n_base}")

    # === Phase 3: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: LP Nesterov n={n_base} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Peak shaving n={n_base} ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_out, _ = peak_shaving(best_seq, t4, seed=777)
        try_update(a_out, "peak_shave_1")

    # === Phase 5: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Mirror descent n={n_base} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_out, _ = mirror_descent_refine(best_seq, t5, eta=0.004, seed=555)
        try_update(a_out, "mirror_1")

    # === Phase 6: L-BFGS on longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    print(f"=== Phase 6: L-BFGS n={n_long} ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        T_schedule_6 = [50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.75, 0.37, 0.18, 0.09,
                        0.045, 0.022, 0.011, 0.005, 0.0025, 0.001]
        a_out, _ = lbfgs_lse_optimize(init_long, t6, T_list=T_schedule_6, max_iter=200)
        try_update(a_out, f"lbfgs_n{n_long}")

    # === Phase 7: L-BFGS on even longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long2 = n_base + 400
    if remaining > 40:
        print(f"=== Phase 7: L-BFGS n={n_long2} ===")
        t7 = min(remaining * 0.10, 150.0)
        if t7 > 25:
            init_long2 = interpolate_sequence(best_seq, n_long2)
            T_schedule_7 = [40.0, 20.0, 10.0, 5.0, 2.5, 1.25, 0.6, 0.3, 0.15, 0.075,
                            0.037, 0.018, 0.01, 0.005, 0.002, 0.001]
            a_out, _ = lbfgs_lse_optimize(init_long2, t7, T_list=T_schedule_7, max_iter=200)
            try_update(a_out, f"lbfgs_n{n_long2}")

    # === Phase 8: Final L-BFGS polish (very low T) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"=== Phase 8: Final L-BFGS polish ===")
        t8 = min(remaining * 0.20, 200.0)
        if t8 > 15:
            T_final = [2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.004,
                       0.002, 0.001, 0.0005, 0.0002, 0.0001]
            a_out, _ = lbfgs_lse_optimize(best_seq, t8, T_list=T_final, max_iter=200)
            try_update(a_out, "final_lbfgs")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.35, 80.0)
        if t9 > 8:
            print(f"=== Phase 9: Final peak shave ===")
            a_out, _ = peak_shaving(best_seq, t9, seed=1001)
            try_update(a_out, "peak_shave_final")

    # === Phase 10: Quick mirror descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t10 = min(remaining * 0.40, 60.0)
        if t10 > 8:
            print(f"=== Phase 10: Final mirror descent ===")
            a_out, _ = mirror_descent_refine(best_seq, t10, eta=0.002, seed=2001)
            try_update(a_out, "mirror_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
