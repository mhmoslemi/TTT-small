Minimizes J = 2*n*max_auto / (sum_sq)^2 
where max_auto is the maximum of the 2*n-1 elements of the linear convolution (autocorrelation) of the sequence with itself, 
sum_sq = sum of the sequence (J is undefined if sum < 0.01).

The approach integrates continuous landscapes and discrete search:
1. Starts from a normalized sine-like sequence (length=1600).
2. Explores different resolutions using sine interpolation (lengths 2000, 2400, 3200, 4000).
3. Uses population-based rescaling via genetic algorithm operations to search within each resolution.
4. Maintains a population centroid and individual vectors to balance exploration and exploitation.
5. Employs a multi-phased optimization with crossover and mutation to continuously improve the candidate sequence.
6. Uses explicit fitness evaluation and aggressive pruning to stay within 1800s time limit.
"""

def normalize_sequence(seq, n):
    """Normalizes sequence to have length n and sum sqrt(2*n)"""]
    old_n = len(seq)
    if old_n == n:
        return seq * np.sqrt(2*n) / np.sum(seq) if np.sum(seq) > 0 else None
    low = np.min(seq * (1 - 0.001))
    high = np.max(seq * (1 + 0.001))
    if low < 0 or high > 1000 or np.any(np.isinf(seq)):
        print("Found unsafe sequence: clipping")
        safe_seq = np.clip(seq, 0.0, 1000.0)
        safe_seq_sum = np.sum(safe_seq)
        if safe_seq_sum < 1e-15: 
            return np.ones(n) * 0.01 / n * np.sqrt(2*n)  # Safe fallback
        seq = safe_seq * np.sqrt(2*n) / safe_seq_sum
    try:
        seq_norm = np.interp(np.linspace(0.001, 0.999, n),
                             np.linspace(0.001, 0.999, old_n), 
                             seq / np.sqrt(2*n) if np.sum(seq) > 0 else np.ones(n) * 0.01)
    except Exception:
        print("Interpolation error: using default")
        seq_norm = np.ones(n) * 0.01 * np.sqrt(2*n) / np.sqrt(2*n)
    return seq_norm

def calculate_fitness(seq, n):
    """Computes J for a normalized sequence via interpolation"""
    try:
        seq_normalized = normalize_sequence(seq, n)
        if seq_normalized is None:
            return float('inf')
        seq_float = [float(x) for x in seq_normalized]
        convolve_result = np.convolve(seq_float, seq_float)
        max_val = np.max(convolve_result)
        sum_val = np.sum(seq_float)
        return float(2*n*max_val/sum_val**2) if sum_val > 0 else float('inf')
    except:
        return float('inf')

def mutation_sequence(p):
    """Mutate a sequence: random rescaler"""
    if random.random() < 0.99:
        s_p = np.sum(p)
        if s_p < 1e-10:
            return np.ones(len(p)) * 0.01 * (np.sqrt(2*len(p)) / 0.01)
        scale = np.sqrt(2*len(p)) / s_p
        p_mut = [x * scale for x in p]
            if random.random() < 0.1:
                sigma = 0.1 * np.std(p)
                mut_point = random.randint(1, len(p)-1)
                p_mut[mut_point] = max(0, p_mut[mut_point] + np.random.normal(0, sigma))
            }
        return p_mut
    # Cross-like mutation
    cutoff = random.randint(1, len(p)-1)
    a = [max(0, 0.9 + 0.05 * random.random()) for _ in range(cutoff)]
    b_part = p[:cutoff] + np.random.normal(0, 0.1, cutoff)
    b_part = [min(max(0, x), 1000) for x in b_part]
    b_slice = p[cutoff:]
    return a + b_slice

        # Step 3.1: Start with base sequence (from parent if available)

        seq_initial = [float(x) for x in (seq_initial + np.random.rand(MAX_CANDIDATES-len(seq_initial))*0.01)[:MAX_CANDIDATES]]

        best_seq_local = None
        best_score_local = float('inf')
        # Step 3.2: Apply candidate-specific crossover and mutation
        for _ in range(50):  # This is fixed at 50 generations per length
            try:
                # Candidate sequence acquisition
                new_seq = crossover_sequence(seq_initial, [max(0, x) for x in seq_initial])
                new_seq_score = evaluate_safe(new_seq)
                if new_seq_score < best_score_local or best_seq_local is None:

                if new_seq_score < current_best_score:
                    current_best_score = new_seq_score
                    current_best = new_seq.copy()
            except:
                continue
            # Periodic progress check
            # This is unlikely to hit, but if time runs out, we return
            if time.time() - start_time > budget_in_seconds:
                break
        # Record the winner for this length
        length_score = best_score_local if best_score_local is not None else float('inf')
        print(f"{idx} Length {n_target} completed: score={length_score:.4f} best world={current_best_score:.4f}")
        if length_score <= GOAL and length_score < (current_best_score if best_seq_local else float('inf')):
            current_best_score = length_score
            current_best = best_seq_local

            if not final_candidate:
                final_candidate = crossover_sequence(best_seq_local, [max(0, x) for x in best_seq_local])
