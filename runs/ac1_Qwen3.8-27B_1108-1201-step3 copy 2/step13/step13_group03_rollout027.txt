def _fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    """Gradient of LSE-smoothed max autocorrelation."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    return 2.0 * cross[n - 1: n - 1 + n]

                t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                mom = (t_k - 1.0) / t_k_new
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                sy = y_new.sum()
                if sy > 1e-15:
                    y_new = y_new / sy
                else:
                    y_new = a_new.copy()
                t_k = t_k_new
            
            staging = a_new
            a = a_new
            y = y_new
            
            if (it + 1) % 200 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _build_conv_matrix(a, n):
    """Build the convolution matrix T such that (T @ b) = conv(a, b)."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _lp_nesterov(a0, time_limit, solver, verbose=False):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa
        
        y = y_new
        a = a_new
        
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if n_iters % 50 == 0:
            score_a = _fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        peak_idx = int(np.argmax(b))
        
        # Find the source of the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-15:
            break
        
        src = idx[np.argmax(contrib)]
        
        # Find a good destination (low contribution to peaks)
        peak_mask = np.zeros(n, dtype=bool)
        top_peaks = np.argsort(b)[::-1][:5]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2+1] = True
        
        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]
        
        # Pick from low-value candidates
        candidates = candidates[np.argsort(a[candidates])[:min(len(candidates), n//4)]]
        dst = candidates[rng.integers(len(candidates))]
        
        improved = False
        for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial * norm / st
            score = _fast_autocorr(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial * norm / st
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            stall += 1
            if stall > 100:
                break
    
    return best_a, best_score

def _make_initializations(n, rng, parent=None):
    """Generate diverse parametric initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # Gaussian windows (different widths)
    center = n / 2
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        a = np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n))**2)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Sine power windows
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.sin(np.pi * x)**p
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Chirp-derived (magnitude of complex chirp, symmetrized)
    for M in [1, 2, 3, 5]:
        idx = np.arange(n)
        a = np.abs(np.cos(2 * np.pi * M * idx**2 / (n**2)))
        a = np.maximum(a, 1e-10)
        # Symmetrize
        a = 0.5 * (a + a[::-1])
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Random smooth (sum of cosines)
    for _ in range(3):
        a = np.ones(n)
        for _f in range(3):
            f = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * x + phase)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Parent-based perturbations
    if parent is not None:
        parent_resampled = np.interp(x, np.linspace(0, 1, len(parent)), parent / parent.sum())
        parent_resampled = np.maximum(parent_resampled, 1e-10)
        parent_resampled = parent_resampled / parent_resampled.sum()
        inits.append(parent_resampled)
        
        for _ in range(3):
            noise_scale = rng.uniform(0.1, 0.5)
            noise = np.ones(n)
            for _f in range(2):
                f = rng.uniform(1.0, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                noise += noise_scale * np.cos(2 * np.pi * f * x + phase)
            a = parent_resampled + 0.1 * np.maximum(noise, 0)
            a = np.maximum(a, 1e-10)
            inits.append(a / a.sum())
    
    return inits
