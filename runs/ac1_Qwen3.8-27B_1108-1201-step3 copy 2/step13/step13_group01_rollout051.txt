def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def lp_best_response_topk(a, M, k, solver):
    """
    LP best response with top-k peak constraints.
    Maximizes sum(b) subject to:
    - b >= 0
    - (a*b)[j] <= M for j in top-k peaks of (a*a)
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    # Build the convolution matrix (sparse would be better but let's use dense for simplicity)
    # T[j, i] = a[j-i] if 0 <= j-i < n, else 0
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[i:end, i] = a[:end-i]
    
    # Find top-k peaks of a*a
    aa = fast_autocorr(a)
    top_k = np.argsort(aa)[::-1][:k]
    
    constraints = [T[top_k] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def spectral_flatten(a, alpha=0.1):
    """
    Apply spectral flattening: blend the power spectrum with its average.
    This reduces time-domain peaks.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    power = np.abs(A) ** 2
    avg_power = power.mean()
    # Blend: new_power = (1-alpha)*power + alpha*avg_power
    # But we need to maintain the phase
    new_power = (1 - alpha) * power + alpha * avg_power
    # Recover signal: new_A = sqrt(new_power/ power) * A (preserve phase)
    # Actually: new_A = A * sqrt(new_power / power) where power > 0
    ratio = np.ones_like(A)
    mask = power > 1e-15
    ratio[mask] = np.sqrt(new_power[mask] / power[mask])
    new_A = A * ratio
    new_a = irfft(new_A, fft_size)[:n]
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-15:
        return a.copy()
    return new_a / s * np.sum(a)

def diverse_init(n, rng):
    """Generate diverse initializations."""
    x = np.linspace(0.0, 1.0, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        # Cosine window
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Exponential decay from left
        a = np.exp(-3.0 * x)
    elif choice == 3:
        # Two Gaussians
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        # Random smooth (Gaussian process-like)
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.3, 1.0, size=5)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.0)
    elif choice == 5:
        # Asymmetric ramp with noise
        a = (1 - x) + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    elif choice == 6:
        # Concentrated on left with tail
        a = np.exp(-5.0 * x) + 0.1
    else:
        # Pure random
        a = rng.uniform(0.1, 1.0, n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def lse_objective_and_grad(a_flat, n, T_temp):
    """Combined objective and gradient for LSE smooth surrogate."""
    a = a_flat.reshape(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a_flat)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros_like(a_flat)
    
    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient via FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()

def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    """LSE smooth optimization with scipy L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 180.0)
        
        x0 = a.copy()
        
        def obj_func(x):
            return lse_objective_and_grad(x, n, T)[0]
        
        def grad_func(x):
            return lse_objective_and_grad(x, n, T)[1]
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func, 
                                  options={'maxiter': 300, 'maxfun': 800,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception as e:
            pass
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Improved coordinate descent:
    - Target top peaks specifically
    - Try both increasing and decreasing coordinates
    - Use line search for optimal step size
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        # For each top peak, try to reduce it
        for k_peak in top_peaks[:6]:
            # Elements that contribute to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Get top contributors
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]
            
            # Build peak mask (elements involved in any top peak)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Try moving mass from contributors to non-peak elements
            for src in top_contrib[:4]:
                if a[src] < 1e-10:
                    continue
                
                # Candidates: elements NOT in peak mask, or with low values
