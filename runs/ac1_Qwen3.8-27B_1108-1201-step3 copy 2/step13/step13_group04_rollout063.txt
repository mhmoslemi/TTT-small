Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response iteration
with Polyak averaging, multiple restarts, and peak-targeting local search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a):
    """Build sparse matrix T where T[k, m] = a[k-m] if 0<=k-m<n, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for m in range(n):
        for k in range(m, m + n):
            idx = k - m  # a[idx]
            rows.append(k)
            cols.append(m)
            vals.append(a[idx])
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

def lp_solve(a, M, method='highs'):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)]*n, method=method)
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_best_response_with_averaging(a0, time_limit, verbose=True):
    """LP best-response with Polyak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        M = fast_autocorr(a).max()
        if M < 1e-10:
            break
        
        b = lp_solve(a, M)
        if b is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b = b / b_sum  # normalize to unit sum
        
        # Update
        a = b
        it += 1
        
        # Track best individual
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Polyak average
        avg_a += a
        n_avg += 1
        
        # Check average
        if n_avg % 10 == 0:
            avg = avg_a / n_avg
            avg_score = fast_autocorr(avg).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it} ({elapsed:.0f}s): current_max={M:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response iteration
with Polyak averaging, diverse restarts, and peak-targeting local search.
Key approach: iterative LP where we fix the first factor and optimize the second
to maximize total mass under the current peak constraint, then normalize and repeat.
Polyak averaging over iterations provides convergence guarantees.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_T(a):
    """Build sparse convolution matrix T where T[k,m] = a[k-m] if 0<=k-m<n."""
    n = len(a)
    rows = np.repeat(np.arange(n), n)
    cols = np.concatenate([np.arange(m, m + n) for m in range(n)])
    data = np.tile(a, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_given_a(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    T = build_T(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def solve_lp_min_M(a, b_init=None):

    Returns (M_opt, b_opt) or (None, None)."""

    T = build_T(a)

    c[-1] = 1.0  # minimize M
    # Constraints: T @ b <= M => T @ b - M * 1 <= 0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])

    # Equality: sum(b) = 1
    A_eq = np.hstack([np.ones(n), np.zeros(1)])
    b_eq = np.array([1.0])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b_opt = np.maximum(res.x[:n], 0.0)
        M_opt = res.x[-1]
        return M_opt, b_opt
    return None, None

                print(f"  LP failed at iter {it}, trying min-M formulation")
            # Fallback: use min-M formulation
            M_opt, b = solve_lp_min_M(a)
            if b is None:
                if verbose:
                    print(f"  Both LP formulations failed, stopping")
                break

        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        b = b / b_sum
        a = b
        it += 1

        # Track best individual iterate
        cur_max = fast_autocorr(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # Polyak average
        avg_a += a
        n_avg += 1

        # Check average periodically
        if n_avg % 5 == 0:
            avg = avg_a / n_avg
            avg_max = fast_autocorr(avg).max()
            if avg_max < best_max:
                best_max = avg_max
                best_a = avg.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  iter={it} ({elapsed:.0f}s): M={M:.10f}, best={best_max:.10f}")

    if verbose:
        print(f"  LP iteration complete: {it} iters, best_max={best_max:.10f}")

    return best_a


def lp_iteration_momentum(a0, deadline, alpha=0.3, verbose=True):
    """LP best-response with momentum (convex combination update)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    it = 0

    while time.monotonic() < deadline:
        M = fast_autocorr(a).max()
        if M < 1e-12:
            break

        b = solve_lp_given_a(a, M)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        b = b / b_sum

        # Momentum update: a = (1-alpha)*a + alpha*b
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = b.copy()
        else:
            a_new = a_new / ss
        a = a_new
        it += 1

        cur_max = fast_autocorr(a).max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg = avg_a / n_avg
            avg_max = fast_autocorr(avg).max()
            if avg_max < best_max:
                best_max = avg_max
                best_a = avg.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  mom-iter={it} ({elapsed:.0f}s): M={M:.10f}, best={best_max:.10f}")

    if verbose:
        print(f"  Momentum LP complete: {it} iters, best_max={best_max:.10f}")

    return best_a


def local_search(a0, deadline, seed=42):
    """Peak-targeting local search: move mass away from peak-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    no_improve = 0

    while time.monotonic() < deadline:
        C = fast_autocorr(a)
        top_ks = np.argsort(C)[::-1][:3]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_src = idx[np.argsort(contrib)[::-1][:5]]

            for src in top_src:
                if a[src] < 1e-10:
                    continue

                # Build peak mask
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 2:
                    candidates = np.where(np.arange(n) != src)[0]

                # Try multiple destinations and fractions
                tried = 0
                while tried < 5 and time.monotonic() < deadline:
                    dst = candidates[rng.integers(len(candidates))]
                    if dst == src:
                        tried += 1
                        continue
                    tried += 1

                    for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_new = a.copy()
                        a_new[src] -= amt
                        a_new[dst] += amt
                        a_new = np.maximum(a_new, 0.0)
                        ss = a_new.sum()
                        if ss < 1e-15:
                            continue
                        a_new = a_new / ss
                        new_max = fast_autocorr(a_new).max()
                        if new_max < best_max:
                            a = a_new
                            best_max = new_max
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 3 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.08)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
            if no_improve > 300:
                break

    return best_a


def lse_gradient(a, T):
    """Compute gradient of LSE surrogate for the max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w = w / w_sum

    # grad[m] = sum_k w[k] * d(conv(a,a)[k])/da[m]
    # d(conv(a,a)[k])/da[m] = a[k-m] + a[m-(k-(n-1))] ... 
    # Actually: conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d a[m] = a[k-m] (if 0<=k-m<n) + a[m-(k-... )] 
    # More precisely: d/d a[m] of sum_i a[i]*a[k-i] = a[k-m] + (if m appears in second factor) a[k-m]
    # Wait: the term a[m]*a[k-m] appears when i=m, and a[i]*a[m] appears when k-i=m, i.e., i=k-m.
    # So d/d a[m] = a[k-m] (from i=m) + a[k-m] (from i=k-m) if k != 2m... 
    # Actually by symmetry: d/d a[m] of conv(a,a)[k] = a[k-m] if 0<=k-m<n, plus 
    #   a[k-m] if k-m != m (the cross term)
    # Let me just use: grad[m] = sum_k w[k] * (a[k-m] * [0<=k-m<n] + a[k-m] * [0<=k-m<n and k-m != m])
    # This simplifies to: for k != 2m: contribution is 2*a[k-m]*w[k], for k=2m: contribution is a[m]*w[k]
    # Actually it's simpler: grad[m] = sum_k w[k] * (a[k-m] if 0<=k-m<n else 0) 
    #   + sum_k w[k] * (a[k-m] if 0<=k-m<n else 0)  [the two terms are the same!]
    # Wait no. Let me be very careful.
    # conv(a,a)[k] = sum_{i} a[i] * a[k-i] where 0<=i<n and 0<=k-i<n
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
    # d/d a[m]: the term a[m]*a[k-m] appears when i=m (contribution a[k-m])
    #           the term a[i]*a[m] appears when k-i=m, i.e., i=k-m (contribution a[k-m])
    # If m != k-m (i.e., m != k/2), both terms are distinct, so total = 2*a[k-m]
    # If m == k-m (i.e., m = k/2, k must be even), the term is a[m]^2, derivative = 2*a[m]
    # So in all cases: d/d a[m] of conv(a,a)[k] = 2*a[k-m] if 0<=k-m<n, else 0
    # Wait that's not right either. When i=m: we get a[k-m]. When i=k-m: we get a[k-m] = a[m]... no.
    # Let me redo: conv(a,a)[k] = sum_i a[i] * a[k-i]
    # term with a[m] in first position: i=m, gives a[m]*a[k-m]
    # term with a[m] in second position: k-i=m => i=k-m, gives a[k-m]*a[m]
    # These are the same term! So the total contribution of a[m] to conv(a,a)[k] is:
    #   a[m]*a[k-m] (appears once, as long as both 0<=m<n and 0<=k-m<n)
    # So d/d a[m] = a[k-m] (from first position) + a[k-m] (from second position) = 2*a[k-m]
    # BUT only if m != k-m. If m = k-m (i.e., 2m=k), then the term is a[m]^2 and derivative is 2*a[m].
    # In both cases: d/d a[m] = 2*a[k-m] when 0<=k-m<n.
    # Hmm wait, when m = k-m: the term a[m]*a[k-m] = a[m]*a[m] = a[m]^2. 
    # In the sum, this appears exactly once (at i=m=k-m). Its derivative w.r.t. a[m] is 2*a[m].
    # In the formula 2*a[k-m] = 2*a[m]. Same! So the formula d/d a[m] = 2*a[k-m] (if 0<=k-m<n) works in all cases.
    
    # So: grad[m] = 2 * sum_{k: 0<=k-m<n} w[k] * a[k-m]
    # = 2 * (w * a_rev)[m + n - 1]  where a_rev = a[::-1]
    # Because (w * a_rev)[j] = sum_k w[k] * a_rev[j-k] = sum_k w[k] * a[n-1-(j-k)]
    # For j = m + n - 1: = sum_k w[k] * a[n-1-m-n+1+k] = sum_k w[k] * a[k-m]
    
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)
    conv = irfft(W * A_rev, fft_size)
    r1 = conv[n - 1: n - 1 + n]
    return 2.0 * r1


def lse_polish(a0, deadline, T_start=5.0, T_end=0.3, verbose=True):
    """LSE gradient descent with Nesterov acceleration as a polishing step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    y = a.copy()
    y_prev = a.copy()
    it = 0

    for si, T in enumerate(temps):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        n_iters = max(20, int(remaining / (len(temps) - si) * 1.5))

        for _ in range(n_iters):
            if time.monotonic() >= deadline:
                break

            # Nesterov extrapolation
            if it == 0:
                y_ext = y
            else:
                mom = (it - 1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / (2 * (it + 2))
                y_ext = y + mom * (y - y_prev)
                y_ext = np.maximum(y_ext, 0)
                ss = y_ext.sum()
                if ss > 1e-15:
                    y_ext = y_ext / ss

            grad = lse_gradient(y_ext, T)
            lr = 1.0 / (it + 3)
            y_new = y - lr * grad
            y_new = np.maximum(y_new, 0.0)
            ss = y_new.sum()
            if ss < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / ss

            y_prev = y.copy()
            y = y_new
            it += 1

            cur_max = fast_autocorr(y).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = y.copy()

        if verbose:
            print(f"  LSE stage {si+1}/{n_stages} T={T:.3f}: best={best_max:.10f}")

    return best_a


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate():
    """
    Main search function. Combines LP best-response with Polyak averaging,
    momentum LP, LSE polishing, and peak-targeting local search across
    multiple initializations and sequence lengths.
    """
    t_start = time.monotonic()
    time_budget = 1710.0  # Leave 90s margin within 1800s

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_eval = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, eval={p_eval:.8f}")
    except Exception:
        pass

    if parent is None:
        n = 1600
        parent = np.ones(n) / n
        p_eval = evaluate_safe(parent)
        print(f"Default uniform init: n={n}, eval={p_eval:.8f}")

    best_seq = parent.copy()
    best_eval = p_eval

    def try_update(a, label):
        nonlocal best_seq, best_eval
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_eval:
            best_eval = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {best_eval:.8f} (n={len(a_norm)}, t={time.monotonic()-t_start:.0f}s)")

    # ============ Phase 1: LP best-response on parent ============
    deadline1 = t_start + min(700.0, time_budget * 0.40)
    print(f"\n{'='*60}")
    print(f"Phase 1: LP best-response on parent (n={len(parent)}, budget={deadline1-t_start:.0f}s)")
    print(f"{'='*60}")
    a_out = lp_iteration(best_seq, deadline1, verbose=True)
    try_update(a_out, "lp_parent")

    # ============ Phase 2: LP with momentum (different dynamics) ============
    remaining = time_budget - (time.monotonic() - t_start)
    deadline2 = t_start + min(700.0 + 350.0, time_budget * 0.60)
    if time.monotonic() < deadline2:
        print(f"\n{'='*60}")
        print(f"Phase 2: LP momentum (n={len(best_seq)}, budget={deadline2-time.monotonic():.0f}s)")
        print(f"{'='*60}")
        a_out = lp_iteration_momentum(best_seq, deadline2, alpha=0.25, verbose=True)
        try_update(a_out, "lp_momentum")

    # ============ Phase 3: LP from perturbed start ============
    remaining = time_budget - (time.monotonic() - t_start)
    deadline3 = t_start + min(700.0 + 350.0 + 300.0, time_budget * 0.78)
    if time.monotonic() < deadline3:
        print(f"\n{'='*60}")
        print(f"Phase 3: LP from perturbed init (n={len(best_seq)}, budget={deadline3-time.monotonic():.0f}s)")
        print(f"{'='*60}")
        rng = np.random.default_rng(2024)
        n_cur = len(best_seq)
        # Create perturbation
        noise_scale = 0.15
        perturbed = best_seq + rng.normal(0, noise_scale, n_cur)
        perturbed = np.maximum(perturbed, 0.0)
        ss = perturbed.sum()
        if ss > 1e-10:
            perturbed = perturbed / ss
            a_out = lp_iteration(perturbed, deadline3, verbose=True)
            try_update(a_out, "lp_perturbed")

    # ============ Phase 4: LSE polishing ============
    remaining = time_budget - (time.monotonic() - t_start)
    deadline4 = min(t_start + time_budget * 0.88, time.monotonic() + min(200.0, remaining * 0.5))
    if time.monotonic() < deadline4:
        print(f"\n{'='*60}")
        print(f"Phase 4: LSE polish (n={len(best_seq)}, budget={deadline4-time.monotonic():.0f}s)")
        print(f"{'='*60}")
        a_out = lse_polish(best_seq, deadline4, T_start=3.0, T_end=0.2, verbose=True)
        try_update(a_out, "lse_polish")

    # ============ Phase 5: Local search ============
    remaining = time_budget - (time.monotonic() - t_start)
    deadline5 = min(t_start + time_budget * 0.95, time.monotonic() + min(150.0, remaining * 0.5))
    if time.monotonic() < deadline5:
        print(f"\n{'='*60}")
        print(f"Phase 5: Local search (n={len(best_seq)}, budget={deadline5-time.monotonic():.0f}s)")
        print(f"{'='*60}")
        a_out = local_search(best_seq, deadline5, seed=12345)
        try_update(a_out, "local_search")

    # ============ Phase 6: Try slightly different length ============
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 80:
        n_orig = len(best_seq)
        for n_try in [n_orig + 200, n_orig + 400]:
            if time.monotonic() > t_start + time_budget * 0.97:
                break
            deadline6 = min(t_start + time_budget * 0.97, time.monotonic() + min(remaining * 0.4, 120))
            if time.monotonic() >= deadline6:
                break
            print(f"\n{'='*60}")
            print(f"Phase 6: Try n={n_try} (budget={deadline6-time.monotonic():.0f}s)")
            print(f"{'='*60}")
            a_interp = interpolate_seq(best_seq, n_try)
            a_out = lp_iteration(a_interp, deadline6, verbose=False)
            try_update(a_out, f"lp_n{n_try}")

    # ============ Final: one more LSE + local search polish ============
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 30:
        deadline7 = min(t_start + time_budget * 0.99, time.monotonic() + min(80.0, remaining * 0.6))
        if time.monotonic() < deadline7:
            a_out = lse_polish(best_seq, deadline7, T_start=2.0, T_end=0.1, verbose=False)
            try_update(a_out, "final_lse")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 15:
            deadline8 = min(t_start + time_budget * 0.99, time.monotonic() + min(60.0, remaining * 0.6))
            if time.monotonic() < deadline8:
                a_out = local_search(best_seq, deadline8, seed=77777)
                try_update(a_out, "final_local")

    # Final evaluation
    final_eval = evaluate_safe(best_seq)
    if np.isfinite(final_eval):
        best_eval = final_eval
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_eval:.8f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
