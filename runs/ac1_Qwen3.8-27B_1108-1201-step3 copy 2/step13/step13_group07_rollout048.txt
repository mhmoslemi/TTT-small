def fast_autocorr(a):
    """Full convolution a*a using FFT."""

    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) / n * target_sum
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]

        score = internal_score(a_new)

            best_a = a_new.copy()
        a = a_new

    return best_a

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    m = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix: T[k, j] = a_fixed[k-j] if 0 <= k-j < n
    # For efficiency, use the Toeplitz-like structure
    T = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = j + n
        T[k_start:k_end, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, max_iter=100)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=200)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Direct subgradient optimization with softmax-smoothed max.
    Uses FFT-based gradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    fft_size = 1 << (3 * n - 2).bit_length()

    # Use adaptive temperature for softmax smoothing
    base_T = 200.0
    v = np.zeros(n)  # momentum accumulator
    beta_mom = 0.85

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        T_smooth = max(1.0, base_T / (1.0 + t * 0.001))

        # Softmax weights
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient via FFT: grad_j = 2 * sum_k w_k * a_{k-j}
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Adaptive learning rate
        lr_t = 0.05 / (1.0 + 0.005 * t)

        # Momentum
        v = beta_mom * v + (1 - beta_mom) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with randomized restarts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: minimize impact on other peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            if not candidates:
                continue

            # Pick from low-value candidates
            cand_arr = np.array([a[j] for j in candidates])
            sorted_idx = np.argsort(cand_arr)
            dst_pool = [candidates[i] for i in sorted_idx[:max(1, len(candidates) // 4)]]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 1000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.6f}")
        results[f'lp_n{n_base}'] = (a_out, sc_out)

    elapsed = time.monotonic() - t_global_start
    print(f"  After P2: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === PHASE 3: Subgradient optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_out, sc_out = subgradient_optimize(best_seq, t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad]: {best_score:.6f}")

    elapsed = time.monotonic() - t_global_start
    print(f"  After P3: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === PHASE 4: Try longer sequence (n+400) ===
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral at longer length
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.25)
        sc = internal_score(a_spec)
        init_for_lp = a_spec if sc < internal_score(init_long) else init_long
        print(f"  Init for long LP: score={internal_score(init_for_lp):.6f}")

        # LP at longer length
        t4_lp = t4 * 0.85
        a_out, sc_out = nesterov_lp_optimize_v2(init_for_lp, t4_lp, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_long}]: {best_score:.6f}")
        results[f'lp_n{n_long}'] = (a_out, sc_out)

    elapsed = time.monotonic() - t_global_start
    print(f"  After P4: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === PHASE 5: Second LP round at best length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP round 2 (n={n_best}, remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 20:
        a_out, sc_out = nesterov_lp_optimize_v2(best_seq, t5, solver, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_r2_n{n_best}]: {best_score:.6f}")

    elapsed = time.monotonic() - t_global_start
    print(f"  After P5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === PHASE 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 80.0)
    if t6 > 5:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak1]: {best_score:.6f}")

    # === PHASE 7: Final LP sweep if time remains ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP (remaining={remaining:.0f}s) ===")
        t7 = min(remaining * 0.60, 150.0)
        a_out, sc_out = nesterov_lp_optimize_v2(best_seq, t7, solver, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_final]: {best_score:.6f}")

        # Final peak shave
        remaining2 = time_budget - (time.monotonic() - t_global_start)
        if remaining2 > 10:
            t8 = min(remaining2 * 0.5, 40.0)
            a_out, sc_out = peak_shaving(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [peak_final]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
