The objective is `min max_k sum_j a[j] * a[k-j]` subject to `a >= 0, sum(a) = 1`.

This is a "minimum peak probability" problem for the sum of two i.i.d. random variables.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (FFT-based iterative method)
2. LP best-response with Nesterov acceleration
3. Systematic coordinate descent with bisection line search

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [x >= 0, cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                return s
        except:
            continue
    _solver = cp.ECOS
    return _solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a, target_sum=None):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) / n
    if target_sum is None:
        target_sum = 1.0
    return a * target_sum / s

def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return float('inf')
    conv = fast_autocorr(a)
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def spectral_flatten(a0, n_iters=2000, time_limit=60.0, verbose=True):
    """
    Iterative FFT-based method to flatten autocorrelation.
    Alternates between:
    - Time domain: enforce a >= 0, normalize
    - Frequency domain: modify |A| to reduce peak
    """
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive threshold
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        P = A.real**2 + A.imag**2  # |A|^2
        c = irfft(P, fft_size)[:2*n-1]  # autocorrelation
        
        c_max = c.max()
        c_mean = c.mean()
        
        # Threshold: reduce |A| where it contributes to peaks
        # Use a soft threshold on the power spectrum
        ratio = c_max / max(c_mean, 1e-15)
        if ratio < 1.05:
            break
        
        # Modify power spectrum: reduce peaks
        # Idea: P_new = P / (1 + alpha * max(0, P/P_mean - 1))
        P_mean = P[P > 0].mean() if (P > 0).any() else 1.0
        alpha = 0.3
        P_new = P / (1.0 + alpha * np.maximum(0, P / P_mean - 1.0))
        
        # Reconstruct A with new magnitudes, keep phases
        A_new = np.sqrt(P_new) * (A / np.maximum(np.sqrt(P), 1e-15))
        
        # Back to time domain
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): peak={best_score:.6f}")
    
    return best_a, best_score

def build_conv_matrix(a):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver, active_rows=None):
    """Solve: max sum(b) s.t. T_a @ b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    
    if active_rows is not None and len(active_rows) < size:
        T = build_conv_matrix(a)
        constraints = [T[active_rows, :] @ b_var <= M]
    else:
        T = build_conv_matrix(a)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b.sum() < 1e-15:
        return None
    return b

    t0 = time.monotonic()

    y_prev = a.copy()

    return best_a, best_peak

def coord_descent(a0, time_limit, verbose=True):
    """
    Systematic coordinate descent: for each coordinate i, find the best
    value of a[i] (with others fixed) that minimizes the peak of conv(a,a).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    size = 2 * n - 1
    
    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute the "influence" of each coordinate
    # Changing a[i] by delta changes conv[k] by delta * (a[k-i] + a[i-k] - 2*a[i] if k==2i... )
    # Actually: conv(a,a)[k] = sum_j a[j]*a[k-j]
    # d/d(a[i]) conv[k] = a[k-i] + a[i-k] (if both indices valid)
    # But we also have the constraint sum(a) = norm, so changing a[i] requires changing something else.
    # Simplest: change a[i] and renormalize.
    
    pass_count = 0
    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-10:
                # Try to increase from 0
                for delta in [norm * 0.001, norm * 0.01, norm * 0.05]:
                    a[i] = delta
                    a = normalize(a, norm)
                    peak = fast_autocorr(a).max()
                    if peak < best_peak - 1e-10:
                        best_peak = peak
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] = 0
                        a = normalize(a, norm)
                continue
            
            # Try decreasing a[i]
            fracs = [0.0, 0.1, 0.3, 0.5, 0.7, 0.9]
            best_local = a[i]
            best_local_peak = best_peak
            
            for f in fracs:
                new_ai = ai * (1.0 - f)
                a[i] = new_ai
                a = normalize(a, norm)
                peak = fast_autocorr(a).max()
                if peak < best_local_peak:
                    best_local_peak = peak
                    best_local = new_ai
            
            if best_local != ai:
                a[i] = best_local
                a = normalize(a, norm)
                if best_local_peak < best_peak - 1e-10:
                    best_peak = best_local_peak
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = ai
                a = normalize(a, norm)
        
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 2:
                break
        else:
            no_improve_count = 0
        
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_count} ({elapsed:.0f}s): peak={best_peak:.6f}")
    
    return best_a, best_peak

def pair_cd(a0, time_limit, rng, verbose=True):

    """

    pass_count = 0
    no_improve = 0

        pass_count += 1
        improved = False
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-10:
                continue
            
            # Try transferring to a few random j's
            for _ in range(4):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Binary search for optimal fraction to transfer
                lo, hi = 0.0, ai
                best_frac = 0.0
                best_val = best_peak
                
                # First check if any transfer helps
                mid = hi * 0.5
                a[i] = ai - mid
                a[j] = a[j] + mid
                a = normalize(a, norm)
                test_peak = fast_autocorr(a).max()
                if test_peak >= best_peak:
                    # Transfer hurts, try small amounts
                    a[i] = ai
                    a[j] = a[j] - mid + a[j]  # revert
                    a[j] = a[j]  # need to properly revert
                    # Let me redo this properly
                    a[i] = ai
                    a = normalize(a, norm)
                    continue
                
                # Binary search
                for _ in range(15):
                    mid = (lo + hi) / 2
                    a[i] = ai - mid
                    a[j] = a[j] + mid
                    a = normalize(a, norm)
                    test_peak = fast_autocorr(a).max()
                    if test_peak < best_val:
                        best_val = test_peak
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_frac > 0:
                    a[i] = ai - best_frac
                    a[j] = a[j] + best_frac
                    a = normalize(a, norm)
                    if best_val < best_peak - 1e-12:

                        best_a = a.copy()
                        improved = True
                    else:
                        # Revert
                        a[j] = a[j] - best_frac
                        a[i] = ai
                        a = normalize(a, norm)
                else:
                    a[i] = ai
                    a = normalize(a, norm)
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and pass_count % 10 == 0:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (FFT-based iterative peak reduction)
2. LP best-response with Nesterov acceleration + constraint screening + warm starts
3. Systematic pair coordinate descent with binary search line search
4. Multi-length exploration with interpolation transfer
5. Multiple initialization strategies (uniform, triangular, ramp, spectral)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_lp_solver = None


def _get_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [x >= 0, cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _normalize(a, target=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    if target is None:
        target = 1.0
    return a * target / s


def _eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    sa = a.sum()
    if sa < 0.01:
        return float('inf')
    conv = _fast_autocorr(a)
    mb = conv.max()
    return 2.0 * n * mb / (sa * sa)


def _build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_solve(b_var, T_sub, M, solver):
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)


def _lp_best_response(a, M, solver, active_rows=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a)
    if active_rows is not None and 0 < len(active_rows) < size:
        b = _lp_solve(b_var, T[active_rows, :], M, solver)
        if b is not None and b.sum() > 1e-15:
            # Verify no serious violations of dropped constraints
            full_conv = T @ b
            max_viol = (full_conv - M).max()
            if max_viol < M * 0.01:
                return b
    b = _lp_solve(b_var, T, M, solver)
    return b


def _spectral_flatten(a0, time_limit, verbose=True):
    n = len(a0)
    a = _normalize(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_peak = _fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = rfft(a, fft_size)
        P = A.real ** 2 + A.imag ** 2
        c = irfft(P, fft_size)[:2 * n - 1]
        c_max = c.max()
        c_mean = c[c > 1e-15].mean() if (c > 1e-15).any() else 1.0
        ratio = c_max / max(c_mean, 1e-15)
        if ratio < 1.02:
            break
        P_pos = np.maximum(P, 1e-30)
        P_mean = P_pos.mean()
        # Soft compression of power spectrum peaks
        alpha = 0.5
        compression = 1.0 + alpha * np.maximum(0.0, P_pos / P_mean - 1.0) ** 0.5
        P_new = P_pos / compression
        mag = np.sqrt(P_new)
        phase = A / np.maximum(np.sqrt(P_pos), 1e-15)
        A_new = mag * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        peak = _fast_autocorr(a_new).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 200 == 0:
            print(f"    Spectral it={it} ({time.monotonic()-t0:.0f}s): peak={best_peak:.6f}")
    return best_a, best_peak

def _pair_cd(a0, time_limit, rng, verbose=True):

        improved = False
        perm = rng.permutation(n)
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue
            # Try 3 random targets
            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                aj = a[j]
                # Try transferring fraction f of ai to j
                # Binary search for best f in [0, 1]
                lo, hi = 0.0, min(ai, norm * 0.5)
                best_f = 0.0
                best_val = best_peak
                # Quick test at midpoint
                mid_test = lo + (hi - lo) * 0.5
                a[i] = ai - mid_test
                a[j] = aj + mid_test
                a = _normalize(a, norm)
                test_peak = _fast_autocorr(a).max()
                a[i] = ai
                a[j] = aj
                a = _normalize(a, norm)
                if test_peak >= best_peak:
                    continue
                # Binary search
                for _bs in range(12):
                    mid = (lo + hi) / 2.0
                    if mid <= 0 or mid >= ai:
                        break
                    a[i] = ai - mid
                    a[j] = a[j] + mid
                    a = _normalize(a, norm)
                    test_peak = _fast_autocorr(a).max()
                    if test_peak < best_val:
                        best_val = test_peak
                        best_f = mid
                        lo = mid
                    else:
                        hi = mid
                if best_f > 0 and best_val < best_peak - 1e-12:
                    a[i] = ai - best_f
                    a[j] = a[j] + best_f
                    a = _normalize(a, norm)
                    actual_peak = _fast_autocorr(a).max()
                    if actual_peak < best_peak:

        if not improved:

def _interp(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return _normalize(a_new)

        print(f"\n=== P3: Pair CD n={n_cur} (t={t3:.0f}s) ===")
        a_out, _ = _pair_cd(best_a, t3, rng, True)
        _try_update(a_out, f"pcd_n{n_cur}")

    # Phase 4: Try longer length with LP
    n_long = min(n0 + 800, 3500)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 300.0)
    if t4 > 30:
        print(f"\n=== P4: LP Nesterov n={n_long} (t={t4:.0f}s) ===")
        a_interp = _interp(best_a, n_long)
        a_out, _ = _nesterov_lp(a_interp, t4, solver, True)
        _try_update(a_out, f"lp_n{n_long}")

    # Phase 5: Pair CD on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        print(f"\n=== P5: Pair CD n={n_cur} (t={t5:.0f}s) ===")
        a_out, _ = _pair_cd(best_a, t5, rng, True)
        _try_update(a_out, f"pcd2_n{n_cur}")

    # Phase 6: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        print(f"\n=== P6: Spectral n={n_cur} (t={t6:.0f}s) ===")
        a_out, _ = _spectral_flatten(best_a, t6, True)
        _try_update(a_out, f"spectral2_n{n_cur}")

    # Phase 7: Single CD for fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 10:
        print(f"\n=== P7: Single CD n={n_cur} (t={t7:.0f}s) ===")
        a_out, _ = _single_cd(best_a, t7, True)
        _try_update(a_out, f"scd_n{n_cur}")

    # Phase 8: Final LP with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 20:
        print(f"\n=== P8: LP Final n={n_cur} (t={t8:.0f}s) ===")
        a_out, _ = _nesterov_lp(best_a, t8, solver, True)
        _try_update(a_out, f"lp_final_n{n_cur}")

    # Final pair CD with any remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    t9 = min(remaining * 0.50, 80.0)
    if t9 > 10:
        print(f"\n=== P9: PCD Final n={n_cur} (t={t9:.0f}s) ===")
        a_out, _ = _pair_cd(best_a, t9, rng, True)
        _try_update(a_out, f"pcd_final_n{n_cur}")

    # Final
    if best_a is not None:
        final_sc = _eval_score(best_a)
        best_score = min(best_score, final_sc)
        best_a = best_a / best_a.sum()
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_a]
