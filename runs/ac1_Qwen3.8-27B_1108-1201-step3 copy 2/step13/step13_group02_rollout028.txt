    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s if s > 1e-15 else np.ones(old_n) / old_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    return b_val


def lp_best_response_reduced(a, M, solver):
    """LP with constraint reduction to active constraints only."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(3):
        circ = T @ b_val
        tol = 1e-8 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 5 or n_active < 20:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-6):
            b_val = b_new
        else:
            break

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gd(a0, time_limit, beta_start, beta_end, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_mult = 1.0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        log_beta = np.log(beta_start) + progress * (np.log(beta_end) - np.log(beta_start))
        beta = np.exp(log_beta)

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        shifted = beta * (b - max_b)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        grad = compute_lse_gradient(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        a_max = a.max()
        base_step = 1.0 / (2.0 * beta * max(a_max, 1e-10) + 1e-15)
        step = base_step * step_mult

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            step_mult = max(0.5 * step_mult, 0.01)
        else:
            a_new = a_new / s_new
            obj_new = 2.0 * n * fast_autocorr(a_new).max()
            if obj_new > best_obj * 1.01:
                # Step was too large, reduce
                step_mult = max(0.5 * step_mult, 0.01)
            else:
                # Step was OK, maybe increase slightly
                step_mult = min(1.1 * step_mult, 5.0)
                a = a_new
                steps += 1
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD[{label}] step={steps} ({elapsed:.0f}s): beta={beta:.1f}, obj={best_obj:.8f}")

    return best_a, best_obj
