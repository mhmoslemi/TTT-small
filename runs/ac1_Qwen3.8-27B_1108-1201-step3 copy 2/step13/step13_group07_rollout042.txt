    """Compute LSE weights: w[k] = exp(T*conv[k]) / sum exp(T*conv)."""
    max_val = conv_vals.max()

    s = exp_vals.sum()
    if s < 1e-300:

    return exp_vals / s

    """Compute (1/T) * log(sum(exp(T*conv_vals)))."""
    max_val = conv_vals.max()
    exp_vals = np.exp(T * (conv_vals - max_val))
    return (max_val + np.log(exp_vals.sum())) / T

The KKT conditions at the optimum are:
- For j in support(a*): grad_j(w, a*) = lambda
- For j not in support(a*): grad_j(w, a*) >= lambda
- sum(a*) = 1, a* >= 0

3. For j in support: set a[j] so that grad_j = lambda
4. Project to simplex
5. Repeat

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a, b=None):
    """Compute full convolution a*b via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def internal_score(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_conv(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-12:
        v[v < 0] = 0
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 1e-15:
            result = result / s * target_sum
    return result


def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64) - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)
    for _ in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = np.clip(target / (B[1:] + 1e-30), 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a


def build_T_indices(n):
    """Precompute row/col indices for convolution matrix T (2n-1 x n).
    T[k,j] = 1 (placeholder, actual value = a[k-j])."""
    n_rows = 2 * n - 1
    row_parts = []
    col_parts = []
    offset_parts = []
    for j in range(n):
        k_start = j
        k_end = j + n
        k_range = np.arange(k_start, k_end)
        row_parts.append(k_range)
        col_parts.append(np.full(n, j, dtype=np.int32))
        offset_parts.append(k_range - j)
    all_rows = np.concatenate(row_parts)
    all_cols = np.concatenate(col_parts)
    all_offsets = np.concatenate(offset_parts)
    return all_rows, all_cols, all_offsets


def solve_lp_best_response(a_fixed, n, row_idx, col_idx, offsets):
    """Solve: min M s.t. (a_fixed*b)_k <= M, sum(b)=1, b>=0.
    Returns (b_val, M_val) or (None, None) on failure."""
    # Variables: x = [M, b_0, ..., b_{n-1}], length n+1
    # A_ub: (2n-1) x (n+1), A_ub[k,0]=-1, A_ub[k,j+1]=a_fixed[k-j]
    a_vals = a_fixed[offsets]
    A_ub_data = np.concatenate([np.full(len(a_vals), -1.0), a_vals])
    A_ub_rows = np.concatenate([row_idx, row_idx])
    A_ub_cols = np.concatenate([np.zeros(len(row_idx), dtype=np.int32), col_idx + 1])
    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_eq = csr_matrix((np.ones(n), (np.zeros(1, dtype=np.int32), np.arange(1, n + 1))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] + [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return None, None
    if res.status != 0 or res.x is None:
        return None, None
    M_val = float(res.x[0])
    b_val = np.asarray(res.x[1:], dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def nesterov_lp(a0, time_limit, row_idx, col_idx, offsets, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0
    stall = 0
    avg_a = np.zeros(n)
    avg_count = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        b_star, M_val = solve_lp_best_response(y, n, row_idx, col_idx, offsets)
        if b_star is None:
            continue
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        if t <= 2:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = project_simplex(a_new)
        y_prev = y.copy()
        y = y_new
        a = a_new
        avg_a += a
        avg_count += 1
        sc_a = internal_score(a)
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        sc_y = internal_score(y)
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        if avg_count % 50 == 0:
            avg_candidate = avg_a / avg_count
            sc_avg = internal_score(avg_candidate)
            if sc_avg < best_score - 1e-12:
                best_score = sc_avg
                best_a = avg_candidate.copy()
                stall = 0
        if stall > 3000:
            y = a.copy()
            y_prev = a.copy()
            avg_a = np.zeros(n)
            avg_count = 0
            stall = 0
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def compute_lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Returns (grad, lse_value)."""
    n = len(a)
    conv_aa = fast_conv(a)
    max_c = conv_aa.max()
    exp_Tc = np.exp(T * (conv_aa - max_c))
    w = exp_Tc / exp_Tc.sum()
    lse_val = (max_c + np.log(exp_Tc.sum())) / T
    # grad[j] = (2/T) * sum_k w[k] * a[k-j]
    # Using: conv(a_rev, w)[n-1+j] = sum_m a[m]*w[m+j]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A_rev * W, fft_size)
    grad = (2.0 / T) * conv_full[n - 1:n - 1 + n]
    return grad, lse_val


def lse_pg_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_steps=5000, verbose=True):
    """Projected gradient on LSE-smoothed max conv."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    for i in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        progress = (i + 1) / n_steps
        T_curr = T_start * (T_end / T_start) ** progress
        eta = 1.0 / (T_curr * n + 1.0)
        grad, _ = compute_lse_grad(a, T_curr)
        a_new = project_simplex(a - eta * grad)
        sc = internal_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if verbose and i % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-PG i={i} T={T_curr:.2f} ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def adam_lse_optimize(a0, time_limit, T_start=30.0, T_end=0.05, n_stages=20, lr=0.004):
    """Adam optimizer on log-parameterized LSE objective."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            conv_aa = fast_conv(a)
            max_c = conv_aa.max()
            exp_Tc = np.exp(stage_T * (conv_aa - max_c))
            w = exp_Tc / exp_Tc.sum()
            obj_val = (max_c + np.log(exp_Tc.sum())) / stage_T
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv_full = irfft(A_rev * W, fft_size)
            grad_a = (2.0 / stage_T) * conv_full[n - 1:n - 1 + n]
            S_val = np.dot(grad_a, a)
            grad_z = a * (grad_a - S_val)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            if t_adam % 20 == 0:
                sc = internal_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
        a_final = softmax_fn(z)
        sc = internal_score(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage={si}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving: move mass from peak contributors to safe spots."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() - t0 < time_limit:
        conv_aa = fast_conv(a)
        top_k = np.argsort(conv_aa)[::-1][:15]
        peak_areas = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            peak_areas.update(range(js, je + 1))
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            cand_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = cand_sorted[:max(1, len(cand_sorted) // 4)]
            for _ in range(3):
                dst = dst_pool[rng.integers(len(dst_pool))]
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = internal_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.002, 0.02)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = internal_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 800:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Phases: spectral init -> Nesterov LP -> LSE PG -> Adam LSE -> peak shaving.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1730.0)

    # --- Initialization ---
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Precompute sparse indices for LP
    row_idx, col_idx, offsets = build_T_indices(n_base)
    print(f"Precomputed T indices for n={n_base}")

    # --- Phase 1: Spectral shaping (multiple alphas) ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 5:
        for alpha_val in [0.5, 0.3, 0.15]:
            a_out = spectral_shaping(best_seq, n_iter=400, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After P1: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 2: Nesterov LP (primary workhorse) ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Nesterov LP (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.50, 800.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp(best_seq, t2, row_idx, col_idx, offsets)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [NLP]: {best_score:.6f}")
    print(f"  After P2: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 3: LSE-projected gradient polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-PG polish (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_out, sc_out = lse_pg_optimize(best_seq, t3, T_start=80.0, T_end=0.5, n_steps=8000)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LSE-PG]: {best_score:.6f}")
    print(f"  After P3: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 4: Adam LSE ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam LSE (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t4, T_start=40.0, T_end=0.05, n_stages=20, lr=0.003)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [Adam]: {best_score:.6f}")
    print(f"  After P4: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 5: Peak shaving ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 5:
        a_out, _ = peak_shaving(best_seq, t5, seed=456)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [peak1]: {best_score:.6f}")
    print(f"  After P5: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 6: Second Nesterov LP round ---
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Nesterov LP round 2 (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 20:
        a_out, _ = nesterov_lp(best_seq, t6, row_idx, col_idx, offsets, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [NLP2]: {best_score:.6f}")
    print(f"  After P6: best={best_score:.6f} (t={time.monotonic()-t_global:.0f}s)")

    # --- Phase 7: Final peak shave ---
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t7 = min(remaining * 0.40, 80.0)
        if t7 > 5:
            a_out, _ = peak_shaving(best_seq, t7, seed=789)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [peak2]: {best_score:.6f}")

    # --- Final verification ---
    best_seq = np.maximum(best_seq, 0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    final_score = internal_score(best_seq)
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
