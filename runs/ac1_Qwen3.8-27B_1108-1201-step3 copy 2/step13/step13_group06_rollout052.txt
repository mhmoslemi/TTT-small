"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov zero-sum game with constraint generation (active set)
2. Momentum-tuned Nesterov acceleration with adaptive restart
3. Multi-length exploration (n=1600, 1800)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(a, b):
    """Linear cross-correlation: result[k] = sum_j a[k-j]*b[j] for valid j."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(np.conj(A) * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_toeplitz_rows(a_fixed, rows):
    """Build rows of Toeplitz matrix: T[i,j] = a_fixed[rows[i] - j] if valid."""
    n = len(a_fixed)
    T = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            T[i, idx] = a_fixed[k - idx]
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_active(y, active_rows, solver, warm_b=None):
    """
    Solve LP: min M s.t. T_active @ b <= M, sum(b)=1, b>=0.
    Returns (b, M_val, all_converted_vals) where all_converted_vals = cross_corr(y,b).
    """
    n = len(y)
    T = build_toeplitz_rows(y, active_rows)

    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)

    # Compute full cross-correlation for constraint generation
    all_vals = fast_cross(y, b_val)

    return b_val, M_val, all_vals

def local_refine(a0, time_limit, seed=123):
    """
    Local coordinate refinement: try paired moves to reduce max autocorrelation.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        top_k = min(8, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (far from danger zones)
            danger = np.zeros(n, dtype=bool)
            for pk in peak_indices[:6]:
                djs = max(0, pk - n + 1)
                dje = min(n - 1, pk)
                danger[djs:dje + 1] = True
            danger[src] = True

            safe = np.where(~danger)[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]

            found = False
            for dst in dst_pool[:6]:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 5e-7):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0 and stall <= 300:
                # Small random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 400:
                break

    return best_a, best_score

            # Actually: (a*a)[k] = sum_{j: valid} a[j]*a[k-j]
            # d/d(a[i]) = sum over terms where i appears
            # Term a[j]*a[k-j] contains a[i] if j=i (gives a[k-i]) or k-j=i (gives a[j]=a[k-i])
            # If i != k-i (i.e., 2i != k), both give the same: a[k-i], counted once
            # If i == k-i (2i == k), it's a[i]^2, derivative is 2*a[i]
        # Let me redo:
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = a[j]
        # Also need to account for the constraint sum(a)=1
        # Simpler: just do a multiplicative update
        gmax = grad.max()
        if gmax > 1e-18:
            grad_norm = grad / gmax
            eta = 0.005
            update = 1.0 - eta * grad_norm
            update = np.maximum(update, 0.01)
            a_new = a * update
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try random frequency-domain perturbation
            A = rfft(a, fft_size)
            noise = rng.normal(0, 0.01 * np.abs(A).mean(), len(A))
            A_new = A.copy()
            A_new[1:] += noise[1:]  # Don't touch DC component
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new / sn
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score * 1.001:  # Allow slight increase for exploration
                    a = a_new
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        improved = True

            if not improved and (int((time.monotonic() - t0) * 10) % 50 == 0):
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Use smooth interpolation
    old_x = np.linspace(0, 1, old_n + 1)
    f_padded = np.concatenate([f_norm, [0]])
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_padded)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x)
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-10 * x ** 2) + np.exp(-10 * (x - 1) ** 2) + 0.1
    elif choice == 5:
        a = 1.0 / (1 + 5 * np.sin(np.pi * x) ** 2)
    elif choice == 6:
        # W-shape: high at edges and middle
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.8 * (np.exp(-30 * x ** 2) + np.exp(-30 * (x - 1) ** 2))
    else:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.7 * np.exp(-3 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining fast LP-CG game, spectral refinement,
    local coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 80.0, 1720.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Get parent or generate initial
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP-CG at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-CG n={n_base} ===")
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_cg(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Local refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Local refine n={len(best_seq)} ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 20:
        a_out, _ = local_refine(best_seq, t2, seed=777)
        try_update(a_out, f"local_n{len(best_seq)}")

    # === Phase 3: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Spectral refine n={len(best_seq)} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_out, _ = spectral_refine(best_seq, t3, seed=456)
        try_update(a_out, f"spectral_n{len(best_seq)}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    print(f"=== Phase 4: LP-CG n={n_long} ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_cg(init_long, t4, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 5: Local refine at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t5 = min(remaining * 0.10, 100.0)
        if t5 > 15:
            print(f"=== Phase 5: Local refine n={len(best_seq)} ===")
            a_out, _ = local_refine(best_seq, t5, seed=888)
            try_update(a_out, f"local2_n{len(best_seq)}")

    # === Phase 6: Multi-start with diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Multi-start ===")
    t6_total = min(remaining * 0.12, 150.0)
    if t6_total > 40:
        for si in range(3):
            if time.monotonic() - t_global > t_global + time_budget - 60:
                break
            t6_each = t6_total / 3
            init = structured_init(n_base, rng)
            # Blend with current best
            blend = 0.5 * init + 0.5 * (best_seq / best_seq.sum())
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            a_out, _ = nesterov_lp_cg(blend, t6_each, solver)
            try_update(a_out, f"multi_{si}")

    # === Phase 7: Final LP-CG polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7 = min(remaining * 0.25, 200.0)
        print(f"=== Phase 7: Final LP-CG polish n={len(best_seq)} ===")
        a_out, _ = nesterov_lp_cg(best_seq, t7, solver)
        try_update(a_out, f"final_lp_n{len(best_seq)}")

    # === Phase 8: Final local refine ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 10:
            print(f"=== Phase 8: Final local refine ===")
            a_out, _ = local_refine(best_seq, t8, seed=999)
            try_update(a_out, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: uniform
        best_seq = np.ones(n_base) / n_base

    return [float(x) for x in best_seq]
