def build_conv_matrix(a):
    """Build sparse convolution matrix T where (T*b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_end = min(n - 1, k)

            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    T = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response_fast(a, M, solver, T_cache=None):

def compute_active_subgrad(a, ac, margin=0.0):
    """Compute subgradient using only the active set (lags at max)."""
    n = len(a)
    M = ac.max()
    # Include lags within margin of the max
    active = np.where(ac >= M - margin)[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])
    w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, T_cache=None):
    """Nesterov-accelerated LP best-response with cache."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    T_cached = T_cache
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b, T_cached = lp_best_response_fast(y, M_curr, solver, T_cached)
        if b is None:
            # Fallback: use subgradient
            ac = fast_autocorr(y)
            g = compute_active_subgrad(y, ac, margin=1e-4)
            step = 1.0 / (np.linalg.norm(g, ord=np.inf) + 1e-10)
            b = y - step * g
            b = np.maximum(b, 0.0)
            b = b * c / b.sum()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m, T_cached

def mirror_descent_optimize(a0, time_limit, top_k=3, beta=2.0, verbose=True):
    """Mirror descent with active-set and soft subgradient."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 1e-15)
    a = a * c / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    n_avg = 1
    gamma = 1.0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        # Alternate between active-set and soft subgradient
        if t % 3 == 0:
            g = compute_active_subgrad(a, ac, margin=ac.max() * 0.01)
        else:
            g = compute_soft_subgrad(a, ac, top_k=top_k, beta=beta)
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def coordinate_polish(a0, time_limit, verbose=True):
    """Randomized coordinate descent: move mass between pairs to reduce peak."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        # Focus on top peaks
        top_idx = np.argsort(ac)[::-1][:10]
        improved = False
        
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destination: minimize contribution to all top peaks
            peak_influence = np.zeros(n)
            for k2 in top_idx:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                peak_influence[idx2] += a[k2 - idx2]
            
            # Pick destination with low influence
            candidates = np.where(peak_influence < np.median(peak_influence))[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Try several random candidates
            n_try = min(5, len(candidates))
            candidates_sample = candidates[rng.choice(len(candidates), n_try, replace=False)]
            
            for dst in candidates_sample:
                if dst == src:
                    continue
                a_src = a[src]
                if a_src < c * 1e-10:
                    continue
            
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a_src * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    a = a * c / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c / a.sum()
        else:
            stall = 0
        
        if verbose and (stall % 50 == 0 or improved):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def diverse_initializations(n, parent, rng, count=5):
    """Generate diverse initializations."""
    inits = []
    c = np.sqrt(2.0 * n)
    
    # 1. Interpolated parent
    if parent is not None:
        p = interpolate_sequence(parent, n)
        inits.append(p * c)
    
    # 2. Sine bump
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    inits.append(f * c / f.sum())
    
    # 3. Triangle
    f = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    inits.append(f * c / f.sum())
    
    # 4. Gaussian bump
    f = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    inits.append(f * c / f.sum())
    
    # 5. Random smooth
    coarse = rng.random(max(10, n // 100))
    f = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
    f = np.maximum(f, 0.01)
    inits.append(f * c / f.sum())
    
    return inits

def try_update(best_state, a, label):
    """Try to update the best state."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_state
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        elapsed = time.monotonic() - best_state['t0']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_state

def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with LP-Nesterov core."""
    t_global = time.monotonic()

    rng = np.random.default_rng(42)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            sc = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={sc:.6f}")
            best_state = try_update(best_state, parent, "parent")
    except Exception:
        pass
    
    if best_state['seq'] is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        init = np.sin(np.pi * x) ** 2
        best_state['seq'] = init / init.sum()
        best_state['score'] = evaluate_safe(best_state['seq'])
        print(f"Default init: n={n_base}, score={best_state['score']:.6f}")
    
    # === Phase 1: LP-Nesterov at parent length ===
    n0 = len(best_state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: LP-Nesterov n={n0} (budget {t1:.0f}s) ===")
    if t1 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_state['seq'] * c0 / best_state['seq'].sum()
        a_out, _, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        best_state = try_update(best_state, a_out, f"lp_n{n0}")

    target_lengths = [800, 1200, 2000, 2400, 3200]

            # Try multiple initializations

            init_budget = per_length / len(inits)
            
            for ii, a_init in enumerate(inits):
                if time.monotonic() - t_global > time_budget - 60:
                    break
                c_nl = np.sqrt(2.0 * nl)
                a_init = a_init * c_nl / a_init.sum()
                a_out, m_out, _ = nesterov_lp_optimize(a_init, init_budget, solver, verbose=False)
                if m_out < best_local_m:
                    best_local_m = m_out
                    best_local = a_out
                best_state = try_update(best_state, a_out, f"explore_n{nl}_init{ii}")
            
            # Also try mirror descent on the best local
            if best_local is not None and remaining > 30:

                c_nl = np.sqrt(2.0 * nl)

        a_out, _, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        best_state = try_update(best_state, a_out, f"lp_final_n{n_best}")
    
    # === Phase 7: Final coordinate polish ===
    n_best = len(best_state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
