def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]

    except Exception:

def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        result = result / result.sum() * target_sum
    return result


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a_fixed[k-j] if valid."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, max_iter=200):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iter)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=300)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 8000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y_old = y
        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a

    # grad_m = 2 * sum_{k: 0<=k-m<n} w_k * a_{k-m}
    # This is the correlation of w with a at lag m
    
    # More efficient: use the fact that it's a convolution
    # sum_k w[k] * a[k-m] = (a_rev * w)[n-1-m] ... let me just use direct FFT

    # sum_k w[k] * a[k-m]: let j=k-m, then k=j+m, sum_j w[j+m]*a[j] = cross_corr(w,a,m)

    # In terms of convolution: cross_corr(w,a,m) = conv(w, a_rev)[(len(w)-1) - m]...

def smoothed_projected_gradient(a0, time_limit, T_start=50.0, T_end=1.0, 
                                  lr_base=0.05, verbose=True):
    """
    Projected gradient on the LogSumExp-smoothed objective.
    f_T(a) = (1/T) * log(sum_k exp(T * (a*a)_k))

        a = a / s

    best_score = internal_score(a)

    t_iter = 0
    
    # Adaptive temperature

        t_iter += 1
        
        # Current temperature (annealing)

        b = fast_autocorr(a)

        a_trial = a - lr_t * v
        a_trial = project_simplex(a_trial)

        if verbose and t_iter % 200 == 0:

def active_set_gradient_descent(a0, time_limit, top_k=5, verbose=True):
    """
    Gradient descent using only the top-k active peaks for the gradient.
    This is a subgradient method for the non-smooth max function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1 + n - 1 - 1).bit_length()
    lr_base = 0.02
    t_iter = 0
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        t_iter += 1
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find top-k peaks with some tolerance
        threshold = b_max - 1e-6 * b_max
        active = np.where(b >= threshold)[0]
        if len(active) > top_k:
            top_indices = np.argsort(b)[::-1][:top_k]
            active = top_indices
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute gradient: average of gradients at active peaks
        # grad_m = (2/|K|) * sum_{k in K} a[k-m] (valid indices only)
        grad = np.zeros(n)
        for k in active:
            # (a*a)_k gradient at position m: 2*a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += 2.0 * a[k - m]
        grad /= len(active)
        
        # Step with backtracking
        lr = lr_base / (1.0 + 0.02 * t_iter)
        improved = False
        for attempt in range(5):
            a_trial = a - lr * grad
            a_trial = project_simplex(a_trial)
            score = internal_score(a_trial)
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                lr_base *= 1.01  # slightly increase on success
                break
            lr *= 0.5
        if not improved:
            lr_base *= 0.9
        
        if verbose and t_iter % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      ASD it={t_iter} (lr={lr_base:.5f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

        # Identify peak "influence zones" (indices that contribute to any significant peak)
        influence = set()
        for k2 in peaks[:20]:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            influence.update(range(js2, je2 + 1))
        
        improved = False
        
        # Strategy 1: Move mass from high-contributing peak positions to low-influence positions
        for ci in range(min(5, len(contrib_sorted))):
            src = idx[contrib_sorted[ci]]
            
            # Find good destinations (outside influence zones, low value)
            candidates = [j for j in range(n) if j not in influence and a[j] < a[src] * 0.5]
            if not candidates:
                candidates = [j for j in range(n) if j not in influence]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Sort by a[j] (prefer moving to low-value positions)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 5)]
            
            for dst in dst_pool[:5]:
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        # Strategy 2: Symmetric redistribution (reduce two symmetric contributors)
        if not improved and len(idx) >= 2:
            # Find the pair (i, j) with i+j=k_peak that has max product
            for ci in range(min(3, len(contrib_sorted))):
                src1 = idx[contrib_sorted[ci]]
                # The symmetric partner
                partner = k_peak - src1
                if 0 <= partner < n and partner != src1:
                    candidates = [j for j in range(n) if j not in influence]
                    if not candidates:
                        candidates = list(range(n))
                    candidates_sorted = sorted(candidates, key=lambda j: a[j])
                    dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 5)]
                    for dst in dst_pool[:3]:
                        for frac in [0.002, 0.005, 0.01]:
                            amt = 0.5 * a[src1] * frac
                            amt2 = 0.5 * a[partner] * frac
                            a_trial = a.copy()
                            a_trial[src1] -= amt
                            a_trial[partner] -= amt
                            a_trial[dst] += amt + amt2
                            if a_trial[src1] < -1e-15 or a_trial[partner] < -1e-15:
                                continue
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = internal_score(a_trial)
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
                        if improved:
                            break
                    if improved:
                        break
            if improved:
                stall = 0
        
        # Strategy 3: Random perturbation when stuck
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
        
        if stall > 300:
            # Try a larger perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = a[i] * 0.1
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall = 0  # reset to avoid premature termination
            if stall > 500:
                break
                
        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave stall={stall} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def dual_guided_optimization(a0, time_limit, solver, verbose=True):
    """
    Use LP dual information to guide a quadratic optimization.
    The equilibrium strategy of the column player defines a weighted sum
    of peaks, giving a tractable optimization subproblem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    # Run a few LP iterations to get the dual (column player equilibrium)
    y = a.copy()
    b_eq = a.copy()  # equilibrium strategy
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        b_star, _ = solve_lp_best_response(y, solver)
        if b_star is None:
            break
        b_eq = 0.9 * b_eq + 0.1 * b_star
        b_eq = b_eq / b_eq.sum()
        y = 0.9 * y + 0.1 * b_eq
        y = y / y.sum()
    
    # Now use b_eq to define a weighted objective: sum_k w_k * (a*a)_k
    # where w_k comes from the "importance" of each peak
    b_current = fast_autocorr(a)
    b_max = b_current.max()
    # Weights: emphasize peaks that are close to the maximum
    T_weight = 50.0
    w = np.exp(T_weight * b_current / max(b_max, 1e-30))
    w = w / w.sum()
    
    # Optimize: min sum_k w_k * (a*a)_k  s.t. a>=0, sum(a)=1
    # Gradient: 2 * sum_k w_k * S_k * a
    # = 2 * cross_corr(w, a) (same as before)
    fft_size = 1 << (2 * n - 1 + n - 1 - 1).bit_length()
    
    lr = 0.05
    v = np.zeros(n)
    beta = 0.8
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        grad, _ = compute_smoothed_gradient(a, b, T_weight)
        
        v = beta * v + (1.0 - beta) * grad
        lr_t = lr / (1.0 + 0.01 * it)
        a_trial = a - lr_t * v
        a_trial = project_simplex(a_trial)
        
        score = internal_score(a_trial)
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial.copy()
        else:
            # Try with accumulated strategy
            y_trial = 0.5 * a + 0.5 * b_eq
            y_trial = project_simplex(y_trial)
            score_y = internal_score(y_trial)
            if score_y < best_score:
                best_score = score_y
                best_a = y_trial.copy()
                a = y_trial.copy()
        
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      DualGuide it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Uses LP for global optimization, then direct gradient methods on the
    actual non-convex objective for local refinement, plus smart peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping (fast, multiple alphas)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.5, 0.3, 0.15, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP (main global optimizer)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} ===")
    t2 = min(remaining * 0.35, 650.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver)
        sc = internal_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 3: Smoothed projected gradient on actual objective
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Smoothed PG n={n_base} ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 10:
        a_out, sc_out = smoothed_projected_gradient(best_seq, t3, T_start=30.0, T_end=0.5, lr_base=0.03)
        sc = internal_score(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)

            if s > 0:

    print(f"  After Phase 5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 6: More LP if time remains (second round from improved start)
    remaining = time_budget - (time.monotonic() - t_global_start)

    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:

        sc = internal_score(a_out)

            s = best_seq.sum()

            print(f"  *** NEW BEST [LP2]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"\n=== Phase 7: Final Peak Shaving ===")
        t7 = min(remaining * 0.40, 80.0)
        if t7 > 5:
            a_out, sc_out = peak_shaving_advanced(best_seq, t7, seed=789)
            sc = internal_score(a_out)
            if np.isfinite(sc) and sc < best_score:

    # Phase 8: Try a slightly different length for potential improvement

    if remaining > 60:
        n_alt = n_base + 100
        print(f"\n=== Phase 8: Alt length n={n_alt} ===")
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_alt = spectral_shaping(init_alt, n_iter=100, alpha=0.2)
            a_out, _ = nesterov_lp_optimize(a_alt, t8 * 0.7, solver, verbose=False)
            sc = internal_score(a_out)
            # Convert back: score for length n_alt
            sc_eval = evaluate_safe(a_out)
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()

                    best_seq = best_seq / s
                print(f"  *** NEW BEST [AltN]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:

        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
