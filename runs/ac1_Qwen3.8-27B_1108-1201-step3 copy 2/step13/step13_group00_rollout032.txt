"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response (primary optimizer)
2. FFT-based LSE Nesterov-Frank-Wolfe (fast gradient method)
3. Grid-search pair coordinate descent for polishing
4. Multi-length exploration with interpolation
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def lse_gradient(a, T):
    """Fast LSE gradient via FFT. grad_i = (2/T) * sum_k w_k * a[k-i]"""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    
    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / T)
    w = exp_c / exp_c.sum()
    
    W = fft(w, N)
    A = fft(a, N)
    
    # sum_k w[k] * a[k-i] = ifft(W * conj(A))[i] for i in [0, n-1]
    cross = np.real(ifft(W * np.conj(A)))[:n]
    
    return 2.0 * cross / T


def lse_nesterov_fw(a0, time_limit, verbose=True):
    """Nesterov-accelerated Frank-Wolfe with LSE smooth approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    it = 0
    
    # Anneal temperature
    T_start = 5.0
    T_end = 0.3
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal temperature
        progress = min(1.0, it / 5000)
        T = T_start * (T_end / T_start) ** progress
        
        grad = lse_gradient(y, T)
        
        # FW: move towards vertex that minimizes <grad, v>
        # v is either e_i * norm (for some i) or 0
        i_min = int(np.argmin(grad))
        i_max = int(np.argmax(grad))
        
        # Use the vertex with minimum gradient
        v = np.zeros(n)
        v[i_min] = norm
        
        # Step size
        alpha = 2.0 / (it + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s
        
        if it == 1:
            x_new = v.copy()
        else:
            momentum = (it - 1.0) / (it + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s = x_new.sum()
            if s < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s
        
        y = y_new
        a = x_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        self_gap = np.dot(grad, a) - np.min(grad)  # FW gap
        if self_gap * norm < 1e-10:
            break
        
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FW it={it} T={T:.3f} ({elapsed:.1f}s): maxC={best_score:.6f}")
    
    return best_a, best_score


def lse_projected_grad(a0, time_limit, verbose=True):
    """Projected gradient descent with LSE smooth approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_to(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T_start = 5.0
    T_end = 0.3
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 5000)
        T = T_start * (T_end / T_start) ** progress
        
        grad = lse_gradient(a, T)
        
        # Learning rate: adaptive
        lr = norm / (T * (1 + 0.1 * it))
        lr = min(lr, 0.5)  # Cap
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtracking
            a_new2 = a - lr * 0.5 * grad
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 * norm / s2
                score2 = fast_autocorr(a_new2).max()
                if score2 < best_score + 1e-10:
                    a = a_new2
    
    return best_a, best_score

def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def normalize_to(a, target_sum):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def lse_gradient_fft(a, T):
    """
    Compute gradient of T*log(sum exp(C_k/T)) where C = autocorr(a).
    Formula: grad_i = (2/T) * sum_k w_k * a[k-i]
    Computed via FFT in O(n log n).
    """
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    conv = fast_autocorr(a)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / T)
    w = exp_c / exp_c.sum()

    W = fft(w, N)
    A = fft(a, N)

    # sum_k w[k] * a[k-i] = ifft(W * conj(A))[i] for i in [0,...,n-1]
    # (verified: for zero-padded real sequences with N >= 2n-1)
    cross = np.real(ifft(W * np.conj(A)))[:n]

    return 2.0 * cross / T


def lse_nesterov_fw(a0, time_limit, verbose=False):
    """Nesterov-accelerated Frank-Wolfe with LSE smooth approximation."""
    n = len(a0)

    a = normalize_to(a0, norm)

def build_conv_matrix_cols(a, n):
    """Build the convolution matrix T where (T @ b)[k] = sum_j b[j]*a[k-j]."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None, tol=0.05):
    """Solve LP best-response with optional constraint screening."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_cols(a, n)

    if active_set is not None and len(active_set) > 0:
        constraints = [T[active_set, :] <= M]
    else:
        conv_curr = T @ a
        active_set = np.where(conv_curr > M * (1.0 - tol))[0]
        if len(active_set) < 5:
            active_set = np.where(conv_curr > M * 0.3)[0]
        if len(active_set) == 0:
            active_set = np.arange(size)
        constraints = [T[active_set, :] <= M]

    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set

    # Verify constraints on full set
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1 + 1e-5))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) < 0.95 * size:
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T[new_active, :] <= M]

                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active
            else:

        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.8) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (t={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE Nesterov-FW (complementary, different landscape) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LSE-FW n={n_best} (t={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_nesterov_fw(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_fw_n{n_best}")

    # === Phase 3: Targeted Pair CD (peak-directed) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: Targeted PCD n={n_best} (t={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # === Phase 4: LP Nesterov again (refine) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: LP Nesterov refine n={n_best} (t={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t4, solver, verbose=False)
        do_try_update(a_out, f"lp_refine_n{n_best}")

    # === Phase 5: Fine Pair CD (final polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Fine PCD n={n_best} (t={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_fine(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pcd_fine_n{n_best}")

    # === Phase 6: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_base + 200
    t6 = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 6: LP Nesterov n={n_alt} (t={t6:.0f}s) ===")
    if t6 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n_alt}")

    # === Phase 7: Final fine polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 7: Final PCD n={n_best} (t={t7:.0f}s) ===")
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_fine(a_norm, t7, rng, verbose=False)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
