"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale initialization (solve small LP, upsample + diverse profiles)
2. Subgradient mirror descent with top-k peak weighting
3. Paired coordinate descent exploiting the pairing structure at active peaks

5. Fine polishing

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lp_solve_best_response(a, solver=None):
    """
    Solve the LP: minimize M subject to (a*b)[k] <= M, sum(b)=1, b>=0.
    Returns the optimal b.
    """
    import cvxpy as cp
    
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix: (a*b)[k] = sum_j a[...] * b[j]
    # (a*b)[k] = sum_j a[k-j] * b[j] (with a[i]=0 for i<0 or i>=n)
    # So the constraint is: for each k, sum_j A[k,j] * b[j] <= M
    # where A[k,j] = a[k-j] if 0<=k-j<n else 0
    
    # For efficiency, use the convolution structure
    # A is a (2n-1) x n matrix where A[k,j] = a[k-j] if valid
    
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a  # T[k,j] = a[k-j] for k in [j, j+n-1]
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def find_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def upsample(a_small, n_target):
    """Upsample a_small to length n_target by linear interpolation."""
    n_small = len(a_small)
    if n_small == n_target:
        return a_small.copy()
    x_small = np.linspace(0, 1, n_small)
    x_target = np.linspace(0, 1, n_target)
    a_target = np.interp(x_target, x_small, a_small)
    return np.maximum(a_target, 0)


def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Linear ramp (various slopes)
    for slope in [0.5, 1.0, 2.0, 3.0]:
        t = np.arange(n, dtype=np.float64) / (n - 1)
        a = 1.0 + slope * t
        a = np.maximum(a, 0.1)
        a = a / a.sum()
        inits.append(a.copy())
    
    # 3. Exponential profiles
    for rate in [0.5, 1.0, 2.0]:
        t = np.arange(n, dtype=np.float64) / (n - 1)
        a = np.exp(rate * t)
        a = a / a.sum()
        inits.append(a.copy())
        a2 = np.exp(-rate * t)
        a2 = a2 / a2.sum()
        inits.append(a2.copy())
    
    # 4. Sigmoid profiles
    for center in [0.3, 0.5, 0.7]:
        for steep in [5, 10, 20]:
            t = np.arange(n, dtype=np.float64) / (n - 1)
            a = 1.0 / (1.0 + np.exp(-steep * (t - center)))
            a = a / a.sum()
            inits.append(a.copy())
    
    # 5. Two-block with smooth transition
    for ratio in [1.5, 2.0, 3.0]:
        t = np.arange(n, dtype=np.float64) / (n - 1)
        a = np.where(t < 0.5, ratio, 1.0)
        # Smooth with a window
        from scipy.ndimage import uniform_filter1d
        a_smooth = uniform_filter1d(a, size=max(2, n // 32))
        a_smooth = np.maximum(a_smooth, 0.01)
        a_smooth = a_smooth / a_smooth.sum()
        inits.append(a_smooth.copy())
    
    # 6. Cosine window (asymmetric)
    for asym in [0.3, 0.5, 0.7]:
        t = np.arange(n, dtype=np.float64)
        center = n * asym
        x = (t - center) / max(n * 0.5, 1)
        x = np.clip(x, -1, 1)
        a = (0.5 * (1 + np.cos(np.pi * x))) ** 1.5
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)
    
    # 7. Random smooth (Gaussian process-like)
    for _ in range(5):
        freqs = rng.random(10) * n
        phases = rng.random(10) * 2 * np.pi
        t = np.arange(n, dtype=np.float64) / n
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        inits.append(a.copy())
    
    return inits


def subgradient_md(a0, time_limit, lr_init=0.05, top_k=30, verbose=True):
    """
    Subgradient mirror descent with top-k peak weighting.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Top-k peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * (a[k - i_end:k - i_start + 1][::-1] + a[i_start:i_end + 1])
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg / max(sg.max(), 1e-10))
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        avg_a = (it * avg_a + a_new) / (it + 1)
        a = a_new
        conv = fast_convolve(a)
        
        # Check running average periodically
        if it % 200 == 0:
            avg_score = 2.0 * n * fast_convolve(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def paired_coord_descent(a0, time_limit, verbose=True):
    """
    Paired coordinate descent: updates pairs (j, k*-j) together to maintain
    the pairing structure at the active peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Process pairs (j, k_star - j)
        processed = set()
        
        for j in range(n):
            if a[j] < 1e-15:
                continue
            partner = k_star - j
            if partner < 0 or partner >= n or partner == j or partner in processed:
                continue
            
            processed.add(j)
            processed.add(partner)
            
            if a[partner] < 1e-15:
                continue
            
            # Try shifts delta: a[j] += delta, a[partner] -= delta
            # This doesn't change (a*a)[k_star]
            base = a[j]
            factors = [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in factors:
                delta = base * factor
                new_j = a[j] + delta
                new_p = a[partner] - delta
                if new_j < -1e-15 or new_p < -1e-15:
                    continue
                
                # Compute change in convolution
                # Delta (a*a)[k] = delta * (a[k-j] + a[j]) - delta * (a[k-partner] + a[partner])
                # But we need to be more careful with the quadratic terms
                
                # For k where both k-j and k-partner are valid:
                # Delta = delta * a[k-j] + delta * a[k-(k_star-j)] - delta * a[k-partner] - delta * a[k-(k_star-j)]
                
                # Actually, the change is:
                # Delta (a*a)[k] = delta * (a[k-j] - a[k-partner]) + delta * (a[j] - a[partner])... 
                # No, let me think more carefully.
                
                # (a*a)[k] = sum_i a[i] * a[k-i]
                # Change in a[j]: contributes delta * a[k-j] (when i=j) and delta * a[j] (when k-i=j, i.e., i=k-j)
                # Change in a[partner]: contributes -delta * a[k-partner] (when i=partner) and -delta * a[partner] (when i=k-partner)
                
                # So total: delta * (a[k-j] + a[k-(k-j)]) - delta * (a[k-partner] + a[k-(k-partner)])
                # = delta * (a[k-j] + a[2j-k]) - delta * (a[k-partner] + a[2*partner-k])  ... this is getting messy
                
                # Let me just compute the affected range directly
                k_start = min(j, partner)
                k_end = max(j, partner) + n - 1
                k_start = max(k_start, 0)
                k_end = min(k_end, 2*n-2)
                
                # Original affected values
                # delta contributes at: k in [j, j+n-1] (from a[k-j]) and k in [n-1-j, n-1-j+n-1] (from a[j] at position k-j)
                # Hmm, let me just do it the brute force way for the affected range
                
                affected_start = min(j, k_star - partner)  # min of the two "starts"
                affected_end = max(j + n - 1, k_star - partner + n - 1)  # This isn't right either
                
                # Let me just compute the full change vector
                # Change at position k: delta * a[k-j] + delta * a[k-(k_star-j)]... 
                
                # Actually, the cleanest way:
                # The change to a is: delta at position j, -delta at position partner
                # So delta_conv[k] = delta * a[k-j] + (-delta) * a[k-partner] + delta * a[j]... 
                
                # No. Let me think again.
                # (a_new * a_new)[k] - (a * a)[k]
                # = (a + da)(a + da) - aa where da[j] = delta, da[partner] = -delta
                # = da*a + a*da + da*da
                # da*a has contributions: delta * a[k-j] + (-delta) * a[k-partner]
                # a*da has contributions: delta * a[k-j] + (-delta) * a[k-partner]  (by symmetry of convolution... wait no)
                
                # (da * a)[k] = sum_i da[i] * a[k-i] = da[j]*a[k-j] + da[partner]*a[k-partner] = delta*a[k-j] - delta*a[k-partner]
                # (a * da)[k] = sum_i a[i] * da[k-i] = a[k-j]*da[j] + a[k-partner]*da[partner] = a[k-j]*delta - a[k-partner]*delta
                
                # So (da*a + a*da)[k] = 2*delta*(a[k-j] - a[k-partner])
                # (da*da)[k] = da[j]*da[k-j] + da[partner]*da[k-partner] = delta*da[k-j] - delta*da[k-partner]
                # da[k-j] is non-zero only if k-j == j (k=2j) or k-j == partner (k = j+partner = k_star)
                # da[k-partner] is non-zero only if k-partner == j (k = j+partner = k_star) or k-partner == partner (k=2*partner)
                
                # So (da*da)[k] is non-zero only at k=2j, k=k_star, k=2*partner
                # At k=k_star: delta*da[k_star-j] - delta*da[k_star-partner] = delta*da[partner] - delta*da[j] = delta*(-delta) - delta*delta = -2*delta^2
                
                # OK this is getting too complicated for the analysis. Let me just compute it numerically for the affected range.
                
                # The affected range of k is: [min(j,partner), max(j,partner)+n-1] intersected with [0, 2n-2]
                k_aff_start = min(j, partner)
                k_aff_end = min(max(j, partner) + n - 1, 2*n-2)
                
                if k_aff_end < 0 or k_aff_start > 2*n-2:
                    continue
                
                # Unaffected max
                if k_aff_start > 0:
                    unaff_left = conv[:k_aff_start].max()
                else:
                    unaff_left = -1e30
                if k_aff_end < 2*n-2:
                    unaff_right = conv[k_aff_end+1:].max()
                else:
                    unaff_right = -1e30
                unaff_max = max(unaff_left, unaff_right)
                
                # Compute new affected values
                new_affected = conv[k_aff_start:k_aff_end+1].copy()
                # Add contribution from delta at j: delta * a[k-j] for k in range
                for k_idx in range(k_aff_start, k_aff_end+1):
                    k_val = k_idx
                    contrib = 0.0
                    # da[j] = delta: contributes delta * a[k_val - j] if 0 <= k_val - j < n
                    if 0 <= k_val - j < n:
                        contrib += delta * a[k_val - j]
                    # da[partner] = -delta: contributes -delta * a[k_val - partner] if 0 <= k_val - partner < n
                    if 0 <= k_val - partner < n:
                        contrib -= delta * a[k_val - partner]
                    # Also the "other side" of convolution: a[k_val-i]*da[i]
                    # This is the same as above by the symmetry of the sum
                    # Wait, no. (da*a)[k] + (a*da)[k] = 2 * (da*a)[k] only if da and a are the same...
                    # Let me be more careful.
                    # (a_new * a_new)[k] - (a*a)[k] where a_new = a + da
                    # = (a+da)*(a+da) - a*a = 2*(da*a) + (da*da) (using commutativity)
                    # (da*a)[k] = da[j]*a[k-j] + da[partner]*a[k-partner] (with valid index checks)
                    # = delta * a[k-j] - delta * a[k-partner] (with checks)
                    # (da*da)[k] = da[j]*da[k-j] + da[partner]*da[k-partner]
                    # = delta * da[k-j] - delta * da[k-partner]
                    # da[k-j] is non-zero only at k-j = j (k=2j) or k-j = partner (k=j+partner=k_star)
                    # da[k-partner] is non-zero only at k-partner = j (k=j+partner=k_star) or k-partner = partner (k=2*partner)
                    
                    # So total at position k:
                    # 2*(delta*a[k-j] - delta*a[k-partner]) + (da*da)[k]
                    
                    # I already computed the 2*(da*a) part as "contrib" above
                    # Now add (da*da)[k]:
                    dd = 0.0
                    if k_val == 2*j:
                        dd += delta * delta  # da[j]*da[j]
                    if k_val == j + partner:  # = k_star
                        dd += delta * (-delta)  # da[j]*da[partner]
                        dd += (-delta) * delta  # da[partner]*da[j]
                    if k_val == 2*partner:
                        dd += (-delta) * (-delta)  # da[partner]*da[partner]
                    
                    new_affected[k_idx - k_aff_start] += 2 * contrib + dd
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                # Apply update
                a[j] = max(a[j] + best_delta, 0.0)
                a[partner] = max(a[partner] - best_delta, 0.0)
                
                # Update convolution (recompute affected range)
                k_aff_start = min(j, partner)
                k_aff_end = min(max(j, partner) + n - 1, 2*n-2)
                # Just recompute the affected range from scratch (correct but slower)
                # For now, let's just recompute the full convolution every few updates
                # Actually, let me just recompute for correctness
                if improved or sweeps % 10 == 0:
                    conv = fast_convolve(a)
                else:
                    # Incremental update
                    for k_idx in range(max(k_aff_start, 0), min(k_aff_end, 2*n-2) + 1):
                        contrib = 0.0
                        if 0 <= k_idx - j < n:
                            contrib += best_delta * a[k_idx - j]
                        if 0 <= k_idx - partner < n:
                            contrib -= best_delta * a[k_idx - partner]
                        dd = 0.0
                        if k_idx == 2*j:
                            dd += best_delta * best_delta
                        if k_idx == j + partner:
                            dd += -2 * best_delta * best_delta
                        if k_idx == 2*partner:
                            dd += best_delta * best_delta
                        conv[k_idx] += 2 * contrib + dd
                
                M = conv.max()
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        # Also do standard coordinate updates for unpaired positions
        for j in range(n):
            partner = k_star - j
            if partner >= 0 and partner < n and partner != j:
                continue  # Already handled as a pair
            if a[j] < 1e-15:
                continue
            
            k_start = j
            k_end = min(j + n - 1, 2*n-2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2*n-2:
                unaff_right = conv[k_end+1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - j:k_end - j + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            base = a[j]
            for factor in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
                delta = base * factor
                new_val = a[j] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv[k_start:k_end+1] + 2.0 * delta * a_shifted
                if 2*j >= k_start and 2*j <= k_end:
                    idx = 2*j - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2*j] + 2.0 * delta * a[j] + delta * delta
                else:
                    new_affected = new_affected.copy()
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[j] = max(a[j] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2*j >= k_start and 2*j <= k_end:
                    idx = 2*j - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[j] - best_delta) + best_delta * best_delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        # Refresh convolution periodically to avoid drift
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = conv.max()
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score
