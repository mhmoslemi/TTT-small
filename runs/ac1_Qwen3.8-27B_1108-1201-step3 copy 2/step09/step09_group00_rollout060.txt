    n = len(a0)

        # Last equation: sum_{j in S} a[j] = 1
        
        # Actually, the system is:

        # Then: M @ a_S = nu * |A| * 1

                    if r_idx < m and S[r_idx] == r:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def symmetrize(a):
    """Enforce symmetry: a_j = a_{n-1-j}."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for j in range(half):
        a_sym[j] = a[j]
        a_sym[n - 1 - j] = a[j]
    if n % 2 == 1:
        a_sym[n // 2] = a[n // 2]
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def build_conv_matrix_sym(a_half, n):
    """Build convolution matrix for symmetric sequences.

    for j in range(h):

        a_full[n - 1 - j] = a_half[j]

        a_full[n // 2] = a_half[n // 2]

    T = np.zeros((n, h))
    for j in range(h):

        if use_symmetry:
            a = symmetrize(a)
            y = symmetrize(y)
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            
            if stall_rounds >= 15:
                noise_scale = 0.02 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                if use_symmetry:
                    y = symmetrize(y)
                stall_rounds = 0
            
            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def kkt_polish(a0, time_limit, max_iters=5000):
    """KKT-based active set polishing.
    
    At optimum, for active set A (where conv = max) and support S (where a > 0):
    sum_{k in A} a[k-j] = lambda * |A| for all j in S
    sum_{k in A} a[k-j] <= lambda * |A| for all j not in S
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: positions within 0.1% of max
        threshold = max_b * (1.0 - 1e-4)
        A = np.where(b >= threshold)[0]
        
        # Support: positions with a > 1e-8
        S = np.where(a > 5e-8)[0]
        
        m = len(S)
        |A_len| = len(A)
        
        if m < 2 or |A_len| < 1:
            break
        
        # Build linear system:
        # For j in S: sum_{k in A} a[k-j] (only counting k-j in S) = nu * |A|
        # sum_{j in S} a[j] = 1
        # Unknowns: a_S (length m) and nu (scalar) -> m+1 unknowns
        
        S_list = S.tolist()
        S_set = set(S_list)
        S_idx = {v: i for i, v in enumerate(S_list)}
        
        M_mat = np.zeros((m, m))
        for i, j in enumerate(S_list):
            for k in A:
                r = k - j
                if r in S_set:
                    M_mat[i, S_idx[r]] += 1
        
        # System: M_mat @ a_S - |A| * nu * 1 = 0
        #          1^T @ a_S = 1
        Full = np.zeros((m + 1, m + 1))
        Full[:m, :m] = M_mat
        Full[:m, m] = -float(|A_len|)
        Full[m, :m] = 1.0
        
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(Full, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(Full, rhs, rcond=None)[0]
            except Exception:
                break
        
        a_S_new = sol[:m]
        nu_new = sol[m]
        
        # Check non-negativity
        min_a = a_S_new.min()
        if min_a < -1e-8:
            # Remove the most negative element from support
            worst = np.argmin(a_S_new)
            S = np.delete(S, worst)
            a_at = np.zeros(n)
            for i, j in enumerate(S):
                a_at[j] = max(0.0, a_S_new[i])
            st = a_at.sum()
            if st > 1e-15:
                a = a_at / st
            continue
        
        # Build new a
        a_new = np.zeros(n)
        for i, j in enumerate(S_list):
            a_new[j] = max(0.0, a_S_new[i])
        st = a_new.sum()
        if st < 1e-15:
            break
        a_new = a_new / st
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Check convergence
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        if abs(max_new - max_b) < 1e-12:
            break
        
        if it % 200 == 0:
            print(f"    KKT iter={it}: score={best_score:.10f}, |S|={m}, |A|={|A_len|}")
    
    return best_a, best_score


def subgradient_polish(a0, time_limit, max_steps=200000):
    """Subgradient method with adaptive step size targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step_size = 0.005
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: positions within 1% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient: average of shifted a over active positions
        g = np.zeros(n)
        for k in active:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                r = k - j
                if 0 <= r < n:
                    g[j] += a[r]
        g /= len(active)
        
        # Projected step: a - eta * g, project to simplex
        a_trial = a - step_size * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 20:
            step_size *= 0.92
            no_improve = 0
        step_size = max(step_size, 1e-8)
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.10f} eta={step_size:.2e}")
    
    return best_a, best_score
