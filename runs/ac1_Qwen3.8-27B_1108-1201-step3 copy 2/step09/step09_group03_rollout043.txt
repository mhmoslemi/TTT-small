import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v, s=1.0):
    """Project vector v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - s)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - s) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.full(n, s / n)


def score_from_conv(a, conv):
    """Compute score = 2*n*max(conv)/(sum(a))^2 from precomputed conv."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def fw_line_search(a, C, j_star, n_evals=30):
    """
    Exact line search for FW step: minimize f(t) = max_k C_new[k]
    where C_new[k] = (1-t)^2*C[k] + 2*t*(1-t)*S[k] + t^2*[k==0]
    and S[k] = a[j_star+k] + a[j_star-k] (with bounds).
    
    Returns optimal t in [0, 1].
    """
    n = len(a)
    j = int(j_star)

    # Precompute S[k] = a[j+k] + a[j-k] for k in [0, 2n-2]
    # Represent as: S_full[k] for k in 0..2n-2
    # a[j+k] is valid for k in [0, n-1-j]
    # a[j-k] is valid for k in [0, j]
    S = np.zeros(2 * n - 1)
    if j < n:
        S[:n - j] += a[j:]       # a[j+k] for k=0..n-1-j
        S[:j + 1] += a[:j + 1][::-1]  # a[j-k] for k=0..j

    # D[k] = 1 if k==0 else 0 (the e_j * e_j term)
    # D only has D[0] = 1

    def eval_f(t):
        # C_new = (1-t)^2 * C + 2*t*(1-t)*S + t^2 * D
        # D only contributes at k=0
        c1 = (1.0 - t) * (1.0 - t)
        c2 = 2.0 * t * (1.0 - t)
        c3 = t * t
        vals = c1 * C + c2 * S
        vals[0] += c3  # D[0] = 1
        return float(vals.max())

    # Ternary search / golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    # First, check if t=0 is optimal (no improvement)
    f0 = eval_f(0.0)
    if f0 >= eval_f(0.01):
        # Minimum is not at 0
        pass

    for _ in range(n_evals):
        c = lo + gr * (hi - lo)
        d = lo + (1.0 - gr) * (hi - lo)
        fc = eval_f(c)
        fd = eval_f(d)
        if fc < fd:
            hi = d
        else:
            lo = c

    t_opt = (lo + hi) / 2.0

    # Verify: check endpoints
    f_lo = eval_f(0.0)
    f_hi = eval_f(1.0)
    f_mid = eval_f(t_opt)
    if f_lo <= f_mid and f_lo <= f_hi:
        t_opt = 0.0
    elif f_hi <= f_mid and f_hi <= f_lo:
        t_opt = 1.0

    return t_opt

        k_star = int(np.argmax(C))

        # Compute subgradient: grad[j] = a[j+k*] + a[j-k*] (with bounds)
        # For the FW step, we want j* = argmin grad[j]
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = 2.0 * a
        else:
            grad[:n - k_star] = a[k_star:]
            grad[k_star:] += a[:n - k_star]

        j_star = int(np.argmin(grad))

        # Line search
        t_opt = fw_line_search(a, C, j_star, n_evals=25)

        if t_opt < 1e-10:
            no_improve += 1
            if no_improve > 500:
                break
            continue

        # Update
        a = (1.0 - t_opt) * a
        a[j_star] += t_opt

        # Update C incrementally:
        # C_new = (1-t)^2 * C_old + 2*t*(1-t) * S + t^2 * D
        # where S[k] = a_old[j+k] + a_old[j-k], D[0]=1
        j = j_star
        S = np.zeros(2 * n - 1)
        # Use old a for S (before the update)
        a_old = (a - t_opt * np.eye(n)[j])  # recover old a at position j
        # Actually, let's just recompute: a_old[j] = a[j] - t_opt (approximately)
        # Better: compute S from the pre-update a
        # a_old = a - t_opt * e_j (since a = (1-t)*a_old + t*e_j, so a_old = (a - t*e_j)/(1-t))
        # This is getting messy. Let's just recompute C from scratch every 100 steps,
        # and use the incremental update otherwise.

        # Actually, the cleanest approach: just recompute C periodically
        if step % 100 == 0:
            C = fast_convolve(a)
        else:
            # Incremental update
            # a_new = (1-t)*a_old + t*e_j
            # C_new = (1-t)^2 * C_old + 2*t*(1-t) * (e_j * a_old) + t^2 * (e_j * e_j)
            # (e_j * a_old)[k] = a_old[j+k] + a_old[j-k]
            # (e_j * e_j)[k] = 1 if k=0 else 0
            # We need a_old = (a_new - t*e_j) / (1-t)
            a_old = np.zeros(n)
            a_old[:] = a
            a_old[j] = (a[j] - t_opt) / (1.0 - t_opt) if (1.0 - t_opt) > 1e-15 else 0.0

            S = np.zeros(2 * n - 1)
            if j < n:
                S[:n - j] += a_old[j:]
                S[:j + 1] += a_old[:j + 1][::-1]

            c1 = (1.0 - t_opt) * (1.0 - t_opt)
            c2 = 2.0 * t_opt * (1.0 - t_opt)
            c3 = t_opt * t_opt
            C = c1 * C + c2 * S
            C[0] += c3

        # Evaluate
        cur_score = 2.0 * n * float(C.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def cd_exact_1d(a_init, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact 1D optimization.
    For each coordinate i, we consider the "away step": decrease a[i] by delta
    and distribute it proportionally to all other coordinates.
    The objective is convex in delta, so we use bisection.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    C = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(C.max())
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-12:
                continue

            # Away step: a[i] decreases by delta, others increase proportionally
            # a_new[i] = a[i] - delta
            # a_new[j] = a[j] * (1 - a[i] + delta) / (1 - a[i]) for j != i
            # Let ratio = (1 - a[i] + delta) / (1 - a[i]) = 1 + delta / (1 - a[i])
            # So a_new[j] = a[j] * (1 + delta / (1-a[i])) for j != i
            # and a_new[i] = a[i] - delta

            # The new autocorrelation can be computed, but it's complex.
            # Alternative: use the FW frame. For coordinate i, the "vertex" to move towards
            # is the one that minimizes the gradient. But that's what FW does.
            # Let's use a simpler approach: try a few delta values and pick the best.

            # Actually, let's use the "proportional away step" approach:
            # a_new = (1 - t) * a + t * (a - a[i]*e_i) / (1 - a[i])
            #      = a + t * (a - a[i]*e_i)/(1-a[i]) - t * a
            #      = (1-t) * a + t * (a - a[i]*e_i)/(1-a[i])
            # This is equivalent to: move from a towards the "average of other coordinates"
            # which stays in the simplex and reduces a[i].

            # Simpler: just try decreasing a[i] and increasing a[j*] where j* is
            # the coordinate with the lowest "sensitivity" to the peaks.

            # For now, use the FW approach: move towards vertex j*.
            # Find j* that minimizes the subgradient.
            M = float(C.max())
            k_star = int(np.argmax(C))

            # Subgradient at a for peak k_star
            if k_star == 0:
                grad = 2.0 * a
            else:
                grad = np.zeros(n)
                grad[:n - k_star] = a[k_star:]
                grad[k_star:] += a[:n - k_star]

            # Exclude i from candidates (we're optimizing coordinate i)
            # Actually, in CD we fix all other coordinates and optimize a[i].
            # The "move" is: transfer delta from i to some j.
            # Best j: the one with minimum grad[j] (excluding j=i).

            # But this is essentially the FW step again. Let me just do FW steps
            # in a round-robin fashion over coordinates.

            # For now, just do a simple 1D search: vary a[i] and distribute to others.
            # This is the "proportional" move.
            
            # Compute the objective as a function of delta (amount removed from i):
            # a_new[i] = a[i] - delta
            # a_new[j] = a[j] * (1 + delta/(1-a[i])) for j != i
            # The new C is a quadratic in delta. We want to minimize max(C).
            
            # This is complex to compute analytically. Let's use a simpler method:
            # Just try a few values of t (the FW parameter) for the best vertex.
            
            j_star = int(np.argmin(grad))
            if j_star == i:
                # Find second best
                grad_copy = grad.copy()
                grad_copy[i] = float('inf')
                j_star = int(np.argmin(grad_copy))

            t_opt = fw_line_search(a, C, j_star, n_evals=15)

            if t_opt > 1e-10:
                a_old = a.copy()
                a = (1.0 - t_opt) * a
                a[j_star] += t_opt
                C = fast_convolve(a)
                cur_score = 2.0 * n * float(C.max())
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    improved = True
                elif cur_score > 2.0 * n * float(fast_convolve(a_old).max()) + 1e-10:
                    # Revert if it got worse (shouldn't happen with exact line search)
                    a = a_old

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweep} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fw_away_steps(a_init, time_limit, verbose=True):
    """
    FW with away steps: at each iteration, either move towards the best vertex
    or away from the worst vertex (or both).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    C = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(C.max())
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        M = float(C.max())
        k_star = int(np.argmax(C))

        # Compute subgradient
        if k_star == 0:
            grad = 2.0 * a
        else:
            grad = np.zeros(n)
            grad[:n - k_star] = a[k_star:]
            grad[k_star:] += a[:n - k_star]

        # Best vertex (minimize gradient)
        j_star = int(np.argmin(grad))

        # Worst vertex (maximize a[j] * grad[j]) - for away step
        away_scores = a * grad
        j_away = int(np.argmax(away_scores))

        # Try FW step
        t_fw = fw_line_search(a, C, j_star, n_evals=20)
        f_fw = 2.0 * n * eval_fw_result(a, C, j_star, t_fw, n)

        # Try away step: move away from j_away
        # a_new = (a - gamma * e_{j_away}) / (1 - gamma)
        # where gamma = t * a[j_away] for some t
        # This is equivalent to the FW step towards the "average of others"
        # For simplicity, try a few gamma values.
        f_away = best_score
        best_away_gamma = 0.0
        if a[j_away] > 1e-10:
            for gamma_frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                gamma = gamma_frac * a[j_away]
                a_test = a.copy()
                a_test[j_away] -= gamma
                a_test = a_test / a_test.sum()
                C_test = fast_convolve(a_test)
                f_away_test = 2.0 * n * float(C_test.max())
                if f_away_test < f_away:
                    f_away = f_away_test
                    best_away_gamma = gamma

        # Try combined: FW step + away step
        # a_new = (1-t)*a + t*e_{j_star} - gamma*e_{j_away}
        # normalized to be in simplex
        f_combined = best_score
        best_params = None
        if t_fw > 1e-5 and a[j_away] > 1e-10:
            for gamma_frac in [0.2, 0.5, 0.8]:
                t_test = t_fw * 0.8
                gamma = gamma_frac * a[j_away]
                a_test = (1 - t_test) * a
                a_test[j_star] += t_test
                a_test[j_away] -= gamma
                if a_test[j_away] < 0:
                    a_test[j_away] = 0
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test = a_test / s_test
                C_test = fast_convolve(a_test)
                f_test = 2.0 * n * float(C_test.max())
                if f_test < f_combined:
                    f_combined = f_test
                    best_params = (t_test, gamma)

        # Pick the best move
        best_f = min(f_fw, f_away, f_combined, best_score)

        if best_f < best_score - 1e-12:
            if best_f == f_fw and t_fw > 1e-10:
                a = (1.0 - t_fw) * a
                a[j_star] += t_fw
            elif best_f == f_away and best_away_gamma > 1e-10:
                a[j_away] -= best_away_gamma
                a = a / a.sum()
            elif best_f == f_combined and best_params is not None:
                t_test, gamma = best_params
                a = (1 - t_test) * a
                a[j_star] += t_test
                a[j_away] -= gamma
                a[j_away] = max(a[j_away], 0.0)
                a = a / a.sum()

            C = fast_convolve(a)
            if 2.0 * n * float(C.max()) < best_score:
                best_score = 2.0 * n * float(C.max())
                best_a = a.copy()

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-Away step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def eval_fw_result(a, C, j_star, t, n):
    """Evaluate the FW result without full FFT (uses the closed form)."""
    j = int(j_star)
    S = np.zeros(2 * n - 1)
    if j < n:
        S[:n - j] += a[j:]
        S[:j + 1] += a[:j + 1][::-1]
    c1 = (1.0 - t) * (1.0 - t)
    c2 = 2.0 * t * (1.0 - t)
    c3 = t * t
    vals = c1 * C + c2 * S
    vals[0] += c3
    return float(vals.max())


def lp_best_response(a_init, time_limit, verbose=True):
    """
    LP-based best response: solve min M subject to (a*b)[k] <= M, sum(b)=1, b>=0.
    Then blend the result with a.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a_init, 2.0 * len(a_init) * float(fast_convolve(a_init).max())

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0

    # Build T matrix: T[k, j] = contribution of b[j] to conv(a, b)[k]
    # conv(a, b)[k] = sum_j a[k-j] * b[j] (for valid indices)
    # So T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iter=50000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        # Evaluate b_val
        conv_b = fast_convolve(b_val)
        score_b = 2.0 * n * float(conv_b.max())

        # Blend
        alpha = 3.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * float(fast_convolve(a_new).max())

        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        # Rebuild T
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    for power in [0.5, 1.0, 2.0, 4.0]:
        a = (0.5 * (1 + np.cos(np.pi * x))) ** power
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Periodic
    for period in [2, 3, 4, 5, 7, 8]:
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = max(0.0, 0.5 + 0.4 * np.cos(2 * np.pi * phase))
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Two humps
    for c1, c2 in [(0.25, 0.75), (0.3, 0.7), (0.2, 0.8)]:
        a = np.exp(-0.5 * ((t - n * c1) / (n * 0.15)) ** 2) + np.exp(-0.5 * ((t - n * c2) / (n * 0.15)) ** 2)
        s = a.sum()
        if s > 1e-10:
            inits.append(a / s)

    # Random
    a = rng.random(n)
    a = a / a.sum()
    inits.append(a)

    # Sparse (only half the positions nonzero)
    mask = rng.random(n) > 0.5
    a = np.zeros(n)
    a[mask] = 1.0
    s = a.sum()
    if s > 10:
        inits.append(a / s)

    return inits


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Nesterov accelerated smoothed subgradient (global)
    2. FW with exact line search (mid-level)
    3. FW with away steps (refinement)
    4. Coordinate descent with exact 1D (fine)
    5. LP best response (polish)
    6. Final CD (last mile)
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Determine sequence length
    n = 1024
    if parent is not None and len(parent) in (512, 1024, 2048):
        n = len(parent)

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()

    # === Phase 1: Nesterov from parent and diverse inits ===

    # === Phase 2: FW with line search ===
    print(f"\n=== Phase 2: FW Line Search (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 400
