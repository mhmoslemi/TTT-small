def lse_value_grad(a, T, t0, time_limit):
    """Compute log-sum-exp smoothed objective and its simplex gradient."""
    if time.monotonic() - t0 > time_limit:
        n = len(a)
        return 1e15, np.zeros(n)
    n = len(a)

    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)

    value = (2.0 * n / T) * log_sum_exp

def lbgfs_refine_a(a0, time_limit, T_start=80.0, T_end=0.05, n_stages=15):
    """L-BFGS with multi-stage temperature annealing on the simplex."""
    n = len(a0)
    a = norm_sequence(a0)
    t0 = time.monotonic()
    z = np.log(np.maximum(a, 1e-300))

    for stage in range(n_stages):
        frac_start = stage / n_stages
        frac_end = (stage + 1) / n_stages
        T = T_start * (T_end / T_start) ** frac_start
        if T < 1.0:
            T = max(T, T_end)

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters_this_stage = max(20, int(500 * remaining / (time_limit * frac_end)))

        def func(z_in):
            a_local = np.exp(z_in - z_in.max())
            a_local = a_local / a_local.sum()
            val, _ = lse_value_grad(a_local, T, t0, time_limit)
            return val

        def jac(z_in):
            a_local = np.exp(z_in - z_in.max())
            a_local = a_local / a_local.sum()
            _, g = lse_value_grad(a_local, T, t0, time_limit)
            # Transform from a-space to z-space: dz->da gives da/dz = a*(1 - a) for softmax
            # Actually z = log(a), so da/dz = a. The gradient in z-space is grad_a / a (element-wise)
            # But we're working with z = log(a) and a = exp(z)/sum(exp(z))
            # grad_z = grad_a * a (chain rule: da_i/dz_j = a_i(d_ij - a_j))
            # Actually for softmax: if a = softmax(z), then dL/dz_j = a_j * (dL/da_j - sum_k a_k * dL/da_k)
            # But we already have the simplex gradient: g_simplex = grad_a - a*(grad_a . a)
            # For softmax parameterization: grad_z = a * g_simplex ... no
            # If a = softmax(z) = exp(z)/sum(exp(z)), and we want dL/dz where L = f(softmax(z)):
            # dL/dz_j = sum_i (dL/da_i) * (da_i/dz_j) = sum_i (dL/da_i) * a_i * (delta_ij - a_j)
            # = a_j * dL/da_j - a_j * sum_i a_i * dL/da_i
            # = a_j * (dL/da_j - (a . dL/da))
            # = a_j * g_simplex_j
            gz = a_local * g
            return gz

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_this_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_this_stage * 500, 'disp': False}
            )
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
        except Exception:
            pass

    return a

    """Direct coordinate descent targeting the peak of the true (unsmoothed) convolution."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:

        # Top contributors (indices in the original array)
        top_local = np.argsort(contribs)[::-1][:10]
        top_global = idx[top_local]

        # Low contribution coordinates (good targets for redistribution)
        all_contrib = np.zeros(n)
        all_contrib[idx] = contribs
        # Also consider contribution to OTHER peaks (top 5 peaks)
        peak_indices = np.argsort(c)[::-1][:5]
        other_contrib = np.zeros(n)
        for pk in peak_indices:
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            idx2 = np.arange(js2, je2 + 1)
            other_contrib[idx2] += a[idx2] * a[pk - idx2]

        low_coords = np.argsort(all_contrib + 0.5 * other_contrib)
        # Use the lowest 1/4 as redistribution targets
        n_low = max(4, n // 4)
        low_set = low_coords[:n_low]

        improved = False
        # Try reducing top contributors
        for gi in top_global:
            if a[gi] < 1e-12:
                continue
            for factor in [0.95, 0.90, 0.85, 0.78, 0.70]:
                new_val = a[gi] * factor
                diff = a[gi] - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[gi] = new_val
                # Redistribute to low-contrib coordinates
                mask = np.isin(low_set, gi)
                targets = low_set[~mask]
                if len(targets) == 0:
                    continue
                t_weights = a_trial[targets] + 1e-12
                a_trial[targets] += diff * t_weights / t_weights.sum()
                a_trial = norm_sequence(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            # Try pair swaps (reduce one hot, increase one cold)
            found = False
            for gi in top_global[:6]:
                if a[gi] < 1e-12:
                    continue
                for gj in low_set[-min(8, n_low // 5):]:
                    if gi == gj or a[gj] >= a[gi]:
                        continue
                    for swap_frac in [0.1, 0.2, 0.3]:
                        amount = a[gi] * swap_frac
                        a_trial = a.copy()
                        a_trial[gi] -= amount
                        a_trial[gj] += amount
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            found = True
                            no_improve = 0
                            break
                    if found:
                        break
                if found:
                    break

            if not found:
                no_improve += 1
                if no_improve > 500:
                    # Perturb and reset counter
                    rng_l = np.random.default_rng(42 + step)
                    a = a + 0.02 * rng_l.standard_normal(n)
                    a = norm_sequence(a)
                    no_improve = 0

    return best_a, best_score


def nesterov_subgrad(a0, time_limit, n_max_iters=50000):
    """Nesterov-accelerated subgradient on the true max objective."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(y)
        k_star = int(np.argmax(c))

        # Subgradient of max(c) at a=y
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k_star - idx]  # dc_k/d(a_i) = 2*a_{k-i}

        # Project gradient onto simplex tangent
        S = np.dot(g, a)
        g_tangent = g - a * S

        step_size = 2.0 / (it + 2)

        # Nesterov update
        y_new = (1 - step_size) * y + step_size * a
        y_new = y_new - step_size * g_tangent
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / sy
        y = y_new

        # Update a (Chebyshev-like)
        a_trial = y
        sc = score_normalized(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0 and stall > 1000:
                # Restart with perturbation
                rng_l = np.random.default_rng(999 + it)
                y = a + 0.05 * rng_l.standard_normal(n)
                y = norm_sequence(y)
                stall = 0

    return best_a, best_score


def make_init_exp_decay(n, rate=0.3):
    """Exponentially decaying initialization."""
    x = np.arange(n)
    a = np.exp(-rate * x / n)
    return norm_sequence(a)


def make_init_ramp(n, power=1.0):
    """Power-law ramp (asymmetric)."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** power
    return norm_sequence(a)


def make_init_bump(n, center=0.3, width=0.2):
    """Off-center Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sequence(a)


def make_init_cosine_taper(n, asym=1.0):
    """Asymmetrized cosine taper."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** (2 * asym)
    return norm_sequence(a)


def make_init_quadratic(n, asym=1.0):
    """Parabolic shape with asymmetry."""
    x = np.linspace(0, 1, n)
    a = np.exp(-asym * x * (1 - x) * 6) * (1 + 0.5 * x)
    return norm_sequence(a)


def make_init_plateau(n, w=0.6):
    """Plateau with smooth edges."""
    x = np.linspace(0, 1, n)
    half = w / 2
    a = np.clip((x - (1 - w)) / 0.1, 0, 1) * np.clip((w + 1 - x - w) / 0.1, 0, 1)
    a = np.clip((x + 0.1) / 0.1, 0, 1) * np.clip((1 - x + 0.1) / 0.1, 0, 1)
    a = a * (1.0 + 0.3 * np.sin(2 * np.pi * x))
    return norm_sequence(a)


def make_init_harmonic(n, k=3):
    """Harmonic-based asymmetric shape."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * k * x) + 0.2 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.01)
    return norm_sequence(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-scale, multi-start optimization for minimizing 2n*max(conv)/sum^2.
    Uses coarse-to-fine scaling, L-BFGS with LSE smoothing, peak-attack, and
    Nesterov subgradient method.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_final = len(best_seq) if best_seq is not None else 960

    # === Phase 1: Coarse-to-fine optimization ===
    print(f"\n=== Phase 1: Coarse-to-Fine (target n={n_final}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Start small and grow
        scales = [64, 128, 256, 512]
        if n_final <= 512:
            scales = [32, 64, 128, n_final]
        elif n_final > 512:
            scales.append(n_final)

        # Initial shape at smallest scale
        inits = [
            make_init_exp_decay(64, 0.5),
            make_init_ramp(64, 1.5),
            make_init_bump(64, 0.35, 0.25),
            make_init_cosine_taper(64, 1.2),
        ]
        # Pick best init at small scale via quick L-BFGS
        best_small = None
        best_small_score = float('inf')
        t_per_init = min(5, rem / (len(inits) * 4))
        for init in inits:
            a_small = lbgfs_refine_a(init, t_per_init, T_start=20, T_end=1.0, n_stages=3)
            sc = score_normalized(a_small)
            if sc < best_small_score:
                best_small_score = sc
                best_small = a_small
        print(f"  Best small-scale score: {best_small_score:.6f}")

        # Grow through scales
        a_cur = best_small
        for target_n in scales[1:]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            a_interp = interpolate_sequence(a_cur, target_n)
            t_alloc = min(rem * 0.6, 120.0)
            if t_alloc > 10:
                a_cur = lbgfs_refine_a(a_interp, t_alloc, T_start=50, T_end=2.0, n_stages=8)
                sc = score_normalized(a_cur)
                print(f"  n={target_n}: score={sc:.6f}")

        try_update(a_cur, "coarse_fine")

    # === Phase 2: Multi-start at target length ===
    print(f"\n=== Phase 2: Multi-Start n={n_final} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 90:
        # Generate diverse initializations at target length
        inits_target = []
        if best_seq is not None:
            inits_target.append(best_seq.copy())
        inits_target.append(interpolate_sequence(best_small, n_final))
        inits_target.append(make_init_exp_decay(n_final, 0.8))
        inits_target.append(make_init_ramp(n_final, 2.0))
        inits_target.append(make_init_bump(n_final, 0.4, 0.3))
        inits_target.append(make_init_cosine_taper(n_final, 1.5))
        inits_target.append(make_init_quadratic(n_final, 1.2))
        inits_target.append(make_init_harmonic(n_final, 2))
        # Random perturbations of best
        for _ in range(3):
            pert = best_seq + 0.05 * rng.standard_normal(n_final) if best_seq is not None else make_init_ramp(n_final, 1.0)
            inits_target.append(norm_sequence(pert))

        # Quick screen: L-BFGS with limited budget on each
        t_per = max(10, min(40, rem / (len(inits_target) * 2)))
        candidates = []
        for i_init, a_init in enumerate(inits_target):
            rem = deadline - time.monotonic()
            if rem < t_per:
                break
            t_alloc = min(t_per, rem * 0.9)
            a_opt = lbgfs_refine_a(a_init, t_alloc, T_start=40, T_end=1.0, n_stages=6)
            sc = score_normalized(a_opt)
            candidates.append((sc, a_opt, i_init))
            print(f"  Init {i_init}: score={sc:.6f}")

        candidates.sort(key=lambda x: x[0])
        # Take top 3 for deeper optimization
        for sc, a_c, i_idx in candidates[:3]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_deep = min(rem * 0.35, 200.0)
            if t_deep > 20:
                a_deep = lbgfs_refine_a(a_c, t_deep, T_start=30, T_end=0.05, n_stages=14)
                try_update(a_deep, f"deep_{i_idx}")

    # === Phase 3: Peak attack on best ===
    print(f"\n=== Phase 3: Peak Attack (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t_pa = min(rem * 0.2, 120.0)
        if t_pa > 15:
            a_pa, _ = peak_attack(best_seq, t_pa, n_steps=50000)
            try_update(a_pa, "peak_attack")

    # === Phase 4: L-BFGS deep polish ===
    print(f"\n=== Phase 4: Deep L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t_lb = min(rem * 0.3, 250.0)
        if t_lb > 20:
            a_polish = lbgfs_refine_a(best_seq, t_lb, T_start=15, T_end=0.02, n_stages=18)
            try_update(a_polish, "deep_polish")

    # === Phase 5: Nesterov subgradient ===
    print(f"\n=== Phase 5: Nesterov Subgrad (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_seq is not None:
        t_ns = min(rem * 0.15, 100.0)
        if t_ns > 15:
            a_ns, _ = nesterov_subgrad(best_seq, t_ns, n_max_iters=50000)
            try_update(a_ns, "nesterov")

    # === Phase 6: Second peak attack ===
    print(f"\n=== Phase 6: Peak Attack 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_pa2 = min(rem * 0.15, 80.0)
        if t_pa2 > 10:
            a_pa2, _ = peak_attack(best_seq, t_pa2, n_steps=40000)
            try_update(a_pa2, "peak_attack_2")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        cur_n = len(best_seq)
        alt_lengths = []
        for mult in [0.75, 1.25, 1.5, 0.5]:
            alt_n = int(cur_n * mult)
            if 128 <= alt_n <= 4096 and alt_n != cur_n:
                alt_lengths.append(alt_n)
        for fixed in [256, 512, 1024, 1280, 2048]:
            if 128 <= fixed <= 4096 and fixed != cur_n:
                alt_lengths.append(fixed)
        alt_lengths = sorted(set(alt_lengths))

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_total = min(rem * 0.25, 150.0)
            if t_total < 20:
                break
            print(f"  Trying n={alt_n}")
            a_interp = interpolate_sequence(best_seq, alt_n)
            a_alt = lbgfs_refine_a(a_interp, t_total, T_start=30, T_end=0.1, n_stages=10)
            try_update(a_alt, f"len_{alt_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        # Quick L-BFGS
        t1 = min(rem * 0.3, 40.0)
        if t1 > 10:
            a_f1 = lbgfs_refine_a(best_seq, t1, T_start=5, T_end=0.01, n_stages=6)
            try_update(a_f1, "final_lb")

        rem = deadline - time.monotonic()
        # Quick peak attack
        t2 = min(rem * 0.3, 30.0)
        if t2 > 10:
            a_f2, _ = peak_attack(best_seq, t2, n_steps=20000)
            try_update(a_f2, "final_pa")

        rem = deadline - time.monotonic()
        # Final Nesterov
        t3 = min(rem * 0.3, 25.0)
        if t3 > 10:
            a_f3, _ = nesterov_subgrad(best_seq, t3, n_max_iters=20000)
            try_update(a_f3, "final_ns")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
