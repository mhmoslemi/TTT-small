import time
from scipy.fft import fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def hio_iterate(a, n_iters=5000, beta=0.9):
    """HIO-style iteration to flatten the auto-convolution spectrum."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    
    for it in range(n_iters):
        # Forward FFT
        F = np.fft.rfft(a, 2*n)
        # Target: flatten |F|^2
        amp = np.abs(F)
        target_amp = np.mean(amp)  # or use median for robustness
        
        # Modify amplitude, keep phase
        phase = F / (np.abs(F) + 1e-30)
        F_new = target_amp * phase
        
        # Inverse FFT
        a_new = np.fft.irfft(F_new, 2*n)[:n]
        
        # Project to non-negative simplex
        a_proj = project_simplex(np.maximum(a_new, 0))
        
        # Relaxation
        a = (1 - beta) * a + beta * a_proj
        a = project_simplex(a)
        
        # Track best
        if it % 10 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def hio_optimize(a0, n_iters=3000, beta_start=0.95, beta_end=0.5, power=1.0):
    """
    HIO (Hybrid Input-Output) to flatten auto-convolution spectrum.
    Alternates: FFT -> flatten amplitude -> IFFT -> project to non-negative simplex.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    fft_len = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = score_normalized(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        # Forward FFT
        F = rfft(a, fft_len)
        amp = np.abs(F)
        phase = F / (amp + 1e-30)

        # Target: compress dynamic range of amplitude
        if power == 1.0:
            target_amp = np.mean(amp)
            F_new = target_amp * phase
        else:
            # Power-law compression
            amp_compressed = amp ** power
            target_amp = np.mean(amp_compressed)
            F_new = (target_amp / (amp_compressed + 1e-30)) * F

        # Inverse FFT, take first n samples
        a_new = irfft(F_new, fft_len)[:n]

        # Project to non-negative simplex
        a_proj = project_simplex(np.maximum(a_new, 0))

        # Ergodic mixing (relaxation)
        a = (1.0 - beta) * a + beta * a_proj
        a = project_simplex(a)

        # Track best
        if it % 20 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a, using FFT."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j a[j] * w[i-j] (circular correlation of a with w, truncated)
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=300.0,
                lr_init=0.06, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            # Nesterov momentum
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            if step_count > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 50 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step % 500 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"        NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.CBC]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_max_iters=30000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 13)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Multi-peak shaving with coordinate and pair adjustments."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        if peak_val <= 0:
            break
        threshold = peak_val - 0.08 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in top_idx[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Simultaneous small reduction of top few
            a_trial = a.copy()
            for ci in top_idx[:4]:
                i = idx[ci]
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Try random perturbation to escape
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score * 1.001:
                    a = a_trial
                else:
                    break

    return best_a, best_score


def make_init_sin2(n, rng):
    """sin^2 based initialization."""
    x = np.linspace(0, 1, n, endpoint=False)
    a = np.sin(np.pi * x) ** 2
    a = a + 1e-10
    a = a / a.sum()
    return a


def make_init_raised_cos(n, rng):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2 * np.pi * x))
    a = a + 1e-10
    a = a / a.sum()
    return a


def make_init_asymmetric(n, alpha, rng):
    """Asymmetric initialization with random shape."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + alpha * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif choice == 1:
        a = 1.0 + alpha * (2 * x - 1)
    elif choice == 2:
        a = 1.0 + alpha * np.sin(4 * np.pi * x)
    elif choice == 3:
        c1 = rng.uniform(0.2, 0.8)
        c2 = 1.0 - c1
        a = np.exp(-4.0 * (x - c1) ** 2) + 0.5 * np.exp(-4.0 * (x - c2) ** 2)
    elif choice == 4:
        a = 1.0 + alpha * np.cos(2 * np.pi * x) * np.sin(np.pi * x)
    else:
        a = np.sin(np.pi * x) ** 2 + alpha * np.sin(2 * np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_init_hio(n, rng, n_hio_iters=500):
    """Use HIO to generate an initial point."""
    base = make_init_sin2(n, rng)
    a, _ = hio_optimize(base, n_iters=n_hio_iters, beta_start=0.9, beta_end=0.6, power=1.0)
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining HIO, Nesterov PG, LP refinement,

    """
    rng = np.random.default_rng(seed)

    best_seq = None

    solver = get_lp_solver()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_init_sin2(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: HIO diversification ===
    print(f"\n=== Phase 1: HIO Diversification n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        hio_inits = []
        t_per_hio = t_alloc / 5
        for h in range(5):
            if time.monotonic() > deadline - 60:
                break
            if h == 0:
                init = best_seq.copy()
            elif h == 1:
                init = make_init_sin2(n_best, rng)
            elif h == 2:
                init = make_init_raised_cos(n_best, rng)
            elif h == 3:
                init = make_init_asymmetric(n_best, 0.5, rng)
            else:
                init = make_init_asymmetric(n_best, 0.8, rng)
            t_hio = min(t_per_hio, deadline - time.monotonic() - 60)
            if t_hio < 2:
                break
            # Run HIO with time limit
            n_iters = int(t_hio * 80)
            a_hio, sc_hio = hio_optimize(init, n_iters=n_iters, beta_start=0.92, beta_end=0.5, power=1.0)
            hio_inits.append(a_hio)
            try_update(a_hio, f"hio_{h}")

        # Also try HIO with power compression
        if time.monotonic() < deadline - 30 and len(hio_inits) > 0:
            a_hio2, _ = hio_optimize(hio_inits[0], n_iters=1000, beta_start=0.85, beta_end=0.4, power=0.7)
            try_update(a_hio2, "hio_power")

    # === Phase 2: Nesterov PG on best candidate ===
    print(f"\n=== Phase 2: Nesterov PG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_pg(best_seq, t_alloc, T_start=1.5, T_end=400.0,

        try_update(a_alt, f"ng_n{alt_n}")
