    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for alt in ['CLARABEL', 'ECOS', 'SCS', 'PDLP']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000, use_sym=True):
    """Nesterov dual averaging LP refinement with symmetrization."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    if use_sym:
        a = symmetrize_norm(a)
    y = a.copy()
    best_a = a.copy()
    best_score = objective_val(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    eval_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        if use_sym:
            b_new = symmetrize_norm(b_new)

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        if use_sym:
            y = symmetrize_norm(y)

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        if use_sym:
            a = symmetrize_norm(a)

        if it % 50 == 0:
            eval_count += 1
            score_a = objective_val(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            if it % 300 == 0:
                if best_score < prev_best - 1e-6:
                    stall_count = 0
                else:
                    stall_count += 1
                prev_best = best_score
                if stall_count >= 10:
                    rng_local = np.random.default_rng(99999 + it)
                    noise = rng_local.random(n) * 0.05
                    y = y + noise
                    y = np.maximum(y, 0)
                    y = y / y.sum()
                    if use_sym:
                        y = symmetrize_norm(y)
                    stall_count = 0

    # Final check
    score_a = objective_val(a)
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.05, n_stages=None, maxiter_total=10000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter_total // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            if max_Tb > 500:
                return 1e15
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    result = softmax_from_z(z)
    result = symmetrize_norm(result)
    return result


def mirror_descent_refine(a0, time_limit, n_max_iters=500000):
    """Nesterov-accelerated mirror descent (multiplicative weights) for fast global moves."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    a = symmetrize_norm(a)
    z = a.copy()
    best_a = a.copy()
    best_score = objective_val(a)
    t0 = time.monotonic()
    no_improve = 0

    for t in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(z)
        k_star = int(np.argmax(b))

        # Subgradient at z
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            g[i] = z[k_star - i]

        # Adaptive step size
        eta = 2.0 / (np.sqrt(t + 1) + 10.0)

        # Mirror descent step
        log_mult = -eta * g
        log_mult -= log_mult.max()
        weights = np.exp(log_mult)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a_new = symmetrize_norm(a_new)

        # Nesterov extrapolation
        alpha = 2.0 / (t + 2)
        z = (1 - alpha) * z + alpha * a_new
        z = np.maximum(z, 0)
        sz = z.sum()
        if sz < 1e-15:
            z = a_new.copy()
        else:
            z = z / sz
        z = symmetrize_norm(z)

        a = a_new

        # Periodic evaluation
        if t % 100 == 0:
            score = objective_val(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                break

    # Final check
    score = objective_val(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def mirror_descent_multi_peak(a0, time_limit, n_max_iters=300000, n_peaks=5):
    """Mirror descent that simultaneously reduces top n_peaks of the convolution."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    a = symmetrize_norm(a)
    z = a.copy()
    best_a = a.copy()
    best_score = objective_val(a)
    t0 = time.monotonic()
    no_improve = 0

    for t in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(z)
        # Get top peaks (exclude edges)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        top_k = np.argsort(b_work)[::-1][:n_peaks]

        # Combined subgradient
        g = np.zeros(n)
        for k_star in top_k:
            k_star = int(k_star)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            for i in range(js, je + 1):
                g[i] += z[k_star - i]

        # Normalize gradient
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max

        eta = 2.0 / (np.sqrt(t + 1) + 20.0)

        log_mult = -eta * g
        log_mult -= log_mult.max()
        weights = np.exp(log_mult)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a_new = symmetrize_norm(a_new)

        alpha = 2.0 / (t + 2)
        z = (1 - alpha) * z + alpha * a_new
        z = np.maximum(z, 0)
        sz = z.sum()
        if sz < 1e-15:
            z = a_new.copy()
        else:
            z = z / sz
        z = symmetrize_norm(z)

        a = a_new

        if t % 200 == 0:
            score = objective_val(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                break

    score = objective_val(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Multi-peak shaving with pair adjustments and redistribution."""
    n = len(a0)
    a = symmetrize_norm(a0)
    best_a = a.copy()
    best_score = objective_val(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Focus on interior peaks
        b_work = b.copy()
        if len(b_work) > 6:
            b_work[:3] = -np.inf
            b_work[-3:] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        if peak_val < 1e-15:
            break

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:10]

        improved = False

        # Try reducing top contributors and redistributing
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if i == j or j < 0 or j >= n:
                continue
            if a[i] < 1e-14:
                continue

            for factor in [0.98, 0.96, 0.93, 0.90, 0.85]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] *= factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue

                # Redistribute to low-contrib positions (symmetrically)
                low_coords = np.argsort(contribs)[:n // 3]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                # Enforce symmetry
                low_mask = low_mask | low_mask[::-1]
                low_mask[i] = False
                low_mask[n - 1 - i] = False

                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-12
                    a_trial[low_mask] = dist + diff * dist / dist.sum()

                a_trial = symmetrize_norm(a_trial)
                score = objective_val(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Also try pair reduction (i, j) simultaneously
        if not improved:
            for ci in top_contrib[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.92, 0.88]:
                    a_trial = a.copy()
                    diff_i = a_trial[i] * (1 - f1)
                    diff_j = a_trial[j] * (1 - f1)
                    a_trial[i] *= f1
                    a_trial[j] *= f1
                    total_diff = diff_i + diff_j
                    if total_diff < 1e-16:
                        continue
                    # Redistribute
                    low_mask = np.zeros(n, dtype=bool)
                    low_coords = np.argsort(contribs)[:n // 4]
                    low_mask[low_coords] = True
                    low_mask[i] = False
                    low_mask[j] = False
                    low_mask = low_mask | low_mask[::-1]
                    low_mask[i] = False
                    low_mask[j] = False
                    low_mask[n - 1 - i] = False
                    low_mask[n - 1 - j] = False
                    if low_mask.sum() > 0:
                        dist = a_trial[low_mask] + 1e-12
                        a_trial[low_mask] = dist + total_diff * dist / dist.sum()
                    a_trial = symmetrize_norm(a_trial)
                    score = objective_val(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Try a random perturbation to escape
                rng_local = np.random.default_rng(step)
                a_pert = a + 0.02 * rng_local.standard_normal(n)
                a_pert = symmetrize_norm(a_pert)
                score_pert = objective_val(a_pert)
                if score_pert < best_score:
                    a = a_pert
                    best_score = score_pert
                    best_a = a.copy()
                    no_improve = 0
                else:
                    a = a_pert  # move anyway to explore
                    no_improve = 0
                if no_improve > 200:
                    break
        else:
            no_improve = 0

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.8 * (1 + 0.1 * np.cos(4 * np.pi * x))
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x)) * (1 + 0.2 * np.sin(3 * np.pi * x))
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.maximum(np.sin(np.pi * x), 0.1) ** 1.5
    else:
        a = np.ones(n) * (1 + 0.05 * np.sin(2 * np.pi * rng.random() * 5 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    a = symmetrize_norm(a)
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization and symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize_norm(f_interp)
    return f_interp

        g_max = g.max()
        if g_max < 1e-15:

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    1. Diverse structured initializations
    2. Mirror descent for fast global convergence
    3. Nesterov LP for structural optimization
    4. L-BFGS with log-sum-exp for smooth local refinement
    5. Peak shaving for final polish
    6. Multi-length exploration
    Exploits symmetry of the auto-convolution problem throughout.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = symmetrize_norm(parent)
            best_score = evaluate_safe(best_seq)
            print(f"After symmetrization: score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = symmetrize_norm(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Mirror descent for fast global convergence ===
    print(f"\n=== Phase 1: Mirror Descent n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_adaptive(best_seq, t_alloc, n_max_iters=500000)
        try_update(a_md, "mirror_desc")

    # === Phase 2: Nesterov LP (main structural optimization) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000, use_sym=True)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS with multi-stage annealing ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 100->0.05, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.05, maxiter_total=12000, n_stages=12)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Second Nesterov LP round ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000, use_sym=True)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: L-BFGS low-temperature polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 15->0.03, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=15.0, T_end=0.03, maxiter_total=8000, n_stages=10)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 6: Peak shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    alt_lengths = []
    for mult in [0.85, 0.9, 1.1, 1.15, 1.25]:
        alt_n = int(n_cur * mult)
        if 384 <= alt_n <= 4000 and alt_n != n_cur:
            alt_lengths.append(alt_n)
    for fixed_n in [640, 800, 1024, 1280, 1440]:
        if 384 <= fixed_n <= 4000 and fixed_n != n_cur:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.25, 180.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Mirror descent first (fast)
        t_md = t_total * 0.25
        a_alt, _ = mirror_descent_adaptive(init_alt, t_md, n_max_iters=100000)
        if evaluate_safe(a_alt) < best_score:
            best_seq = symmetrize_norm(a_alt)
            best_score = evaluate_safe(best_seq)
            print(f"    MD improved: {best_score:.6f}")

        # LP refinement
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_total * 0.5)
        if t_lp > 20:
            a_alt2, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=50000, use_sym=True)
            if not try_update(a_alt2, f"lp_n{alt_n}"):
                pass

        # L-BFGS
        rem = deadline - time.monotonic()
        if rem > 30 and len(best_seq) == alt_n:
            t_lb = min(rem * 0.4, 60.0)
            if t_lb > 15:
                a_alt3 = lbgfs_refine(best_seq, t_lb, T_start=20.0, T_end=0.05, maxiter_total=4000, n_stages=8)
                try_update(a_alt3, f"lbgfs_n{alt_n}")

        # Go back to original length if we found something at alt_n
        if best_seq is not None and len(best_seq) != n_cur:
            # Interpolate back
            pass  # We'll just use whatever length is best

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_pert_budget = min(rem * 0.15, 120.0)

        for scale in [0.03, 0.07, 0.12]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = symmetrize_norm(perturbed)

            t_sub = min(t_pert_budget / 3, rem * 0.7)
            t_lp_pert = t_sub * 0.6
            a_pert, _ = nesterov_lp_refine(perturbed, t_lp_pert, solver=solver, n_max_iters=20000, use_sym=True)
            try_update(a_pert, f"pert_lp_{scale}")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_md_pert = min(rem * 0.5, t_sub * 0.3)
                if t_md_pert > 10:
                    a_pert2, _ = mirror_descent_adaptive(perturbed, t_md_pert, n_max_iters=50000)
                    try_update(a_pert2, f"pert_md_{scale}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.02, maxiter_total=5000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.2, 40.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.3, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=10000, use_sym=True)
        try_update(a_lp_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_md_fin = min(rem * 0.3, 30.0)
        a_md_fin, _ = mirror_descent_multi_peak(best_seq, t_md_fin, n_max_iters=50000, n_peaks=3)
        try_update(a_md_fin, "final_md_mp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
