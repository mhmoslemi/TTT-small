```python
def lp_best_response(a, T):
    """Solve the LP: min M s.t. T@b <= M, sum(b)=1, b>=0"""
    n = len(a)
    from scipy.optimize import linprog
    
    # x = [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub @ x <= b_ub: T@b - M <= 0
    # A_ub = [T, -I]
    A_ub = np.hstack([T, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    # A_eq @ x = b_eq: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, M free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        b_opt = result.x[:n]
        M_opt = result.x[n]
        return b_opt, M_opt
    return None, None
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    c[-1] = 1.0

    A_ub = np.hstack([T, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return None, None

def nesterov_qp(a_init, w, n_iter=1000):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using Nesterov accelerated PG.
    W[j,k] = w[j+k] (Hankel matrix).
    """
    n = len(a_init)
    # FFT setup for Hankel matvec
    N = 1 << (2*n).bit_length()
    W_f = rfft(w, N)
    
    def hankel_matvec(x):
        X_f = rfft(x, N)
        corr = irfft(W_f * np.conj(X_f), N)
        return corr[:n]
    
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    
    y = a.copy()
    t_k = 1.0
    a_prev = a.copy()
    
    for k in range(n_iter):
        # Gradient: 2 * W * y
        grad = 2.0 * hankel_matvec(y)
        
        # Step size (for projected gradient on simplex)
        # Lipschitz constant of the gradient: 2 * ||W||_2 <= 2 * sum(w) * n (rough)
        # Use a conservative step
        L = 2.0 * w.sum() * 2  # rough upper bound
        step = 1.0 / L
        
        # Projected gradient step
        a_new = y - step * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = simplex_project(a_new)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        beta = (t_k - 1) / t_new
        y = a_new + beta * (a_new - a_prev)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / y_sum
        
        a_prev = a
        a = a_new
        t_k = t_new
    
    return a
```

Or even better: since this is a QP (possibly non-convex), Nesterov acceleration might not be applicable in the standard sense. Let me use a different approach.

**Alternative: Use the KKT conditions directly**

At the optimum of min a^T W a s.t. sum(a)=1, a>=0:
- Let A be the active set (where a[i] > 0)
- KKT: 2*(W*a)[i] + nu = 0 for i in A
       2*(W*a)[i] + nu >= 0 for i not in A
       sum(a) = 1, a >= 0

If we know the active set A, the first condition gives:
(W*a)[i] = -nu/2 for all i in A

Since (W*a)[i] = sum_{j in A} w[i+j]*a[j] (only active j contribute), this is a system of |A| linear equations in |A| unknowns (the a[j] for j in A) plus the unknown nu.

So: sum_{j in A} w[i+j]*a[j] = -nu/2 for all i in A
And: sum_{j in A} a[j] = 1

This is a linear system! I can solve it directly.

The system is:
[W_A  -1/2] [a_A]   [0]
[1^T   0  ] [nu ] = [1]

where W_A is the |A| x |A| submatrix of W with rows and columns indexed by A.

If this system gives a_A >= 0 and the KKT conditions are satisfied for inactive variables, then we have the exact solution.

This is the **active-set method** for QP! And it's exact (no iterations needed for a given active set).

The algorithm:
1. Start with an active set A (e.g., all variables)
2. Solve the linear system
3. Check if the solution is feasible (a_A >= 0, KKT for inactive)
4. If not, update the active set (add/remove variables)
5. Repeat until convergence

The linear system involves W_A which is a submatrix of the Hankel matrix. I can compute W_A @ a_A using the FFT-based approach (just restrict to the active set).

This is a much better approach than iterative gradient methods! Let me implement this.

```python
def active_set_qp(w, a_init, max_iter=1000):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 where W[j,k]=w[j+k]."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    a = a / s
    
    # Initial active set: all indices with a[i] > threshold
    threshold = 1e-10
    A = set(np.where(a > threshold)[0])
    
    for it in range(max_iter):
        if len(A) < 2:
            # Degenerate case
            A = set(range(n))
        
        A_list = sorted(A)
        m = len(A_list)
        
        # Build the linear system:
        # [W_A] [a_A]   [0]
        # [1^T] [nu ] = [1]
        
        # W_A[i,j] = w[A_list[i] + A_list[j]]
        # Compute W_A @ a_A for each row
        
        # Method 1: Build W_A explicitly (O(m^2))
        W_A = np.array([[w[A_list[i] + A_list[j]] for j in range(m)] for i in range(m)])
        
        # Solve [W_A, -0.5; 1^T, 0] [a_A; nu] = [0; 1]
        # Equivalent to: W_A @ a_A = 0.5 * nu * 1, and 1^T @ a_A = 1
        # From second: nu = 2 * (W_A @ a_A) / m ... no that's not right.
        # Actually: W_A @ a_A = -nu/2 * 1 (vector of ones), and sum(a_A) = 1
        # So: W_A @ a_A is a constant vector. Let's call it c * 1 where c = -nu/2.
        # Sum of all equations: 1^T W_A a_A = m * c → c = (1^T W_A a_A) / m
        # But this doesn't directly give us a_A.
        
        # Better: Just solve the (m+1) x (m+1) system directly.
        M_sys = np.zeros((m+1, m+1))
        M_sys[:m, :m] = W_A
        M_sys[:m, m] = -0.5
        M_sys[m, :m] = 1.0
        M_sys[m, m] = 0.0
        
        b_sys = np.zeros(m+1)
        b_sys[m] = 1.0
        
        try:
            sol = np.linalg.solve(M_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(M_sys, b_sys, rcond=None)[0]
            except:
                break
        
        a_A = sol[:m]
        nu = sol[m]
        
        # Check feasibility
        if np.any(a_A < -1e-10):
            # Remove the most negative variable
            worst = A_list[np.argmin(a_A)]
            A.remove(worst)
            # Reset a[worst] = 0 and redistribute
            a[worst] = 0
            continue
        
        # Check KKT for inactive variables
        inactive = [i for i in range(n) if i not in A]
        if inactive:
            # Compute gradient at inactive: 2*(W*a)[i] + nu >= 0
            # (W*a)[i] = sum_{j in A} w[i+j] * a[j]
            grad_inactive = np.array([sum(w[i+j]*a[j] for j in A_list) for i in inactive])
            if np.any(2*grad_inactive + nu < -1e-10):
                # Add the most violated variable
                worst = inactive[np.argmin(2*grad_inactive + nu)]
                A.add(worst)
                continue
        
        # Solution is optimal
        a_new = np.zeros(n)
        a_new[A_list] = np.maximum(a_A, 0)
        a_new = a_new / a_new.sum()
        return a_new
    
    # If we didn't converge, return best effort
    a_new = np.zeros(n)
    a_new[A_list] = np.maximum(a_A, 0) if 'a_A' in dir() else a
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new
```
