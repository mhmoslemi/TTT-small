        # More precisely: db_k/da_i = a_{k-i} (if 0<=k-i<n) + a_i (if 0<=k-i<n) ... 
        # Actually b_k = sum_j a_j a_{k-j}, so db_k/da_i = a_{k-i} + a_i (both terms when valid)
        # Wait: b_k = sum_j a_j * a_{k-j}. db_k/da_i: the term where j=i gives a_{k-i},
        # and the term where k-j=i, i.e., j=k-i, gives a_{k-i}. So if i != k-i (i.e., i != k/2),
        # db_k/da_i = 2 * a_{k-i} (if 0 <= k-i < n)
        # If i == k/2, db_k/da_i = a_i (only one term)
        # Actually more carefully: b_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j a_{k-j}
        # db_k/da_i = a_{k-i} [if 0<=k-i<n] + a_{k-i} [if 0<=k-i<n, from the other term where j=k-i]
        # Wait no. Let me be very careful.
        # b_k = sum_j a_j * a_{k-j} where sum is over j such that 0<=j<n AND 0<=k-j<n
        # = sum_j a_j * a_{k-j}
        # db_k/da_i: differentiate w.r.t. a_i
        # Term 1: when j = i, we get a_{k-i} (if 0 <= k-i < n)
        # Term 2: when k-j = i, i.e., j = k-i, we get a_{k-i} (if 0 <= k-i < n)
        # If i = k-i (i.e., k = 2i), these are the same term, so db_k/da_i = a_i
        # Otherwise db_k/da_i = 2 * a_{k-i} (if 0 <= k-i < n), else 0
        
        # For the subgradient, we just need: grad_a[i] = db_peak/da_i
        k = peak_idx
        grad_a = np.zeros(n)
        # For each i in [0, n-1]:
        # if 0 <= k-i < n: grad_a[i] += a[k-i]
        # if 0 <= k-i < n and i != k-i: grad_a[i] += a[k-i]  (second term)
        # Simplified: 
        valid_mask = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        if np.any(valid_mask):
            vals = a_y[k - np.arange(n)]
            vals[~valid_mask] = 0
            grad_a[valid_mask] += vals[valid_mask]
            # Second term (when k-i != i)
            not_self = (np.arange(n) != k - np.arange(n)) & valid_mask
            grad_a[not_self] += a_y[k - np.arange(n)[not_self]]

        # Map gradient from a-space to z-space (softmax):
        # d/dz F(softmax(z)) = softmax(z) * (grad_a - softmax(z) @ grad_a)
        a_z = a_y
        proj = np.dot(grad_a, a_z)
        grad_z = a_z * (grad_a - proj)

        # Nesterov update
        t_val = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2
        lr = lr_base / (1 + 0.01 * it)
        z_new = y - lr * grad_z
        theta = (t_val - 1) / (it + 2)
        y = z_new + theta * (z_new - z)
        z = z_new

        # Track best
        if it % 50 == 0:
            a_cur = z_max_to_a(z)
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
            a_y_cur = z_max_to_a(y)
            sc_y = compute_score(a_y_cur)
            if sc_y < best_score:
                best_score = sc_y
                best_a = a_y_cur.copy()

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NestSubgrad iter={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a


def multiplicative_weights(a0, time_limit, n_iters=5000, eta=0.5):
    """Multiplicative weights for the zero-sum game formulation.
    min_a max_k (Ta)_k s.t. a in simplex"""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    size = 2 * n - 1
    y = np.ones(size) / size  # dual variable (distribution over lags)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Running averages
    a_avg = a.copy()
    y_avg = y.copy()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Update a: a_i <- a_i * exp(-eta * g_i) where g = T^T y
        # g_i = sum_k y_k * db_k/da_i
        # Efficient computation: g = T^T y means g_i = sum_k y_k * (contrib of a_i to b_k)
        # b_k = sum_j a_j a_{k-j}, so the "linearization" at current a gives:
        # db_k/da_i contributes to g_i via: g_i = sum_k y_k * (a_{k-i} + a_{k-i} if i != k-i)
        # This is expensive. Instead, use the gradient of y^T (Ta) = (T^T y)^T a
        # (T^T y)_i = sum_k y_k * (db_k/da_i) -- but this is the gradient at the current point
        # For MW, we use: g_i = sum_k y_k * a_{k-i} (just one term, the "response" term)
        # Actually for the game, the payoff is y^T (Ta) = sum_k y_k * sum_j a_j a_{k-j}
        # The gradient w.r.t. a_i is: sum_k y_k * (a_{k-i} + a_{k-i} if k != 2i)
        # For simplicity and speed, let's use: g_i = 2 * sum_k y_k * a_{k-i} (approx, valid when k != 2i mostly)
        
        # Efficient: g = 2 * correlate(y, a, 'valid') approximately
        # correlate(y, a, 'valid')[i] = sum_j y[i+j] * a[j] ... hmm not quite
        # Let's use: g[i] = sum_k y[k] * a[k-i] for valid k-i
        # This is a cross-correlation
        # g[i] = sum_{k=i}^{i+size-1} y[k] * a[k-i] where k-i in [0,n-1]
        # = sum_{j=0}^{n-1} y[i+j] * a[j] (where i+j < size)
        # This is np.correlate(y, a, 'valid')!
        
        g = np.correlate(y, a, mode='valid')
        g = np.maximum(g, 1e-300)
        
        # Update a (multiplicative weights for minimizer)
        log_a = np.log(np.maximum(a, 1e-300)) - eta * g
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()
        
        # Update y (multiplicative weights for maximizer)
        b = fast_convolve(a)
        log_y = np.log(np.maximum(y, 1e-300)) + eta * b
        log_y -= log_y.max()
        y = np.exp(log_y)
        y = y / y.sum()
        
        # Running averages
        w = 1.0 / (it + 1)
        a_avg = (1 - w) * a_avg + w * a
        y_avg = (1 - w) * y_avg + w * y
        
        # Track best
        if it % 100 == 0:
            sc = compute_score(a_avg)
            if sc < best_score:
                best_score = sc
                best_a = a_avg.copy()
            sc_direct = compute_score(a)
            if sc_direct < best_score:
                best_score = sc_direct
                best_a = a.copy()

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a


def active_set_peak_shave(a0, time_limit, n_rounds=20000):
    """Targeted peak shaving: identify top peaks and reduce contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        
        # Find top peaks (excluding endpoints which are single terms)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_val = b_work.max()
        threshold = peak_val - 0.02 * peak_val
        peak_indices = np.where(b_work >= threshold)[0]
        
        if len(peak_indices) == 0:
            break

        improved = False

        # For each significant peak, find top contributors and reduce them
        for k in peak_indices[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[k - idx]
            top_i = idx[np.argmax(contribs)]
            top_j = k - top_i
            
            if top_i >= n or top_j >= n or top_i == top_j:
                # Single coordinate reduction
                for factor in [0.98, 0.96, 0.94, 0.92, 0.90]:
                    a_trial = a.copy()
                    diff = a_trial[top_i] * (1 - factor)
                    if diff < 1e-16:
                        continue
                    a_trial[top_i] -= diff
                    mask = np.ones(n, dtype=bool)
                    mask[top_i] = False
                    others = a_trial[mask].sum()
                    if others < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair reduction
                for f1 in [0.97, 0.95, 0.93]:
                    for f2 in [0.97, 0.95, 0.93]:
                        a_trial = a.copy()
                        a_trial[top_i] *= f1
                        a_trial[top_j] *= f2
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

            if improved:
                break

        if not improved:
            # Try small random perturbation
            if rnd % 200 == 0:
                rng = np.random.default_rng(rnd)
                a_trial = a.copy()
                n_pert = rng.integers(2, 8)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial[indices] *= (1.0 + rng.uniform(-0.03, 0.03, n_pert))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            if not improved:
                if rnd > 500:
                    break

        if rnd % 5000 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rnd} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  ** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Quick p-norm L-BFGS warmup
    2. Multiplicative weights (game-theoretic coarse optimization)
    3. High-p L-BFGS (fine optimization)
    4. Nesterov subgradient (direct max optimization)
    5. Peak shaving (targeted local improvements)
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(-1, 1, n_init)
        parent = np.exp(-x ** 2 * 3)
        parent = parent / parent.sum()
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    print(f"Starting optimization (budget={budget_s:.0f}s)")

    # === Phase 1: Quick p-norm L-BFGS warmup ===
    print(f"\n=== Phase 1: PN-LBFGS warmup n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_out = pnorm_lbfgs(best_state['seq'], t_alloc, p_start=10.0, p_end=300.0, n_stages=6)
        try_update(best_state, a_out, "pn_warmup")

    # === Phase 2: Multiplicative weights (game-theoretic) ===
    print(f"\n=== Phase 2: Multiplicative Weights (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_mw = multiplicative_weights(best_state['seq'], t_alloc, n_iters=20000, eta=0.3)
        try_update(best_state, a_mw, "mw")
        # Also try with different eta
        rem = deadline - time.monotonic()
        if rem > 20:
            a_mw2 = multiplicative_weights(best_state['seq'], min(rem*0.5, 100.0), n_iters=15000, eta=0.8)
            try_update(best_state, a_mw2, "mw_eta0.8")

    # === Phase 3: High-p L-BFGS (main refinement) ===
    print(f"\n=== Phase 3: High-p LBFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        a_hp = pnorm_lbfgs(best_state['seq'], t_alloc, p_start=50.0, p_end=3000.0, n_stages=10)
        try_update(best_state, a_hp, "pn_high")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_ns = nesterov_subgrad(best_state['seq'], t_alloc, lr_base=0.05)
        try_update(best_state, a_ns, "nesterov_subgrad")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 300.0)
    if t_alloc > 20:
        a_ps, _ = active_set_peak_shave(best_state['seq'], t_alloc, n_rounds=30000)
        try_update(best_state, a_ps, "peak_shave")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_state['score']:.6f}) ===")
    n_cur = len(best_state['seq'])
    candidate_lengths = []
    for delta in [-64, -32, -16, 16, 32, 64, 96, 128, 192]:
        alt_n = n_cur + delta
        if 400 <= alt_n <= 2500:
            candidate_lengths.append(alt_n)
    # Add some round numbers
    for rn in [512, 640, 768, 896, 1024, 1152, 1280, 1536]:
        if 400 <= rn <= 2500 and rn not in candidate_lengths:
            candidate_lengths.append(rn)
    
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_cur))
    
    n_explored = 0
    for alt_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.20, 200.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n} (budget={t_total:.0f}s)")

        init_alt = interpolate_seq(best_state['seq'], alt_n)
        
        # Quick p-norm refinement
        t_pn = t_total * 0.6
        a_alt = pnorm_lbfgs(init_alt, t_pn, p_start=30.0, p_end=1000.0, n_stages=6)
        improved = try_update(best_state, a_alt, f"pn_n{alt_n}")

        # If improved, do peak shaving
        if improved:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_ps = min(rem * 0.3, 80.0)
                a_alt2, _ = active_set_peak_shave(best_state['seq'], t_ps, n_rounds=10000)
                try_update(best_state, a_alt2, f"ps_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur2 = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur2)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.3, 80.0)
            a_pert = pnorm_lbfgs(perturbed, t_sub, p_start=50.0, p_end=1500.0, n_stages=5)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final high-p polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = pnorm_lbfgs(best_state['seq'], t_fin, p_start=300.0, p_end=5000.0, n_stages=5)
        try_update(best_state, a_fin, "final_pn")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.35, 60.0)
        a_ps_fin, _ = active_set_peak_shave(best_state['seq'], t_ps, n_rounds=15000)
        try_update(best_state, a_ps_fin, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_mw = min(rem * 0.5, 30.0)
        a_mw_fin = multiplicative_weights(best_state['seq'], t_mw, n_iters=5000, eta=0.5)
        try_update(best_state, a_mw_fin, "final_mw")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
