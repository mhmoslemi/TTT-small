`w_k = exp(-beta * (max_c - c_k))`

For large beta, this approximates the max. For moderate beta, it gives a smoother gradient that might help with convergence.

This is essentially a softmax/smoothing parameter, similar to the p-norm but applied to the peak weighting rather than the objective itself.

Actually, this is equivalent to optimizing `(sum_k exp(beta * c_k))^{1/beta}` which is a smoothed max. The gradient is:
`g_i = (1/beta) * (sum_k beta * exp(beta*c_k) * dc_k/da_i) / (sum_k exp(beta*c_k))`
`= sum_k (exp(beta*c_k) / sum_j exp(beta*c_j)) * dc_k/da_i`
`= sum_k w_k * dc_k/da_i`

where `w_k = exp(beta*c_k) / sum_j exp(beta*c_j)`.

This is exactly the softmax weighting I described. And the gradient of the smoothed objective is:
`d/d(a_i) [log(sum_k exp(beta*c_k))] / beta`

Which equals `sum_k w_k * dc_k/da_i = 2 * correlate(w, a, 'valid')`.

So my "soft peak subgradient" is actually the gradient of a smoothed max function! This is a well-known technique (LogSumExp smoothing).

The smoothing parameter beta controls the approximation quality:
- beta → ∞: exact max
- beta moderate: smooth approximation, easier to optimize
- beta small: very smooth, poor approximation

I'll use a schedule: start with small beta (easy optimization) and increase to large beta (accurate max).

This is essentially the same as the p-norm approach, but with a different parameterization. The p-norm uses `(sum c^p)^{1/p}` while I'm using `(1/beta) * log(sum exp(beta*c))`. Both converge to max as p→∞ or beta→∞.

The difference is in the gradient behavior:
- p-norm: gradient has a `c^{p-1}` factor, which can be very peaked
- LogSumExp: gradient has a softmax distribution, which is always smooth

The LogSumExp might be numerically more stable for large parameters.

and targeted peak-shaving local search with exploit of convolution structure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the score 2*n*max(conv)/(sum^2)."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def logsumexp_smoothed_obj(a, beta):
    """(1/beta)*log(sum_k exp(beta * c_k)) where c = convolve(a,a)."""
    c = fast_convolve(a)
    c_max = c.max()
    val = c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta
    n = len(a)
    s = a.sum()
    return 2.0 * n * val / (s * s)


def logsumexp_smoothed_grad(a, beta):
    """Gradient of 2n/(sum^2) * (1/beta)*log(sum exp(beta*c_k))."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    # Gradient of sum_k w_k * c_k w.r.t. a_i = 2 * correlate(w, a, 'valid')
    g_conv = 2.0 * np.correlate(w, a, mode='valid')
    s = a.sum()
    # d/da [2n * f(c) / s^2] = 2n/s^2 * g_conv - 2*2n*f(c)/s^3 * 1
    f_val = c_max + np.log(exp_bc.sum()) / beta
    grad = (2.0 * n / (s * s)) * g_conv - (4.0 * n * f_val / (s * s * s)) * np.ones(n)
    return grad


def logsumexp_lbfgs(a0, time_limit, beta_start=50.0, beta_end=5000.0, n_stages=8):
    """L-BFGS with LogSumExp smoothing schedule."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    betas = np.geomspace(beta_start, beta_end, n_stages)
    iters_per_stage = max(50, 6000 // n_stages)

    for stage_beta in betas:
        if time.monotonic() - t0 > time_limit:

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:

                method='L-BFGS-B',

                         'maxfun': iters_per_stage * 300, 'disp': False}

def structured_init_asymmetric(n, rng, variant=0):

    if variant == 0:
        # Concentrated at left
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x)
    elif variant == 1:
        # Concentrated at right
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (1 - x))
    elif variant == 2:
        # Skewed Gaussian (left heavy)
        x = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2)
    elif variant == 3:
        # Power law decay
        x = np.linspace(1e-6, 1, n)
        a = x ** (-1.5)
    elif variant == 4:
        # Triangle (peak at left)
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif variant == 5:
        # Bimodal asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.2) ** 2) + 0.5 * np.exp(-10 * (x - 0.7) ** 2)
    elif variant == 6:
        # Cosine at left only
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 3
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def symmetric_init(n, variant=0):
    """Create symmetric structured initializations."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2 / 0.08)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score'] - 1e-10:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LogSumExp L-BFGS, multiplicative descent,
    asymmetric initialization, local peak shaving, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = symmetric_init(n_init, variant=0)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: LogSumExp L-BFGS on current best ===
    print(f"\n=== Phase 1: LSE-LBFGS (n={n_best}, best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        a1 = logsumexp_lbfgs(best_state['seq'], t_alloc,
                             beta_start=50.0, beta_end=5000.0, n_stages=8)
        try_update(best_state, a1, "lse_lbfgs_1")

    # === Phase 2: Multi-start with asymmetric inits ===
    print(f"\n=== Phase 2: Multi-start (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_per_start = min(rem * 0.20 / 4, 200.0)
    if t_per_start > 15:
        for variant in range(4):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.15, t_per_start)
            if t_sub < 10:
                break
            init = structured_init_asymmetric(n_best, rng, variant=variant)
            a_m = logsumexp_lbfgs(init, t_sub,
                                  beta_start=30.0, beta_end=3000.0, n_stages=6)
            try_update(best_state, a_m, f"asym_v{variant}")

    # === Phase 3: Multiplicative descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a_md = multiplicative_descent(best_state['seq'], t_alloc,
                                      beta_start=50.0, beta_end=3000.0, n_stages=6)
        try_update(best_state, a_md, "mirror_descent")

    # === Phase 4: LSE-LBFGS round 2 (higher beta) ===
    print(f"\n=== Phase 4: LSE-LBFGS r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a2 = logsumexp_lbfgs(best_state['seq'], t_alloc,
                             beta_start=500.0, beta_end=10000.0, n_stages=6)
        try_update(best_state, a2, "lse_lbfgs_2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-128, -96, -64, -48, -32, 32, 48, 64, 96, 128]:
        alt_n = base + delta
        if 512 <= alt_n <= 2560:
            candidate_lengths.add(alt_n)

    n_explored = 0
    for alt_n in sorted(candidate_lengths, key=lambda x: abs(x - base)):
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 150.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        a_alt = logsumexp_lbfgs(init_alt, t_total,
                                beta_start=50.0, beta_end=3000.0, n_stages=6)
        try_update(best_state, a_alt, f"length_{alt_n}")

    # === Phase 6: Local peak shaving ===
    print(f"\n=== Phase 6: Local Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 10:
        a_ls = local_peak_shaving(best_state['seq'], t_alloc, n_steps=80000)
        try_update(best_state, a_ls, "local_shave_1")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 10:
                break
            a_p = logsumexp_lbfgs(perturbed, t_sub,
                                  beta_start=100.0, beta_end=5000.0, n_stages=5)
            try_update(best_state, a_p, f"pert_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 100.0)
        a_fin = logsumexp_lbfgs(best_state['seq'], t_fin,
                                beta_start=1000.0, beta_end=20000.0, n_stages=5)
        try_update(best_state, a_fin, "final_lse")

    # Final local search
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.30, 80.0)
        a_lr = local_peak_shaving(best_state['seq'], t_lr, n_steps=50000)
        try_update(best_state, a_lr, "final_local")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")

    return [float(max(0.0, min(1000.0, x))) for x in best_state['seq']]
