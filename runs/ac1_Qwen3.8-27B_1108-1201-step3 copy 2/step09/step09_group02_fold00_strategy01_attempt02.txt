def simplex_project(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if n == 0:
        return np.ones(1)/1 if 1 else np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - u[0]
    cond = u > (cssv - 1) / (np.arange(1, n+1))
    rho = np.max(np.nonzero(np.diff(cond)) + 1) if cond.any() else 0
    theta = (cssv[rho] - 1) / (rho + 1)
