"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-strategy approach:
- p-norm smoothed L-BFGS with adaptive schedule
- Subgradient descent on the max of the convolution
- LP-based Nesterov alternating optimization

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def normalize_nonneg(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def pnorm_smoothed_obj(z, p):
    """Objective: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def pnorm_smoothed_grad(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_lbfgs_refine(a0, time_limit, p_start=15.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return pnorm_smoothed_obj(z_val, stage_p)

        def jac(z_val):
            return pnorm_smoothed_grad(z_val, stage_p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

            break

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()

        c = fast_convolve(a)
        max_c = c.max()
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Compute subgradient of max in z-space
        active_tol = tol * max_c + 1e-20
        active = np.where(c >= max_c - active_tol)[0]

        # g_z[i] = 2n * sum_{k in active} dc_k/da_i * da_i/dz_i - (same) * sum(a)
        # dc_k/da_i = 2*a[k-i] if 0<=k-i<n
        # da_i/dz_j = a_i*(delta_ij - a_j)
        # So g_z_j = 2n * a_j * [sum_{k in active} 2*a[k-j] - sum_{k in active} sum_i a_i*2*a[k-i]]
        # = 2n * a_j * [2*sum_{k in active} a[k-j] - 2*sum_{k in active} (a * a)[k] * ... ]
        # Actually simpler: g_a[i] = 2 * sum_{k in active, 0<=k-i<n} a[k-i]
        # g_z = a * (g_a - dot(g_a, a))

        g_a = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                idx = np.arange(i_start, i_end + 1)
                g_a[idx] += 2.0 * a[k - idx]

        g_z = a * (g_a - np.dot(g_a, a))

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * it)
        # Nesterov-style momentum
        v = momentum * v - lr * g_z
        z = z + v
        z = z - z.max()

        # Periodic check
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0 and it > 0:
            if best_score < prev_stall_score - 1e-7:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 4:
                rng = np.random.default_rng(it + 42)
                noise_scale = 0.03 + 0.01 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def local_peak_shave(a0, time_limit, n_steps=30000):
    """Aggressive local refinement: coordinate descent on peak contributors."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        peak_k = int(np.argmax(c))

        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_coords = idx[np.argsort(contribs)[::-1][:12]]

        improved = False

        # Try reducing top contributors
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.92, 0.88, 0.83, 0.78, 0.70]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                a_trial[i] = max(0, a[i] - diff)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair adjustments
        if not improved:
            top_pairs = idx[np.argsort(contribs)[::-1][:8]]
            for ci in range(len(top_pairs)):
                i = top_pairs[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial /= st
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Random perturbation
        if not improved and step % 50 == 0:
            rng = np.random.default_rng(step * 7 + 3)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.04, 0.04))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial /= st
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved and step > 200:
            break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def diverse_init(n, rng, variant):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.exp(-3 * x * (1 - x))
    elif variant == 2:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 3:
        # Bimodal at edges
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        # Gaussian center
        xc = (x - 0.5) * 2
        a = np.exp(-xc ** 2 * 2)
    elif variant == 5:
        # Slightly anti-symmetric
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x * 2 + 0.5)
    elif variant == 6:
        # Power law
        a = x ** 0.3 + (1 - x) ** 0.3
    elif variant == 7:
        # Random smooth
        a = np.exp(-((x - rng.uniform(0.3, 0.7)) ** 2) / (2 * rng.uniform(0.05, 0.15) ** 2))
    elif variant == 8:
        # Double Gaussian
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization: p-norm L-BFGS, subgradient, LP-Nesterov,
    basin hopping, local peak shaving, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    def try_update(a, label):
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        parent = diverse_init(n_init, rng, 4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    solver = get_lp_solver()
    print(f"LP solver: {solver}, starting n={n_best}, score={best_state['score']:.6f}")

    # === Phase 1: p-norm L-BFGS on parent ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 1: p-norm L-BFGS (budget={min(rem*0.25, 400):.0f}s) ===")
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        a1 = pnorm_lbfgs_refine(best_state['seq'], t_alloc, p_start=10.0, p_end=500.0, n_stages=9)
        try_update(a1, "pnorm_p1")

    # === Phase 2: Subgradient optimization ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"\n=== Phase 2: Subgradient (budget={t_alloc:.0f}s) ===")
        a2, _ = subgradient_optimize(best_state['seq'], t_alloc, n_iters=80000, tol=1e-4)
        try_update(a2, "subgrad_p2")

    # === Phase 3: Nesterov LP ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"\n=== Phase 3: Nesterov LP (budget={t_alloc:.0f}s) ===")
        a3, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=80000)
        try_update(a3, "nesterov_p3")

    # === Phase 4: Local peak shaving ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"\n=== Phase 4: Peak Shaving (budget={t_alloc:.0f}s) ===")
        a4, _ = local_peak_shave(best_state['seq'], t_alloc, n_steps=50000)
        try_update(a4, "shave_p4")

    # === Phase 5: Basin hopping ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 5: Basin Hopping (budget={min(rem*0.20, 250):.0f}s) ===")
    t_total_basin = min(rem * 0.20, 250.0)
    n_basin = 5
    t_per_basin = t_total_basin / n_basin

    for bi in range(n_basin):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(t_per_basin, rem * 0.8)
        if t_sub < 20:
            break

        # Create perturbation
        noise_scale = 0.05 + 0.03 * bi
        perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_best)
        perturbed = np.maximum(perturbed, 0)
        perturbed /= perturbed.sum()

        # Quick optimize
        a_b = pnorm_lbfgs_refine(perturbed, t_sub * 0.6, p_start=30.0, p_end=300.0, n_stages=5)
        try_update(a_b, f"basin_{bi}")

        if rem - (deadline - time.monotonic()) - t_sub * 0.6 > 20:
            a_b2, _ = local_peak_shave(best_state['seq'], min(rem * 0.1, 30.0), n_steps=10000)
            try_update(a_b2, f"basin_shave_{bi}")

    # === Phase 6: Length exploration ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Length Exploration (budget={min(rem*0.20, 250):.0f}s) ===")
    candidate_lengths = []
    for dl in [-128, -64, 64, 128, 192, 256, -192]:
        alt_n = n_best + dl
        if 640 <= alt_n <= 2048:
            candidate_lengths.append(alt_n)
    for rn in [768, 896, 1024, 1152, 1280, 1408, 1536]:
        if 640 <= rn <= 2048 and abs(rn - n_best) > 64:
            candidate_lengths.append(rn)

    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.18, 150.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        t_pn = t_total * 0.55
        a_alt = pnorm_lbfgs_refine(init_alt, t_pn, p_start=20.0, p_end=300.0, n_stages=6)
        improved = try_update(a_alt, f"len_{alt_n}")

        if improved:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.12, 80.0)
                a_alt2, _ = nesterov_lp_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=20000)
                try_update(a_alt2, f"len_{alt_n}_lp")

    # === Phase 7: Fresh diverse starts ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 7: Diverse Starts (budget={min(rem*0.15, 150):.0f}s) ===")
    n_diverse = 3
    t_per_diverse = min(rem * 0.15, 150.0) / n_diverse

    for dv in range(n_diverse):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(t_per_diverse, rem * 0.7)
        if t_sub < 20:
            break

        variant = dv + 5  # variants 5, 6, 7
        a_div = diverse_init(n_best, rng, variant)
        # Add some randomness
        a_div += 0.05 * rng.standard_normal(n_best)
        a_div = np.maximum(a_div, 0)
        a_div /= a_div.sum()

        a_opt = pnorm_lbfgs_refine(a_div, t_sub, p_start=15.0, p_end=400.0, n_stages=7)
        try_update(a_opt, f"diverse_{variant}")

    # === Phase 8: Final polish ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    if rem > 20:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = pnorm_lbfgs_refine(best_state['seq'], t_fin, p_start=100.0, p_end=1000.0, n_stages=4)
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.3, 40.0)
        a_lr, _ = local_peak_shave(best_state['seq'], t_lr, n_steps=20000)
        try_update(a_lr, "final_shave")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.3, 30.0)
        a_sg, _ = subgradient_optimize(best_state['seq'], t_sg, n_iters=30000, tol=1e-5)
        try_update(a_sg, "final_subgrad")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
        print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")

    return [float(max(0.0, min(1000.0, x))) for x in best_state['seq']]
