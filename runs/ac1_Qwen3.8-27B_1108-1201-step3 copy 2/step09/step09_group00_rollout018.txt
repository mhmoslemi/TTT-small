    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _softmax(z):
    z2 = z - z.max()
    e = np.exp(z2)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def _lse_obj(z, n, T):
    a = _softmax(z)
    b = fast_convolve(a)
    Tb = T * b
    mx = Tb.max()
    e = np.exp(Tb - mx)
    return (2.0 * n / T) * (mx + np.log(e.sum()))


def _lse_grad(z, n, T):
    a = _softmax(z)
    b = fast_convolve(a)
    Tb = T * b
    mx = Tb.max()
    e = np.exp(Tb - mx)
    w = e / e.sum()
    ga = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        ga[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(ga, a)
    return a * (ga - S)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    z = _z_from_a(a0).copy()
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 3:
            break

        def _f(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return _lse_obj(z, n, stage_T)

        def _j(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return _lse_grad(z, n, stage_T)

        try:
            res = minimize(_f, z, jac=_j, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    return _softmax(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    if solver is None:
        solver = _get_lp_solver()
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score
            if stall_rounds >= 8:
                noise = rng.standard_normal(n) * (0.03 / np.sqrt(stall_rounds))
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall_rounds = 0

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=80000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    y = a.copy()
    t_k = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * 0.9998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k - idx][::-1]
        g /= len(active)
        z = np.maximum(y - lr * g, 0)
        sz = z.sum()
        if sz < 1e-15:
            z = a.copy()
        else:
            z = z / sz
        score_z = 2.0 * n * fast_convolve(z).max()
        if score_z < best_score - 1e-15:
            best_score = score_z
            best_a = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 20:
            lr *= 0.92
            no_improve = 0
        lr = max(1e-7, lr)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = z + (t_k - 1.0) / t_new * (z - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = z
        t_k = t_new
    return best_a, best_score


def peak_directed_descent(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.995, 0.99, 0.98, 0.96]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            a_trial = a.copy()
            for pos in top_positions[:4]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.995
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 300:
                    break
        if improved:
            stall = 0
    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.4 + 0.25 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-6 * (x - 0.5) ** 2) * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: FISTA subgradient, LP-alternating,
    LogSumExp L-BFGS, peak-directed descent, multi-length sweep,
    and structured perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = _get_lp_solver()
    print(f"LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1000
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Phase 1: FISTA subgradient on parent
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"\n=== Phase 1: FISTA subgradient n={len(best_seq)} budget={t1:.0f}s ===")
        a_sg, _ = subgradient_refine(best_seq, t1, n_steps=120000)
        try_update(a_sg, "fista1")

    # Phase 2: LP-alternating
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 450.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-alternating n={len(best_seq)} budget={t2:.0f}s ===")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver=solver, n_max_iters=200000)
        try_update(a_lp, "lp1")

    # Phase 3: LogSumExp L-BFGS
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        print(f"\n=== Phase 3: L-BFGS n={len(best_seq)} budget={t3:.0f}s ===")
        T_stages = [30.0, 15.0, 7.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        a_lb = lbgfs_refine(best_seq, t3, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "lbgfs1")

    # Phase 4: Peak-directed descent
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Peak descent n={len(best_seq)} budget={t4:.0f}s ===")
        a_pd, _ = peak_directed_descent(best_seq, t4, n_steps=60000)
        try_update(a_pd, "peak1")

    # Phase 5: Multi-length exploration
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"\n=== Phase 5: Multi-length exploration ===")
        n_base = len(best_seq)
        candidate_lengths = []
        for factor in [0.75, 0.85, 1.0, 1.1, 1.25, 1.4, 1.6]:
            nl = int(n_base * factor)
            if 400 <= nl <= 3000 and nl not in candidate_lengths:
                candidate_lengths.append(nl)
        candidate_lengths = sorted(set(candidate_lengths))
        t_each = rem * 0.7 / max(1, len(candidate_lengths))
        for nl in candidate_lengths:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            if nl == n_base:
                continue
            t_alloc = min(t_each, rem * 0.4)
            if t_alloc < 15:
                continue
            print(f"  n={nl} budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, nl)
            a_alt, _ = nesterov_lp_refine(init_alt, t_alloc * 0.5, solver=solver, n_max_iters=30000)
            try_update(a_alt, f"lp_n{nl}")
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                a_alt2 = lbgfs_refine(best_seq, min(rem * 0.3, 60.0),
                                      T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                                      maxiter_per_stage=2000)
                try_update(a_alt2, f"lbgfs_n{nl}")

    # Phase 6: Structured perturbation restarts
    rem = deadline - time.monotonic()
    if rem > 80:
        print(f"\n=== Phase 6: Perturbation restarts ===")
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        t_pert = min(rem * 0.3, 180.0)
        perturbations = []
        for freq in [1, 2, 3, 5]:
            for amp in [0.01, 0.03, 0.05]:
                perturbations.append(amp * np.cos(freq * x + rng.uniform(0, 2 * np.pi)))
        for i, pert in enumerate(perturbations):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_each_pert = min(t_pert / max(1, len(perturbations) - i), rem * 0.4)
            if t_each_pert < 10:
                continue
            perturbed = np.maximum(best_seq + pert, 0)
            perturbed = perturbed / perturbed.sum()
            a_p, _ = nesterov_lp_refine(perturbed, t_each_pert, solver=solver, n_max_iters=15000)
            try_update(a_p, f"pert_{i}")

    # Phase 7: Final polish
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final polish ===")
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin,
                                 T_stages=[3.0, 1.5, 0.7, 0.3, 0.15, 0.07, 0.03],
                                 maxiter_per_stage=4000)
            try_update(a_fin, "fin_lbgfs")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_pg, _ = peak_directed_descent(best_seq, min(rem * 0.4, 30.0), n_steps=30000)
            try_update(a_pg, "fin_peak")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sg2, _ = subgradient_refine(best_seq, min(rem * 0.5, 20.0), n_steps=30000)
            try_update(a_sg2, "fin_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
