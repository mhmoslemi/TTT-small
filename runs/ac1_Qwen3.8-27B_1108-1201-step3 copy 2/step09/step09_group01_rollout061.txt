def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""

    A = rfft(a, fft_size)

    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix T such that T @ b = conv(f, b)."""
    n = len(f)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = f
    return T


def solve_lp_max_sum(f, M):
    """Solve: maximize sum(b) subject to T_f @ b <= M, b >= 0.
    Returns the optimal b and its sum."""
    n = len(f)
    T = build_conv_matrix(f)
    c = -np.ones(n)
    A_ub = T
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = np.maximum(result.x, 0)
        return b, float(b.sum())
    return None, 0.0


def solve_lp_min_M(f):
    """Solve: minimize M subject to T_f @ b <= M * 1, sum(b) = 1, b >= 0.
    Returns optimal b and M."""
    n = len(f)
    T = build_conv_matrix(f)
    # Variables: b (n vars), M (1 var)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        b = np.maximum(result.x[:n], 0)
        s = b.sum()
        if s < 1e-15:
            return None, None
        b = b / s
        return b, float(result.x[n])
    return None, None


def lp_iteration_sota(f, max_backtracks=15):
    """One state-of-the-art LP iteration:
    1. Compute M = ||f*f||_inf
    2. Solve LP: max sum(b) s.t. T_f @ b <= M, b >= 0
    3. Normalize g = b/sum(b) * sum(f)
    4. Backtracking line search: f_new = (1-t)*f + t*g
    Returns (f_new, M_new, M_old)
    """
    n = len(f)
    s_f = f.sum()
    conv_f = fast_convolve(f)
    M = float(conv_f.max())

    b, b_sum = solve_lp_max_sum(f, M)
    if b is None or b_sum < 1e-15:
        return f, M, M

    g = b / b_sum * s_f

    # Backtracking line search
    best_t = 0.0
    best_M = M
    for bt in range(max_backtracks):
        t = 1.0 / (2.0 ** bt)
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        M_new = float(fast_convolve(f_new).max())
        if M_new <= M + 1e-14:
            return f_new, M_new, M
        if M_new < best_M:
            best_M = M_new
            best_t = t

    if best_t > 0:
        f_new = (1.0 - best_t) * f + best_t * g
        return np.maximum(f_new, 0), best_M, M
    return f, M, M

def fast_coord_descent(f0, time_limit, verbose=False, label=""):
    """Fast coordinate descent minimizing max of auto-convolution.
    Works on the normalized sequence (sum = 1).
    """
    n = len(f0)
    a = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_impt = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1].copy()
            a_arr = a.copy()

            # Try multiple deltas
            best_delta = 0.0
            best_new_M = M

            factors = [0.0, -0.5, -0.2, -0.05, 0.05, 0.2, 0.5, 1.0, 2.0]
            for fac in factors:
                delta = fac * a[i]
                new_val = a[i] + delta
                if new_val < -1e-18:
                    continue
                if new_val > 2.0:
                    continue

                new_aff = affected + 2.0 * delta * a_arr
                # Handle the diagonal term (k = 2i)
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_d = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx_d] += delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a_new = a[i] + best_delta
                a[i] = max(a_new, 0.0)

                # Update conv
                changes = 2.0 * best_delta * a_arr
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_d = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_d] = 2.0 * best_delta * (a_new - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_impt += 1
            if no_impt > 30:
                break
        else:
            no_impt = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): "
                  f"M={M:.8f}, score={best_score:.6f}")

    return best_a, best_score

                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_impt += 1
            if no_impt > 20:
                break
        else:
            no_impt = 0

    return best_a, best_score


def multiplicative_weight_update(f, b_response, eta):
    """One MWU step: f_new[i] proportional to f[i] * exp(-eta * b_response[k*-i]).
    Uses the peak position of conv(f, b_response) as the subgradient."""
    n = len(f)
    conv_fb = convolve_ab(f, b_response)
    k_star = int(np.argmax(conv_fb))

    # Subgradient: g[i] = b_response[k_star - i] for valid i
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = b_response[j]

    # MWU update
    log_f = np.log(f + 1e-30)
    log_f -= eta * g
    f_new = np.exp(log_f - log_f.max())
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def hybrid_mwu_lp(f0, time_limit, lp_every=50, eta=0.1, verbose=False, label=""):
    """Hybrid: MWU for speed, periodically corrected by LP best response."""
    n = len(f0)
    s0 = f0.sum()
    f = np.maximum(f0.copy(), 0)
    f = f / f.sum()  # Work with sum=1 for MWU

    best_f = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    t0 = time.monotonic()
    steps = 0

    while time.monotonic() - t0 < time_limit:
        steps += 1

        if steps % lp_every == 1:
            # Do an LP best response
            b, _ = solve_lp_min_M(f)
            if b is not None:
                f_mwu = multiplicative_weight_update(f, b, eta)
            else:
                f_mwu = f.copy()
        else:
            # Fast approximation: use the convolution peak as gradient
            conv_f = fast_convolve(f)
            k_star = int(np.argmax(conv_f))
            # Pseudo best response: concentrate mass where it doesn't contribute to peak
            g = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    g[i] = f[j]
            f_mwu = multiplicative_weight_update(f, g, eta)

        f = f_mwu

        # Evaluate periodically
        if steps % 100 == 0:
            sc = float(evaluate_sequence(f.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hybrid {label} step={steps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_f, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
