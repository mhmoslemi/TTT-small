    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    n = len(a)
    c = fast_convolve(a)

    return 2.0 * n * c.max() / (s * s)

    S = cp.sum()
    if S < 1e-300:
        return 0.0, np.zeros(n)
    obj = S ** (1.0 / p)
    # Gradient: d/d(a_i) S^(1/p) = (1/p) * S^(1/p - 1) * sum_k p * c_k^(p-1) * d(c_k)/d(a_i)

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum a)^2 for non-negative a."""
    n = len(a)
    c = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def compute_score_normalized(a):
    """Score assuming sum(a)=1, i.e., 2*n*max(conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def p_norm_and_grad(a, p):
    """Compute p-norm of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-300)
    if p > 50:
        log_cp = (p - 1.0) * np.log(c_safe)
        max_log = log_cp.max()
        exp_shifted = np.exp(log_cp - max_log)
        S = np.exp(max_log - (p - 1.0) / p * 0) # We need S^(1/p)
        # Actually let's compute more carefully
        # S = sum(c^p), we want S^(1/p)
        # log(S) = log(sum(exp(p*log(c)))) = max_log_c*p + log(sum(exp(p*log(c) - p*max_log_c)))
        pass
    # Direct computation for all p

    S = cp.sum()
    if S < 1e-300:

    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * (S ** (1.0 / p - 1.0)) * corr
    return obj, grad


def p_norm_smoothed_obj_z(z, n, p):
    """Objective in log-domain: 2*n*(sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()

    return 2.0 * n * S ** (1.0 / p)


def p_norm_grad_z(z, n, p):

def high_p_norm_and_grad(a, p):
    """Compute p-norm using log-sum-exp for numerical stability at high p."""
    n = len(a)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-300)
    log_c = np.log(c_safe)
    # log(S) = log(sum(c^p)) = log(sum(exp(p*log(c))))
    p_log_c = p * log_c
    max_p_log_c = p_log_c.max()
    log_S = max_p_log_c + np.log(np.exp(p_log_c - max_p_log_c).sum())
    obj = np.exp(log_S / p)
    # Gradient: w_k = c_k^(p-1), S^(1/p - 1)
    # In log space: log(w_k) = (p-1)*log(c_k), log(S^(1/p-1)) = (1/p-1)*log_S

    factor = np.exp(log_factor)
    corr = np.correlate(w_shifted, a, mode='valid')

    return obj, grad

def pnorm_lbfgs_optimize(a0, time_limit, p_schedule=None):
    """L-BFGS optimization with adaptive p-norm schedule in log domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    z = np.log(a)
    z = z - z.max()
    t0 = time.monotonic()

    if p_schedule is None:
        p_schedule = [20, 50, 100, 200, 500, 1000, 3000, 8000]

    best_a = a.copy()
    best_score = compute_score_normalized(a)

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Use high-p version for p > 100
        if p > 100:
            func = lambda z_val: p_norm_z_high(z_val, n, p)
            jac = lambda z_val: p_norm_grad_z_high(z_val, n, p)
        else:
            func = lambda z_val: p_norm_smoothed_obj_z(z_val, n, p)
            jac = lambda z_val: p_norm_grad_z(z_val, n, p)

        max_iter = max(200, int(remaining * 50))
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score_cur = compute_score_normalized(a_cur)
        if score_cur < best_score:
            best_score = score_cur
            best_a = a_cur.copy()

        elapsed = time.monotonic() - t0
        print(f"    p={p:<6d} ({elapsed:5.0f}s): score={score_cur:.8f} (best={best_score:.8f})")

def subgradient_optimize(a0, time_limit, n_iter=None):
    """Zoutendijk's generalized gradient method for max of convex functions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()

    if n_iter is None:
        n_iter = 50000

    # Zoutendijk's method: maintain a set of subgradients (active constraints)
    active_grads=[]
    step_size = 0.1
    stall_count = 0
    prev_best = best_score

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        k_star = int(c.argmax())
        grad = np.zeros(n)
        # Subgradient of C(k_star) w.r.t. a
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_star - idx]

        # Zoutendijk: pick the steepest descent direction from active gradients
        if len(active_grads) < 50:
            active_grads.append(grad.copy())

        # Find the direction that best decreases all active constraints
        best_dir = -grad
        best_dir_obj = np.inf
        for ag in active_grads[-20:]:
            dir_candidate = -ag
            # Evaluate: we want to minimize the max of active constraints
            # Use the one with steepest descent on current objective
            val = np.dot(dir_candidate, a)
            if val < best_dir_obj:
                best_dir_obj = val
                best_dir = dir_candidate

        # Line search
        alpha = step_size
        improved = False
        for trial in range(20):
            a_trial = a + alpha * best_dir
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / st
            score_trial = compute_score_normalized(a_trial)
            if score_trial < best_score - 1e-12:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                improved = True
                step_size = min(step_size * 1.1, 0.5)
                break
            alpha *= 0.7

        if not improved:
            step_size *= 0.8
            stall_count += 1
            if stall_count > 100:
                # Perturb and reset
                rng = np.random.default_rng(it)
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                active_grads = []
                step_size = 0.1
                stall_count = 0
        else:
            stall_count = 0

        if it % 5000 == 0 and it > 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={it:<6d} ({elapsed:5.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shaving_refine(a0, time_limit, n_steps=50000):
    """Aggressive peak shaving with multiple move types."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        peak_k = int(c.argmax())
        peak_val = c[peak_k]

        improved = False

        # Move 1: Reduce top contributor to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = idx[np.argmax(contribs)]

        for frac in [0.95, 0.90, 0.85, 0.80, 0.70]:
            reduce_amt = a[top_i] * (1 - frac)
            if reduce_amt < 1e-16:
                break
            a_trial = a.copy()
            a_trial[top_i] -= reduce_amt
            # Redistribute to random positions far from peak
            n_target = 5
            target_idx = rng.choice(n, size=n_target, replace=False)
            weights = rng.dirichlet(np.ones(n_target))
            for ti, w in zip(target_idx, weights):
                a_trial[ti] += reduce_amt * w
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = compute_score_normalized(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        # Move 2: Reduce multiple contributors
        if not improved and len(idx) > 3:
            top_3 = idx[np.argsort(contribs)[::-1][:3]]
            for frac in [0.90, 0.85, 0.80]:
                a_trial = a.copy()
                total_reduce = 0
                for ti in top_3:
                    reduce_amt = a_trial[ti] * (1 - frac)
                    a_trial[ti] -= reduce_amt
                    total_reduce += reduce_amt
                # Spread to neighbors of current small values
                small_idx = np.argsort(a_trial)[:10]
                for si in small_idx:
                    a_trial[si] += total_reduce / 10.0
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = compute_score_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        # Move 3: Block swap
        if not improved and step % 10 == 0:
            block_size = min(n // 10, 64)
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            if abs(i - j) < block_size:
                j = (j + n // 2) % (n - block_size)
            a_trial = a.copy()
            a_trial[i:i + block_size], a_trial[j:j + block_size] = \
                a[j:j + block_size].copy(), a[i:i + block_size].copy()
            score = compute_score_normalized(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True

        # Move 4: Smooth perturbation
        if not improved and step % 20 == 0:
            a_trial = a.copy()
            # Gaussian smoothing
            kernel = np.exp(-0.5 * (np.arange(-3, 4)) ** 2)
            kernel /= kernel.sum()
            a_trial = np.convolve(a_trial, kernel, mode='same')
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = compute_score_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Random restart perturbation
            if step % 500 == 499:
                a = a + 0.03 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            if step > 2000 and not any(
                compute_score_normalized(a + 0.01 * rng.standard_normal(n) / (a + 0.01 * rng.standard_normal(n)).sum() * (a+0.01*rng.standard_normal(n)).sum() ** 0.5)
                < best_score for _ in range(3)
            ):
                pass

        if not improved and step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step:<6d} ({elapsed:5.0f}s): score={best_score:.8f}")

        if step > 5000 and (step % 5000 == 0):
            # Try random perturbation + re-peak-shave
            a = a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def diverse_local_search(a0, time_limit):
    """Local search with diverse move generators."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    consecutive_fails = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        peak_k = int(c.argmax())
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False

        # Try multiple move strategies
        strategies = []

        # Strategy A: Reduce single top contributor, spread to random
        top_i = idx[np.argmax(contribs)]
        for frac in [0.92, 0.85, 0.75]:
            strategies.append(('reduce_spread', top_i, frac, None))

        # Strategy B: Reduce pair of contributors
        if len(idx) >= 2:
            top_2 = idx[np.argsort(contribs)[::-1][:2]]
            for frac in [0.90, 0.82]:
                strategies.append(('reduce_pair', top_2[0], frac, top_2[1]))

        # Strategy C: Mass transfer from peak contributor to anti-peak
        # Find position that contributes least to the peak
        if len(idx) > 2:
            anti_peak_i = idx[np.argmin(contribs)]
            if a[anti_peak_i] > 1e-10:
                strategies.append(('transfer', top_i, 0.5, anti_peak_i))

        # Strategy D: Random pair adjustment
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 != i2:
            for f1 in [0.9, 0.8, 0.7]:
                for f2 in [1.3, 1.5, 1.8]:
                    strategies.append(('random_pair', i1, f1, (i2, f2)))

        # Strategy E: Neighborhood smoothing around peak
        if peak_k > 2 and peak_k < n - 2:
            center_pos = peak_k // 2
            if 1 <= center_pos < n - 1:
                strategies.append(('smooth', center_pos, 0.3, None))

        rng.shuffle(strategies)

        for strategy in strategies:
            if time.monotonic() - t0 > time_limit:
                break
            a_trial = a.copy()
            stype = strategy[0]

            if stype == 'reduce_spread':
                i, frac, _ = strategy[1], strategy[2], strategy[3]
                reduce_amt = a_trial[i] * (1 - frac)
                if reduce_amt < 1e-15:
                    continue
                a_trial[i] -= reduce_amt
                # Spread to 10 random positions
                targets = rng.choice(n, size=min(10, n), replace=False)
                a_trial[targets] += reduce_amt / len(targets)

            elif stype == 'reduce_pair':
                i1, frac, i2 = strategy[1], strategy[2], strategy[3]
                r1 = a_trial[i1] * (1 - frac)
                r2 = a_trial[i2] * (1 - frac)
                a_trial[i1] -= r1
                a_trial[i2] -= r2
                targets = rng.choice(n, size=min(10, n), replace=False)
                a_trial[targets] += (r1 + r2) / len(targets)

            elif stype == 'transfer':
                i_from, frac, i_to = strategy[1], strategy[2], strategy[3]
                amt = a_trial[i_from] * frac
                a_trial[i_from] -= amt
                a_trial[i_to] += amt

            elif stype == 'random_pair':
                i1, f1, rest = strategy[1], strategy[2], strategy[3]
                i2, f2 = rest
                a_trial[i1] *= f1
                a_trial[i2] *= f2

            elif stype == 'smooth':
                center, width, _ = strategy[1], strategy[2], strategy[3]
                lo = max(0, center - 2)
                hi = min(n, center + 3)
                a_trial[lo:hi] *= 0.95
                total = a_trial[lo:hi].sum() * 0.05
                a_trial[lo:hi] += total / (hi - lo)

            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = compute_score_normalized(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                consecutive_fails = 0
                break

        if not improved:
            consecutive_fails += 1
            if consecutive_fails > 200:
                # Restart with perturbation
                pert_scale = min(0.1, 0.02 * (consecutive_fails // 100))
                a = a + pert_scale * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                consecutive_fails = 0

        if consecutive_fails > 2000:
            break

        if consecutive_fails > 0 and consecutive_fails % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DivLocal fails={consecutive_fails} ({elapsed:5.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        xc = (x - 0.5) * 2
        a = np.exp(-xc ** 2 * 3)
    elif variant == 5:
        # Asymmetric: heavy left tail
        a = np.exp(-3 * x) + 0.3 * np.exp(-3 * (1 - x))
    elif variant == 6:
        # W-shaped
        a = 1.0 - np.abs(np.sin(2 * np.pi * x)) ** 2
    elif variant == 7:
        # Two bumps
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    elif variant == 8:
        # Power law
        a = (x + 0.01) ** (-0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum a)^2.
    
    Strategy:
    1. Load parent or create diverse initializations
    2. High-p L-BFGS optimization (main workhorse)
    3. Subgradient method (Zoutendijk) for direct max optimization
    4. Peak shaving local search
    5. Diverse local search with multiple move types
    6. Length exploration
    7. Final polish with very high p
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4, rng=rng)
        best_state['seq'] = parent.copy()
        best_state['score'] = compute_score_normalized(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.8f}")

    n_best = len(best_state['seq'])

    def try_update(a_new, label):
        a_new = np.maximum(a_new, 0).astype(np.float64)
        s = a_new.sum()
        if s < 1e-15:
            return False
        a_new = a_new / s
        sc = compute_score_normalized(a_new)
        if sc < best_state['score'] - 1e-12:
            best_state['score'] = sc
            best_state['seq'] = a_new.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}] ({elapsed:.0f}s): {sc:.8f} (n={len(a_new)}) ***")
            return True
        return False

    # === Phase 1: Enhanced p-norm L-BFGS ===
    print(f"\n=== Phase 1: p-norm L-BFGS n={n_best} (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        p_sched = [30, 80, 200, 500, 1500, 5000, 15000]
        a_pn, _ = pnorm_lbfgs_optimize(best_state['seq'], t_alloc, p_schedule=p_sched)
        try_update(a_pn, "pnorm_p1")

    # === Phase 2: Subgradient (Zoutendijk) ===
    print(f"\n=== Phase 2: Subgradient ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_optimize(best_state['seq'], t_alloc)
        try_update(a_sg, "subgrad")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shaving_refine(best_state['seq'], t_alloc)
        try_update(a_ps, "peak_shave")

    # === Phase 4: Diverse local search ===
    print(f"\n=== Phase 4: Diverse Local Search ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_dl, _ = diverse_local_search(best_state['seq'], t_alloc)
        try_update(a_dl, "diverse_local")

    # === Phase 5: Second p-norm pass with higher p ===
    print(f"\n=== Phase 5: p-norm L-BFGS r2 ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        p_sched2 = [500, 2000, 10000, 30000]
        a_pn2, _ = pnorm_lbfgs_optimize(best_state['seq'], t_alloc, p_schedule=p_sched2)
        try_update(a_pn2, "pnorm_p2")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ({best_state['score']:.8f}) ===")
    candidate_lengths = []
    base = n_best
    for delta in [-128, -64, 64, 128, 256, -256]:
        alt_n = base + delta
        if 512 <= alt_n <= 2560:
            candidate_lengths.append(alt_n)
    # Round numbers
    for rn in [768, 896, 1024, 1152, 1280, 1536]:
        if 512 <= rn <= 2560 and rn not in candidate_lengths:
            candidate_lengths.append(rn)

    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 150.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)
        t_pn = t_total * 0.6
        a_alt, _ = pnorm_lbfgs_optimize(init_alt, t_pn, p_schedule=[50, 200, 1000, 5000])
        improved = try_update(a_alt, f"pn_n{alt_n}")

        if improved and rem > 20:
            a_alt2, _ = peak_shaving_refine(best_state['seq'], 20)
            try_update(a_alt2, f"ps_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.25, 60.0)
            a_pert, _ = pnorm_lbfgs_optimize(perturbed, t_sub, p_schedule=[100, 1000, 5000])
            try_update(a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ({best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        a_fin, _ = pnorm_lbfgs_optimize(best_state['seq'], t_fin, p_schedule=[5000, 20000])
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.4, 40.0)
        a_lr, _ = peak_shaving_refine(best_state['seq'], t_lr)
        try_update(a_lr, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_dl = min(rem * 0.5, 30.0)
        a_dl2, _ = diverse_local_search(best_state['seq'], t_dl)
        try_update(a_dl2, "final_diverse")

    # Final verification
    if best_state['seq'] is not None:
        final_score = evaluate_safe(best_state['seq'])
        if np.isfinite(final_score):
            print(f"\nFinal verified score: {final_score:.8f} (n={len(best_state['seq'])})")
        else:
            print(f"\nWARNING: Final score is not finite, using computed value {best_state['score']:.8f}")

    elapsed_total = time.monotonic() - t_start
    print(f"\nTOTAL TIME: {elapsed_total:.0f}s")
    print(f"FINAL SCORE: {best_state['score']:.8f}")

    return [float(x) for x in best_state['seq']]
