def fast_convolve(a):

def build_conv_matrix(a_fixed):

def compute_subgradient_topk(a, top_k_frac=0.05):
    """Compute subgradient focusing on top-K autocorrelation lags."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b, b
    n_active = max(1, int(top_k_frac * (2 * n - 1)))
    # Get top K lags
    top_indices = np.argsort(b)[::-1][:n_active]
    threshold = b[top_indices[-1]]
    active = np.where(b >= threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b, b

    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    momentum = 0.9
    v = np.zeros(n)
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, max_b, b = compute_subgradient_topk(a, top_k_frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        gap = best_score - f_lb
        if gap < 1e-12:
            break
        
        # Adaptive learning rate
        polyak_lr = gap / (g_norm * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)
        
        # Momentum
        v = momentum * v - alpha * g
        z_new = z + v
        a_new = softmax_from_z(z_new)
        
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            if step % 500 == 0:
                lr = min(lr * 1.1, 5.0)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.8
                no_improve = 0
            lr = max(lr, 0.005)
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK-SD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f} gap={gap:.6f}")
    
    return best_a, best_score


def bidirectional_lp_nesterov(a0, time_limit, solver=None, n_max_iters=200000):
    """Bidirectional LP alternating: optimize b|a AND a|b with Nesterov acceleration."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    b = a.copy()
    y_a = a.copy()
    y_b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    
    # Pre-build LP for both directions
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Optimize b given a (best response)
        b_new, M_b = solve_lp_best_response(y_a, solver)
        
        # Step 2: Optimize a given b (best response)
        a_new, M_a = solve_lp_best_response(y_b, solver)
        
        if b_new is None and a_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        
        if b_new is not None:
            y_b = (1 - alpha) * y_b + alpha * b_new
            y_b = np.maximum(y_b, 0)
            sy_b = y_b.sum()
            if sy_b > 1e-15:
                y_b = y_b / sy_b
        
        if a_new is not None:
            y_a = (1 - alpha) * y_a + alpha * a_new
            y_a = np.maximum(y_a, 0)
            sy_a = y_a.sum()
            if sy_a > 1e-15:
                y_a = y_a / sy_a
        
        # Evaluate both
        score_ya = 2.0 * n * fast_convolve(y_a).max()
        score_yb = 2.0 * n * fast_convolve(y_b).max()
        
        if score_ya < best_score:
            best_score = score_ya
            best_a = y_a.copy()
            stall_count = 0
        if score_yb < best_score:
            best_score = score_yb
            best_a = y_b.copy()
            stall_count = 0
        
        if it % 500 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                rng = np.random.default_rng(it * 7 + 13)
                noise = rng.random(n) * 0.03
                y_a = y_a + noise
                y_a = np.maximum(y_a, 0)
                y_a = y_a / y_a.sum()
                y_b = y_a.copy()
                stall_count = 0
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BiDirLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=10):
    """L-BFGS on log-sum-exp smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def obj_and_grad(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        f_val = (2.0 * n / T_val) * log_sum_exp
        
        # Gradient
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return f_val, grad_z
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, 5000 // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            val, _ = obj_and_grad(z_in, stage_T)
            return val
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            _, grad = obj_and_grad(z_in, stage_T)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_restarts=5000):
    """Coordinate descent with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    for step in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break
        
        # Pick a coordinate (cycle through with some randomness)
        j = (step + rng.integers(0, min(n, 50))) % n
        if a[j] < 1e-12:
            no_improve += 1
            continue
        
        # Try moving mass from j to a random target
        found_improvement = False
        for _ in range(10):
            i = rng.integers(n)
            if i == j:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amount = a[j] * frac
                if amount > a[j]:
                    continue
                a_trial = a.copy()
                a_trial[j] -= amount
                a_trial[i] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    found_improvement = True
                    break
            if found_improvement:
                break
        
        if not found_improvement:
            no_improve += 1
    
    return best_a, best_score


def spectral_init(n, rng):
    """Initialize using spectral approach: near-flat magnitude spectrum."""
    # Create a near-flat magnitude spectrum with random phases
    # In frequency domain, |A_k|^2 being flat means autocorrelation is a delta
    # We want non-negative time domain, so use real signals
    N = n
    # Start with all positive frequency components of equal magnitude
    mags = np.ones(N // 2 + 1)
    # Add small random perturbation to break symmetry
    mags = mags * (1 + 0.1 * rng.random(N // 2 + 1))
    
    # Random phases (but ensure real signal: conjugate symmetry)
    phases = 2 * np.pi * rng.random(N // 2 + 1)
    
    # Build spectrum
    freq = np.zeros(N, dtype=np.complex128)
    freq[:N//2 + 1] = mags * np.exp(1j * phases)
    if N % 2 == 0:
        freq[N//2 + 1:] = np.conj(freq[N//2-1:0:-1])
    else:
        freq[N//2 + 1:] = np.conj(freq[N//2-1:0:-1])
    
    # Inverse FFT
    a = np.real(ifft(freq))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_init(n, rng):
    """Gaussian-like initialization."""
    x = np.linspace(-1, 1, n)
    sigma = 0.3 + 0.4 * rng.random()
    a = np.exp(-x**2 / (2 * sigma**2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_init(n, rng):
    """Cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    p = 1 + rng.integers(2, 5)
    a = np.cos(x / 2) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fourier_init(n, rng):
    """Fourier series initialization."""
    x = np.linspace(0, 2 * np.pi, n)
    a = np.ones(n)
    n_terms = rng.integers(2, 6)
    for _ in range(n_terms):
        k = rng.integers(1, 10)
        amp = 0.1 + 0.3 * rng.random()
        phase = 2 * np.pi * rng.random()
        a += amp * np.cos(k * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving integral."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization for low-peak autocorrelation sequences."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    n_best = len(best_seq) if best_seq is not None else 960
    
    if parent is None:
        # Generate multiple diverse initializations and pick the best
        n_init = 960
        candidates = [
            spectral_init(n_init, rng),
            gaussian_init(n_init, rng),
            cosine_init(n_init, rng),
            fourier_init(n_init, rng),
            np.ones(n_init) / n_init,
        ]
        best_init = None
        best_init_score = float('inf')
        for c in candidates:
            sc = evaluate_safe(c)
            if sc < best_init_score:
                best_init_score = sc
                best_init = c
        best_seq = best_init
        best_score = best_init_score
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: Bidirectional LP with Nesterov ===
    print(f"\n=== Phase 1: Bidirectional LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 480.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = bidirectional_lp_nesterov(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "bidir_lp_1")
    
    # === Phase 2: Top-K subgradient descent ===
    print(f"\n=== Phase 2: TopK Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = topk_subgradient_descent(best_seq, t_alloc, top_k_frac=0.03, lr_init=1.5)
        try_update(a_sg, "topk_sg_1")
    
    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 330.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t_alloc, T_start=80.0, T_end=0.3, n_stages=8)
        try_update(a_lb, "lbgfs_1")
    
    # === Phase 4: Second LP pass with tighter focus ===
    print(f"\n=== Phase 4: LP Pass 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        a_lp2, _ = bidirectional_lp_nesterov(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp2, "bidir_lp_2")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_restarts=20000)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 1.75, 0.8, 2.0, 1.1, 1.4]:
        alt_n = int(n_best * mult)
        if 400 <= alt_n <= 3500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]
    
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interp improvement: {sc_alt:.6f}")
        
        # LP on new length
        t_lp = t_total * 0.55
        a_alt, _ = bidirectional_lp_nesterov(cur_best, t_lp, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")
        
        # Quick subgradient
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None:
            t_sg = min(rem * 0.25, 80.0)
            if t_sg > 15:
                a_sg_alt, _ = topk_subgradient_descent(best_seq, t_sg, top_k_frac=0.04, lr_init=1.0)
                try_update(a_sg_alt, f"sg_n{alt_n}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = lbgfs_logsumexp(best_seq, t_fin, T_start=10.0, T_end=0.2, n_stages=5)
        try_update(a_fin, "final_lbgfs")
    
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.20, 60.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd, n_restarts=15000)
        try_update(a_cd2, "final_cd")
    
    rem = deadline - time.monotonic()
    if rem > 25:
        t_lp_fin = min(rem * 0.35, 50.0)
        a_lp_fin, _ = bidirectional_lp_nesterov(best_seq, t_lp_fin, solver=solver, n_max_iters=20000)
        try_update(a_lp_fin, "final_lp")
    
    # Final: try a quick subgradient on the best
    rem = deadline - time.monotonic()
    if rem > 20:
        a_sg_fin, _ = topk_subgradient_descent(best_seq, min(rem - 5, 60.0), top_k_frac=0.02, lr_init=0.5)
        try_update(a_sg_fin, "final_sg")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
