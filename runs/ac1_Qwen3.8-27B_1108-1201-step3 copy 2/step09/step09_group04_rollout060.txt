"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast multiplicative weights (entropy mirror descent) with FFT-based gradients
2. LP-based Nesterov alternating optimization with scipy HiGHS solver
3. Log-sum-exp smoothed gradient refinement via L-BFGS
4. Multi-length exploration with interpolation
5. Aggressive local peak-shaving refinement
Key innovations: O(n log n) per-step MW via FFT correlation, adaptive temperature
annealing, periodic LP oracle corrections, and structured Fourier-domain search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Full linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, variant=0, rng=None):

    if rng is None:

    x = np.linspace(0, 1, n)

    elif variant == 3:

    elif variant == 4:

        a = np.exp(-xc ** 2 * 3)

def multiplicative_weights_refine(a0, time_limit, n_anneal_steps=None, eps_start=None, eps_end=None):
    """
    Fast multiplicative weights (entropy mirror descent) with log-sum-exp smoothing.
    Uses FFT for O(n log n) gradient computation per step.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30).copy()
    a = a / a.sum()

    if n_anneal_steps is None:
        n_anneal_steps = max(500, int(time_limit * 50))
    if eps_start is None:
        c0 = fast_convolve(a)
        eps_start = max(c0.max() / 10.0, 1e-6)
    if eps_end is None:
        eps_end = max(eps_start * 0.001, 1e-10)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    eps_values = np.geomspace(eps_start, eps_end, n_anneal_steps)
    steps_per_eps = 3

    for idx, eps in enumerate(eps_values):
        if time.monotonic() - t0 > time_limit:
            break

        for sub in range(steps_per_eps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_convolve(a)
            c_clipped = np.clip(c, 0, None)

            # Softmax weights
            log_p = c_clipped / eps
            log_p -= log_p.max()
            p = np.exp(log_p)
            p /= p.sum()

            # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (correlation)
            # This is 2 * correlate(p, a) in 'valid' mode
            # p has length 2n-1, a has length n
            # correlate(p, a, 'valid')[i] = sum_j p[i+j] * a[j] for j=0..n-1
            # = sum_j p[i+j] * a[j]
            # We want sum_k p[k] * a[k-i] for valid k-i
            # Let k = i + j: sum_j p[i+j] * a[j]  -- that's correlate(p, a) at offset i
            # Actually we want: sum_k p[k] * a[k-i] where 0 <= k-i < n
            # = sum_j p[i+j] * a[j] where j = k-i, 0 <= j < n
            # This is the 'valid' correlation of p with a, length = len(p) - n + 1 = n
            corr = np.correlate(p, a, mode='valid')
            g = 2.0 * corr

            # Step size: adaptive
            # eta ~ 1 / ||g||_inf for stability
            g_max = g.max()
            if g_max < 1e-15:
                eta = 1.0
            else:
                eta = min(2.0 / g_max, 50.0 / max(g.std(), 1e-15))

            # Multiplicative update
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_new = log_a - eta * g
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s

            # Check improvement
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

            a = a_new

        if idx % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW step {idx}/{n_anneal_steps} eps={eps:.2e} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lp_best_response(a_fixed):
    """Given fixed a (sum=1, a>=0), solve LP to find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1

    # Build T matrix: T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    # LP: min M s.t. T@b - M <= 0, sum(b) = 1, b >= 0
    # Variables: x = (M, b_0, ..., b_{n-1})
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0

    # A_ub @ x <= b_ub
    # (2n-1) rows: -M + T[k,:] @ b <= 0
    # 1 row: -sum(b) <= -1
    # 1 row: sum(b) <= 1
    A_ub = np.zeros((size + 2, n + 1))
    A_ub[:size, 0] = -1.0
    A_ub[:size, 1:] = T
    b_ub = np.zeros(size + 2)
    A_ub[size, 1:] = -1.0
    b_ub[size] = -1.0
    A_ub[size + 1, 1:] = 1.0
    b_ub[size + 1] = 1.0

    bounds = [(None, None)] + [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b_val = np.maximum(res.x[1:], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
            return b_val, float(res.x[0])
    except Exception:
        pass
    return None, None


def nesterov_lp_refine(a0, time_limit, n_max_iters=100000):
    """
    Nesterov alternating LP optimization.
    Alternates between solving best-response LPs for a and b.
    Uses running average for the Nesterov primal iterates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    b = a.copy()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Nesterov averages
    avg_a = a.copy()
    avg_b = a.copy()
    weight_sum = 0.0

    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response of a (given b)
        a_br, _ = lp_best_response(b)
        if a_br is None:
            a_br = a.copy()

        # Solve LP for best response of b (given a)
        b_br, _ = lp_best_response(a)
        if b_br is None:
            b_br = b.copy()

        # Nesterov step size
        alpha = 2.0 / (it + 2)

        # Update dual variables
        a = (1 - alpha) * a + alpha * a_br
        b = (1 - alpha) * b + alpha * b_br

        # Nesterov averaging
        avg_a = (avg_a * weight_sum + alpha * a_br) / (weight_sum + alpha)
        avg_b = (avg_b * weight_sum + alpha * b_br) / (weight_sum + alpha)
        weight_sum += alpha

        # Clip and renormalize
        for arr in [a, b, avg_a, avg_b]:
            arr[:] = np.maximum(arr, 0)
            s = arr.sum()
            if s > 1e-15:
                arr /= s

        # Evaluate
        sc_a = compute_score(a)
        sc_avg_a = compute_score(avg_a)
        sc_avg_b = compute_score(avg_b)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_avg_a < best_score:
            best_score = sc_avg_a
            best_a = avg_a.copy()
        if sc_avg_b < best_score:
            best_score = sc_avg_b
            best_a = avg_b.copy()

        # Stall detection and restart
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
                prev_best = best_score
            else:
                stall_count += 1
            if stall_count >= 5:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.05
                a = a + noise
                b = b + noise
                a = np.maximum(a, 0)
                b = np.maximum(b, 0)
                a /= a.sum()
                b /= b.sum()
                stall_count = 0
                prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def logsumexp_gradient_refine(a0, time_limit, eps_start=None, eps_end=None, n_total_steps=5000):
    """
    L-BFGS optimization of smoothed objective using log-sum-exp.
    Works in log-space for non-negativity constraint.
    """
    n = len(a0)

    if eps_start is None:
        c0 = fast_convolve(a0)
        eps_start = max(c0.max() / 5.0, 1e-5)
    if eps_end is None:
        eps_end = max(eps_start * 0.01, 1e-8)

    z = np.log(np.maximum(a0, 1e-30))
    z -= z.max()  # center for stability

    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()

    n_stages = 6
    eps_schedule = np.geomspace(eps_start, eps_end, n_stages)
    steps_per_stage = max(50, n_total_steps // n_stages)

    for stage_idx, eps in enumerate(eps_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_clipped = np.maximum(c, 0)
            val = eps * (np.log(np.sum(np.exp(c_clipped / eps))) - np.log(len(c)))
            return val

        def jac(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a = exp_z / exp_z.sum()
            c = fast_convolve(a)
            c_clipped = np.maximum(c, 1e-300)
            log_p = c_clipped / eps
            log_p -= log_p.max()
            p = np.exp(log_p)
            p /= p.sum()
            corr = np.correlate(p, a, mode='valid')
            g_a = 2.0 * corr
            # Gradient w.r.t. z: a * (g_a - dot(g_a, a))
            return a * (g_a - np.dot(g_a, a))

        iters = min(steps_per_stage, max(50, int(remaining / 0.5)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        elapsed = time.monotonic() - t0
        print(f"      LSE stage {stage_idx+1}/{n_stages} eps={eps:.2e} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def local_peak_shaving(a0, time_limit, n_steps=50000):
    """
    Aggressive local refinement: coordinate descent on peak contributors.
    Identifies the lag with maximum convolution value, finds top contributors,
    and redistributes mass away from them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        peak_k = c.argmax()
        peak_val = c[peak_k]

        # Find near-peak lags
        threshold = peak_val * 0.95
        near_peaks = np.where(c >= threshold)[0][:10]

        # Compute contribution scores
        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                contrib[i] += a[i] * a[k - i]

        # Try reducing top contributors
        top_idx = np.argsort(contrib)[::-1][:20]
        improved = False

        for i in top_idx:
            if a[i] < 1e-14:
                continue
            for factor in [0.90, 0.80, 0.70, 0.50]:
                a_trial = a.copy()
                removed = a_trial[i] * (1 - factor)
                a_trial[i] -= removed
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    removed = a[i]
                # Redistribute to non-peak contributors
                mask = np.ones(n, dtype=bool)
                mask[top_idx[:5]] = False  # don't add to other top contributors
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += removed * a_trial[mask] / others_sum
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial /= s
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try small random perturbation every 50 steps
            if step % 50 == 0 and step > 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(1, 8)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng.uniform(-0.1, 0.1))
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial /= s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True

        if not improved and step > 200:
            break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalRefine step={step} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fourier_perturb_search(a0, time_limit, n_trials=50):
    """
    Generate diverse candidates by perturbing in the Fourier domain,
    then refining each with a quick MW pass.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Generate random Fourier perturbation
        freq = rng.standard_normal((n, 2))
        freq[0, :] = 0  # don't perturb DC
        # Low-frequency bias
        freq_weight = np.exp(-np.linspace(0, 3, n) * rng.uniform(0.5, 2.0))
        freq *= freq_weight[:, None]

        z = fft(freq)
        a_trial = a0 + 0.05 * np.real(ifft(z))
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial /= s

        # Quick MW refinement
        mw_limit = min(5.0, (time_limit - (time.monotonic() - t0)) / max(1, n_trials - trial))
        a_ref, sc_ref = multiplicative_weights_refine(a_trial, mw_limit, n_anneal_steps=200,
                                                       eps_start=1e-4, eps_end=1e-7)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"      Fourier trial {trial}: new best={best_score:.6f}")

        # Also check direct
        sc_direct = compute_score(a_trial)
        if sc_direct < best_score:
            best_score = sc_direct
            best_a = a_trial.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines MW, LP-Nesterov, gradient refinement, and local search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_state_score = float('inf')
    best_state_a = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state_a = parent.copy()
            best_state_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4, rng=rng)
        best_state_a = parent.copy()
        best_state_score = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state_score:.6f}")

    def try_update(a, label):
        nonlocal best_state_score, best_state_a
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state_score:
            best_state_score = sc
            best_state_a = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    n_best = len(best_state_a)
    print(f"\nStarting optimization: n={n_best}, initial score={best_state_score:.6f}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Multiplicative Weights (fast, high iteration count) ===
    print(f"\n=== Phase 1: MW Refinement (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, sc_mw = multiplicative_weights_refine(best_state_a, t_alloc)
        try_update(a_mw, "MW_1")

    # === Phase 2: Nesterov LP (medium speed, high quality) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_state_a, t_alloc, n_max_iters=50000)
        try_update(a_lp, "NLP_1")

    # === Phase 3: Log-sum-exp gradient refinement ===
    print(f"\n=== Phase 3: LSE Gradient (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, sc_lse = logsumexp_gradient_refine(best_state_a, t_alloc)
        try_update(a_lse, "LSE_1")

    # === Phase 4: MW round 2 (from improved starting point) ===
    print(f"\n=== Phase 4: MW Round 2 (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw2, sc_mw2 = multiplicative_weights_refine(best_state_a, t_alloc,
                                                        eps_start=1e-5, eps_end=1e-9)
        try_update(a_mw2, "MW_2")

    # === Phase 5: Local peak shaving ===
    print(f"\n=== Phase 5: Local Peak Shaving (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ls, sc_ls = local_peak_shaving(best_state_a, t_alloc)
        try_update(a_ls, "Local_1")

    # === Phase 6: Nesterov LP round 2 ===
    print(f"\n=== Phase 6: NLP Round 2 (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = nesterov_lp_refine(best_state_a, t_alloc, n_max_iters=30000)
        try_update(a_lp2, "NLP_2")

    # === Phase 7: Fourier perturbation search ===
    print(f"\n=== Phase 7: Fourier Search (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ft, sc_ft = fourier_perturb_search(best_state_a, t_alloc, n_trials=40)
        try_update(a_ft, "Fourier_1")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Length Exploration (best={best_state_score:.6f}) ===")
    candidate_lengths = sorted(set([
        max(512, n_best - 128),
        max(512, n_best - 64),
        n_best + 64,
        n_best + 128,
        n_best + 256,
        1024,
        1280,
        1536,
    ]))
    candidate_lengths = [cn for cn in candidate_lengths if 384 <= cn <= 2560 and cn != n_best]
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 120.0)
        if t_total < 20:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state_a, alt_n)

        # Quick MW
        t_mw = t_total * 0.5
        a_alt, _ = multiplicative_weights_refine(init_alt, t_mw,
                                                   eps_start=1e-4, eps_end=1e-7,
                                                   n_anneal_steps=300)
        improved = try_update(a_alt, f"MW_n{alt_n}")

        # If improved, do LP
        if improved and rem > 30:
            t_lp = min(rem * 0.10, 60.0)
            a_alt2, _ = nesterov_lp_refine(best_state_a, t_lp, n_max_iters=10000)
            try_update(a_alt2, f"NLP_n{alt_n}")

        # Local refinement on best for this length
        rem = deadline - time.monotonic()
        if rem > 15:
            t_lr = min(rem * 0.05, 30.0)
            a_alt3, _ = local_peak_shaving(best_state_a, t_lr, n_steps=10000)
            try_update(a_alt3, f"Local_n{alt_n}")

    # === Phase 9: Final polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_state_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # High-precision LSE
        t_fin = min(rem * 0.3, 80.0)
        a_fin, _ = logsumexp_gradient_refine(best_state_a, t_fin,
                                              eps_start=1e-6, eps_end=1e-10)
        try_update(a_fin, "LSE_fin")

    rem = deadline - time.monotonic()
    if rem > 30:
        # Final MW with very low temperature
        t_mw_fin = min(rem * 0.3, 50.0)
        a_mw_fin, _ = multiplicative_weights_refine(best_state_a, t_mw_fin,
                                                     eps_start=1e-6, eps_end=1e-10,
                                                     n_anneal_steps=200)
        try_update(a_mw_fin, "MW_fin")

    rem = deadline - time.monotonic()
    if rem > 20:
        # Final local
        t_lr_fin = min(rem * 0.4, 40.0)
        a_lr_fin, _ = local_peak_shaving(best_state_a, t_lr_fin, n_steps=20000)
        try_update(a_lr_fin, "Local_fin")

    rem = deadline - time.monotonic()
    if rem > 15:
        # One more NLP
        t_lp_fin = min(rem * 0.5, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state_a, t_lp_fin, n_max_iters=10000)
        try_update(a_lp_fin, "NLP_fin")

    # Final verification
    if best_state_a is not None:
        fs = evaluate_safe(best_state_a)
        if np.isfinite(fs):
            best_state_score = fs

    print(f"\nFINAL SCORE: {best_state_score:.6f} (n={len(best_state_a)})")
    return [float(x) for x in best_state_a]
