        n_stages = 4

        x_curr = x0.copy() + 0.01 * np.random.default_rng(restart).standard_normal(n)
        
        for stage in range(n_stages):
            if time.monotonic() - t0 > time_limit:
                break
            T = temps[stage]
            stage_time = (time_limit - (time.monotonic() - t0)) / (n_stages - stage)
            
            def objective_and_grad(x):
                val, grad = lse_value_and_grad_x(x, T)
                return val, grad
            
            try:
                res = sp_minimize(objective_and_grad, x_curr, method='L-BFGS-B',
                                  jac=True, options={'maxiter': 300, 'maxfun': 1000,
                                                     'ftol': 1e-12, 'gtol': 1e-10,
                                                     'maxls': 50})
                x_curr = res.x
            except Exception:
                break
            
            a_curr = softmax_from_x(x_curr)
            sc = score_normalized(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                elapsed = time.monotonic() - t0
                print(f"      L-BFGS r={restart} s={stage} T={T:.1f} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with adaptive step size."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        a = project_simplex(a)
        
        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
        
        # Stall detection with perturbation
        if (it + 1) % 5000 == 0:
            rng = np.random.default_rng(it)
            y = y + 0.01 * rng.standard_normal(n)
            y = project_simplex(y)
    
    return best_a, best_score


def nested_lp_refine(a0, time_limit, solver=None):
    """
    Nested LP refinement: run LP refine, then re-initialize from best
    with perturbation and run again.
    """
    t0 = time.monotonic()
    current = a0.copy()
    best = a0.copy()
    best_score = score_normalized(a0)
    
    n_rounds = 3
    time_per_round = (time_limit - 10) / n_rounds
    
    for round_i in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_round, rem)
        if t_sub < 20:
            break
        a_res, sc = lp_refine(current, t_sub, solver=solver)
        if sc < best_score:
            best_score = sc
            best = a_res.copy()
        # Perturb for next round
        rng = np.random.default_rng(round_i + 100)
        current = best + 0.005 * rng.standard_normal(len(best))
        current = project_simplex(current)
    
    return best, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Multi-peak shaving with coordinate adjustments."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        if peak_val < 1e-15:
            break
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:8]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair modifications
            for ci in top_contrib[:4]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation to escape
            perturb_idx = rng.integers(0, n, size=5)
            a_trial = a.copy()
            for pi in perturb_idx:
                a_trial[pi] *= (0.9 + 0.2 * rng.random())
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step > 0 and step % 200 == 0:
                break
    
    return best_a, best_score


def make_init_sin(n, rng):
    """Initialize with 1 - sin(2*pi*j/n) shape (continuous optimal insight)."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 - np.sin(2 * np.pi * j / n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_init_cos_shift(n, rng):
    """Initialize with 1 + sin(2*pi*j/n + phase) shape."""
    j = np.arange(n, dtype=np.float64)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.8 * np.cos(2 * np.pi * j / n + phase)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_init_parabola(n, rng):
    """Parabolic (raised cosine) shape."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    a = 1.0 - (2 * x - 1) ** 2
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_init_flat_taper(n, taper_frac, rng):
    """Flat middle with tapered edges."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    a = np.ones(n)
    # Taper edges
    taper_n = int(taper_frac * n)
    for k in range(taper_n):
        fade = 0.5 * (1 + np.cos(np.pi * k / taper_n))
        a[k] = fade
        a[n - 1 - k] = fade
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_init_asymmetric(n, rng):
    """Asymmetric initialization."""
    choice = rng.integers(0, 4)
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    if choice == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + 1.0)
    elif choice == 1:
        a = 1.0 + 0.3 * (2 * x - 1) + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 2:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2) + 0.2
    else:
        a = 1.0 + 0.4 * np.sin(3 * np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def nesterov_pg_simplex(a0, time_limit, T_start=2.0, T_end=300.0,
                        lr_init=0.05, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient on simplex."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0
    
    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)
            
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            _, grad = lse_value_and_grad_a(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent or create diverse initializations
    2. L-BFGS on softmax-parametrized objective
    3. LP-based best-response refinement
    4. Nesterov projected gradient
    5. Peak shaving
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = make_init_sin(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: L-BFGS on softmax-parametrized objective ===
    print(f"\n=== Phase 1: L-BFGS softmax (best={best_score:.6f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, sc_lb = lbfqs_optimize(best_seq, t_alloc, T_start=3.0, T_end=150.0, n_restarts=3)
        try_update(a_lb, "l-bfgs_main")
    
    # === Phase 2: LP-based refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nested_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "lp_nested")
    
    # === Phase 3: Nesterov PG with different params ===
    print(f"\n=== Phase 3: Nesterov PG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_ng, _ = nesterov_pg_simplex(best_seq, t_alloc, T_start=5.0, T_end=400.0,
                                       lr_init=0.04, n_stages=10, steps_per_stage=2000)
        try_update(a_ng, "ng")
    
    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse Inits (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        inits = [
            (make_init_sin(n_best, rng), "sin"),
            (make_init_cos_shift(n_best, rng), "cos_shift"),
            (make_init_parabola(n_best, rng), "parabola"),
            (make_init_flat_taper(n_best, 0.15, rng), "flat_taper"),
            (make_init_asymmetric(n_best, rng), "asym"),
        ]
        for init_a, label in inits:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.2, 100.0)
            if t_sub < 30:
                break
            sc_init = score_normalized(init_a)
            print(f"  {label}: init_score={sc_init:.6f}, budget={t_sub:.0f}s")
            # Quick L-BFGS
            a_sub, sc_sub = lbfqs_optimize(init_a, t_sub * 0.6, T_start=3.0, T_end=100.0, n_restarts=2)
            try_update(a_sub, f"lbfgs_{label}")
            rem = deadline - time.monotonic()
            if rem > 30 and t_sub * 0.3 > 15:
                a_sub2, _ = lp_refine(a_sub, t_sub * 0.3, solver=solver, n_max_iters=10000)
                try_update(a_sub2, f"lp_{label}")
    
    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_ps, sc_ps = peak_shave_local(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")
    
    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for factor in [0.6, 0.7, 0.8, 1.1, 1.2, 1.3, 1.5, 2.0]:
        alt_n = int(n_best * factor)
        alt_n = max(256, min(4096, alt_n))
        if alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))
    
    for alt_n in alt_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.18, 120.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # L-BFGS then LP
        a_alt, _ = lbfqs_optimize(init_alt, t_total * 0.5, T_start=3.0, T_end=100.0, n_restarts=2)
        try_update(a_alt, f"lbfgs_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 40.0), solver=solver, n_max_iters=8000)
            try_update(a_alt2, f"lp_n{alt_n}")
    
    # === Phase 7: Second round of LP on best ===
    print(f"\n=== Phase 7: LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 100.0)
        a_fin, _ = lp_refine(best_seq, t_fin, solver=solver, n_max_iters=20000)
        try_update(a_fin, "final_lp")
    
    # === Phase 8: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.15, 30.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=10000)
        try_update(a_ps2, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
