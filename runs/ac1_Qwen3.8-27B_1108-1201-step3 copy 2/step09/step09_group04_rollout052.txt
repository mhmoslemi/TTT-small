"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Adaptive p-norm L-BFGS with FFT-based gradients
2. Peak-targeting local search (direct peak reduction)
3. Simulated annealing for escape from local minima
4. Multi-length exploration with interpolation
Key innovation: "peak targeting" that directly reduces the top convolution
peaks via targeted coordinate adjustments, combined with an adaptive p-schedule
that focuses optimization on the critical (highest) peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_obj_and_grad(z, p):
    """Objective and gradient for p-norm smoothing with softmax parameterization.
    Objective: 2n * (sum c_k^p)^(1/p) where a = softmax(z) (sum a = 1).
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        val = 2.0 * n
        return val, np.zeros_like(z)
    val = 2.0 * n * S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    grad_z = a * (g_a - np.dot(g_a, a))
    return val, grad_z


def refine_pnorm(a0, time_limit, p_specs):
    """
    L-BFGS refinement with specified p schedule.
    p_specs: list of (p_start, p_end, n_stages, max_iters)
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for (p_start, p_end, n_stages, max_iters) in p_specs:
        temps = np.geomspace(p_start, p_end, n_stages)
        for stage_p in temps:
            if time.monotonic() - t0 > time_limit:
                return z
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 3:
                return z
            iters = min(max_iters, max(200, int(remaining * 50)))

            def func(z_val):
                return p_norm_obj_and_grad(z_val, stage_p)[0]

            def jac(z_val):
                return p_norm_obj_and_grad(z_val, stage_p)[1]

            try:
                res = minimize(func, z, jac=jac, method='L-BFGS-B',
                               options={'maxiter': iters, 'ftol': 1e-20,
                                        'gtol': 1e-16, 'maxfun': iters * 100,
                                        'disp': False})
                z = res.x
            except Exception:
                pass

    return z


def z_to_a(z):
    """Convert z (log-domain) to normalized a."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def score(a):
    """Compute the objective value 2n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def peak_targeting(a0, time_limit, n_target=3, trial_factors=None):
    """
    Directly target the top convolution peaks by adjusting coordinates
    that contribute most to each peak.
    """
    if trial_factors is None:
        trial_factors = [0.98, 0.95, 0.90, 0.85, 0.80, 0.70]
    
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_convolve(a)
        # Find top peaks (exclude the very first and last which are small)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        peak_indices = np.argsort(c_work)[::-1][:n_target]
        peak_val = c[peak_indices[0]]
        
        # Identify coordinates contributing to peaks
        contrib = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib[j] += a[j] * a[k - j]
        
        # Try reducing top contributors
        top_coords = np.argsort(contrib)[::-1][:20]
        improved = False
        
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for f in trial_factors:
                delta = a[i] * (1 - f)
                if delta < 1e-16:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                if a_new[i] < 0:
                    a_new[i] = 0
                    delta = a[i]
                # Redistribute
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                other_sum = a_new[mask].sum()
                if other_sum < 1e-15:
                    continue
                a_new[mask] += delta * a_new[mask] / other_sum
                sc = score(a_new)
                if sc < best_score - 1e-15:
                    a = a_new
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair adjustments
            k = peak_indices[0]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_pairs = np.argsort(contribs)[::-1][:8]
            
            for ci in top_pairs:
                i = int(idx[ci])
                j = int(k - i)
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_new = a.copy()
                        a_new[i] *= f1
                        a_new[j] *= f2
                        a_new = np.maximum(a_new, 0)
                        st = a_new.sum()
                        if st < 0.01:
                            continue
                        a_new /= st
                        sc = score(a_new)
                        if sc < best_score - 1e-15:
                            a = a_new
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation every 200 steps
            if step % 200 == 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(1, 6)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_new = a.copy()
                for i in indices:
                    a_new[i] *= (1.0 + rng.uniform(-0.08, 0.08))
                a_new = np.maximum(a_new, 0)
                st = a_new.sum()
                if st > 0.01:
                    a_new /= st
                    sc = score(a_new)
                    if sc < best_score - 1e-15:
                        a = a_new
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            
            if not improved and step % 500 == 0:
                break
    
    return best_a, best_score


def simulated_annealing(a0, time_limit, T0=0.1, T_end=0.001, n_iter=100000):
    """
    Simulated annealing: randomly perturb coordinates and accept with
    probability exp(-Δ/T).
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = score(a)
    current_score = best_score
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, it / (n_iter * 0.8))
        T = T0 * (T_end / T0) ** progress
        
        # Random perturbation
        n_pert = rng.integers(1, 4)
        indices = rng.choice(n, size=n_pert, replace=False)
        a_new = a.copy()
        for i in indices:
            a_new[i] *= (1.0 + rng.uniform(-0.1, 0.1))
        a_new = np.maximum(a_new, 0)
        st = a_new.sum()
        if st < 0.01:
            continue
        a_new /= st
        
        sc_new = score(a_new)
        delta = sc_new - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
    
    return best_a, best_score


def subgradient_nesterov(a0, time_limit, n_steps=500000):
    """
    Nesterov accelerated subgradient method with adaptive step size.
    Uses the true subgradient at the argmax of the convolution.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    a /= s
    y = a.copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    # Track running average for Polyak step size
    f_star_est = best_score  # estimate of optimal value
    G2 = 0.0  # accumulated gradient squared norm
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        c = fast_convolve(y)
        k_star = int(np.argmax(c))
        M = c[k_star]
        S = y.sum()
        
        # grad c_k/da_i = 2 * a[k-i] if 0 <= k-i < n
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * y[k_star - j]
        
        # Full gradient of f = 2n*M/S^2:
        # df/da = 2n * (S * grad_c - 2*M*1) / S^3
        f_val = 2.0 * n * M / (S * S)
        grad_f = (2.0 * n / (S ** 3)) * (S * grad - 2.0 * M * np.ones(n))
        
        # Project onto simplex direction
        g_norm_sq = np.dot(grad_f, grad_f)
        G2 += g_norm_sq
        
        # Polyak step size
        f_diff = max(f_val - f_star_est, 1e-10)
        eta = f_diff * S / (G2 + 1e-10)
        eta = min(eta, 1.0)  # cap step size
        
        # Nesterov momentum
        alpha = 2.0 / (it + 2.0)
        
        # Update
        a_new = y - eta * grad_f
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa > 1e-15:
            a_new /= sa
        else:
            a_new = y.copy()
        
        y = (1 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        
        # Evaluate
        sc = score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            f_star_est = best_score
    
    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    f_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False
