"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-strategy approach:
diverse initialization, adaptive p-norm L-BFGS, multiplicative subgradient descent,
LP-based alternating optimization, and coordinate-level local refinement.
Key innovations: efficient FFT-based gradients, multiplicative weight updates for
the exact max subgradient, and adaptive scheduling across optimization phases.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective score for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def p_norm_value_and_grad(a, p):
    """
    Compute p-norm of convolve(a,a) and its gradient w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_pow = c_pos ** p
    S = p_pow.sum()
    if S < 1e-300:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient: g[i] = 2 * factor * sum_k w[k] * a[k-i] (valid range)
    # This is 2 * factor * correlate(w, a, 'valid')
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * factor * corr
    return val, grad


def lbfgs_objective(z, p):
    """Objective in log-space: 2n * (sum c^p)^(1/p) where a = softmax(z)."""
    zmax = z.max()
    exp_z = np.exp(z - zmax)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (c_pos ** p).sum() ** (1.0 / p)
    return 2.0 * n * val


def lbfgs_gradient(z, p):
    """Gradient of log-space objective w.r.t. z."""
    zmax = z.max()
    exp_z = np.exp(z - zmax)
    a = exp_z / exp_z.sum()
    n = len(a)
    val, grad_a = p_norm_value_and_grad(a, p)
    # Full gradient of 2n * p_norm w.r.t. a
    g_a = 2.0 * n * grad_a / val * p_norm_value_and_grad(a, 1)[0] if val > 0 else np.zeros_like(a)
    # Actually, let's recompute properly
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_pow = c_pos ** p
    S = p_pow.sum()
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * n * 2.0 * factor * corr
    # Gradient w.r.t. z: a * (g_a - sum(g_a * a))
    return a * (g_a - np.dot(g_a, a))


def lbfgs_refine(a0, time_limit, p_start=15.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm in log-space parameterization."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        p_cur = p_start * (p_end / p_start) ** (stage / max(1, n_stages - 1))
        iters = max(50, int(6000 / n_stages))

        def fun(zv):
            return lbfgs_objective(zv, p_cur)

        def jac(zv):
            return lbfgs_gradient(zv, p_cur)

        try:
            res = minimize(
                fun, z, jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        zmax = z.max()
        a_cur = np.exp(z - zmax)
        a_cur = a_cur / a_cur.sum()
        sc = score_of(a_cur)
        if stage % 2 == 0:
            print(f"    L-BFGS p={p_cur:.1f} ({time.monotonic()-t0:.0f}s): score={sc:.6f}")

    zmax = z.max()
    a = np.exp(z - zmax)
    a = a / a.sum()
    return a


def subgradient_multiplicative(a0, time_limit, n_iters_max=200000):
    """
    Multiplicative exponentiated gradient method for minimizing max_k c[k].
    Uses exact subgradient of the max function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    eta = 1.0 / (50.0 * n ** 0.5)

    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        max_c = c.max()
        # Subgradient: for each i, g[i] = 2 * a[k*-i] where k* is argmax
        # Use soft version: weight by c[k]^p for large p, then normalize
        # For exact subgradient, use uniform over argmax set
        threshold = max_c - 1e-10
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # Compute subgradient as average over peaks
        g = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= len(peak_indices)

        # Multiplicative update
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        if it % 5000 == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Adaptive eta
        if it % 10000 == 0 and it > 0:
            eta = eta * 0.8  # Shrink step size

    sc = score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                print(f"  LP solver: {solver_name}")
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    print(f"  LP solver (fallback): {solver_name}")
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build T @ b = convolve(a_fixed, b)
    # T is (2n-1) x n, T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0
    # We build it as a matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov-style LP alternating with adaptive step size and stall detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    lp_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        lp_count += 1
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 1e-300)
        y = y / y.sum()

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        sc_a = score_of(a)
        sc_y = score_of(y)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0

        if it % 2000 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
                prev_best = best_score
            else:
                stall_count += 1

            if stall_count >= 3:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * (0.03 + 0.02 * stall_count)
                y = y + noise
                y = np.maximum(y, 1e-300)
                y = y / y.sum()
                stall_count = 0
                prev_best = best_score

            if it % 5000 == 0:
                print(f"    NesterovLP iter={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, LPs={lp_count}")

    return best_a, best_score


def local_refine(a0, time_limit, n_steps=50000):
    """
    Coordinate descent with peak-shaving. Focuses on reducing the dominant peaks
    by adjusting the coordinates that contribute most to them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find the peak (excluding endpoints which are squares)
        c_inner = c.copy()
        c_inner[0] = -1
        c_inner[-1] = -1
        peak_k = np.argmax(c_inner)
        peak_val = c_inner[peak_k]

        # Find indices contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-30:
            break

        # Try reducing top contributors
        top_frac = np.sort(contribs / total_contrib)[::-1]
        n_top = max(3, int(0.1 * len(idx)))
        top_local = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for ci in top_local:
            i = idx[ci]
            j = peak_k - i
            if j < 0 or j >= n or i == j:
                continue
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue
            for f in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try single coordinate
            for ci in top_local:
                i = idx[ci]
                if a[i] < 1e-15:
                    continue
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    diff = a[i] * (1 - f)
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial[mask] += diff * others / others.sum()
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Random perturbation to escape local minima
        if not improved and step % 200 == 0:
            rng = np.random.default_rng(step)
            n_pert = rng.integers(1, 5)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 100:
            # Try to break by random coordinate swap
            rng = np.random.default_rng(step + 1)
            i1, i2 = rng.choice(n, size=2, replace=False)
            delta = 0.01 * a.sum()
            a_trial = a.copy()
            a_trial[i1] = max(0, a_trial[i1] - delta)
            a_trial[i2] += delta
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

        if step % 10000 == 0 and step > 0:
            print(f"    LocalRefine step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        a = np.cos(np.pi * x) ** 2  # Bimodal at ends
    elif variant == 4:
        x2 = np.linspace(-1, 1, n)
        a = np.exp(-x2 ** 2 * 3)
    elif variant == 5:
        # Flat with slight bump in middle
        a = np.ones(n) * 0.8
        a[n//4:3*n//4] += 0.4
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return np.inf
        return float(evaluate_sequence(a_list))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: diverse init, p-norm L-BFGS, multiplicative
    subgradient, LP alternating, local refinement, length exploration.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(100.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 20.0  # Safety margin

    # Solve for LP solver early
    solver = get_lp_solver()

    best_score = np.inf
    best_seq = None

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_score = parent_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_score = evaluate_safe(parent)
        best_seq = parent.copy()
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 0: Diverse initialization exploration ===
    print(f"\n=== Phase 0: Diverse Init (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for variant in [0, 1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            a_var = structured_init(n_best, variant)
            # Quick L-BFGS
            t_sub = min(rem * 0.12, 30.0)
            a_opt = lbfgs_refine(a_var, t_sub, p_start=20, p_end=100, n_stages=3)
            sc = evaluate_safe(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Variant {variant}: NEW BEST {sc:.6f}")

    # === Phase 1: p-norm L-BFGS (main) ===
    print(f"\n=== Phase 1: p-norm L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out = lbfgs_refine(best_seq, t_alloc, p_start=10, p_end=300, n_stages=10)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  L-BFGS: NEW BEST {sc:.6f}")

    # === Phase 2: Multiplicative subgradient ===
    print(f"\n=== Phase 2: Subgradient Multiplicative (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sub, _ = subgradient_multiplicative(best_seq, t_alloc, n_iters_max=50000)
        sc = evaluate_safe(a_sub)
        if sc < best_score:
            best_score = sc
            best_seq = a_sub.copy()
            print(f"  Subgrad: NEW BEST {sc:.6f}")

    # === Phase 3: Nesterov LP ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=30000)
        sc = evaluate_safe(a_lp)
        if sc < best_score:
            best_score = sc
            best_seq = a_lp.copy()
            print(f"  NesterovLP: NEW BEST {sc:.6f}")

    # === Phase 4: p-norm L-BFGS round 2 (higher p) ===
    print(f"\n=== Phase 4: L-BFGS r2 high-p (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out2 = lbfgs_refine(best_seq, t_alloc, p_start=100, p_end=800, n_stages=6)
        sc = evaluate_safe(a_out2)
        if sc < best_score:
            best_score = sc
            best_seq = a_out2.copy()
            print(f"  L-BFGS r2: NEW BEST {sc:.6f}")

    # === Phase 5: Local refinement ===
    print(f"\n=== Phase 5: Local Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_lr, _ = local_refine(best_seq, t_alloc, n_steps=30000)
        sc = evaluate_safe(a_lr)
        if sc < best_score:
            best_score = sc
            best_seq = a_lr.copy()
            print(f"  LocalRefine: NEW BEST {sc:.6f}")

    # === Phase 6: Nesterov LP round 2 ===
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=20000)
        sc = evaluate_safe(a_lp2)
        if sc < best_score:
            best_score = sc
            best_seq = a_lp2.copy()
            print(f"  NesterovLP r2: NEW BEST {sc:.6f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    candidate_lengths = sorted(set([
        n_best - 32, n_best - 16, n_best + 16, n_best + 32,
        n_best - 64, n_best + 64, n_best - 96, n_best + 96,
        896, 928, 960, 992, 1024, 1056, 1088
    ]), key=lambda x: abs(x - n_best))
    candidate_lengths = [x for x in candidate_lengths if 512 <= x <= 2048 and x != n_best]

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 120.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_to_length(best_seq, alt_n)
        t_pn = t_total * 0.5
        a_alt = lbfgs_refine(init_alt, t_pn, p_start=30, p_end=200, n_stages=4)
        sc = evaluate_safe(a_alt)
        if sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            print(f"  Length {alt_n}: NEW BEST {sc:.6f}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for noise_scale in [0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 1e-10)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.25, 50.0)
            a_pert = lbfgs_refine(perturbed, t_sub, p_start=50, p_end=300, n_stages=4)
            sc = evaluate_safe(a_pert)
            if sc < best_score:
                best_score = sc
                best_seq = a_pert.copy()
                print(f"  Perturb {noise_scale}: NEW BEST {sc:.6f}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 60.0)
        a_fin = lbfgs_refine(best_seq, t_fin, p_start=200, p_end=1000, n_stages=4)
        sc = evaluate_safe(a_fin)
        if sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.30, 40.0)
        a_lr2, _ = local_refine(best_seq, t_lr, n_steps=20000)
        sc = evaluate_safe(a_lr2)
        if sc < best_score:
            best_score = sc
            best_seq = a_lr2.copy()

    rem = deadline - time.monotonic()
    if rem > 10:
        a_lp_fin, _ = nesterov_lp_refine(best_seq, rem, solver=solver, n_max_iters=10000)
        sc = evaluate_safe(a_lp_fin)
        if sc < best_score:
            best_score = sc
            best_seq = a_lp_fin.copy()

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()- (deadline - total_budget + 20.0):.0f}s")
    return [float(x) for x in best_seq]
