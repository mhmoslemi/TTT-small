import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix T such that T @ b = convolve(a, b) (full linear)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def score_of(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def lp_best_response(a, solver=None):
    """
    Solve: min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    import cvxpy as cp
    
    n = len(a)
    T = build_conv_matrix(a)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s_b = b_val.sum()
    if s_b < 1e-15:
        return None, None
    b_val = b_val / s_b
    M_val = float(M_var.value)
    
    return b_val, M_val

def alternating_lp(a_init, time_limit, solver=None, verbose=True, alpha_init=1.0):
    """
    Alternating LP: repeatedly solve the best-response LP and update.

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    alpha = alpha_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_val, M_val = lp_best_response(a, solver=solver)
        
        if b_val is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        
        # Evaluate the LP solution
        score_b = score_of(b_val)
        
        # Update: use the LP solution directly or damped
        if alpha >= 0.99:
            a_new = b_val.copy()
        else:
            a_new = (1 - alpha) * a + alpha * b_val
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = b_val.copy()
            else:
                a_new = a_new / s_new
        
        score_new = score_of(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step size
            alpha = max(alpha * 0.8, 0.1)
        
        # Periodically reset alpha to 1.0 to allow larger steps
        if no_improve % 5 == 0:
            alpha = min(alpha * 1.2, 1.0)
        
        a = a_new
        
        if verbose and (it % 10 == 1 or it <= 5):
            elapsed = time.monotonic() - t0
            print(f"    ALP it={it} ({elapsed:.0f}s/{time_limit:.0f}s): "
                  f"best={best_score:.8f} M_lp={M_val:.8f} alpha={alpha:.3f}")
    
    return best_a, best_score

def cosine_init(n, rng):
    """Cosine window initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / (center + 1e-10)
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    """Two-peak initialization."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_init(n, rng):
    """Periodic pattern initialization."""
    period = int(rng.integers(3, 12))
    a = np.zeros(n)
    for i in range(n):
        a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * i / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    conv_aa = fast_convolve(a)

    else:

        T_active = np.zeros((len(active_set), n))

        b_var = cp.Variable(n, nonneg=True)

            T_active @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

            if solver is not None:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)

            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return b_opt, M_opt

            break

        b_val = np.maximum(b_val, 0.0)

        if s_b < 1e-15:

        for k in range(num_total):

                if conv_ab[k] > M_val + 1e-10:

    return b_val, M_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP best response (fast, active-set based)
2. Alternating LP with dual-based convergence certificates
3. Coordinate descent with exact 1D subproblem solving
4. Fine-grained CD with ternary search for last-mile improvements
5. Multi-start with diverse initializations at multiple lengths
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_rows(a, indices):
    """Build rows of the convolution matrix for given indices."""
    n = len(a)
    rows = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows[idx, j] = a[k - j]
    return rows


def score_of(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def lp_best_response_cp(a, solver=None, max_cp=80, top_k=30):
    """
    LP best response via cutting plane.
    min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt, lambda_dual) or (None, None, None).
    """
    import cvxpy as cp

    n = len(a)
    num_total = 2 * n - 1
    conv_aa = fast_convolve(a)

    # Initial active set: top-k peaks of a*a
    if top_k < num_total:
        active_list = list(np.argsort(conv_aa)[-top_k:])
    else:
        active_list = list(range(num_total))

    b_val = None
    M_val = np.inf
    lam_val = None

    for cp_iter in range(max_cp):
        active_set = list(set(active_list))
        T_active = build_conv_matrix_rows(a, active_set)

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T_active @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        try:
            if solver is not None:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
            solved = True
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                solved = True
            except Exception:
                pass

        if not solved or prob.status not in ('optimal', 'optimal_inaccurate'):
            if b_val is not None:
                return b_val, M_val, lam_val
            return None, None, None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b
        M_val = float(M_var.value)

        # Get dual variables (lambda for the inequality constraints)
        try:
            lam_val = np.asarray(constraints[0].dual_value, dtype=np.float64)
            lam_val = np.maximum(lam_val, 0.0)
            s_lam = lam_val.sum()
            if s_lam > 1e-15:
                lam_full_map = {}
                for idx, k in enumerate(active_set):
                    lam_full_map[k] = lam_val[idx]
            else:
                lam_full_map = {}
        except Exception:
            lam_full_map = {}

        # Check for violated constraints
        conv_ab = convolve_ab(a, b_val)
        max_violation = 0.0
        worst_k = -1
        violations = []
        for k in range(num_total):
            if k not in set(active_set):
                viol = conv_ab[k] - M_val
                if viol > 1e-9:
                    violations.append((viol, k))
                    if viol > max_violation:
                        max_violation = viol
                        worst_k = k

        if not violations:
            # KKT satisfied: return with full dual
            return b_val, M_val, lam_full_map

        # Add most violated constraints
        violations.sort(reverse=True)
        for v in violations[:5]:
            if v[1] not in active_list:
                active_list.append(v[1])

        # If active set too large, add all remaining
        if len(active_list) > num_total // 2:
            active_list = list(range(num_total))

    return b_val, M_val, lam_full_map


def find_best_solver():
    """Find the fastest working LP solver."""
    import cvxpy as cp
    candidates = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK]
    for s in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver: SCS (fallback)")
    return cp.SCS


def alternating_lp(a_init, time_limit, solver=None, verbose=True):
    """
    Alternating LP optimization.
    At each iteration: solve best-response LP, update a toward the solution.
    Uses dual certificate for convergence detection.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    alpha = 1.0  # step size for update
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        iter_start = time.monotonic()

        b_val, M_lp, lam_map = lp_best_response_cp(a, solver=solver, top_k=30)

        if b_val is None:
            print(f"    ALP: LP failed at iter {it}")
            break

        # Compute dual lower bound
        dual_lb = M_lp  # LP optimal value is a lower bound on optimal ||a*a||_inf

        # Update
        a_new = (1.0 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = b_val.copy()
        else:
            a_new = a_new / s_new

        score_new = score_of(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 3:
                alpha = max(alpha * 0.7, 0.2)
            elif stall_count % 5 == 0:
                alpha = min(alpha * 1.3, 1.0)

        a = a_new
        iter_time = time.monotonic() - iter_start

        # Check gap between primal and dual
        gap = score_new / (2.0 * n) - dual_lb  # in terms of max convolution value
        if gap < 1e-7 and it > 5:
            if verbose:
                print(f"    ALP: converged (gap={gap:.2e}) at iter {it}")
            break

        # Adaptive: if LP is too slow, reduce remaining iterations
        if iter_time > (time_limit - (time.monotonic() - t0)) * 0.3:
            if verbose:
                print(f"    ALP: LP too slow ({iter_time:.1f}s), stopping")
            break

        if verbose and (it <= 5 or it % 20 == 0):
            elapsed = time.monotonic() - t0
            print(f"    ALP it={it} ({elapsed:.0f}/{time_limit:.0f}s): "
                  f"best={best_score:.8f} M_lp={M_lp:.8f} alpha={alpha:.2f} "
                  f"gap={gap:.2e} cp_iter_time={iter_time:.2f}s")

    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, fine=False):
    """
    Coordinate descent with exact 1D subproblem solving.
    For each coordinate i, find the optimal delta by searching over a grid.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    if fine:
        factors = [0.999, 0.998, 0.997, 0.995, 0.99, 0.98, 0.95,
                   1.001, 1.002, 1.003, 1.005, 1.01, 1.02, 1.05]
        tol = 1e-16
    else:
        factors = [0.99, 0.98, 0.97, 0.95, 0.9,
                   1.01, 1.02, 1.03, 1.05, 1.1]
        tol = 1e-15

    max_sweeps_no_improve = 5 if fine else 3

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # a_shifted[k - k_start] = a[k - i] for k in [k_start, k_end]
            a_shifted = a[k_start - i: k_end - i + 1]
            affected = conv[k_start: k_end + 1]
            has_diag = (2 * i >= k_start) and (2 * i <= k_end)
            diag_idx = 2 * i - k_start if has_diag else -1

            best_delta = 0.0
            best_new_M = M

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -tol:
                    continue

                # New affected values
                new_aff = affected + 2.0 * delta * a_shifted
                if has_diag:
                    new_aff = new_aff.copy()
                    new_aff[diag_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - tol:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - tol:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update convolution
                changes = 2.0 * best_delta * a_shifted
                if has_diag:
                    changes = changes.copy()
                    old_a_i = a[i] - best_delta
                    changes[diag_idx] = 2.0 * best_delta * old_a_i + best_delta * best_delta
                conv[k_start: k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start: k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= max_sweeps_no_improve:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {'FineCD' if fine else 'CD'} sweep={sweeps} "
                  f"({elapsed:.0f}s/{time_limit:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def cosine_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / (center + 1e-10)
    power = rng.uniform(0.5, 3.0)
    a = (0.5 * (1.0 + np.cos(np.pi * x))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def two_hump_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def periodic_init(n, rng):
    period = int(rng.integers(3, 15))
    a = np.zeros(n)
    for i in range(n):
        a[i] = 0.5 + 0.4 * np.cos(2.0 * np.pi * i / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
