def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        for idx_j, j in enumerate(P_arr):

def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with log-sum-exp continuation (adaptive smoothing)
2. Exact 1D line search via pairwise peak equalization
3. FFT-based subgradient for multiplicative mirror descent
4. Structured perturbations for escape from local optima
5. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_subgradient(a, weights):
    """Compute g[i] = 2 * sum_j a[j] * weights[j+i] via FFT cross-correlation."""
    n = len(a)
    if len(weights) != 2 * n - 1:
        # Pad or trim
        w = np.zeros(2 * n - 1)
        m = min(len(weights), 2 * n - 1)
        w[:m] = weights[:m]
        weights = w
    w_flipped = weights[::-1]
    conv_size = n + len(w_flipped) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_flipped, fft_size)
    conv_result = irfft(A * W, fft_size)[:conv_size]
    g = 2.0 * conv_result[n-1 : 2*n-1]
    return g


def fista_continuation(a0, time_limit, beta_start=5.0, beta_end=500.0, verbose=False):
    """
    FISTA with log-sum-exp continuation for the smoothed max problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    x_prev = a.copy()
    x = a.copy()
    t_nest = 1.0
    best_x = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.7))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = x + (t_nest - 1.0) / t_new * (x - x_prev)
        y = simplex_project(y)
        
        # Compute smooth gradient
        conv_y = fast_convolve(y)
        max_cv = conv_y.max()
        scaled = beta * (conv_y - max_cv)
        scaled = np.clip(scaled, -40, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z
        
        # Gradient via FFT
        g = compute_subgradient(y, weights)
        
        # Learning rate: 1/L where L is Lipschitz constant
        # For smooth max with parameter beta, L ~ beta * max_k sum_i |2*a[k-i]|
        # = beta * 2 * max_k (sum of a) = 2*beta (since sum a = 1)
        # Actually L = beta * ||Hessian|| <= beta * 2 * max(a) * 2n ... 
        # Let's just use a safe learning rate
        eta = 1.0 / (2.0 * beta * max(y.max() * n, 1.0) + 1e-10)
        # Simpler: eta = 1 / (2 * beta)
        eta = 0.5 / (beta + 1e-10)
        
        x_new = simplex_project(y - eta * g)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(x_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if no_improve > 200:
            # Reset Nesterov momentum
            t_nest = 1.0
            x_prev = x_new.copy()
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    FISTA it={it} ({elapsed:.0f}s) beta={beta:.1f}: score={best_score:.6f}")
    
    return best_x, best_score

    a = a / s
