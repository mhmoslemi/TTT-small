def lse_gradient(a, T):

def nesterov_subgrad(a0, time_budget, T=50.0, n_max_iters=200000):
    """Nesterov accelerated subgradient with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_seq(a0).copy()
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    mu = 0.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        # Nesterov extrapolation
        mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y = a + ((mu - 1.0) / (mu_new + 1.0)) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        mu = mu_new

        # Compute smoothed gradient
        b = fast_convolve(y)
        Tb = T * b
        m = Tb.max()
        exp_Tb = np.exp(Tb - m)
        w = exp_Tb / exp_Tb.sum()

        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * y[k - idx]

        # Adaptive step size
        lr = 1.0 / (np.sqrt(it + 1.0) * T * 0.5)

        # Update
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        a_prev = a
        a = a_new

        # Track best
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def lbgfs_refine(a0, time_budget, T_start=80.0, T_end=0.05, maxiter=8000, n_stages=12):
    """L-BFGS with log-sum-exp smoothing and geometric temperature annealing."""
    n = len(a0)
    a = norm_seq(a0).copy()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_budget:
                return 1e15
            aa = np.exp(z_in - z_in.max())
            aa = aa / aa.sum()
            b = fast_convolve(aa)
            Tbb = stage_T * b
            mm = Tbb.max()
            return (2.0 * n / stage_T) * (mm + np.log(np.exp(Tbb - mm).sum()))

        def jac(z_in):
            if time.monotonic() - t0 > time_budget:
                return np.zeros_like(z_in)
            aa = np.exp(z_in - z_in.max())
            aa = aa / aa.sum()
            b = fast_convolve(aa)
            Tbb = stage_T * b
            mm = Tbb.max()
            exp_Tb = np.exp(Tbb - mm)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * aa[k - idx]
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def peak_shave(a0, time_budget, n_steps=30000):
    """Multi-peak coordinate descent with pair moves."""
    n = len(a0)
    a = norm_seq(a0).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find near-peaks
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0]
        if len(near_peaks) == 0:
            near_peaks = np.array([peak_k])

        # Compute contributions for top peaks
        contrib = np.zeros(n)
        for k in near_peaks[:6]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            contrib += contribs

        # Try reducing top contributors
        top_coords = np.argsort(contrib)[::-1][:20]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80, 0.75]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to low-contrib positions
                low_coords = np.argsort(contrib)[:n // 3]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-12
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair moves at the peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_c = np.argsort(contribs)[::-1][:10]
            found = False
            for ci in top_c:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.85, 0.78]:
                    for f2 in [0.92, 0.85, 0.78]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_seq(a_trial)
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            if not improved:
                no_improve += 1
                if no_improve > 300:
                    break
        else:
            no_improve = 0

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # === Phase 1: Nesterov subgradient (fast global) ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.20, 300.0)
        print(f"\n=== Phase 1: Nesterov Subgrad (budget={t_alloc:.0f}s) ===", flush=True)
        a1, _ = nesterov_subgrad(best_seq, t_alloc, T=40.0, n_max_iters=200000)
        try_update(a1, "nesterov_1")

    # === Phase 2: L-BFGS smooth refinement ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.25, 400.0)
        print(f"\n=== Phase 2: L-BFGS (budget={t_alloc:.0f}s) ===", flush=True)
        a2 = lbgfs_refine(best_seq, t_alloc, T_start=60.0, T_end=0.05, maxiter=10000, n_stages=10)
        try_update(a2, "lbgfs_1")

    # === Phase 3: Peak shaving (direct) ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.12, 180.0)
        print(f"\n=== Phase 3: Peak Shave (budget={t_alloc:.0f}s) ===", flush=True)
        a3, _ = peak_shave(best_seq, t_alloc, n_steps=50000)
        try_update(a3, "peak_shave_1")

    # === Phase 4: Subgradient local (different T) ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.10, 120.0)
        print(f"\n=== Phase 4: Subgrad Local T=20 (budget={t_alloc:.0f}s) ===", flush=True)
        a4, _ = subgrad_local(best_seq, t_alloc, T=20.0)
        try_update(a4, "subgrad_local")

    # === Phase 5: Second L-BFGS polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.10, 120.0)
        print(f"\n=== Phase 5: L-BFGS Polish (budget={t_alloc:.0f}s) ===", flush=True)
        a5 = lbgfs_refine(best_seq, t_alloc, T_start=15.0, T_end=0.03, maxiter=8000, n_stages=8)
        try_update(a5, "lbgfs_polish")

    # === Phase 6: Peak shave again ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.08, 100.0)
        print(f"\n=== Phase 6: Peak Shave 2 (budget={t_alloc:.0f}s) ===", flush=True)
        a6, _ = peak_shave(best_seq, t_alloc, n_steps=30000)
        try_update(a6, "peak_shave_2")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.8f}) ===", flush=True)
    alt_lengths = sorted(set([
        int(n_best * 1.25), int(n_best * 1.5), int(n_best * 0.75),
        512, 768, 1024, 1280, 1536, 1920, 2048, 2560
    ]))
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 3000 and x != n_best]

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 45:
            break
        t_total = min(rem * 0.18, 180.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s", flush=True)

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick subgradient
        t_sub = t_total * 0.4
        a_alt, _ = nesterov_subgrad(init_alt, t_sub, T=50.0, n_max_iters=50000)
        try_update(a_alt, f"subgrad_n{alt_n}")

        # L-BFGS on result
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.12, 80.0)
            if t_lb > 10:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=30.0, T_end=0.1, maxiter=5000, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

        # Quick peak shave
        rem = deadline - time.monotonic()
        if rem > 10:
            t_ps = min(rem * 0.06, 40.0)
            if t_ps > 5:
                a_alt3, _ = peak_shave(best_seq, t_ps, n_steps=15000)
                try_update(a_alt3, f"peak_n{alt_n}")

    # === Phase 8: Perturbation escape ===
    rem = deadline - time.monotonic()
    if rem > 40:
        print(f"\n=== Phase 8: Perturbation (best={best_score:.8f}) ===", flush=True)
        n_cur = len(best_seq)
        for scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_seq(perturbed)

            t_sub = min(rem * 0.15, 60.0)
            if t_sub > 5:
                a_p, _ = nesterov_subgrad(perturbed, t_sub, T=30.0, n_max_iters=30000)
                try_update(a_p, f"pert_{scale}")

            rem = deadline - time.monotonic()
            if rem > 10:
                t_lb = min(rem * 0.10, 40.0)
                if t_lb > 5:
                    a_p2 = lbgfs_refine(best_seq, t_lb, T_start=10.0, T_end=0.1, maxiter=3000, n_stages=6)
                    try_update(a_p2, f"pert_lb_{scale}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===", flush=True)
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=8.0, T_end=0.02, maxiter=5000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.20, 50.0)
        a_ps = peak_shave(best_seq, t_ps, n_steps=20000)
        try_update(a_ps[0], "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_sg = min(rem * 0.15, 40.0)
        a_sg, _ = subgrad_local(best_seq, t_sg, T=15.0)
        try_update(a_sg, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
