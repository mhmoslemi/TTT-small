def fast_convolve_with_fft(a):
    """Equivalent to the original discrete convolution with full mode."""
    n = len(a)
    # Make FFT size power of two, covering full convolution output (length 2*n-1)
    size_needed = 2 * n - 1
    fft_size = 1 << (size_needed - 1).bit_length()
    A = rfft(a, fft_size)
    conv_result = irfft(A * A.conjugate(), fft_size)[:size_needed]
    return conv_result
