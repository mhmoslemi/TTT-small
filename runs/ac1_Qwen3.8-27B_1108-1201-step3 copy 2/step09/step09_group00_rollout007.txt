# Global cache for LP solver
_lp_solver = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:

                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

            sy = 1.0

def kkt_waterfilling(a0, time_limit, max_iters=100000):
    """KKT-based water-filling refinement. Equalizes the subgradient on the support."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        
        # Find active peaks (within 0.1% of max)
        thresh = max_c * 0.999
        active = np.where(c >= thresh)[0]
        
        # Compute average subgradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # KKT: g should be constant on support
        support_thresh = 1e-10
        support = a > support_thresh
        if support.sum() > 0:
            mu = g[support].mean()
            
            # Water-filling step: reduce where g > mu, increase where g < mu
            a_new = a.copy()
            for i in range(n):
                if a[i] > support_thresh:
                    # On support: reduce if g[i] > mu
                    if g[i] > mu:
                        a_new[i] = max(0, a[i] - lr * (g[i] - mu))
                else:
                    # Off support: add if g[i] < mu
                    if g[i] < mu - 1e-6:
                        a_new[i] = lr * (mu - g[i]) * 0.1
            
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / sn
            
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= 20:
                lr *= 0.8
                no_improve = 0
            lr = max(1e-8, lr)
        
        if it % 5000 == 0:
            print(f"    KKT iter={it}: score={best_score:.8f} lr={lr:.2e}")
    
    return best_a, best_score

class AdamOptimizer:
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, x, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(x)
            self.v = np.zeros_like(x)
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def adam_smoothed(a0, time_limit, T_stages=None, max_iters=50000, lr=0.01):
    """Adam optimizer on log-sum-exp smoothed objective in probability simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        iters_this_stage = max(100, int(remaining * 100 / (len(T_stages))))
        opt = AdamOptimizer(lr=lr / np.sqrt(n))
        
        for it in range(iters_this_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute log-sum-exp objective and gradient
            c = fast_convolve(a)
            max_c = c.max()
            exp_c = np.exp((c - max_c) / T)
            lse = max_c + T * np.log(exp_c.sum())
            
            # Gradient of lse w.r.t. a
            w = exp_c / exp_c.sum()
            grad = np.zeros(n)
            for k in range(len(c)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += w[k] * a[k - idx]
            
            # Projected Adam step (maintain simplex: a >= 0, sum = 1)
            a_new = opt.step(a, grad)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / sn
            
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                a = a_new  # still update a even if not improving (Adam momentum)
        
        if time.monotonic() - t0 < time_limit:
            print(f"    Adam T={T}: score={best_score:.8f}")
    
    return best_a, best_score

    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Two-bump structure
        a = np.exp(-10 * x**2) + np.exp(-10 * (x-1)**2)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_shaping(a, time_limit, n_iters=5000):
    """Try to flatten the power spectrum to reduce auto-convolution peak."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        fft_size = 1 << (2 * n - 1).bit_length() if (2*n-1).bit_length() > 1 else 2
        # Actually let's use a power of 2 >= 2n-1
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        p = mag ** 2  # power spectrum
        
        # Target: flat power spectrum
        # But we need to maintain |A[0]|^2 = (sum a)^2 = 1 (DC component)
        # Actually, sum(a) = A[0] * (1/fft_size) * fft_size = A[0]... 
        # In numpy's FFT convention: A[k] = sum_j a[j] * exp(-2πi*j*k/N)
        # So A[0] = sum(a)
        # And the auto-conv via IFFT(|A|^2) gives the circular autocorrelation
        # which equals the linear autocorrelation for k in [0, n-1] (with zero padding)
        
        # Strategy: flatten p while keeping p[0] fixed and a >= 0
        # Simple approach: interpolate p towards flat, then take IFFT and clip
        
        p_target = np.ones_like(p) * (p.sum() / len(p))
        p_target[0] = p[0]  # keep DC fixed
        
        # Blend current p with target
        blend = 0.1  # small step
        p_new = (1 - blend) * p + blend * p_target
        
        # Reconstruct A from p_new and current phase
        A_new = np.sqrt(p_new) * np.exp(1j * phase)
        
        # Take IFFT to get new a
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        
        if it % 2000 == 0:
            print(f"    Spectral iter={it}: score={best_score:.8f}")
    
    return best_a, best_score
