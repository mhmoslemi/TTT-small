import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the actual score (scale-invariant, so normalize sum=1 first)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a = a / s
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max()


def p_norm_obj_grad_z(z, p):
    """Overflow-safe p-norm objective and gradient in softmax parameterization."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    if c_max < 1e-300:
        return 2.0 * n * c_max, np.zeros_like(z)
    c_norm = c / c_max
    # Avoid log issues
    c_norm_safe = np.maximum(c_norm, 1e-300)
    log_c = np.log(c_norm_safe)
    p_log_c = p * log_c
    p_log_c_max = p_log_c.max()
    exp_p = np.exp(p_log_c - p_log_c_max)
    S = exp_p.sum()
    if S < 1e-300:
        S = 1e-300
    obj = 2.0 * n * c_max * np.exp(p_log_c_max) * S ** (1.0 / p) / np.exp(p_log_c_max)
    # Simplify: obj = 2*n*c_max*S^(1/p)
    obj = 2.0 * n * c_max * S ** (1.0 / p)
    # Gradient w.r.t. a: proportional to correlate(c_norm^(p-1), a)

        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
    else:
        w = c_norm_safe ** (p - 1)
    corr = np.correlate(w, a, mode='valid')
    factor = 4.0 * n * c_max * S ** (1.0 / p - 1.0)
    ga = factor * corr
    gz = a * (ga - np.dot(ga, a))
    return obj, gz


def run_pnorm_lbfgs(a0, time_limit, p_schedule):
    """Multi-stage p-norm L-BFGS with given p schedule (list of p values)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    for p_val in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # More iterations for lower p (easier landscape), fewer for high p
        if p_val < 50:
            max_iter = 300
        elif p_val < 500:
            max_iter = 200
        else:
            max_iter = 100

        def func(zv):
            return p_norm_obj_grad_z(zv, p_val)[0]

        def jac(zv):
            return p_norm_obj_grad_z(zv, p_val)[1]

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-16,

        z_max = z.max()

        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        elapsed = time.monotonic() - t0
        print(f"    p={p_val:.0f} ({elapsed:.0f}s): score={sc:.6f}")

    return best_a

    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute full convolution for active set identification
        c = fast_convolve(y)
        c_max = c.max()
        # Identify active set: lags within 1% of max
        threshold = c_max * 0.99
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]

        if len(active_indices) == 0:
            break

        # Subgradient: sg[i] = 2 * sum_{k in active} a[k-i] (where 0<=k-i<n)
        # This equals 2 * correlate(indicator_active, y, 'valid')
        indicator = np.zeros(2 * n - 1)
        indicator[active_indices] = 1.0
        sg = 2.0 * np.correlate(indicator, y, mode='valid')

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Nesterov step
        alpha = 2.0 / (it + 2)
        lr = 1.0 / (sg_norm * (it + 1) ** 0.5 + 1e-8)
        lr = min(lr, 0.5)

        y_new = y - lr * sg
        # Project onto simplex
        y_new = project_simplex(y_new)

        # Update a (averaging)
        a = ((it + 1) * a + alpha * (it + 2) * y_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        # Check improvement
        if it % 500 == 0 or it < 100:
            sc_a = compute_score(a)
            sc_y = compute_score(y_new)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y_new.copy()
            if it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovSG iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, n_active={len(active_indices)}")

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    sw = w.sum()
    if sw < 1e-15:
        return np.ones(n) / n
    return w / sw


def peak_balance_refine(a0, time_limit, n_iter=50000):
    """Fast local refinement using subgradient-driven coordinate moves."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        # Active set: top peaks within 0.5% of max
        threshold = c_max * 0.995
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]

        if len(active_indices) == 0:
            break

        # Compute subgradient
        indicator = np.zeros(2 * n - 1)
        indicator[active_indices] = 1.0
        sg = 2.0 * np.correlate(indicator, a, mode='valid')

        # Find elements with highest subgradient (most responsible for peaks)
        # and elements with lowest subgradient (safe to increase)
        top_k = min(10, n)
        top_indices = np.argsort(sg)[::-1][:top_k]
        bottom_indices = np.argsort(sg)[:top_k]

        improved = False
        # Try moving mass from high-sg to low-sg elements
        for i in top_indices:
            if a[i] < 1e-10:
                continue
            for j in bottom_indices:
                if i == j:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        a_trial[i] = 0
                        a_trial[j] += a[i]
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try uniform redistribution among top-sg elements
            if it % 100 == 50 and len(active_indices) > 5:
                top_5 = np.argsort(sg)[::-1][:5]
                a_trial = a.copy()
                total_top = a_trial[top_5].sum()
                avg = total_top / 5
                for i in top_5:
                    a_trial[i] = avg
                st = a_trial.sum()
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and it % 1000 == 0:
            # Random perturbation to escape
            rng = np.random.default_rng(it + 12345)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and it > 5000:
            break

        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakBalance iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n):
    """Create a good initial sequence."""
    x = np.linspace(0, 1, n)
    # Sine window with power tuning
    a = np.sin(np.pi * x) ** 1.2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. p-norm L-BFGS (p: 20->2000) - main workhorse
    2. Nesterov subgradient with active set
    3. Peak-balancing local search
    4. Multi-length exploration
    5. Final high-p polish
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init)
        best_state['seq'] = parent.copy()
        best_state['score'] = compute_score(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    def try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        sc = compute_score(a)
        if sc < best_state['score'] - 1e-12:
            # Verify with actual evaluator
            sc_real = evaluate_safe(a)
            if np.isfinite(sc_real) and sc_real < best_state['score']:
                best_state['score'] = sc_real
                best_state['seq'] = a / s
                print(f"  NEW BEST [{label}]: {sc_real:.6f} (n={len(a)})")
                return True
        return False

    # === Phase 1: p-norm L-BFGS (p: 20 -> 500) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 350.0)
    if t1 > 30:
        print(f"\n=== Phase 1: p-norm L-BFGS (p:20-500), budget={t1:.0f}s ===")
        p_sched = [20, 40, 80, 150, 300, 500]
        a1 = run_pnorm_lbfgs(best_state['seq'], t1, p_sched)
        try_update(a1, "pnorm_p1")

    # === Phase 2: p-norm L-BFGS high-p (p: 500 -> 5000) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        print(f"\n=== Phase 2: p-norm L-BFGS high-p (p:500-5000), budget={t2:.0f}s ===")
        p_sched = [500, 1000, 2000, 3000, 5000]
        a2 = run_pnorm_lbfgs(best_state['seq'], t2, p_sched)
        try_update(a2, "pnorm_p2")

    # === Phase 3: Nesterov subgradient ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Nesterov Subgradient, budget={t3:.0f}s ===")
        a3, _ = nesterov_subgradient(best_state['seq'], t3, n_max_iter=100000)
        try_update(a3, "nesterov_sg")

    # === Phase 4: Peak-balancing local search ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Peak Balance, budget={t4:.0f}s ===")
        a4, _ = peak_balance_refine(best_state['seq'], t4, n_iter=80000)
        try_update(a4, "peak_balance_1")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== Phase 5: Length Exploration ===")
        base = len(best_state['seq'])
        candidate_lengths = sorted(set([
            base - 64, base - 32, base + 32, base + 64,
            base - 128, base + 128,
            896, 960, 1024, 1088, 1152
        ]))
        # Filter valid lengths
        candidate_lengths = [x for x in candidate_lengths if 512 <= x <= 2048 and x != base]
        candidate_lengths.sort(key=lambda x: abs(x - base))

        n_explored = 0
        for alt_n in candidate_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 2:
                break
            n_explored += 1
            t_alt = min(rem * 0.20, 120.0)
            if t_alt < 30:
                break
            print(f"  Exploring n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_state['seq'], alt_n)
            # Quick p-norm
            p_sched_short = [50, 200, 1000]
            a_alt = run_pnorm_lbfgs(init_alt, t_alt * 0.6, p_sched_short)
            improved = try_update(a_alt, f"pnorm_n{alt_n}")
            if improved:
                rem = deadline - time.monotonic()
                if rem > 30:
                    a_alt2, _ = peak_balance_refine(best_state['seq'], min(rem * 0.3, 60.0), n_iter=20000)
                    try_update(a_alt2, f"peak_n{alt_n}")

    # === Phase 6: Final high-p polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 6: Final Polish, budget={rem:.0f}s ===")
        t_fin = min(rem * 0.5, 100.0)
        p_sched_fin = [2000, 5000, 10000]
        a_fin = run_pnorm_lbfgs(best_state['seq'], t_fin, p_sched_fin)
        try_update(a_fin, "final_pnorm")

    # === Phase 7: Final local search ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.5, 60.0)
        a_lr, _ = peak_balance_refine(best_state['seq'], t_lr, n_iter=30000)
        try_update(a_lr, "final_local")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}, time={elapsed:.0f}s")
    return [float(x) for x in best_state['seq']]
