def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    2. Intensive accelerated LSE refinement
    3. LP direction injection rounds
    4. L-BFGS polish at very low temperature
    5. Peak shaving for final micro-improvements
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    solver = _get_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_score(parent)
            print(f"Parent: n={len(parent)}, local_score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, mode=0)
        best_score = local_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_nonneg(np.asarray(a, dtype=np.float64))
        sc = local_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Quick multi-length assessment (30s) ===
    print(f"\n=== Phase 1: Length Assessment (best={best_score:.8f}) ===")
    t0 = time.monotonic()
    t_phase1 = min(30.0, (deadline - time.monotonic()) * 0.05)
    
    alt_lengths = []
    for mult in [0.8, 0.9, 1.1, 1.25, 1.15]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    # Add some specific lengths
    for fn in [768, 896, 1024, 1152, 1280]:
        if 384 <= fn <= 3000 and fn != n_best:
            alt_lengths.append(fn)
    alt_lengths = sorted(set(alt_lengths))
    
    # Just evaluate interpolated versions (quick)
    length_scores = {}
    for alt_n in alt_lengths[:6]:
        interp = interpolate_sequence(best_seq, alt_n)
        sc = local_score(interp)
        length_scores[alt_n] = sc
        print(f"  n={alt_n}: interpolated score={sc:.8f}")
    
    rem = deadline - time.monotonic()
    
    # === Phase 2: Intensive accelerated LSE on best length (60% of time) ===
    print(f"\n=== Phase 2: Accelerated LSE n={n_best} (best={best_score:.8f}) ===")
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, T: 300->0.3")
        a_r1, _ = accelerated_lse_refine(best_seq, t_alloc, T_start=300.0, T_end=0.3,
                                          n_max_iters=200000, eta0=0.3, solver=solver)
        try_update(a_r1, "accel_lse_1")
    
    rem = deadline - time.monotonic()
    
    # === Phase 3: Second round with different parameters (20% of time) ===
    print(f"\n=== Phase 3: LSE Round 2 (best={best_score:.8f}) ===")
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s, T: 50->0.2, eta=0.15")
        a_r2, _ = accelerated_lse_refine(best_seq, t_alloc, T_start=50.0, T_end=0.2,
                                          n_max_iters=150000, eta0=0.15, solver=solver)
        try_update(a_r2, "accel_lse_2")
    
    rem = deadline - time.monotonic()
    
    # === Phase 4: L-BFGS polish (15% of time) ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 15->0.05")
        a_lb = lbgfs_polish(best_seq, t_alloc, T_start=15.0, T_end=0.05,
                            maxiter=10000, n_stages=10)
        try_update(a_lb, "lbgfs_polish")
    
    rem = deadline - time.monotonic()
    
    # === Phase 5: Peak shaving (5% of time) ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.8f}) ===")
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_fast(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")
    
    rem = deadline - time.monotonic()
    
    # === Phase 6: Perturbation + re-optimize (10% of time) ===
    print(f"\n=== Phase 6: Perturbation Escape (best={best_score:.8f}) ===")
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        n_cur = len(best_seq)
        # Try several perturbations
        for scale in [0.02, 0.05, 0.1]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_nonneg(perturbed)
            sc_pert = local_score(perturbed)
            
            t_sub = min(t_alloc / 3, rem * 0.7)
            if t_sub < 10:
                continue
            a_pert, _ = accelerated_lse_refine(perturbed, t_sub, T_start=30.0, T_end=0.2,
                                                n_max_iters=50000, eta0=0.2, solver=solver)
            try_update(a_pert, f"pert_{scale}")
    
    rem = deadline - time.monotonic()
    
    # === Phase 7: Alternative length with quick optimization (10% of time) ===
    print(f"\n=== Phase 7: Alt Length Optimization (best={best_score:.8f}) ===")
    # Pick the best alternative length from Phase 1
    if length_scores:
        best_alt_n = min(length_scores, key=length_scores.get)
        if best_alt_n != n_best:
            t_alloc = min(rem * 0.10, 120.0)
            if t_alloc > 20:
                print(f"  Trying n={best_alt_n}, budget={t_alloc:.0f}s")
                init_alt = interpolate_sequence(best_seq, best_alt_n)
                a_alt, _ = accelerated_lse_refine(init_alt, t_alloc, T_start=50.0, T_end=0.2,
                                                   n_max_iters=50000, eta0=0.2, solver=solver)
                try_update(a_alt, f"alt_n{best_alt_n}")
                
                rem = deadline - time.monotonic()
                if rem > 15 and best_seq is not None and len(best_seq) == best_alt_n:
                    a_alt2 = lbgfs_polish(best_seq, min(rem * 0.5, 40.0),
                                           T_start=10.0, T_end=0.1, maxiter=3000, n_stages=6)
                    try_update(a_alt2, f"alt_lbgfs_n{best_alt_n}")
    
    rem = deadline - time.monotonic()
    
    # === Phase 8: Final micro-polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if rem > 10:
        t_fin = min(rem * 0.3, 40.0)
        a_fin = lbgfs_polish(best_seq, t_fin, T_start=5.0, T_end=0.03,
                             maxiter=4000, n_stages=6)
        try_update(a_fin, "final_lbgfs")
    
    rem = deadline - time.monotonic()
    if rem > 8:
        t_ps = min(rem * 0.3, 20.0)
        a_ps2, _ = peak_shave_fast(best_seq, t_ps, n_steps=10000)
        try_update(a_ps2, "final_ps")
    
    rem = deadline - time.monotonic()
    if rem > 8:
        a_r3, _ = accelerated_lse_refine(best_seq, min(rem * 0.4, 20.0),
                                          T_start=5.0, T_end=0.1,
                                          n_max_iters=20000, eta0=0.1, solver=None)
        try_update(a_r3, "final_accel")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            print(f"\nFINAL VERIFIED SCORE: {fs:.8f} (n={len(best_seq)})")
        else:
            print(f"\nWARNING: Final verification gave inf, using local score: {best_score:.8f}")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(256) / 256.0
    
    return [float(x) for x in best_seq]
