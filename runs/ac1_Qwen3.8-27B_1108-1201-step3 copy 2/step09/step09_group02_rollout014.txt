"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smoothing and continuation (beta annealing)
2. Exponential (softmax) parameterization for automatic simplex constraint
3. Efficient FFT-based gradient computation O(n log n) per evaluation
4. Exact coordinate descent with golden section for final polishing
5. Basin hopping with structured perturbations
6. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def next_pow2(x):
    return 1 << max(0, (x - 1).bit_length())


def fft_convolve(a, b):
    """Linear convolution of two arrays via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fsize = next_pow2(size)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    return irfft(A * B, fsize)[:size]


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:size]


def softmax(u):
    """Numerically stable softmax."""
    u = u - u.max()
    e = np.exp(u)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(u)) / len(u)
    return e / s


def lse_gradient(u, beta):
    """
    Compute f(u) = 2*n*LSE(beta, conv(softmax(u), softmax(u)))
    and its gradient w.r.t. u.
    """
    a = softmax(u)
    n = len(a)

    # Compute autocorrelation
    conv = fast_autocorr(a)

    # Compute LSE and softmax weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    L = (bc_max + np.log(exp_bc.sum())) / beta

    # f = 2*n*L (since sum(a)=1)
    f = 2.0 * n * L

    # Compute dL/da:
    # dL/da[m] = (1/beta) * sum_k p_k * dc_k/da[m]
    # dc_k/da[m] = a[k-m] if 0<=k-m<n, plus extra a[m] if k=2m
    #
    # g1[m] = sum_k p_k * a[k-m] = cross-correlation
    # g2[m] = p[2m] * a[m] (diagonal correction)

    # g1 via FFT: c[m] = sum_j a[j] * p[m+j]
    # c[m] = full_conv[2n-2-m] where full_conv = fft_convolve(a, p_rev)
    p_rev = p[::-1]
    full_conv = fft_convolve(a, p_rev)
    # c has length 2n-1, c[m] = full_conv[2n-2-m]
    idx = slice(2 * n - 2, -1, -1)
    g1 = full_conv[idx].copy()
    # g1 should have length 2n-1, but we only need first n components
    # Actually g1[m] for m=0..n-1 is what we need for dL/da
    g1 = g1[:n]

    # g2: p[2m] * a[m] for m where 2m <= 2n-2
    m_idx = np.arange(n)
    valid = 2 * m_idx < 2 * n - 1
    g2 = np.zeros(n)
    g2[valid] = p[2 * m_idx[valid]] * a[m_idx[valid]]

    dL_da = (g1 + g2) / beta

    # df/da = 2*n*dL_da - 4*n*L (since sum(a)=1)
    df_da = 2.0 * n * dL_da - 4.0 * n * L

    # df/du_j = a_j * (df_da_j - <df_da, a>)
    Ga = df_da @ a
    df_du = a * (df_da - Ga)

    return f, df_du


def lse_objective(u, beta):
    """Objective for scipy minimize (value only)."""
    f, _ = lse_gradient(u, beta)
    return f


def lse_objgrad(u, beta):
    """Objective and gradient for scipy minimize."""
    return lse_gradient(u, beta)


def lse_lbfgs(u0, betas, max_iter_each=200, gtol=1e-10, time_limit=None, t_start=None):
    """
    Run L-BFGS with continuation over increasing beta values.
    Returns best u and best score.
    """
    if t_start is None:
        t_start = time.monotonic()
    u = u0.copy()
    n = len(u)
    best_score = float('inf')
    best_a = softmax(u).copy()

    for beta in betas:
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        # Define wrapper
        class ObjGrad:
            def __init__(self, b, u_arr):
                self.beta = b
                self.u_arr = u_arr
            def fun(self, u):
                f, g = lse_gradient(u, self.beta)
                return f
            def jac(self, u):
                f, g = lse_gradient(u, self.beta)
                return g

        og = ObjGrad(beta, None)

        try:
            res = minimize(og.fun, u, jac=og.jac, method='L-BFGS-B',
                          options={'maxiter': max_iter_each, 'ftol': 1e-14,
                                   'gtol': gtol, 'maxls': 50})
            if res.success or res.fun < best_score:
                u = res.x.copy()
        except Exception:
            pass

        a = softmax(u)
        score = float(fast_autocorr(a).max() * 2.0 * n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    L-BFGS beta={beta}: score={score:.8f}")

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=False, label=""):

            if k_lo > 0:

                    c_vals[idx2] = conv[k] - 2.0 * v * a_partners[idx2]

            quad_idx = -1
            if has_quad:
                quad_idx = 2 * i - k_lo

            # Golden section search
            upper = max(3.0 * v, 4.0 / n, 1.0 / (n * 0.5))

            c = upper - gr * upper
            d = gr * upper
            fc = _cd_score(c, c_vals, a_partners, unaff_max, has_quad, quad_idx, v, n)

                    fd = fc

                changes = 2.0 * delta * a_partners

def _cd_score(x, c_vals, a_partners, unaff_max, has_quad, quad_idx, v, n):
    """Compute score for coordinate descent golden section."""
    ns = 1.0 + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)

def init_flat_spectrum(n, rng):
    """Flat power spectrum (random phase) initialization."""
    n_fft = next_pow2(n)
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.5, 1.5, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def init_tilted_uniform(n, alpha, rng):

    a = 1.0 + alpha * np.cos(2.0 * np.pi * t / n + phase)

def init_linear(n, rng):
    """Linearly increasing sequence."""
    direction = rng.integers(0, 2)
    t = np.arange(n, dtype=np.float64)
    if direction == 0:
        a = t + 1.0
    else:
        a = n - t
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def init_exponential(n, rng):
    """Exponentially shaped sequence."""
    t = np.arange(n, dtype=np.float64)
    rate = rng.uniform(0.5, 3.0)
    direction = rng.integers(0, 2)
    if direction == 0:
        a = np.exp(rate * t / n)
    else:
        a = np.exp(rate * (n - t) / n)
    s = a.sum()
    return a / s


def init_cosine(n, rng):

def init_fourier_mix(n, rng):

    s = a.sum()

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-15)
    ss = f_interp.sum()
    return f_interp / ss


def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.max(np.nonzero(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def subgradient_step(a, conv, lr, n_peaks=20):
    """One subgradient step with active peak weighting."""
    n = len(a)
    k_act = min(n_peaks, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = 0.01 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    a_new = a - lr * grad
    a_new = simplex_project(a_new)
    return a_new


def basin_hopping(a0, time_limit, rng, n_perturbs=12, verbose=False):

            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, 120.0)

        sigma = 0.03 + 0.04 * rng.random()
        perturbed_a = a + rng.standard_normal(n) * sigma
        perturbed_a = np.maximum(perturbed_a, 1e-10)
        perturbed_a = perturbed_a / perturbed_a.sum()

        # Also try peak-breaking perturbation
        conv_p = fast_autocorr(a)
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv_p, -top_k)[-top_k:]
        for k in peaks:
            i_start = max(0, int(k) - n + 1)
            i_end = min(n - 1, int(k))
            idx_range = np.arange(i_start, i_end + 1)
            n_pick = max(1, len(idx_range) // 6)
            picks = rng.choice(idx_range, size=min(n_pick, len(idx_range)), replace=False)
            for ii in picks:
                perturbed_a[ii] *= (1.0 + rng.standard_normal() * 0.08)
        perturbed_a = np.maximum(perturbed_a, 1e-10)
        perturbed_a = perturbed_a / perturbed_a.sum()

        u_pert = np.log(perturbed_a)
        a_opt, s_opt = lse_lbfgs(u_pert, [20, 50, 100], max_iter_each=100,
                                  time_limit=t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max conv(a,b)."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix T: T @ b = conv(a, b)
    # conv(a,b)[k] = sum_j a[j]*b[k-j] for valid j
    # As a matrix: row k has a[j] at position (k-j)
    # T[k, j] = a[k-j] if 0<=k-j<n, else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    ss = b_val.sum()
    if ss < 1e-15:
        return None
    return b_val / ss


def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = float(2.0 * n * fast_autocorr(a).max())
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        b = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Line search
        best_t = 0.0
        best_sc = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = float(2.0 * n * fast_autocorr(cand).max())
            if sc < best_sc:
                best_sc = sc
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = float(2.0 * n * fast_autocorr(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=False, label=""):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = float(2.0 * n * fast_autocorr(a).max())
    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_autocorr(a)
        a_new = subgradient_step(a, conv, lr, n_peaks=15)
        sc = float(2.0 * n * fast_autocorr(a_new).max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
                no_improve = 0
        if it % 2000 == 0:
            lr = max(lr * 0.95, 1e-6)

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining L-BFGS with continuation,
    exact coordinate descent, basin hopping, and LP fictitious play.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: L-BFGS with continuation from diverse initializations ===
    print("\n=== Phase 1: L-BFGS Continuation ===")
    t1_start = time.monotonic()

    n_values = [512, 768, 1024, 1280]
    if parent is not None:
        np_par = len(parent)
        if np_par not in n_values:
            n_values.insert(0, np_par)
        # Add nearby lengths
        for d in [-128, -64, 64, 128]:
            alt = np_par + d
            if 256 <= alt <= 2048 and alt not in n_values:
                n_values.append(alt)

    n_values = sorted(set(n_values))[:6]
    print(f"  Trying n values: {n_values}")

    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 300:
            break
        if time.monotonic() > deadline - 600:
            break

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat_spec', init_flat_spectrum(n, rng)))
        inits.append(('tilted', init_tilted_uniform(n, 0.5, rng)))
        inits.append(('linear', init_linear(n, rng)))
        inits.append(('exp', init_exponential(n, rng)))
        inits.append(('cosine', init_cosine(n, rng)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('two_pole', init_asymmetric_two_pole(n, rng)))
        inits.append(('flat_spec2', init_flat_spectrum(n, rng)))
        inits.append(('tilted2', init_tilted_uniform(n, 0.7, rng)))

        for itype, cand in inits:
            t_left = 300.0 - (time.monotonic() - t1_start)
            if t_left < 5:
                break
            if time.monotonic() > deadline - 500:
                break

            u0 = np.log(np.maximum(cand, 1e-30))
            betas = [5, 10, 20, 50, 100]
            a_opt, s_opt = lse_lbfgs(u0, betas, max_iter_each=150,
                                      time_limit=min(60.0, t_left * 0.4))
            if s_opt < best_score:
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                         f"p1_n{n}_{itype}")

    print(f"  Phase 1 done ({time.monotonic()-t1_start:.0f}s), best={best_score:.8f}")

    # === Phase 2: Intensive L-BFGS on best ===
    print(f"\n=== Phase 2: Intensive L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        n_best = len(best_seq)
        t2_start = time.monotonic()

        for rnd in range(3):
            rem = deadline - time.monotonic() - 350
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 150.0)

            u0 = np.log(np.maximum(best_seq, 1e-30))
            betas = [50, 100, 200, 400, 800]
            a_opt, s_opt = lse_lbfgs(u0, betas, max_iter_each=200,
                                      time_limit=t_alloc)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                     f"p2_r{rnd}")

        print(f"  Phase 2 done ({time.monotonic()-t2_start:.0f}s), best={best_score:.8f}")

    # === Phase 3: Exact coordinate descent polishing ===
    print(f"\n=== Phase 3: ECD Polish (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_best = len(best_seq)
        t3_start = time.monotonic()

        for rnd in range(3):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 180.0)
            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(best_seq, t_alloc,
                                                        verbose=True,
                                                        label=f"p3_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                     f"ecd_r{rnd}")

        print(f"  Phase 3 done ({time.monotonic()-t3_start:.0f}s), best={best_score:.8f}")

    # === Phase 4: Basin hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        rem = deadline - time.monotonic() - 180
        if rem > 40:
            t_bh = min(rem * 0.7, 250.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, n_perturbs=15,
                                        verbose=True)
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score,
                                                     "basin_hop")

    # === Phase 5: LP Fictitious Play ===
    print(f"\n=== Phase 5: LP Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 120
        if rem > 40:
            t_fp = min(rem * 0.6, 150.0)
            print(f"  FP n={n_best}, budget={t_fp:.0f}s")
            solver = get_lp_solver()
            a_fp, s_fp = fictitious_play(best_seq, t_fp, solver=solver,
                                           verbose=True)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score,
                                                     "fp")

            # ECD after FP
            rem = deadline - time.monotonic() - 80
            if rem > 20:
                a_cd, s_cd, _ = exact_coord_descent(a_fp, min(rem * 0.5, 60.0),
                                                     label="post_fp")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score,
                                                         "ecd_after_fp")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = len(best_seq)
        alt_lengths = []
        for alt in [n_best - 128, n_best - 64, n_best + 64, n_best + 128, 1024, 768, 1536]:
            if 256 <= alt <= 2048 and alt != n_best and alt not in alt_lengths:
                alt_lengths.append(alt)

        for n_alt in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 60.0)

            cand = interpolate_seq(best_seq, n_alt)
            u0 = np.log(np.maximum(cand, 1e-30))
            a_opt, s_opt = lse_lbfgs(u0, [20, 50, 100, 200], max_iter_each=100,
                                      time_limit=t_alloc)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score,
                                                     f"alt_n{n_alt}")

    # === Phase 7: Subgradient refinement ===
    print(f"\n=== Phase 7: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_sg = min(rem * 0.5, 60.0)
            a_sg, s_sg = subgradient_optimize(best_seq, t_sg)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score,
                                                     "subgrad")

    # === Phase 8: Final intensive polishing ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_fine = min(rem * 0.4, 40.0)
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                    verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score,
                                                     "fine_cd")

        # Another ECD round
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.6, 20.0),
                                                   label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score,
                                                     "ecd_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
