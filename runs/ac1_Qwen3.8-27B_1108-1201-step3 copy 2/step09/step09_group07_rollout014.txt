    return b_val, float(M_var.value)


def solve_lp_dual(a_fixed, solver):
    """Solve dual LP: max nu s.t. T^T @ lambda >= nu*1, sum(lambda)=1, lambda>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    lam = cp.Variable(size, nonneg=True)
    nu = cp.Variable()
    constraints = [T.T @ lam >= nu, cp.sum(lam) == 1]
    prob = cp.Problem(cp.Maximize(nu), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    lambda_val = np.asarray(lam.value, dtype=np.float64)
    lambda_val = np.maximum(lambda_val, 0)
    s = lambda_val.sum()
    if s < 1e-15:
        return None, None
    lambda_val = lambda_val / s
    return lambda_val, float(nu.value)


def nesterov_lp_iteration(a0, solver, time_limit, alpha_decay=2.0,
                          use_nesterov_avg=True, momentum=0.0, max_iters=50000):
    """LP-based best response with Nesterov dual averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    y_prev = a.copy() if momentum > 0 else None
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            break

        alpha = alpha_decay / (it + 1 + alpha_decay)

        if momentum > 0:
            y = y + momentum * (y - y_prev) + alpha * (b_new - y)
            y_prev = project_simplex(y)
            y = project_simplex(y)
        else:
            y = project_simplex((1 - alpha) * y + alpha * b_new)

        if use_nesterov_avg:
            w = alpha * (it + 1 + alpha_decay)
            a = ((it + 1) * a + w * b_new) / (it + 1 + w)
            a = project_simplex(a)

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 3:
                rng = np.random.default_rng(it * 7919)
                y = y + 0.05 * rng.standard_normal(n)
                y = project_simplex(y)
                stall = 0
            elapsed = time.monotonic() - t0
            if (it + 1) % 1000 == 0:
                print(f"        LP iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def nesterov_projected_gradient(a0, time_limit, T_start=3.0, T_end=500.0,
                                lr_init=0.06, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed max."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count)) / 2.0
            t_prev = (1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2)) / 2.0
            beta = (t_prev - 1.0) / t_curr
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()

            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(y, fft_size)
            W = rfft(w, fft_size)
            corr = irfft(np.conj(W) * A, fft_size)[:n]
            grad = 2.0 * corr

            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def local_refine(a0, time_limit, n_trials=50000):
    """Coordinate descent and pair modification for local refinement."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    trial = 0
    while trial < n_trials:
        if time.monotonic() - t0 > time_limit:
            break
        trial += 1

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        if rng.random() < 0.5:
            # Single coordinate reduction
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            idx_pick = idx[rng.integers(len(idx))]
            factors = [0.95, 0.90, 0.85, 0.98]
            for f in factors:
                a_t = a.copy()
                a_t[idx_pick] *= f
                a_t = project_simplex(a_t)
                sc = score_normalized(a_t)
                if sc < best_score - 1e-12:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    break
        else:
            # Pair modification
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            i1 = idx[rng.integers(len(idx))]
            i2 = idx[rng.integers(len(idx))]
            if i1 == i2 or a[i1] < 1e-14 or a[i2] < 1e-14:
                continue
            f1 = [0.9, 0.95, 0.85][rng.integers(3)]
            f2 = [0.9, 0.95, 0.85][rng.integers(3)]
            a_t = a.copy()
            a_t[i1] *= f1
            a_t[i2] *= f2
            a_t = project_simplex(a_t)
            sc = score_normalized(a_t)
            if sc < best_score - 1e-12:
                a = a_t
                best_score = sc
                best_a = a.copy()

        if trial % 5000 == 0:
            print(f"        Local refine trial={trial} ({time.monotonic()-t0:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_gaussian_init(n, center=0.4, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_sinusoidal_init(n, freq=1.0, phase=0.0, amp=0.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_linear_init(n, slope=1.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + slope * (2 * x - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_init(n, rng):
    a = rng.random(n) + 0.1
    return a / a.sum()


def make_two_gaussian_init(n, c1=0.3, c2=0.7, w=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - c1) / w) ** 2) + 0.7 * np.exp(-((x - c2) / w) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP best-response with Nesterov acceleration,
    projected gradient, local refinement, multi-length and multi-start search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        best_seq = make_gaussian_init(n_init, 0.4, 0.35)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # === Phase 1: LP refinement on parent (most important) ===
    print(f"\n=== Phase 1: LP Refine n={n_best} (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_iteration(best_seq, solver, t_alloc,
                                            alpha_decay=3.0, momentum=0.0)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: Nesterov PG for further refinement ===
    print(f"\n=== Phase 2: NG Refine (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_ng, _ = nesterov_projected_gradient(best_seq, t_alloc,
                                              T_start=10.0, T_end=800.0,
                                              lr_init=0.03, n_stages=6,
                                              steps_per_stage=1500)
        try_update(a_ng, "ng_refine")

    # === Phase 3: Second LP round with momentum ===
    print(f"\n=== Phase 3: LP with momentum (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_iteration(best_seq, solver, t_alloc,
                                         alpha_decay=4.0, momentum=0.3)
        try_update(a_lp2, "lp_momentum")

    # === Phase 4: Local refinement ===
    print(f"\n=== Phase 4: Local Refine (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_loc, _ = local_refine(best_seq, t_alloc, n_trials=80000)
        try_update(a_loc, "local")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [100, 200, 300, 500, -100, -200]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 4096:
            alt_lengths.append(alt_n)
    # Also try some round numbers
    for cn in [512, 768, 1024, 1280, 1536, 2048, 2560, 3072]:
        if 256 <= cn <= 4096 and cn not in alt_lengths:
            alt_lengths.append(cn)
    alt_lengths = sorted(set(alt_lengths))[:6]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_sub = min(rem * 0.25, 150.0)
        if t_sub < 30:
            break
        print(f"  n={alt_n}, budget={t_sub:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)

        # LP on this length
        a_alt, _ = nesterov_lp_iteration(init_alt, solver, t_sub * 0.7,
                                         alpha_decay=3.0)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = local_refine(a_alt, min(rem * 0.3, 20.0), n_trials=10000)
            try_update(a_alt2, f"loc_n{alt_n}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Inits (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    inits = [
        ("gauss_0.3", make_gaussian_init(n_best, 0.3, 0.25)),
        ("gauss_0.6", make_gaussian_init(n_best, 0.6, 0.25)),
        ("lin_pos", make_linear_init(n_best, 0.8)),
        ("lin_neg", make_linear_init(n_best, -0.8)),
        ("sin1", make_sinusoidal_init(n_best, 1.0, 0.0, 0.6)),
        ("sin2", make_sinusoidal_init(n_best, 2.0, 0.5, 0.4)),
        ("two_gauss", make_two_gaussian_init(n_best)),
        ("random", make_random_init(n_best, rng)),
    ]

    for name, init in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.15, 80.0)
        if t_sub < 15:
            break
        # Quick NG then LP
        a_sub, _ = nesterov_projected_gradient(init, t_sub * 0.4,
                                               T_start=2.0, T_end=200.0,
                                               lr_init=0.08, n_stages=5,
                                               steps_per_stage=1000)
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sub2, _ = nesterov_lp_iteration(a_sub, solver, min(rem * 0.4, 30.0),
                                              alpha_decay=3.0)
            try_update(a_sub2, name)

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 120.0)
        a_fin, _ = nesterov_lp_iteration(best_seq, solver, t_fin,
                                         alpha_decay=5.0)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final local refine ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin2, _ = local_refine(best_seq, min(rem * 0.5, 30.0), n_trials=20000)
        try_update(a_fin2, "final_loc")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
