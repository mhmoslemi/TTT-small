import numpy as np

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Full linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def nesterov_md(a0, time_limit, lr_init=0.05, top_k=30, verbose=True):
    """
    Nesterov-accelerated multiplicative weight updates (mirror descent)
    with multi-peak subgradient.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    # Nesterov momentum state
    a_prev = a.copy()
    a_momentum = a.copy()

    # Adaptive temperature for peak weighting
    base_temp = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-9:
                break

        # Compute multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        temp = base_temp * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Nesterov lookahead
        if it > 1:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * (it - 1) + 2.0)
            a_lookahead = a + beta * (a - a_prev)
        else:
            a_lookahead = a.copy()

        a_prev = a.copy()

        # Mirror descent step (multiplicative weight update)
        a_new = a_lookahead * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        a = a_new
        conv = fast_convolve(a)

        # Check running average
        if it % 200 == 0:
            avg_a = (a + best_a) * 0.5
            avg_score = 2.0 * n * fast_convolve(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score

                # Update convolution
                changes = 2.0 * best_delta * sensitivities
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    def get_unaff_max(k_start, k_end):
        if k_start > 0 and k_end < 2 * n - 2:
            return max(conv[:k_start].max(), conv[k_end + 1:].max())
        elif k_start > 0:
            return conv[:k_start].max()
        elif k_end < 2 * n - 2:
            return conv[k_end + 1:].max()
        else:
            return 0.0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = get_unaff_max(k_start, k_end)

            affected = conv[k_start:k_end + 1]
            sensitivities = a[k_start - i:k_end - i + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine steps around current value
            for factor in [0.995, 0.99, 0.985, 0.992, 1.008, 1.015, 1.01, 1.005]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * sensitivities
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-16:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * sensitivities
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30 or sweeps > 5:
                break
        else:
            no_improve = 0

    return best_a, best_score


def block_cd(a0, time_limit, block_size=8, verbose=True):
    """
    Block coordinate descent: optimize small blocks of coordinates simultaneously.
    This can escape local minima that single-coordinate CD gets stuck in.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for block_start in range(0, n, block_size):
            block_end = min(block_start + block_size, n)
            block = a[block_start:block_end].copy()
            if block.max() < 1e-15:
                continue

            # Try redistributing within the block
            # The block's contribution to the convolution depends on surrounding context
            # Simple approach: try a few redistribution patterns

            block_sum = block.sum()
            if block_sum < 1e-15:
                continue

            # Generate a few candidate redistributions
            candidates = []

            # 1: Uniform within block
            candidates.append(np.ones(block_size) * block_sum / block_size)

            # 2: Linear ramp up
            t = np.linspace(0.5, 1.5, block_size)
            c = t / t.sum() * block_sum
            candidates.append(c)

            # 3: Linear ramp down
            t = np.linspace(1.5, 0.5, block_size)
            c = t / t.sum() * block_sum
            candidates.append(c)

            # 4: Keep as is but with small perturbation
            candidates.append(block)

            best_candidate = block
            best_candidate_M = None

            for cand in candidates:
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand = cand / cs * block_sum

                # Compute effect on convolution
                delta = cand - block
                # The change in conv is convolve(a, delta) + convolve(delta, a) approximately
                # But for simplicity, just compute the new conv directly for the affected range
                new_a = a.copy()
                new_a[block_start:block_end] = cand

                # Only compute convolution in the affected range
                k_min = block_start
                k_max = min(block_end - 1 + n - 1, 2 * n - 2)

                # For efficiency, compute full conv (n=1024 is small enough)
                new_conv = fast_convolve(new_a)
                new_M = float(new_conv.max())

                if best_candidate_M is None or new_M < best_candidate_M:
                    best_candidate_M = new_M
                    best_candidate = cand.copy()

            if best_candidate_M is not None and best_candidate_M < M - 1e-14:
                a[block_start:block_end] = best_candidate
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def diverse_init(n, rng, seed_type=0):
    """Generate diverse initializations."""
    if seed_type == 0:
        # Raised cosine
        t = np.arange(n, dtype=np.float64)
        x = t / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif seed_type == 1:
        # Two humps
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif seed_type == 2:
        # Periodic with random period
        period = int(rng.integers(3, 15))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase + rng.uniform(0, 2 * np.pi))
    elif seed_type == 3:
        # Chirp
        t = np.arange(n, dtype=np.float64)
        freq = rng.uniform(1, 5)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t / n)
    elif seed_type == 4:
        # Notch (high at edges, low in middle)
        t = np.arange(n, dtype=np.float64)
        x = t / (n - 1)
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    else:
        # Random with smoothing
        a = np.abs(rng.standard_normal(n))
        # Simple moving average smoothing
        kernel_size = max(3, n // 200)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """
    Direct subgradient method on the original problem (not alternating).
    Uses the subgradient of max_k c[k] with respect to a directly.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    no_improve = 0
    top_k = 50

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1000:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-10:
                break

        # Get active peaks
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]

        # Weight peaks by how close they are to the max
        pv = conv[peaks]
        margin = M - pv.max() + 1e-10
        temp = 0.003 * M
        w = np.exp(-(M - pv) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient of max(c) at a:
        # d(c[k])/d(a[i]) = a[k-i] * 1[0<=k-i<n] + a[i-k] * 1[0<=i-k<n]
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            # a[k-i] for i in [max(0, k-n+1), min(n-1, k)]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # d(c[k])/d(a[i]) = a[k-i] + a[i-k] (if both in range)
            # a[k-i] for i in [i_start, i_end] = a[k-i_start : k-i_end+1][::-1]
            part1 = a[k - i_end:k - i_start + 1][::-1]
            # a[i-k] for i in [max(0,k), min(n-1, 2n-1-k)]
            i_start2 = max(0, k)
            i_end2 = min(n - 1, 2 * n - 1 - k)
            if i_start2 <= i_end2:
                part2 = np.zeros(n)
                part2[i_start2:i_end2 + 1] = a[i_start2 - k:i_end2 - k + 1]
            else:
                part2 = np.zeros(n)
            sg[i_start:i_end + 1] += w[idx] * part1
            if i_start2 <= i_end2:
                sg[i_start2:i_end2 + 1] += w[idx] * part2[i_start2:i_end2 + 1]

        # Multiplicative weight update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines FW, subgradient, coordinate descent, and block CD.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Uniform init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Quick FW to establish baseline ===
    remaining = deadline - time.monotonic()
    t_fw1 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 1: FW Global (budget={t_fw1:.0f}s, best={best_score:.6f}) ===")
    a_fw, s_fw = fw_global_optimize(best_seq, t_fw1, n_fw_per_round=200, verbose=True)
    best_seq, best_score = check_and_update(a_fw, best_seq, best_score, "fw1")

    # === Phase 2: Diversified FW from multiple inits ===
    remaining = deadline - time.monotonic()
    t_fw2_total = min(remaining * 0.2, 300.0)
    if t_fw2_total > 30:
        print(f"\n=== Phase 2: Diverse FW (budget={t_fw2_total:.0f}s, best={best_score:.6f}) ===")
        inits = [diverse_init(n_best, rng, i) for i in range(5)]
        t_each = t_fw2_total / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            print(f"  Diversified FW init {i}, budget={t_each:.0f}s")
            a_d, s_d = fw_global_optimize(init, t_each, n_fw_per_round=150, verbose=False)
            best_seq, best_score = check_and_update(a_d, best_seq, best_score, f"div_fw_{i}")

    # === Phase 3: Nesterov subgradient ===
    remaining = deadline - time.monotonic()
    t_md = min(remaining * 0.2, 250.0)
    if t_md > 30:
        print(f"\n=== Phase 3: Nesterov MD (budget={t_md:.0f}s, best={best_score:.6f}) ===")
        a_md, s_md = nesterov_md(best_seq, t_md, lr_init=0.04, top_k=40, verbose=True)
        best_seq, best_score = check_and_update(a_md, best_seq, best_score, "nesterov_md")

    # === Phase 4: Direct subgradient polish ===
    remaining = deadline - time.monotonic()
    t_sg = min(remaining * 0.15, 150.0)
    if t_sg > 20:
        print(f"\n=== Phase 4: Direct Subgradient (budget={t_sg:.0f}s, best={best_score:.6f}) ===")
        a_sg, s_sg = subgradient_polish(best_seq, t_sg, verbose=True)
        best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad")

    # === Phase 5: Coordinate descent ===
    remaining = deadline - time.monotonic()
    t_cd = min(remaining * 0.2, 200.0)
    if t_cd > 20:
        print(f"\n=== Phase 5: Coordinate Descent (budget={t_cd:.0f}s, best={best_score:.6f}) ===")
        a_cd, s_cd = coord_descent_optimal(best_seq, t_cd, verbose=True, label="main")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_main")

    # === Phase 6: Block CD ===
    remaining = deadline - time.monotonic()
    t_bc = min(remaining * 0.1, 80.0)
    if t_bc > 15:
        print(f"\n=== Phase 6: Block CD (budget={t_bc:.0f}s, best={best_score:.6f}) ===")
        a_bc, s_bc = block_cd(best_seq, t_bc, block_size=16, verbose=True)
        best_seq, best_score = check_and_update(a_bc, best_seq, best_score, "block_cd")

    # === Phase 7: Fine polishing ===
    remaining = deadline - time.monotonic()
    t_fine = min(remaining * 0.25, 150.0)
    if t_fine > 15:
        print(f"\n=== Phase 7: Fine Polish (budget={t_fine:.0f}s, best={best_score:.6f}) ===")
        a_fine, s_fine = fine_polish(best_seq, t_fine, verbose=True)
        best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_polish")

    # === Phase 8: Another round of CD + polish from best ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD+Polish (budget={remaining:.0f}s, best={best_score:.6f}) ===")
        t_cd2 = min(remaining * 0.5, 80.0)
        a_cd2, s_cd2 = coord_descent_optimal(best_seq, t_cd2, verbose=False, label="final")
        best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_final")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_fine2 = min(remaining * 0.7, 50.0)
            a_fine2, s_fine2 = fine_polish(best_seq, t_fine2, verbose=False)
            best_seq, best_score = check_and_update(a_fine2, best_seq, best_score, "fine_final")

    # === Phase 9: Try perturbation + quick re-optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 9: Perturbation + Re-optimize (budget={remaining:.0f}s) ===")
        n_perturb = min(3, int(remaining // 20))
        for p in range(n_perturb):
            perturbed = best_seq + rng.standard_normal(n_best) * 0.001
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_sub = min((deadline - time.monotonic()) * 0.3, 20.0)
            if t_sub > 5:
                a_p, s_p = coord_descent_optimal(perturbed, t_sub, verbose=False, label=f"pert_{p}")
                best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
