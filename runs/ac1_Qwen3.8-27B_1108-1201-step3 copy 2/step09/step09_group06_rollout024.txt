One approach: use a sequence whose Fourier transform has a flat magnitude. By the convolution theorem, if |F(a)| is flat, then the circular autocorrelation is a delta function. But we're dealing with linear convolution, not circular, and we want the convolution (not autocorrelation) to be flat.

So we can't make the convolution perfectly flat. The best we can do is minimize the peak-to-average ratio of the convolution.

from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s
    return b_val, float(M_var.value)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

        s = 1.0

    t0 = time.monotonic()

        if b_new is None:
