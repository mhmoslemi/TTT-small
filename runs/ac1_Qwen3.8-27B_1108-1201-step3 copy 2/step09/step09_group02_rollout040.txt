import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_gradient(a, c, tau):
    """
    Compute gradient of 2n * tau * logsumexp(c/tau) where c = conv(a,a).
    Uses FFT cross-correlation for O(n log n) computation.
    grad[i] = (4n/tau) * sum_k softmax(c[k]/tau)[k] * a[k-i]
    """
    n = len(a)
    c_max = c.max()
    c_shifted = c / tau - c_max / tau
    exp_vals = np.exp(c_shifted)
    w = exp_vals / exp_vals.sum()

    # Cross-correlation: result[i] = sum_j w[j] * a[j-i] (with a[j-i]=0 if out of [0,n-1])
    N = 1 << ((len(w) + n - 1) - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    result = np.fft.ifft(W * np.conj(A)).real
    g = result[:n]

    return (4.0 * n / tau) * g


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def golden_section_search(f, lo, hi, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]. Returns optimal x."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes with random coefficients."""
    n_harmonics = rng.integers(3, 15)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_harmonics, replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    # Temperature schedule
    c_max_initial = float(conv.max())
    tau_start = c_max_initial * 0.1
    tau_end = c_max_initial * 0.005
    total_planned_steps = 20000
    min_iter = 500

    while time.monotonic() - t0 < time_limit:
        step_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Adaptive temperature
        frac = min(1.0, step_count / total_planned_steps)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-10)

        # Recompute convolution periodically (for accuracy)
        if step_count % 100 == 1 or step_count == 1:
            conv = fast_convolve(a)
            current_score = 2.0 * n * float(conv.max())
            if current_score < best_score - 1e-14:
                best_score = current_score
                best_a = a.copy()

        # Compute smooth gradient
        grad = compute_smooth_gradient(a, conv, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** step_count)
        v_hat = v / (1.0 - beta2 ** step_count)

        # Adaptive learning rate
        effective_lr = lr / (1.0 + step_count * 0.0001)
        update = effective_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Normalized gradient step (maintain sum=1 approximately)
        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        # Check if improvement
        if step_count % 50 == 0:
            a = a_new
            conv_check = fast_convolve(a)
            sc = 2.0 * n * float(conv_check.max())
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                if verbose and step_count % 500 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    Adam {label} step={step_count} ({elapsed:.0f}s): "
                          f"score={best_score:.8f}, tau={tau:.6f}")
            else:
                # Decay learning rate if stuck
                lr *= 0.95
        else:
            a = a_new

        # Safety: ensure no NaN
        if not np.isfinite(a).all() or (a < 0).any():
            a = best_a.copy()
            break

    # Final check
    conv_final = fast_convolve(a)
    sc = 2.0 * n * float(conv_final.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def smooth_adam_phase2(a0, time_limit, verbose=False, label=""):
    """
    Second phase of smooth optimization with different hyperparameters
    and mirror descent flavor.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    beta1 = 0.8
    beta2 = 0.99
    eps = 1e-10
    lr = 0.005
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    c_max_initial = float(conv.max())
    tau = c_max_initial * 0.02

    while time.monotonic() - t0 < time_limit:
        step_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Decrease temperature slowly
        tau = max(tau * 0.9995, c_max_initial * 0.001)

        if step_count % 50 == 1:
            conv = fast_convolve(a)
            current_score = 2.0 * n * float(conv.max())
            if current_score < best_score - 1e-14:
                best_score = current_score
                best_a = a.copy()
                if verbose and step_count % 1000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    Adam2 {label} step={step_count} ({elapsed:.0f}s): "
                          f"score={best_score:.8f}")

        grad = compute_smooth_gradient(a, conv, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** step_count)
        v_hat = v / (1.0 - beta2 ** step_count)

        # Mirror descent style: multiplicative update
        log_update = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a * np.exp(-log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        a = a_new

        if not np.isfinite(a).all():
            a = best_a.copy()
            break

    conv_final = fast_convolve(a)
    sc = 2.0 * n * float(conv_final.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def fictitious_play_with_linesearch(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with Nesterov acceleration and line search."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0

    # Nesterov momentum
    v_ = np.zeros(n)
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Nesterov: x_{k+1} = (1-momentum)*x_k + momentum*(x_k + velocity) + step*(b - x_k)
        # Simplified: blend towards b with adaptive step
        alpha = 2.0 / (iters + 2.0)  # Harvey-like averaging
        v_ = momentum * v_ + (b - a)
        a_new = a + alpha * v_
        a_new = np.maximum(a_new, 0)
        cs = a_new.sum()
        if cs < 1e-10:
            a_new = a
        else:
            a_new = a_new / cs

        # Line search
        best_t = 0.5
        best_ls_score = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs2 = cand.sum()
            if cs2 < 1e-10:
                continue
            cand = cand / cs2
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = a_new
            else:
                a = a / cs
        else:
            a = a_new

        for cand in [a, a_new, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def targeted_perturbation(a, conv, rng, n_perturb=5):
    """Perturb coordinates that contribute most to peak lags."""
    n = len(a)
    a_pert = a.copy()

    # Find top peak lags
    top_k = min(5, 2 * n - 1)
    peaks = np.argpartition(conv, -top_k)[-top_k:]

    for k in peaks:
        k_int = int(k)
        # Find contributing indices
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 4)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for ii in picks:
            a_pert[ii] *= (1.0 + rng.standard_normal() * 0.08)

    a_pert = np.maximum(a_pert, 0)
    ps = a_pert.sum()
    if ps < 1e-10:
        return a.copy()
    return a_pert / ps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with small multiplicative steps for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_delta = 0.0
            cur_local_max = float(max(unaff_max, float(conv[k_lo:k_hi + 1].max())))

            for factor in [0.995, 0.99, 0.985, 0.97, 0.95, 1.05, 1.03, 1.015, 1.01, 1.005]:
                delta = v * (factor - 1.0)
                x = v + delta
                if x < -1e-15:
                    continue
                nv = c_vals.copy() + 2.0 * x * a_partners
                if has_quad:
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                if local_max < cur_local_max - 1e-16:
                    cur_local_max = local_max
                    best_delta = delta

            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                current_sum += best_delta

                changes = 2.0 * best_delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = best_delta * (2.0 * v + best_delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Smooth Adam gradient (log-sum-exp + FFT gradient)
    2. LP fictitious play with Nesterov
    3. Exact coordinate descent
    4. Targeted basin hopping
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Smooth Adam optimization ===
    print("\n=== Phase 1: Smooth Adam (log-sum-exp + FFT) ===")
    t1_start = time.monotonic()

    n_values = [1024]
    if parent is not None:
        np_ = len(parent)
        n_values = [np_]
        for alt in [768, 1280, 512]:
            if alt != np_ and 256 <= alt <= 2048:
                n_values.append(alt)

    # Focus on the parent length first
    n_main = n_values[0]

    # Initialize from parent with small perturbations
    if parent is not None:
        inits = []
        inits.append(('parent', interpolate_seq(parent, n_main)))
        # Small perturbation
        pert1 = parent + rng.standard_normal(n_main) * 0.01
        pert1 = np.maximum(pert1, 0)
        inits.append(('pert1', pert1 / pert1.sum()))
        pert2 = parent + rng.standard_normal(n_main) * 0.03
        pert2 = np.maximum(pert2, 0)
        inits.append(('pert2', pert2 / pert2.sum()))
        # Flat spectrum
        inits.append(('flat', init_flat_spectrum(n_main, rng)))
        # Fourier mix
        inits.append(('fourier', init_fourier_mix(n_main, rng)))
    else:
        inits = [('flat', init_flat_spectrum(n_main, rng)),
                 ('cosine', init_cosine(n_main, rng))]

    phase1_best_a = None
    phase1_best_s = float('inf')

    for itype, initial in inits[:5]:
        rem = deadline - time.monotonic() - 600
        if rem < 10:
            break
        t_alloc = min(rem * 0.3, 120.0)
        print(f"  Adam phase1: {itype}, n={n_main}, budget={t_alloc:.0f}s")
        a_opt, s_opt = smooth_adam_optimize(initial, t_alloc, verbose=True, label=f"p1_{itype}")
        if s_opt < phase1_best_s:
            phase1_best_s = s_opt
            phase1_best_a = a_opt.copy()
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p1_{itype}")

    # Second Adam phase with different params
    if phase1_best_a is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  Adam phase2: n={n_main}, budget={t_alloc:.0f}s")
            a_opt2, s_opt2 = smooth_adam_phase2(phase1_best_a, t_alloc,
                                                 verbose=True, label="p2")
            if s_opt2 < phase1_best_s:
                phase1_best_s = s_opt2
                phase1_best_a = a_opt2.copy()
            best_seq, best_score = check_and_update(a_opt2, best_seq, best_score, "p2_adam")

    print(f"  Phase 1 complete. Best so far: {best_score:.8f}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None:
        solver = get_lp_solver()
        print(f"  Using LP solver: {solver}")
        n_best = len(best_seq)

        fp_initial = best_seq.copy()

        if rem > 60:
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  FP n={n_best}, budget={t_alloc:.0f}s")
            a_fp, score_fp, iters = fictitious_play_with_linesearch(
                fp_initial, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "fp")

            # Post-FP coordinate descent
            rem = deadline - time.monotonic() - 250
            if rem > 30:

                a_cd, s_cd, _ = exact_coord_descent(a_fp, t_cd, label="post_fp")

    # === Phase 4: Targeted Basin Hopping ===
    print(f"\n=== Phase 4: Targeted Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 60:
            n_hops = min(6, int(rem / 30))
            for hop in range(n_hops):
                rem = deadline - time.monotonic() - 50
                if rem < 20:
                    break
                conv_cur = fast_convolve(best_seq)
                perturbed = targeted_perturbation(best_seq, conv_cur, rng)
                # Also try pure noise
                if rng.random() < 0.3:
                    pert_noise = best_seq + rng.standard_normal(n_best) * 0.02
                    pert_noise = np.maximum(pert_noise, 0)
                    pert_noise = pert_noise / pert_noise.sum()
                    perturbed = pert_noise

                t_alloc = min(rem * 0.5, 60.0)
                # Use smooth Adam for quick re-optimization
                a_hop, s_hop = smooth_adam_optimize(perturbed, t_alloc * 0.6, label=f"hop{hop}")
                # Then polish with CD
                rem2 = deadline - time.monotonic() - 40
                if rem2 > 10:
                    a_hop2, s_hop2, _ = exact_coord_descent(a_hop, min(rem2 * 0.4, 30.0),
                                                            label=f"hop{hop}_cd")
                    a_hop = a_hop2
                    s_hop = min(s_hop, s_hop2)

                best_seq, best_score = check_and_update(a_hop, best_seq, best_score, f"hop{hop}")

    # === Phase 5: Final Polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_fine = min(rem * 0.5, 60.0)
            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="final_fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another ECD round
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd2 = min(rem * 0.6, 40.0)
            print(f"  ECD-polish, budget={t_cd2:.0f}s")
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
