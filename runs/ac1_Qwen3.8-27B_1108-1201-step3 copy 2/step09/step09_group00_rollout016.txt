def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response_fixed_a(a_fixed, solver):

    prob = cp.Problem(cp.Minimize(M_var), constraints)

def bidirectional_lp_alternation(a0, time_limit, solver, max_iters=200000):
    """Alternating minimization solving both LP subproblems with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best b given current y_a
        b_new, M1 = solve_lp_best_response_fixed_a(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            # Nesterov combined point for b
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            b = normalize(b)

        # Step 2: Best a given current y_b
        a_new, M2 = solve_lp_best_response_fixed_b(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1.0 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
            a = ((it + 1) * a + (alpha * (it + 3)) * a_new) / (it + 1 + alpha * (it + 3))
            a = normalize(a)

        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_b = 2.0 * n * fast_convolve(b).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 10:
                noise_scale = 0.02 / math.sqrt(stall)
                y_a = y_a + rng.standard_normal(n) * noise_scale
                y_a = normalize(y_a)
                y_b = y_b + rng.standard_normal(n) * noise_scale
                y_b = normalize(y_b)
                stall = 0

            if (it + 1) % 2000 == 0:
                print(f"    BiDirLP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def peak_flattening_lp(a0, solver, n_steps=500):
    """Perform LP-based peak flattening: at each step, solve an LP to find
    the direction that most reduces the peak while keeping all peaks bounded."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > 60:
            break

        b = fast_convolve(a)
        M = b.max()
        size = 2 * n - 1

        # Build the linearization matrix: gradient of b[k] w.r.t. a[j] is a[k-j]
        # For the LP: minimize sum of (linearized b[k]) over active peaks
        # subject to sum(delta)=0, a+delta >= 0, and linearized b[k] <= M for all k

        # Only include positions where b[k] > M * 0.9 (near the peak)
        threshold = M * 0.9
        active_k = np.where(b >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(b))])

        # Build constraint matrix for active positions
        # Linearized b[k] at a: b[k] + 2 * sum_j a[k-j] * delta_j
        # We want: b[k] + 2 * sum_j a[k-j] * delta_j <= M + epsilon (for all active k)
        # And minimize: sum over active k of (b[k] + 2 * sum_j a[k-j] * delta_j) / len(active)

        m_active = len(active_k)
        # G[k_idx, j] = 2 * a[k - j] if k-j in [0, n-1], else 0
        G = np.zeros((m_active, n))
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            G[i, idx] = 2.0 * a[k - idx]

        delta_var = cp.Variable(n)
        eps_var = cp.Variable()

        # Minimize mean of linearized peaks + epsilon
        obj = cp.Minimize(cp.sum(G @ delta_var) / m_active + eps_var)

        constraints = [
            cp.sum(delta_var) == 0,
            a + delta_var >= 0,
            G @ delta_var <= m_active * eps_var,  # each linearized peak <= M + eps
        ]

        # Also constrain non-active peaks to not exceed current M
        inactive_k = np.where(b < threshold)[0]
        # Only add constraints for positions where b is within 20% of M
        near_inactive = inactive_k[(b[inactive_k] >= M * 0.7)]
        if len(near_inactive) > 0 and len(near_inactive) < 500:
            G_2 = np.zeros((len(near_inactive), n))
            for i, k in enumerate(near_inactive):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                G_2[i, idx] = 2.0 * a[k - idx]
            constraints.append(G_2 @ delta_var <= 0)  # don't increase these

        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                delta_val = np.asarray(delta_var.value, dtype=np.float64)
                # Take a step in this direction
                a_trial = a + 0.5 * delta_val
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                else:
                    # Try smaller step
                    a_trial = a + 0.2 * delta_val
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
        except Exception:
            pass

    return best_a, best_score


def fourier_parametric_refine(a0, n_freq=20, time_limit=120):
    """Refine in the Fourier domain: optimize a small number of frequency coefficients."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Get DFT
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    # Work with the first n_freq+1 coefficients (DC + n_freq positive freqs)
    # A[0] is real, A[1..n_freq] are complex
    # Total free parameters: 1 + 2*n_freq = 2*n_freq + 1

    n_free = 2 * n_freq + 1
    # Initial values
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag

    # Remaining coefficients stay fixed
    A_rest = A[n_freq + 1:].copy()

    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest

        # Inverse DFT
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()

    # Optimize using Nelder-Mead (derivative-free, good for low-dim)
    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 5000, 'xatol': 1e-14, 'fatol': 1e-14,
                     'adaptive': True}
        )
        x_opt = res.x
        # Reconstruct
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass

    # Also try with BFGS using numerical gradients
    if time.monotonic() - t0 < time_limit - 5:
        try:
            res2 = minimize(
                fourier_obj, x_opt, method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12}
            )
            x_opt2 = res2.x
            A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
            A_new[0] = x_opt2[0]
            for i in range(1, n_freq + 1):
                A_new[i] = x_opt2[2 * i - 1] + 1j * x_opt2[2 * i]
            A_new[n_freq + 1:] = A_rest
            a_candidate = irfft(A_new, fft_size)[:n]
            a_candidate = np.maximum(a_candidate, 0)
            s = a_candidate.sum()
            if s > 1e-10:
                a_candidate = a_candidate / s
                sc = 2.0 * n * fast_convolve(a_candidate).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
        except Exception:
            pass

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_stages=None):
    """L-BFGS on the LogSumExp-smoothed objective in the simplex (log-parametrization)."""
    n = len(a0)
    a = normalize(a0)
    # Log-parametrization: a = exp(z) / sum(exp(z))
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = max(20, int(remaining * 20))

        def obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse

        def grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            # Gradient w.r.t. a: 2 * sum_k w[k] * a[k-j] for each j
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            # Chain rule for softmax: grad_z = a * (grad_a - sum(grad_a * a))
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        try:
            res = minimize(
                obj, z, jac=grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def symmetric_lp_optimize(a0, solver, time_limit):
    """Optimize under the symmetry constraint a[i] = a[n-1-i]."""
    n = len(a0)
    a = normalize(a0)
    # Enforce symmetry
    a_sym = 0.5 * (a + a[::-1])
    a_sym = normalize(a_sym)
    half_n = (n + 1) // 2

    best_a = a_sym.copy()
    best_score = 2.0 * n * fast_convolve(a_sym).max()
    t0 = time.monotonic()

    # Optimize only the first half (the second half is determined)
    a_half = a_sym[:half_n].copy()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        # Reconstruct full symmetric sequence
        a_full = np.zeros(n)
        a_full[:half_n] = a_half
        a_full[half_n:] = a_half[:n - half_n][::-1]
        a_full = normalize(a_full)

        # Compute convolution and find best response under symmetry
        b = fast_convolve(a_full)
        M = b.max()

        # LP: find symmetric a' minimizing max conv(a', a')
        # Parameterize by first half only
        hp = (n + 1) // 2
        a_var = cp.Variable(hp, nonneg=True)

        # Build the symmetric convolution constraints
        # conv(a', a')[k] for symmetric a'
        # This is complex to build as an LP directly, so use the full formulation
        # with the symmetry constraint a[j] = a[n-1-j]

        # Alternative: just use the full LP with added symmetry constraints
        a_full_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(a_full)
        constraints = [
            T @ a_full_var <= M_var,
            cp.sum(a_full_var) == 1,
        ]
        # Add symmetry constraints
        for j in range(half_n):
            other = n - 1 - j
            if other < j:
                break
            constraints.append(a_full_var[j] == a_full_var[other])

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_full_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                a_val = a_val / a_val.sum()
                sc = 2.0 * n * fast_convolve(a_val).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_val.copy()
                    a_half = a_val[:half_n].copy()
        except Exception:
            break

    return best_a, best_score


def simple_subgradient(a0, time_limit, n_steps=30000):
    """Simple subgradient method with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Find all peaks within 0.5% of max
        threshold = M * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.95
                no_improve = 0
        lr = max(1e-7, lr)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    else:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses bidirectional LP alternation, peak-flattening LP, Fourier parametric
    refinement, very-low-T LogSumExp L-BFGS, subgradient method, and
    systematic multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Bidirectional LP alternation ===
    print(f"\n=== Phase 1: Bidirectional LP (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_lp, sc_lp = bidirectional_lp_alternation(best_seq, t1, solver)
        try_update(a_lp, "bidi_lp")

    # === Phase 2: Peak flattening LP ===
    print(f"\n=== Phase 2: Peak Flattening LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        a_pf, sc_pf = peak_flattening_lp(best_seq, solver, n_steps=300)
        try_update(a_pf, "peak_flat_lp")

    # === Phase 3: Fourier parametric refinement ===
    print(f"\n=== Phase 3: Fourier Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 20:
        for n_freq in [16, 32, 64]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t3_sub = min(rem * 0.3, 40.0)
            a_four, sc_four = fourier_parametric_refine(best_seq, n_freq=n_freq, time_limit=t3_sub)
            try_update(a_four, f"fourier_K{n_freq}")

    # === Phase 4: LogSumExp L-BFGS (very low T) ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, 400.0)
    if t4 > 30:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        print(f"  T_stages: {T_stages}, budget={t4:.0f}s")
        a_lb, sc_lb = lbgfs_logsumexp(best_seq, t4, T_stages=T_stages)
        try_update(a_lb, "lbgfs_lowT")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 10:
        a_sg, sc_sg = simple_subgradient(best_seq, t5, n_steps=20000)
        try_update(a_sg, "subgrad")

    # === Phase 6: Second peak flattening ===
    print(f"\n=== Phase 6: Peak Flattening Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 120.0)
    if t6 > 15:
        a_pf2, sc_pf2 = peak_flattening_lp(best_seq, solver, n_steps=400)
        try_update(a_pf2, "peak_flat_2")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_candidates = []
        # Try several n values around the current best
        base_n = n_best
        for mult in [0.75, 0.85, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 200 <= cand_n <= 5000 and cand_n not in n_candidates:
                n_candidates.append(cand_n)

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if cand_n == n_best:
                continue
            t_sub = min(rem * 0.2, 80.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
            init_cand = interpolate_sequence(best_seq, cand_n)
            # Quick LP + L-BFGS
            a_cand, _ = bidirectional_lp_alternation(init_cand, t_sub * 0.4, solver)
            try_update(a_cand, f"lp_n{cand_n}")
            rem = deadline - time.monotonic()
            if rem > 20:
                T_st4 = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
                a_cand2, _ = lbgfs_logsumexp(best_seq if len(best_seq) == cand_n else a_cand,
                                           min(rem * 0.4, 60.0), T_stages=T_st4)
                try_update(a_cand2, f"lbgfs_n{cand_n}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Short subgradient
        t_sg = min(rem * 0.2, 30.0)
        if t_sg > 10:
            a_final_sg, _ = simple_subgradient(best_seq, t_sg, n_steps=10000)
            try_update(a_final_sg, "final_sg")

        # Very low T L-BFGS
        rem = deadline - time.monotonic()
        if rem > 20:
            t_fin = min(rem * 0.4, 60.0)
            a_fin, _ = lbgfs_logsumexp(best_seq, t_fin,
                                       T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
            try_update(a_fin, "final_lbgfs")

        # Final peak flattening
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin_pf, _ = peak_flattening_lp(best_seq, solver, n_steps=200)
            try_update(a_fin_pf, "final_pf")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        # Also call evaluate_sequence to confirm
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
