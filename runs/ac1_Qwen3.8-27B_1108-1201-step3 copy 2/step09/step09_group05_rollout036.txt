def norm_sequence(a):
    a = np.maximum(a, 0).astype(np.float64)

        return np.ones(len(a)) / len(a)

def build_palindromic(half):
    """Build full palindromic sequence from half sequence."""
    n_half = len(half)
    n = 2 * n_half - 1
    full = np.empty(n)
    full[:n_half] = half
    full[n_half:] = half[:n_half-1][::-1]
    return full

def palindromic_half(a):
    """Extract independent half of palindromic sequence."""
    n = len(a)

    return np.allclose(a, a[::-1], atol=tol)

def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    return (a + a[::-1]) / 2

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS

def subgradient_refine(a0, time_limit, n_steps=100000):
    """Subgradient method with averaging for the exact (unsmoothed) objective."""
    n = len(a0)
    a = norm_sequence(a0)
    a = make_palindromic(a)
    a = norm_sequence(a)
    
    n_half = (n + 1) // 2
    y = palindromic_half(a).copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Mirror descent on the simplex using the half-sequence
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        # Subgradient of max c[k] w.r.t. a
        grad_a = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = mir = min(n - 1, k_star)
        for i in range(js, je + 1):
            grad_a[i] = 2.0 * a[k_star - i]
        
        # Project to half
        grad_half = np.zeros(n_half)
        for j in range(n_half):
            grad_half[j] = grad_a[j]
            if n - 1 - j != j:
                grad_half[j] += grad_a[n - 1 - j]
        
        # Mirror descent step on the simplex
        eta = 1.0 / np.sqrt(it + 1)
        y = y - eta * grad_half / (np.max(np.abs(grad_half)) + 1e-10)
        
        # Project to simplex
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n_half) / n_half
        else:
            y = y / sy
        
        a_candidate = build_palindromic(y)
        a_candidate = norm_sequence(a_candidate)
        
        score = 2.0 * n * fast_convolve(a_candidate).max()
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
        
        a = a_candidate
    
    return best_a, best_score

def parametric_init(n, rng, mode=0):
    """Create parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Bell (Gaussian)
        a = np.exp(-4 * (x - 0.5)**2 / 0.1)
    elif mode == 1:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = a ** 2
    elif mode == 2:
        # Triangle
        a = 1 - 2 * np.abs(x - 0.5)
    elif mode == 3:
        # Flat top with rounded edges
        w = 0.3
        a = np.ones(n)
        mask = (x < w) | (x > 1-w)
        a[mask] = 0.5 * (1 + np.cos(np.pi * np.minimum(x[mask], 1-x[mask]) / w))
    elif mode == 4:
        # Sine window
        a = np.sin(np.pi * x) ** 2
    elif mode == 5:
        # Dual bell
        a = np.exp(-4*(x-0.3)**2/0.05) + np.exp(-4*(x-0.7)**2/0.05)
    elif mode == 6:
        # Power law
        a = (1 - np.abs(2*x - 1)) ** 3
    else:
        a = 1 + 0.2 * np.sin(2*np.pi*rng.random()*5*x)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    # Let's compute conv(a, p) first: conv(a,p)[m] = sum_j a[j] * p[m-j]
    # result[i] = sum_k p[k] * a[k-i] = sum_j a[j] * p[j+i] (letting j=k-i)
    # 
    # Now, conv(a,p)[j+i] = sum_m a[m] * p[j+i-m]
    # That's not directly what we want.
    #
    # What about conv(p, a)?
    # conv(p,a)[m] = sum_k p[k] * a[m-k]
    # conv(p,a)[2*(n-1)-1+i]... hmm this is getting messy.
    
    # Let me just do it the direct way with numpy broadcasting for small n:
    # For n <= 2000, O(n^2) with numpy is fast enough (~1-2 ms for n=1000)
    
    # Create index matrix
    # p has length 2n-1, indexed 0 to 2n-2
    # a has length n, indexed 0 to n-1
    # We want: for each i in 0..n-1: sum over k where 0<=k and 0<=k-i<n
    # i.e., max(0,i) <= k <= min(2n-2, i+n-1)
    
    # Vectorized: for each i, the valid k range is [max(0,i), min(2n-2, i+n-1)]
    # And we sum p[k] * a[k-i]
    
    # This is equivalent to: for each valid (i, j) pair where j = k-i, 0<=j<n:
    #   contribute p[j+i] * a[j] to result[i]
    
    # So result[i] = sum_j p[j+i] * a[j] for j=0..n-1 (where j+i < len(p))
    
    # This is: result = a @ P where P[i,j] = p[i+j]
    # P is a Hankel matrix!
    
    # For FFT: result[i] = sum_j a[j] * p[j+i]
    # = cross_correlation of a and p at lag i
    # = ifft(fft(a) * conj(fft(p)))[i]... let me verify
    
    # Standard: cross_corr[x,y][n] = sum_m x[m+n] * conj(y[m])
    # In the real case: cross_corr[x,y][n] = sum_m x[m+n] * y[m]
    # So cross_corr[p, a][i] = sum_m p[m+i] * a[m] = sum_j p[j+i] * a[j] = result[i] ✓
    
    # And cross_corr[p,a] = ifft(conj(fft(p)) * fft(a))... 
    # Actually: cross_corr[x,y] = ifft(fft(x) * conj(fft(y)))... 
    
    # Let me just use the definition:
    # FFT of cross-correlation: FFT[xcorr(x,y)] = FFT(x) * conj(FFT(y))
    # where xcorr(x,y)[n] = sum_m x[m+n] * y[m]** (for complexes)
    # For reals: xcorr(x,y)[n] = sum_m x[m+n] * y[m]
    
    # So: result = irfft(rfft(p, fft_size) * np.conj(rfft(a, fft_size)), fft_size)
    
    fft_size = 1 << (2*n - 1).bit_length()  # Need at least 2n-1 + n - 1 = 3n-2... 
    # Actually for cross-correlation, we need fft_size >= max(len(p), len(a)) + min(len(p), len(a)) - 1
    # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    
    P = rfft(p, fft_size)
