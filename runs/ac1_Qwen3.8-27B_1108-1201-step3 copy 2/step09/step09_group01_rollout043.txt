def fast_convolve(a):
    n = int(len(a))
    if n == 0:
        return np.array([])
    size = 2*n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    c = irfft(A * A, L)
    c = c[:size]
    # remove small negatives
    c = np.where(c > 0.0, c, 0.0)
    return c

def peak_targeted_perturb(a, rng, strength=0.05):
   a=normalize; c=fast_convolve(a); M=c.max(); mean=c.mean(); gap=0.1*(M-mean); thresh=M-gap
   pressure=np.zeros(len(a))
   # To avoid O(peaks*n^2), use gradient-like pressure: p=soft weights, then contribution to each peak? Below approximate.
   p=np.maximum(c-thresh,0)/max(1e-15,(c-thresh).sum())
   # pressure_i = sum_k p_k a_{k-i}
   g=corr_grad(p,a)
   # Reduce high g, increase low
   g_norm=(g - g.min()) / max(1e-15, g.max()-g.min())
   noise=rng.standard_normal(len(a))*0.2
   a_new=a*(1.0 - strength*g_norm + 0.3*strength*noise)
   a_new=np.maximum(a_new,0); return normalize

"""
Search for low-peak non-negative step-function sequences.

Objective minimized:
    2*n*max(conv(a,a))/(sum(a))^2.

The routine combines:
  - FFT-based exact convolution and gradients,
  - active-set top-K L-BFGS optimization on a simplex softmax,
  - LSE-smoothed max L-BFGS optimization,
  - linearized least-squares / LP local steps from scipy.optimize.linprog,
  - multiplicative-exponentiated gradient peak balancing,
  - fast coordinate descent polishing,
  - random / peak-targeted restarts and limited length exploration.

The previous construction is used as the main seed when available.
"""

import time
import numpy as np

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = int(len(a))
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    c = irfft(A * A, L)
    c = c[:size]
    c = np.where(c > 0.0, c, 0.0)
    return c


def corr_grad(p, a):
    """
    Return g_i = sum_l a_l p_{l+i}, i=0..n-1.
    This is used for gradients of convolution objectives.
    """
    n = int(len(a))
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    P = rfft(p, L)
    g = irfft(np.conj(A) * P, L)
    return g[:n]


def normalize_seq(a):
    """Clip to nonnegative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        n = int(a.size)
        if n <= 0:
            return np.array([1.0], dtype=np.float64)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def exact_score_fast(a):
    """Scale-invariant objective computed directly from FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-300:
        return np.inf
    c = fast_convolve(a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    val = 2.0 * float(n) * m / (s * s)
    if not np.isfinite(val):
        return np.inf
    return float(val)


def update_best(a, best_a, best_score, label=""):
    """Evaluate a candidate and update global best if better."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return best_a, best_score
    s = float(a.sum())
    if s < 1e-300:
        return best_a, best_score
    a = a / s
    sc = exact_score_fast(a)
    if np.isfinite(sc) and sc < best_score:
        best_a = a.copy()
        best_score = float(sc)
        print(f"NEW BEST [{label}]: {sc:.7f} n={len(a)}")
    return best_a, best_score


def z_from_a(a, base):
    """
    Simplex parameterization using a fixed base coordinate.
    If a_base = 1/(1+sum exp z_j), then a_j = a_base exp z_j for j != base.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    if base < 0 or base >= n:
        base = int(np.argmax(a))
    if a[base] <= 1e-300:
        a = a.copy()
        a[base] = 1e-300
    ratios = a / a[base]
    mask = np.ones(n, dtype=bool)
    mask[base] = False
    z = np.log(np.maximum(ratios[mask], 1e-300))
    z = np.where(np.isfinite(z), z, -50.0)
    return np.clip(z, -50.0, 50.0)


def softmax_from_z(z, n, base):
    """Recover a probability vector from simplex parameterization."""
    n = int(n)
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    if z.size != n - 1:
        return np.ones(n, dtype=np.float64) / float(n)
    e = np.exp(z - float(np.max(z)))
    denom = 1.0 + float(e.sum())
    a_base = 1.0 / denom
    mask = np.ones(n, dtype=bool)
    mask[base] = False
    a = np.zeros(n, dtype=np.float64)
    a[mask] = e * a_base
    a[base] = a_base
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def smooth_valgrad(z, beta, n, base):
    """
    Return [value, gradient] for logsumexp(beta convolution)/beta
    under the simple base-index simplex parameterization.
    """
    n = int(n)
    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)
    if z.size != n - 1:
        return 1e300, np.zeros(max(0, n - 1), dtype=np.float64)

    a = softmax_from_z(z, n, base)
    c = fast_convolve(a)
    if c.size == 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    beta = float(beta)
    if beta <= 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    xc = beta * c
    m = float(np.max(xc))
    w = np.exp(xc - m)
    Z = float(w.sum())
    if Z <= 1e-300:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        Z = 1.0
    p = w / Z
    val = (m + float(np.log(Z))) / beta

    g = corr_grad(p, a)
    grad_a = 2.0 * g
    s_grad = float(np.dot(a, grad_a))
    mask = np.ones(n, dtype=bool)
    mask[base] = False
    grad_z = a[mask] * (grad_a[mask] - s_grad)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n - 1, dtype=np.float64)
    if not np.isfinite(val):
        val = 1e300
    return float(val), grad_z


def topk_valgrad(z, idx, n, base):
    """
    Return [value, gradient] for the mean of a fixed active top-K set
    of convolution coefficients.
    """
    n = int(n)
    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)
    if z.size != n - 1:
        return 1e300, np.zeros(max(0, n - 1), dtype=np.float64)

    a = softmax_from_z(z, n, base)
    c = fast_convolve(a)
    idx = np.asarray(idx, dtype=np.int64)
    if idx.size == 0:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    val = float(np.mean(c[idx]))

    p = np.zeros(c.size, dtype=np.float64)
    p[idx] = 1.0 / float(idx.size)
    g = corr_grad(p, a)
    grad_a = 2.0 * g
    s_grad = float(np.dot(a, grad_a))
    mask = np.ones(n, dtype=bool)
    mask[base] = False
    grad_z = a[mask] * (grad_a[mask] - s_grad)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n - 1, dtype=np.float64)
    if not np.isfinite(val):
        val = 1e300
    return float(val), grad_z


def run_minimize_segment(fun, z, args, maxiter=80):
    """Run one bounded L-BFGS segment and return the final x."""
    if not SCIPY_OPT_OK:
        return z
    z = np.asarray(z, dtype=np.float64)
    try:
        res = minimize(
            fun,
            z,
            args=args,
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-14,
                "gtol": 1e-7,
                "maxls": 40,
            },
        )
        x = np.asarray(res.x, dtype=np.float64)
        if x.size == z.size and np.all(np.isfinite(x)):
            return x
    except Exception:
        pass
    return z


def run_lbfgs_schedule(a0, time_budget, betas=None):
    """Annealed LSE-smoothed max L-BFGS."""
    a0 = normalize_seq(a0)
    n = int(a0.size)
    if n <= 2 or not SCIPY_OPT_OK or time_budget < 5:
        return a0.copy(), exact_score_fast(a0)

    deadline = time.monotonic() + float(time_budget)
    a = a0.copy()
    best_a = a.copy()
    best_score = exact_score_fast(a)

    if betas is None:
        betas = (5000.0, 15000.0, 40000.0)
    betas = tuple(float(x) for x in betas)
    per = max(10.0, float(time_budget) / float(len(betas)))

    for beta in betas:
        beta_deadline = min(deadline, time.monotonic() + per)
        base = int(np.argmax(a))
        z = z_from_a(a, base)
        cur_score = exact_score_fast(a)
        outer = 0

        while time.monotonic() < beta_deadline - 1.0 and outer < 12:
            outer += 1
            seg_deadline = min(beta_deadline, time.monotonic() + max(5.0, per * 0.25))
            inner = 0
            improved = False

            while time.monotonic() < seg_deadline - 0.5 and inner < 100:
                z_old = z
                z_new = run_minimize_segment(smooth_valgrad, z, (beta, n, base), 80)

                if z_new is z_old or np.allclose(z_new, z_old, rtol=1e-12, atol=1e-12):
                    break

                a_new = softmax_from_z(z_new, n, base)
                sc = exact_score_fast(a_new)

                if np.isfinite(sc) and sc <= cur_score + 1e-8:
                    a = a_new
                    cur_score = sc
                    z = z_new
                    improved = True
                    inner += 1
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                else:
                    z = z_from_a(a, base)
                    break

            if not improved and outer >= 3:
                break

        a = best_a.copy()
        a = normalize_seq(a)

    a = best_a.copy()
    a = normalize_seq(a)
    sc = exact_score_fast(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return a, best_score


def run_topk_lbfgs(a0, time_budget, K_list=None):
    """
    Active-set optimization: repeatedly fix the top K convolution peaks
    and minimize their mean with L-BFGS on the simplex.
    """
    a0 = normalize_seq(a0)
    n = int(a0.size)
    if n <= 2 or not SCIPY_OPT_OK or time_budget < 5:
        return a0.copy(), exact_score_fast(a0)

    deadline = time.monotonic() + float(time_budget)
    a = a0.copy()
    best_a = a.copy()
    best_score = exact_score_fast(a)

    if K_list is None:
        K_list = (16, 32, 64, 128)
    K_list = tuple(int(k) for k in K_list if int(k) >= 1)
    if not K_list:
        K_list = (32,)

    per = max(10.0, float(time_budget) / float(len(K_list)))

    for K in K_list:
        k_deadline = min(deadline, time.monotonic() + per)
        outer = 0

        while time.monotonic() < k_deadline - 1.0 and outer < 12:
            outer += 1
            c = fast_convolve(a)
            kuse = min(K, int(c.size))
            if kuse <= 0:
                break
            idx = np.argsort(c)[::-1][:kuse].astype(np.int64, copy=False)
            base = int(np.argmax(a))
            z = z_from_a(a, base)
            cur_score = exact_score_fast(a)

            seg_deadline = min(k_deadline, time.monotonic() + max(5.0, per * 0.25))
            inner = 0
            improved = False

            while time.monotonic() < seg_deadline - 0.5 and inner < 100:
                z_old = z
                z_new = run_minimize_segment(topk_valgrad, z, (idx, n, base), 80)

                if z_new is z_old or np.allclose(z_new, z_old, rtol=1e-12, atol=1e-12):
                    break

                a_new = softmax_from_z(z_new, n, base)
                sc = exact_score_fast(a_new)

                if np.isfinite(sc) and sc <= cur_score + 1e-8:
                    a = a_new
                    cur_score = sc
                    z = z_new
                    improved = True
                    inner += 1
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                else:
                    z = z_from_a(a, base)
                    break

            if not improved and outer >= 3:
                break

        a = best_a.copy()
        a = normalize_seq(a)

    a = best_a.copy()
    a = normalize_seq(a)
    sc = exact_score_fast(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return a, best_score


def solve_linprog_lp(c_obj, A_ub, b_ub, A_eq, b_eq, bounds):
    """Solve an LP with scipy if available."""
    if not SCIPY_OPT_OK:
        return None
    try:
        return linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": 2.5, "presolve": True},
        )
    except Exception:
        pass
    try:
        return linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        pass
    try:
        return linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="revised simplex",
        )
    except Exception:
        return None


def lp_linearize(a0, time_budget, top_k=64, max_step_frac=0.2):
    """
    Local nonconvex step: linearize selected peak convolutions around a,
    solve an LP for a step v with sum(v)=0 and nonnegativity, then line-search.
    """
    a0 = normalize_seq(a0)
    n = int(a0.size)
    if n <= 1 or n > 2048 or not SCIPY_OPT_OK or time_budget < 5:
        return a0.copy(), exact_score_fast(a0)

    deadline = time.monotonic() + float(time_budget)
    a = a0.copy()
    best_a = a.copy()
    best_score = exact_score_fast(a)

    k = int(top_k)
    step_frac = float(max_step_frac)
    fails = 0
    calls = 0

    while time.monotonic() < deadline - 2.0 and calls < 35 and fails < 10:
        calls += 1
        c = fast_convolve(a)
        m = float(np.max(c))
        score = 2.0 * float(n) * m

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        kuse = min(k, int(c.size))
        if kuse <= 0:
            break
        idx = np.argsort(c)[::-1][:kuse].astype(np.int64, copy=False)

        rows = []
        for kk in idx:
            kk = int(kk)
            i0 = max(0, kk - n + 1)
            i1 = min(n - 1, kk)
            row = np.zeros(n, dtype=np.float64)
            if i1 >= i0:
                ia = np.arange(i0, i1 + 1, dtype=np.int64)
                row[ia] = 2.0 * a[kk - ia]
            rows.append(row)

        A = np.vstack(rows).astype(np.float64, copy=False)
        b_ub = -c[idx].astype(np.float64, copy=False)
        A_full = np.hstack((A, -np.ones((len(rows), 1), dtype=np.float64)))
        A_eq = np.ones((1, n + 1), dtype=np.float64)
        A_eq[0, n] = 0.0

        step = max(1e-7, step_frac * float(np.max(a)))
        bounds = []
        for i in range(n):
            lower = -float(a[i])
            bounds.append((lower, step))
        bounds.append((0.0, None))

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0

        sol = solve_linprog_lp(c_obj, A_full, b_ub, A_eq, [0.0], bounds)
        if sol is None or not getattr(sol, "success", False):
            fails += 1
            continue

        v = np.asarray(sol.x[:n], dtype=np.float64)
        v = np.where(np.isfinite(v), v, 0.0)

        cand_best = a
        cand_score = score

        for alpha in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
            cand = a + float(alpha) * v
            cand = np.maximum(cand, 0.0)
            s = float(cand.sum())
            if s < 1e-300:
                continue
            cand = cand / s
            sc = exact_score_fast(cand)
            if np.isfinite(sc) and sc < cand_score:
                cand_score = sc
                cand_best = cand

        if cand_best is not a:
            a = cand_best.copy()
            fails = 0
            sc = exact_score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            fails += 1
            if fails == 3:
                k = min(256, max(64, 2 * k))
                step_frac = min(1.0, max(0.2, 1.4 * step_frac))
                fails = 0

    a = best_a.copy()
    a = normalize_seq(a)
    sc = exact_score_fast(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return a, best_score


def coord_descent(a0, time_budget, factors=NORMAL_FACTORS, verbose=False, label=""):
    """
    Fast coordinate descent on the homogeneous objective.
    Each coordinate is changed independently and the exact score is tested.
    """
    a = normalize_seq(a0)
    n = int(a.size)
    if n == 0:
        return np.array([1.0], dtype=np.float64), np.inf
    if time_budget <= 0:
        return a.copy(), exact_score_fast(a)

    if not isinstance(factors, tuple):
        factors = tuple(float(x) for x in factors)

    deadline = time.monotonic() + float(time_budget)
    a = a.copy()
    S = 1.0
    conv = fast_convolve(a)
    M = float(np.max(conv))
    best_score = 2.0 * float(n) * M
    best_a = a.copy()

    sweep = 0
    stagnant = 0

    while time.monotonic() < deadline - 0.5:
        sweep += 1
        improved = False

        for i in range(n):
            v = float(a[i])
            k0 = i
            k1 = i + n - 1
            last = 2 * n - 2

            unaff_left = float(np.max(conv[:k0])) if k0 > 0 else -1.0
            unaff_right = float(np.max(conv[k1 + 1:])) if k1 < last else -1.0
            unaff = unaff_left if unaff_left >= unaff_right else unaff_right

            old_aff = conv[k0:k1 + 1]
            aset = a[:n]
            has_diag = (2 * i >= k0 and 2 * i <= k1)
            diag_off = 2 * i - k0

            if S <= 0:
                cur_score = np.inf
            else:
                cur_score = 2.0 * float(n) * M / (S * S)

            best_score_c = cur_score
            best_d = 0.0

            if v > 1e-18:
                for f in factors:
                    d = float(f) * v
                    if v + d < -1e-15:
                        continue
                    S2 = S + d
                    if S2 <= 1e-300:
                        continue
                    new_aff = old_aff + 2.0 * d * aset
                    if has_diag:
                        new_aff[diag_off] = new_aff[diag_off] + d * d
                    aff_max = float(np.max(new_aff))
                    new_M = aff_max if aff_max > unaff else unaff
                    sc = 2.0 * float(n) * new_M / (S2 * S2)
                    if sc < best_score_c - 1e-15:
                        best_score_c = sc
                        best_d = d
            else:
                mean_val = S / float(n)
                base_delta = max(1e-8, 1e-4 * mean_val)
                for mult in (0.01, 0.1, 1.0, 10.0):
                    d = float(mult) * base_delta
                    if d <= 0.0:
                        continue
                    S2 = S + d
                    if S2 <= 1e-300:
                        continue
                    new_aff = old_aff + 2.0 * d * aset
                    if has_diag:
                        new_aff[diag_off] = new_aff[diag_off] + d * d
                    aff_max = float(np.max(new_aff))
                    new_M = aff_max if aff_max > unaff else unaff
                    sc = 2.0 * float(n) * new_M / (S2 * S2)
                    if sc < best_score_c - 1e-15:
                        best_score_c = sc
                        best_d = d

            if best_d != 0.0 and best_score_c < cur_score - 1e-14:
                changes = 2.0 * best_d * aset
                if has_diag:
                    changes = changes.copy()
                    changes[diag_off] = changes[diag_off] + best_d * best_d

                conv[k0:k1 + 1] = conv[k0:k1 + 1] + changes
                a[i] = max(v + best_d, 0.0)
                S += best_d

                aff_max = float(np.max(conv[k0:k1 + 1]))
                M = aff_max if aff_max > unaff else unaff
                cur_score = 2.0 * float(n) * M / (S * S)

                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            stagnant += 1
            if stagnant > 2:
                break
        else:
            stagnant = 0

        if sweep % 5 == 0 or abs(S - 1.0) > 1e-6:
            if S > 0:
                a = a / S
                S = 1.0
            conv = fast_convolve(a)
            M = float(np.max(conv))
            sc = 2.0 * float(n) * M
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()

        if verbose and (sweep % 20 == 0):
            print(f"CD {label} sweep={sweep} t={time.monotonic() - (deadline - float(time_budget)):.0f}s score={best_score:.7f}")

    best_a = normalize_seq(best_a)
    sc = exact_score_fast(best_a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best_a, best_score


def soft_peak_weights(c):
    """Weight convolution positions near the current maximum."""
    if c.size == 0:
        return np.zeros(0, dtype=np.float64)
    M = float(np.max(c))
    mean = float(np.mean(c))
    gap = 0.15 * max(1e-300, M - mean)
    thresh = M - gap
    w = np.maximum(c - thresh, 0.0)
    s = float(w.sum())
    if s < 1e-15:
        k = max(1, int(c.size) // 16)
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c, dtype=np.float64)
        w[idx] = 1.0
        s = float(k)
    return w / s


def eg_mirror(a0, time_budget, verbose=False, label=""):
    """
    Multiplicative weights / exponentiated gradient steps using soft top-peak
    gradients. This helps flatten a broad active peak plateau.
    """
    a = normalize_seq(a0)
    n = int(a.size)
    if n == 0 or time_budget <= 0:
        return a.copy(), exact_score_fast(a)

    deadline = time.monotonic() + float(time_budget)
    a = a.copy()
    best_a = a.copy()
    best_score = exact_score_fast(a)

    fail = 0
    it = 0

    while time.monotonic() < deadline - 1.0 and fail < 80:
        it += 1
        c = fast_convolve(a)
        M = float(np.max(c))
        score = 2.0 * float(n) * M
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        p = soft_peak_weights(c)
        g = corr_grad(p, a)
        gmax = float(np.max(np.abs(g)))
        if gmax < 1e-25:
            break

        g_u = g - float(np.max(g))
        base_eta = 1.0 / gmax

        best_score_c = score
        best_a_c = a
        best_factor = 0.0

        for factor in (0.02, 0.05, 0.1, 0.2, 0.4):
            eta = float(factor) * base_eta
            log_a = np.log(a + 1e-300) + eta * g_u
            a_new = np.exp(log_a - float(np.max(log_a)))
            s = float(a_new.sum())
            if s < 1e-300:
                continue
            a_new = a_new / s
            sc = exact_score_fast(a_new)
            if np.isfinite(sc) and sc < best_score_c - 1e-14:
                best_score_c = sc
                best_a_c = a_new
                best_factor = factor

        if best_factor != 0.0:
            a = best_a_c.copy()
            fail = 0
            if best_score_c < best_score:
                best_score = best_score_c
                best_a = a.copy()
        else:
            fail += 1
            if fail > 50:
                break

        if verbose and (it % 200 == 0):
            print(f"EG {label} it={it} t={it * 0.1:.0f} score={best_score:.7f}")

    best_a = normalize_seq(best_a)
    sc = exact_score_fast(best_a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best_a, best_score


def init_bimodal(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = max(1.0, n * rng.uniform(0.05, 0.15))
    w2 = max(1.0, n * rng.uniform(0.05, 0.15))
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return normalize_seq(a)


def init_ridges(n, rng):
    period = int(max(2, n / rng.integers(3, 8)))
    t = np.arange(n, dtype=np.float64)
    amp = rng.uniform(0.1, 0.35)
    a = 0.5 + amp * np.cos(2.0 * np.pi * t / float(period))
    return normalize_seq(a)


def init_symmetric(n, rng):
    half = max(1, n // 2)
    t = np.linspace(0.0, 1.0, half, dtype=np.float64)
    profile = np.ones(half, dtype=np.float64)
    ncomp = int(rng.integers(2, 7))
    for _ in range(ncomp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.45)
        phase = rng.uniform(0.0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.0)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(float(profile[-1]), float(profile[0])) * rng.uniform(0.3, 1.5)
        a = np.concatenate([profile, np.array([mid], dtype=np.float64), profile[::-1]])
    return normalize_seq(a)


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = int(f.size)
    new_n = int(new_n)
    if old_n == new_n or old_n <= 0:
        return normalize_seq(f)
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    g = np.interp(x_new, x_old, f)
    return normalize_seq(g)


def peak_targeted_perturb(a, rng, strength=0.05):
    a = normalize_seq(a)
    c = fast_convolve(a)
    p = soft_peak_weights(c)
    g = corr_grad(p, a)
    gmin = float(np.min(g))
    gmax = float(np.max(g))
    span = max(1e-300, gmax - gmin)
    gnorm = (g - gmin) / span
    noise = rng.standard_normal(a.size) * 0.2
    a_new = a * (1.0 - float(strength) * gnorm + 0.3 * float(strength) * noise)
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    return a_new / s


def random_restructure(a, rng, frac=0.15):
    a = normalize_seq(a)
    n = int(a.size)
    keep = rng.random(n) > float(frac)
    if int(np.count_nonzero(keep)) < max(8, n // 10):
        keep = np.ones(n, dtype=bool)
    a_new = a.copy()
    a_new[~keep] = 0.0
    a_new[keep] = a_new[keep] * (1.0 + 0.2 * rng.standard_normal(int(np.count_nonzero(keep))))
    a_new = np.maximum(a_new, 0.0)

    zero_mask = a_new < 1e-18
    if np.any(zero_mask):
        zidx = np.where(zero_mask)[0]
        nadd = max(1, int(zidx.size * 0.05))
        if nadd <= zidx.size:
            pick = rng.choice(zidx, size=nadd, replace=False)
            a_new[pick] = rng.random(nadd) * max(1e-12, float(np.max(a_new)) * 0.25)

    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    return a_new / s


def gaussian_perturb(a, rng, sigma=0.04):
    a = normalize_seq(a)
    noise = rng.standard_normal(a.size) * float(sigma)
    a_new = np.maximum(a * (1.0 + noise), 0.0)
    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    return a_new / s


def threshold_perturb(a, rng, frac=0.5):
    a = normalize_seq(a)
    t = np.quantile(a, float(frac))
    a_new = (a > t).astype(np.float64)
    if int(np.count_nonzero(a_new)) < 16:
        a_new = np.ones_like(a_new)
    s = float(a_new.sum())
    if s < 1e-300:
        return a.copy()
    return a_new / s


def propose_candidate():
    """
    Main search entry point. Returns the best non-negative vector found
    within about 1780 seconds.
    """
    t_start = time.monotonic()
    deadline = t_start + 1780.0
    rng = np.random.default_rng(20240528)

    best_a = None
    best_score = np.inf

    # Use previous construction when available.
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim != 1:
            h = h.ravel()
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.clip(h, 0.0, 1000.0)
        if h.size > 0 and float(h.sum()) > 0.01:
            candidate = normalize_seq(h)
            sc = exact_score_fast(candidate)
            if np.isfinite(sc):
                best_a = candidate.copy()
                best_score = float(sc)
                print(f"Parent: n={len(best_a)} score={best_score:.7f}")
    except Exception:
        pass

    # Fallback if no usable parent.
    if best_a is None:
        n = 1024
        starts = (
            np.ones(n, dtype=np.float64),
            init_bimodal(n, rng),
            init_ridges(n, rng),
            init_symmetric(n, rng),
        )
        for st in starts:
            st = normalize_seq(st)
            sc = exact_score_fast(st)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = st.copy()
                print(f"Fallback start: n={len(best_a)} score={best_score:.7f}")
        if best_a is None:
            best_a = np.ones(1024, dtype=np.float64) / 1024.0
            best_score = exact_score_fast(best_a)

    # If no parent evolved, do a bit of broad optimization first.
    if "Parent" not in "".join(
        line for line in []
    ) and time.monotonic() < deadline - 120:
        rem = deadline - time.monotonic()
        b = min(60.0, 0.2 * rem)
        a1, _ = coord_descent(best_a, b, label="fb_cd")
        best_a, best_score = update_best(a1, best_a, best_score, "fb_cd")
        if time.monotonic() < deadline - 60:
            a2, _ = run_topk_lbfgs(best_a, min(40.0, deadline - time.monotonic() - 20.0), K_list=(32, 64))
            best_a, best_score = update_best(a2, best_a, best_score, "fb_topk")

    # Phase 1: quick coordinate descent to polish the seed.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        b = min(45.0, 0.20 * rem)
        a1, _ = coord_descent(best_a, b, label="p1_cd")
        best_a, best_score = update_best(a1, best_a, best_score, "p1_cd")

    # Phase 2: linearized LP step.
    rem = deadline - time.monotonic()
    if rem > 120.0 and len(best_a) <= 2048:
        b = min(90.0, 0.15 * rem)
        a1, _ = lp_linearize(best_a, b, top_k=64)
        best_a, best_score = update_best(a1, best_a, best_score, "p2_lp")

    # Phase 3: active-set top-K L-BFGS.
    rem = deadline - time.monotonic()
    if rem > 220.0:
        b = min(420.0, 0.35 * rem)
        a1, _ = run_topk_lbfgs(best_a, b, K_list=(16, 32, 64, 128))
        best_a, best_score = update_best(a1, best_a, best_score, "p3_topk")

    # Phase 4: LSE-smoothed L-BFGS.
    rem = deadline - time.monotonic()
    if rem > 160.0:
        b = min(220.0, 0.25 * rem)
        a1, _ = run_lbfgs_schedule(best_a, b, betas=(5000.0, 15000.0, 40000.0))
        best_a, best_score = update_best(a1, best_a, best_score, "p4_lse")

    # Phase 5: multiplicative peak balancing.
    rem = deadline - time.monotonic()
    if rem > 120.0:
        b = min(110.0, 0.20 * rem)
        a1, _ = eg_mirror(best_a, b, label="p5_eg")
        best_a, best_score = update_best(a1, best_a, best_score, "p5_eg")

    # Phase 6: fine coordinate polish.
    rem = deadline - time.monotonic()
    if rem > 100.0:
        b = min(230.0, 0.35 * rem)
        a1, _ = coord_descent(best_a, b, factors=FINE_FACTORS, label="p6_fine")
        best_a, best_score = update_best(a1, best_a, best_score, "p6_fine")

    # Phase 7: restarts from the best found so far.
    rem = deadline - time.monotonic()
    if rem > 240.0:
        total = min(360.0, 0.40 * rem)
        base = best_a.copy()
        per = total / 5.0
        perturb_names = ("perturb_peak", "restructure", "gaussian", "threshold", "small_noise")
        for rname in perturb_names:
            rem = deadline - time.monotonic()
            if rem < 60.0:
                break
            b = max(2.0, min(per, rem * 0.6))
            if rname == "perturb_peak":
                cur = peak_targeted_perturb(base, rng, 0.05)
            elif rname == "restructure":
                cur = random_restructure(base, rng, 0.18)
            elif rname == "gaussian":
                cur = gaussian_perturb(base, rng, 0.05)
            elif rname == "threshold":
                cur = threshold_perturb(base, rng, 0.45)
            else:
                cur = gaussian_perturb(base, rng, 0.02)

            rem_sub = deadline - time.monotonic() - 20.0
            if rem_sub > 30.0:
                cur, _ = coord_descent(cur, min(0.35 * b, max(5.0, 0.35 * rem_sub)))
                best_a, best_score = update_best(cur, best_a, best_score, f"r_cd_{rname}")
                cur = best_a.copy()

            rem_sub = deadline - time.monotonic() - 20.0
            if rem_sub > 45.0 and len(cur) <= 1600:
                cur, _ = run_topk_lbfgs(cur, min(0.45 * b, max(10.0, 0.55 * rem_sub)), K_list=(32, 64))
                best_a, best_score = update_best(cur, best_a, best_score, f"r_topk_{rname}")
                cur = best_a.copy()

            rem_sub = deadline - time.monotonic() - 15.0
            if rem_sub > 20.0:
                cur, _ = coord_descent(
                    cur,
                    min(0.30 * b, max(5.0, 0.80 * rem_sub)),
                    factors=FINE_FACTORS,
                    label=f"r_fine_{rname}",
                )
                best_a, best_score = update_best(cur, best_a, best_score, f"r_fine_{rname}")

            base = best_a.copy()

    # Phase 8: limited length exploration.
    rem = deadline - time.monotonic()
    if rem > 180.0:
        n_best = int(len(best_a))
        alt_ns = []
        for cand in (n_best + 128, n_best + 256, n_best - 128, 1024, 1152, 1280, 1536):
            cand = int(cand)
            if 512 <= cand <= 2048 and cand not in alt_ns:
                alt_ns.append(cand)

        alt_total = min(220.0, 0.28 * rem)
        per_alt = alt_total / max(1, len(alt_ns[:4]))
        base = best_a.copy()

        for n_alt in alt_ns[:4]:
            rem = deadline - time.monotonic()
            if rem < 45.0:
                break
            b = max(5.0, min(per_alt, rem * 0.8))
            cur = interpolate_seq(base, n_alt)

            rem_sub = deadline - time.monotonic() - 15.0
            if rem_sub > 15.0:
                cur, _ = coord_descent(cur, min(0.40 * b, max(5.0, 0.55 * rem_sub)), label=f"alt_{n_alt}")
                best_a, best_score = update_best(cur, best_a, best_score, f"alt_cd_{n_alt}")
                cur = best_a.copy()

            rem_sub = deadline - time.monotonic() - 15.0
            if rem_sub > 45.0 and len(cur) <= 1600:
                cur, _ = run_topk_lbfgs(cur, min(0.50 * b, max(10.0, 0.65 * rem_sub)), K_list=(32, 64))
                best_a, best_score = update_best(cur, best_a, best_score, f"alt_topk_{n_alt}")
                cur = best_a.copy()

            rem_sub = deadline - time.monotonic() - 10.0
            if rem_sub > 10.0:
                cur, _ = coord_descent(
                    cur,
                    min(0.30 * b, max(3.0, 0.85 * rem_sub)),
                    factors=FINE_FACTORS,
                    label=f"alt_fine_{n_alt}",
                )
                best_a, best_score = update_best(cur, best_a, best_score, f"alt_fine_{n_alt}")

            base = best_a.copy()

    # Phase 9: final LP + polishing.
    rem = deadline - time.monotonic()
    if rem > 90.0 and len(best_a) <= 2048:
        b = min(70.0, 0.35 * rem)
        a1, _ = lp_linearize(best_a, b, top_k=128)
        best_a, best_score = update_best(a1, best_a, best_score, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 30.0:
        b = max(5.0, min(90.0, 0.65 * rem))
        a1, _ = coord_descent(best_a, b, factors=FINE_FACTORS, label="final_cd1")
        best_a, best_score = update_best(a1, best_a, best_score, "final_cd1")

    rem = deadline - time.monotonic()
    if rem > 20.0:
        a1, _ = coord_descent(best_a, max(3.0, min(30.0, 0.85 * rem)), factors=FINE_FACTORS, label="final_cd2")
        best_a, best_score = update_best(a1, best_a, best_score, "final_cd2")

    # Final safe normalization and validation.
    best_a = normalize_seq(best_a)
    try:
        fs = evaluate_sequence(best_a.tolist())
        if np.isfinite(fs):
            best_score = float(fs)
    except Exception:
        pass

    print(f"FINAL SCORE: {best_score:.7f} n={len(best_a)}")
    return [float(x) for x in best_a]
