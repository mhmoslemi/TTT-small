5. Adaptive time allocation based on progress

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_subgradient(conv, a, top_k=3):
    """
    Compute weighted subgradient of max(conv) w.r.t. a.
    Returns gradient vector of length n.
    Uses top-k peaks with softmax weights.
    """
    n = len(a)
    M = conv.max()
    
    # Get top-k peak indices
    if top_k >= 2 * n - 1:
        peaks = np.arange(2 * n - 1)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    pv = conv[peaks]
    temp = 0.01 * max(M, 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # sg[i] += w[k] * 2 * a[k-i]
        sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
    
    return sg, M

            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Nesterov lookahead
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute subgradient at y (using current conv as approximation)
        sg, M = compute_subgradient(conv, a, top_k=top_k)
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a_prev = a.copy()
        a = a_new
        conv = fast_convolve(a)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-mom it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def peak_cd(a0, time_limit, verbose=True):
    """
    Coordinate descent focused on reducing peak convolution values.
    Uses the structure: changing a[i] affects conv[k] for k in [i, i+n-1].
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    # Precompute: for each i, the affected range is [i, i+n-1]
    # conv[k] changes by 2*delta*a[k-i] when a[i] changes by delta
    # (plus delta^2 at k=2i)
    
    # For efficiency, precompute the "influence" of each coordinate
    # coord i affects conv[i..i+n-1] with weights 2*a[i..i+n-1-i] = 2*a[0..n-1] shifted
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            affected_len = k_end - k_start + 1
            
            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            # The affected convolution values
            # conv[k] for k in [k_start, k_end] depends on a[i] through:
            # conv[k] = ... + 2*a[i]*a[k-i] + ... (for k != 2i)
            # conv[2i] = ... + a[i]^2 + ... 
            # When we change a[i] by delta:
            # new_conv[k] = conv[k] + 2*delta*a[k-i] + (delta^2 if k==2i)
            # for k in [k_start, k_end]
            
            a_shifted = a[k_start - i:k_end - i + 1]  # a[k-i] for k in [k_start, k_end]
            
            # Try to find optimal delta
            # We want to minimize max(unaff_max, max_k (conv[k] + 2*delta*a[k-i] + delta^2*I(k=2i)))
            # subject to a[i] + delta >= 0
            
            # For large n, the quadratic term is small, so approximate as linear:
            # new_conv[k] ≈ conv[k] + 2*delta*a[k-i]
            # The max of (conv[k] + 2*delta*a[k-i]) is minimized when we balance
            # the top affected values.
            
            # Simple approach: try a few step sizes
            cur_affected = conv[k_start:k_end + 1]
            cur_affected_max = cur_affected.max()
            
            if cur_affected_max < M - 1e-14 and unaff_max >= M - 1e-14:
                # This coordinate's affected range doesn't contain the max
                # Reducing a[i] won't help (it might even hurt by reducing sum)
                # Actually, we're working with sum=1, so we need to be careful
                # Let's just try small reductions
                pass
            
            # Binary search for the best delta
            # The affected max as a function of delta is:
            # f(delta) = max(k) [conv[k] + 2*delta*a[k-i] + delta^2 * I(2i in [k_start,k_end] and k==2i)]
            # This is convex in delta (max of linear/quadratic functions)
            # So we can use ternary search or evaluate at a few points
            
            base = a[i]
            best_delta = 0.0
            best_new_M = M
            
            # Try several factors
            for factor in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3, -0.5, 0.5, -0.1, 0.1, -0.2, 0.2]:
                delta = base * factor
                new_val = base + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = cur_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = cur_affected[idx] + 2.0 * delta * base + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-13:
                a[i] = max(a[i] + best_delta, 0.0)
                
                # Update convolution
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                # Renormalize (since we're changing sum)
                # Actually, the score depends on (sum a)^2, so we need to track this
                # But for the local optimization, we work with sum=1 and score = 2n*max(conv)
                # After changing a[i], sum changes. Let's renormalize periodically.
                
                cur_score = 2.0 * n * M / (a.sum()**2) if a.sum() > 1e-15 else float('inf')
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        # Renormalize after each sweep
        s = a.sum()
        if s > 1e-15:
            a = a / s
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def fine_cd(a0, time_limit, verbose=True):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - i:k_end - i + 1]
            cur_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            # Very fine steps around current value
            for factor in [0.995, 0.99, 0.998, 1.002, 1.005, 1.01, 0.997, 1.003]:
                delta = a[i] * (factor - 1.0)
                if a[i] + delta < -1e-15:
                    continue
                
                new_affected = cur_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = cur_affected[idx] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_a_score if False else best_a, best_score

def spectral_init(n, rng):
    """
    Generate a random non-negative sequence via spectral method.
    Random phase in frequency domain, IFFT, clip to non-negative.
    """
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, n)
    # Flat amplitude (gives white noise in time domain)
    A = np.exp(1j * phase)
    # Make it conjugate symmetric for real output
    # Actually, let's use rfft/irfft
    # Generate random real sequence via random phase
    a = np.cos(phase) + 0.1  # bias to ensure non-negativity after IFFT
    
    # Alternative: just use a random non-negative sequence
    # a = np.abs(rng.standard_normal(n)) + 0.1
    
    # Better: use a structured initialization
    # U-shaped: high at edges, low in middle
    t = np.linspace(-1, 1, n)
    shape_param = rng.uniform(0.5, 3.0)
    a = 1.0 / (1.0 + np.exp(shape_param * t))  # sigmoid - high on left, low on right
    
    # Or U-shaped
    if rng.random() < 0.5:
        a = 0.3 + 0.7 * (2 * np.abs(t) - 1) ** 2  # U-shaped
    else:
        a = 0.3 + 0.7 * (1 - (2 * t) ** 2)  # inverted U (hump)
    
    a = np.maximum(a, 0.01)
    s = a.sum()
    a = a / s
    return a

def diverse_init(n, rng, idx):
    """Generate diverse initializations based on index."""
    t = np.linspace(0, 1, n)
    
    if idx == 0:
        # U-shaped (bathtub)
        a = 0.2 + 0.8 * (2*t - 1)**2
    elif idx == 1:
        # W-shaped
        a = 0.2 + 0.5*(2*t-1)**2 + 0.3*np.exp(-50*(t-0.5)**2)
    elif idx == 2:
        # Cosine window
        a = 0.5 * (1 + np.cos(2*np.pi*t - np.pi))
    elif idx == 3:
        # Two peaks
        a = np.exp(-100*(t-0.2)**2) + np.exp(-100*(t-0.8)**2) + 0.1
    elif idx == 4:
        # Periodic
        a = 0.5 + 0.3*np.cos(2*np.pi*3*t)
    elif idx == 5:
        # Random
        a = np.abs(rng.standard_normal(n)) + 0.1
    elif idx == 6:
        # Raised cosine
        a = 0.5 + 0.5*np.cos(2*np.pi*(t-0.5))
    else:
        # Mix
        a = 0.3 + 0.3*(2*t-1)**2 + 0.2*np.cos(2*np.pi*2*t)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def subsidiary_momentum_descent(a0, time_limit, lr_init=0.01, top_k=3):
    """
    Subgradient mirror descent without momentum (simpler, more stable).
    Focuses on the top peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            lr *= 0.8
            if lr < 1e-10:
                break
            no_improve = 0
        
        # Compute subgradient focusing on top peaks
        if top_k == 1:
            k_max = int(np.argmax(conv))
            i_start = max(0, k_max - n + 1)
            i_end = min(n - 1, k_max)
            sg = np.zeros(n)
            sg[i_start:i_end + 1] = 2.0 * a[k_max - i_end:k_max - i_start + 1][::-1]
        else:
            sg, M = compute_subgradient(conv, a, top_k=top_k)
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve(a)
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.2e}")
    
    return best_a, best_score

def lp_alternate(a0, time_limit, verbose=True):
    """
    LP-based alternating minimization.
    Given a, solve: min M s.t. conv(a,b) <= M, b >= 0, sum(b) = 1.
    Then update a towards the solution.
    """
    try:
        import cvxpy as cp
    except ImportError:
        print("    cvxpy not available, skipping LP")
        return a0, 2.0 * len(a0) * fast_convolve(a0).max()
    
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Build convolution matrix: T @ b = conv(a, b)
    # T has shape (2n-1, n), T[k, j] = a[k-j] if 0 <= k-j < n else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    # Find solver
    solver = None
    for s_name in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(2, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    print(f"    LP solver: {solver}")
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b
        
        # Evaluate
        score_b = 2.0 * n * fast_convolve(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()
        
        # Damped update
        alpha = 2.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        # Rebuild T
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = a
        
        if it % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def propose_candidate():

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_subgradient(conv, a, top_k=3, temp_scale=0.01):
    """Compute weighted subgradient of max(conv) w.r.t. a using top-k peaks."""
    n = len(a)
    total_len = 2 * n - 1
    M = conv.max()
    if top_k >= total_len:
        peaks = np.arange(total_len)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    temp = temp_scale * max(M, 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / ws
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-15:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
    return sg, M

def spectral_init(n, rng):
    """Spectral initialization: random phase, flat amplitude, IFFT, clip."""
    phase = rng.uniform(0, 2 * np.pi, n)
    # Create conjugate-symmetric spectrum
    A = np.exp(1j * phase)
    # Ensure hermitian symmetry for real output
    if n % 2 == 0:
        A[n // 2] = np.abs(A[n // 2])
    for k in range(1, n // 2 + 1):
        A[n - k] = np.conj(A[k])
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    # For larger n, just do linear interpolation in time
    t_old = np.linspace(0, 1, n_orig, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(t_new, t_old, a, period=1)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new


def evaluate_internal(a):
    """Fast internal score computation (same formula as evaluator for normalized sequences)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s ** 2)


def propose_candidate():
    """
    Multi-strategy search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    Combines subgradient descent, coordinate descent, and LP alternating.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(7)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_internal(parent)
            print(f"Parent loaded: n={len(parent)}, internal_score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = evaluate_internal(parent)
        try:
            es = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(es):
                best_score = es
                print(f"Parent eval: {best_score:.6f}")
        except Exception:
            pass
    else:
        best_seq = np.ones(1024) / 1024
        best_score = 2.0

    def check_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            score = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(candidate)}) ***")

    # === Phase 1: Subgradient descent on parent ===
    print(f"\n=== Phase 1: Subgradient on parent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 500
    if remaining > 100:
        # Multiple SGD runs with different parameters
        sg_configs = [
            {'lr_init': 0.01, 'top_k': 3, 'temp_scale': 0.01},
            {'lr_init': 0.005, 'top_k': 1, 'temp_scale': 0.005},
            {'lr_init': 0.02, 'top_k': 5, 'temp_scale': 0.02},
        ]
        for ci, cfg in enumerate(sg_configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.3, 150)
            print(f"  SGD config {ci}, budget={t_alloc:.0f}s, {cfg}")
            a_sg, s_sg = subgradient_descent(best_seq, t_alloc, verbose=False, **cfg)
            check_update(a_sg, f"sgd_{ci}")

    # === Phase 2: Coordinate descent ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 350
    if remaining > 80:
        t_cd = min(remaining * 0.5, 300)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd, verbose=True)
        check_update(a_cd, "cd_main")

    # Second CD pass on result
    remaining = deadline - time.monotonic() - 250
    if remaining > 60:
        t_cd2 = min(remaining * 0.4, 150)
        a_cd2, s_cd2 = coord_descent(best_seq, t_cd2, verbose=False)
        check_update(a_cd2, "cd_pass2")

    # === Phase 3: Fine polishing ===
    print(f"\n=== Phase 3: Fine Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 40:
        t_fine = min(remaining * 0.4, 120)
        print(f"  FineCD budget={t_fine:.0f}s")
        a_fine, s_fine = fine_polish(best_seq, t_fine, verbose=True)
        check_update(a_fine, "fine_polish")

    # === Phase 4: Try larger n (2048) ===
    print(f"\n=== Phase 4: Larger n exploration (best={best_score:.6f}) ===")
    n_large = 2048
    remaining = deadline - time.monotonic() - 150
    if remaining > 100 and parent is not None:
        a_large = interpolate_sequence(parent, n_large)
        # Add diverse inits
        inits_large = [a_large]
        inits_large.append(diverse_init(n_large, rng, 0))
        inits_large.append(diverse_init(n_large, rng, 6))
        inits_large.append(spectral_init(n_large, rng))

        for ii, init in enumerate(inits_large):
            remaining = deadline - time.monotonic() - 100
            if remaining < 50:
                break
            t_per = min(remaining * 0.25, 100)
            # Quick SGD
            t_sg = t_per * 0.4
            a_t, s_t = subgradient_descent(init, t_sg, lr_init=0.005, top_k=3, verbose=False)
            check_update(a_t, f"large_n_{ii}_sgd")
            # CD
            remaining = deadline - time.monotonic() - 80
            if remaining < 30:
                break
            t_cd_l = min(t_per * 0.5, remaining * 0.3)
            a_t2, s_t2 = coord_descent(a_t, t_cd_l, verbose=False)
            check_update(a_t2, f"large_n_{ii}_cd")
            # Fine polish
            remaining = deadline - time.monotonic() - 60
            if remaining > 20:
                t_fp = min(remaining * 0.3, 30)
                a_t3, s_t3 = fine_polish(a_t2, t_fp, verbose=False)
                check_update(a_t3, f"large_n_{ii}_fine")

    # === Phase 5: Try n=4096 if time permits ===
    n_xlarge = 4096
    remaining = deadline - time.monotonic() - 100
    if remaining > 80 and parent is not None:
        print(f"\n=== Phase 5: n={n_xlarge} (best={best_score:.6f}) ===")
        a_xl = interpolate_sequence(parent, n_xlarge)
        inits_xl = [a_xl, diverse_init(n_xlarge, rng, 0)]
        for ii, init in enumerate(inits_xl):
            remaining = deadline - time.monotonic() - 60
            if remaining < 40:
                break
            t_per = min(remaining * 0.4, 60)
            a_t, s_t = subgradient_descent(init, t_per * 0.5, lr_init=0.003, top_k=3, verbose=False)
            check_update(a_t, f"xl_n_{ii}")
            remaining = deadline - time.monotonic() - 40
            if remaining > 20:
                a_t2, s_t2 = coord_descent(a_t, min(remaining * 0.4, 30), verbose=False)
                check_update(a_t2, f"xl_n_{ii}_cd")

    # === Phase 6: LP refinement on best ===
    remaining = deadline - time.monotonic() - 40
    if remaining > 50:
        print(f"\n=== Phase 6: LP Refinement (best={best_score:.6f}) ===")
        t_lp = min(remaining * 0.5, 80)
        a_lp, s_lp = lp_alternate(best_seq, t_lp, verbose=True)
        check_update(a_lp, "lp_final")

    # === Phase 7: Final fine polish ===
    remaining = deadline - time.monotonic() - 15
    if remaining > 20:
        print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
        t_fin = min(remaining * 0.6, 40)
        a_fin, s_fin = fine_polish(best_seq, t_fin, verbose=True)
        check_update(a_fin, "abs_final")

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(max(0.0, x)) for x in best_seq]
