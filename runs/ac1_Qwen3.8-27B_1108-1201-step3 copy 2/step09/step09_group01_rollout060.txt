"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def eval_score(a):
    """Compute the score 2n*M/s^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    M = float(conv.max())
    return 2.0 * n * M / (s * s)


def precise_cd(a0, time_limit, verbose=False, label="", rng_seed=42):

    For each coordinate, tries multiple scales and refines locally.

    else:
        a = a / s
    s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

        sweeps += 1
        improved = False
        order = rng.permutation(n)

                continue

            ks = i
            ke = min(i + n - 1, 2 * n - 2)

            u_max = 0.0
            if ks > 0:

            if ke < 2 * n - 2:
                u_max = max(u_max, float(conv[ke + 1:].max()))

def smooth_gradient_descent(a0, time_limit, beta_init=40.0, lr_init=0.005, verbose=False, label=""):
    """
    Gradient descent using logsumexp as smooth surrogate for max.
    Operates on the simplex with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    a_prev = a.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        if it == 1:
            y = a.copy()
        else:
            gamma = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
            y = (1.0 + gamma) * a - gamma * a_prev
            a_prev = a.copy()

        # Project to simplex
        y = simplex_project(y)

        # Compute smooth gradient via logsumexp
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -60, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient of max(conv(a,a)) w.r.t. a
        grad = np.zeros(n)
        active = np.where(weights > 1e-12)[0]
        for k in active:
            w = weights[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y[k - j]

        # Multiplicative update (mirror descent)
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        # Also subtract mean to maintain sum ~ 1
        log_a -= log_a.mean()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 100 == 0:
            beta *= 1.01

    return best_a, best_score


def spectral_perturb(a0, rng, strength=0.3):
    """
    Perturb in frequency domain: modify phase of FFT to reduce time-domain peaks.
    This helps escape local optima by exploring very different sequences.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s

    # Zero-pad to 2n for linear convolution
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    # Get magnitude and phase
    mag = np.abs(A)
    phase = np.angle(A)

    # Modify phase to reduce peaks (iterative amplitude reduction in time domain)
    # Try adding a random phase offset
    if rng is not None:
        random_phase = rng.standard_normal(len(A)) * strength * 0.1
        new_phase = phase + random_phase
    else:
        new_phase = phase

    # Reconstruct
    A_new = mag * np.exp(1j * new_phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss < 1e-15:
        return a0.copy()
    return a_new / ss


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_diverse(n, rng, parent=None):
    """Generate diverse initializations."""
    cands = []
    if parent is not None:
        cands.append(interpolate_seq(parent, n))
    # Uniform
    cands.append(np.ones(n) / n)
    # Bimodal
    for _ in range(3):
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w1 = n * rng.uniform(0.05, 0.12)
        w2 = n * rng.uniform(0.05, 0.12)
        t = np.arange(n)
        a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            cands.append(a / ss)
    # Periodic
    for _ in range(2):
        period = max(3, int(n / rng.integers(4, 12)))
        t = np.arange(n)
        amp = rng.uniform(0.1, 0.4)
        a = np.ones(n) * 0.5 + amp * np.cos(2 * np.pi * t / period + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            cands.append(a / ss)
    # Random
    for _ in range(2):
        a = rng.random(n)
        ss = a.sum()
        if ss > 1e-15:
            cands.append(a / ss)
    return cands


def pair_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Pair coordinate descent: update pairs of coordinates that jointly
    contribute to the same peak positions.
    """
    rng = np.random.default_rng(999)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:

    else:
        a = a / s

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    stale = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        # Find top peaks and their contributing pairs
        top_k = np.argsort(conv)[-20:]

        for k in top_k:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end - i_start < 2:
                continue

            # Pick two coordinates that jointly contribute most to this peak
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if contributions.max() < 1e-15:
                continue
            top2_idx = np.argsort(contributions)[-2]
            i1 = i_start + top2_idx[0]
            i2 = i_start + top2_idx[1]

            a1, a2 = a[i1], a[i2]
            if a1 < 1e-14 or a2 < 1e-14:
                continue

            # Try reducing both
            for f1 in [0.3, 0.5, 0.7]:
                for f2 in [0.3, 0.5, 0.7]:
                    v1 = a1 * f1
                    v2 = a2 * f2
                    d1 = v1 - a1
                    d2 = v2 - a2
                    ns = s + d1 + d2
                    if ns < 1e-12:
                        continue

                    # Compute new conv at affected positions
                    # This is complex for pairs, so just do a quick FFT check
                    a_trial = a.copy()
                    a_trial[i1] = v1
                    a_trial[i2] = v2
                    conv_trial = fast_convolve(a_trial)
                    nM = float(conv_trial.max())
                    sc = 2.0 * n * nM / (ns * ns)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        conv = conv_trial
                        M = nM
                        s = ns
                        best_score = sc
                        best_a = a.copy()
                        improved = True

                    break

            stale += 1
            if stale > 50:

        else:

def check_and_update(a, best_seq, best_score, label=""):
