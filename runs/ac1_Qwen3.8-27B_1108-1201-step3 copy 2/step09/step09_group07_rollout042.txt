def fast_conv(a, b=None):
    """Linear convolution of a with b (or with itself if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a, b=None):
    """Linear convolution of a with b (or a with itself)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Score = 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_logsumexp(a, T):

    # cross_corr = ifft(conj(FFT(a_pad)) * FFT(h)) ...

def grad_logsumexp_efficient(a, T):
    """Efficient gradient using the structure of the convolution."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # We need g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # Equivalently: g[i] = 2 * (a * w_reversed_shifted)[i]
    # sum_k w[k] * a[k-i]: let j = k-i, then k = j+i, sum_j w[j+i] * a[j]
    # This is the cross-correlation of a and w at lag i
    # Using FFT: cross_corr(i) = sum_j a[j] * w[j+i]
    # = ifft(conj(fft(a_extended)) * fft(w)) evaluated at appropriate indices
    # For linear cross-correlation, pad to length >= n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    a_ext = np.zeros(N)
    a_ext[:n] = a
    w_ext = np.zeros(N)
    w_ext[:2 * n - 1] = w
    A = rfft(a_ext, N)
    W = rfft(w_ext, N)
    corr = irfft(np.conj(A) * W, N)
    # corr[i] = sum_j a[j] * w[j+i] for i = 0...N-1 (linear, since we padded enough)
    return 2.0 * corr[:2 * n - 1]


def mw_best_response(a, n_iters=300, eta=None):
    """
    Find b >= 0, sum b = 1, minimizing max_k conv(a, b)
    using multiplicative weights (exponential weights) algorithm.
    Returns (best_b, best_max_val).
    """
    n = len(a)
    size = 2 * n - 1
    if eta is None:
        eta = 3.0 * np.sqrt(np.log(n) / n_iters)
    
    b = np.ones(n) / n
    y = np.ones(size) / size
    best_b = b.copy()
    best_val = float('inf')
    
    prefix = np.zeros(size + 1)
    idx_hi = np.minimum(np.arange(n) + n, size)
    idx_lo = np.arange(n)
    
    for it in range(n_iters):
        conv_ab = fast_conv(a, b)
        val = conv_ab.max()
        if val < best_val:
            best_val = val
            best_b = b.copy()
        
        # Update row player strategy y proportional to payoffs
        y = conv_ab / conv_ab.sum()
        
        # Compute (T^T y)[i] = a[i] * C[i] where C[i] = sum_{k=i}^{i+n-1} y[k] (clipped)
        prefix[0] = 0.0
        prefix[1:] = np.cumsum(y)
        C = prefix[idx_hi] - prefix[idx_lo]
        
        paid = a * C
        
        # Update column player: b[i] prop to exp(-eta * paid[i])
        shift = paid.max()
        b_new = np.exp(-eta * (paid - shift))
        s = b_new.sum()
        if s < 1e-300:
            b_new = np.ones(n) / n
        else:
            b_new /= s
        b = b_new
    
    return best_b, best_val


def nesterov_gradient(a0, time_limit, T_start=1.0, T_end=200.0,
                       lr_init=0.05, n_stages=8, steps_per_stage=3000):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_of(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0
    
    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.003 * step_count)
            
            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            grad = grad_logsumexp_efficient(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            
            if step_count % 100 == 0:
                sc = score_of(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    
    return best_a, best_score


def alternating_mw(a0, time_limit, mw_iters=200, alpha_init=0.5):
    """
    Alternating optimization: Nesterov outer loop + multiplicative weights inner.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for it in range(1000000):

        if it > 0:
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr
        else:
            beta = 0.0
        y = project_simplex(a + beta * (a - a_prev))
        
        # Best response via multiplicative weights

        a_prev = a

        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=50000):
    """Aggressive peak shaving with coordinate and pair moves."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_conv(a)
        # Find the peak (exclude first and last which are single terms)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        
        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        
        # Try reducing top contributors
        top_idx = np.argsort(contribs)[::-1][:min(8, len(idx))]
        for ti in top_idx:
            i = idx[ti]
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair reduction
            for ti in top_idx[:4]:
                i = idx[ti]
                j = peak_k - i
                if i == j or j < 0 or j >= n or i < 0 or i >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = project_simplex(a_trial)
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try small random perturbation
            if step % 500 == 0:
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = project_simplex(a_trial)
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            else:
                break
    
    return best_a, best_score


def make_structured_init(n, seed=42):
    """Create various structured initializations."""
    rng = np.random.default_rng(seed)
    method = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    
    if method == 0:
        # Two-level alternating
        alpha = rng.uniform(1.5, 4.0)
        a = np.where(np.arange(n) % 2 == 0, 1.0, alpha)
    elif method == 1:
        # Sine modulation
        freq = rng.uniform(1, 5)
        amp = rng.uniform(0.3, 0.9)
        a = 1.0 + amp * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    elif method == 2:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
    elif method == 3:
        # Triangle wave
        freq = rng.uniform(2, 10)
        a = np.abs(np.sin(np.pi * freq * x))
    elif method == 4:
        # Random with smooth noise
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x * rng.uniform(1, 3)) + 0.3 * rng.standard_normal(n) * np.exp(-x * 3)
    else:
        # Power law
        gamma = rng.uniform(0.5, 2.0)
        a = (x + 0.01) ** gamma
    
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization using multiplicative weights best-response
    and Nesterov-accelerated methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = make_structured_init(n_init, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: Nesterov Gradient ===
    print(f"\n=== Phase 1: Nesterov PG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_gradient(best_seq, t_alloc, T_start=0.5, T_end=100.0,
                                    lr_init=0.06, n_stages=8, steps_per_stage=5000)
        try_update(a_ng, "ng_1")
    
    # === Phase 2: Alternating Multiplicative Weights ===
    print(f"\n=== Phase 2: Alternating MW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        # Run with different mw_iters for exploration
        a_mw1, _ = alternating_mw(best_seq, t_alloc * 0.6, mw_iters=100, alpha_init=0.4)
        try_update(a_mw1, "mw_fast")
        
        rem = deadline - time.monotonic()
        if rem > 20:
            a_mw2, _ = alternating_mw(a_mw1, min(rem, t_alloc * 0.4), mw_iters=300, alpha_init=0.3)
            try_update(a_mw2, "mw_slow")
    
    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=80000)
        try_update(a_ps, "peak_shave")
    
    # === Phase 4: Second Nesterov round ===
    print(f"\n=== Phase 4: NG Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        a_ng2, _ = nesterov_gradient(best_seq, t_alloc, T_start=5.0, T_end=300.0,
                                      lr_init=0.03, n_stages=6, steps_per_stage=3000)
        try_update(a_ng2, "ng_2")
    
    # === Phase 5: Multi-start with structured inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 20:
                break
            a_init = make_structured_init(n_best, seed=seed + 100 + trial * 7)
            print(f"  Trial {trial+1}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_gradient(a_init, t_sub * 0.5, T_start=0.5, T_end=80.0,
                                          lr_init=0.08, n_stages=5, steps_per_stage=2000)
            try_update(a_sub, f"struct_{trial}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2, _ = alternating_mw(a_sub, min(rem * 0.4, 40.0), mw_iters=150, alpha_init=0.4)
                try_update(a_sub2, f"struct_mw_{trial}")
    
    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-120, -60, -30, 30, 60, 120, 200]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 5000:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))
    
    for alt_n in alt_lengths[:4]:

        if rem > 15 and t_total * 0.3 > 10:

            try_update(a_alt2, f"mw_n{alt_n}")
    
    # === Phase 7: Final refinement with MW + peak shave ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 80.0)
        a_fin, _ = alternating_mw(best_seq, t_fin, mw_iters=200, alpha_init=0.2)
        try_update(a_fin, "final_mw")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.4, 40.0)
        a_ps2, _ = peak_shave(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
