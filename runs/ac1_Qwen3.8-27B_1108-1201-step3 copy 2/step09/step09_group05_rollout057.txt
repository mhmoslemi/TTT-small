def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.05, n_stages=10):

            log_sum_exp = max_Tb + np.log(exp_Tb.sum())

        return func

                grad_a[idx] += w[k] * a[k - idx]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov-accelerated LP dual averaging for global structure
2. L-BFGS with log-sum-exp smoothing and efficient gradient via correlation

5. Multi-length exploration with interpolation warm-start
Key: efficient gradient computation using np.correlate, adaptive time allocation,

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max_b/sum_a^2 assuming sum_a=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lss_func_obj(z_in, T_val, n, t0, time_limit):
    """Log-sum-exp smoothed objective."""
    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lss_func_jac(z_in, T_val, n, t0, time_limit):
    """Log-sum-exp smoothed gradient using efficient correlation."""
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # grad_a[i] = sum_k w[k] * a[k-i] = correlate(w, a, 'valid')[i]
    grad_a = np.correlate(w, a, mode='valid')
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start, T_end, n_stages):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        maxit = max(20, int(remaining / n_stages * 80))
        try:
            res = minimize(
                fun=lambda x: lss_func_obj(x, stage_T, n, t0, time_limit),
                x0=z,
                jac=lambda x: lss_func_jac(x, stage_T, n, t0, time_limit),
                method='L-BFGS-B',
                options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxit * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def structured_init(n, mode):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2.5)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 1.5 + 0.05
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) ** 2
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 0.6 + 0.1 * x
    else:
        a = np.ones(n) * (1 + 0.15 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
        a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
