def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_cross(a, b):
    """Compute linear convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # T is size x n, where T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        # T[k, j] = a_fixed[k-j] for valid k
        # k-j is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = min(size, j + n)
        T[k_start:k_end, j] = a_fixed[:k_end - k_start]
    return T

def solve_lp_fast(a_fixed):
    """
    Solve the LP: min M s.t. T@b <= M, sum(b) = 1, b >= 0
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    size = 2 * n - 1
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Minimize: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: T @ b - M <= 0
    # i.e., [T, -1] @ [b, M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1, M is free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M is free
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            b_opt = res.x[:n]
            M_opt = res.x[n]
            b_opt = np.maximum(b_opt, 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt /= s
            return b_opt, float(M_opt)
    except Exception:
        pass
    
    # Fallback to cvxpy
    try:
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_opt = np.asarray(b_var.value, dtype=np.float64)
            b_opt = np.maximum(b_opt, 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt /= s
            return b_opt, float(M_var.value)
    except Exception:
        pass
    
    return None, None

from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

    return T

                b_opt /= s
            return b_opt, float(res.x[n]), T
    except Exception:
        pass
    return None, None, T


def solve_lp_active_set(a_fixed, active_indices, T=None, max_rounds=30, max_add=100):
    n = len(a_fixed)
    size = 2 * n - 1
    if T is None:
        T = build_conv_matrix(a_fixed)

    for _ in range(max_rounds):
        A_sub = T[active_indices, :]
        A_ub = np.hstack([A_sub, -np.ones((len(active_indices), 1))])
        b_ub = np.zeros(len(active_indices))
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break
            b_sol = np.maximum(res.x[:n], 0)
            s = b_sol.sum()
            if s < 1e-15:
                break
            b_sol /= s
            M_sol = float(res.x[n])
        except Exception:
            break

        T_full = T @ b_sol
        violations = T_full - M_sol
        violated = np.where(violations > 1e-9)[0]
        if len(violated) == 0:
            return b_sol, M_sol, active_indices, T

        new_violated = violated[~np.isin(violated, active_indices)]
        if len(new_violated) == 0:
            return b_sol, M_sol, active_indices, T

        worst = new_violated[np.argsort(violations[new_violated])[::-1]][:max_add]
        active_indices = np.unique(np.concatenate([active_indices, worst]))

    # Fallback to full LP
    result = solve_lp_full(a_fixed, T)
    if result[0] is not None:
        return result[0], result[1], np.arange(size), T
    return None, None, active_indices, T


def compute_multi_peak_subgradient(a, n_peaks=10):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - 0.1 * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    elif len(active) > n_peaks:
        order = np.argsort(b[active])[::-1]
        active = active[order[:n_peaks]]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        a_kj = np.array([a[k - j] for j in idx])
        g[idx] += a_kj + a[idx]
    g /= len(active)
    return g, max_b


def nesterov_lp_main(a0, time_limit, T=None, active_set=None, use_momentum=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a /= s
    y = a.copy()
    prev_y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    beta_momentum = 0.7

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val, active_set, T = solve_lp_active_set(y, active_set, T, max_rounds=20, max_add=80)
        if b_new is None:
            b_new, M_val, active_set, T = solve_lp_full(y, T)
            if b_new is None:
                continue

        alpha = 2.0 / (it + 2.0)
        if use_momentum:
            y_new = (1 - alpha) * y + alpha * b_new
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy < 1e-15:
                break
            y_new /= sy
            prev_y = y
            y = y_new
        else:
            y = (1 - alpha) * y + alpha * b_new
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                break
            y /= sy

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 10:
                rng = np.random.default_rng(it * 31 + 17)
                noise = rng.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y /= y.sum()
                stall_count = 0
                active_set = np.arange(2 * n - 1)

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score, active_set, T


def coordinate_descent_polish(a0, time_limit, n_rounds=5):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)

        for j in idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            partner = peak_k - j
            if partner == j:
                continue
            if partner < 0 or partner >= n or a[partner] < 1e-14:
                continue

            # Try transferring mass from j to partner or vice versa
            for frac in [0.01, 0.03, 0.05, 0.1]:
                for direction in [1, -1]:
                    amount = a[j] * frac * direction
                    if amount < 0:
                        continue
                    if a[j] - amount < -1e-15 or a[partner] + amount < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j] -= amount
                    a_trial[partner] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

        # Also try redistributing to random low-contribution positions
        contrib = np.zeros(n)
        for k in [peak_k, peak_k - 1, peak_k + 1, 0, 2 * n - 2]:
            if 0 <= k < 2 * n - 1:
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                for j in range(js2, je2 + 1):
                    contrib[j] += a[j] * a[k - j]

        top_contrib = np.argsort(contrib)[::-1][:10]
        low_contrib = np.argsort(contrib)[:10]
        rng = np.random.default_rng(rnd + 42)

        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            si = top_contrib[rng.integers(len(top_contrib))]
            di = low_contrib[rng.integers(len(low_contrib))]
            if si == di or a[si] < 1e-14:
                continue
            for frac in [0.02, 0.05, 0.1]:
                amount = a[si] * frac
                a_trial = a.copy()
                a_trial[si] -= amount
                a_trial[di] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    break

        if rnd % 2 == 0:
            print(f"    CD round {rnd}: score={best_score:.8f}")

    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=6):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_loc = np.exp(z_in - z_in.max())
            a_loc /= a_loc.sum()
            b = fast_convolve(a_loc)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return (2.0 * n / stage_T) * (max_Tb + np.log(exp_Tb.sum()))

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_loc = np.exp(z_in - z_in.max())
            a_loc /= a_loc.sum()
            b = fast_convolve(a_loc)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                a_kj = np.array([a_loc[k - j] for j in idx])
                grad_a[idx] += 2.0 * w[k] * (a_kj + a_loc[idx])
            S = np.dot(grad_a, a_loc)
            return a_loc * (grad_a - S)

        iters = max(20, min(500, int(remaining * 50)))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-12,
                                    'maxfun': iters * 100, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out /= a_out.sum()
    return a_out


def structured_init(n, rng, style=0):
    if style == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif style == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 1.5)
    elif style == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif style == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x * (1 - x))
    elif style == 4:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = np.maximum(a, 0) ** 1.2
    elif style == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
        a = np.maximum(a, 0)
    elif style == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a ** 1.5
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x * (1 - x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

        pass

    def try_update(a_cand, label):

        a_cand = np.maximum(a_cand, 0).astype(np.float64)
        s = a_cand.sum()
        if s < 0.01:
            return
        a_cand = a_cand / s
        sc = evaluate_safe(a_cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cand.copy()

    # Quick diverse initializations - pick the best
    print(f"\n=== Phase 0: Diverse Init Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        inits = []
        for style in range(7):
            a_s = structured_init(n_best, rng, style)
            inits.append(a_s)
        # Quick mirror descent on each
        for i, a_init in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            g, mb = compute_multi_peak_subgradient(a_init, n_peaks=8)
            a_trial = a_init - 0.005 * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss
                sc_trial = evaluate_safe(a_trial)

                    try_update(a_trial, f"init_{i}")

    # Phase 1: Nesterov LP with active set (main workhorse)
    print(f"\n=== Phase 1: Nesterov LP + Active Set (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()

        T = build_conv_matrix(best_seq)

        a_lp, _, active_set, T = nesterov_lp_main(best_seq, t_alloc, T=T, active_set=active_set, use_momentum=False)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 2: Second Nesterov LP round with momentum

        a_lp2, _, _, _ = nesterov_lp_main(best_seq, t_alloc, T=T, active_set=active_set, use_momentum=True)

    # Final verification
    if best_seq is not None:

            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
