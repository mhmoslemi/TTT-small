import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix_a(a_fixed):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_a(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def ultra_fast_cd(a0, time_limit, verbose=False, label=""):
    """
    Ultra-fast coordinate descent with analytical optimal delta computation.
    Key: for each coordinate, compute the exact best decrease analytically.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    # Precompute prefix and suffix max for fast unaffected max
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        current_M = float(conv.max())

        for i in range(n):
            if a[i] < 1e-12:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_left = conv[:k_start].max() if k_start > 0 else -1e30
            unaff_right = conv[k_end + 1:].max() if k_end < 2 * n - 2 else -1e30
            unaff_max = float(max(unaff_left, unaff_right))

            affected = conv[k_start:k_end + 1]
            a_shift = a  # Since k_start = i, the shift is a[0:n]

            # Analytical optimal decrease:
            # We want the largest delta > 0 such that:
            #   affected[t] - 2*delta*a_shift[t] <= target for all t
            # where target = max(unaff_max, 0)
            # => delta >= (affected[t] - target) / (2*a_shift[t]) for all t where affected[t] > target
            target = unaff_max
            if target < 0:
                target = 0.0

            # For positions where affected > target, we need delta >= (affected[t] - target)/(2*a_shift[t])
            mask_above = affected > target + 1e-15
            mask_valid = a_shift > 1e-15

            if mask_above.any():
                combined_mask = mask_above & mask_valid
                if combined_mask.any():
                    required_deltas = (affected[combined_mask] - target) / (2.0 * a_shift[combined_mask])
                    delta_needed = float(required_deltas.max())
                    # Also need delta <= a[i] (can't go below zero)
                    delta_needed = min(delta_needed, a[i] * 0.999)

                    if delta_needed > 1e-14:
                        # Compute new affected
                        new_affected = affected - 2.0 * delta_needed * a_shift
                        # Handle diagonal
                        diag_pos = 2 * i - k_start
                        if 0 <= diag_pos < len(new_affected):
                            new_affected = new_affected.copy()
                            new_affected[diag_pos] = conv[2 * i] - 2.0 * delta_needed * a[i] + delta_needed * delta_needed

                        new_max = max(target, float(new_affected.max()))
                        if new_max < current_M - 1e-14:
                            # Accept the update
                            a[i] -= delta_needed
                            sum_a -= delta_needed
                            conv[k_start:k_end + 1] = new_affected
                            current_M = new_max
                            improved = True

                            cur_score = 2.0 * n * current_M / (sum_a * sum_a) if sum_a > 1e-10 else float('inf')
                            if cur_score < best_score - 1e-14:
                                best_score = cur_score
                                best_a = a.copy()
                elif mask_above.any() and not mask_valid.all():
                    # Some positions above target have a_shift = 0, can't reduce them by changing a[i]
                    # Try a smaller delta that at least reduces the max
                    pass
            else:
                # All affected are already <= target, try to reduce further
                # Just try a small reduction to see if it helps
                pass

            # Also try small random perturbations occasionally
            # (handled outside main loop)

        # Renormalize periodically to avoid drift
        if sweeps % 50 == 0 and sum_a > 0:
            a = a / sum_a
            conv = fast_convolve(a)
            current_M = float(conv.max())
            sum_a = 1.0
            cur_score = 2.0 * n * current_M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
            M = current_M

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 50:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    UCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={current_M:.10f}")

    # Final renormalize
    s_final = best_a.sum()
    if s_final > 1e-10:
        best_a = best_a / s_final
        best_score = 2.0 * n * float(fast_convolve(best_a).max())

    return best_a, best_score, sweeps

def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False, nesterov_ave=True):
    """
    LP fictitious play with Nesterov (Aginag-Tseng) averaging.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    # Aginag-Tsenssequent
    x_sequent = a.copy()  # This is the averaged iterate we return
    iters = 0
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Best response for a given b
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            continue

        # Nesterov/Aginag-Tseng sequent update
        gamma_it = 2.0 / (iters + 2.0)
        if nesterov_ave:
            x_sequent = (1.0 - gamma_it) * x_sequent + gamma_it * a_new

        # Evaluate and track best
        candidates = [a_new, b_new, x_sequent, 0.5 * (x_sequent + a_new), 0.5 * (x_sequent + b_new)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * float(fast_convolve(cand_norm).max())
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        a = a_new

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters


def peak_targeted_perturb(a0, rng, strength=0.05):
    """Structured perturbation targeting convolution peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    # Identify peak positions
    peak_threshold = M * 0.90
    peak_positions = np.where(conv >= peak_threshold)[0]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if j < 0 or j >= n:
                continue
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Apply: reduce high-pressure, increase low-pressure
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.2):
    """Randomly zero out and redistribute."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    if keep.sum() < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] *= (1.0 + rng.standard_normal(keep.sum()) * 0.15)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def soft_max_perturb(a0, rng, strength=0.03):
    """Perturb based on soft-max of convolution peaks."""
    n = len(a0)
    conv = fast_convolve(a0)
    M = conv.max()
    # Soft pressure
    beta = 20.0
    weights = np.exp(beta * (conv - M) / M)
    pressure = np.zeros(n)
    for k in range(len(conv)):
        w = weights[k]
        if w < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                pressure[i] += w * a0[i] * a0[j]
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure /= p_max
    a_new = a0 * (1.0 - strength * pressure) + strength * 0.5 * rng.random(n) * a0.max()
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric (palindromic) initialization."""
    half = n // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = profile[-1] * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):

    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)

    w2 = n * rng.uniform(0.04, 0.12)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 1e-6)
    s = a.sum()
    return a / s


def init_optimal_shape(n, rng):

        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    if best_seq is not None:
        print(f"  After Phase 1: best={best_score:.6f} (n={len(best_seq)})")

    # === Phase 2: LP Fictitious Play (main engine) ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        # Run LP FP for a while
        lp_budget = min(400.0, deadline - time.monotonic() - 300)
        lp_budget = max(lp_budget, 60.0)
        print(f"  LP-FP budget={lp_budget:.0f}s, n={n_best}")

        a_fp, score_fp, iters = lp_fictitious_play(
            n_best, lp_budget, initial=current, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"lpfp_n{n_best}")

        # Try a different length for LP
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            alt_n = n_best + 128
            if alt_n <= 2048:
                t_alt = min(rem * 0.4, 150.0)
                print(f"  LP-FP alt n={alt_n}, budget={t_alt:.0f}s")
                initial_alt = interpolate_seq(best_seq, alt_n)
                a_fp2, score_fp2, _ = lp_fictitious_play(
                    alt_n, t_alt, initial=initial_alt, solver=solver, verbose=False
                )
                best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"lpfp_n{alt_n}")

    # === Phase 3: Intensive CD with perturbations ===
    print(f"\n=== Phase 3: Intensive CD + Perturb (best={best_score:.6f}) ===")
    t3_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(8):
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 80.0)
            t_alloc = max(t_alloc, 5.0)
            print(f"  CD round {rnd+1}/8, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")

            a_opt, s_opt, _ = ultra_fast_cd(current, t_alloc, verbose=False, label=f"ph3r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd_r{rnd}")

            # Perturb to escape
            if rnd < 6:
                pert_type = rnd % 3
                if pert_type == 0:
                    current = peak_targeted_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                elif pert_type == 1:
                    current = soft_max_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                else:
                    current = random_restructure(best_seq, rng, frac=0.1 + 0.05 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 4: More LP + CD ===
    print(f"\n=== Phase 4: LP + CD (best={best_score:.6f}) ===")
    t4_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Another LP round
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_lp2 = min(rem * 0.4, 150.0)
            print(f"  LP-FP round 2, n={n_best}, budget={t_lp2:.0f}s")
            a_fp3, score_fp3, _ = lp_fictitious_play(
                n_best, t_lp2, initial=best_seq, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp3, best_seq, best_score, "lpfp_r2")

        # CD polish after LP
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_cd = min(rem * 0.5, 80.0)
            if t_cd > 5:
                a_cd, s_cd, _ = ultra_fast_cd(best_seq, t_cd, verbose=False, label="ph4_cd")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "ph4_cd")

    # === Phase 5: Final polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    t5_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple fast CD passes
        for pass_i in range(3):
            rem = deadline - time.monotonic() - 20
            if rem < 5:
                break
            t_pass = min(rem * 0.3, 40.0)
            if t_pass > 3:
                a_p, s_p, _ = ultra_fast_cd(best_seq, t_pass, label=f"polish{pass_i}")
                best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"polish{pass_i}")

        # Try nearby lengths
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            for delta_n in [64, -64, 128]:
                alt_n = n_best + delta_n
                if alt_n < 256 or alt_n > 2500:
                    continue
                t_alt = min(rem * 0.3, 30.0)
                if t_alt < 3:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = ultra_fast_cd(init_alt, t_alt, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
                rem = deadline - time.monotonic() - 15
                if rem < 10:
                    break

    # === Phase 6: Two-sided CD for final squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_ts = min(rem * 0.7, 30.0)
            if t_ts > 5:
                a_ts, s_ts, _ = two_sided_cd(best_seq, t_ts, label="ts_cd")
                best_seq, best_score = check_and_update(a_ts, best_seq, best_score, "two_sided")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
