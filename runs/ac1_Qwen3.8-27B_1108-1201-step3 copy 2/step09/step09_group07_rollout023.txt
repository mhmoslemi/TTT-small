def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_topk_weighted(a, T, K=5):
    """Gradient of top-K weighted smooth max of conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # Focus on top K peaks
    top_K = np.argsort(w)[::-1][:K]
    w_focused = np.zeros_like(w)
    w_focused[top_K] = w[top_K]
    w_focused /= w_focused.sum()
    
    fft_size = 1 << (n + len(w_focused) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_focused, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def grad_smooth_max(a, T):
    """Standard log-sum-exp gradient."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

def build_conv_matrix(a):

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

                    a_trial[j] *= (1 + 0.1 * f * a[i] / max(a[j], 1e-15))
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                # Accept if no improvement for many steps, break
                if step > 100:
                    break
    
    return best_a, best_score


def make_structured_init(n, rng):
    """Create structured initializations inspired by additive combinatorics."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        # Quadratic residue inspired
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x * rng.integers(1, 5))
    elif choice == 1:
        # Asymmetric ramp
        a = 1.0 + 0.5 * (2*x - 1) * rng.uniform(0.5, 1.5)
    elif choice == 2:
        # Multi-scale
        freqs = rng.uniform(1, 8, size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        amps = rng.uniform(0.1, 0.4, size=3)
        a = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.sin(2*np.pi*f*x + p)
    elif choice == 3:
        # Bump functions
        centers = rng.uniform(0.1, 0.9, size=3)
        widths = rng.uniform(0.05, 0.2, size=3)
        a = 1.0
        for c, w in zip(centers, widths):
            a += rng.uniform(0.2, 0.5) * np.exp(-((x-c)**2) / (2*w**2))
    elif choice == 4:
        # Step-like
        a = 1.0 + 0.3 * np.sin(2*np.pi*x) + 0.2 * np.tanh(3*(2*x-1))
    else:
        # Power law asymmetry
        a = 1.0 + 0.5 * np.power(x, rng.uniform(0.5, 3.0)) - 0.25
    
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive Nesterov accelerated
gradient with peak-focused weighting, LP best-response dynamics, aggressive
pair coordinate descent, and structured multi-start exploration.
Key innovations: adaptive temperature with stall detection, top-K peak
weighting for gradient focus, pair-coordinate descent with peak awareness,
and diverse structured initializations from additive combinatorics.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of log-sum-exp smoothing of max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def grad_topk_weighted(a, T, K):
    """Gradient with top-K peak weighting for focused optimization."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # Focus on top K peaks
    top_idx = np.argsort(w)[::-1][:K]
    w_focused = np.zeros_like(w)
    w_focused[top_idx] = w[top_idx]
    w_focused /= w_focused.sum()
    fft_size = 1 << (n + len(w_focused) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_focused, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_adaptive(a0, time_limit, T_start, T_end, lr_init,
                      n_stages, steps_per_stage, stall_threshold,
                      T_increase, use_topk, K):
    """Nesterov accelerated projected gradient with adaptive temperature."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0).copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()

    T = T_start
    stall_count = 0
    t_k = 1.0

    for stage in range(n_stages):
        # Geometric temperature schedule as baseline
        T_sched = T_start * (T_end / T_start) ** (stage / max(n_stages - 1, 1))
        T = max(T, T_sched)

        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            lr = lr_init / (1.0 + 0.003 * (stage * steps_per_stage + step))

            # Nesterov momentum
            t_prev = t_k
            t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            beta = (t_prev - 1.0) / t_k
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys

            # Gradient
            if use_topk:
                grad = grad_topk_weighted(y, T, K)
            else:
                grad = grad_smooth_max(y, T)

            x_prev = x.copy()
            x_new = y - lr * grad
            x = project_simplex(x_new)

            # Monitor progress
            if (stage * steps_per_stage + step) % 200 == 0:
                sc = score_normalized(x)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = x.copy()
                    stall_count = 0
                elif (stage * steps_per_stage + step) % 1000 < 200:
                    stall_count += 1

            # Adaptive temperature increase on stall
            if stall_count >= stall_threshold:
                T = min(T * T_increase, 10000.0)
                stall_count = 0
                rng_local = np.random.default_rng(stage * steps_per_stage + step)
                x = x + 0.015 * rng_local.standard_normal(n)
                x = project_simplex(x)
                if (stage * steps_per_stage + step) % 2000 < 200:
                    print(f"    Adaptive: step={stage*steps_per_stage+step} T={T:.1f} score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def lp_refine(a0, time_limit, solver, n_max_iters):
    """LP best-response dynamics with Nesterov dual averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        score_y = score_normalized(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                rng_local = np.random.default_rng(it + 7)
                y = y + 0.025 * rng_local.standard_normal(n)
                sig = np.linspace(0, 2 * np.pi, n)
                y = y + 0.015 * np.sin(3.0 * sig + rng_local.random() * 2 * np.pi)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 2000 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, n_steps):
    """Aggressive pair coordinate descent with peak-aware priority and SA acceptance."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0
    current_temp = 0.01

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        # Contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        top_n = min(10, len(idx))
        order = np.argsort(contribs)[::-1][:top_n]

        for ci in order:
            i = idx[ci]
            j = peak_k - i
            if i >= n or j >= n or i < 0 or j < 0:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue

            if i == j:
                for factor in [0.995, 0.99, 0.98, 0.96, 0.93]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-16:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
            else:
                for f1 in [0.995, 0.99, 0.97, 0.94]:
                    for f2 in [0.995, 0.99, 0.97, 0.94]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-16:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Simulated annealing: occasionally accept worse moves
            if no_improve > 5:
                current_temp *= 0.95
                rng_local = np.random.default_rng(step)
                if rng_local.random() < current_temp:
                    i = int(rng_local.integers(0, n))
                    j = int(rng_local.integers(0, n))
                    if i != j and a[i] > 1e-10 and a[j] > 1e-10:
                        a_trial = a.copy()
                        delta = 0.05 * a[i]
                        a_trial[i] -= delta
                        a_trial[j] += delta
                        if a_trial[i] >= 0:
                            a_trial = project_simplex(a_trial)
                            sc = score_normalized(a_trial)
                            if sc < best_score + current_temp * 0.01:
                                a = a_trial
                if no_improve > 50:
                    break
            elif step > 200:
                break

    return best_a, best_score


def make_structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)

    if choice == 0:
        # Linear asymmetry
        a = 1.0 + 0.6 * (2 * x - 1) * float(rng.uniform(0.5, 1.2))
    elif choice == 1:
        # Sinusoidal
        freq = float(rng.uniform(1.0, 6.0))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.2, 0.5))
        a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)
    elif choice == 2:
        # Multi-scale
        a = 1.0
        for _ in range(3):
            f = float(rng.uniform(1, 8))
            p = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.1, 0.3))
            a += amp * np.sin(2 * np.pi * f * x + p)
    elif choice == 3:
        # Gaussian bumps
        a = np.ones(n) * 0.5
        for _ in range(int(rng.integers(2, 5))):
            c = float(rng.uniform(0.1, 0.9))
            w = float(rng.uniform(0.03, 0.15))
            amp = float(rng.uniform(0.3, 0.8))
            a += amp * np.exp(-((x - c) ** 2) / (2 * w ** 2))
    elif choice == 4:
        # Power law
        p = float(rng.uniform(0.3, 3.0))
        a = 1.0 + 0.5 * np.power(x, p) - 0.25
    elif choice == 5:
        # Tangent asymmetry
        a = 1.0 + 0.3 * np.tanh(3.0 * (2 * x - 1)) + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 6:
        # Cosine with phase
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x + float(rng.uniform(0, 2 * np.pi)))
    else:
        # Random smooth
        coeff = rng.standard_normal(5) * 0.15
        a = 1.0
        for k, c in enumerate(coeff):
            a += c * np.cos(2 * np.pi * (k + 1) * x)

    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: adaptive Nesterov, LP best-response dynamics,
    pair coordinate descent with SA, structured multi-start, alt lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is None:
        n_init = 960
        best_seq = make_structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Adaptive Nesterov PG ===
    print(f"\n=== Phase 1: Adaptive NG n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_adaptive(best_seq, t_alloc, T_start=5.0, T_end=400.0,
                                     lr_init=0.08, n_stages=20, steps_per_stage=1500,
                                     stall_threshold=25, T_increase=1.3,
                                     use_topk=False, K=5)
        try_update(a_ng, "ng_main")

    # === Phase 2: LP best-response refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Top-K focused Nesterov ===
    print(f"\n=== Phase 3: Top-K NG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 20:
        a_ng2, _ = nesterov_adaptive(best_seq, t_alloc, T_start=15.0, T_end=600.0,
                                      lr_init=0.04, n_stages=12, steps_per_stage=1000,
                                      stall_threshold=15, T_increase=1.5,
                                      use_topk=True, K=3)
        try_update(a_ng2, "topk_ng")

    # === Phase 4: Pair coordinate descent ===
    print(f"\n=== Phase 4: Pair CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, _ = pair_coordinate_descent(best_seq, t_alloc, n_steps=200000)
        try_update(a_cd, "pair_cd")

    # === Phase 5: Second LP round from best ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        a_lp2, _ = lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=40000)
        try_update(a_lp2, "lp_2")

    # === Phase 6: Multi-start ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_starts = min(3, int(rem // 90))
        for s in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.18, 80.0)
            if t_sub < 30:
                break
            a_init = make_structured_init(n_best, rng)
            print(f"  Start {s+1}/{n_starts}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_adaptive(a_init, t_sub * 0.55, T_start=3.0, T_end=250.0,
                                          lr_init=0.1, n_stages=10, steps_per_stage=800,
                                          stall_threshold=20, T_increase=1.4,
                                          use_topk=False, K=5)
            try_update(a_sub, f"start_{s}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2, _ = lp_refine(a_sub, min(rem * 0.3, 30.0), solver=solver, n_max_iters=6000)
                try_update(a_sub2, f"start_{s}_lp")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    deltas = [-80, -40, 40, 80, 120]
    alt_lengths = [n_best + d for d in deltas if 256 <= n_best + d <= 4096]
    for alt_n in sorted(set(alt_lengths))[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.12, 80.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_adaptive(init_alt, t_total * 0.55, T_start=3.0, T_end=200.0,
                                      lr_init=0.08, n_stages=8, steps_per_stage=800,
                                      stall_threshold=20, T_increase=1.4,
                                      use_topk=False, K=5)
        try_update(a_alt, f"alt_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 25.0), solver=solver, n_max_iters=4000)
            try_update(a_alt2, f"alt_n{alt_n}_lp")

    # === Phase 8: Final polish (LP + CD) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin, _ = lp_refine(best_seq, min(rem * 0.3, 50.0), solver=solver, n_max_iters=8000)
        try_update(a_fin, "final_lp")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin2, _ = pair_coordinate_descent(best_seq, min(rem * 0.2, 25.0), n_steps=50000)
        try_update(a_fin2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
