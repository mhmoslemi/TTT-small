import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000, top_k_interval=5000):
    """
    Enhanced Nesterov dual averaging LP refinement with:
    - Adaptive restarts on stall
    - Periodic top-k LP variant
    - Dual-guided momentum

    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(98765)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Every top_k_interval iterations, use top-2 LP variant
        tk = 1
        if it % top_k_interval < top_k_interval // 5:
            tk = 2

        b_new, M_b, dual_y = solve_lp_best_response(y, solver, top_k=tk)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)

        # Standard Nesterov update
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 6:
                # Adaptive noise: scale with stall severity
                noise_scale = 0.05 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy < 1e-15:
                    y = best_a.copy()
                else:
                    y = y / sy
                stall_rounds = 0

            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def kkt_local_search(a0, time_limit, n_steps=50000):
    """
    KKT-verified local search: check if current point is a local minimum
    by looking for a descent direction in the active set.
    If found, take a step in that direction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    rng = np.random.default_rng(54321)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active set (positions within 0.5% of max)
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient: average of gradients over active positions
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Projected gradient step (maintain sum=1, a>=0)
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            if no_improve >= 20:
                lr *= 0.85
                no_improve = 0

        # Periodic: try a random perturbation + projection
        if step % 5000 == 0 and step > 0:
            pert = a + rng.standard_normal(n) * 0.01
            pert = np.maximum(pert, 0)
            ps = pert.sum()
            if ps > 1e-15:
                pert = pert / ps
                sc = 2.0 * n * fast_convolve(pert).max()
                if sc < best_score:
                    best_score = sc
                    best_a = pert.copy()
                    a = pert

        lr = max(1e-8, lr)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT search {elapsed:.0f}s: score={best_score:.10f} lr={lr:.8f} active={len(active)}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=30000):
    """
    Fine-grained coordinate descent: try small adjustments to individual
    or pairs of coordinates to reduce the max convolution value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    # Pre-compute which positions contribute to the peak
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributing positions
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(12, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        # Try reducing individual top contributors
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute proportionally to others
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try pair swaps
        if not improved and len(top_positions) >= 2:
            for i in range(min(4, len(top_positions))):
                for j in range(i + 1, min(4, len(top_positions))):
                    p, q = top_positions[i], top_positions[j]
                    if a[p] < 1e-12 or a[q] < 1e-12:
                        continue
                    for eps in [0.001, 0.002, 0.003, 0.005]:
                        delta = eps * min(a[p], a[q])
                        a_trial = a.copy()
                        a_trial[p] = a[p] - delta
                        a_trial[q] = a[q] + delta
                        if a_trial[p] < 0 or a_trial[q] < 0:
                            continue
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                # Try reverse swap
                for i in range(min(4, len(top_positions))):
                    p = top_positions[i]
                    if a[p] < 1e-12:
                        continue
                    # Move mass from p to a random non-peak position
                    others = list(set(range(n)) - set(top_positions))
                    if not others:
                        continue
                    target = rng.choice(others)
                    for eps in [0.002, 0.005, 0.01]:
                        delta = eps * a[p]
                        a_trial = a.copy()
                        a_trial[p] = a[p] - delta
                        a_trial[target] = a[target] + delta
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Add small random perturbation to escape
                a = a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f}")

    return best_a, best_score


def logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with low temperature LogSumExp for smooth optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    T_matrix = build_conv_matrix(a)  # Will rebuild each iteration

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            T_in = build_conv_matrix(a_in)
            b_in = fast_convolve(a_in)
            Tb = T_in * b_in
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            T_in = build_conv_matrix(a_in)
            Tb = T_in * b_in
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    result = softmax_from_z(z)
    return result


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.05
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.6 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # Two-hump: less mass in center
        a = np.exp(-4 * (x - 0.25) ** 2) + np.exp(-4 * (x - 0.75) ** 2) + 0.1
    else:
        # Cosine taper with bump
        a = np.cos(np.pi * x / 2) ** 0.5 * (1 + 0.2 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def momentum_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """
    LP alternating with heavy ball momentum and adaptive step size.
    Different from Nesterov: uses a fixed momentum parameter with
    periodic adaptation based on progress.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)  # velocity
    beta = 0.9  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall = 0
    rng = np.random.default_rng(11111)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Heavy ball update
        v = beta * v + (1 - beta) * (b_new - a)
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # If worse, dampen velocity
            v *= 0.5
            stall += 1
        a = a / a.sum()

        if (it + 1) % 2000 == 0:
            if best_score < prev_score - 1e-9:
                stall = 0
                prev_score = best_score
            else:
                stall += 1
                if stall >= 5:
                    # Reset momentum and add noise
                    v = np.zeros(n)
                    a = a + rng.standard_normal(n) * 0.02
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MomentumLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Nesterov LP (main workhorse)
    Phase 2: Momentum LP (different dynamics)
    Phase 3: KKT local search (fine-grained)
    Phase 4: Coordinate descent (very fine)
    Phase 5: LogSumExp L-BFGS (smooth)
    Phase 6: Try alternative lengths
    Phase 7: Final combined polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a_arr, label):
        nonlocal best_seq, best_score
        a_arr = np.maximum(np.asarray(a_arr, dtype=np.float64), 0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_arr = a_arr / s
        sc = evaluate_local(a_arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a_arr)})")

    # === Phase 1: Nesterov LP (main) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.38, 650.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 2: Momentum LP ===
    print(f"\n=== Phase 2: Momentum LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mom, _ = momentum_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_mom, "momentum_lp")

    # === Phase 3: KKT local search ===
    print(f"\n=== Phase 3: KKT Local Search (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_kkt, _ = kkt_local_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_kkt, "kkt_search")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: LogSumExp L-BFGS ===
    print(f"\n=== Phase 5: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_lb = logsumexp_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=6000)
        try_update(a_lb, "logsumexp")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt Length (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        # Try a slightly different length
        alt_n = int(n_best * 1.2)
        t_alt = min(rem * 0.3, 250.0)
        if t_alt > 60:
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.7, solver=solver, n_max_iters=30000)
            try_update(a_alt, f"lp_n{alt_n}")

    # === Phase 7: Second Nesterov pass on best ===
    print(f"\n=== Phase 7: Second Nesterov (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_2 = min(rem * 0.4, 200.0)
        print(f"  Budget: {t_2:.0f}s")
        a_2, _ = nesterov_lp_refine(best_seq, t_2, solver=solver, n_max_iters=50000)
        try_update(a_2, "nesterov_2nd")

    # === Phase 8: Final coordinate descent polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 60.0)
        if t_fin > 15:
            a_fin, _ = coordinate_descent_refine(best_seq, t_fin, n_steps=30000)
            try_update(a_fin, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_kkt = min(rem * 0.5, 40.0)
        if t_kkt > 10:
            a_kkt2, _ = kkt_local_search(best_seq, t_kkt, n_steps=20000)
            try_update(a_kkt2, "final_kkt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
