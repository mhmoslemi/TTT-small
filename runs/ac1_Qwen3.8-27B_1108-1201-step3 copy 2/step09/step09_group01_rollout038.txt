To handle this, we can use:

3. **Optimistic step**: Use y_{t-1} (the previous iterate) to compute the gradient, not the current one.

Algorithm:
```

    # Compute smoothed gradient using y_{t-1}
    conv = x_{t-1} * y_{t-1}  (FFT)
    weights = softmax(beta * (conv - max(conv)))
    grad = sum_k weights[k] * shift(y_{t-1}, k)
    
    # Mirror descent update
    x_t = x_{t-1} * exp(-eta_t * grad) / Z
    
    # Similarly update y using x_{t-1}
    y_t = y_{t-1} * exp(-eta_t * grad_y) / Z
    
    # Nesterov sequent update
    seq_t = (1 - alpha_t) * seq_{t-1} + alpha_t * x_t
    
Evaluation: use seq_t (or average of x_t and y_t)
```

    y = np.ones(n) / n
    x_seq = x.copy()
    y_seq = y.copy()
    
    best_x = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()

    t = 0

    eta = eta_init

        t += 1

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_gradient_smooth(conv, other_seq, beta, threshold=1e-8):
    """
    Compute smoothed gradient of max(conv) w.r.t. one player's sequence.
    grad[j] = sum_k softmax_k * other_seq[k-j] for valid (j, k-j).
    
    conv: the convolution of x * y (length 2n-1)
    other_seq: the opponent's sequence (length n)
    beta: smoothing parameter
    """
    n = len(other_seq)
    max_c = conv.max()
    scaled = beta * (conv - max_c)
    scaled = np.clip(scaled, -60, 0)
    weights = np.exp(scaled)
    Z = weights.sum()
    if Z < 1e-30:
        # Fallback: use only the argmax
        k_star = conv.argmax()
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        j = np.arange(i_start, i_end + 1)
        grad[j] = other_seq[k_star - j]
        return grad
    
    weights = weights / Z
    
    # Only process significant weights
    sig_idx = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    
    for k in sig_idx:
        w = weights[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += w * other_seq[k - j]
    
    return grad


def optimistic_mirror_descent_fp(n, time_limit, initial=None, beta_init=40.0, 
                                  eta_init=0.05, seed=42, verbose=True, label=""):
    """
    Optimistic Mirror Descent Fictitious Play.
    Uses smoothed max (log-sum-exp) for differentiability.
    Nesterov-accelerated averaging for the final answer.
    """
    rng = np.random.default_rng(seed)
    
    if initial is not None:
        x = np.maximum(np.asarray(initial, dtype=np.float64), 0).copy()
        y = np.maximum(np.asarray(initial, dtype=np.float64), 0).copy()
    else:
        x = np.ones(n) / n
        y = np.ones(n) / n
    
    x = x / x.sum()
    y = y / y.sum()
    
    x_prev = x.copy()
    y_prev = y.copy()
    
    # Nesterov sequents
    x_seq = x.copy()
    y_seq = y.copy()
    
    best_seq = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()
    
    t0 = time.monotonic()
    t = 0
    beta = beta_init
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute cross-convolution x_prev * y_prev
        conv_xy = fast_convolve(x_prev, y_prev)
        
        # Optimistic: use delayed opponent for gradient
        # x gradient uses y_prev (delayed)
        # y gradient uses x_prev (delayed)
        grad_x = compute_gradient_smooth(conv_xy, y_prev, beta)
        grad_y = compute_gradient_smooth(conv_xy, x_prev, beta)
        
        # Normalize gradients for stability
        gx = grad_x / (max(grad_x.max(), 1e-15))
        gy = grad_y / (max(grad_y.max(), 1e-15))
        
        # Mirror descent update (entropic / multiplicative weights)
        log_x = np.log(x + 1e-30) - eta * gx
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x = simplex_project(x_new)
        
        log_y = np.log(y + 1e-30) - eta * gy
        log_y -= log_y.max()
        y_new = np.exp(log_y)
        y = simplex_project(y_new)
        
        # Nesterov sequent update
        alpha = 2.0 / (t + 2.0)
        x_seq = (1.0 - alpha) * x_seq + alpha * x
        y_seq = (1.0 - alpha) * y_seq + alpha * y
        
        # Evaluate candidates periodically
        if t % 50 == 0 or t <= 10:
            for cand in [x, y, x_seq, y_seq, 0.5 * (x_seq + y_seq)]:
                c = np.maximum(cand, 0)
                s = c.sum()
                if s < 1e-10:
                    continue
                cn = c / s
                score = 2.0 * n * fast_convolve(cn).max()
                if score < best_score:
                    best_score = score
                    best_seq = cn.copy()
        
        # Adaptive parameters
        if t % 200 == 0:
            beta *= 1.15
            eta *= 0.97
            if eta < 1e-6:
                break
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    OMD-FP {label} t={t} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.1f} eta={eta:.5f}")
    
    # Final evaluation
    for cand in [x, y, x_seq, y_seq, 0.5 * (x_seq + y_seq)]:
        c = np.maximum(cand, 0)
        s = c.sum()
        if s < 1e-10:
            continue
        cn = c / s
        score = 2.0 * n * fast_convolve(cn).max()
        if score < best_score:
            best_score = score
            best_seq = cn.copy()
    
    return best_seq, best_score, t


def fast_coord_descent(a0, time_limit, verbose=False, label="", max_sweeps=5000):
    """
    Fast coordinate descent minimizing 2n*max(conv)/sum^2.
    Uses analytical candidate steps for each coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and sweeps < max_sweeps:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-14:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            if k_start > 0 and k_end < 2 * n - 2:
                unaff_max = max(float(conv[:k_start].max()) if k_start > 0 else 0,
                                float(conv[k_end+1:].max()) if k_end < 2*n-2 else 0)
            elif k_start > 0:
                unaff_max = float(conv[:k_start].max())
            elif k_end < 2 * n - 2:
                unaff_max = float(conv[k_end+1:].max())
            else:
                unaff_max = 0.0
            
            affected = conv[k_start:k_end+1].copy()
            a_ref = a.copy()
            
            best_delta = 0.0
            best_new_score = best_score if s > 0 else float('inf')
            
            # Candidate deltas: fractions of a[i]
            fracs = [0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 0.99,
                     -0.01, -0.03, -0.05, -0.1, -0.15, -0.25, -0.4, -0.6, -0.8, -0.99]
            
            for f in fracs:
                delta = f * a[i]
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                if new_val > 2.0:
                    continue
                
                new_sum = s + delta
                if new_sum < 1e-6:
                    continue
                
                new_aff = affected + 2.0 * delta * a_ref
                # Check if 2i is in the affected range
                two_i = 2 * i
                if k_start <= two_i <= k_end:
                    idx = two_i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[two_i] + 2.0 * delta * a[i] + delta * delta
                    if new_aff[idx] < 0:
                        new_aff[idx] = 0
                
                new_M = max(unaff_max, float(new_aff.max()))
                new_score = 2.0 * n * new_M / (new_sum * new_sum)
                
                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = delta
            
            if best_delta != 0.0:
                delta = best_delta
                a[i] += delta
                if a[i] < 0:
                    a[i] = 0
                    delta = -a[i]  # adjust
                s += delta
                
                # Update convolution
                changes = 2.0 * delta * a_ref
                two_i = 2 * i
                if k_start <= two_i <= k_end:
                    idx = two_i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * (a[i] - delta) + delta * delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))
                
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, sweeps

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng, n_modes=3):
    """Create a symmetric initialization with smooth profile."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    profile = np.ones(half) * 0.8
    for j in range(n_modes):
        freq = rng.uniform(1.0, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    
    if n % 2 == 0:
        a = np.concatenate([profile[:n//2], profile[:n//2][::-1]])
    else:
        a = np.concatenate([profile[:n//2], [profile[n//2]], profile[:n//2][::-1]])
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Bimodal initialization (mass at edges)."""
    width = n * rng.uniform(0.1, 0.3)
    t = np.arange(n)
    a = np.exp(-((t - 0) / width) ** 2) + np.exp(-((t - (n-1)) / width) ** 2)
    a = np.maximum(a, 1e-6)
    s = a.sum()
    return a / s


def init_barkers_like(n, rng):

    period = int(n * rng.uniform(0.1, 0.3))

    a = np.maximum(a, 0.05)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    s = a.sum()
    return a / s

                    continue
                cn = c / sc
                score = 2.0 * n * fast_convolve(cn).max()
                if score < best_score:
                    best_score = score
                    best_seq = cn.copy()
        
        x_prev = x.copy()
        y_prev = y.copy()
        eta *= 0.999
    
    # Phase B: Smooth gradient (log-sum-exp) for refinement
    beta = 30.0
    eta = 0.03
    x = best_seq.copy()
    y = best_seq.copy()
    x_prev = x.copy()
    y_prev = y.copy()
    x_seq = x.copy()
    y_seq = y.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv_xy = fast_convolve(x_prev, y_prev)
        
        grad_x = compute_gradient_smooth(conv_xy, y_prev, beta)
        grad_y = compute_gradient_smooth(conv_xy, x_prev, beta)
        
        gx = grad_x / max(grad_x.max(), 1e-15)
        gy = grad_y / max(grad_y.max(), 1e-15)
        
        log_x = np.log(x + 1e-30) - eta * gx
        log_x -= log_x.max()
        x = simplex_project(np.exp(log_x))
        
        log_y = np.log(y + 1e-30) - eta * gy
        log_y -= log_y.max()
        y = simplex_project(np.exp(log_y))
        
        alpha = 2.0 / (t + 2.0)
        x_seq = (1 - alpha) * x_seq + alpha * x
        y_seq = (1 - alpha) * y_seq + alpha * y
        
        if t % 500 == 0:
            for cand in [x_seq, y_seq, 0.5*(x_seq+y_seq), x, y]:
                c = np.maximum(cand, 0)
                sc = c.sum()
                if sc < 1e-10:
                    continue
                cn = c / sc
                score = 2.0 * n * fast_convolve(cn).max()
                if score < best_score:
                    best_score = score
                    best_seq = cn.copy()
        
        x_prev = x.copy()
        y_prev = y.copy()
        
        if t % 2000 == 0:
            beta *= 1.2
            eta *= 0.95
            if eta < 1e-6:
                break
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MS-MD t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Quick MD-FP at target length to get a good starting point
    Phase 2: Coordinate descent polishing
    Phase 3: Multi-scale MD-FP for further improvement
    Phase 4: Fine coordinate descent
    Phase 5: Multi-length exploration
    Phase 6: Final intensive polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    
    best_seq = None
    best_score = float('inf')
    
    # Start from parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Starting from parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    n_target = 1024 if parent is None else len(parent)
    
    # === Phase 1: Quick Mirror Descent FP ===
    print(f"\n=== Phase 1: MD-FP (n={n_target}, current best={best_score:.6f}) ===")
    t1_alloc = min(250, (deadline - time.monotonic()) * 0.15)
    
    if best_seq is not None:
        init1 = best_seq.copy() if len(best_seq) == n_target else interpolate_seq(best_seq, n_target)
    else:
        init1 = init_symmetric(n_target, rng)
    
    a1, s1 = optimistic_mirror_descent_fp(n_target, t1_alloc, initial=init1, 
                                            beta_init=30, eta_init=0.06, 
                                            verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "md_fp_p1")
    
    # Quick CD polish
    rem = deadline - time.monotonic() - 1400
    t_cd1 = max(5, min(60, rem))
    a_cd1, s_cd1, _ = fast_coord_descent(a1, t_cd1, label="p1cd")
    best_seq, best_score = check_and_update(a_cd1, best_seq, best_score, "cd_p1")
    
    # === Phase 2: Continued MD-FP with different params ===
    print(f"\n=== Phase 2: MD-FP variant (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 1200
    t2_alloc = max(10, min(200, rem * 0.5))
    
    # Try with symmetric init perturbation
    perturb = best_seq * (1.0 + 0.05 * rng.standard_normal(n_target))
    perturb = np.maximum(perturb, 0)
    perturb = perturb / perturb.sum()
    
    a2, s2 = optimistic_mirror_descent_fp(n_target, t2_alloc, initial=perturb,
                                            beta_init=50, eta_init=0.04,
                                            seed=7, verbose=True, label="p2")
    best_seq, best_score = check_and_update(a2, best_seq, best_score, "md_fp_p2")
    
    # CD polish
    rem = deadline - time.monotonic() - 1100
    t_cd2 = max(5, min(80, rem))
    a_cd2, s_cd2, _ = fast_coord_descent(a2, t_cd2, label="p2cd")
    best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_p2")
    
    # === Phase 3: Multi-scale MD-FP ===
    print(f"\n=== Phase 3: Multi-scale MD-FP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 800
    t3_alloc = max(10, min(300, rem * 0.6))
    
    a3, s3 = multi_scale_md_fp(n_target, t3_alloc, initial=best_seq, verbose=True)
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "msmd_p3")
    
    # CD polish
    rem = deadline - time.monotonic() - 700
    t_cd3 = max(5, min(100, rem))
    a_cd3, s_cd3, _ = fast_coord_descent(a3, t_cd3, label="p3cd")
    best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, "cd_p3")
    
    # === Phase 4: Fine coordinate descent (intensive) ===
    print(f"\n=== Phase 4: Fine CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    t4_alloc = max(10, min(200, rem * 0.7))
    
    a4, s4, _ = fine_coord_descent(best_seq, t4_alloc, label="fine1")
    best_seq, best_score = check_and_update(a4, best_seq, best_score, "fine_cd_1")
    
    # Another round of MD-FP starting from polished point
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_fp = min(rem * 0.5, 120)
        a_fp, s_fp = optimistic_mirror_descent_fp(n_target, t_fp, initial=best_seq,
                                                    beta_init=60, eta_init=0.02,
                                                    seed=99, verbose=False, label="p4fp")
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "md_fp_p4")
        
        # Fine CD again
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_f4, s_f4, _ = fine_coord_descent(best_seq, min(rem * 0.6, 80), label="fine2")
            best_seq, best_score = check_and_update(a_f4, best_seq, best_score, "fine_cd_2")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    
    for alt_n in [n_target + 128, n_target - 128, n_target + 256]:
        if alt_n < 512 or alt_n > 2048:
            continue
        if rem < 30:
            break
        t_alt = min(rem * 0.3, 60)
        rem = deadline - time.monotonic() - 50
        if rem < 20:
            break
        
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt, s_alt = optimistic_mirror_descent_fp(alt_n, t_alt, initial=init_alt,
                                                     beta_init=40, eta_init=0.05,
                                                     seed=alt_n % 100, verbose=False, label=f"n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
        
        # Quick CD on the new length
        if rem > 20 and len(best_seq) == alt_n:
            a_alt2, s_alt2, _ = fast_coord_descent(a_alt, min(rem * 0.4, 30), label=f"cd_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"cd_alt_n{alt_n}")
    
    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10
    
    if best_seq is not None and rem > 30:
        # Best CD with many sweeps
        t_final_cd = min(rem * 0.4, 100)
        a_fin, s_fin, _ = fine_coord_descent(best_seq, t_final_cd, label="final_fine")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_fine_cd")
        
        rem = deadline - time.monotonic() - 5
        if rem > 20:
            # One more MD-FP push
            t_push = min(rem * 0.5, 60)
            a_push, s_push = optimistic_mirror_descent_fp(len(best_seq), t_push, 
                                                           initial=best_seq,
                                                           beta_init=80, eta_init=0.01,
                                                           seed=200, verbose=False, label="push")
            best_seq, best_score = check_and_update(a_push, best_seq, best_score, "final_push")
            
            rem = deadline - time.monotonic() - 2
            if rem > 10:
                a_fc, s_fc, _ = fine_coord_descent(best_seq, rem * 0.7, label="final_cd2")
                best_seq, best_score = check_and_update(a_fc, best_seq, best_score, "final_cd2")
        
        # Final fast CD round
        rem = deadline - time.monotonic() - 2
        if rem > 5:
            a_fcd, s_fcd, _ = fast_coord_descent(best_seq, rem * 0.8, label="final_fast")
            best_seq, best_score = check_and_update(a_fcd, best_seq, best_score, "final_fast_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification with the actual evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
