                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:

    return softmax_from_z(z)

def mirror_descent_refine(a0, time_limit, eta_init=0.01, n_steps=50000):
    """Mirror descent with negative entropy (multiplicative weight updates)."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta_init
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Subgradient of max(b) w.r.t. a
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * n * a[k_star - idx]
        
        # Mirror descent update: a_i <- a_i * exp(-eta * g_i)
        # This is the "exponentiated gradient" step
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # Slowly decrease eta
            if step % 500 == 0:
                eta *= 0.95
        else:
            # Don't update, decrease eta
            eta *= 0.99
        
        if best_score < 2.0 * n * fast_convolve(a_new).max() * 0.999 and step % 1000 == 0:
            a = a_new  # Only commit if close to best
        else:
            a = a_new  # Actually always commit for exploration
    
    return best_a, best_score

def peak_shave_analytic(a0, time_limit, n_steps=20000):
    """Peak shaving with analytical optimal step size via quadratic approximation."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Consider interior peaks (exclude edges which are naturally lower)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]
        
        # Find the secondary peak
        b_work2 = b_work.copy()
        b_work2[peak_k] = -np.inf
        sec_k = int(np.argmax(b_work2))
        
        # For each peak, find the most impactful coordinate to reduce
        for pk in [peak_k, sec_k]:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_i = idx[np.argmax(contribs)]
            
            if a[top_i] < 1e-15:
                continue
            
            # Quadratic approximation:
            # b(pk) after reducing a[top_i] by delta and adding to a[j]:
            # b(pk) new ≈ b(pk) - delta * a[pk-top_i] - delta * (a[pk-top_i] if top_i != pk-top_i else 0)
            # Actually: b(pk) = sum_i a_i * a_{pk-i}
            # d(b(pk))/d(a_i) = a_{pk-i} + a_{pk-(n-i)} ... no wait
            # b(pk) = sum_{i=0}^{n-1} a_i * a_{pk-i} (where a_j = 0 if j not in [0,n-1])
            # d(b(pk))/d(a_i) = a_{pk-i} (for the term a_i * a_{pk-i})
            # But also a_i appears in the term a_{pk-i} * a_i, which is the same term
            # So d(b(pk))/d(a_i) = a_{pk-i} (counting once, since it's the same product)
            # Wait, b(pk) = sum_i a_i * a_{pk-i}. The term involving a_i is a_i * a_{pk-i}.
            # So d(b(pk))/d(a_i) = a_{pk-i}.
            # But we also have the constraint that sum(a) = 1, so if we decrease a_i by delta,
            # we must increase some other a_j by delta.
            
            # Let's try: decrease a[top_i] by delta, increase a[j] by delta
            # where j is a coordinate with low contribution to both peaks.
            
            # Find j with minimum contribution to all peaks
            min_contrib = np.inf
            best_j = 0
            for j in range(n):
                if j == top_i:
                    continue
                c = 0
                for pk2 in [peak_k, sec_k]:
                    if 0 <= pk2 - j < n:
                        c += a[pk2 - j]
                if c < min_contrib:
                    min_contrib = c
                    best_j = j
            
            # Optimal delta: we want to minimize max of the affected peaks
            # Linear approx: b(pk) - delta * a[pk - top_i] + delta * a[pk - best_j] (for each peak)
            # For peak_k: change = -delta * a[peak_k - top_i] + delta * a[peak_k - best_j] (if valid)
            
            dk_top = a[peak_k - top_i] if 0 <= peak_k - top_i < n else 0
            dk_bot = a[peak_k - best_j] if 0 <= peak_k - best_j < n else 0
            ds_top = a[sec_k - top_i] if 0 <= sec_k - top_i < n else 0
            ds_bot = a[sec_k - best_j] if 0 <= sec_k - best_j < n else 0
            
            # b(peak_k) delta = -delta * dk_top + delta * dk_bot
            # b(sec_k) delta = -delta * ds_top + delta * ds_bot
            
            # We want to maximize the reduction of the max, so:
            # If peak_k is the main peak, we want -dk_top + dk_bot < 0, i.e., dk_top > dk_bot
            net_reduction_k = dk_top - dk_bot
            net_reduction_s = ds_top - ds_bot
            
            if net_reduction_k <= 0 and net_reduction_s <= 0:
                continue
            
            # Maximum delta before a[top_i] hits 0
            max_delta = a[top_i] * 0.5
            
            # Try a few fractions of max_delta
            improved = False
            for frac in [0.3, 0.15, 0.08, 0.04, 0.02]:
                delta = max_delta * frac
                a_trial = a.copy()
                a_trial[top_i] -= delta
                a_trial[best_j] += delta
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try small random perturbations
            if step % 100 == 0:
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = norm_sequence(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
        
        if step > 5000:
            b = fast_convolve(a)
            if np.std(b[1:n-1]) < 0.01 * b.max():
                break  # Autocorrelation is very flat, hard to improve
    
    return best_a, best_score

def spectral_init(n, rng, mode=0):
    """Generate initial sequence via spectral shaping."""
    if mode == 0:
        # Flat spectrum with random phase
        magnitude = np.ones(n) / np.sqrt(n)
        phase = 2 * np.pi * rng.random(n)
        A = magnitude * np.exp(1j * phase)
        a = irfft(A, n)
    elif mode == 1:
        # Low-pass shaped spectrum
        freqs = np.arange(n)
        magnitude = np.exp(-freqs ** 2 / (2 * (n/4) ** 2))
        magnitude = magnitude / magnitude.sum() * n
        phase = 2 * np.pi * rng.random(n)
        A = magnitude * np.exp(1j * phase)
        a = irfft(A, n)
    elif mode == 2:
        # "V" shaped sequence (high at edges)
        x = np.linspace(0, 1, n)
        a = (1 - 2*x + 1)**2  # = 1 everywhere, not useful
        a = np.abs(x - 0.5) ** 2  # V shape
    elif mode == 3:
        # Cosine taper
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 4:
        # Two-peaked
        x = np.linspace(0, 1, n)
        a = np.exp(-((x-0.25)/0.1)**2) + np.exp(-((x-0.75)/0.1)**2)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def lse_multi_stage(a0, time_limit, T_values=None, maxiter_per_stage=2000):
    """Multi-stage LSE-annealed L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    if T_values is None:
        T_values = [200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2]
    
    for T_val in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            return (2.0 * n / T_val) * (m + np.log(np.sum(np.exp(Tb - m))))
        
        def jac(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w = w / w.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        iters = max(20, min(maxiter_per_stage, int(remaining / 0.01)))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-11,
                                   'maxfun': iters * 400, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log-domain representation."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lbgfs_power_refine(a0, time_limit, p=16, maxiter=5000):
    """
    L-BFGS with power smoothing: minimize 2n * (sum b_k^p)^(1/p).
    Uses corrected gradient with factor of 2 from chain rule through convolution.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def func(z_in):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        b = np.clip(b, 0, None)
        S = np.sum(b ** p)
        if S < 1e-300:
            return 0.0
        return 2.0 * n * S ** (1.0 / p)

    def jac(z_in):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = np.clip(fast_convolve(a), 0, None)
        S = np.sum(b ** p)
        if S < 1e-300:
            return np.zeros_like(z_in)
        # df/db_k = 2n * S^(1/p - 1) * b_k^(p-1)
        df_db = 2.0 * n * (S ** (1.0 / p - 1.0)) * (b ** (p - 1))
        # d(f)/d(a_j) = sum_k df_db[k] * 2 * a_{k-j} (factor of 2 from convolution symmetry)
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if df_db[k] < 1e-30:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * df_db[k] * a[k - idx]
        # Chain rule through softmax
        S2 = np.dot(grad_a, a)
        return a * (grad_a - S2)

    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 500, 'disp': False}
        )
        z = res.x
    except Exception:
        pass

    return softmax_from_z(z)


def lse_multi_stage(a0, time_limit, T_values=None, maxiter_per_stage=2000):
    """
    Multi-stage LSE-annealed L-BFGS. Starts with low T (smooth) and increases
    to very high T (sharp, approximating exact max). Correct gradient with factor 2.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if T_values is None:
        T_values = [200, 100, 50, 20, 10, 5, 2, 1, 0.5]

    for T_val in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            return (2.0 * n / T_val) * (m + np.log(np.sum(np.exp(Tb - m))))

        def jac(z_in):
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w = w / w.sum()
            # d(f)/d(a_j) = sum_k 2n * w_k * 2 * a_{k-j}
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 4.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        iters = max(20, min(maxiter_per_stage, int(remaining / 0.05)))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-11,
                                   'maxfun': iters * 400, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)
