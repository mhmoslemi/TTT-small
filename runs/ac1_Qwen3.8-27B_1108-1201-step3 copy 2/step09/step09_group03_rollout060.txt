"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smart coordinate descent with exact optimal delta computation
2. Multi-peak mirror descent (multiplicative weight updates)
3. LP-based refinement (if cvxpy available)
4. Structured initializations
5. Multi-start with diverse initializations

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve_auto(a):
    """Linear auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_ab(a, b):
    """Linear convolution a*b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def score_from_conv(n, conv):
    """Compute score from convolution and n."""
    return 2.0 * n * conv.max()

def smart_coordinate_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact optimal delta computation.
    For each coordinate i, compute the delta that minimizes the new max convolution value.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve_auto(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improved = False
        
        for i in range(n):
            ai = a[i]
            if ai < 1e-16:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            # Affected region
            # conv[k] for k in [k_start, k_end] changes by 2*delta*a[k-i] (+ delta^2 at k=2i)
            a_shifted = a[k_start - i : k_end - i + 1]  # a[k-i] for k in [k_start, k_end]
            conv_aff = conv[k_start:k_end + 1]
            
            current_aff_max = float(conv_aff.max())
            
            if current_aff_max <= unaff_max + 1e-14:
                # Affected region is already below unaffected max, no improvement possible
                continue
            
            # Find optimal delta to bring affected max down to unaff_max
            # Need: conv_aff[k] + 2*delta*a_shifted[k] <= unaff_max for all k
            # => delta <= (unaff_max - conv_aff[k]) / (2*a_shifted[k]) for a_shifted[k] > 0
            valid = a_shifted > 1e-16
            if not valid.any():
                continue
            
            ub = (unaff_max - conv_aff[valid]) / (2.0 * a_shifted[valid])
            delta_opt = float(ub.min())
            
            # Constraint: a[i] + delta >= 0 => delta >= -a[i]
            delta_lower = -ai * 0.999  # stay slightly positive
            
            if delta_opt < delta_lower:
                delta = delta_lower
            else:
                delta = delta_opt
            
            # Handle the delta^2 correction at k=2i
            # After applying delta, check if k=2i exceeds unaff_max
            if 2 * i >= k_start and 2 * i <= k_end:
                # conv[2i] + 2*delta*ai + delta^2 should be <= unaff_max
                # delta^2 + 2*ai*delta <= unaff_max - conv[2i]
                # Solve: delta = -ai + sqrt(ai^2 - (conv[2i] - unaff_max))
                rhs = unaff_max - conv[2 * i]
                disc = ai * ai - rhs
                if disc > 0:
                    delta_q = -ai + np.sqrt(disc)
                    delta = min(delta, delta_q)
                else:
                    # Can't fix with this coordinate alone (would need negative a[i])
                    # Just use the linear solution
                    pass
            
            if abs(delta) < 1e-16:
                continue
            
            # Apply update
            new_ai = max(ai + delta, 0.0)
            actual_delta = new_ai - ai
            if abs(actual_delta) < 1e-16:
                continue
            
            # Update convolution
            changes = 2.0 * actual_delta * a_shifted
            if 2 * i >= k_start and 2 * i <= k_end:
                idx = 2 * i - k_start
                changes = changes.copy()
                changes[idx] = 2.0 * actual_delta * ai + actual_delta * actual_delta
            
            conv[k_start:k_end + 1] += changes
            a[i] = new_ai
            
            new_M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
            if new_M < M - 1e-14:
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                any_improved = True
        
        if not any_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweep} ({elapsed:.1f}s): score={best_score:.6f} M={M:.8f}")
    
    return best_a, best_score


def multi_peak_mirror_descent(a0, time_limit, lr_init=0.05, top_k=20, verbose=True):
    """
    Mirror descent with multi-peak subgradient.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve_auto(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break
        
        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act * 2 < (2 * n - 1):
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)
        
        pv = conv[peaks]
        temp = 0.01 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += wi * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        # Mirror descent step
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn
        
        a = a_new
        conv = fast_convolve_auto(a)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.1f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def fw_alternate(a0, b0, n_steps, verbose=False):
    """
    Frank-Wolfe alternating: optimize b given a, then a given b.
    Simple linearized FW without away steps (more robust).
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    b = normalize(b0.astype(np.float64))
    
    # Optimize b given a
    b_conv = fast_convolve_ab(a, b)
    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad))
        
        t = 2.0 / (step + 2.0)
        
        # Update: b_new = (1-t)*b + t*e_j
        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t
    
    # Optimize a given b
    a_conv = fast_convolve_ab(b, a)
    for step in range(n_steps):
        k_star = int(np.argmax(a_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = b[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad))
        
        t = 2.0 / (step + 2.0)
        
        a_conv = (1.0 - t) * a_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        a_conv[ks_s:ks_e + 1] += t * b
        a = (1.0 - t) * a
        a[j_star] += t
    
    return a, b


def lp_refine(a0, time_limit, verbose=True):
    """LP-based refinement using cvxpy."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0, 2.0 * len(a0) * fast_convolve_auto(normalize(a0)).max()
    
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    it = 0
    
    # Build convolution matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    # Find a solver
    solver = None
    for s in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            solver = s
            break
        except:
            continue
    if solver is None:
        solver = cp.SCS
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b
        
        score_b = 2.0 * n * fast_convolve_auto(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()
        
        # Damped update
        alpha = 2.0 / (it + 1.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve_auto(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Rebuild T
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a
        
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.1f}s): best={best_score:.6f}")
    
    return best_a, best_score


def structured_init(n, rng, style="uniform"):
    """Generate structured initializations."""
    if style == "uniform":
        return np.ones(n) / n
    elif style == "cosine":
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = (0.5 * (1 + np.cos(np.pi * x))) ** power
        return normalize(a)
    elif style == "two_hump":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.2)

    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    return score, a_norm

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact-delta coordinate descent (computes optimal step analytically)
2. Multi-peak mirror descent with softmax subgradients
3. Frank-Wolfe alternating minimization
4. LP-based best response refinement (if cvxpy available)
5. Multi-start with diverse structured initializations
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve_auto(a):
    """Linear auto-convolution a*a via FFT, returns full length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution a*b via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smart_coordinate_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact optimal delta computation.
    For coordinate i, the convolution changes as:
      conv[i:i+n] += 2*delta*a  (element-wise)
      conv[2i] += delta^2       (correction)
    We compute the largest delta (possibly negative) that keeps all affected
    values <= unaff_max, which minimizes the new global max.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve_auto(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    rng = np.random.default_rng(7 + sweep)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improved = False
        total_improvements = 0

        # Randomize order every 10 sweeps to avoid cycles
        if sweep % 10 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            ai = a[i]
            if ai < 1e-16:
                continue

            k_start = i
            k_end = i + n - 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Affected region values
            conv_aff = conv[k_start:k_end + 1]
            current_aff_max = float(conv_aff.max())

            if current_aff_max <= unaff_max + 1e-13:
                continue

            # Compute optimal delta
            # Linear constraints: conv_aff[j] + 2*delta*a[j] <= unaff_max
            valid = a > 1e-16
            if not valid.any():
                continue

            ub = (unaff_max - conv_aff[valid]) / (2.0 * a[valid])
            delta_opt = float(ub.min())

            # Quadratic constraint at k=2i (j=i in affected array):
            # conv[2i] + 2*delta*a[i] + delta^2 <= unaff_max
            rhs = unaff_max - conv[2 * i]
            disc = ai * ai - rhs
            if disc >= 0:
                delta_q = -ai + np.sqrt(disc)
                delta_opt = min(delta_opt, delta_q)

            # Lower bound: keep a[i] + delta >= 0
            delta_lower = -ai * 0.999

            if delta_opt < delta_lower:
                delta = delta_lower
            else:
                delta = delta_opt

            if abs(delta) < 1e-15:
                continue

            new_ai = ai + delta
            if new_ai < 0:
                new_ai = 0.0
            delta = new_ai - ai

            if abs(delta) < 1e-15:
                continue

            # Apply update to convolution
            conv[k_start:k_end + 1] += 2.0 * delta * a
            conv[2 * i] += delta * delta
            a[i] = new_ai

            # Update M
            new_M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
            if new_M < M - 1e-13:
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                any_improved = True
                total_improvements += 1

        if not any_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweep} ({elapsed:.1f}s): "
                  f"score={best_score:.6f} M={M:.8f} impr={total_improvements}")

    return best_a, best_score

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            conv_aff = conv[k_start:k_end + 1]
            target_max = max(unaff_max, float(conv_aff.max()))
            if target_max <= M + 1e-13:
                target_max = M

            best_delta = 0.0
            best_new_M = float(conv_aff.max())
            if best_new_M < unaff_max:
                best_new_M = unaff_max

            # Try multiple step sizes
            steps = [0.7, 0.5, 0.3, 0.15, -0.15, -0.3, -0.5, -0.7,
                     0.85, 0.6, 0.4, -0.4, -0.6, -0.85]
            for f in steps:
                delta = ai * f
                new_ai = ai + delta
                if new_ai < -1e-15:
                    continue

                # Compute new affected max
                new_aff = conv_aff + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * ai + delta * delta

                new_M_val = max(unaff_max, float(new_aff.max()))
                if new_M_val < best_new_M - 1e-14:
                    best_new_M = new_M_val
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end + 1] += 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                any_improved = True

        if not any_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD[{label}] sweep={sweep} ({elapsed:.1f}s): "
                  f"score={best_score:.6f} M={M:.8f}")

    return best_a, best_score


def multi_peak_mirror_descent(a0, time_limit, lr_init=0.04, top_k=25, verbose=True):
    """
    Mirror descent (multiplicative weight updates) with multi-peak subgradient.
    Uses softmax weighting over top-k peaks for robust gradient estimation.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    conv = fast_convolve_auto(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    avg_a = a.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act * 3 < (2 * n - 1):
            peaks = np.argpartition(conv, -k_act)[-k_act:]
        else:
            peaks = np.arange(2 * n - 1)

        pv = conv[peaks]
        temp = 0.008 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient: sg[i] = sum_k w_k * 2 * a[k-i]
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += wi * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Mirror descent update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / sn

        # Running average
        avg_a = (it * avg_a + a_new) / (it + 1)

        a = a_new
        conv = fast_convolve_auto(a)

        # Check running average periodically
        if it % 100 == 0:
            avg_score = 2.0 * n * fast_convolve_auto(avg_a).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_a.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.1f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def fw_alternate(a0, b0, n_steps, verbose=False):
    """
    Frank-Wolfe alternating minimization (simple linearized version).
    Optimizes b given a, then optimizes a given b.
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    b = normalize(b0.astype(np.float64))

    # Optimize b given fixed a
    b_conv = fast_convolve_ab(a, b)
    for step in range(n_steps):
        k_star = int(np.argmax(b_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = a[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad))
        t = 2.0 / (step + 2.0)

        b_conv = (1.0 - t) * b_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        b_conv[ks_s:ks_e + 1] += t * a
        b = (1.0 - t) * b
        b[j_star] += t

    # Optimize a given fixed b
    a_conv = fast_convolve_ab(b, a)
    for step in range(n_steps):
        k_star = int(np.argmax(a_conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        grad = b[k_star - i_end:k_star - i_start + 1][::-1]
        j_star = i_start + int(np.argmin(grad))
        t = 2.0 / (step + 2.0)

        a_conv = (1.0 - t) * a_conv
        ks_s = j_star
        ks_e = min(ks_s + n - 1, 2 * n - 2)
        a_conv[ks_s:ks_e + 1] += t * b
        a = (1.0 - t) * a
        a[j_star] += t

    if verbose:
        sa = 2.0 * n * fast_convolve_auto(a).max()
        sb = 2.0 * n * fast_convolve_auto(b).max()
        print(f"    FW done: a_score={sa:.6f}, b_score={sb:.6f}")

    return a, b


def lp_refine(a0, time_limit, verbose=True):
    """LP-based best response refinement using cvxpy."""
    try:
        import cvxpy as cp
    except ImportError:
        sc = 2.0 * len(a0) * fast_convolve_auto(normalize(a0)).max()
        return normalize(a0), sc

    n = len(a0)
    a = normalize(a0.astype(np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    it = 0

    # Find a working solver
    solver = None
    for s_name in ['SCS', 'ECOS', 'PDLP']:
        try:
            import cvxpy as cp
            x = cp.Variable(2, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s_name
                break
        except Exception:
            continue
    if solver is None:
        solver = 'SCS'

    # Build sparse convolution matrix
    import scipy.sparse as sp
    T = sp.lil_matrix((2 * n - 1, n))
    for j in range(n):
        for i in range(n):
            T[i, j] = a[j]  # T @ b gives conv(a, b)
    T = T.tocsr()

    import cvxpy as cp
    while time.monotonic() - t0 < time_limit:
        it += 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        score_b = 2.0 * n * fast_convolve_auto(b_val).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_val.copy()

        # Damped update
        alpha = 2.0 / (it + 1.0)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve_auto(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

        # Rebuild T
        T = sp.lil_matrix((2 * n - 1, n))
        for j in range(n):
            for i in range(n):
                T[i, j] = a[j]
        T = T.tocsr()

        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.1f}s): best={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng, style):
    """Generate structured initializations."""
    if style == "uniform":
        return np.ones(n) / n
    elif style == "cosine":
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 2.5)
        a = (0.5 * (1 + np.cos(np.pi * x))) ** power
        return normalize(a)
    elif style == "two_hump":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        return normalize(a)
    elif style == "periodic":
        period = int(rng.integers(2, 6))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase)
        return normalize(a)
    elif style == "notch":
        t = np.arange(n, dtype=np.float64)
        center = n / 2.0
        a = 1.0 - 0.4 * np.exp(-0.5 * ((t - center) / (n * 0.1)) ** 2)
        return normalize(a)
    elif style == "noise":
        a = np.abs(rng.standard_normal(n))
        return normalize(a)
    elif style == "geometric":
        r = rng.uniform(0.95, 0.999)
        a = r ** np.arange(n)
        return normalize(a)
    else:
        return np.ones(n) / n


def check_score(a, label=""):
    """Evaluate using evaluate_sequence and return (score, normalized_a)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf'), a
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    return score, a_norm
