from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute 2n * max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

    # Clip to avoid overflow (shouldn't be needed after subtraction, but safety)
    exponents = np.minimum(exponents, 500)
    lse = c_max + np.log(np.sum(np.exp(exponents))) / temp
    return 2.0 * len(a) * lse


def lse_gradient(a, temp):
    """Gradient of 2n * LSE(conv(a,a), temp) w.r.t. a."""
    c = fast_convolve(a)
    c_max = c.max()
    exponents = temp * (c - c_max)
    exponents = np.minimum(exponents, 500)
    exp_vals = np.exp(exponents)
    weights = exp_vals / exp_vals.sum()  # softmax(temp * (c - c_max))
    n = len(a)
    # Gradient of LSE w.r.t. a_i = sum_k weights_k * dc_k/da_i = sum_k weights_k * 2*a_{k-i}
    # where the sum is over valid k
    # This is 2 * correlate(weights, a, 'valid')
    corr = np.correlate(weights, a, mode='valid')
    g_a = 4.0 * n * corr  # 2n * 2 * corr
    return g_a


def lse_lbfgs(a0, time_limit, temp_start=1.0, temp_end=200.0, n_stages=10, iters_per_stage=2000):
    """L-BFGS with log-sum-exp smoothing and increasing temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
  
    # Use parameterization: a = exp(z) / sum(exp(z))
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_score = score(a)
    best_a = a.copy()
    
    temps = np.geomspace(temp_start, temp_end, n_stages)
    
    for stage_temp in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        def func(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a_cur = exp_z / exp_z.sum()
            c = fast_convolve(a_cur)
            c_max = c.max()
            exponents = stage_temp * (c - c_max)
            exponents = np.minimum(exponents, 500)
            lse_val = c_max + np.log(np.sum(np.exp(exponents))) / stage_temp
            return 2.0 * n * lse_val
        
        def jac(z_val):
            z_max = z_val.max()
            exp_z = np.exp(z_val - z_max)
            a_cur = exp_z / exp_z.sum()
            c = fast_convolve(a_cur)
            c_max = c.max()
            exponents = stage_temp * (c - c_max)
            exponents = np.minimum(exponents, 500)
            exp_vals = np.exp(exponents)
            weights = exp_vals / exp_vals.sum()
            corr = np.correlate(weights, a_cur, mode='valid')
            g_a = 4.0 * n * corr
            # Gradient w.r.t. z: a * (g_a - dot(g_a, a))
            return a_cur * (g_a - np.dot(g_a, a_cur))
        
        max_iter = min(iters_per_stage, max(50, int(remaining * 5)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-LBFGS temp={stage_temp:.1f} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def pnorm_loss_z(z, p):
    """p-norm objective in log domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_grad_z(z, p):
    """p-norm gradient in log domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_p = c_pos ** p
    S = np.sum(c_p)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_lbfgs(a0, time_limit, p_start=50.0, p_end=5000.0, n_stages=10):
    """L-BFGS with p-norm smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_score = score(a)
    best_a = a.copy()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        max_iter = max(100, int(remaining * 0.8))
        try:
            res = minimize(
                fun=znorm_loss_z, x0=z, jac=pnorm_grad_z,
                method='L-BFGS-B',
                args=(stage_p,),
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    PNORM-LBFGS p={stage_p:.0f} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, time_limit, n_steps=50000):
    """Aggressive peak shaving via mass redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, score(a)
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        # Find the current max
        max_idx = c.argmax()
        max_val = c[max_idx]
        
        # Find contributing indices for this peak
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        total_contrib = contribs.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Find the position with highest contribution
        top_contrib_pos = idx[np.argmax(contribs)]
        
        # Also find the position with lowest "importance" (least contribution to any peak)
        # Compute total contribution to all peaks for each position
        importance = np.zeros(n)
        for k in [max_idx]:  # Only the main peak for efficiency
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            idx2 = np.arange(js2, je2 + 1)
            importance[idx2] += a[idx2] * a[k - idx2]
        
        # Try moving mass from high-importance to low-importance positions
        # Sort by importance
        sorted_idx = np.argsort(importance)
        low_pos = sorted_idx[:5]  # Least important positions
        high_pos = sorted_idx[-5:]  # Most important positions
        
        improved = False
        for hi in high_pos:
            if a[hi] < 1e-10:
                continue
            for lo in low_pos:
                if lo == hi or a[lo] > a[hi]:
                    continue
                # Try moving a small fraction of mass from hi to lo
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    delta = a[hi] * frac
                    a_trial = a.copy()
                    a_trial[hi] -= delta
                    a_trial[lo] += delta
                    if a_trial[hi] < 0:
                        continue
                    sc = score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        # Also try uniform redistribution at the peak
        if not improved:
            # Reduce the two most contributing elements at the peak
            top2_local = np.argsort(contribs)[::-1][:2]
            for ci in top2_local:
                pos = idx[ci]
                for factor in [0.85, 0.7, 0.5]:
                    delta = a[pos] * (1 - factor)
                    a_trial = a.copy()
                    a_trial[pos] -= delta
                    if a_trial[pos] < 0:
                        continue
                    # Redistribute to positions furthest from the peak
                    # (positions that don't contribute to this peak)
                    mask = np.ones(n, dtype=bool)
                    mask[js:je+1] = False
                    # Also exclude positions that contribute to nearby peaks
                    if (n - 1 - js) > 0 and (je + 1) < n:
                        pass
                    if mask.sum() > 0:
                        # Give more weight to positions far from peak
                        dist = np.abs(np.arange(n) - pos)
                        weights = np.zeros(n)
                        weights[mask] = dist[mask] ** 2
                        w_sum = weights.sum()
                        if w_sum > 0:
                            a_trial += delta * weights / w_sum
                    else:
                        # Redistribute uniformly to others
                        mask2 = np.ones(n, dtype=bool)
                        mask2[pos] = False
                        a_trial[mask2] += delta * a_trial[mask2] / a_trial[mask2].sum()
                    
                    sc = score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Perturb
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                stall = 0
                    
            if stall > 200:
                break
        else:
            stall = 0
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShaving step={step} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def spectral_smooth(a0, time_limit):
    """Try different levels of spectral smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, score(a)
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (n - 1).bit_length()
    
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        freqs = np.arange(fft_size // 2 + 1)
        # Low-pass: attenuate high frequencies
        cutoff = int(alpha * fft_size // 2)
        for f in range(1, fft_size // 2 + 1):
            if f > cutoff:
                A[f] *= 0.5
            elif f > cutoff // 2:
                A[f] *= 0.8
        
        a_trial = np.fft.irfft(A, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st
        sc = score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_alternating(a0, time_limit, n_max_iters=20000):
    """Nesterov accelerated alternating minimization.
    At each step: given a, find b that minimizes max conv(a, b) subject to sum(b)=1, b>=0.
    This is an LP. Use a simple approximation: b = a * exp(-lambda * conv(a,a)) normalized.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov momentum variable
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute "best response" for y: find b that minimizes max_k conv(y, b)
        # Approximation: b_i proportional to exp(-lambda * (conv(y, y))_i)
        # where (conv(y,y))_i captures how much adding to position i would increase peaks
        c_yy = fast_convolve(y)
        # The "cost" of putting mass at position i is related to the peaks it would create
        # Approximate: cost_i = sum_k conv(y,y)_k * |dy_k/dy_i| = sum_k conv(y,y)_k * 2*y_{k-i}
        # = 2 * correlate(c_yy, y, 'valid')[i]
        cost = np.correlate(c_yy, y, mode='valid')
        cost = np.maximum(cost, 1e-15)
        
        # Exponential tilting: b_i = y_i * cost_i^(-temperature) / sum
        temp = 1.0 + 0.5 * it / n_max_iters  # gradually increase temperature
        b_raw = y * (cost ** (-temp))
        b_raw = np.maximum(b_raw, 0)
        sb = b_raw.sum()
        if sb < 1e-15:
            b_new = y.copy()
        else:
            b_new = b_raw / sb
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_new.copy()
        else:
            y = y / sy
        
        # Actual iterate
        a = y  # Simplified: just use y directly
        
        sc = score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovAlt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 3:
        # Flat-top window
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        transition = 0.15
        for i in range(n):
            if x[i] < transition:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * x[i] / transition)
            elif x[i] > 1 - transition:
                a[i] = 0.5 + 0.5 * np.cos(np.pi * (x[i] - (1 - transition)) / transition)
    elif variant == 4:
        # Kaiser-like
        x = np.linspace(-1, 1, n)
        beta = 5.0
        from scipy.special import i0
        a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def lse_objective_z(z, temp):
    """Log-sum-exp smoothed objective in log domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exponents = temp * (c - c_max)
    exponents = np.clip(exponents, -500, 500)
    lse_val = c_max + np.log(np.sum(np.exp(exponents))) / temp
    return 2.0 * n * lse_val


def lse_gradient_z(z, temp):
    """Gradient of LSE-smoothed objective w.r.t. log-domain parameters."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exponents = temp * (c - c_max)
    exponents = np.clip(exponents, -500, 500)
    exp_vals = np.exp(exponents)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros_like(z)
    weights = exp_vals / total
    corr = np.correlate(weights, a, mode='valid')
    g_a = 4.0 * n * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_objective_z(z, p):
    """p-norm objective in log domain."""
    z_max = z.max()

    max_log = np.max(p * log_c)
    log_sum = max_log + np.log(np.sum(np.exp(p * log_c - max_log)))
    log_val = log_sum / p
    return 2.0 * n * np.exp(log_val)

    """p-norm gradient in log domain with log-space computation."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    log_c = np.log(c_pos)
    p_log_c = p * log_c
    max_plc = np.max(p_log_c)
    exp_shifted = np.exp(p_log_c - max_plc)
    S_shifted = np.sum(exp_shifted)
    if S_shifted < 1e-300:
        return np.zeros_like(z)

                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
            if stall > 1000:
                break
        else:
            stall = 0
        
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      MassRedist step={step} ({elapsed:.0f}s): {best_score:.6f}")
    return best_a, best_score


def nesterov_tilt(a0, time_limit, n_max_iters=50000):
    """Nesterov accelerated method with exponential tilting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute cost: how much each position contributes to peaks
        c_yy = fast_convolve(y)
        cost = np.correlate(c_yy, y, mode='valid')
        cost = np.maximum(cost, 1e-20)
        
        # Exponential tilting: reduce mass at high-cost positions
        temp = 0.5 + 0.5 * (it / max(1, n_max_iters))
        c_max = cost.max()
        if c_max > 0:
            tilt_exp = np.exp(-temp * (cost - cost.min()) / c_max)
        else:
            tilt_exp = np.ones(n)
        b_raw = y * tilt_exp
        b_raw = np.maximum(b_raw, 0)
        sb = b_raw.sum()
        if sb < 1e-15:
            b_new = y.copy()
        else:
            b_new = b_raw / sb
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y = b_new.copy()
        else:
            y = y_new / sy
        
        a = y
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NestTilt it={it+1} ({elapsed:.0f}s): {best_score:.6f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Cubic-ish interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3.0)
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif variant == 3:
        # One-sided: increasing
        x = np.linspace(0, 1, n)
        a = x ** 2
    elif variant == 4:
        # Bimodal
        x = np.linspace(-1, 1, n)
        a = np.exp(-(x - 0.4) ** 2 * 4) + np.exp(-(x + 0.4) ** 2 * 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
