        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Stall detection and recovery
        if it % stall_threshold == 0 and it > 0:
            if best_score < prev_stall_score - 1e-8:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1

        if stall_count >= 3:
            rng = np.random.default_rng(it * 31 + 17)
            noise_scale = 0.02 + 0.015 * min(stall_count, 5)
            noise = rng.standard_normal(n) * noise_scale
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = y.copy()
            stall_count = 0
            prev_stall_score = best_score

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shaving_local(a0, time_limit, max_steps=50000):
    """Aggressive peak-shaving: reduce contributions at the autocorrelation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autoconv(a)
        # Find the peak (excluding endpoints which are a[0]^2 and a[n-1]^2)
        c_inner = c.copy()
        if len(c_inner) > 2:
            c_inner[0] = -np.inf
            c_inner[-1] = -np.inf
        peak_idx = int(np.argmax(c_inner))
        peak_val = c[peak_idx]

        # Find contributing pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Try to reduce top contributors
        top_k = min(20, len(idx))
        top_indices = idx[np.argsort(contribs)[::-1][:top_k]]
        improved = False

        for i in top_indices:
            j = peak_idx - i
            if i == j or j >= n or j < 0:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue

            # Try reducing the pair
            for f in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 0.01:
                    continue
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Single coordinate reduction
        if not improved:
            for i in top_indices[:10]:
                if a[i] < 1e-14:
                    continue
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial = a_trial / st
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Random small perturbation if stuck
        if not improved:
            rng = np.random.default_rng(step * 7 + 3)
            n_pert = rng.integers(2, 6)
            pidx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in pidx:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 100:
            # Larger perturbation
            rng = np.random.default_rng(step * 13 + 5)
            frac = min(0.15, 0.02 * np.sqrt(step / 100.0))
            mask = rng.random(n) < frac
            a_trial = a.copy()
            a_trial[mask] *= (1.0 + rng.uniform(-0.3, 0.3, size=mask.sum()))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                else:
                    break  # Give up

    return best_a, best_score


def structured_init(n, variant):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        xm = np.linspace(-1, 1, n)
        a = np.exp(-xm ** 2 * 3)
    elif variant == 4:
        a = 1.0 / (1 + 10 * (x - 0.5) ** 2)
    elif variant == 5:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LSE-LBFGS + Enhanced Nesterov LP + peak shaving,
    with multi-length exploration and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, 3)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    def try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Quick LSE-LBFGS on parent ===
    print(f"\n=== Phase 1: LSE-LBFGS n={n_best} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.15, 150.0)
    if t1 > 30:
        t_vals = [0.05, 0.02, 0.008, 0.003, 0.001]
        a1 = lse_lbfgs_refine(best_state['seq'], t1, t_vals)
        try_update(a1, "lse_p1")

    # === Phase 2: Enhanced Nesterov LP (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 500.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a2, _ = nesterov_lp_enhanced(best_state['seq'], t2, solver, max_iters=500000)
        try_update(a2, "nesterov_p2")

    # === Phase 3: LSE polish (high temperature -> low) ===
    print(f"\n=== Phase 3: LSE polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        t_vals = [0.01, 0.004, 0.001, 0.0003]
        a3 = lse_lbfgs_refine(best_state['seq'], t3, t_vals)
        try_update(a3, "lse_p3")

    # === Phase 4: Second Nesterov round ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 20:
        a4, _ = nesterov_lp_enhanced(best_state['seq'], t4, solver, max_iters=300000)
        try_update(a4, "nesterov_p4")

    # === Phase 5: Peak shaving local refinement ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 10:
        a5, _ = peak_shaving_local(best_state['seq'], t5, max_steps=100000)
        try_update(a5, "peak_shave_p5")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-128, -96, -64, -32, 32, 64, 96, 128, 192]:
        alt_n = base + delta
        if 640 <= alt_n <= 1600:
            candidate_lengths.add(alt_n)
    for rn in [768, 832, 896, 1024, 1088, 1152, 1280]:
        if 640 <= rn <= 1600:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.15, 120.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_seq(best_state['seq'], alt_n)

        # LSE quick
        t_lse = t_total * 0.4
        a_alt = lse_lbfgs_refine(init_alt, t_lse, [0.03, 0.01, 0.003])
        improved = try_update(a_alt, f"lse_n{alt_n}")

        if improved or t_total > 60:
            rem = deadline - time.monotonic()
            if rem > 20:
                t_lp = min(rem * 0.3, 80.0)
                a_alt2, _ = nesterov_lp_enhanced(
                    best_state['seq'] if improved else a_alt,
                    t_lp, solver, max_iters=100000)
                try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Final Nesterov + polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 200.0)
        a_fin, _ = nesterov_lp_enhanced(best_state['seq'], t_fin, solver, max_iters=200000)
        try_update(a_fin, "final_nest")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.3, 100.0)
        a_ps, _ = peak_shaving_local(best_state['seq'], t_ps, max_steps=50000)
        try_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lse_final = min(rem * 0.4, 60.0)
        a_lse_final = lse_lbfgs_refine(best_state['seq'], t_lse_final, [0.002, 0.0005])
        try_update(a_lse_final, "final_lse")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
