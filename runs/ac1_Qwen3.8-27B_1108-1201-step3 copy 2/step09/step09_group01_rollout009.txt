def fast_convolve(a):
    """Linear auto-convolution via FFT."""

    size = 2 * n - 1

    A = rfft(a, fft_size)

def convolve_ab(a, b):

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def init_frequency_flat(n, rng, n_iter=50):

        # Scale to have flat magnitude
        A_new = mag_target * A / (mag + 1e-30)
        # Time domain: project back

        a_new = a_new / s
        # Check if converged
        if np.allclose(a_new, a, atol=1e-8):

        a = a_new

def init_parent_interpolate(parent, n):

    old_n = len(parent)

    s = parent.sum()

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

        # dC[k]/d(a[i]) = 2*a[k-i] (with the convention that a is the full sequence)

        # = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i < n)
        # Wait, that's only if both terms are valid. Let me be more careful.
        # C[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # dC[k]/d(a[i]) = a[k-i] (from the first term, j=i) + a[k-i] (from the second term, k-j=i, i.e., j=k-i)
        # But these are the same term if i = k-i, i.e., k = 2i. In that case, dC[k]/d(a[i]) = 2*a[i] (derivative of a[i]^2)
        # Actually no: if k = 2i, then the term a[i]*a[i] appears once in the sum, and its derivative is 2*a[i].
        # If k != 2i, then a[k-i] appears as both a[j] (j=i) and a[k-j] (j=k-i=k-i), giving 2*a[k-i].
        # Wait, I think the correct formula is:
        # dC[k]/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n and k-i != i
        # dC[k]/d(a[i]) = 2*a[i] if k-i == i (i.e., k = 2i)
        # Which is the same: 2*a[k-i] = 2*a[i] when k = 2i.
        # So in all cases: dC[k]/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0.
        
        # But wait, this isn't quite right for the auto-convolution. Let me verify:
        # C[k] = sum_j a[j] * a[k-j]
        # dC[k]/d(a[i]) = sum_j [d(a[j])/d(a[i]) * a[k-j] + a[j] * d(a[k-j])/d(a[i])]
        # = sum_j [delta(j-i) * a[k-j] + a[j] * delta(k-j-i)]
        # = a[k-i] + a[k-i]  (if 0 <= k-i < n)
        # = 2 * a[k-i]
        # Yes, that's correct.
        
        grad = np.zeros(n)
        # For each k, add 2*weights[k]*a[k-i] to grad[i]
        # Equivalent to: grad = 2 * convolve(weights, y) (truncated to length n)
        # convolve(weights, y)[i] = sum_k weights[k] * y[k-i] (with appropriate bounds)
        # Hmm, that's the cross-correlation, not convolution.
        # Let me think: grad[i] = 2 * sum_k weights[k] * y[k-i] for 0 <= k-i < n
        # = 2 * sum_k weights[k] * y[k-i]
        # This is the cross-correlation of weights and y, evaluated at lag i.
        # cross_corr[i] = sum_k weights[k] * y[k-i] = (weights * y_flipped)[i] in convolution terms
        # Actually: convolve(weights, y_flipped)[i] where y_flipped = y[::-1]
        # convolve(w, y[::-1])[i] = sum_k w[k] * y_flipped[i-k] = sum_k w[k] * y[n-1-(i-k)]

        # This is a cross-correlation: grad[i] = 2 * sum_j weights[i+j] * y[j]
        # = 2 * (weights ⊛ y)[i] where ⊛ denotes cross-correlation
        # In terms of convolution: (w ⊛ y)[i] = (w * flip(y))[i] = convolve(w, y_flipped)[i + n - 1]
        # Hmm, let me just use the direct formula.
        
        # Actually, the simplest way: grad = 2 * np.correlate(weights, y, mode='valid') 
        # No, that's not right either. Let me just compute it as:

        # then the cross-correlation can be computed via FFT.
        # But it's simpler to note that grad = 2 * (T^T @ weights) where T is the convolution matrix.
        # (T @ y)[k] = sum_j T[k,j] * y[j] = sum_j (y if j in valid range for k) * ... 
        # Actually (T^T @ w)[i] = sum_k T[k,i] * w[k] = sum_k (y[k-i] if valid) * w[k]
        # = sum_k y[k-i] * w[k] for valid k
        # = sum_j y[j] * w[i+j] (substituting j = k-i)
        # Yes! So grad = 2 * T^T @ weights.
        
        # Compute T^T @ weights efficiently:
        # (T^T @ w)[i] = sum_{k=i}^{i+n-1} w[k] * y[k-i]  (with bounds checking)
        # = sum_{j=0}^{n-1} w[i+j] * y[j]  (with w having length 2n-1, so i+j < 2n-1)
        # This is the cross-correlation of w and y at lag -i (or equivalently, the 
        # "anti-convolution")
        
        # Simplest: use np.correlate
        # np.correlate(w, y, mode='full')[i + n - 1] = sum_j w[i+j] * y[j]
        # So grad[i] = 2 * np.correlate(weights, y, mode='full')[i + n - 1]

        grad = 2.0 * full_corr[n - 1:2 * n - 1]

        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = simplex_project(a_new)

    return best_a, best_score

def intensive_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Intensive coordinate descent with line search.
    For each coordinate, find the optimal value via ternary search on the 1D subproblem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            # Determine affected range in convolution
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # The affected part: conv[k_start:k_end+1] changes when a[i] changes
            # conv[k_start + j] = ... + a[i] * a[j] + ... (linear in a[i] except at k=2i)
            # Actually: d conv[k] / d a[i] = 2 * a[k-i] (as derived before)
            # But conv[k] itself = old_conv[k] + 2*delta*a[k-i] + delta^2*(1 if k==2i else 0)
            
            # We want to find delta that minimizes max(unaff_max, max of new affected)
            # The new affected[k] = conv[k] + 2*delta*a[k-i] + delta^2*(1 if k==2i)
            # = conv[k] + 2*delta*a[k-i] + delta^2*(delta_{k,2i})
            
            # For the k != 2i terms, it's linear in delta.
            # For k = 2i (if in range), it's quadratic.
            
            # Strategy: try several candidate values of a[i] and pick the best.
            # Current value: a[i]. Try scaling by various factors.
            
            best_delta = 0.0
            best_new_max = M

            # Try logarithmically spaced factors
            for factor in [0.0, 0.1, 0.25, 0.4, 0.5, 0.65, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25, 1.5]:
                new_val = a[i] * factor
                delta = new_val - a[i]
                if new_val < 0:
                    continue
                
                # Compute new affected
                # For k in [k_start, k_end]: new_conv[k] = conv[k] + 2*delta*a[k-i] + delta^2 * (1 if k==2i)
                k_range = np.arange(k_start, k_end + 1)
                idx = k_range - i  # These are the indices j where a[j] pairs with a[i]
                idx_valid = (idx >= 0) & (idx < n)
                
                new_affected = conv[k_start:k_end + 1].copy()
                new_affected[idx_valid] += 2.0 * delta * a[idx[idx_valid]]
                
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos_2i = 2 * i - k_start
                    new_affected[pos_2i] += delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()
                
                # Update convolution
                k_range = np.arange(k_start, k_end + 1)
                idx = k_range - i
                idx_valid = (idx >= 0) & (idx < n)
                
                changes = np.zeros(k_end - k_start + 1)
                changes[idx_valid] = 2.0 * best_delta * a[idx[idx_valid]]
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos_2i = 2 * i - k_start
                    changes[pos_2i] += best_delta * best_delta
                
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ICD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score
