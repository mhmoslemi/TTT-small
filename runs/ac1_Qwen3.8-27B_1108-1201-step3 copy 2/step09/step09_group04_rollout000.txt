def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def subgradient_at(a, peak_indices):
    """Compute subgradient of max_k (a*a)[k] at given peak indices.
    grad[i] = 2 * mean(a[k-i] for k in peak_indices, valid indices)"""
    n = len(a)
    g = np.zeros(n)
    count = 0
    for k in peak_indices:
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # a[k-i] for i in [i_start, i_end] is a[k-i_start], a[k-i_start-1], ...
            # a[k - i] = a[k - i_start], a[k - (i_start+1)], ...
            # This reverses a[k-i_end], a[k-(i_end-1)], ..., a[k-i_start]
            segment = a[k - i_end : k - i_start + 1]
            g[i_start:i_end + 1] += segment
            count += 1
    if count > 0:
        g *= 2.0 / count
    return g


def subgradient_descent(a0, time_limit, n_iterations=None):
    """Projected subgradient descent with Nesterov acceleration on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()

    if n_iterations is None:
        n_iterations = 200000

    # Lower bound for Polyak step: use 1/(2n-1) as optimistic bound
    lower_bound = 1.0 / (2 * n - 1)
    gap = best_score / (2.0 * n) - lower_bound  # gap in max_b

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        peak_val = b.max()
        score = 2.0 * n * peak_val

        # Active set: peaks within 1% of maximum
        threshold = peak_val * 0.995
        active_mask = b >= threshold
        peak_indices = np.where(active_mask)[0]

        # Limit active set to top 20 for stability
        if len(peak_indices) > 20:
            top_vals = b[peak_indices]
            order = np.argsort(top_vals)[::-1][:20]
            peak_indices = peak_indices[order]

        g = subgradient_at(y, peak_indices)

        # Polyak step size
        current_gap = peak_val - lower_bound
        g_norm_sq = np.dot(g - y, g - y)
        if g_norm_sq < 1e-30:
            break
        eta = min(current_gap / g_norm_sq, 1.0)  # Cap step size
        # Also use diminishing step as backup
        eta_dim = 1.0 / (np.sqrt(it + 1) * 0.1)
        eta = min(eta, eta_dim)

        # Gradient step
        y_step = y - eta * g
        y_step = np.maximum(y_step, 0.0)
        sy = y_step.sum()
        if sy < 1e-15:
            y_step = y.copy()
            sy = 1.0
        y_step = y_step / sy

        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        a_new = (1.0 - alpha) * a + alpha * y_step
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_step.copy()
        else:
            a_new = a_new / sa

        a = a_new
        y = y_step

        # Track best
        new_score = 2.0 * n * fast_convolve(a).max()
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def p_norm_objective_value(a, p):
    """Compute (sum c_k^p)^(1/p) for c = convolve(a, a)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_smoothing_z(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=15.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(50, max(500, int(6000 / n_stages)))

    for stage_i, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Scale iterations based on remaining time and stages left
        stages_left = n_stages - stage_i
        adaptive_iters = max(30, iters_per_stage * 2 // max(1, stages_left))

        def func(z_val):
            return p_norm_smoothing_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': adaptive_iters, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': adaptive_iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"        P-norm L-BFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.8f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def active_set_peak_shaving(a0, time_limit, n_steps=50000):
    """Aggressive active-set peak shaving with secondary redistribution.
    Identifies top peaks and reduces contributing coordinates,
    then redistributes mass to minimize new peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()

        # Active set: top peaks
        threshold = peak_val * 0.998
        active_peaks = np.where(b >= threshold)[0]
        if len(active_peaks) > 15:
            active_peaks = active_peaks[np.argsort(b[active_peaks])[::-1][:15]]

        if len(active_peaks) == 0:
            break

        # Compute contribution scores for each position
        contrib = np.zeros(n)
        for k in active_peaks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                contrib[i_start:i_end+1] += a[i_start:i_end+1] * a[k-i_end:k-i_start+1][::-1]

        # Strategy 1: Reduce top contributors
        improved = False
        top_coords = np.argsort(contrib)[::-1]

        for idx_i in top_coords[:30]:
            if a[idx_i] < 1e-15:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.83, 0.78]:
                diff = a[idx_i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[idx_i] = max(0, a[idx_i] - diff)
                mask = np.ones(n, dtype=bool)
                mask[idx_i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = a_trial[mask] + diff * others / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Pair adjustment at strongest peak
        if not improved:
            k_peak = int(active_peaks[0])
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx_range = np.arange(i_start, i_end + 1)
            if len(idx_range) == 0:
                continue
            contribs = a[idx_range] * a[k_peak - idx_range]
            top_pairs = np.argsort(contribs)[::-1][:10]
            for pos in top_pairs:
                i = int(idx_range[pos])
                j = k_peak - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94, 0.90]:
                    for f2 in [0.97, 0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial /= st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Multi-coordinate simultaneous reduction
        if not improved and step % 50 == 0:
            # Reduce top 3 contributors simultaneously
            top3 = top_coords[:3]
            top3 = [t for t in top3 if a[t] > 1e-10]
            if len(top3) >= 2:
                a_trial = a.copy()
                total_reduced = 0.0
                for t in top3:
                    red = a_trial[t] * 0.05
                    a_trial[t] -= red
                    total_reduced += red
                mask = np.ones(n, dtype=bool)
                for t in top3:
                    mask[t] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum > 1e-15:
                    a_trial[mask] += total_reduced * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True

        # Strategy 4: Small random perturbation for escape
        if not improved and step % 200 == 0:
            rng = np.random.default_rng(step * 31 + 7)
            n_pert = rng.integers(2, 8)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial /= st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved and step > 100:
            # Try slightly larger perturbations after stall
            rng = np.random.default_rng(step * 17 + 3)
            scale = 0.02 + 0.01 * min((step // 200), 5)
            a_trial = a + scale * rng.standard_normal(n)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial /= st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Check if we've stalled for too long
            if step > 500:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"        PeakShave step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_best_response(a_fixed):
    """Given fixed a, find b = argmin max_k conv(a,b)[k] subject to sum(b)=1, b>=0.
    Uses simple iterative projection (avoiding cvxpy for speed)."""
    n = len(a_fixed)
    b = np.ones(n) / n
    for _ in range(200):
        ab = np.convolve(a_fixed, b, mode='full')
        peak = ab.max()
        if peak < 1e-10:
            break
        # Reduce b at positions contributing most to peak
        k_peak = int(np.argmax(ab))
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        for i in range(i_start, i_end + 1):
            contrib = a_fixed[i] * b[k_peak - i]
            if contrib > 0.1 * peak:
                b[i] *= 0.95
        b = np.maximum(b, 0)
        b /= b.sum()
    return b


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3.0)
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif variant == 3:
        # Skewed
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.8
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_local(a):
    """Fast local evaluation (assumes a is already valid: non-negative, finite)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2 * len(a) * b.max() / (s * s))


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: subgradient + p-norm L-BFGS + peak shaving +
    length exploration + evolutionary restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=1)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.8f}")

    n_best = len(best_state['seq'])

    def try_update(a, label):
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Subgradient descent (fast, good for fine-tuning) ===
    print(f"\n=== Phase 1: Subgradient Descent n={n_best} (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_descent(best_state['seq'], t_alloc, n_iterations=500000)
        try_update(a_sg, "subgradient_1")

    # === Phase 2: P-norm L-BFGS (main workhorse) ===
    print(f"\n=== Phase 2: P-norm L-BFGS (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 20 -> 600")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=20.0, p_end=600.0, n_stages=10)
        try_update(a_pn, "pnorm_1")

    # === Phase 3: Peak shaving (local refinement) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        a_ps, _ = active_set_peak_shaving(best_state['seq'], t_alloc, n_steps=80000)
        try_update(a_ps, "peakshave_1")

    # === Phase 4: Subgradient round 2 ===
    print(f"\n=== Phase 4: Subgradient r2 (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        a_sg2, _ = subgradient_descent(best_state['seq'], t_alloc, n_iterations=400000)
        try_update(a_sg2, "subgradient_2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_state['score']:.8f}) ===")
    base = n_best
    candidate_lengths = []
    for delta in [-48, -32, -24, -16, -8, 8, 16, 24, 32, 48, 64, 96]:
        alt_n = base + delta
        if 512 <= alt_n <= 2560:
            candidate_lengths.append(alt_n)
    # Round numbers
    for rn in [896, 928, 960, 992, 1024, 1056, 1088, 1120, 1152]:
        if 512 <= rn <= 2560 and rn != base:
            if rn not in candidate_lengths:
                candidate_lengths.append(rn)

    # Sort by proximity to current best
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - base))

    n_explored = 0
    for alt_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 150.0)
        if t_total < 50:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # Quick optimization: subgradient + peak shaving
        t_sg = t_total * 0.4
        a_alt, _ = subgradient_descent(init_alt, t_sg, n_iterations=100000)

        if rem > 20:
            t_ps = min((deadline - time.monotonic()) * 0.3, t_total * 0.4)
            if t_ps > 15:
                a_alt, _ = active_set_peak_shaving(a_alt, t_ps, n_steps=30000)

        try_update(a_alt, f"len_{alt_n}")

    # === Phase 6: P-norm L-BFGS round 2 (higher p) ===
    print(f"\n=== Phase 6: P-norm L-BFGS r2 (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 200 -> 2000")
        a_pn2 = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=200.0, p_end=2000.0, n_stages=5)
        try_update(a_pn2, "pnorm_2")

    # === Phase 7: Evolutionary restarts ===
    print(f"\n=== Phase 7: Evolutionary Restarts (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_state['seq'])
        for trial in range(4):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            noise_scales = [0.02, 0.05, 0.10, 0.15]
            scale = noise_scales[trial % len(noise_scales)]
            perturbed = best_state['seq'] + scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed /= perturbed.sum()

            t_sub = min(rem * 0.25, 60.0)
            if t_sub < 15:
                break
            # Quick subgradient + peak shave on perturbed
            a_pert, _ = subgradient_descent(perturbed, t_sub * 0.5, n_iterations=50000)
            if rem > 15:
                t_ps2 = min((deadline - time.monotonic()) * 0.3, t_sub * 0.4)
                if t_ps2 > 10:
                    a_pert, _ = active_set_peak_shaving(a_pert, t_ps2, n_steps=20000)
            try_update(a_pert, f"restart_{trial}_{scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        a_fin, _ = subgradient_descent(best_state['seq'], t_fin, n_iterations=200000)
        try_update(a_fin, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps_fin = min(rem * 0.5, 40.0)
        a_fin, _ = active_set_peak_shaving(best_state['seq'], t_ps_fin, n_steps=30000)
        try_update(a_fin, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_pn_fin = min(rem * 0.8, 30.0)
        a_pnf = p_norm_lbfgs_refine(best_state['seq'], t_pn_fin, p_start=500.0, p_end=3000.0, n_stages=3)
        try_update(a_pnf, "final_pnorm")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 20.0)
    print(f"\nFINAL SCORE: {best_state['score']:.8f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")

    # Ensure we return a valid list
    result = [float(max(0.0, min(1000.0, x))) for x in best_state['seq']]
    if not result or np.sum(result) < 0.01:
        # Fallback
        return [1.0 / 960] * 960
    return result
