def compute_gradient_smooth(x, beta):

    n = len(x)

    # This is the cross-correlation of weights and x

3. Fast coordinate descent with vectorized 1D line search
4. LP best-response with Nesterov averaging for global exploration
5. Multi-scale and multi-length exploration
6. Structured perturbation based on convolution error pattern
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time

# Global cache for LP solver
_lp_solver = None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def smooth_objective_and_grad(x, beta):
    """
    Compute smoothed objective and gradient.
    f(x) = (1/beta) * log(sum_k exp(beta * c_k)) where c_k = (x*x)_k
    grad_i = 2 * sum_j w_{i+j} * x_j  (cross-correlation of weights and x)
    """
    n = len(x)
    conv = fast_convolve(x)
    max_conv = conv.max()
    scaled = beta * (conv - max_conv)
    scaled = np.clip(scaled, -500, 0)
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    weights = exp_scaled / Z
    obj = max_conv + np.log(Z) / beta

    # Gradient via FFT cross-correlation
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * x[j]
    N = 1 << (len(weights) + n - 1 - 1).bit_length()
    W = rfft(weights, N)
    X = rfft(x, N)
    corr = irfft(np.conj(W) * X, N)
    grad = 2.0 * corr[:n]

    return obj, grad

def nesterov_smoothed_descent(a0, time_limit, beta_start=5.0, beta_end=500.0,
                                lr_start=0.1, lr_end=0.01):
    """
    Nesterov-accelerated projected gradient on smoothed objective.
    Adaptive beta schedule from smooth to sharp.
    """
    n = len(a0)
    x = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s

    x_prev = x.copy()
    best_x = x.copy()
    _, best_conv = smooth_objective_and_grad(x, beta_end)
    best_score = 2.0 * n * fast_convolve(x).max()

    t0 = time.monotonic()
    t_nest = 1.0
    it = 0

    total_beta_steps = 200
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive beta: logarithmic schedule
        frac = min(1.0, it / total_beta_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        # Adaptive learning rate: decrease over time
        lr = lr_start * (lr_end / lr_start) ** frac

        # Nesterov extrapolation
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = (t_nest - 1.0) / t_new

        y = x + momentum * (x - x_prev)
        y_proj = simplex_project(y)

        # Compute gradient at projected y
        _, grad = smooth_objective_and_grad(y_proj, beta)

        # Gradient step + projection
        x_new = simplex_project(y_proj - lr * grad)

        # Check improvement
        conv_new = fast_convolve(x_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_x = x_new.copy()

        x_prev = x
        x = x_new
        t_nest = t_new

    return best_x, best_score

def coord_descent_line_search(a0, time_limit, max_sweeps=200, verbose=False, label=""):
    """
    Coordinate descent with bisection line search for the 1D subproblem.
    For each coordinate i, find the optimal t that minimizes max_conv
    subject to a[i] + t >= 0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit and sweeps < max_sweeps:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            affected = conv[k_start:k_end + 1].copy()
            a_seg = a[0:n].copy()

            # Define the 1D objective: f(t) = max(unaff_max, max_r [affected[r] + 2*t*a_seg[r] + t^2 * (r==i)])
            # subject to t >= -a[i]
            self_idx = i  # relative index where r == i (the self-term)

            def f_1d(t):
                new_affected = affected + 2.0 * t * a_seg
                new_affected = new_affected.copy()
                new_affected[self_idx] = affected[self_idx] + 2.0 * t * a_seg[self_idx] + t * t
                return max(unaff_max, float(new_affected.max()))

            # Bisection to find optimal t in [-a[i], upper_bound]
            t_low = -a[i]
            # Estimate upper bound: current M / (2 * max(a_seg)) gives rough scale
            max_a_seg = a_seg.max()
            if max_a_seg < 1e-15:
                t_high = 0.01
            else:
                t_high = min(0.5, 2.0 * M / max_a_seg)

            # Golden section search (robust for convex 1D)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = t_high - gr * (t_high - t_low)
            d = t_low + gr * (t_high - t_low)
            fc = f_1d(c)
            fd = f_1d(d)

            for _ in range(30):
                if fc < fd:
                    t_high = d
                    d = c
                    fd = fc
                    c = t_high - gr * (t_high - t_low)
                    fc = f_1d(c)
                else:
                    t_low = c
                    c = d
                    fc = fd
                    d = t_low + gr * (t_high - t_low)
                    fd = f_1d(d)

            t_opt = (t_low + t_high) / 2.0
            f_opt = f_1d(t_opt)

            if f_opt < M - 1e-14:
                a[i] = max(a[i] + t_opt, 0.0)
                sum_a = a.sum()

                # Update convolution
                changes = 2.0 * t_opt * a_seg
                changes = changes.copy()
                changes[self_idx] = 2.0 * t_opt * (a[i] - t_opt) + t_opt * t_opt
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-Linesearch {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with vectorized step size evaluation.
    Tries multiple step sizes per coordinate and picks the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Step size fractions to try
    fracs = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, -0.02, -0.05, -0.1, -0.2, -0.3, -0.5, -0.7, -0.9]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_seg = a[0:n]
            self_idx = i

            best_delta = 0.0
            best_new_max = M

            for f in fracs:
                delta = f * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # Vectorized: new_affected = affected + 2*delta*a_seg
                new_affected = affected + 2.0 * delta * a_seg
                # Fix the self-term
                new_affected = new_affected.copy()
                new_affected[self_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_seg
                changes = changes.copy()
                changes[self_idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:
            if alt != solver:
                try:
                    prob.solve(solver=alt)
                    break
                except Exception:
                    continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    return b_val, float(M_var.value)

def fictitious_play(a0, time_limit, solver=None, verbose=False, label=""):
    """
    Fictitious play with Nesterov averaging.
    Alternates LP best responses and uses Nesterov-style averaging.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    a_avg = a.copy()
    b_avg = b.copy()
    best_x = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # LP best responses
        b_new, M_b = lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Nesterov averaging (Agmon-Varaiya style)
        gamma = 2.0 / (iters + 2.0)
        a_avg = (1.0 - gamma) * a_avg + gamma * a
        b_avg = (1.0 - gamma) * b_avg + gamma * b

        # Evaluate candidates
        candidates = [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-15:
                best_score = score
                best_x = cand.copy()

        # Periodic CD polish
        if iters % 5 == 0 and time.monotonic() - t0 < time_limit - 10:
            a_cd, s_cd, _ = fast_coord_descent(best_x, min(5.0, time_limit - (time.monotonic() - t0)), label=f"fp_cd_{iters}")
            if s_cd < best_score:
                best_score = s_cd
                best_x = a_cd

        if verbose and iters % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_x, best_score, iters

def peak_reduction(a0, steps=50, strength=0.1):
    """
    Iteratively reduce the peak of the convolution by targeting
    the coordinates that contribute most to the maximum lag.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)

    for _ in range(steps):
        k_star = int(conv.argmax())
        M = conv[k_star]
        avg = 1.0 / (2 * n - 1)

        if M <= avg * 1.01:
            break

        # Find contributing pairs
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)

        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = k_star - i
            contrib = a[i] * a[j]
            pressure[i] += contrib
            if j != i:
                pressure[j] += 0.5 * contrib

        # Reduce high-pressure coordinates, increase low-pressure
        p_max = pressure.max()
        if p_max < 1e-20:
            break
        pressure = pressure / p_max

        a_new = a * (1.0 - strength * pressure)
        # Redistribute to low-pressure areas
        low_mask = pressure < 0.3
        if low_mask.any():
            redistribute = (a.sum() - a_new.sum()) / max(low_mask.sum(), 1)
            a_new[low_mask] += redistribute * a[low_mask] / max(a[low_mask].sum(), 1e-15)

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        conv = fast_convolve(a)

    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def init_flat_optimal(n):
    """
    Initialize with a shape that approximately flattens the convolution.
    Uses the fact that a raised-cosine-like shape gives smoother convolution.
    """
    t = np.arange(n) / (n - 1)
    # Raised cosine (Hann window) - gives smoother convolution
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 0)
    # Add a small uniform base to avoid zeros
    a += 0.1
    s = a.sum()
    return a / s

def init_optimal_shape(n, rng):
    """
    Initialize with a shape based on the theoretical optimum.
    The optimum for minimizing peak auto-correlation with non-negativity
    is related to the Fejér kernel shape.
    """
    t = np.arange(n) / (n - 1)
    # Try different profiles
    profiles = []
    # Hann
    profiles.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    # Flat top
    x = (t - 0.5) * 2
    a_flat = 1.0 - 1.93 * x**2 + 1.29 * x**4 - 0.39 * x**6
    profiles.append(np.maximum(a_flat, 0))
    # Gaussian-like
    profiles.append(np.exp(-2.0 * ((t - 0.5) * 2) ** 2))
    # Sine
    profiles.append(np.sin(np.pi * t))
    # Power law
    profiles.append(np.maximum(1.0 - (2 * t - 1) ** rng.uniform(1, 4), 0))

    # Pick a random profile and add noise
    idx = rng.integers(0, len(profiles))
    a = profiles[idx].copy()
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Nesterov Smoothed Gradient Descent ===
    print(f"\n=== Phase 2: Nesterov Smoothed GD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_md = min(rem * 0.6, 400.0)
            print(f"  Nesterov MD, budget={t_md:.0f}s, n={len(best_seq)}")
            a_md, s_md = nesterov_smoothed_descent(best_seq, t_md,
                                                    beta_start=3.0, beta_end=300.0,
                                                    lr_start=0.05, lr_end=0.005)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "nesterov_md")

            # Second pass with different params
            rem = deadline - time.monotonic() - 500
            if rem > 30:
                t_md2 = min(rem * 0.4, 200.0)
                a_md2, s_md2 = nesterov_smoothed_descent(best_seq, t_md2,
                                                          beta_start=10.0, beta_end=800.0,
                                                          lr_start=0.03, lr_end=0.003)
                best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "nesterov_md2")

    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768, 1152]:
            if alt_n != n_best:
                fp_lengths.append(alt_n)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 400
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 250.0)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp) if best_seq is not None else None
            a_fp, score_fp, iters = fictitious_play(initial, t_alloc, solver=solver,
                                                      verbose=False, label=f"fp{n_fp}")
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

    # === Phase 4: Peak Reduction + CD ===
    print(f"\n=== Phase 4: Peak Reduction + CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Apply peak reduction
        a_peak = peak_reduction(best_seq, steps=100, strength=0.08)
        best_seq, best_score = check_and_update(a_peak, best_seq, best_score, "peak_reduce")

        # CD with line search on the result
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_cd = min(rem * 0.5, 150.0)
            a_cd, s_cd, _ = coord_descent_line_search(best_seq, t_cd,
                                                       verbose=False, label="cd_ls")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd_linesearch")

        # More fast CD
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd2 = min(rem * 0.5, 120.0)
            a_cd2, s_cd2, _ = fast_coord_descent(best_seq, t_cd2, label="fcd4")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "fcd_ph4")

    # === Phase 5: Multi-round refine with perturbation ===
    print(f"\n=== Phase 5: Multi-round Refine (best={best_score:.6f}) ===")
    if best_seq is not None:
        current = best_seq.copy()
        for rnd in range(5):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break

            # Perturb
            strength = 0.02 + 0.01 * rnd
            conv_cur = fast_convolve(current)
            M_cur = conv_cur.max()
            peak_thresh = M_cur * 0.9
            peak_pos = np.where(conv_cur >= peak_thresh)[0]

            pressure = np.zeros(len(current))
            for k in peak_pos:
                i_s = max(0, k - len(current) + 1)
                i_e = min(len(current) - 1, k)
                for i in range(i_s, i_e + 1):
                    j = k - i
                    prod = current[i] * current[j]
                    pressure[i] += prod
                    if j != i:
                        pressure[j] += 0.5 * prod

            p_max = pressure.max()
            if p_max > 1e-20:
                pressure = pressure / p_max
            noise = rng.standard_normal(len(current))
            current = current * (1.0 - strength * pressure + 0.2 * strength * noise)
            current = np.maximum(current, 0)
            s = current.sum()
            if s > 1e-15:
                current = current / s

            # Optimize
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 80.0)
            a_opt, s_opt, _ = fast_coord_descent(current, t_alloc, label=f"r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"refine_r{rnd}")
            if s_opt < best_score:
                current = best_seq.copy()

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best + 128, n_best + 256, n_best - 128, 1536, 2048]
        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 60
            if rem < 20 or alt_n < 512 or alt_n > 3000:
                continue
            t_alt = min(rem * 0.3, 60.0)
            if t_alt < 10:
                break
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_final = min(rem * 0.6, 100.0)
            print(f"  Final CD, budget={t_final:.0f}s")
            a_fin, s_fin, _ = fast_coord_descent(best_seq, t_final, label="final1")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd1")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_fin2, s_fin2, _ = fast_coord_descent(best_seq, min(rem * 0.7, 40.0), label="final2")
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
