"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid approach:
1. Multi-initialization with lopsided (asymmetric) sequences
2. p-norm L-BFGS for fast initial convergence
3. LP-based Nesterov alternating method for fine optimization
4. Subgradient refinement for final polish
5. Multi-length exploration
Key insight: optimal sequences are asymmetric (lopsided), not symmetric.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    max_b = fast_convolve(a_norm).max()
    return 2.0 * n * max_b

def p_norm_value_and_grad(a, p):
    n = len(a)
    c = fast_convolve(a)
    c = np.maximum(c, 1e-30)
    p_pow = c ** p
    S = p_pow.sum()
    if S < 1e-30:
        return 0.0, np.zeros_like(a)
    factor = S ** (1.0/p - 1.0) * p
    w = c ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * factor * corr
    val = 2.0 * n * S ** (1.0/p)
    return val, grad

def pnorm_lbfgs(a0, time_limit, p_start, p_end, n_stages):
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    p_values = np.geomspace(p_start, p_end, n_stages)
    
    for p_val in p_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 200))
        
        def func(x):
            return p_norm_value_and_grad(x, p_val)[0]
        
        def jac(x):
            return p_norm_value_and_grad(x, p_val)[1]
        
        try:
            res = minimize(func, a, jac=jac, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-20, 
                                   'gtol': 1e-12, 'maxls': 50})
            a_new = np.maximum(res.x, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                if compute_score(a_new) <= compute_score(a) + 1e-10:
                    a = a_new
        except:
            pass
    
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def build_conv_tilde(a):
    """Build matrix T such that (T @ b)[k] = convolve(a, b)[k] for k=0..2n-2."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a_fixed):
    """Find b >= 0, sum(b)=1 minimizing max_k convolve(a_fixed, b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_tilde(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_alternate(a0, time_limit, max_iters=50000):
    """Nesterov accelerated alternating LP method."""
    n = len(a0)
    a = a0.copy()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_new = lp_best_response(y)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Averaged iterate
        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 1) + 2)
        # Wait, that's not the standard Nesterov averaging. Let me use the standard one.
        # Standard: a_t = (1/(t+1)) * sum_{s=0}^{t} x_s where x_s is the "iterative" point.
        # In Nesterov: x_{t+1} = y_t + alpha_t * (z_t - y_t) where z_t = argmin at y_t.
        # The averaged iterate is (1/(t+2)) * sum_{s=0}^{t+1} x_s.
        
        # Actually let me use the simpler version:
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = compute_score(a)
        score_y = compute_score(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, n_steps=100000):
    """Subgradient method for final refinement."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Use diminishing step size
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        active = np.where(c >= max_c - 1e-10)[0]
        
        # Subgradient: average of gradients of active components
        if len(active) == 1:
            k = active[0]
            # grad_i = 2 * a[k-i] for valid k-i
            grad = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] = 2.0 * a[j]
        else:
            grad = np.zeros(n)
            for k in active:

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid multi-phase approach:
1. Multi-initialization with lopsided (asymmetric) sequences - key insight: optimal
   sequences are NOT symmetric, they are "lopsided" to reduce center auto-correlation.
2. p-norm L-BFGS for fast initial convergence with adaptive p-schedule.
3. LP-based Nesterov alternating method (zero-sum game formulation) for fine optimization.
4. Subgradient refinement using active-set identification for final polish.
5. Multi-length exploration with interpolation between lengths.
The LP alternation solves: given a, find b minimizing max_k conv(a,b)_k via linear program,
then alternates. Nesterov acceleration gives O(1/T^2) convergence rate.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(convolve(a,a)) with sum(a)=1 normalization."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    max_b = fast_convolve(a_norm).max()
    return 2.0 * n * max_b


def pnorm_value_gradient(a, p):
    """Value and gradient of 2n * (sum c_k^p)^(1/p) where c = convolve(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    c = np.maximum(c, 1e-30)
    cp = c ** p
    S = cp.sum()
    if S < 1e-30:
        return 0.0, np.zeros_like(a)
    factor = S ** (1.0 / p - 1.0) * p
    w = c ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * factor * corr
    val = 2.0 * n * S ** (1.0 / p)
    return val, grad


def pnorm_lbfgs(a0, time_limit, p_start=20.0, p_end=300.0, n_stages=8):
    """Multi-stage p-norm L-BFGS optimization."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    t0 = time.monotonic()
    p_values = np.geomspace(p_start, p_end, n_stages)

    for pv in p_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 150))

        def func(x):
            return pnorm_value_gradient(x, pv)[0]

        def jac(x):
            return pnorm_value_gradient(x, pv)[1]

        try:
            res = minimize(func, a, jac=jac, method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxls': 40})
            a_new = np.maximum(res.x, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                sc_new = compute_score(a_new)
                sc_old = compute_score(a)
                if sc_new <= sc_old + 1e-9:
                    a = a_new
        except Exception:
            pass

    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def build_conv_matrix(a):
    """Build T matrix where (T @ b)[k] = convolve(a, b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_cache = {}


def lp_best_response(a_fixed, max_time=5.0):
    """Solve LP: minimize M s.t. convolve(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    key = n
    if key not in _lp_cache:
        _lp_cache[key] = build_conv_matrix(np.ones(n) / n)
    # We need to rebuild with actual a values
    T = build_conv_matrix(a_fixed)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    t0 = time.monotonic()
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        if time.monotonic() - t0 > max_time:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None
    return b_val / sv


def nesterov_lp_alternate(a0, time_limit, max_iters=30000, stall_limit=8):
    """Nesterov accelerated LP alternating minimization."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = lp_best_response(y)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = compute_score(a)
        sc_y = compute_score(y)

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if it % 300 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= stall_limit:

                noise = rng.standard_normal(n) * (0.01 + 0.005 * stall)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall = 0
                prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(1 - u / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_refine(a0, time_limit, max_steps=50000):
    """Subgradient refinement using active-set of max convolution entries."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        # Find active set (entries within tolerance of max)
        tol = max(1e-10, 0.01 * max_c)
        active = np.where(c >= max_c - tol)[0]

        # Compute subgradient: average of gradients of active components
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        # Step size: diminishing with backtracking
        eta = 0.05 / (1.0 + 0.005 * step)
        for try_cnt in range(4):
            a_trial = a - eta * grad
            a_trial = project_simplex(a_trial)
            sc = compute_score(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
            eta *= 0.5

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        for noise_scale in [0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_sub = min(rem * 0.25, 80.0)
            a_pert = pnorm_lbfgs(perturbed, t_sub, p_start=50, p_end=400, n_stages=5)
            try_update(a_pert, f"pert_{noise_scale}")

    # Phase 8: Final intensive LP
    print(f"\n=== Phase 8: Final LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_fin = min(rem * 0.35, 200.0)
        print(f"  Budget: {t_fin:.0f}s")
        a_fin, _ = nesterov_lp_alternate(best_state['seq'], t_fin, max_iters=40000, stall_limit=15)
        try_update(a_fin, "final_lp")

    # Phase 9: Final subgradient polish
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sg = min(rem * 0.25, 60.0)
        a_sg, _ = subgradient_refine(best_state['seq'], t_sg, max_steps=30000)
        try_update(a_sg, "final_subgrad")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
