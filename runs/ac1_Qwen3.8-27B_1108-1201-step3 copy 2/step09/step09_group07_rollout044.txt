import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.SCS


def solve_lp_best_response(a_fixed, solver, max_iters=None):

    n = len(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    s = b_val.sum()
    if s < 1e-15:

def lp_fixed_point(a0, time_limit, solver, n_max_iters=100000):
    """Accelerated LP fixed-point iteration with adaptive step size."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    patience = 800

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        # Nesterov-style: x_{k+1} = (1 - alpha)*y + alpha*b, y = x + beta*(x - y)
        alpha = 2.0 / (it + 2)
        beta = (alpha - 1.0) / (alpha + 1.0) if alpha < 1 else 0.0
        y_new = y + beta * (y - a)
        a_new = (1 - alpha) * y_new + alpha * b_new
        a_new = project_simplex(np.maximum(a_new, 0))
        y = a_new
        a = a_new

        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[k_star - idx]
                top = np.argsort(contribs)[::-1][:max(3, n // 100)]

                stall_count = 0

        b = fast_convolve(x)
        k_star = int(np.argmax(b))

        # Subgradient: g_j = a_{j+k*} + a_{j-k*} (with bounds)
        g = np.zeros(n)
        for j in range(n):
            v = 0.0
            jp = j + k_star
            jm = j - k_star
            if 0 <= jp < n:
                v += x[jp]
            if 0 <= jm < n:
                v += x[jm]
            g[j] = v

        # Nesterov
        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
        if step_count > 0:
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr
        else:
            beta = 0.0
        y = x + beta * (x - x_prev)
        y = project_simplex(y)

        lr = 1.0 / (1.0 + 2.0 * step_count)
        x_prev = x.copy()
        x = project_simplex(y - lr * g / max(1.0, g.max()))
        step_count += 1

        if step_count % 100 == 0:
            sc = score_normalized(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=200000):
    """Coordinate descent with analytic line search for final polish."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    coord_order = np.arange(n)
    np.random.default_rng(0).shuffle(coord_order)
    ci = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        i = coord_order[ci % n]
        ci += 1
        if a[i] < 1e-12:
            continue

        # Analytic: try reducing a[i] by moving mass to random other coordinate
        # f(t) = score when a[i] -= t, distributed to random j
        # We do a small line search
        b_obj = best_score
        for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.95]:
            a_trial = a.copy()
            reduction = a[i] * (1.0 - factor)
            a_trial[i] *= factor
            # Distribute reduction to the coordinate with smallest value
            j = int(np.argmin(a_trial))
            a_trial[j] += reduction
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < b_obj - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break
        # Also try just scaling and re-projecting
        a_trial = a.copy()
        a_trial[i] *= 0.995
        a_trial = project_simplex(a_trial)
        sc = score_normalized(a_trial)
        if sc < best_score - 1e-12:
            a = a_trial
            best_score = sc
            best_a = a.copy()

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 0:
                print(f"    CD step={step} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def make_sin_init(n, freq, phase, alpha):
    """Sinusoidal modulation initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_init(n, rng):
    """Random non-negative initialization."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def make_trap_init(n, center=0.5, width=0.3):
    """Triangular initialization."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1.0 - np.abs(x - center) / width)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence with safe bounds."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def structured_perturb(a, rng, strength=0.1):
    """Perturb a using structured noise correlated with the peak structure."""
    n = len(a)
    b = fast_convolve(a)
    k_star = int(np.argmax(b))
    # Create perturbation that reduces peak contribution
    pert = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[k_star - idx]
    pert[idx] = -contribs
    # Add global noise
    pert += strength * rng.standard_normal(n)
    a_new = a + strength * pert / max(1.0, np.abs(pert).max())
    return project_simplex(np.maximum(a_new, 0))


def multi_freq_lp(a0, time_limit, solver, freqs=None):
    """LP fixed point with periodic structured perturbation."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(123)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = y.copy()

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0

        if it % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 5:
                # Structured perturbation
                a = structured_perturb(a, rng, strength=min(0.05, 0.01 * stall))
                y = a.copy()
                stall = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MFLP iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: subgradient warm-start, accelerated LP fixed-point,
    structured perturbation, multi-length exploration, coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_sin_init(n_init, 1, 0, 0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # === Phase 0: Quick subgradient warm-start ===
    print(f"\n=== Phase 0: Subgradient Warm-Start (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_sub = min(rem * 0.05, 45.0)
    if t_sub > 5:
        print(f"  Budget: {t_sub:.0f}s")
        a_sg, _ = subgradient_optimize(best_seq, t_sub, n_steps=20000)
        try_update(a_sg, "sg_warm")

    # === Phase 1: Main LP fixed-point with structured perturbation ===
    print(f"\n=== Phase 1: LP Fixed-Point (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = multi_freq_lp(best_seq, t_alloc, solver)
        try_update(a_lp, "lp_main")

    # === Phase 2: Second LP round with different perturbation ===
    print(f"\n=== Phase 2: LP Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        rng2 = np.random.default_rng(seed + 7)
        perturbed = structured_perturb(best_seq, rng2, strength=0.08)
        a_lp2, _ = multi_freq_lp(perturbed, t_alloc, solver)
        try_update(a_lp2, "lp_pert")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_steps=100000)
        try_update(a_cd, "cd")

    # === Phase 4: Alternative lengths ===
    print(f"\n=== Phase 4: Alternative Lengths (best={best_score:.7f}) ===")
    alt_lengths = []
    for delta in [-60, -40, -20, 0, 20, 40, 60, 80, 120]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 3072:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))
    # Prioritize lengths close to current best
    alt_lengths = sorted(alt_lengths, key=lambda x: abs(x - n_best))[:5]

    for alt_n in alt_lengths:
        if alt_n == n_best:
            continue
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.18, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick subgradient then LP
        t_sg = t_total * 0.15
        a_alt, _ = subgradient_optimize(init_alt, t_sg, n_steps=3000)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lp = min(rem * 0.25, t_total * 0.7)
            a_alt2, _ = multi_freq_lp(a_alt, t_lp, solver)
            try_update(a_alt2, f"lp_n{alt_n}")
        else:
            try_update(a_alt, f"sg_n{alt_n}")

    # === Phase 5: Multi-start with different frequencies ===
    print(f"\n=== Phase 5: Multi-Start Frequencies (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    freqs_to_try = [(2, 0.0, 0.4), (3, 1.0, 0.5), (1, 0.7, 0.6), (4, 0.3, 0.3)]
    for freq, phase, alpha in freqs_to_try[:3]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_sub = min(rem * 0.15, 120.0)
        if t_sub < 30:
            break
        a_init = make_sin_init(n_best, freq, phase, alpha)
        print(f"  freq={freq}, phase={phase}, alpha={alpha}, budget={t_sub:.0f}s")
        a_ms, _ = multi_freq_lp(a_init, t_sub, solver)
        try_update(a_ms, f"ms_f{freq}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_fin = min(rem * 0.35, 200.0)
        a_fin, _ = multi_freq_lp(best_seq, t_fin, solver)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.25, 60.0)
        a_ps, _ = coordinate_descent(best_seq, t_ps, n_steps=50000)
        try_update(a_ps, "final_cd")

    # === Phase 8: One more LP with very small perturbation ===
    rem = deadline - time.monotonic()
    if rem > 40:
        rng3 = np.random.default_rng(seed + 99)
        tiny_pert = best_seq + 0.005 * rng3.standard_normal(len(best_seq))
        tiny_pert = project_simplex(np.maximum(tiny_pert, 0))
        t_tiny = min(rem * 0.3, 60.0)
        a_tiny, _ = multi_freq_lp(tiny_pert, t_tiny, solver)
        try_update(a_tiny, "tiny_pert")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
