def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Let me think about an alternative: instead of solving the full LP, use a first-order approximation. Given the current a, the best response b satisfies:
- (a*b)[k] <= M for all k
- sum(b) = 1
- b >= 0
- At optimality, the set of k where (a*b)[k] = M is the "active set"

We can approximate this with a few iterations of a projected gradient method on b:
- Start with b = a
- Iterate: b = proj_simplex(b - eta * (a*b > M ? grad : 0))

This is essentially a "soft" version of the LP that might be faster per iteration.

Actually, let me try a completely different approach:

**Approach: Iterative "peak shaving" with analytical sub-solves**

At each step:
1. Compute c = a*a
2. Find the peak k* = argmax c
3. The peak value is c[k*] = sum_i a[i]*a[k*-i]
4. To reduce c[k*], we need to reduce some of the products a[i]*a[k*-i]
5. The most efficient way is to reduce the largest product
6. But reducing a[i] also affects other convolution values

This is essentially what the local refinement does, but we can make it smarter by:
- Considering multiple peaks simultaneously
- Using a "rebalance" operation that moves mass from high-contribution positions to low-contribution ones

Let me try yet another angle. What if I formulate this as a quadratic program?

minimize (1/k) * sum_{j in top_k} (a*a)[j]
subject to sum(a) = 1, a >= 0

This is a QP if the top-k set is fixed! The objective is:
(1/k) * sum_{j in S} (sum_i a[i] * a[j-i]) = (1/k) * a^T Q_S a

where Q_S is a matrix that depends on S. This is a quadratic form in a, so it's a QP!

If I can solve this QP efficiently, I can use it as a subproblem in a block coordinate / active set method.

But the QP is non-convex in general (Q_S might not be PSD). However, if S is the set of peaks and we're looking for a local minimum, the QP might be locally convex.

Actually, Q_S = (1/k) * sum_{j in S} C_j where C_j is the matrix with (C_j)_{i,l} = 1 if j-i = l (i.e., C_j is a shifted identity). This is a sum of permutation-like matrices, which is symmetric but not necessarily PSD.

1. Start from the parent solution
2. Use top-k gradient with momentum (faster than p-norm for high p)
3. Periodically apply Nesterov LP
4. Try a few different lengths
5. Finish with local refinement

Let me also add a "Fourier flattening" step: in the Fourier domain, multiply the spectrum by a function that flattens it, then transform back and project to the simplex. This is a non-local operation that might escape local minima.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using top-k peak momentum gradient
descent, Nesterov LP alternating, Fourier-domain spectral equalization, and
multi-length exploration. Combines fast per-iteration cost with diverse
optimization strategies to escape local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_topk_grad(a, k):
    n = len(a)
    c = fast_convolve(a)
    # For auto-conv of non-negative seq, max is in the middle
    # Use top-k from the full convolution
    topk_idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for ki in topk_idx:
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[ki - idx]
    grad /= float(k)
    return grad, c

def topk_momentum(a0, time_limit, k_start=3, k_end=10, lr0=1.0, mu0=0.90):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr0
    mu = mu0
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        k = max(1, int(k_start + (k_end - k_start) * frac))
        
        grad, c = compute_topk_grad(a, k)
        
        v = mu * v - lr * grad
        a_new = project_simplex(a + v)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                lr *= 0.9
                v *= 0.5
                stall = 0
        
        a = a_new
        lr *= 0.9995
        mu = min(0.999, mu0 + 0.099 * min(1.0, step / 200.0))
        step += 1
        
        if step % 1000 == 0:
            print(f"      TopK-MD step={step} ({elapsed:.0f}s): score={best_score:.6f}, k={k}, lr={lr:.6f}")
    
    return best_a, best_score

def nesterov_lp(a0, time_limit, n_max_iters=100000):
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    y = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score
    
    # Try to find a good solver
    solver = None
    for sn in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x_test = cp.Variable(10, nonneg=True)
            prob_test = cp.Problem(cp.Minimize(cp.sum(x_test)), [cp.sum(x_test) == 1])
            prob_test.solve(solver=sn)
            if prob_test.status in ('optimal', 'optimal_inaccurate'):
                solver = sn
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        size = 2 * n - 1
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j+n, j] = a
        
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            continue
        b_val /= sb
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 500 == 0:
            if best_score < prev_stall_score - 1e-7:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 5:
                rng2 = np.random.default_rng(it + 42)
                noise = rng2.standard_normal(n) * 0.03
                y = np.maximum(y + noise, 0)
                y /= y.sum()
                stall_count = 0
                prev_stall_score = best_score
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def local_refine(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        top_coords = idx[np.argsort(contribs)[::-1][:10]]
        
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            top_pairs = np.argsort(contribs)[::-1][:5]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial /= st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            if step > 200:
                break
            rng2 = np.random.default_rng(step + 7)
            n_pert = int(rng2.integers(1, 4))
            indices = rng2.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng2.uniform(-0.03, 0.03))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial /= st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fourier_equalize(a0, time_limit, n_iters=500):
    """Apply Fourier-domain spectral equalization to flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: flatten the power spectrum
        # Only modify frequencies where power is above average
        avg_power = power.mean()
        # Soft flattening: reduce peaks in spectrum
        target = np.sqrt(avg_power)  # target amplitude
        A_new = A.copy()
        amp = np.abs(A_new)
        phase = np.angle(A_new)
        # Where amplitude is above target, reduce it
        excess = np.maximum(amp - target, 0)
        amp_new = amp - 0.1 * excess
        A_new = amp_new * np.exp(1j * phase)
        
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new /= s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using excess-gradient momentum,
Nesterov LP alternating, p-norm LBFGS polish, Fourier spectral equalization,
local refinement, and multi-length exploration.
Key innovations: excess-gradient (soft top-k) for fast first-order convergence,
combined with Nesterov LP for global progress, and adaptive multi-phase scheduling.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def excess_gradient(a):
    """Compute gradient of sum_k max(0, (a*a)[k] - mean) w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_mean = c.mean()
    excess = np.maximum(c - c_mean, 0.0)
    fft_size = 1 << (2 * n - 2).bit_length()
    E = rfft(excess, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(E * np.conj(A), fft_size)[:n]
    return 2.0 * corr, c


def excess_momentum(a0, time_limit, eta0=0.5, mu0=0.85, decay=0.9995):

def p_norm_objective(z, p, n):
    """Objective: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_jac(z, p, n):
    """Jacobian of p-norm objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=7):
    """L-BFGS with adaptive p-norm smoothing in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        max_iter = max(50, int(6000 / n_stages))

        try:
            res = minimize(
                fun=lambda z_val: p_norm_objective(z_val, stage_p, n),
                x0=z,
                jac=lambda z_val: p_norm_jac(z_val, stage_p, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': max_iter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm LBFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def nesterov_lp(a0, time_limit, n_max_iters=100000):
    """Nesterov dual averaging with LP best-response."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    y = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    # Find working solver
    solver = None
    for sn in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            xt = cp.Variable(5, nonneg=True)
            pt = cp.Problem(cp.Minimize(cp.sum(xt)), [cp.sum(xt) == 1])
            pt.solve(solver=sn)
            if pt.status in ('optimal', 'optimal_inaccurate'):
                solver = sn
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        size = 2 * n - 1
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = a

        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sb = b_val.sum()
        if sb < 1e-15:
            continue
        b_val /= sb

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_stall_score - 1e-7:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 5:
                rng2 = np.random.default_rng(it + 42)
                noise = rng2.standard_normal(n) * 0.03
                y = np.maximum(y + noise, 0)
                y /= y.sum()
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def local_refine(a0, time_limit, n_steps=50000):
    """Local refinement: coordinate descent + pair adjustments + peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        top_coords = idx[np.argsort(contribs)[::-1][:12]]

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            top_pairs = np.argsort(contribs)[::-1][:6]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.97, 0.94, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    st = a_trial.sum()
                    if st < 0.01:
                        continue
                    a_trial /= st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Random restart within local neighborhood
                rng2 = np.random.default_rng(step + 7)
                n_pert = int(rng2.integers(2, 8))
                indices = rng2.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for i in indices:
                    a_trial[i] *= (1.0 + rng2.uniform(-0.04, 0.04))
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 0.01:
                    a_trial /= st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                        improved = True
                if not improved and no_improve > 2000:
                    break
        else:
            no_improve = 0

    return best_a, best_score


def fourier_equalize(a0, time_limit, n_iters=2000):
    """Fourier-domain spectral equalization to flatten power spectrum."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        amp = np.abs(A)
        phase = np.angle(A)
        avg_amp = amp.mean()
        # Softly flatten: reduce peaks toward average
        strength = 0.05 + 0.05 * (it / max(1, n_iters))
        excess = np.maximum(amp - avg_amp, 0)
        amp_new = amp - strength * excess
        # Also boost valleys slightly
        deficit = np.maximum(avg_amp - amp, 0)
        amp_new = amp_new + 0.3 * strength * deficit
        amp_new = np.maximum(amp_new, 1e-10)

        A_new = amp_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new /= s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def eval_safe(a):
    """Evaluate sequence using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining excess-gradient momentum, Nesterov LP,
    p-norm LBFGS, Fourier equalization, local refinement, and multi-length
    exploration. Designed to escape local minima via diverse optimization
    strategies and adaptive scheduling.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(-1, 1, n_init)
        parent = np.exp(-x ** 2 * 3)
        parent = np.maximum(parent, 0)
        parent /= parent.sum()
        best_seq = parent.copy()
        best_score = eval_safe(parent)
        print(f"Default init: n={len(parent)}, score={best_score:.6f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Nesterov LP (proven strong method) ===
    print(f"\n=== Phase 1: Nesterov LP (n={n_best}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res, _ = nesterov_lp(best_seq, t_alloc, n_max_iters=100000)
        update_best(a_res, "nesterov_lp_1")

    # === Phase 2: Excess gradient momentum ===
    print(f"\n=== Phase 2: Excess Momentum (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res, _ = excess_momentum(best_seq, t_alloc, eta0=0.6, mu0=0.85, decay=0.9995)
        update_best(a_res, "excess_mom_1")

    # === Phase 3: p-norm LBFGS polish ===
    print(f"\n=== Phase 3: p-norm LBFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res = p_norm_lbfgs(best_seq, t_alloc, p_start=30.0, p_end=600.0, n_stages=8)
        update_best(a_res, "pnorm_lbfgs_1")

    # === Phase 4: Fourier equalization ===
    print(f"\n=== Phase 4: Fourier Equalize (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_res, _ = fourier_equalize(best_seq, t_alloc, n_iters=5000)
        update_best(a_res, "fourier_eq")

    # === Phase 5: Local refinement ===
    print(f"\n=== Phase 5: Local Refine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 130.0)
    if t_alloc > 10:
        a_res, _ = local_refine(best_seq, t_alloc, n_steps=50000)
        update_best(a_res, "local_1")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_cur = len(best_seq)
        alt_lengths = []
        for delta in [-64, -32, 32, 64, -128, 128, 192]:
            alt_n = n_cur + delta
            if 512 <= alt_n <= 2048:
                alt_lengths.append(alt_n)

            if 512 <= rn <= 2048 and rn not in alt_lengths:
                alt_lengths.append(rn)
        alt_lengths = sorted(set(alt_lengths), key=lambda x: abs(x - n_cur))

        n_explored = 0
        for alt_n in alt_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 80 or n_explored >= 3:
                break
            n_explored += 1
            t_total = min(rem * 0.18, 180.0)
            if t_total < 40:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_seq(best_seq, alt_n)

            # Excess momentum
            t_em = t_total * 0.4
            a_alt, _ = excess_momentum(init_alt, t_em, eta0=0.4, mu0=0.85, decay=0.9995)
            improved = update_best(a_alt, f"excess_n{alt_n}")

            # If improved, LP polish
            rem = deadline - time.monotonic()
            if improved and rem > 30:
                t_lp = min(rem * 0.15, 100.0)
                a_alt2, _ = nesterov_lp(best_seq, t_lp, n_max_iters=30000)
                update_best(a_alt2, f"lp_n{alt_n}")

    if rem > 80:

        for noise_scale in [0.02, 0.05, 0.08]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed /= perturbed.sum()

            t_sub = min(rem * 0.22, 80.0)
            a_pert, _ = excess_momentum(perturbed, t_sub, eta0=0.4, mu0=0.88, decay=0.9995)
            update_best(a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = p_norm_lbfgs(best_seq, t_fin, p_start=100.0, p_end=1000.0, n_stages=5)
        update_best(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.35, 60.0)
        a_lr, _ = local_refine(best_seq, t_lr, n_steps=30000)
        update_best(a_lr, "final_local")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_em = min(rem * 0.4, 40.0)
        a_em, _ = excess_momentum(best_seq, t_em, eta0=0.1, mu0=0.95, decay=0.9998)
        update_best(a_em, "final_excess")

    # Final verification
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed_total:.0f}s")
    return [float(x) for x in best_seq]
