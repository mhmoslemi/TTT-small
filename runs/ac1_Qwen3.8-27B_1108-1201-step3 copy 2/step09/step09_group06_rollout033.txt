def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_full(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    af = np.asarray(a_fixed, dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = af
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_value(a, T):
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (1.0 / T) * (max_Tb + np.log(exp_Tb.sum()))


def lse_gradient(a, T):
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    p = exp_Tb / exp_Tb.sum()
    n = len(a)
    g = 2.0 * sp_correlate(p, a, mode='valid')
    return g, b


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_bar = np.max(rho[cond])
    theta = (cssv[rho_bar - 1] - 1) / rho_bar
    return np.maximum(v - theta, 0)

            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fista_lse(a0, time_limit, T_start=80.0, T_end=2.0, n_stages=6,
              iters_per_stage=2000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        x = a.copy()
        x_prev = a.copy()
        t_mom = 1.0
        step = 0.05 / stage_T

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 50 == 0:
                remaining = time_limit - (time.monotonic() - t0)
                if remaining < 1:
                    break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            momentum = (t_mom - 1.0) / t_new
            z = x + momentum * (x - x_prev)
            z = project_simplex(z)

            g, b_z = lse_gradient(z, stage_T)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            x_trial = project_simplex(z - step * g)

            val_trial = fast_convolve(x_trial).max()
            val_x = fast_convolve(x).max()
            if 2.0 * n * val_trial < 2.0 * n * val_x:
                x_prev = x
                x = x_trial
                step = min(step * 1.1, 1.0 / stage_T)
            else:
                step *= 0.7

            t_mom = t_new

            score = 2.0 * n * fast_convolve(x).max()
            if score < best_val:
                best_val = score
                best_a = x.copy()

        a = x
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_val:
            best_val = score
            best_a = a.copy()
        print(f"    FISTA T={stage_T:.1f}: score={best_val:.6f}")

    return best_a, best_val


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    lp_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        lp_count += 1
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2.0)) * b_new) / (it + 1.0 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 1000 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 10:
                rng = np.random.default_rng(it * 13 + 7)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            elapsed = time.monotonic() - t0
            if it % 5000 == 0:
                print(f"    NesterovLP iter={it+1}/{lp_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=300.0, T_end=0.1, n_stages=12,
                 maxiter_total=20000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_soft = softmax_from_z(z_in)
            b = fast_convolve(a_soft)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_soft = softmax_from_z(z_in)
            b = fast_convolve(a_soft)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            n_local = len(a_soft)
            grad_a = np.zeros(n_local)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n_local + 1)
                je = min(n_local - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_soft[k - idx]
            S = np.dot(grad_a, a_soft)
            return a_soft * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter_total // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
            a_curr = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_curr).max()
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if np.log10(stage_T) < 0:
            print(f"    L-BFGS T={stage_T:.3f} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_directed_search(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 20000 > time_limit - (time.monotonic() - t0) + 5:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        if peak_val < 1e-15:
            break

        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        improved = False
        for si in top_src:
            if improved:
                break
            for di in low_dst:
                if improved:
                    break
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            n_random = 5
            for _ in range(n_random):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * rng.uniform(0.01, 0.06)
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and step > 10000:
            if step % 2000 == 0:
                rng2 = np.random.default_rng(step)
                idx = rng2.integers(n, size=n // 10)
                a[idx] *= rng2.uniform(0.8, 1.2, size=len(idx))
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSearch step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
        a[n // 4:3 * n // 4] *= 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 1: FISTA on LSE (fast warm-up)
    print(f"\n=== Phase 1: FISTA LSE (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fista, _ = fista_lse(best_seq, t_alloc, T_start=100.0, T_end=3.0,
                                n_stages=5, iters_per_stage=1500)
        try_update(a_fista, "fista_1")

    # Phase 2: Nesterov LP (main workhorse)
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: L-BFGS multi-stage annealing
    print(f"\n=== Phase 3: L-BFGS Anneal (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 300->0.1, 12 stages")
        a_lb, _ = lbgfs_refine(best_seq, t_alloc, T_start=300.0, T_end=0.1,
                                n_stages=12, maxiter_total=25000)
        try_update(a_lb, "lbgfs_1")

    # Phase 4: Second LP round
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 5: Peak-directed local search
    print(f"\n=== Phase 5: Peak-Directed Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_pk, _ = peak_directed_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_pk, "peak_search")

    # Phase 6: Alternative lengths
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0, 1.1]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.3, 180.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interp improved: {sc_alt:.6f}")

        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(cur_best, t_lp, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_fista = min(rem * 0.2, 60.0)
            if t_fista > 15:
                a_alt2, _ = fista_lse(best_seq, t_fista, T_start=50.0, T_end=1.0,
                                       n_stages=4, iters_per_stage=1000)
                try_update(a_alt2, f"fista_n{alt_n}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.05,
                             n_stages=8, maxiter_total=10000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_ls = min(rem * 0.25, 50.0)
        a_ls2, _ = peak_directed_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.35, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=15000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
