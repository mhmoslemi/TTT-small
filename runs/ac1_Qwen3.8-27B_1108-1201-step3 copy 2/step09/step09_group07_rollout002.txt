import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent(a0, time_limit, T_start=10.0, T_end=500.0, n_stages=10, steps_per_stage=2000):
    """Mirror descent (exponential weighting) with temperature annealing."""
    n = len(a0)
    a = project_simplex(a0.copy())
    a = np.maximum(a, 1e-15)  # ensure strictly positive for log
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage in range(n_stages):
        T = temps[stage]
        # Adaptive step size
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            b = fast_convolve(a)
            Tb = T * b
            m = Tb.max()
            w = np.exp(Tb - m)
            w = w / w.sum()
            
            # Gradient: 2 * cross_corr(a, w)
            fft_size = 1 << (n + len(w) - 1).bit_length()
            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            grad = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
            
            # Mirror descent update: a * exp(-eta * grad)
            eta = 1.0 / (T * (np.max(np.abs(grad)) + 1e-10))
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            # Check improvement
            if (step + 1) % 20 == 0:
                sc = score_normalized(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_score

    b_var = cp.Variable(n, nonneg=True)

        constraints = [cp.sum(b_var) == 1]
        for k in active_ks:
            # Build row k of T: T[k, i] = a_fixed[k-i] if 0 <= k-i < n
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            row = np.zeros(n)
            row[i_min:i_max+1] = a_fixed[k - np.arange(i_min, i_max+1)]
            constraints.append(row @ b_var <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        
        # Check violations
        full_conv = np.convolve(a_fixed, b_val)
        M_val = M_var.value
        violations = np.where(full_conv > M_val + 1e-7)[0]

import numpy as np
import time

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(a, b):
    """Compute cross-correlation (a * b)[i] = sum_j a[j] * b[j+i] truncated to len(a)."""
    n = len(a)
    m = len(b)
    fft_size = 1 << (n + m - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(np.conj(B) * A, fft_size)[:n]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent_refine(a0, time_limit, T_start=5.0, T_end=300.0, n_stages=12, steps_per_stage=3000):
    """Mirror descent (exponential weighting) with adaptive step size and temperature annealing."""

def nesterov_mirror_descent(a0, time_limit, T_start=5.0, T_end=400.0, n_stages=10, steps_per_stage=3000):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 1e-15).copy())
    x = np.maximum(x, 1e-12)
    x = x / x.sum()
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov momentum
            if step_count > 0:
                t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0

            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-12)
            y = y / y.sum()

            # Gradient at y
            b = fast_convolve(y)
            Tb = T * b
            m = Tb.max()
            exp_terms = np.exp(Tb - m)
            w = exp_terms / exp_terms.sum()
            grad = 2.0 * cross_correlate(y, w)

            x_prev = x.copy()
            grad_max = np.max(np.abs(grad))
            eta = 1.0 / (T * (grad_max + 1e-10))
            eta = min(eta, 0.8)

            log_ratio = -eta * grad
            log_ratio = np.clip(log_ratio, -15, 15)
            x = y * np.exp(log_ratio)
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            step_count += 1

            if (step + 1) % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def solve_lp_best_response_cutting_plane(a_fixed, solver, max_rounds=30, initial_peaks=15):
    """Solve min_b max_k conv(a_fixed, b)[k] using cutting plane method."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Find initial peaks
    b_self = fast_convolve(a_fixed)
    top_ks = np.argsort(b_self)[::-1][:min(initial_peaks, 2 * n - 1)]
    active_ks = list(top_ks)

    for round_num in range(max_rounds):
        constraints = [cp.sum(b_var) == 1]
        for k in active_ks:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            row = np.zeros(n)
            row[idx] = a_fixed[k - idx]
            constraints.append(row @ b_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s

        M_val = float(M_var.value)

        # Check violations using FFT
        full_conv = np.convolve(a_fixed, b_val)
        violations = np.where(full_conv > M_val + 1e-6)[0]
        if len(violations) == 0:
            return b_val, M_val

        # Add top violated constraints
        sorted_viol = violations[np.argsort(full_conv[violations])[::-1]][:10]
        for k in sorted_viol:
            if k not in active_ks:
                active_ks.append(k)

        if len(active_ks) > 2 * n - 1:
            # All constraints active, just return
            return b_val, M_val

    return np.asarray(b_var.value, dtype=np.float64) if b_var.value is not None else None, float(M_var.value)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            try_update(a_sub, f"init_m{mode}")
            rem = deadline - time.monotonic()
            if rem > 20 and t_sub * 0.3 > 10:
                a_sub2, _ = lp_best_response_refine(a_sub, t_sub * 0.3, solver=solver, n_max_iters=3000)
                try_update(a_sub2, f"init_m{mode}_lp")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 120.0)
        a_fin, _ = lp_best_response_refine(best_seq, t_fin, solver=solver, n_max_iters=10000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.15, 60.0)
        a_ps2, _ = peak_shave_refine(best_seq, t_ps, n_steps=50000)
        try_update(a_ps2, "final_ps")

    # === Phase 9: One more mirror descent + LP cycle ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t_md = rem * 0.3
        t_lp = rem * 0.5
        if t_md > 20:
            a_final_md, _ = nesterov_mirror_descent(best_seq, t_md, T_start=10.0, T_end=600.0,
                                                    n_stages=8, steps_per_stage=5000)
            try_update(a_final_md, "final_md")
            rem = deadline - time.monotonic()
            if rem > 30 and t_lp > 20:
                a_final_lp, _ = lp_best_response_refine(best_seq, t_lp, solver=solver, n_max_iters=10000)
                try_update(a_final_lp, "final_lp2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - time_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, elapsed={elapsed:.0f}s)")

    return [float(x) for x in best_seq]
