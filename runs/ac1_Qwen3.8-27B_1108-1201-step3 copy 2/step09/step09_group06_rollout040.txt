import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Compute a * a (self-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Compute a * b (cross-convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build the Toeplitz matrix T such that T @ b = a * b."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def alternating_lp(a_init, time_budget, solver, n_iters=50000):
    """Nesterov-accelerated alternating LP."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    y = a.copy()  # Nesterov momentum point
    best_a = a.copy()
    best_score = 2 * n * fast_convolve(a).max()
    
    T = build_conv_matrix(a)  # Will be rebuilt each iteration
    
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP: min M s.t. (y*b)[k] <= M, b >= 0, sum(b) = 1
        # Using y as the fixed variable
        T_y = build_conv_matrix(y)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_y @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
        b_val = b_val / b_val.sum()
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_val
        a_new = ((it+1) * a + alpha * (it+2) * b_val) / (it + 1 + alpha * (it+2))
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        score = 2 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        score_y = 2 * n * fast_convolve(y_new).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
        
        a = a_new
        y = y_new
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated
alternating LP with dual warm-start, subgradient polishing, and
multi-length exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A*A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A*B, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2*n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver

def solve_best_response_lp(y, solver):
    """Solve: min M s.t. (y*b)_k <= M for all k, b>=0, sum(b)=1."""
    n = len(y)
    T = build_conv_matrix(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        b_val, M_val = solve_best_response_lp(y, solver)
        if b_val is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        y /= y.sum()
        
        a = ((it+1)*a + alpha*(it+2)*b_val) / (it + 1 + alpha*(it+2))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = y.copy()
        else:
            a /= s
        
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_y = 2.0 * n * fast_convolve(y).max()
        
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        
        if (it+1) % 500 == 0:

                y = np.maximum(y, 0)
                y /= y.sum()
                a = y.copy()
                stall = 0

```python
def multiplicative_weights_best_response(y, eta=0.1, max_iters=2000):
    """Approximate best response using multiplicative weights."""
    n = len(y)
    b = np.ones(n) / n
    fft_size = 1 << (2*n - 2).bit_length()
    Y_fft = rfft(y, fft_size)
    
    for it in range(max_iters):
        # Compute y * b
        B_fft = rfft(b, fft_size)
        conv = irfft(Y_fft * B_fft, fft_size)[:2*n-1]
        k = int(np.argmax(conv))
        max_val = conv[k]
        
        if max_val < 1e-15:
            break
        
        # Update: decrease b[j] proportionally to y[k-j]
        # The "loss" for coordinate j is y[k-j] (contribution to the max)
        # Multiply by exp(-eta * loss / max_val)
        for j in range(n):
            m = k - j

def solve_best_response_dual(y, solver):
    """Solve dual: max lambda s.t. sum(y_lam)=1, y_lam>=0, (T^T y_lam)_j >= lambda."""
    n = len(y)
    size = 2*n - 1
    y_lam_var = cp.Variable(size, nonneg=True)
    lambda_var = cp.Variable()
    
    # Compute T^T @ y_lam using the sliding window formula
    # (T^T @ y_lam)_j = sum_{m=0}^{n-1} y_lam[j+m] * y[m]
    # This is a linear expression in y_lam
    
    # Build the constraint matrix for T^T
    # (T^T)[j, k] = y[k-j] if 0 <= k-j < n, else 0
    # So (T^T @ y_lam)_j = sum_{k: 0<=k-j<n} y_lam[k] * y[k-j]
    # = sum_{m=0}^{n-1} y_lam[j+m] * y[m]
    
    # The matrix T^T has shape (n, 2n-1)
    # (T^T)[j, j+m] = y[m] for m = 0, ..., n-1 (when j+m < 2n-1)
    
    TT = np.zeros((n, size))
    for j in range(n):
        max_m = min(n, size - j)
        for m in range(max_m):
            TT[j, j+m] = y[m]
    
    constraints = [
        cp.sum(y_lam_var) == 1,
        TT @ y_lam_var >= lambda_var
    ]
    prob = cp.Problem(cp.Maximize(lambda_var), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    # Recover primal solution from dual
    # By complementary slackness: b[j] > 0 ⟹ (T^T y_lam)_j = lambda
    y_lam_val = np.asarray(y_lam_var.value, dtype=np.float64)
    lambda_val = float(lambda_var.value)
    
    # The primal solution b is supported on the "tight" constraints
    # (T^T y_lam)_j = lambda
    tight = np.where((TT @ y_lam_val) <= lambda_val + 1e-8)[0]
    
    # For tight constraints, set b[j] > 0
    # The exact values require solving a sub-problem
    # For now, just set b uniformly on the tight set
    b_val = np.zeros(n)
    b_val[tight] = 1.0 / len(tight)
    
    return b_val, lambda_val
```

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver


def solve_best_response_lp(y, solver):
    n = len(y)
    T = build_conv_matrix(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)


def alternating_lp_nesterov(a0, time_budget, solver, max_iters=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score + 1e-10
    noise_level = 0.01

    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break

        b_val, M_val = solve_best_response_lp(y, solver)
        if b_val is None:
            break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_val
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y /= sy

        a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_y = 2.0 * n * fast_convolve(y).max()

        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-7:
                prev_best = best_score
                stall = 0
            else:
                stall += 1
            if stall >= 15:
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.random(n) * noise_level
                y = y + noise
                y = np.maximum(y, 0)
                y /= y.sum()
                a = y.copy()
                stall = 0
                prev_best = best_score
                noise_level = max(0.005, noise_level * 0.7)

    return best_a, best_score


def compute_subgradient_vectorized(a, active_indices):
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_indices)
    return g


def subgradient_polish(a0, time_budget, n_active_fracs=None):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()

    if n_active_fracs is None:
        n_active_fracs = [0.05, 0.1, 0.2, 0.5]

    for frac in n_active_fracs:
        if time.monotonic() - t0 > time_budget:
            break
        phase_budget = (time_budget - (time.monotonic() - t0)) / len(n_active_fracs)
        lr = 0.3
        no_improve = 0
        max_iters = 100000

        for step in range(max_iters):
            if time.monotonic() - t0 > time_budget:
                break

            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b

            gap = best_score - f_lb
            if gap < 1e-10:
                break

            threshold = max_b - frac * max_b
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(b))])

            g = compute_subgradient_vectorized(a, active)
            g_norm = np.abs(g).max()
            if g_norm < 1e-15:
                break

            polyak_lr = gap / (2.0 * g_norm * best_score + 1e-15)
            alpha = max(min(lr, polyak_lr), 0.005)

            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()

            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                lr = min(lr * 1.03, 3.0)
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 30:
                    lr *= 0.6
                    no_improve = 0
                lr = max(lr, 0.005)

            if step % 20000 == 0:
                print(f"    SG frac={frac:.2f} step={step}: score={best_score:.7f} lr={lr:.4f}")

    return best_a, best_score


def local_mass_transfer(a0, time_budget, n_steps=80000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0][:5]

        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        improved = False
        top_src = np.argsort(contrib)[::-1][:8]
        low_dst = np.argsort(contrib)[:8]

        for si in top_src:
            if improved:
                break
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    sa = a_trial.sum()
                    if sa < 1e-15:
                        continue
                    a_trial /= sa
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-14:
                    continue
                amount = a[i] * 0.02
                a_trial = a.copy()
                a_trial[i] -= amount
                a_trial[j] += amount
                a_trial = np.maximum(a_trial, 0)
                sa = a_trial.sum()
                if sa < 1e-15:
                    continue
                a_trial /= sa
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and step > 10000:
            break

    return best_a, best_score


def lbgfs_smoothed(a0, time_budget, T_start=100.0, T_end=0.1, maxiter=10000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_budget:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_in):
            if time.monotonic() - t0 > time_budget:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

        a = 1.0 / (1.0 + np.exp(-20 * (x - 0.5)))
    else:
        a = np.ones(n) * 0.8 + 0.2 * np.cos(2 * np.pi * 3 * np.linspace(0, 1, n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def run_pipeline(a_init, time_budget, solver, label=""):
    n = len(a_init)
    a = a_init.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()

    # Phase 1: Alternating LP
    rem = time_budget - (time.monotonic() - t0)
    t_lp = rem * 0.55
    if t_lp > 30:
        a_lp, sc_lp = alternating_lp_nesterov(a, t_lp, solver)
        sc_eval = evaluate_safe(a_lp)
        if sc_eval < best_score:
            best_score = sc_eval
            a = a_lp
            print(f"    LP [{label}]: {sc_eval:.7f}")

    # Phase 2: Subgradient polish
    rem = time_budget - (time.monotonic() - t0)
    t_sg = rem * 0.25
    if t_sg > 20:
        a_sg, sc_sg = subgradient_polish(a, t_sg, n_active_fracs=[0.05, 0.1, 0.2])
        sc_eval = evaluate_safe(a_sg)
        if sc_eval < best_score:
            best_score = sc_eval
            a = a_sg
            print(f"    SG [{label}]: {sc_eval:.7f}")

    # Phase 3: L-BFGS smoothed
    rem = time_budget - (time.monotonic() - t0)
    t_lb = rem * 0.12
    if t_lb > 15:
        a_lb = lbgfs_smoothed(a, t_lb, T_start=50.0, T_end=0.1, maxiter=5000)
        sc_eval = evaluate_safe(a_lb)
        if sc_eval < best_score:
            best_score = sc_eval
            a = a_lb
            print(f"    LBGFS [{label}]: {sc_eval:.7f}")

    # Phase 4: Local search
    rem = time_budget - (time.monotonic() - t0)
    t_ls = rem * 0.10
    if t_ls > 10:
        a_ls, sc_ls = local_mass_transfer(a, t_ls, n_steps=30000)
        sc_eval = evaluate_safe(a_ls)
        if sc_eval < best_score:
            best_score = sc_eval
            a = a_ls
            print(f"    LOCAL [{label}]: {sc_eval:.7f}")

    return a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.7f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    n_best = len(best_seq)

    # Main optimization at parent length
    print(f"\n=== Main: n={n_best}, initial={best_score:.7f} ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.45, 700.0)
    if t_main > 100:
        print(f"  Budget: {t_main:.0f}s")
        a_main, sc_main = run_pipeline(best_seq, t_main, solver, "main")
        try_update(a_main, "main")

    # Second round at parent length with different subgradient params
    rem = deadline - time.monotonic()
    t_sg2 = min(rem * 0.12, 150.0)
    if t_sg2 > 30:
        print(f"\n=== SG Round 2: budget={t_sg2:.0f}s ===")
        a_sg2, _ = subgradient_polish(best_seq, t_sg2, n_active_fracs=[0.03, 0.08, 0.15, 0.3])
        try_update(a_sg2, "sg_round2")

    # Alternative lengths
    print(f"\n=== Alternative Lengths (best={best_score:.7f}) ===")
    alt_lengths = []
    for nn in [n_best - 80, n_best - 40, n_best + 40, n_best + 80, n_best + 140, n_best + 200]:
        if 400 <= nn <= 2000 and nn != n_best:
            alt_lengths.append(nn)
    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alt = min(rem * 0.22, 200.0)
        if t_alt < 40:
            break
        print(f"\n  n={alt_n}, budget={t_alt:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        a_alt, sc_alt = run_pipeline(init_alt, t_alt, solver, f"n{alt_n}")
        try_update(a_alt, f"alt_n{alt_n}")

    # Final polish rounds
    print(f"\n=== Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 100.0)
        a_fin, _ = subgradient_polish(best_seq, t_fin, n_active_fracs=[0.03, 0.05, 0.1])
        try_update(a_fin, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin2 = min(rem * 0.25, 60.0)
        a_fin2 = lbgfs_smoothed(best_seq, t_fin2, T_start=30.0, T_end=0.05, maxiter=6000)
        try_update(a_fin2, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin3 = min(rem * 0.30, 50.0)
        a_fin3, _ = local_mass_transfer(best_seq, t_fin3, n_steps=40000)
        try_update(a_fin3, "final_local")

    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_fp = min(rem * 0.40, 60.0)
        a_fp, _ = alternating_lp_nesterov(best_seq, t_fp, solver, max_iters=20000)
        try_update(a_fp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
