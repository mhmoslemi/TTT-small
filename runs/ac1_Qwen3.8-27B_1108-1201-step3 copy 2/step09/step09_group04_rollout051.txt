    """Compute subgradient of max(convolve(a,a)) and the max value."""
    c = fast_convolve(a)
    max_val = c.max()

    i_end = min(n, k_max + 1)

    hi = k_max - i_start + 1

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        else:
            a = y.copy()

        c_a = fast_convolve(a)
        f_a = float(c_a.max())
        if f_a < best_f:
            best_f = f_a
            best_a = a.copy()
            stall = 0
        c_y2 = fast_convolve(y)
        f_y = float(c_y2.max())
        if f_y < best_f:
            best_f = f_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 2000 == 0:
            if best_f > prev_best - 1e-10:
                stall += 1
            else:
                prev_best = best_f
                stall = 0
            if stall >= 5:
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall = 0
                prev_best = best_f
            elapsed = time.monotonic() - t0
            if (it + 1) % 10000 == 0:
                print(f"      ActiveNesterov iter={it+1} ({elapsed:.0f}s): best={best_f:.8f}")

    return best_a, best_f


def p_norm_obj_z(z, p):
    """Compute 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_jac_z(z, p):
    """Gradient of p_norm_obj_z w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm schedule in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        res = minimize(
            fun=lambda z_val: p_norm_obj_z(z_val, stage_p),
            x0=z,
            jac=lambda z_val: p_norm_jac_z(z_val, stage_p),
            method='L-BFGS-B',
            options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': iters_per_stage * 100, 'disp': False}
        )
        z = res.x
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * float(fast_convolve(a_cur).max())
        elapsed = time.monotonic() - t0
        print(f"      p-norm p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def local_refine(a0, time_limit, n_max_steps=50000):
    """Coordinate descent + peak shaving local refinement."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    best_a = a.copy()
    c = fast_convolve(a)
    best_f = float(c.max())
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_idx = np.argsort(b_work)[::-1][:3]
        peak_val = b_work[peak_idx[0]]
        threshold = peak_val * 0.95
        near_peaks = [int(k) for k in range(len(b_work)) if b_work[k] >= threshold][:6]

        improved = False
        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                contrib[i] += a[i] * a[k - i]
        top_coords = np.argsort(contrib)[::-1][:10]

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-16:
                    continue
                a_t = a.copy()
                a_t[i] = max(0.0, a[i] - diff)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_t[mask].sum()
                if others < 1e-15:
                    continue
                a_t[mask] += diff * a_t[mask] / others
                st = a_t.sum()
                if st < 0.01:
                    continue
                a_t = a_t / st
                f_t = float(fast_convolve(a_t).max())
                if f_t < best_f - 1e-15:
                    a = a_t
                    best_f = f_t
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            peak_k = int(peak_idx[0])
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = range(js, je + 1)
            pairs = sorted(idx, key=lambda i: a[i] * a[peak_k - i], reverse=True)[:5]
            for i in pairs:
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.88]:
                    for f2 in [0.93, 0.88]:
                        a_t = a.copy()
                        a_t[i] *= f1
                        a_t[j] *= f2
                        a_t = np.maximum(a_t, 0)
                        st = a_t.sum()
                        if st < 0.01:
                            continue
                        a_t = a_t / st
                        f_t = float(fast_convolve(a_t).max())
                        if f_t < best_f - 1e-15:
                            a = a_t
                            best_f = f_t
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve >= 50:
                if step % 200 == 0:
                    rng = np.random.default_rng(step + 1)
                    a_t = a + rng.standard_normal(n) * 0.02
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st > 0.01:
                        a_t = a_t / st
                        f_t = float(fast_convolve(a_t).max())
                        if f_t < best_f:
                            a = a_t
                            best_f = f_t
                            best_a = a.copy()
                            no_improve = 0
                if no_improve >= 200:
                    break
        else:
            no_improve = 0

    return best_a, best_f


def interpolate_seq(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0)
    ss = f_i.sum()
    if ss > 1e-15:
        f_i = f_i / ss
    return f_i


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best, a, label):
    """Try updating best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best['score']:
        best['score'] = sc
        best['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: subgradient, active-set Nesterov LP,
    p-norm L-BFGS, length exploration, and local refinement.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best['seq'] = parent.copy()
            best['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        x = np.linspace(-1, 1, n_init)
        parent = np.exp(-x ** 2 * 3.0)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best['seq'] = parent.copy()
        best['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best['score']:.6f}")

    n_best = len(best['seq'])
    print(f"\nStarting: n={n_best}, score={best['score']:.6f}")

    # === Phase 1: Subgradient (fast, no LP) ===
    print(f"\n=== Phase 1: Subgradient (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_optimize(best['seq'], t_alloc, n_max_iters=500000)
        try_update(best, a_sg, "subgrad_1")

    # === Phase 2: Active-set Nesterov LP ===
    print(f"\n=== Phase 2: Active-Set Nesterov LP (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, m=15")
        a_lp, _ = active_set_nesterov(best['seq'], t_alloc, m_active=15, n_max_iters=80000)
        try_update(best, a_lp, "nesterov_1")

    # === Phase 3: p-norm L-BFGS ===
    print(f"\n=== Phase 3: p-norm L-BFGS (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, p: 10->500")
        a_pn = p_norm_lbfgs(best['seq'], t_alloc, p_start=10.0, p_end=500.0, n_stages=8)
        try_update(best, a_pn, "p_norm_1")

    # === Phase 4: Subgradient round 2 (from improved start) ===
    print(f"\n=== Phase 4: Subgradient r2 (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg2, _ = subgradient_optimize(best['seq'], t_alloc, n_max_iters=300000)
        try_update(best, a_sg2, "subgrad_2")

    # === Phase 5: Active-set Nesterov with larger active set ===
    print(f"\n=== Phase 5: Nesterov LP m=25 (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, m=25")
        a_lp2, _ = active_set_nesterov(best['seq'], t_alloc, m_active=25, n_max_iters=40000)
        try_update(best, a_lp2, "nesterov_2")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best['score']:.6f}) ===")
    candidate_lengths = []
    for delta in [-64, -32, 32, 64, 128]:
        alt_n = n_best + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.append(alt_n)
    # Add some round numbers
    for rn in [768, 896, 1024, 1152, 1280, 1400]:
        if 512 <= rn <= 2048 and rn not in candidate_lengths:
            candidate_lengths.append(rn)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 120.0)
        if t_total < 30:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_seq(best['seq'], alt_n)

        # Quick subgradient
        t_sub = t_total * 0.4
        a_alt, _ = subgradient_optimize(init_alt, t_sub, n_max_iters=100000)
        improved = try_update(best, a_alt, f"subgrad_n{alt_n}")

        # If improved, do p-norm
        if improved:
            rem = deadline - time.monotonic()
            if rem > 20:
                t_pn = min(rem * 0.08, 60.0)
                a_alt2 = p_norm_lbfgs(best['seq'], t_pn, p_start=30.0, p_end=300.0, n_stages=4)
                try_update(best, a_alt2, f"pn_n{alt_n}")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbation (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for noise_scale in [0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            n_cur = len(best['seq'])
            perturbed = best['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            st = perturbed.sum()
            if st < 0.01:
                continue
            perturbed = perturbed / st
            t_sub = min(rem * 0.25, 80.0)
            a_pert, _ = subgradient_optimize(perturbed, t_sub, n_max_iters=100000)
            try_update(best, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final p-norm with very high p ===
    print(f"\n=== Phase 8: High p-norm polish (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = p_norm_lbfgs(best['seq'], t_fin, p_start=100.0, p_end=1000.0, n_stages=5)
        try_update(best, a_fin, "high_p")

    # === Phase 9: Local refinement ===
    print(f"\n=== Phase 9: Local Refinement (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.30, 60.0)
        a_lr, _ = local_refine(best['seq'], t_lr, n_max_steps=30000)
        try_update(best, a_lr, "local_1")

    # === Phase 10: Final Nesterov polish ===
    print(f"\n=== Phase 10: Final Nesterov (best={best['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_final = min(rem * 0.40, 40.0)
        a_fi, _ = active_set_nesterov(best['seq'], t_final, m_active=20, n_max_iters=20000)
        try_update(best, a_fi, "final_nesterov")

    # Final verification
    if best['seq'] is not None:
        fs = evaluate_safe(best['seq'])
        if np.isfinite(fs):
            best['score'] = fs

    elapsed_total = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best['score']:.6f} (n={len(best['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best['seq']]
