def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def nesterov_mirror_descent(a0, time_limit, lr_init=0.005, top_k=50, beta_init=50.0):
    """
    Nesterov-accelerated mirror descent with smooth max approximation.
    Uses logsumexp as smooth surrogate for max, with adaptive beta.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0  # momentum parameter

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov step: y = (1+gamma)/gamma * a - 1/gamma * a_prev
        # Simplified: y = a + momentum * (a - a_prev)
        if it == 1:
            y = a.copy()
        else:
            y = (1.0 + gamma) * a - gamma * y

        # Project y to simplex for valid evaluation
        y_proj = simplex_project(y)

        # Compute smooth gradient using logsumexp
        conv_y = fast_convolve(y_proj)
        # Smooth the max
        max_conv = conv_y.max()
        scaled = beta * (conv_y - max_conv)
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient of smooth max w.r.t. a
        grad = np.zeros(n)
        peak_indices = np.where(weights > 1e-10)[0]
        for k in peak_indices:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent update: a_new = a * exp(-lr * grad)
        # Apply to a (not y) for stability
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and beta
        if no_improve > 100:
            lr *= 0.95
            beta *= 1.5
            no_improve = 0
            if lr < 1e-8:
                break
        elif it % 50 == 0:
            # Gradually increase beta (sharper approximation)
            beta *= 1.02

        # Update momentum (polyak step size)
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

def peak_targeted_perturb(a0, rng, strength=0.05):
    """
    Structured perturbation that specifically targets the peak regions
    of the convolution. Reduces values that contribute to peaks and
    increases values that don't.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    # Identify top contributing positions to the max peaks
    peak_threshold = M * 0.95
    peak_positions = np.where(conv >= peak_threshold)[0]

    # For each peak at k, the pairs (i, k-i) contribute: a[i]*a[k-i]
    # We want to reduce high-product pairs
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5  # Less pressure on the other side

    # Normalize pressure
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Apply perturbation: reduce high-pressure coords, increase low-pressure
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.3):
    """
    Randomly zero out a fraction of coordinates and redistribute mass.
    This helps escape from local optima by fundamentally changing the support.
    """
    n = len(a0)
    a = a0.copy()

    # Choose which coords to keep
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n

    a[~keep] = 0
    # Add some noise to kept elements
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.2)
    a = np.maximum(a, 0)

    # Add small random mass to some zeroed positions
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.1))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5

    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Fictitious play with Nesterov-style averaging for faster convergence.
    Uses the "Optimistic" variant: the average tracks a sequent rather than the iterates.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    # Nesterov sequent
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best responses
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Nesterov sequent update (Aginag-Tseng style)
        gamma_it = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma_it) * a_sequent + gamma_it * a
        b_sequent = (1.0 - gamma_it) * b_sequent + gamma_it * b

        # Evaluate candidates
        for cand in [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization (a_i = a_{n-1-i})."""
    half = n // 2
    # Random profile for half the sequence
    t = np.linspace(0, 1, half)
    # Use a superposition of cosines for the half-profile
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    # Extend to full symmetric sequence
    if n % 2 == 0:

    else:

        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with bimodal distribution (mass at two locations)."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_flat_with_ridges(n, rng):
    """Flat base with periodic ridges (tries to create flat convolution)."""
    period = int(n / rng.integers(3, 8))
    t = np.arange(n)
    a = np.ones(n) * 0.5
    ripple_amp = rng.uniform(0.1, 0.3)
    a += ripple_amp * np.cos(2 * np.pi * t / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
