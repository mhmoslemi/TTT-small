import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

# LP solver selection
_lp_solver = None
def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s

import numpy as np

import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:

    T = build_conv_matrix(a_fixed)

    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=1000000,
                       use_adaptive_lr=True, lr_scale=1.0):
    """
    Nesterov dual averaging LP refinement with adaptive step sizes.
    Uses the LP optimal value to scale the step.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    mean_b = np.zeros(n)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Adaptive step size: use M_b to scale
        if use_adaptive_lr:
            # The LP value M_b gives us information about the gap
            # Use a step that balances exploration and exploitation
            alpha = 2.0 * lr_scale / (it + 2)
        else:
            alpha = 1.0 / np.sqrt(it + 1)
        
        # Update running average
        mean_b = (it * mean_b + b_new) / (it + 1)
        
        # Nesterov-style update for y (dual)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        
        # Dual averaging update for a (primal)
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a_new)
        
        # Also track the running average as a candidate
        mean_b = project_simplex(mean_b)
        
        # Check scores
        score_a = score_normalized(a)
        score_y = score_normalized(y)
        score_mean = score_normalized(mean_b)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        if score_mean < best_score:
            best_score = score_mean
            best_a = mean_b.copy()
            stall_count = 0
        
        # Stall detection and restart
        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                rng = np.random.default_rng(it * 7 + 13)
                y = y + 0.03 * rng.standard_normal(n)
                y = project_simplex(y)
                a = a + 0.02 * rng.standard_normal(n)
                a = project_simplex(a)
                stall_count = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_b={M_b:.6f}")

def mirror_descent_lp(a0, time_limit, solver=None, n_max_iters=500000):
    """
    Mirror descent (multiplicative weights) with LP best response.
    Uses exponential tilting instead of Euclidean projection.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Log-parameters for multiplicative updates
    log_a = np.log(a + 1e-15)
    
    lr = 0.5
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        
        # Multiplicative weights update: a_j *= exp(-lr * grad_j)
        # grad_j = 2 * a_{k*-j} where k* = argmax conv(a,a)
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        # grad_j = 2 * a_{k_star - j} if 0 <= k_star - j < n else 0
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
                if j == k_star - j:
                    grad[j] = a[idx]  # diagonal correction
        
        # Update log-parameters
        log_a = log_a - lr * grad / (2.0 * n)
        # Normalize (softmax)
        log_a -= np.max(log_a)
        new_a = np.exp(log_a)
        s = new_a.sum()
        if s > 1e-15:
            new_a = new_a / s
            a = new_a
        
        score = score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Decay learning rate
        if it > 0 and it % 500 == 0:
            lr *= 0.95
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_steps=100000):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce the peak.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]
        
        # Find coordinates contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        if contribs.max() < 1e-15:
            break
        
        # Try reducing top contributors
        top_k = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_k:
            i = idx[ci]
            j = k_peak - i
            if i >= n or j >= n or i == j:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Try several reduction factors
            for f in [0.99, 0.98, 0.97, 0.95, 0.93]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                a_trial = project_simplex(a_trial)
                s = score_normalized(a_trial)
                if s < best_score - 1e-12:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try single coordinate reduction
            for ci in top_k[:5]:
                i = idx[ci]
                j = k_peak - i
                for f in [0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial = project_simplex(a_trial)
                    s = score_normalized(a_trial)
                    if s < best_score - 1e-12:
                        a = a_trial
                        best_score = s
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try adding to low coordinates (redistribution)
            low_coords = np.argsort(a)[:5]
            for li in low_coords:
                if a[li] > 0.5 / n:
                    continue
                for ci in top_k[:3]:
                    i = idx[ci]
                    if i == li:
                        continue
                    for move in [0.001, 0.002, 0.005]:
                        a_trial = a.copy()
                        a_trial[i] -= move * a[i]
                        a_trial[li] += move * a[i]
                        if a_trial[i] < 0 or a_trial[li] < 0:
                            continue
                        a_trial = project_simplex(a_trial)
                        s = score_normalized(a_trial)
                        if s < best_score - 1e-12:
                            a = a_trial
                            best_score = s
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # No improvement in this step - try a random perturbation
            if step % 200 == 199:
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = project_simplex(a_trial)
                s = score_normalized(a_trial)
                if s < best_score:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                else:
                    # If we're truly stuck, break
                    if step > 500:
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """
    Create initialization based on spectral considerations.
    Try to make the DFT magnitude as flat as possible.
    """
    # Start with a random phase in frequency domain
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Generate a target flat spectrum with random phases
    target_mag = np.ones(fft_size) / np.sqrt(fft_size)
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    
    # Enforce Hermitian symmetry
    spectrum = target_mag * np.exp(1j * phases)
    spectrum[fft_size // 2] = target_mag[fft_size // 2]  # Nyquist real
    for k in range(1, fft_size // 2):
        spectrum[fft_size - k] = np.conj(spectrum[k])
    
    # Inverse FFT to get time domain
    a = np.real(irfft(spectrum, fft_size))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback: use a smoother initialization
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        return a
    a = a / s
    return a


def smooth_init(n, rng):
    """Create a smooth initialization with random modulation."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 2*np.pi))
    elif choice == 1:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
    elif choice == 2:
        a = 1.0 + 0.5 * (2 * x - 1)
    elif choice == 3:
        a = np.exp(-2 * (x - 0.3)**2) + 0.5 * np.exp(-2 * (x - 0.7)**2) + 0.2
    elif choice == 4:
        a = 1.0 + 0.3 * np.sin(2*np.pi*x) + 0.2 * np.sin(4*np.pi*x + 0.5)
    else:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def dual_based_iteration(a0, time_limit, solver=None, n_max_iters=500000):
    """
    Alternative: track the dual variable (distribution over peaks) explicitly.
    This implements the primal-dual method more directly.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Dual variable: distribution over the 2n-1 convolution positions
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    mean_a = a.copy()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        b = fast_convolve(a)
        
        # Update dual: p_k proportional to b_k (tilted)
        lr_dual = 5.0 / (it + 10)
        p = p * np.exp(lr_dual * b / b.max())
        p = p / p.sum()
        
        # Primal update: a_j proportional to sum_k p_k * T[k,j]
        # T[k,j] = a_{k-j} if 0 <= k-j < n (this is the coefficient of b_j in conv(a,b)_k)
        # Actually, we want: min_b sum_k p_k * conv(a,b)_k = min_b sum_k p_k * sum_j a_{k-j} b_j
        # = min_b sum_j (sum_k p_k * a_{k-j}) b_j
        # The gradient w.r.t. b_j is sum_k p_k * a_{k-j}
        # To minimize, set b_j proportional to exp(-lr * gradient_j)
        
        # Compute gradient vector
        grad_b = np.zeros(n)
        for j in range(n):
            # grad_b[j] = sum_k p[k] * a[k-j] for valid k
            k_min = max(0, j)
            k_max = min(2 * n - 2, j + n - 1)
            if k_min <= k_max:
                grad_b[j] = np.dot(p[k_min:k_max+1], a[k_max-j:k_min-j-1:-1] if k_max > k_min else [a[j]])
        
        # Actually let me just use FFT for this
        # grad_b[j] = sum_k p_k * a_{k-j} = (p correlate a)[j]
        fft_size = 1 << (2 * n - 1).bit_length()
        P = rfft(p, fft_size)
        A = rfft(a, fft_size)
        # Correlation: (p * a_flipped) in frequency domain
        a_flip = a[::-1]
        AF = rfft(a_flip, fft_size)
        grad_b = irfft(P * AF, fft_size)[:n]
        
        # Multiplicative update for a (mirror descent)
        lr_primal = 2.0 / (it + 10)
        log_update = -lr_primal * grad_b / max(grad_b.max(), 1e-15)
        log_update -= log_update.max()
        a_new = np.exp(log_update)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            a = project_simplex(a_new)
        
        # Running average
        mean_a = (it * mean_a + a) / (it + 1)
        mean_a = project_simplex(mean_a)
        
        # Check scores
        score_a = score_normalized(a)
        score_mean = score_normalized(mean_a)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_mean < best_score:
            best_score = score_mean
            best_a = mean_a.copy()
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Dual iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with LP best response, mirror descent,
    multi-length exploration, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = spectral_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Generated init: n={n_init}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: Main LP-based Nesterov refinement on parent ===
    print(f"\n=== Phase 1: LP Nesterov (n={n_best}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                       n_max_iters=2000000, use_adaptive_lr=True, lr_scale=1.0)
        try_update(a_out, "lp_main")
    
    # === Phase 2: LP with different step size ===
    print(f"\n=== Phase 2: LP Nesterov v2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                       n_max_iters=2000000, use_adaptive_lr=True, lr_scale=0.7)
        try_update(a_out, "lp_v2")
    
    # === Phase 3: Mirror descent variant ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, _ = mirror_descent_lp(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_out, "mirror")
    
    # === Phase 4: Alternative lengths ===
    print(f"\n=== Phase 4: Alternative Lengths (best={best_score:.8f}) ===")
    alt_lengths = [n_best - 100, n_best - 50, n_best + 50, n_best + 100,
                   n_best + 200, n_best - 200]
    alt_lengths = [x for x in alt_lengths if 400 <= x <= 2000]
    alt_lengths = sorted(set(alt_lengths))
    
    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.15, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_lp_refine(init_alt, t_total * 0.7, solver=solver,
                                       n_max_iters=500000, use_adaptive_lr=True, lr_scale=1.0)
        try_update(a_alt, f"lp_n{alt_n}")
        
        rem = deadline - time.monotonic()
        if rem > 20 and t_total * 0.25 > 10:
            a_alt2, _ = coordinate_descent_polish(a_alt, t_total * 0.25, n_steps=50000)
            try_update(a_alt2, f"cd_n{alt_n}")
    
    # === Phase 5: Different initializations at best length ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_starts = min(3, max(1, int(rem / 100)))
    for si in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.2, 150.0)
        if t_sub < 30:
            break
        a_init = spectral_init(n_best, rng)
        # Quick warm-start with a few steps from best_seq
        a_init = 0.5 * a_init + 0.5 * best_seq
        a_init = project_simplex(a_init)
        print(f"  Start {si+1}, budget={t_sub:.0f}s")
        a_sub, _ = nesterov_lp_refine(a_init, t_sub, solver=solver,
                                       n_max_iters=200000, use_adaptive_lr=True, lr_scale=1.0)
        try_update(a_sub, f"multi_{si}")
    
    # === Phase 6: Final LP polish on best ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 120.0)
        print(f"  Budget: {t_fin:.0f}s")
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver,
                                       n_max_iters=500000, use_adaptive_lr=True, lr_scale=0.8)
        try_update(a_fin, "final_lp")
    
    # === Phase 7: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.30, 90.0)
        print(f"\n=== Phase 7: CD Polish (budget={t_cd:.0f}s) ===")
        a_cd, _ = coordinate_descent_polish(best_seq, t_cd, n_steps=200000)
        try_update(a_cd, "cd_final")
    
    # === Phase 8: One more LP round if time remains ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_extra = min(rem * 0.5, 60.0)
        print(f"\n=== Phase 8: Extra LP (budget={t_extra:.0f}s) ===")
        a_ex, _ = nesterov_lp_refine(best_seq, t_extra, solver=solver,
                                       n_max_iters=200000, use_adaptive_lr=True, lr_scale=1.2)
        try_update(a_ex, "extra_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
