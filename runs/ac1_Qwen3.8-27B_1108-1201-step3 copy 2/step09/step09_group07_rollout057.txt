def solve_best_response_lp(a, solver):
    """Given a (sum=1, nonneg), find b minimizing max_k conv(a,b)_k subject to b>=0, sum(b)=1."""
    n = len(a)
    # conv(a,b)_k = sum_i a_i * b_{k-i} for valid i, k-i
    # This is linear in b: conv(a,b)_k = sum_j M[k,j] * b_j
    # where M[k,j] = a_{k-j} if 0 <= k-j < n, else 0
    
    # Build the constraint matrix efficiently
    # For each k (0 to 2n-2), the coefficients for b_j (j=0 to n-1) are:
    # M[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Use scipy linprog
    from scipy.optimize import linprog
    # Variables: b_0, ..., b_{n-1}, t (t = max constraint value)
    # Minimize: t (or -sum(b) to maximize sum while keeping t fixed... 
    # Actually: minimize t subject to M[k,:] @ b <= t, sum(b) = 1, b >= 0, t >= 0
    
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: M[k,:] for b part, -1 for t part
    # M[k,j] - t <= 0  =>  M[k,:] @ b - t <= 0
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
        A_ub[k, -1] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    t_val = res.x[-1]
    return b_val, t_val
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-based best-response with Nesterov acceleration
2. Multi-length exploration
3. Coordinate descent polishing
Key: scipy HiGHS solver for fast LP, sparse constraints, adaptive stepping.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    from scipy.fft import rfft, irfft
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix where M @ b = convolve(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    M = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return M


def solve_best_response_lp(a, method='highs'):
    """Solve the best-response LP: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0."""
    n = len(a)
    M = build_sparse_conv_matrix(a)
    
    # Add t variable: minimize t
    # Constraints: M @ b - t <= 0  =>  [M | -I] @ [b; t] <= 0
    # sum(b) = 1, b >= 0, t >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: M @ b - t <= 0
    A_ub = sp.hstack([M, sp.csr_matrix(-np.eye(2 * n - 1))], format='csc')
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method=method)
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    t_val = res.x[-1]
    return np.maximum(b_val, 0), float(t_val)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-based best-response with Nesterov acceleration (scipy HiGHS)
2. Subgradient method with proximal correction as complement

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * float(b.max())


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_pattern(n):
    """Precompute (row, col) indices for the (2n-1)xn convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_values(a, rows, cols):
    """Get values for the convolution matrix given the pattern."""
    return a[rows - cols]


def solve_best_response_lp(a, rows, cols, method='highs'):
    """Solve: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0. Returns (b, t)."""
    n = len(a)
    vals = a[rows - cols]
    M = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    neg_I = sp.csr_matrix(-np.eye(2 * n - 1))
    A_ub = sp.hstack([M, neg_I], format='csc')
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method, options={'presolve': True})
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub.toarray(), b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return None, None
    
    if res.status != 0:
        return None, None
    b_val = np.maximum(res.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[-1])

def subgradient_optimize(a0, time_limit, lr_init=0.1, decay=0.01):
    """Subgradient method with projection for the max auto-correlation."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    a_avg = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    y = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        max_val = b.max()
        peaks = np.where(b >= max_val - 1e-12 * max_val)[0]
        
        grad = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(peaks)
        
        t_curr = it + 1
        t_prev = it
        beta = (np.sqrt(8 * t_prev * t_prev + 1) - 1) / 2.0 if it > 0 else 0.0
        y = a + beta * (a - y) if it > 0 else a.copy()
        
        lr = lr_init / (1.0 + decay * it)
        a_new = project_simplex(y - lr * grad)
        y = a_new
        
        alpha = 2.0 / (it + 2)
        a_avg = (1.0 - alpha) * a_avg + alpha * a_new
        a = a_avg.copy()
        it += 1
        
        if it % 100 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    a = np.maximum(a, 1e-8)
    a = a / a.sum()
