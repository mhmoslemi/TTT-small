    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solve_kwargs = {}
    if solver == cp.ECOS:
        solve_kwargs['max_iters'] = 20000
        solve_kwargs['abstol'] = 1e-8
        solve_kwargs['reltol'] = 1e-8
    elif solver == cp.SCS:
        solve_kwargs['max_iters'] = 50000
        solve_kwargs['eps'] = 1e-9

    try:
        prob.solve(solver=solver, **solve_kwargs)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def grad_logsumexp(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k(a))) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j a[j] * w[j+i] for valid j
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def lp_best_response_loop(a0, time_budget, solver=None):
    """
    Iterate: solve LP best response, update a via adaptive averaging.
    Uses a "heavy ball" variant with best-iterate tracking.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    a_mom = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_budget:
            break

        b_new, M_val = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # Adaptive step: start aggressive, decay over time
        alpha = 2.0 / (it + 2.0)
        # Heavy ball momentum
        a_mom = a_mom + alpha * (b_new - a)
        a = project_simplex(a_mom)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            # Perturb to escape local optimum
            rng = np.random.default_rng(it * 12345)
            pert = 0.05 * rng.standard_normal(n)
            a = project_simplex(a + pert)
            a_mom = a.copy()
            stall = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")

    return best_a, best_score


def smoothed_gradient_loop(a0, time_budget, T_start=5.0, T_end=500.0,
                           lr_init=0.1, n_stages=15):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = project_simplex(np.maximum(a0, 0))
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_budget:
            break
        T = temps[stage]
        steps = 3000
        for step in range(steps):
            if time.monotonic() - t0 > time_budget:
                break

            global_step = stage * steps + step
            lr = lr_init / (1.0 + 0.003 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_logsumexp(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def coordinate_descent_polish(a0, time_budget, max_steps=50000):
    """
    Greedy coordinate descent: iteratively reduce coordinates that
    contribute most to the peak of the convolution.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for step in range(max_steps):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        # Find positions near the peak
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b >= threshold)[0]
        # Exclude first and last (edges are less impactful to reduce)
        interior_peaks = [k for k in near_peaks if 0 < k < len(b) - 1]
        if not interior_peaks:
            interior_peaks = list(near_peaks)

        # Compute contribution of each coordinate to the peaks
        contrib = np.zeros(n)
        for k in interior_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contrib[j] += a[j] * a[k - j]

        # Try reducing top-contributing coordinates
        top_coords = np.argsort(contrib)[::-1][:15]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.93, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try 2-coordinate moves
            if len(top_coords) >= 2:
                for i in top_coords[:5]:
                    for j in top_coords[:5]:
                        if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                            continue
                        for f1 in [0.95, 0.90]:
                            for f2 in [0.95, 0.90]:
                                a_trial = a.copy()
                                a_trial[i] *= f1
                                a_trial[j] *= f2
                                a_trial = project_simplex(a_trial)
                                sc = score_normalized(a_trial)
                                if sc < best_score - 1e-15:
                                    a = a_trial
                                    best_score = sc
                                    best_a = a.copy()
                                    improved = True
                                    break
                            if improved:
                                break
                        if improved:
                            break
                    if improved:
                        break

        if not improved:
            # Add small random perturbation
            n_pert = max(1, n // 100)
            idx = rng.choice(n, size=n_pert, replace=False)
            for i in idx:
                a[i] *= (1.0 + 0.01 * rng.standard_normal())
            a = project_simplex(a)

            # If no improvement for too long, break
            if step > 100 and not improved:
                # Try a larger perturbation
                a = project_simplex(a + 0.02 * rng.standard_normal(n))

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_variety_init(n, variant, rng):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif variant == 1:
        a = 1.0 + 0.7 * (2 * x - 1)
    elif variant == 2:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif variant == 3:
        a = 1.0 + 0.4 * np.cos(4 * np.pi * x)
    elif variant == 4:
        a = 1.0 + 0.6 * np.sin(2 * np.pi * x + 1.0)
    elif variant == 5:
        a = np.ones(n)
        a[:n//4] *= 1.3
        a[3*n//4:] *= 1.3
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP best response, smoothed gradients,
    coordinate descent, and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_variety_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP Best Response (40% budget) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 600.0)
    print(f"\n=== Phase 1: LP Best Response (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_lp, _ = lp_best_response_loop(best_seq, t1, solver=solver)
        try_update(a_lp, "lp_main")
    else:
        print("  Skipping (insufficient time)")

    # === Phase 2: Smoothed Gradient (15% budget) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 250.0)
    print(f"\n=== Phase 2: Smoothed Gradient (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_sg, _ = smoothed_gradient_loop(best_seq, t2, T_start=5.0, T_end=400.0,
                                          lr_init=0.08, n_stages=12)
        try_update(a_sg, "smoothed_grad")

    # === Phase 3: Second LP round on improved solution (12% budget) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 3: LP Round 2 (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_lp2, _ = lp_best_response_loop(best_seq, t3, solver=solver)
        try_update(a_lp2, "lp_round2")

    # === Phase 4: Coordinate Descent Polish (10% budget) ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    print(f"\n=== Phase 4: Coordinate Descent (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_cd, _ = coordinate_descent_polish(best_seq, t4, max_steps=80000)
        try_update(a_cd, "coord_descent")

    # === Phase 5: Multi-start with different inits ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-Start (remaining={rem:.0f}s) ===")
    if rem > 120:
        variants = [1, 2, 3, 4, 5]
        for v in variants[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 30:
                break
            print(f"  Variant {v}, budget={t_sub:.0f}s")
            a_init = make_variety_init(n_best, v, rng)
            # Quick LP then polish
            a_sub, _ = lp_best_response_loop(a_init, t_sub * 0.7, solver=solver)
            try_update(a_sub, f"multi_v{v}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_sub2, _ = coordinate_descent_polish(a_sub, min(rem * 0.4, 30.0), max_steps=10000)
                try_update(a_sub2, f"multi_v{v}_polish")

    # === Phase 6: Alternative lengths ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Alternative Lengths (remaining={rem:.0f}s) ===")
    if rem > 100:
        deltas = [-5, -3, -1, 1, 3, 5, 8, 12]
        tried = 0
        for d in deltas:
            rem = deadline - time.monotonic()
            if rem < 40 or tried >= 3:
                break
            alt_n = n_best + d
            if 256 <= alt_n <= 4096:
                t_alt = min(rem * 0.22, 80.0)
                if t_alt < 20:
                    break
                print(f"  n={alt_n} (delta={d}), budget={t_alt:.0f}s")
                init_alt = interpolate_sequence(best_seq, alt_n)
                a_alt, _ = lp_best_response_loop(init_alt, t_alt, solver=solver)
                try_update(a_alt, f"n{alt_n}")
                tried += 1

    # === Phase 7: Final LP polish ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final Polish (remaining={rem:.0f}s) ===")
    if rem > 30:
        t_fin = min(rem * 0.4, 90.0)
        a_fin, _ = lp_best_response_loop(best_seq, t_fin, solver=solver)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final coordinate shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.5, 40.0)
        a_ps, _ = coordinate_descent_polish(best_seq, t_ps, max_steps=20000)
        try_update(a_ps, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    else:
        print("ERROR: No valid sequence found!")

    return [float(x) for x in best_seq]
