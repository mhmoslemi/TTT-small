```python
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def make_symmetric(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0

def evaluate_local(a):
    """Evaluate using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    """
    Multiplicative weights (mirror descent) for minimizing max(conv(p,p))
    over the probability simplex.
    """
    n = len(a0)
    p = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    p = p / p.sum()
    
    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    
    eta = 1.0  # learning rate
    best_eta = eta
    patience = 0
    max_patience = 500
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(p)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_p = p.copy()
            patience = 0
        else:
            patience += 1

            eta *= 0.7
            patience = 0
            if eta < 1e-6:
```

- If `f(x_new) > f(prev) + <g, x_new - prev> + eta/2 * ||x_new - prev||^2`, halve eta

Actually, for subgradient methods, the standard approach is:
- `x_{t+1} = proj_C(x_t - eta_t * g_t)`
- `eta_t = c / sqrt(t)` or `eta_t = c / t`

With Nesterov acceleration:
- `y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})`
- `x_{t+1} = proj_C(y_t - eta * g(y_t))`

The convergence rate is O(1/t) for the subgradient method with Nesterov acceleration (under appropriate conditions).

For our problem, the "Lipschitz constant" of the subgradient is bounded by:
`||g(p1) - g(p2)|| <= L * ||p1 - p2||`

The subgradient `g_k(p) = 2 * shift(p, k)` has norm `2*||p||_2 <= 2*||p||_1 = 2` (since sum(p) = 1 and p >= 0).

So the subgradients are bounded in norm by 2. This gives us a step size of `eta_t = R / (L * sqrt(t))` where R is the diameter of the feasible set.

For the simplex, R = 1 (diameter is at most 1 in L1, but in L2 it's at most 1/sqrt(2) for the projection).

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1:
        # Just clip and renormalize (not exact but fast)
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 0:
            return v / s
        return np.ones(n) / n
    
    # Exact projection using bisection
    mu = 0.0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
1. FISTA-accelerated subgradient descent (fast first-order method)

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    t0 = time.monotonic()
    base_eta = 1.0

    for t in range(1, n_steps + 1):

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            sy = 1.0

def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    a_start = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a_start = a_start / a_start.sum()
    z = np.log(a_start)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        t_stage = stage_T

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            log_sum_exp = max_b + np.log(exp_b.sum())
            return (2.0 * n / t_stage) * log_sum_exp

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, int(k) - n + 1)
                je = min(n - 1, int(k))
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[int(k) - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_final = softmax_from_z(z)
    return a_final

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def coordinate_descent_polish(a0, time_limit, n_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Pick a random top contributor to modify
        n_top = min(5, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:n_top]]
        pos = top_positions[rng.integers(0, len(top_positions))]
        
        # Try shrinking this position
        improved = False
        for factor in [0.99, 0.98, 0.97, 0.95]:
            a_trial = a.copy()
            a_trial[pos] *= factor
            diff = a[pos] - a_trial[pos]
            if diff < 1e-16:
                break
            mask = np.ones(n, dtype=bool)
            mask[pos] = False
            others = a_trial[mask]
            if others.sum() < 1e-15:
                break
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FISTA-accelerated subgradient descent with exact simplex projection (fast, many iterations)
2. Nesterov LP alternating with adaptive restarts (exact convex subproblem refinement)
3. Multi-stage L-BFGS with LogSumExp smoothing at aggressively decreasing temperatures
4. Stochastic coordinate descent polish
5. Multi-start exploration with diverse structured initializations and adaptive length
Key innovations: efficient FISTA with top-k peak subgradients, exact Laplacian-free
simplex projection, and temperature schedule tuned for the near-non-smooth regime.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

_lp_solver = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Exact projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    """Evaluate the objective using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng, idx):

    elif idx == 5:

    elif idx == 6:

    a = np.maximum(a, 0)

def fista_subgradient(p0, time_limit, n_steps, top_k):
    """FISTA-accelerated subgradient descent for minimizing max(auto-conv)."""
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 1e-300).copy()
    p = p / p.sum()
    z = p.copy()
    p_prev = p.copy()
    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    base_eta = 2.0
    no_improve = 0
    max_no_improve = 5000
    eta_multiplier = 1.0

    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(z)
        max_b = b.max()

        # Get top-k peaks for stable subgradient
        sorted_idx = np.argsort(b)[::-1][:top_k]
        peak_indices = sorted_idx[b[sorted_idx] >= max_b * 0.98]
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(b))])

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in peak_indices:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * z[k_int - idx]
        g /= len(peak_indices)

        # Adaptive step size with decay
        eta = base_eta * eta_multiplier / np.sqrt(t)
        eta = max(eta, 1e-10)

        # Gradient step + projection
        p_candidate = z - eta * g
        p_new = project_simplex(p_candidate)

        # Nesterov extrapolation
        momentum = (t - 1) / (t + 2)
        z = p_new + momentum * (p_new - p)
        z = np.maximum(z, 1e-300)
        z = z / z.sum()
        p = p_new

        score = 2.0 * n * fast_convolve(p).max()
        if score < best_score - 1e-15:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart mechanism
        if no_improve >= max_no_improve:
            p = best_p.copy()
            z = p.copy()
            eta_multiplier *= 0.5
            no_improve = 0
            if eta_multiplier < 0.001:
                break

        if t % 200000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA t={t} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.2e}")

def get_lp_solver():

    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP refinement with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(54321)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
