    A = rfft(a_padded)

    c = irfft(A * A, L)[:2*n-1]
    peak = c.max()

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def pnorm_obj_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def pnorm_grad_z(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def fl_spectral_flatten(a, alpha=0.15):
    """Fuchs-Levenberg style spectral flattening step."""
    n = len(a)
    L = 1 << (2 * n - 2).bit_length()
    if L < 2 * n - 1:
        L *= 2
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = rfft(a_pad)
    A_mag = np.abs(A)
    A_mag_sq = A_mag ** 2
    mean_mag = np.mean(A_mag_sq)
    if mean_mag < 1e-300:
        return a
    ratio = A_mag_sq / (A_mag_sq + mean_mag * 1e-10)
    target = np.ones_like(A_mag_sq)
    blend = (1 - alpha) * ratio + alpha * target
    A_mag_new = A_mag_sq * np.sqrt(blend / ratio)
    A_new = A_mag_new * np.exp(1j * np.angle(A))
    a_new = irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def pnorm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(zv):
            return pnorm_obj_z(zv, stage_p)

        def jac(zv):
            return pnorm_grad_z(zv, stage_p)

        max_iter = max(50, int(6000 / n_stages))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Periodic FL step every 2 stages
        if n_stages >= 4 and list(temps).index(stage_p) % 2 == 1:
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            a_fl = fl_spectral_flatten(a_cur, alpha=0.1)
            if compute_score(a_fl) < compute_score(a_cur):
                z = np.log(np.maximum(a_fl, 1e-300))
                z = z - z.max()

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def asymmetric_init(n, variant, rng):
    x = np.linspace(0, 1, n)
    if variant == 0:
        # Exponential decay
        a = np.exp(-3.0 * x)
    elif variant == 1:
        # Power law
        a = (1 - x) ** 2.5
    elif variant == 2:
        # Skewed Gaussian (shifted left)
        a = np.exp(-((x - 0.3) ** 2) / 0.08)
    elif variant == 3:
        # Step-like: large first half, small second half
        a = np.where(x < 0.5, 1.0, np.exp(-4 * (x - 0.5)))
    elif variant == 4:
        # Ramp
        a = 1.0 - x
    elif variant == 5:
        # Bimodal asymmetric
        a = np.exp(-((x - 0.25) ** 2) / 0.04) + 0.5 * np.exp(-((x - 0.75) ** 2) / 0.06)
    elif variant == 6:
        # Cosine window (symmetric but different shape)
        a = np.sin(np.pi * x / 2) ** 3
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    # Add small noise for diversity
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = asymmetric_init(n_init, 0, rng)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    def try_update(a, label):
        a = np.maximum(a, 0.0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_state['score']:
            best_state['score'] = sc
            best_state['seq'] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Fine length exploration from parent ===
    print(f"\n=== Phase 1: Fine Length Exploration (best={best_state['score']:.6f}) ===")
    # Try lengths around the parent length
    length_candidates = set()
    base = n_best
    for delta in range(-96, 97, 16):
        alt_n = base + delta
        if 640 <= alt_n <= 1400:
            length_candidates.add(alt_n)
    # Also some round numbers
    for rn in [768, 832, 896, 960, 1024, 1088, 1152, 1216, 1280]:
        if 640 <= rn <= 1400:
            length_candidates.add(rn)

    length_candidates = sorted(length_candidates)
    # Sort by proximity to current best
    length_candidates.sort(key=lambda x: abs(x - n_best))

    n_trial = 0
    max_trials = 7
    for alt_n in length_candidates[:max_trials]:
        rem = deadline - time.monotonic()
        if rem < 100 or n_trial >= max_trials:
            break
        rem = deadline - time.monotonic()
        t_total = min(rem / (max_trials - n_trial), 220.0)
        if t_total < 50:
            break
        n_trial += 1
        print(f"  Trial {n_trial}: n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_seq(best_state['seq'], alt_n)

        # Stage A: p-norm L-BFGS (medium p)
        t_a = t_total * 0.4
        a_cur = pnorm_lbfgs(init_alt, t_a, p_start=20.0, p_end=300.0, n_stages=6)
        try_update(a_cur, f"pn_{alt_n}_A")

        # Stage B: FL step + high p L-BFGS
        rem = deadline - time.monotonic()
        if rem > 30:
            t_b = min(rem * 0.3, t_total * 0.35)
            a_fl = fl_spectral_flatten(a_cur, alpha=0.12)
            a_cur2 = pnorm_lbfgs(a_fl, t_b, p_start=100.0, p_end=600.0, n_stages=5)
            try_update(a_cur2, f"pn_{alt_n}_B")
            # Also try the FL result directly
            try_update(a_fl, f"fl_{alt_n}")

        # Stage C: Local refinement
        rem = deadline - time.monotonic()
        if rem > 20:
            t_c = min(rem * 0.2, 60.0)
            a_loc, _ = local_refine(best_state['seq'], t_c, n_steps=15000)
            try_update(a_loc, f"local_{alt_n}")

    # === Phase 2: Asymmetric initializations ===
    print(f"\n=== Phase 2: Asymmetric Inits (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_state['seq'])
        variants_to_try = [0, 1, 2, 3, 4, 5]
        n_var = min(len(variants_to_try), int(rem / 80))
        for vi in range(n_var):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            variant = variants_to_try[vi]
            t_sub = min(rem * 0.25, 120.0)
            a_init = asymmetric_init(n_cur, variant, rng)
            print(f"  Asymmetric variant {variant}, budget={t_sub:.0f}s")

            # Quick p-norm
            a_opt = pnorm_lbfgs(a_init, t_sub * 0.6, p_start=15.0, p_end=200.0, n_stages=5)
            improved = try_update(a_opt, f"asym_{variant}")

            if improved:
                rem = deadline - time.monotonic()
                if rem > 30:
                    t_loc = min(rem * 0.15, 50.0)
                    a_loc, _ = local_refine(a_opt, t_loc, n_steps=10000)
                    try_update(a_loc, f"asym_loc_{variant}")

    # === Phase 3: High-p refinement on best ===
    print(f"\n=== Phase 3: High-p Refinement (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_hp = min(rem * 0.3, 200.0)
        print(f"  Budget: {t_hp:.0f}s, p: 200->1000")
        a_hp = pnorm_lbfgs(best_state['seq'], t_hp, p_start=200.0, p_end=1000.0, n_stages=6)
        try_update(a_hp, "high_p")

    # === Phase 4: Aggressive local search ===
    print(f"\n=== Phase 4: Aggressive Local (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_loc = min(rem * 0.3, 150.0)
        a_loc, _ = local_refine(best_state['seq'], t_loc, n_steps=30000)
        try_update(a_loc, "aggressive_local")

    # === Phase 5: FL + L-BFGS sandwich iterations ===
    print(f"\n=== Phase 5: FL Sandwich (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_sand = min(rem * 0.35, 200.0)
        a_sand = best_state['seq'].copy()
        t_step = t_sand / 8
        for it in range(8):
            if time.monotonic() > deadline - 30:
                break
            # FL step
            a_sand = fl_spectral_flatten(a_sand, alpha=0.05 + 0.03 * it)
            # Short p-norm
            a_sand = pnorm_lbfgs(a_sand, t_step, p_start=max(200 + 100 * it, 200),
                                  p_end=min(200 + 300 * (it + 1), 2000), n_stages=3)
            try_update(a_sand, f"sandwich_{it}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 100.0)
        a_fin = pnorm_lbfgs(best_state['seq'], t_fin, p_start=500.0, p_end=1500.0, n_stages=4)
        try_update(a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_loc2 = min(rem * 0.5, 60.0)
        a_loc2, _ = local_refine(best_state['seq'], t_loc2, n_steps=20000)
        try_update(a_loc2, "final_local")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
