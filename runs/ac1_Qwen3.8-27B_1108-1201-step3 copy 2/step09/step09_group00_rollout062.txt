import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_gain_solver_cached = None

    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:

                _gain_solver_cached = solver_name

    return _gain_solver_cached


def solve_lp_best_response(a_fixed, solver=None):

def lbgfs_refine_fft(a0, time_limit, T_stages=None, maxiter_per_stage=10000):
    """L-BFGS with FFT-accelerated LogSumExp gradient in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(50, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_s = softmax_from_z(z)
            b = fast_convolve(a_s)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / stage_T) * lse

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a_s = softmax_from_z(z)
            lse_val, grad_a = lse_objective_and_grad_a(a_s, stage_T)
            S = np.dot(grad_a, a_s)
            return a_s * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000):
    """Nesterov dual averaging LP refinement with adaptive restarts."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(54321)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_rounds = 0
            else:
                stall_rounds += 1
            prev_best = best_score

            if stall_rounds >= 5:
                noise_scale = 0.05 / np.sqrt(stall_rounds)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

            if (it + 1) % 5000 == 0
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_block_descent(a0, time_limit, n_blocks=12, n_steps=50000):
    """Peak-aware block coordinate descent with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active_peaks = np.where(b >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([int(np.argmax(b))])

        pick_peak = active_peaks[step % len(active_peaks)]
        js = max(0, pick_peak - n + 1)
        je = min(n - 1, pick_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[pick_peak - idx]

        top_k = min(n_blocks, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_k]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            # Line search: try several reduction factors
            for factor in [0.98, 0.96, 0.94, 0.92, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                # Redistribute to non-peak-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[:top_k] = False  # rough mask
                # Better: redistribute to positions with lowest contribution to all peaks
                mask = np.ones(n, dtype=bool)
                for pk in active_peaks:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    contrib2 = a[je2-js2+1:] * a[pk - np.arange(je2-js2+1)][::-1] if False else None
                # Simple: redistribute proportional to current values excluding top
                mask = np.ones(n, dtype=bool)
                mask[np.sort(top_positions)] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous perturbation
            n_pert = min(5, n)
            pert_positions = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for pos in pert_positions:
                multiplier = 1.0 + rng.uniform(-0.03, 0.03)
                a_trial[pos] *= multiplier
            a_trial = np.maximum(a_trial, 0)
            ts = a_trial.sum()
            if ts > 1e-15:
                a_trial = a_trial / ts
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 500:
                        break
            else:
                stall += 1
                if stall > 500:
                    break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PBD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def dual_guided_subgradient(a0, time_limit, n_steps=100000):
    """Subgradient method using dual weights from active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Soft weights: exponential of normalized b
        temp = max_b * 0.01 + 1e-10
        norm_b = (b - b.min()) / (max_b - b.min() + 1e-15)
        w = np.exp(norm_b / temp)
        w /= w.sum()

        # Gradient in a-space using FFT
        B = hankel_product(w, a)
        grad = 2.0 * B

        # Projected gradient step
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.85
            no_improve = 0
        lr = max(1e-7, lr)

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence based on continuous analysis."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Cosine profile (continuous analysis suggests this is good)
        a = 1.0 + np.cos(np.pi * x)
    elif choice == 1:
        # Quadratic (V-shape inverted)
        a = 1.0 - 0.5 * (2 * x - 1) ** 2
    elif choice == 2:
        # Bimodal
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 3:
        # Cosine with frequency 2 (anti-symmetric pairs)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Power law
        a = (1 - x) ** 0.5 + 0.1
    elif choice == 5:
        # Sinusoidal modulation
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Parabolic
        a = 4.0 * x * (1 - x) + 0.1
    else:
        a = np.cos(np.pi * x / 2) ** 1.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-alternating, FFT-accelerated L-BFGS
    LogSumExp, peak-aware block coordinate descent, and dual-guided subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1000
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick LP-alternating warmup ===
    print(f"\n=== Phase 1: LP warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "lp_warmup")

    # === Phase 2: Deep LP-alternating with Nesterov ===
    print(f"\n=== Phase 2: Deep LP Nesterov (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp2, "lp_deep")

    # === Phase 3: FFT-accelerated L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS FFT LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 40:
        T_stages = [40.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        print(f"  T_stages: {T_stages}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine_fft(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=15000)
        try_update(a_lb, "lbgfs_fft")

    # === Phase 4: Peak-aware block coordinate descent ===
    print(f"\n=== Phase 4: Peak Block Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_pb, _ = peak_block_descent(best_seq, t_alloc, n_blocks=10, n_steps=50000)
        try_update(a_pb, "peak_block")

    # === Phase 5: Dual-guided subgradient ===
    print(f"\n=== Phase 5: Dual Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_sg, _ = dual_guided_subgradient(best_seq, t_alloc, n_steps=100000)
        try_update(a_sg, "dual_sg")

    # === Phase 6: Second L-BFGS at lower temperature ===
    print(f"\n=== Phase 6: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        T_stages2 = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lb2 = lbgfs_refine_fft(best_seq, t_alloc, T_stages=T_stages2, maxiter_per_stage=12000)
        try_update(a_lb2, "lbgfs_final")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        n_cur = len(best_seq)
        alt_lengths = [int(n_cur * 0.85), int(n_cur * 1.15), int(n_cur * 0.75)]
        for alt_n in alt_lengths:
            if alt_n < 100 or alt_n > 5000:
                continue
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alt = min(rem * 0.25, 150.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_alt * 0.5, solver=solver, n_max_iters=30000)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_alt2 = lbgfs_refine_fft(best_seq, min(rem * 0.4, 80.0),
                                          T_stages=[10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                                          maxiter_per_stage=5000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Structured perturbation + re-optimize ===
    print(f"\n=== Phase 8: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.3, 150.0)
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        # Try cosine perturbations at different frequencies
        for freq in [1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbation = 0.03 * np.cos(freq * x)
            perturbed = best_seq + perturbation
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_sub = min(rem * 0.3, 40.0)
            a_pert = lbgfs_refine_fft(perturbed, t_sub,
                                      T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                                      maxiter_per_stage=3000)
            try_update(a_pert, f"pert_f{freq}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 40.0)
        if t_fin > 10:
            a_fin = lbgfs_refine_fft(best_seq, t_fin,
                                     T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02],
                                     maxiter_per_stage=5000)
            try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.4, 30.0)
        if t_ps > 10:
            a_ps, _ = peak_block_descent(best_seq, t_ps, n_blocks=8, n_steps=30000)
            try_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_sg = min(rem * 0.4, 20.0)
        if t_sg > 5:
            a_sg2, _ = dual_guided_subgradient(best_seq, t_sg, n_steps=50000)
            try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
