Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov dual averaging (minimax game equilibrium)
2. Nesterov accelerated projected gradient on log-sum-exp smoothed objective
3. Mirror descent (multiplicative weights) for shape preservation
4. Peak shaving coordinate descent for final polish
5. Multi-length exploration with interpolation

Key improvements over previous:
- Better step size schedules with adaptive learning
- Mirror descent phase for better convergence geometry
- Warm-start LP with previous dual solution

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights / mirror descent step on simplex."""
    n = len(a)
    log_a = np.where(a > 1e-300, np.log(a), -300.0)
    new_log = log_a - eta * grad
    new_log -= new_log.max()  # numerical stability
    new_a = np.exp(new_log)
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a / s


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def grad_exact_max(a):
    """Subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    peak_k = int(np.argmax(b))
    # Subgradient: 2 * a[peak_k - i] for valid i
    grad = np.zeros(n)
    for i in range(n):
        j = peak_k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def nesterov_projected_gradient(a0, time_limit, T_start=3.0, T_end=500.0,
                                 lr_init=0.06, n_stages=10, steps_per_stage=3000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = step_count
            lr = lr_init / (1.0 + 0.008 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0
                beta = (t_prev_calc - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, eta_start=0.5, n_stages=8,
                            steps_per_stage=2000, T_start=2.0, T_end=200.0):
    """Mirror descent (multiplicative weights) with smoothed objective."""
    n = len(a0)
    x = project_simplex(a0.copy())
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage in range(n_stages):
        T = temps[stage]
        eta = eta_start / (1.0 + stage * 0.5)
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            grad = grad_smooth_max(x, T)
            # Normalize gradient for stable updates
            grad_norm = np.abs(grad).max()
            if grad_norm > 1e-15:
                grad = grad / grad_norm
            x = mirror_descent_step(x, grad, eta)
            if (step + 1) % 50 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    prev_best = best_score
    consecutive_no_improve = 0

            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:

            best_a = y.copy()
            improved = True

        if improved:
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if it % 200 == 0:

            else:
                stall_count += 1

                y = project_simplex(y + noise)

                consecutive_no_improve = 0

        if (it + 1) % 2000 == 0:

def peak_shave_local(a0, time_limit, n_steps=50000):
    """Aggressive peak shaving with coordinate and pair moves."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        if peak_val <= 0:
            break

        # Find top contributing pairs for this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1]

        improved = False

        # Try single coordinate reduction on top contributors
        for ci in top_contrib[:8]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.90, 0.87, 0.83]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction on top contributing pairs
            for ci in top_contrib[:6]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or i < 0 or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Try reducing all top contributors simultaneously
            a_trial = a.copy()
            for ci in top_contrib[:4]:
                i = idx[ci]
                if a[i] > 1e-14:
                    a_trial[i] *= 0.95
            a_trial = project_simplex(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step > 200:
                # Try a broader move: reduce a few random elements
                rng = np.random.default_rng(step)
                a_trial = a.copy()
                for _ in range(3):
                    i = rng.integers(n)
                    a_trial[i] *= 0.90
                a_trial = project_simplex(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def fine_peak_shave(a0, time_limit):
    """Very fine peak shaving with small factors for final polish."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        # Find all peaks above 95% of max
        b_max = b.max()
        threshold = 0.95 * b_max
        peak_positions = np.where(b >= threshold)[0]

        for pk in peak_positions[:3]:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_idx = np.argsort(contribs)[::-1][:5]

            for ci in top_idx:
                i = idx[ci]
                if a[i] < 1e-15:
                    continue
                # Try very small reductions
                for factor in [0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = project_simplex(a_trial)
                    score = score_normalized(a_trial)
                    if score < best_score - 1e-16:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break

        # Periodic global check
        if step % 50 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based Nesterov, mirror descent,
    projected gradient, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP-based Nesterov (primary workhorse) ===
    print(f"\n=== Phase 1: LP Nesterov n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: Mirror descent for shape refinement ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc, eta_start=0.3,
                                          n_stages=6, steps_per_stage=3000,
                                          T_start=2.0, T_end=150.0)
        try_update(a_md, "mirror")

    # === Phase 3: Second LP Nesterov round ===
    print(f"\n=== Phase 3: LP Nesterov Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "lp_phase2")

    # === Phase 4: Nesterov projected gradient with high T ===
    print(f"\n=== Phase 4: NG High-T (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_ng, _ = nesterov_projected_gradient(best_seq, t_alloc, T_start=50.0, T_end=1000.0,
                                              lr_init=0.02, n_stages=6, steps_per_stage=2000)
        try_update(a_ng, "ng_hight")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=100000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Fine peak shaving ===
    print(f"\n=== Phase 6: Fine Peak Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_fps, _ = fine_peak_shave(best_seq, t_alloc)
        try_update(a_fps, "fine_ps")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_fin, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Very fine peak shave ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 50.0)
    if t_alloc > 5:
        a_vf, _ = fine_peak_shave(best_seq, t_alloc)
        try_update(a_vf, "vfinal_ps")

    # === Phase 9: Try nearby lengths ===
    print(f"\n=== Phase 9: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        alt_lengths = [n_best - 20, n_best - 10, n_best + 10, n_best + 20,
                       n_best - 50, n_best + 50]
        alt_lengths = [l for l in set(alt_lengths) if 512 <= l <= 2048]
        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.25, 60.0)
            if t_sub < 15:
                break
            print(f"  n={alt_n}, budget={t_sub:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_lp_refine(init_alt, t_sub * 0.6, solver=solver, n_max_iters=10000)
            try_update(a_alt, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 10:
                a_alt2, _ = fine_peak_shave(a_alt, min(rem * 0.3, 15.0))
                try_update(a_alt2, f"ps_n{alt_n}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
