- Given b, find a that minimizes max_k (b*a)[k] (best response for the first factor)

At a fixed point, a is a best response to itself, which means:
max_k (a*a)[k] = min_b max_k (a*b)[k]

This is the value of the "single-shot" game where one player has fixed strategy a and the other wants to minimize the max convolution.

By the minimax theorem for this game (where one player's strategy is fixed):
min_b max_k (a*b)[k] = max_{p in simplex} min_b sum_k p[k]*(a*b)[k] = max_p min_b (p^T T(a) b)

where p is a distribution over k. The inner min over b (subject to b >= 0, sum b = 1) gives:
min_b sum_k p[k] sum_i T(a)[k,i] b[i] = min_b sum_i b[i] sum_k p[k] T(a)[k,i]

= min_i sum_k p[k] T(a)[k,i] (since b is a distribution, the min is at the minimum coordinate)

= min_i (p * T(a))[i] where (p*T(a))[i] = sum_k p[k] * a[k-i] (with appropriate boundary handling)

So the value of the game is: max_p min_i (p * T(a))[i]

This is a saddle point characterization. At the optimum, there exists a distribution p over k such that:
- For all i: (p*T(a))[i] >= V (where V is the game value)
- The set of i where (p*T(a))[i] = V is the support of the optimal b
- p is supported on the k's where (a*b)[k] = V

This gives us a KKT-like condition for the fixed point!

3. The optimal b is supported on argmin g (put mass on the minimum)
4. Update a based on the KKT conditions

def full_convolve(a, b=None):
    """Full convolution via FFT."""
    if b is None:
        b = a
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_best_response_lp(a, solver=None):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (best-response) method with sparse convolution matrices
2. Dual LP for optimality verification
3. Multi-start with diverse initializations at multiple lengths
4. Adaptive damping with restarts
5. Coordinate descent polishing with quadratic probing
6. Exploits the non-palindromic structure to shift convolution peaks
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def full_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute the score 2n*max_conv/sum^2 for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = full_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a):
    """Build matrix T where T @ b = convolve(a, b). Shape (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _cached_solver = None


def find_solver():
    global _cached_solver
    if _cached_solver is not None:
        return _cached_solver
    candidates = [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]
    for s in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _cached_solver = cp.ECOS
    print(f"  Falling back to ECOS")
    return _cached_solver


def solve_best_response(a, solver=None, verbose=False):
    """
    Solve LP: min M s.t. convolve(a,b) <= M, b>=0, sum(b)=1.
    Returns optimal b and M value.
    """
    n = len(a)
    a_norm = a / a.sum()
    
    T = build_conv_matrix(a_norm)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-10:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else full_convolve(a_norm, b_val).max()
    return b_val, M_val


def solve_dual_lp(a, solver=None):
    """
    Solve dual LP: max sum(p) s.t. T^T @ p <= 1, p >= 0.
    Returns optimal p and dual objective (lower bound on M).
    """
    n = len(a)
    a_norm = a / a.sum()
    T = build_conv_matrix(a_norm)
    
    p_var = cp.Variable(2 * n - 1, nonneg=True)
    
    constraints = [
        T.T @ p_var <= 1.0,
    ]
    
    prob = cp.Problem(cp.Maximize(cp.sum(p_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    p_val = np.asarray(p_var.value, dtype=np.float64)
    p_val = np.maximum(p_val, 0.0)
    dual_obj = float(prob.value) if prob.value is not None else float(np.sum(p_val))
    return p_val, dual_obj


def alternating_lp_optimize(a_init, time_budget, solver=None, verbose=True):
    """
    Alternating best-response LP optimization.
    Iteratively: fix a, solve LP for best b; update a towards b.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_of(a)
    
    if solver is None:
        solver = find_solver()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        step_start = time.monotonic()
        
        # Solve best response LP
        b, M_val = solve_best_response(a, solver)
        if b is None:
            break
        
        # Evaluate b
        b_score = score_of(b)
        if b_score < best_score - 1e-12:
            best_score = b_score
            best_a = b.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive damping
        if no_improve > 20:
            alpha = 0.05
            no_improve = 0
        else:
            alpha = min(1.0, 2.5 / (it + 2.0))
        
        # Damped update
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        a_score = score_of(a_new)
        if a_score < best_score - 1e-12:
            best_score = a_score
            best_a = a_new.copy()
        
        # Also try the average of current best and b
        avg = 0.5 * (best_a + b)
        avg = np.maximum(avg, 0)
        s_avg = avg.sum()
        if s_avg > 1e-10:
            avg = avg / s_avg
            avg_score = score_of(avg)
            if avg_score < best_score - 1e-12:
                best_score = avg_score
                best_a = avg.copy()
        
        a = a_new
        
        step_time = time.monotonic() - step_start
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      ALP it={it} ({elapsed:.0f}s, step={step_time:.1f}s): "
                  f"best={best_score:.6f}, M={M_val:.8f}")
        
        # Early stopping if converged
        if M_val < best_score / (2.0 * n) * (a.sum()**2) / (2.0 * n) + 1e-10:
            if it > 50:
                break
    
    return best_a, best_score


def coord_descent_polish(a0, time_budget, verbose=True, label=""):
    """
    Coordinate descent with quadratic probing for local optimization.
    Tries to reduce max convolution by adjusting individual coordinates.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64).copy()
    a = a / a.sum()
    
    conv = full_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        sweeps += 1
        improved = False
        sweep_best_M = M
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            a_shifted = a[k_start - i:k_end - i + 1]
            affected = conv[k_start:k_end + 1].copy()
            cur_M = max(unaff_max, float(affected.max()))
            
            if cur_M < sweep_best_M + 1e-14:
                sweep_best_M = cur_M
            
            # Try several step sizes
            best_delta = 0.0
            best_new_M = cur_M
            
            base = a[i]
            factors = [-0.3, -0.15, -0.05, -0.01, 0.01, 0.03, 0.05, 0.1, 0.2]
            
            for f in factors:
                delta = base * f
                new_val = base + delta
                if new_val < -1e-15:
                    continue
                new_val = max(new_val, 0.0)
                delta = new_val - base
                if abs(delta) < 1e-15:
                    continue
                
                new_aff = affected + 2.0 * delta * a_shifted
                # Fix quadratic term at position 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                if M < best_score / (2.0 * n):
                    best_score = 2.0 * n * M
                    best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
        
        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def init_ramp(n, direction=1, power=1.0):
    """Linear ramp initialization."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64) ** power
    else:
        a = np.arange(n, 0, -1, dtype=np.float64) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_two_hump(n, rng):
    """Two Gaussian humps."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.35)
    c2 = n * rng.uniform(0.65, 0.8)
    w = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine_power(n, rng):
    """Cosine window raised to a power."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    p = rng.uniform(1.0, 5.0)
    a = (0.5 * (1 + np.cos(np.pi * x))) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_asymmetric(n, rng):
    """Asymmetric shape - high on one side, decaying on the other."""
    t = np.arange(n, dtype=np.float64)

    a = np.maximum(a, 0)

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    best_seq = None
    best_score = float('inf')
    
    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Determine which lengths to try
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    if len(parent) != 2048:
        lengths_to_try.append(2048)
    if len(parent) != 1536:
        lengths_to_try.append(1536)
    
    # Allocate time
    total_opt_time = 1500.0
    n_starts = min(len(lengths_to_try), 3)
    
    for length_idx, n in enumerate(lengths_to_try[:n_starts]):
        remaining = deadline - time.monotonic() - 100.0
        if remaining < 120:
            break
        
        # Allocate time for this length
        if length_idx == 0:
            t_alloc = min(remaining * 0.5, 600.0)
        else:
            t_alloc = min(remaining * 0.35, 400.0)
        
        if t_alloc < 60:
            break
        
        print(f"\n=== Optimizing n={n}, budget={t_alloc:.0f}s ===")
        
        # Generate initializations for this length
        inits = []
        
        if parent is not None and len(parent) == n:
            inits.append(('parent', parent.copy()))
            noisy = parent + rng.standard_normal(n) * 0.1
            noisy = np.maximum(noisy, 0)
            noisy = noisy / noisy.sum()
            inits.append(('noisy_parent', noisy))
        
        inits.append(('ramp_up', init_ramp(n, 1, 1.0)))
        inits.append(('ramp_down', init_ramp(n, -1, 1.0)))
        inits.append(('ramp_quad', init_ramp(n, 1, 1.5)))
        inits.append(('two_hump', init_two_hump(n, rng)))
        inits.append(('cosine', init_cosine_power(n, rng)))
        inits.append(('asym', init_asymmetric(n, rng)))
        inits.append(('mixed', init_mixed(n, rng, parent if parent is not None and len(parent) == n else None)))
        inits.append(('uniform', np.ones(n) / n))
        
        # Sort by initial score (try best first)
        scored_inits = []
        for name, a in inits:
            s = score_of(a)
            scored_inits.append((s, name, a))
        scored_inits.sort(key=lambda x: x[0])
        
        # Run optimization on top initializations
        n_trials = min(len(scored_inits), 5)
        t_per_trial = t_alloc / n_trials
        
        for trial_idx in range(n_trials):
            remaining = deadline - time.monotonic() - 50.0
            if remaining < 30:
                break
            
            init_score, init_name, init_a = scored_inits[trial_idx]
            t_trial = min(t_per_trial, remaining * 0.7)
            if t_trial < 20:
                break
            
            print(f"  Trial {trial_idx}: {init_name} (init_score={init_score:.6f}), budget={t_trial:.0f}s")
            
            # Phase 1: Alternating LP
            t_lp = t_trial * 0.6
            a_lp, s_lp = alternating_lp_optimize(init_a, t_lp, verbose=True)
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"ALP_{init_name}")
            
            # Phase 2: Coordinate descent
            remaining = deadline - time.monotonic() - 30.0
            if remaining > 20:
                t_cd = min(t_trial * 0.25, remaining * 0.5)
                a_cd, s_cd = coord_descent_polish(a_lp, t_cd, verbose=False, label=f"cd_{init_name}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"CD_{init_name}")
            
            # Phase 3: Fine polishing
            remaining = deadline - time.monotonic() - 20.0
            if remaining > 15 and s_cd < best_score + 0.01:
                t_fine = min(t_trial * 0.15, remaining * 0.5)
                a_fine, s_fine = fine_coord_descent(a_cd, t_fine, verbose=False)
                best_seq, best_score = check_and_update(a_fine, best_seq, best_score, f"fine_{init_name}")
                # Also polish the best sequence overall
                if s_fine > best_score:
                    a_fine2, s_fine2 = fine_coord_descent(best_seq, t_fine * 0.5, verbose=False)
                    best_seq, best_score = check_and_update(a_fine2, best_seq, best_score, f"fine_best")
    
    # Final global polishing on the best sequence found
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final polishing (budget={remaining:.0f}s) ===")
        t_final_cd = min(remaining * 0.4, 60.0)
        a_fin, s_fin = coord_descent_polish(best_seq, t_final_cd, verbose=True, label="final_cd")
        best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")
        
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 15:
            t_final_fine = min(remaining * 0.5, 30.0)
            a_ff, s_ff = fine_coord_descent(best_seq, t_final_fine, verbose=False)
            best_seq, best_score = check_and_update(a_ff, best_seq, best_score, "final_fine")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
