import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

    return cp.ECOS

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.GLOP, cp.GLPK, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fast_coord_descent(a0, time_limit, verbose=False, label=""):

    For each coordinate, finds the optimal value by evaluating g(t) at multiple points
    including the analytical upper bound.

    else:
        a = a / s

    while time.monotonic() - t0 < time_limit:

        improved = False

        for i in coord_order:
            ai = a[i]
            if ai < 1e-15:
                # Even zero coords can be increased
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)

                if k_start > 0:
                    unaff_max = float(conv[:k_start].max())
                else:
                    unaff_max = 0.0
                if k_end < 2 * n - 2:
                    unaff_right = float(conv[k_end + 1:].max())
                    unaff_max = max(unaff_max, unaff_right)

                affected_len = k_end - k_start + 1
                a_slice = a[:affected_len]

                # Try small positive values
                best_t = 0.0
                best_g = max(unaff_max, 0.0)
                for t_trial in [1e-4, 5e-4, 1e-3, 5e-3]:

                        break
                    vals = 2.0 * t_trial * a_slice

                        vals = vals.copy()
                        vals[2 * i - k_start] = t_trial * t_trial
                    g_val = max(unaff_max, float(vals.max()))
                    if g_val < best_g - 1e-15:
                        best_g = g_val
                        best_t = t_trial

                if best_t > 0 and best_g < M - 1e-14:

                    a[i] = best_t

                        changes[2 * i - k_start] = delta * delta

                    M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = 0.0
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            affected_len = k_end - k_start + 1
            a_slice = a[:affected_len]

            # C_k = conv[k] - 2*ai*a[k-i] (for k != 2i)
            c_vals = conv[k_start:k_end + 1] - 2.0 * ai * a_slice

            # Adjust for k = 2i: should be conv[2i] - ai^2
            has_2i = (2 * i >= k_start and 2 * i <= k_end)
            idx_2i = 2 * i - k_start if has_2i else -1
            if has_2i:
                c_vals = c_vals.copy()
                c_vals[idx_2i] = conv[2 * i] - ai * ai

            # Find optimal t: minimize max(unaff_max, max(c_vals + 2*t*a_slice), c_2i + t^2)
            # Analytical upper bound: t such that c_vals + 2*t*a_slice <= unaff_max
            # => t <= (unaff_max - c_vals) / (2*a_slice) for a_slice > 0
            eps_val = 1e-15

            if positive_mask.any():

                    t_quad = np.sqrt(unaff_max - c_2i)
                    t_analytical = min(t_analytical, t_quad)

            candidates = [0.0, ai, t_analytical]
            if t_analytical > 0:

            best_g = M  # current value

            for t_trial in candidates:
                if t_trial < -1e-15:
                    continue
                t_trial = max(0.0, t_trial)
                # g(t) = max(unaff_max, max(c_vals + 2*t*a_slice))
                vals = c_vals + 2.0 * t_trial * a_slice
                if has_2i:
                    vals = vals.copy()
                    vals[idx_2i] = c_vals[idx_2i] + t_trial * t_trial
                g_val = max(unaff_max, float(vals.max()))
                if g_val < best_g - 1e-15:
                    best_g = g_val
                    best_t = t_trial

            if best_t != ai and best_g < M - 1e-14:
                delta = best_t - ai
                a[i] = best_t
                changes = 2.0 * delta * a_slice
                if has_2i:
                    changes = changes.copy()
                    changes[idx_2i] = delta * (ai + best_t)
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.10f}")

    return best_a, best_score, sweeps

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):

                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:

            else:
                unaff_max = 0.0
            if k_end < 2 * n - 2:

                unaff_max = max(unaff_max, unaff_right)

            affected_len = k_end - k_start + 1

            has_2i = (2 * i >= k_start and 2 * i <= k_end)

                if t_trial < 0:
                    continue
                vals = c_vals + 2.0 * t_trial * a_slice

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Structured perturbation targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    peak_threshold = M * 0.90
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.2):
    """Randomly zero out and redistribute mass to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        chosen = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[chosen] = rng.random(len(chosen)) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def init_cosine(n, rng):
    """Initialize using cosine structure (exploits flat-convolution property)."""
    # Base: cos^2 profile
    t = np.arange(n)
    a = (1.0 + np.cos(2.0 * np.pi * t / n)) / 2.0

    # Add random harmonics
    n_harm = int(rng.integers(1, 4))
    for _ in range(n_harm):
        freq = rng.uniform(1.0, 3.0)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):

    if s < 1e-15:

    return a / s

def init_symmetric(n, rng):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None):
    """Fictitious play with Nesterov averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_seq = a.copy()
    b_seq = b.copy()
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        gamma_it = 2.0 / (iters + 2.0)
        a_seq = (1.0 - gamma_it) * a_seq + gamma_it * a
        b_seq = (1.0 - gamma_it) * b_seq + gamma_it * b

        for cand in [a, b, a_seq, b_seq, 0.5 * (a_seq + b_seq)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver=None):
    """Damped LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization at multiple lengths + quick CD
    Phase 2: LP fictitious play for global direction
    Phase 3: Intensive coordinate descent with perturbation cycles
    Phase 4: Fine polishing
    Phase 5: Alternative length exploration
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse Init + Quick CD ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()

    n_values = [512, 768, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 120:
            break

        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("cosine", init_cosine(n, rng)))
        init_cands.append(("bimodal", init_bimodal(n, rng)))
        init_cands.append(("ridges", init_flat_ridges(n, rng)))
        init_cands.append(("symmetric", init_symmetric(n, rng)))

        for name, cand in init_cands:
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 2:
                break
            t_alloc = max(1.5, min(4.0, t_left / 2.0))
            a_opt, s_opt, _ = fast_coord_descent(cand, t_alloc, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top-5: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:5]]))

    for s_c, a_c, n_c, it_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt in [1024, 768, 1280]:
            if alt != n_best and 256 <= alt <= 2048:
                fp_lengths.append(alt)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 600
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  NFP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp) if best_seq is not None else None
            a_fp, score_fp = nesterov_fictitious_play(n_fp, t_alloc, initial=initial, solver=solver)
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"nfp_n{n_fp}")

            # Damped LP polish
            rem = deadline - time.monotonic() - 500
            if rem > 30:
                a_dl, _ = damped_lp_refine(a_fp, min(rem * 0.3, 60.0), solver=solver)
                best_seq, best_score = check_and_update(a_dl, best_seq, best_score, f"dlp_n{n_fp}")

    # === Phase 3: Intensive CD + Perturbation ===
    print(f"\n=== Phase 3: Intensive CD + Perturbation (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(8):
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 100.0)
            print(f"  CD round {rnd + 1}/8, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")

            a_opt, s_opt, _ = fast_coord_descent(current, t_alloc, verbose=(rnd % 2 == 0), label=f"p3r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p3_cd_r{rnd}")

            # Perturb for next round
            if rnd < 5:
                if rnd % 3 == 0:
                    current = peak_targeted_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                elif rnd % 3 == 1:
                    current = random_restructure(best_seq, rng, frac=0.1 + 0.05 * rnd)
                else:
                    # Small noise
                    noisy = best_seq + rng.standard_normal(n_best) * 0.005
                    noisy = np.maximum(noisy, 0)
                    s = noisy.sum()
                    if s > 1e-10:
                        current = noisy / s
                    else:
                        current = best_seq.copy()
            else:
                current = best_seq.copy()

    # === Phase 4: Fine Polishing ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Multiple fine passes
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_fine = min(rem * 0.4, 120.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_polish(best_seq, t_fine, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd_1")

        # Another round of fast CD
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem * 0.5, 80.0)
            a_cd, s_cd, _ = fast_coord_descent(best_seq, t_cd, label="fcd_polish")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "fcd_polish")

        # Final fine pass
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_f2, s_f2 = fine_polish(best_seq, min(rem * 0.7, 40.0), label="fine2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd_2")

    # === Phase 5: Alternative Length ===
    print(f"\n=== Phase 5: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for delta in [128, 256, -128, -256, 384]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 3072 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = alt_lengths[:4]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alt = min(rem * 0.4, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

            # Quick fine polish on alt
            rem = deadline - time.monotonic() - 10
            if rem > 8 and s_alt < best_score:
                a_alt2, _ = fine_polish(a_alt, min(rem * 0.5, 20.0), label=f"altfine{alt_n}")
                best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"altfine_{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
