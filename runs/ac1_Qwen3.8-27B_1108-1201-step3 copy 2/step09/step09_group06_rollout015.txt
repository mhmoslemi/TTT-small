def smoothed_objective_and_grad(z, T):
    """Compute smoothed objective and gradient w.r.t. z in log-parameterization."""
    n = len(z)

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    S = exp_z.sum()
    a = exp_z / S

    # Convolution
    b = fast_convolve(a)

    # Log-sum-exp
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_sum = exp_Tb.sum()
    objective = (2.0 * n / T) * (max_Tb + np.log(lse_sum))

    # Weights
    w = exp_Tb / lse_sum

    return objective, grad, a, b, w

def nesterov_lp(a0, time_limit, solver=None, max_iters=100000, noise_interval=300, noise_mag=0.02):

    a = np.maximum(a0, 0).copy()

    T_matrix = build_conv_matrix(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # LP best response
        b_new, M_val = solve_lp_best_response(y, T_matrix, solver)

        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        a_new = np.maximum(a_new, 0)

        if sa < 1e-15:

        else:
            a_new = a_new / sa

        a = a_new

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

                rng = np.random.default_rng(it)
                noise = rng.normal(0, noise_mag, n)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} M_lp={M_val:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.3, maxiter=5000):
    """L-BFGS refinement on log-sum-exp smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _, _, _, _ = smoothed_objective_and_grad(z_in, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            _, grad, _, _, _ = smoothed_objective_and_grad(z_in, stage_T)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a = np.exp(z - z.max())
    a = a / a.sum()
    return a


def subgradient_mirror_descent(a0, time_limit, active_frac=0.05, lr_start=0.3):
    """Mirror descent with subgradients of max."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lr = lr_start
    t0 = time.monotonic()
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.8
            no_improve = 0

        # Subgradient: average over active lags
        threshold = max_b - active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break

        alpha = lr / g_norm
        z = z - alpha * g

        # Softmax
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step={step} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def skewed_init(n, rng, style=None):
    """Generate skewed/non-symmetric initialization."""
    if style is None:
        style = rng.integers(0, 8)

    if style == 0:
        # Exponential growth
        a = np.exp(np.linspace(0, 3.0, n))
    elif style == 1:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3.0, n))
    elif style == 2:
        # Skewed Gaussian (peak at 0.3)
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2)
    elif style == 3:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 2
    elif style == 4:
        # Power law reversed
        x = np.linspace(1, 0.01, n)
        a = x ** 2
    elif style == 5:
        # Bimodal skewed
        x = np.linspace(0, 1, n)
        a = np.exp(-15 * (x - 0.25) ** 2) + 0.5 * np.exp(-8 * (x - 0.7) ** 2)
    elif style == 6:
        # Sine power (non-symmetric)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x * 0.8 + 0.3) ** 2
        a = np.maximum(a, 0)
    else:
        # Random smooth
        a = np.random.dirichlet(np.ones(n) * 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Quick multi-length exploration ===
    print(f"\n=== Phase 1: Multi-length exploration (best={best_score:.6f}) ===")
    lengths_to_try = [800, 1000, 1200, 1500, 2000]

    if parent is not None:
        # Include parent length
        lengths_to_try.insert(0, len(parent))
    lengths_to_try = sorted(set(lengths_to_try))

    t0_phase1 = time.monotonic()
    phase1_budget = min(120.0, (deadline - time.monotonic()) * 0.15)

    for n_try in lengths_to_try:
        if time.monotonic() - t0_phase1 > phase1_budget:
            break
        if time.monotonic() > deadline - 300:
            break

        remaining = phase1_budget - (time.monotonic() - t0_phase1)
        t_per = max(5.0, remaining / max(1, len(lengths_to_try) - lengths_to_try.index(n_try)))

        print(f"  n={n_try}, budget={t_per:.0f}s")

        # Generate diverse initializations
        best_here = None
        best_here_score = float('inf')

        n_inits = min(4, max(2, int(t_per / 15)))
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 300:
                break
            a_init = skewed_init(n_try, rng, style=init_i)

            # Quick Adam optimization
            t_adam = max(1.0, t_per / n_inits - 2)
            a_opt, _ = adam_smoothed_optimize(
                a_init, t_adam,
                T_schedule=np.geomspace(50, 1, 4),
                lr=0.05
            )
            sc = evaluate_safe(a_opt)
            if sc < best_here_score:
                best_here_score = sc
                best_here = a_opt.copy()

        if best_here is not None:
            try_update(best_here, f"explor_n{n_try}")

    # === Phase 2: Focused optimization on best length ===
    print(f"\n=== Phase 2: Focused optimization (best={best_score:.6f}) ===")
    n_best = len(best_seq) if best_seq is not None else 1000

    # Try a few nearby lengths
    candidate_lengths = [n_best]
    for mult in [0.8, 1.2, 1.5, 0.7]:
        alt = int(n_best * mult)
        if 400 <= alt <= 4000 and alt != n_best:
            candidate_lengths.append(alt)
    candidate_lengths = sorted(set(candidate_lengths))[:3]

    for n_focus in candidate_lengths:
        if time.monotonic() > deadline - 100:
            break

        # Get a good starting point for this length
        if n_focus == len(best_seq):
            a_start = best_seq.copy()
        else:
            a_start = interpolate_sequence(best_seq, n_focus)
            # Also try fresh skewed init
            a_fresh = skewed_init(n_focus, rng, style=int(rng.integers(0, 5)))
            if evaluate_safe(a_fresh) < evaluate_safe(a_start):
                a_start = a_fresh

        rem = deadline - time.monotonic()
        if rem < 60:
            break

        # Sub-phase 2a: Adam with fine temperature schedule
        t_adam = min(rem * 0.3, 180.0)
        if t_adam > 10:
            print(f"  n={n_focus}: Adam ({t_adam:.0f}s)")
            a_r, _ = adam_smoothed_optimize(
                a_start, t_adam,
                T_schedule=np.geomspace(200, 0.3, 10),
                lr=0.03
            )
            try_update(a_r, f"adam_n{n_focus}")

        # Sub-phase 2b: Nesterov LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.35, 300.0)
        if t_lp > 20:
            print(f"  n={n_focus}: Nesterov LP ({t_lp:.0f}s)")
            a_lp, _ = nesterov_lp(best_seq, t_lp, solver=solver, max_iters=50000)
            try_update(a_lp, f"nlp_n{n_focus}")

        # Sub-phase 2c: L-BFGS refinement
        rem = deadline - time.monotonic()
        t_lb = min(rem * 0.15, 100.0)
        if t_lb > 10:
            print(f"  n={n_focus}: L-BFGS ({t_lb:.0f}s)")
            a_lb = lbgfs_refine(best_seq, t_lb, T_start=30.0, T_end=0.2, maxiter=5000)
            try_update(a_lb, f"lbgfs_n{n_focus}")

        # Sub-phase 2d: Mirror descent
        rem = deadline - time.monotonic()
        t_md = min(rem * 0.1, 60.0)
        if t_md > 10:
            print(f"  n={n_focus}: Mirror Desc ({t_md:.0f}s)")
            a_md, _ = subgradient_mirror_descent(best_seq, t_md, active_frac=0.04, lr_start=0.2)
            try_update(a_md, f"md_n{n_focus}")

    # === Phase 3: Intensive LP on best ===
    print(f"\n=== Phase 3: Intensive LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_lp2 = min(rem * 0.5, 400.0)
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp(best_seq, t_lp2, solver=solver, max_iters=100000, noise_interval=500, noise_mag=0.015)
        try_update(a_lp2, "intensive_lp")

    # === Phase 4: Fine-tuning with low-temperature L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 120.0)
        print(f"  Budget: {t_fin:.0f}s")
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.05, maxiter=8000)
        try_update(a_fin, "fine_lbgfs")

    # === Phase 5: Final Adam polish ===
    print(f"\n=== Phase 5: Final Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = min(rem * 0.5, 60.0)
        print(f"  Budget: {t_final:.0f}s")
        a_fin2, _ = adam_smoothed_optimize(
            best_seq, t_final,
            T_schedule=np.geomspace(5, 0.1, 5),
            lr=0.01
        )
        try_update(a_fin2, "final_adam")

    # === Phase 6: Quick LP polish ===
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        t_p = min(rem * 0.4, 40.0)
        a_pol, _ = nesterov_lp(best_seq, t_p, solver=solver, max_iters=20000)
        try_update(a_pol, "polish_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
