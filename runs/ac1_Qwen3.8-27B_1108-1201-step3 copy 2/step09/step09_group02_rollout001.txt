Actually, let me reconsider the problem from scratch. The objective is:
J(a) = 2n * max_k (a*a)[k] / (sum a)^2

where (a*a)[k] = sum_i a_i * a_{k-i} (valid indices).

Key properties:
- J is scale-invariant (J(ca) = J(a) for c > 0)
- J(a) >= 2 (as we showed, this is achievable)
- Wait, we showed J can be less than 2 (the n=3 example gave 1.875).

Let me re-examine. For n=3, a = (0.5, 0.25, 0.25):
conv = [0.25, 0.25, 0.3125, 0.125, 0.0625]
max = 0.3125
sum = 1
J = 2*3*0.3125/1 = 1.875

So J < 2 is possible. The lower bound is not 2.

What is the actual lower bound? For large n, we can approximate:
J(a) ≈ 2 * max_t R_f(t) / (∫f)^2

where f is the "shape function" and R_f is its autocorrelation.

To minimize this, we want f to have a "flat" autocorrelation (like a Kautz sequence or Barker sequence in the discrete case). For non-negative f, the best we can do is related to the "minimum peak sidelobe" problem.

For a "top-hat" function (square pulse), the autocorrelation is a triangle, and the peak is at t=0. The ratio is 1/(∫f)^2 * ∫f^2 = (∫f^2)/(∫f)^2. For a top-hat of width W: ∫f^2 = W, (∫f)^2 = W^2, ratio = 1/W. As W → ∞, this goes to 0... but that's not right because we're working on a finite interval.

Actually, I think the right way to think about it is:

For a sequence a of length n with sum(a) = 1:
- conv[k] for k near n-1 has about n terms, each of size about (1/n)^2 = 1/n^2, so conv[n-1] ≈ n/n^2 = 1/n.
- To get conv[k] < 1/n for all k, we need the sequence to be "anti-correlated" in some sense.

The key insight is: if the sequence is symmetric (a_i = a_{n-1-i}), then conv[n-1] = sum a_i^2 >= 1/n. But if it's asymmetric, conv[n-1] = sum a_i * a_{n-1-i} can be less than 1/n.

For example, if a is concentrated in the first half (a_i > 0 for i < n/2, a_i = 0 for i >= n/2), then conv[n-1] = sum_{i=0}^{n/2-1} a_i * a_{n-1-i} = 0 (since n-1-i >= n/2 for i < n/2). But then conv[0] = a_0^2 and conv[n/2 - 1] = sum of many terms might be large.

So there's a trade-off. The optimal solution balances these.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def init_raised_cosine(n, rng=None):
    t = (np.arange(n) + 0.5) / n
    a = (1.0 + np.cos(2.0 * np.pi * t)) / 2.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def init_parabolic(n, rng=None):
    t = np.arange(n, dtype=np.float64)
    a = (t + 1.0) * (n - t)
    return a / a.sum()

def init_gaussian(n, rng=None):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * 0.25
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def init_tukey(n, rng, r=0.5):
    # Tukey window (tapered cosine)
    t = np.arange(n, dtype=np.float64)
    half = n // 2
    a = np.ones(n)
    # Left taper
    for i in range(int(r * n)):
        a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
    # Right taper
    for i in range(int(r * n)):
        a[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def compute_subgradient(conv, a, n_peaks=20):
    """Compute subgradient of max(conv) using active set."""
    n = len(a)
    conv_len = 2 * n - 1
    M = conv.max()
    # Active set: coefficients within 1% of max
    threshold = M * 0.99
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                g[i] += a[i]  # derivative of a_i^2
            else:
                g[i] += 2.0 * a[j]
    g /= len(active)
    return g, active

def fast_subgradient(conv, a, n_active=20):
    """Fast subgradient using FFT-based computation for top peaks."""
    n = len(a)
    M = conv.max()
    # Get top n_active peaks
    if n_active < 2 * n - 1:
        top_idx = np.argpartition(conv, -n_active)[-n_active:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    # Create weight vector
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / len(top_idx)
    
    # Compute gradient via correlation
    # g[i] = sum_k w[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] + a[i-(k-i)] ... this is getting complex
    # Let's just do it directly for small n_active
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        partners = k - np.arange(i_start, i_end + 1)
        idx = np.arange(i_start, i_end + 1)
        # d(conv[k])/d(a[i]) = a[k-i] for i != k-i, and a[i] for i = k-i
        contrib = a[partners] * 2.0
        # Fix diagonal terms
        diag = idx == partners
        contrib[diag] = a[idx[diag]]
        g[idx] += contrib / len(top_idx)
    
    return g

def active_set_subgradient(a0, time_limit, verbose=False, label=""):
    """Subgradient descent with active set and line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:

    best_score = 2.0 * n * float(conv.max())

    lr = 0.01
    n_active_max = 30
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        M = conv.max()
        
        # Adaptive active set size
        n_act = min(n_active_max, max(5, int(0.02 * (2 * n - 1))))
        
        # Get active peaks (top n_act)
        top_idx = np.argpartition(conv, -n_act)[-n_act:]

            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            partners = k - idx
            contrib = 2.0 * a[partners]
            diag = idx == partners
            if diag.any():

                contrib[diag] = a[idx[diag]]
            g[idx] += contrib

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            
            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]
            
            if has_quad:
                quad_idx = 2 * i - k_lo
            
            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)
            
            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=35)
            
            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score, sweeps

def basin_hopping(a0, time_limit, optimizer, rng, n_perturbs=10, verbose=False):
    """Basin hopping with structured perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    
    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 120.0)
        
        # Multiple perturbation strategies
        strategy = p % 4
        
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce the most correlated pairs
            conv_p = fast_convolve(a)
            perturbed = a.copy()
            top_k = min(10, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(2, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 + rng.standard_normal() * 0.08)
        elif strategy == 2:
            # Block swap: swap values in two random blocks
            perturbed = a.copy()
            b1 = rng.integers(0, n // 2)
            b2 = rng.integers(n // 2, n)
            size = rng.integers(max(1, n // 20), n // 5)
            b1 = min(b1, n - size)
            b2 = min(b2, n - size)
            perturbed[b1:b1+size], perturbed[b2:b2+size] = perturbed[b2:b2+size].copy(), perturbed[b1:b1+size].copy()
        else:
            # Smooth perturbation (low-pass filtered noise)
            noise = rng.standard_normal(n)
            # Simple moving average filter
            kernel = np.ones(5) / 5.0
            noise = np.convolve(noise, kernel, mode='same')
            sigma = 0.03
            perturbed = a + noise * sigma
        
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps
        
        a_opt, s_opt, _ = optimizer(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p} (strat={strategy}): improved to {best_score:.8f}")
    
    return best_a, best_score
