1. Nesterov-accelerated smoothed subgradient descent (warm-up)
2. LP best-response iteration with dual-variable-guided Nesterov averaging

4. Coordinate-level mass redistribution polish

import numpy as np

from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def conv_gradient_weighted(a, w):
    """Gradient of sum_k w[k]*(a*a)_k w.r.t. a.
    Returns vector g where g[i] = 2 * sum_k w[k] * a[k-i]."""
    n = len(a)
    # Need: g[i] = 2 * sum_k w[k] * a[k-i] (a[j]=0 for j<0 or j>=n)
    # This is cross-correlation: (w cross-correlate a)[i] = sum_k w[k]*a[k-i]
    # Using FFT: ifft(conj(FFT(w)) * FFT(a)) gives this (with proper padding)
    size = 1 << (2 * n - 1).bit_length()
    W = rfft(w, size)
    A = rfft(a, size)
    corr = irfft(np.conj(W) * A, size)[:n]
    return 2.0 * corr


def grad_smoothed_max(a, T):
    """Gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    return conv_gradient_weighted(a, w)


def nesterov_smoothed_subgrad(a0, time_limit, T_start=2.0, T_end=500.0,
                               lr_init=0.05, n_stages=12, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = step_count
            lr = lr_init / (1.0 + 0.003 * global_step)

            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            if step_count > 0:
                t_prev_c = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) ** 2) / 2.0
                beta = (t_prev_c - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            grad = grad_smoothed_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            if step_count % 100 == 0:

                if step_count % 1000 == 0:

def mirror_descent_subgrad(a0, time_limit, T=50.0, lr_init=0.02, steps=10000):
    """Mirror descent (multiplicative updates) on the simplex interior."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(steps):
        if time.monotonic() - t0 > time_limit:
            break
        lr = lr_init / (1.0 + 0.01 * step)
        grad = grad_smoothed_max(a, T)
        # Multiplicative update: a_i <- a_i * exp(-lr * grad_i) / Z
        log_a = np.log(a) - lr * grad
        m = log_a.max()
        exp_log_a = np.exp(log_a - m)
        z = exp_log_a.sum()
        if z < 1e-30 or not np.isfinite(z):
            break
        a = exp_log_a / z
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        if step % 200 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if step % 2000 == 0:
                print(f"        Mirror step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Return the convolution matrix T (2n-1) x n such that T@b = convolve(a,b)."""
    n = len(a)
    size = 2 * n - 1
    # Use structured approach for efficiency
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b).
    Returns (b_val, M_val, active_set)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    # Find active constraints
    conv_ab = T @ b_val
    tol = 0.01 * M_val
    active = np.where(conv_ab >= M_val - tol)[0]
    return b_val, M_val, active


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """LP best-response with Nesterov-dual-averaging schedule."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_start = time.monotonic()
        b_new, M_b, active = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - t_start
        if b_new is None:
            continue
        lp_times.append(lp_time)

        # Nesterov-dual-averaging update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        # Weighted average of all iterates (Nesterov)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.01 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-50:]) if lp_times else 0
            print(f"        LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} avg_lp={avg_lp*1000:.0f}ms")

    return best_a, best_score


def coordinate_mass_redistribute(a0, time_limit, n_steps=20000):

            break

        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        top_peaks = np.argsort(b_work)[::-1][:3]

        improved = False
        for peak_k in top_peaks:
            if a.max() - a.min() < 1e-12:
                break
            # Find contributors to this peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            # Top contributor
            top_i = idx[np.argmax(contribs)]
            if a[top_i] < 1e-10:
                continue

            # Find "quiet" positions (low contribution to all top peaks)
            total_contrib = np.zeros(n)
            for pk in top_peaks:
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                idx2 = np.arange(js2, je2 + 1)
                total_contrib[idx2] += a[idx2] * a[pk - idx2]

            # Quiet positions: low total contribution, not the donor
            quiet_mask = total_contrib < np.median(total_contrib)
            quiet_mask[top_i] = False
            quiet_indices = np.where(quiet_mask)[0]
            if len(quiet_indices) == 0:
                continue

            # Choose recipient: highest a value among quiet positions (to avoid creating new peaks)
            recipient = quiet_indices[np.argmax(a[quiet_indices])]

            # Try moving mass
            for frac in [0.05, 0.1, 0.15, 0.2]:
                move_amt = a[top_i] * frac
                a_trial = a.copy()
                a_trial[top_i] -= move_amt
                a_trial[recipient] += move_amt
                if a_trial.min() < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                a_trial = a_trial / ss
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Try reducing the absolute peak
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_i = idx[np.argmax(contribs)]
            for factor in [0.95, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[top_i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try pairwise adjustment
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top2 = np.argsort(contribs)[::-1][:3]
            for ci in range(len(top2)):
                for cj in range(ci + 1, len(top2)):
                    i1 = idx[top2[ci]]
                    i2 = idx[top2[cj]]
                    if i1 == i2:
                        continue
                    for f1 in [0.95, 0.90]:
                        for f2 in [0.95, 0.90]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            a_trial = project_simplex(a_trial)
                            sc = score_normalized(a_trial)
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Small random perturbation to escape local minima
            rng = np.random.default_rng(step)
            a_trial = a + 0.005 * rng.standard_normal(n)
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                if step > 500 and not improved:
                    # Try larger perturbation
                    a_trial = a + 0.02 * rng.standard_normal(n)
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                    else:
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_diverse_init(n, mode, rng):
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = 0.3 + 0.7 * a / a.max()
    elif mode == 3:
        a = 1.0 + 0.3 * (2 * x - 1) + 0.2 * np.sin(6 * np.pi * x)
    elif mode == 4:
        # Two-bump
        c1, c2 = 0.3, 0.7
        a = np.exp(-4 * (x - c1) ** 2) + np.exp(-4 * (x - c2) ** 2)
        a = 0.2 + 0.8 * a / a.max()
    else:
        a = np.ones(n) + rng.uniform(-0.2, 0.2, n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Nesterov subgradient, LP refinement,
    mirror descent, and coordinate mass redistribution with multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_diverse_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov Subgradient Warm-up ===
    print(f"\n=== Phase 1: Nesterov Subgrad n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_smoothed_subgrad(best_seq, t_alloc, T_start=2.0, T_end=400.0,
                                            lr_init=0.06, n_stages=10, steps_per_stage=3000)
        try_update(a_ng, "subgrad_main")

    # === Phase 2: Mirror Descent ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_subgrad(best_seq, t_alloc, T=50.0, lr_init=0.015, steps=20000)
        try_update(a_md, "mirror")

    # === Phase 3: LP Refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "lp_main")

    # === Phase 4: Nesterov Subgrad Round 2 (higher temps) ===
    print(f"\n=== Phase 4: Subgrad Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_ng2, _ = nesterov_smoothed_subgrad(best_seq, t_alloc, T_start=10.0, T_end=800.0,
                                              lr_init=0.02, n_stages=8, steps_per_stage=2000)
        try_update(a_ng2, "subgrad_2")

    # === Phase 5: Coordinate Mass Redistribution ===
    print(f"\n=== Phase 5: Coord Redistribution (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_mass_redistribute(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "coord_redist")

    # === Phase 6: Multi-start with diverse inits ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        modes = [1, 3, 4, 5, 2]
        for mode in modes:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.20, 120.0)
            if t_sub < 25:
                break
            a_init = make_diverse_init(n_best, mode, rng)
            print(f"  mode={mode}, budget={t_sub:.0f}s, init_score={score_normalized(a_init):.6f}")
            # Quick subgrad + LP
            a_sub, _ = nesterov_smoothed_subgrad(a_init, t_sub * 0.5, T_start=2.0, T_end=200.0,
                                                  lr_init=0.08, n_stages=6, steps_per_stage=1500)
            try_update(a_sub, f"start_m{mode}_sg")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = nesterov_lp_refine(a_sub, min(rem * 0.4, 40.0), solver=solver, n_max_iters=3000)
                try_update(a_sub2, f"start_m{mode}_lp")

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = sorted(set([n_best + d for d in [-48, -24, -12, 12, 24, 48, 96]
                              if 256 <= n_best + d <= 4096]))
    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.18, 100.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_smoothed_subgrad(init_alt, t_total * 0.5, T_start=2.0, T_end=200.0,
                                              lr_init=0.06, n_stages=6, steps_per_stage=1500)
        try_update(a_alt, f"alt_n{alt_n}_sg")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.4, 30.0), solver=solver, n_max_iters=2000)
            try_update(a_alt2, f"alt_n{alt_n}_lp")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 60.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=10000)
        try_update(a_fin, "final_lp")

    # === Phase 9: Final coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.20, 30.0)
        a_ps, _ = coordinate_mass_redistribute(best_seq, t_ps, n_steps=5000)
        try_update(a_ps, "final_coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
