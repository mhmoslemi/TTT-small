def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_obj_and_grad(a, beta):
    """Compute smoothed score and its gradient w.r.t. a.
    
    Score(a) = 2*n*max(conv(a,a)) / (sum a)^2
    Since score is scale-invariant, we work with x = a/sum(a):
    Score = 2*n*max(conv(x,x))
    Smoothed: 2*n*(1/beta)*log(sum_k exp(beta*conv_k(x)))
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    x = a / s
    n = len(a)
    conv = fast_convolve(x)
    scaled = beta * conv
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp  # softmax weights

    # Smoothed objective value
    obj = 2.0 * n * (max_s + np.log(sum_exp)) / beta

    # Gradient computation
    # (grad_x)_i = sum_{j=0}^{n-1} w[i+j] * x[j]
    idx = np.arange(n)
    M = idx[:, None] + idx[None, :]  # M[i,j] = i+j, shape (n,n)
    M = np.minimum(M, 2 * n - 2)  # clip to valid range
    grad_x = np.sum(w[M] * x[None, :], axis=1)

    # weighted_avg = sum_k w_k * conv_k
    weighted_avg = np.dot(w, conv)

    # dS/da_i = (2n/s) * (grad_x_i - weighted_avg)
    grad = (2.0 * n / s) * (grad_x - weighted_avg)

    return obj, grad


def l-bfgs_optimize(a0, beta, time_budget, max_iter=500):
    """Run L-BFGS-B on the smoothed objective."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s

    best_obj = 2.0 * len(a) * float(fast_convolve(a).max())
    best_a = a.copy()

    def obj_func(x):
        val, _ = smooth_obj_and_grad(x, beta)
        return val

    def grad_func(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g

    result = minimize(
        obj_func, a, method='L-BFGS-B', jac=grad_func,
        bounds=[(0.0, None)] * len(a),
        options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-10}
    )

    if result.x is not None:
        a_new = np.maximum(result.x, 0.0)
        sn = a_new.sum()
        if sn > 1e-10:
            a_new = a_new / sn
            true_score = 2.0 * len(a_new) * float(fast_convolve(a_new).max())
            if true_score < best_obj:
                best_obj = true_score
                best_a = a_new.copy()

    elapsed = time.monotonic() - t0
    return best_a, best_obj, elapsed


def mirror_descent(a0, beta, time_budget, eta0=0.1):
    """Mirror descent with multiplicative weights update."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    a = a / s
    n = len(a)
    best_score = 2.0 * n * float(fast_convolve(a).max())
    best_a = a.copy()
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        x = a
        conv = fast_convolve(x)
        scaled = beta * conv
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        w = exp_s / exp_s.sum()

        # Compute gradient (same as in smooth_obj_and_grad but normalized by n)
        idx = np.arange(n)
        M = idx[:, None] + idx[None, :]
        M = np.minimum(M, 2 * n - 2)
        grad_x = np.sum(w[M] * x[None, :], axis=1)
        weighted_avg = np.dot(w, conv)
        grad = grad_x - weighted_avg

        # Multiplicative update
        eta = eta0 / (1.0 + 0.01 * it)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        # Check score periodically
        if it % 20 == 0:
            sc = 2.0 * n * float(fast_convolve(a).max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # Final evaluation
    sc = 2.0 * n * float(fast_convolve(a).max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5


def exact_coord_descent(a0, time_limit, label=""):
    """Exact coordinate descent with golden section 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx2 in range(n_aff):
                k = k_lo + idx2
                if k == 2 * i:
                    c_vals[idx2] = conv[k] - v * v
                else:
                    c_vals[idx2] = conv[k] - 2.0 * v * a_partners[idx2]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx if has_quad else 0,
                              _cs=current_sum, _v=v, _nn=n):
                ns = _cs + (x - _v)
                if ns < 0.01:

                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _nn * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)

                changes = 2.0 * delta * a_partners

                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

def init_flat_spectrum(n, rng):
    """Flat power spectrum initialization."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_nonneg(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    k_max = int(np.argmax(conv))

    # Find the pairs (i, k_max - i) that contribute to the peak
    i_start = max(0, k_max - n + 1)
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1)
    contributions = a[idx] * a[k_max - idx]

    # Reduce the largest contributors
    top_k = max(1, len(idx) // 4)
    top_indices = idx[np.argsort(contributions)[-top_k:]]

    for ii in top_indices:
        a[ii] *= (1.0 - strength * rng.uniform(0.5, 1.0))

        if 0 <= partner < n and partner != ii:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed (softmax) objective with L-BFGS-B - multi-temperature schedule
2. Mirror descent (multiplicative weights) with adaptive step size
3. Exact coordinate descent with golden section search for final polishing
4. Targeted peak-breaking heuristic to escape local minima

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_obj_and_grad(a, beta):
    """Compute smoothed score and gradient w.r.t. a.
    
    Score(a) = 2*n*max(conv(a,a)) / (sum a)^2
    Scale-invariant, so work with x = a/sum(a): Score = 2*n*max(conv(x,x))
    Smoothed: 2*n*(1/beta)*log(sum_k exp(beta*conv_k(x)))
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    x = a / s
    n = len(a)
    conv = fast_convolve(x)
    scaled = beta * conv
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp

    obj = 2.0 * n * (max_s + np.log(sum_exp)) / beta

    idx = np.arange(n)
    M = idx[:, None] + idx[None, :]
    M = np.minimum(M, 2 * n - 2)
    grad_x = np.sum(w[M] * x[None, :], axis=1)
    weighted_avg = np.dot(w, conv)
    grad = (2.0 * n / s) * (grad_x - weighted_avg)

    return obj, grad
