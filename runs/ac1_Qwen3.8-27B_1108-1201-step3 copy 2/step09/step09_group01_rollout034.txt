```python
def peak_reduction(a0, time_limit):
    n = len(a0)
    a = a0.copy() / a0.sum()
    
    while time_remaining():
        conv = fast_conv(a)
        M = conv.max()
        
        # Find active peaks
        active = conv >= 0.9 * M
        peak_indices = np.where(active)[0]
        
        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for k in peak_indices:
            for i in range(max(0, k-n+1), min(n, k+1)):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[i] * a[j]
        
        # Update: reduce high-pressure coords, increase low-pressure
        p_mean = pressure.mean()
        eta = 0.01 * (M - conv.min())  # adaptive step
        a_new = a * (1 - eta * (pressure - p_mean) / (pressure.std() + 1e-10))
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        if fast_conv(a_new).max() < M:
            a = a_new
    
    return a
```

```python
import numpy as np

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if np_ not in n_values:
            n_values.insert(0, np_)
    
    candidates = []
    
    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 120:
            break
        
        # Generate diverse inits
        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
        for idx in range(6):
            inits.append((f"div{idx}", init_diverse(n, rng, idx)))
        
        for name, cand in inits:
            t_left = 120.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(3.0, min(6.0, t_left / 4.0))
            a_opt, s_opt, _ = fast_coord_descent(cand, t_alloc, use_bisection=False, label=f"p1_{name}_{n}")
            candidates.append((s_opt, a_opt, n, name))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:3]]))
    
    for s_c, a_c, n_c, it_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")
    
    # === Phase 2: Intensive fast CD with bisection ===
    print(f"\n=== Phase 2: Intensive CD+Bisection (best={best_score:.8f}) ===")
    t2_start = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()
        
        for rnd in range(8):
            rem = deadline - time.monotonic() - 600
            if rem < 10:
                break
            t_alloc = min(rem * 0.2, 100.0)
            a_opt, s_opt, sweeps = fast_coord_descent(current, t_alloc, use_bisection=True,
                                                      verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_r{rnd}")
            
            if s_opt < best_score * (1 + 1e-6):
                current = a_opt.copy()
            else:
                # Perturb to escape
                conv = fast_conv(current)
                M = conv.max()
                peaks = np.where(conv >= 0.95 * M)[0]
                rng2 = np.random.default_rng(rnd * 100)
                pressure = np.zeros(n_best)
                for k in peaks:
                    for i in range(max(0, k - n_best + 1), min(n_best, k + 1)):
                        pressure[i] += current[k - i]
                pmax = pressure.max()
                if pmax > 1e-20:
                    pressure /= pmax
                current = current * (1.0 - 0.05 * pressure + 0.02 * rng2.standard_normal(n_best))
                current = np.maximum(current, 0)
                current = current / current.sum()
    
    # === Phase 3: Peak-reduction optimization ===
    print(f"\n=== Phase 3: Peak Reduction (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_pr = min(rem * 0.35, 200.0)
            print(f"  PR n={len(best_seq)}, budget={t_pr:.0f}s")
            a_pr, s_pr, steps = peak_reduction_optimize(best_seq, t_pr, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_pr, best_seq, best_score, "peak_reduction")
            
            # Follow up with CD
            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_cd = min(rem * 0.3, 60.0)
                a_cd, s_cd, _ = fast_coord_descent(a_pr, t_cd, use_bisection=True, label="post_pr")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "post_pr_cd")
    
    # === Phase 4: Nesterov Mirror Descent ===
    print(f"\n=== Phase 4: Nesterov MD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_md = min(rem * 0.4, 150.0)
            print(f"  NMD n={len(best_seq)}, budget={t_md:.0f}s")
            a_md, M_md = nesterov_mirror_descent(best_seq, t_md, verbose=True)
            score_md = 2.0 * len(best_seq) * M_md
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "nesterov_md")
            
            # Polish after MD
            rem = deadline - time.monotonic() - 250
            if rem > 15:
                t_pol = min(rem * 0.3, 40.0)
                a_pol, s_pol, _ = fast_coord_descent(a_md, t_pol, use_bisection=True, label="post_md")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "post_md_cd")
    
    # === Phase 5: Simulated Annealing ===
    print(f"\n=== Phase 5: Simulated Annealing (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_sa = min(rem * 0.5, 200.0)
            print(f"  SA n={len(best_seq)}, budget={t_sa:.0f}s")
            a_sa, s_sa, iters = simulated_annealing(best_seq, t_sa, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "simulated_anneal")
            
            # Polish after SA
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_pol = min(rem * 0.4, 50.0)
                a_pol, s_pol, _ = fast_coord_descent(a_sa, t_pol, use_bisection=True, label="post_sa")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "post_sa_cd")
    
    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = [n_cur + 256, n_cur + 512, n_cur - 256, n_cur // 2, n_cur * 3 // 2]
        alt_lengths = [x for x in alt_lengths if 384 <= x <= 3072 and x != n_cur]
        
        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alt = min(rem * 0.4, 40.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alt, use_bisection=True, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
            
            # Quick CD polish on the best alt
            if s_alt < best_score * 1.01:
                rem = deadline - time.monotonic() - 60
                if rem > 10:
                    a_p, s_p, _ = fast_coord_descent(a_alt, min(rem * 0.4, 20.0),
                                                     use_bisection=True, label=f"alt_p{alt_n}")
                    best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"alt_p{alt_n}")
    
    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Multiple rounds of fine CD
        for polish_round in range(4):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            t_fin = min(rem * 0.3, 50.0)
            a_fin, s_fin, _ = fast_coord_descent(best_seq, t_fin, use_bisection=True,
                                                 verbose=False, label=f"fin{polish_round}")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, f"final_{polish_round}")
            
            # Tiny perturbation between rounds to escape micro-local optima
            if polish_round < 3:
                rng3 = np.random.default_rng(200 + polish_round)
                n_b = len(best_seq)
                tiny_pert = 0.005 * rng3.standard_normal(n_b)
                perturbed = best_seq * (1.0 + tiny_pert)
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                # Quick CD on perturbed
                rem = deadline - time.monotonic() - 15
                if rem > 5:
                    a_q, s_q, _ = fast_coord_descent(perturbed, min(rem * 0.3, 15.0),
                                                     use_bisection=True, label=f"fin_p{polish_round}")
                    best_seq, best_score = check_and_update(a_q, best_seq, best_score, f"fin_pert_{polish_round}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
```
