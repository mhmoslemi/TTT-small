def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that (T @ b)[k] = sum_i a_i * b[k-i]."""
    n = len(a)
    size = 2 * n - 1
    # T[k, i] = a[i] if 0 <= k-i <= n-1, else 0
    # T[k, i] = a[i] if k-n+1 <= i <= k
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T


def solve_lp_best_response(y, solver):
    """Given fixed y (sum=1, nonneg), find b minimizing max conv(y,b)."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(y)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """Nesterov dual averaging with LP best response."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (it + 2) * b_new) / (it + 1 + it + 2)
        a = project_simplex(a)
        
        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(hash((it,)) % 2**32)
                y = y + 0.01 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv)."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

                    lr_init=0.06, n_stages=12, steps_per_stage=2000):

    n = len(a0)
    x = project_simplex(a0.copy().astype(np.float64))
    x_prev = x.copy()
    best_a = x.copy()

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

        T = temps[stage]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(y)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """Nesterov dual averaging with LP best response - improved version."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (it + 2) * b_new) / (2.0 * it + 3)
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 6:
                # Perturb to escape local optimum
                noise_scale = 0.005 + 0.005 * (stall_count % 3)
                y = y + noise_scale * rng.standard_normal(n)
                y = project_simplex(y)
                a = a + noise_scale * 0.5 * rng.standard_normal(n)
                a = project_simplex(a)
                stall_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def active_peak_cd(a0, time_limit, n_steps=50000):
    """Coordinate descent focusing on reducing the maximum convolution peak."""
    n = len(a0)
    a = project_simplex(a0.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Exclude edges (less important)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair modification
            for ci in top_idx[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n or i < 0 or i >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try reducing multiple top contributors slightly
            a_trial = a.copy()
            for ci in top_idx[:4]:
                i = idx[ci]
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                # Try next peak
                b_work[peak_k] = -np.inf
                peak_k2 = int(np.argmax(b_work))
                if b_work[peak_k2] < 0:
                    break
                js2 = max(0, peak_k2 - n + 1)
                je2 = min(n - 1, peak_k2)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[peak_k2 - idx2]
                top_idx2 = np.argsort(contribs2)[::-1][:3]
                a_trial2 = a.copy()
                for ci in top_idx2:
                    i2 = idx2[ci]
                    if a_trial2[i2] > 1e-14:
                        a_trial2[i2] *= 0.95
                a_trial2 = project_simplex(a_trial2)
                sc2 = score_normalized(a_trial2)
                if sc2 < best_score - 1e-12:
                    a = a_trial2
                    best_score = sc2
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def make_window_init(n, window_type='hamming'):
    """Create window function initialization."""
    x = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.array([1.0])
    if window_type == 'hamming':
        w = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    elif window_type == 'hanning':
        w = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))
    elif window_type == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    elif window_type == 'blackman_harris':
        w = (0.35875 - 0.48829 * np.cos(2 * np.pi * x / (n - 1))
             + 0.14128 * np.cos(4 * np.pi * x / (n - 1))
             - 0.01168 * np.cos(6 * np.pi * x / (n - 1)))
    elif window_type == 'flattop':
        w = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x / (n - 1))
             + 0.277263158 * np.cos(4 * np.pi * x / (n - 1))
             - 0.083578947 * np.cos(6 * np.pi * x / (n - 1))
             + 0.006947368 * np.cos(8 * np.pi * x / (n - 1)))
    elif window_type == 'triangular':
        w = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    else:
        w = np.ones(n)
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP-based Nesterov refinement, smoothed gradient,
    active peak CD, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = make_window_init(n_init, 'blackman_harris')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP-based Nesterov refinement (main workhorse) ===
    print(f"\n=== Phase 1: LP Refinement n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "lp_main")

    # === Phase 2: Smoothed Nesterov gradient (different parameters) ===
    print(f"\n=== Phase 2: NG Round (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 240.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_smooth(best_seq, t_alloc, T_start=5.0, T_end=500.0,
                                  lr_init=0.04, n_stages=10, steps_per_stage=3000)
        try_update(a_ng, "ng_1")

    # === Phase 3: LP refinement again on updated best ===
    print(f"\n=== Phase 3: LP Refinement Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp2, "lp_2")

    # === Phase 4: Active peak coordinate descent ===
    print(f"\n=== Phase 4: Active Peak CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = active_peak_cd(best_seq, t_alloc, n_steps=50000)
        try_update(a_cd, "peak_cd")

    # === Phase 5: Multi-start with window inits ===
    print(f"\n=== Phase 5: Multi-Start Windows (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        windows = ['hamming', 'blackman', 'hanning', 'flattop']
        for wtype in windows[:3]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_sub = min(rem * 0.25, 180.0)
            if t_sub < 40:
                break
            a_init = make_window_init(n_best, wtype)
            print(f"  Window={wtype}, budget={t_sub:.0f}s")
            # Quick NG
            a_sub, _ = nesterov_smooth(a_init, t_sub * 0.5, T_start=2.0, T_end=300.0,
                                       lr_init=0.08, n_stages=8, steps_per_stage=2000)
            try_update(a_sub, f"win_{wtype}")
            # LP refine
            rem = deadline - time.monotonic()
            if rem > 40:
                a_sub2, _ = nesterov_lp_refine(a_sub, min(rem * 0.3, 60.0), solver=solver,
                                               n_max_iters=10000)
                try_update(a_sub2, f"win_{wtype}_lp")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_offsets = [-80, -40, 40, 80, 120]
    tried = set()
    for delta in alt_offsets:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        alt_n = n_best + delta
        if alt_n in tried or alt_n < 256 or alt_n > 4096:
            continue
        tried.add(alt_n)
        t_total = min(rem * 0.20, 150.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_smooth(init_alt, t_total * 0.5, T_start=3.0, T_end=300.0,
                                   lr_init=0.06, n_stages=8, steps_per_stage=2000)
        try_update(a_alt, f"ng_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 40:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.3, 60.0), solver=solver,
                                           n_max_iters=5000)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 120.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=20000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final peak CD ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.40, 60.0)
        a_ps, _ = active_peak_cd(best_seq, t_ps, n_steps=20000)
        try_update(a_ps, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
