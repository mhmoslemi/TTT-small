    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

        return np.ones(n) / n

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

        try:

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.GLPK, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def coord_descent_fine(a0, time_limit, verbose=False, label=""):
    """
    Very fine coordinate descent with tiny multiplicative steps.
    Tries factors like 0.999, 0.998, 1.001, 1.002 etc.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    # Adaptive step size
    step_size = 0.005

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        total_improvement = 0

        factors = [1.0 - step_size, 1.0 - step_size * 0.5,
                   1.0 + step_size * 0.5, 1.0 + step_size]

        for i in range(n):
            vi = a[i]
            if vi < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            linear_coeff = 2.0 * a[:]
            quad_idx = i

            best_delta = 0.0
            best_f = M

            for f in factors:
                delta = vi * (f - 1.0)
                new_vi = vi + delta
                if new_vi < 0 or new_vi > 1.0:
                    continue

                new_aff = affected + delta * linear_coeff
                new_aff = new_aff.copy()
                new_aff[quad_idx] += delta * delta
                f_val = max(unaff_max, float(new_aff.max()))
                if f_val < best_f - 1e-17:
                    best_f = f_val
                    best_delta = delta

            if best_delta != 0.0 and best_f < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * linear_coeff
                conv[k_start + quad_idx] += best_delta * best_delta
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                total_improvement += 1
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()

        if total_improvement == 0:
            step_size *= 0.7
            if step_size < 1e-5:
                break
        elif sweeps % 100 == 0:
            # Gradually reduce step
            step_size *= 0.95

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CDF {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} step={step_size:.6f}")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
        best_score = 2.0 * n * fast_convolve(best_a).max()

    return best_a, best_score

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    # Temperature schedule
    T0 = 0.01
    T_end = 0.0001
    n_steps = max(1000, int(time_limit * 50))
    cooling = (T_end / T0) ** (1.0 / n_steps)

    T = T0
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and T > T_end:
        it += 1
        T *= cooling

        # Choose a move type
        move_type = rng.random()

        if move_type < 0.4:
            # Peak-targeted: reduce a coordinate that contributes to a peak
            peak_pos = int(rng.integers(0, 2 * n - 1))
            # Coordinates that contribute to conv[peak_pos]
            i_start = max(0, peak_pos - n + 1)
            i_end = min(n - 1, peak_pos)
            if i_end < i_start:
                continue
            i = int(rng.integers(i_start, i_end + 1))
            if a[i] < 1e-16:
                continue
            # Reduce a[i], increase another random coordinate
            j = int(rng.integers(0, n))
            if j == i:
                j = (j + 1) % n
            delta = a[i] * rng.uniform(0.01, 0.1)
            delta = min(delta, a[i])
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            if new_a[i] < 0 or new_a[j] < 0:
                continue
            s_new = new_a.sum()
            if s_new < 0.01:
                continue
            new_a = new_a / s_new
            new_M = float(fast_convolve(new_a).max())
            new_score = 2.0 * n * new_M
            old_score = 2.0 * n * M
            delta_obj = new_score - old_score
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
                a = new_a
                conv = fast_convolve(a)
                M = new_M
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        elif move_type < 0.7:
            # Random coordinate swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            delta = min(a[i], a[j]) * rng.uniform(0.05, 0.5)
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            if new_a[i] < 0 or new_a[j] < 0:
                continue
            new_a = new_a / new_a.sum()
            new_M = float(fast_convolve(new_a).max())
            new_score = 2.0 * n * new_M
            old_score = 2.0 * n * M
            delta_obj = new_score - old_score
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
                a = new_a
                conv = fast_convolve(a)
                M = new_M
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        else:
            # Gaussian perturbation
            noise = rng.standard_normal(n) * T * 0.5
            new_a = a + noise
            new_a = np.maximum(new_a, 0)
            s_new = new_a.sum()
            if s_new < 0.01:
                continue
            new_a = new_a / s_new
            new_M = float(fast_convolve(new_a).max())
            new_score = 2.0 * n * new_M
            old_score = 2.0 * n * M
            delta_obj = new_score - old_score
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
                a = new_a
                conv = fast_convolve(a)
                M = new_M
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): T={T:.6f} best={best_score:.6f} cur={2.0*n*M:.6f}")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
        best_score = 2.0 * n * fast_convolve(best_a).max()

    return best_a, best_score


def lp_nesterov_refine(a0, time_limit, solver=None, verbose=False, label=""):
    """LP best-response with Nesterov (Aggag-Tseng) averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Nesterov sequent (step size 2/(t+2))
        gamma = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma) * a_sequent + gamma * a
        b_sequent = (1.0 - gamma) * b_sequent + gamma * b

        # Evaluate candidates
        candidates = [a, b, a_sequent, b_sequent,
                      0.5 * (a_sequent + b_sequent),
                      0.7 * a_sequent + 0.3 * b_sequent,
                      0.3 * a_sequent + 0.7 * b_sequent]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPN {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
        best_score = 2.0 * n * fast_convolve(best_a).max()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_targeted_perturb(a0, rng, strength=0.03):
    """Reduce coordinates that contribute to peaks, increase others."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    peak_threshold = M * 0.98
    peak_positions = np.where(conv >= peak_threshold)[0]

    if len(peak_positions) == 0:
        return a.copy()

    pressure = np.zeros(n)
    for k in peak_positions[:50]:  # Limit to top 50 peaks
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            pressure[i] += a0[i] * a0[k - i]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n) * 0.1
    a_new = a * (1.0 - strength * pressure + strength * 0.2 * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-scale init + quick CD
    Phase 2: Intensive line-search CD
    Phase 3: Simulated annealing for global exploration
    Phase 4: LP Nesterov refinement
    Phase 5: Fine polishing
    Phase 6: Alternative lengths
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-scale initialization ===
    print("\n=== Phase 1: Multi-scale Init ===")
    t1_start = time.monotonic()
    p1_budget = 180.0

    # Solve small problems hierarchically
    small_best = None
    for n_small in [32, 64, 128, 256, 512]:
        t_left = p1_budget - (time.monotonic() - t1_start)
        if t_left < 10:
            break
        t_alloc = min(t_left * 0.4, 25.0)

        if small_best is not None:
            init = interpolate_seq(small_best, n_small)
        else:
            init = np.ones(n_small) / n_small

        a_opt, s_opt, _ = coord_descent_linesearch(init, t_alloc, label=f"ms_{n_small}")
        if small_best is None or s_opt < 2.0 * 32 * fast_convolve(np.ones(32)/32).max():
            small_best = a_opt
        print(f"  Multi-scale n={n_small}: score={s_opt:.6f}")

    # Interpolate to target lengths
    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))
    if small_best is not None:
        for n_t in [768, 1024]:
            interp = interpolate_seq(small_best, n_t)
            candidates.append((f"ms_{n_t}", interp))

    # Quick CD on each candidate
    for name, cand in candidates:
        t_left = p1_budget - (time.monotonic() - t1_start)
        if t_left < 5:
            break
        t_alloc = min(t_left * 0.3, 20.0)
        a_opt, s_opt, _ = coord_descent_linesearch(cand, t_alloc, label=f"p1_{name}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p1_{name}")

    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: Intensive line-search coordinate descent ===
    print(f"\n=== Phase 2: Intensive CD ({best_score:.6f}) ===")
    t2_start = time.monotonic()
    p2_budget = 650.0

    if best_seq is not None:
        current = best_seq.copy()
        n_best = len(current)

        for rnd in range(10):
            t_left = p2_budget - (time.monotonic() - t2_start)
            if t_left < 15:
                break
            t_alloc = min(t_left * 0.15, 80.0)
            print(f"  CD round {rnd+1}/10, budget={t_alloc:.0f}s, score={best_score:.6f}")

            a_opt, s_opt, sweeps = coord_descent_linesearch(current, t_alloc,
                                                              verbose=True, label=f"cd_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"cd_r{rnd}")

            # Perturb to escape local optimum
            if rnd < 7 and best_seq is not None:
                if rnd % 2 == 0:
                    current = peak_targeted_perturb(best_seq, rng, strength=0.02 + 0.01 * rnd)
                else:
                    # Random restructure
                    n_r = len(best_seq)
                    frac = 0.1 + 0.05 * rnd
                    keep = rng.random(n_r) > frac
                    n_keep = keep.sum()
                    if n_keep < n_r // 4:
                        keep[:] = True
                    a_r = best_seq.copy()
                    a_r[~keep] = 0
                    a_r[keep] *= (1.0 + rng.standard_normal(n_keep) * 0.15)
                    a_r = np.maximum(a_r, 0)
                    s_r = a_r.sum()
                    if s_r > 0.01:
                        current = a_r / s_r
                    else:
                        current = best_seq.copy()
            else:
                current = best_seq.copy() if best_seq is not None else current

    # === Phase 3: Simulated annealing ===
    print(f"\n=== Phase 3: Simulated Annealing ({best_score:.6f}) ===")
    t3_start = time.monotonic()
    p3_budget = 250.0

    if best_seq is not None:
        t_left = p3_budget - (time.monotonic() - t3_start)
        if t_left > 30:
            t_alloc = min(t_left * 0.6, 150.0)
            print(f"  SA budget={t_alloc:.0f}s, n={len(best_seq)}")
            a_sa, s_sa = simulated_annealing(best_seq, t_alloc, rng, verbose=True, label="sa1")
            best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "sa1")

            # Another SA round from perturbed best
            if best_seq is not None and time.monotonic() < deadline - 500:
                perturbed = peak_targeted_perturb(best_seq, rng, strength=0.05)
                t_left2 = p3_budget - (time.monotonic() - t3_start)
                if t_left2 > 20:
                    t_alloc2 = min(t_left2 * 0.5, 80.0)
                    a_sa2, s_sa2 = simulated_annealing(perturbed, t_alloc2, rng, label="sa2")
                    best_seq, best_score = check_and_update(a_sa2, best_seq, best_score, "sa2")

    # === Phase 4: LP Nesterov refinement ===
    print(f"\n=== Phase 4: LP Nesterov ({best_score:.6f}) ===")
    t4_start = time.monotonic()
    p4_budget = 400.0

    if best_seq is not None:
        n_best = len(best_seq)
        # Try LP at current length and maybe one alternative
        lp_lengths = [n_best]
        if n_best != 1024:
            lp_lengths.append(1024)
        lp_lengths = lp_lengths[:2]

        for n_lp in lp_lengths:
            t_left = p4_budget - (time.monotonic() - t4_start)
            if t_left < 60:
                break
            t_alloc = min(t_left * 0.4, 150.0)
            print(f"  LP-N n={n_lp}, budget={t_alloc:.0f}s")

            init_lp = interpolate_seq(best_seq, n_lp)
            a_lp, s_lp = lp_nesterov_refine(init_lp, t_alloc, solver=solver, label=f"lp_{n_lp}")
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, f"lp_{n_lp}")

    # === Phase 5: Final fine polishing ===
    print(f"\n=== Phase 5: Fine Polish ({best_score:.6f}) ===")
    t5_start = time.monotonic()
    p5_budget = 200.0

    if best_seq is not None:
        # Fine CD
        t_left = p5_budget - (time.monotonic() - t5_start)
        if t_left > 30:
            t_alloc = min(t_left * 0.5, 100.0)
            print(f"  FineCD budget={t_alloc:.0f}s")
            a_fine, s_fine = coord_descent_fine(best_seq, t_alloc, verbose=True, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine1")

        # More line-search CD
        t_left = p5_budget - (time.monotonic() - t5_start)
        if t_left > 20:
            t_alloc = min(t_left * 0.4, 70.0)
            a_cd, s_cd, _ = coord_descent_linesearch(best_seq, t_alloc, label="final_cd")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_cd")

    # === Phase 6: Try alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths ({best_score:.6f}) ===")
    t6_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [n_best + 128, n_best - 128, n_best + 256, 896, 1152]:
            if alt_n < 256 or alt_n > 2048:
                continue
            if alt_n == n_best:
                continue
            t_left = deadline - time.monotonic() - 30
            if t_left < 20:
                break
            t_alloc = min(t_left * 0.3, 40.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = coord_descent_linesearch(init_alt, t_alloc, label=f"alt_{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
