```python
def fienup_init(n, n_iters=500):
    N = 1 << (2 * n - 1 - 1).bit_length()
    # Start with a symmetric template
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = a / a.sum()
    
    for it in range(n_iters):
        # FFT
        A = np.fft.rfft(a, N)
        # Target: flat magnitude (geometric mean)
        mag = np.abs(A)
        target_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))
        # Keep the phase
        phase = np.angle(A)
        # New spectrum
        A_new = target_mag * np.exp(1j * phase)
        # IFFT
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a
```

The second approach is what the "local swap search" does, but we can do it more efficiently.

For a swap: increase a[i] by δ and decrease a[j] by δ (maintaining sum = 1).
The change in (a*a)_k is:
δ * (d_k^(i) - d_k^(j))

where d_k^(i) is as defined above (the "sensitivity" of (a*a)_k to a[i]).

We want to choose i, j, δ to minimize max_k (b_k + δ * (d_k^(i) - d_k^(j))).

This is still a minimax problem, but now in one variable δ (with the choice of i, j discrete).

For fixed i, j: minimize_δ max_k (b_k + δ * e_k) where e_k = d_k^(i) - d_k^(j).
This is a 1D convex minimax (max of linear functions), solvable in closed form.

The optimal δ is where the two "worst" constraints intersect. Specifically:
- Let K1 = argmax_k (b_k + 0 * e_k) = argmax_k b_k (the current peak)
- As δ increases, some k with e_k > e_{K1} will become the new peak
- The optimal δ balances these

This is a standard "minimax of linear functions" problem, solvable by finding the upper envelope of lines.

For efficiency, we can approximate: the optimal δ is approximately (b_{K1} - b_{K2}) / (e_{K2} - e_{K1}) where K1 is the current peak and K2 is the "next" peak when we change a[i] and a[j].

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):

    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

def nesterov_bidirectional_lp(a0, time_limit, solver=None, n_max_iters=200000):
    """Bidirectional Nesterov LP: alternates LP for a and for b."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: LP to find best response b to y
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov update for y
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Step 2 (NEW - bidirectional): LP to find best response a to y
        a_lp, M_a = solve_lp_best_response(y, solver)
        if a_lp is not None:
            # Use a_lp directly as the new a (it's already on the simplex)
            a = a_lp.copy()
        else:
            # Fallback: Nesterov formula
            a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
            else:
                a = a / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
                prev_score = best_score
            else:
                stall_count += 1
            if stall_count >= 10:
                rng = np.random.default_rng(it * 7 + 13)
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    BidirLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            if max_Tb > 700:
                return (2.0 * n / T_val) * 700
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            if max_Tb > 700:
                return np.zeros_like(z_in)
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def fienup_spectral_init(n, n_iters=300, seed=0):
    """Fienup-style alternating projections: flat spectrum -> nonneg time domain."""
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1 - 1).bit_length()
    # Start with a smooth template
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0)
    a = a / a.sum()

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: geometric mean magnitude (flat spectrum)
        log_mag = np.log(np.maximum(mag, 1e-300))
        target_mag = np.exp(np.mean(log_mag))
        # Blend: move 10% toward target
        new_mag = mag + 0.1 * (target_mag - mag)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(arr)})")

    # === Phase 0: Quick Fienup spectral init to see if it helps ===
    print(f"\n=== Phase 0: Spectral Init (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_spec = fienup_spectral_init(n_best, n_iters=500, seed=seed)
        try_update(a_spec, "fienup_init")

    # === Phase 1: Mirror Descent ===
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent(best_seq, t_alloc, n_active_frac=0.05)
        try_update(a_md, "mirror_desc_1")

    # === Phase 2: Bidirectional Nesterov LP (key new method) ===
    print(f"\n=== Phase 2: Bidirectional LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_bi, _ = nesterov_bidirectional_lp(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_bi, "bidir_lp_1")

    # === Phase 3: L-BFGS LogSumExp ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=150.0, T_end=0.1, maxiter=12000)
        try_update(a_lb, "lbgfs_1")

    # === Phase 4: More Mirror Descent with tighter active set ===
    print(f"\n=== Phase 4: MD r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_md2, _ = mirror_descent(best_seq, t_alloc, n_active_frac=0.03)
        try_update(a_md2, "mirror_desc_2")

    # === Phase 5: Sensitivity-aware swap search ===
    print(f"\n=== Phase 5: Sensitivity Swap (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_sw, _ = sensitivity_swap_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_sw, "sens_swap")

    # === Phase 6: Alternate lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0, 1.1]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.30, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        cur_best = init_alt.copy()
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        # Bidirectional LP for alt length
        t_lp = t_total * 0.60
        a_alt, _ = nesterov_bidirectional_lp(cur_best, t_lp, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"bidirlp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_md = min(rem * 0.25, 80.0)
            if t_md > 15:
                a_alt2, _ = mirror_descent(best_seq, t_md, n_active_frac=0.04)
                try_update(a_alt2, f"md_n{alt_n}")

    # === Phase 7: Final aggressive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.05, maxiter=10000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_sw = min(rem * 0.25, 60.0)
        a_sw2, _ = sensitivity_swap_search(best_seq, t_sw, n_steps=30000)
        try_update(a_sw2, "final_swap")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_bi_fin = min(rem * 0.35, 50.0)
        a_bi_fin, _ = nesterov_bidirectional_lp(best_seq, t_bi_fin, solver=solver, n_max_iters=20000)
        try_update(a_bi_fin, "final_bidir")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
