    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    momentum = 1.0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Adaptive Nesterov step with momentum
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Nesterov average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                #Perturb to escape
                noise_scale = 0.05 / np.sqrt(stall_count)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def quadratic_smooth_refine(a0, time_limit, maxiter=3000):
    """Minimize L2 norm of autocorrelation using L-BFGS in log domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()
    
    # Work in log domain: a = softmax(z), sum constraint handled by normalization
    z = np.log(np.maximum(a, 1e-300)).copy()
    
    def quad_obj(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_local = softmax_from_z(z)
        c = fast_convolve(a_local)
        return float(np.sum(c * c))
    
    def quad_grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a_local = softmax_from_z(z)
        c = fast_convolve(a_local)
        # grad of ||c||^2 w.r.t. a: 4 * (c ⋆ a)
        h = cross_corr_conv(c, a_local)
        grad_a = 4.0 * h
        # Chain rule through softmax: grad_z = a * (grad_a - a @ grad_a)
        S = np.dot(grad_a, a_local)
        return a_local * (grad_a - S)
    
    res = minimize(
        fun=quad_obj, x0=z, jac=quad_grad, method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                 'maxfun': maxiter * 100, 'disp': False}
    )
    a_new = softmax_from_z(res.x)
    return a_new


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def cross_corr_conv(c, a):
    """Compute h[m] = sum_j c[j+m]*a[j] for m=0..n-1 using FFT.
    c has length 2n-1, a has length n."""
    n = len(a)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    C = rfft(c, fft_size)
    A = rfft(a, fft_size)
    h = irfft(C * np.conj(A), fft_size)[:n]
    return h


def logsumexp_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing for precise max optimization."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_local):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_local = softmax_from_z(z_local)
            b = fast_convolve(a_local)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / stage_T)
            log_sum_exp = max_b + stage_T * np.log(exp_b.sum() / len(b))
            return (2.0 * n) * log_sum_exp

        def jac(z_local):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_local)
            a_local = softmax_from_z(z_local)
            b = fast_convolve(a_local)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / stage_T)
            w = exp_b / exp_b.sum()
            # grad of logsumexp w.r.t. a[m]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)

        iters = max(20, maxiter_per_stage // len(T_stages))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def kkt_polish(a0, time_limit, n_steps=20000):
    """KKT-based active set polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()
        # Active set: positions within 0.1% of max
        threshold = M * 0.9999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute weighted subgradient
        # Weight by proximity to max
        weights = c[active] / c[active].sum()
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            w = weights[i]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w * a[k - idx]
        
        # Projected gradient step
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            lr *= 0.95
            lr = max(lr, 1e-8)

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=30000):
    """Adaptive coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()
        peak_k = int(np.argmax(c))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random small perturbation of top 5
            a_trial = a.copy()
            for pos in top_positions[:5]:
                if a[pos] > 1e-12:
                    a_trial[pos] *= (1.0 + rng.standard_normal() * 0.005)
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break
        else:
            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
