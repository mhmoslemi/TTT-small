Let me also think about the spectral flattening more carefully:

Given a (non-negative, sum=1):
1. Compute A = FFT(a, padded to 2n)
2. The power spectrum is |A|^2
3. The autocorrelation is the IFFT of |A|^2 (with appropriate windowing for linear vs circular)
4. To flatten the autocorrelation, we want to flatten |A|^2... but that's circular autocorrelation.

Actually, for linear convolution, the relationship is:
`linear_conv(a,a)[k] = circular_conv(pad(a), pad(a))[k]` where pad adds n-1 zeros.

So in the DFT domain (with padding to length 2n-1 or next power of 2):
`DFT(circ_conv(pad(a), pad(a))) = |DFT(pad(a))|^2`

So the linear autocorrelation is the IDFT of |DFT(pad(a))|^2.

To flatten the autocorrelation, we want |DFT(pad(a))|^2 to be flat, which means |DFT(pad(a))| should be constant.

The Gerchberg-Saxton approach:
1. Start with some A (DFT coefficients) with |A| roughly flat and A[0] = sum(a) = 1.
2. Compute a_padded = IDFT(A).
3. Keep only the first n elements, clip to non-negative, renormalize.
4. Recompute A = DFT of the padded sequence.
5. Modify |A| to be more flat (e.g., |A| = |A|^0.5, or clip |A| to [min, max]).
6. Repeat.

This is a reasonable heuristic. Let me implement it.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def p_norm_smooth(z, p):
    """Objective: 2n * (sum c_k^p)^(1/p) with a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_grad(z, p):
    """Gradient of p-norm objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=6, max_iter_per_stage=500):
    """L-BFGS with adaptive p-norm smoothing."""
    n = len(a0)
    a = np.maximum(a0, 1e-300).astype(np.float64)
    s = a.sum()
    a = a / s
    z = np.log(a)
    z = z - z.max()
    t0 = time.monotonic()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_it = min(max_iter_per_stage, int(remaining * 50))
        if max_it < 10:
            break
        
        try:
            res = minimize(
                fun=lambda z_val: p_norm_smooth(z_val, stage_p),
                x0=z,
                jac=lambda z_val: p_norm_grad(z_val, stage_p),
                method='L-BFGS-B',
                options={'maxiter': max_it, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_it * 100, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at p={stage_p:.1f}: {e}")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def spectral_flatten(a0, time_limit, n_iters=200, flatten_power=0.7):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Forward FFT of zero-padded a
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        
        # Flatten magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress dynamic range of magnitude
        mag_min = mag.max() * 0.1  # floor at 10% of max
        mag = np.maximum(mag, mag_min)
        mag = mag ** flatten_power
        # Normalize so DC component gives sum = 1
        mag = mag / mag[0] * a.sum()  # Keep DC proportional
        
        # Actually, let's just scale to keep sum = 1
        # DC component A[0] = sum(a) should remain 1 after inverse
        A_new = mag * np.exp(1j * phase)
        A_new[0] = 1.0  # Ensure sum = 1
        
        # Inverse FFT
        a_padded_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_padded_new[:n], 0)

        if s_new < 1e-15:

        a_new = a_new / s_new
        
        sc = score(a_new)
        if sc < best_sc:

            best_a = a_new.copy()
        a = a_new

def bottleneck_refine(a0, time_limit, max_steps=5000):

    n = len(a0)

            mask[top_i] = False
            others = a_trial[mask]
            osum = others.sum()
            if osum < 1e-15:
                continue
            a_trial[mask] = others + diff * others / osum
            sc = score(a_trial)
            if sc < best_sc - 1e-14:
                a = a_trial
                best_sc = sc
                best_a = a.copy()
                improved = True
                break

                j = k_star - i
                if i == j or j < 0 or j >= n:

                            best_a = a.copy()

                    break

def interpolate(a, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    return a_new

    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def _pnorm_grad(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


class PNormFunc:
    def __init__(self, p):
        self.p = p
    def __call__(self, z):
        return _pnorm_val(z, self.p)

class PNormJac:
    def __init__(self, p):
        self.p = p
    def __call__(self, z):
        return _pnorm_grad(z, self.p)


def pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=6, max_iter_per_stage=400):
    n = len(a0)
    a = np.maximum(a0, 1e-300).astype(np.float64)
    s = a.sum()
    a = a / s
    z = np.log(a)
    z = z - z.max()
    t0 = time.monotonic()
    
    temps = np.geomspace(p_start, p_end, n_stages)
    
    for si, stage_p in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_it = min(max_iter_per_stage, max(10, int(remaining * 30)))
        
        pf = PNormFunc(stage_p)
        pj = PNormJac(stage_p)
        
        try:
            res = minimize(
                fun=pf,
                x0=z,
                jac=pj,
                method='L-BFGS-B',
                options={'maxiter': max_it, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_it * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


# --- Spectral flattening ---

def spectral_flatten(a0, time_limit, n_iters=300, flatten_power=0.6):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        mag = np.maximum(mag, mag.max() * 0.05)
        mag = mag ** flatten_power
        
        # Preserve DC
        A_new = mag * np.exp(1j * phase)
        A_new[0] = 1.0
        
        a_new_padded = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new_padded[:n], 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Annealing: increase flatten power over time
        # (already fixed, but could adapt)
        a = a_new
    
    return best_a, best_sc


# --- Bottleneck refinement ---

def bottleneck_refine(a0, time_limit, max_steps=10000):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        k_star = int(np.argmax(c))
        
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contribs = a[idx] * a[k_star - idx]
        
        improved = False
        
        # Single coordinate
        top_i = idx[np.argmax(contribs)]
        for factor in [0.97, 0.93, 0.88, 0.80]:
            a_trial = a.copy()
            diff = a_trial[top_i] * (1 - factor)
            if diff < 1e-16:
                continue
            a_trial[top_i] -= diff
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others = a_trial[mask]
            osum = others.sum()
            if osum < 1e-15:
                continue
            a_trial[mask] = others + diff * others / osum
            sc = compute_score(a_trial)
            if sc < best_sc - 1e-14:
                a = a_trial
                best_sc = sc
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
        
        # Pair reduction
        if not improved:
            top_pairs = np.argsort(contribs)[::-1][:5]
            for ci in top_pairs:
                i = int(idx[ci])
                j = k_star - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.95, 0.95), (0.90, 0.95), (0.95, 0.90), (0.88, 0.92)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial /= st
                    sc = compute_score(a_trial)
                    if sc < best_sc - 1e-14:
                        a = a_trial
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Perturb to escape
                rng = np.random.default_rng(step + 12345)
                scale = min(0.05, 0.01 * (no_improve // 100))
                a_trial = a + scale * rng.standard_normal(n)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a = a_trial / st
                no_improve = 0
                if step > max_steps * 0.8:
                    break
    
    return best_a, best_sc


# --- Utilities ---

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    return a_new


def make_initial(n, variant, rng):
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.ones(n) * 0.9 + 0.1 * rng.random(n)
    elif variant == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 2:
        a = np.exp(-3 * x * (1 - x))
    elif variant == 3:
        a = np.exp(-2 * x) + 0.1
    elif variant == 4:
        a = np.exp(-3 * (x - 0.3))
    elif variant == 5:
        a = (1 - x) ** 0.5 + 0.05
    elif variant == 6:
        a = np.exp(-10*(x-0.2)**2) + np.exp(-10*(x-0.8)**2) + 0.05
    elif variant == 7:
        a = x + 0.1
    elif variant == 8:
        a = np.exp(-5 * x) + 0.2
    elif variant == 9:
        # Two-scale: broad + fine structure
        a = np.exp(-3*x*(1-x)) + 0.1 * np.sin(20*np.pi*x)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def eval_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(state, a, label, start_time):

    return False

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
- Multi-length screening with diverse structured initializations

- Spectral flattening (Gerchberg-Saxton style heuristic)

- Length neighborhood exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


# --- Core computation ---

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


# --- p-norm smoothing functions ---

def _pnorm_val(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def _pnorm_grad(z, p):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


class PNormObj:
    def __init__(self, p):
        self.p = p
    def __call__(self, z):
        return _pnorm_val(z, self.p)


class PNormJacobian:
    def __init__(self, p):
        self.p = p
    def __call__(self, z):
        return _pnorm_grad(z, self.p)


def pnorm_lbfgs(a0, time_limit, p_start=10.0, p_end=500.0, n_stages=6, max_iter_per_stage=400):
    """L-BFGS with adaptive p-norm smoothing schedule in log-softmax space."""
    n = len(a0)

    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)

        pj = PNormJacobian(stage_p)
