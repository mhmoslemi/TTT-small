```python
def nesterov_subgrad(a0, time_limit, lr0=0.01, momentum0=0.9):
    n = len(a0)
    a = a0.copy()
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    lr = lr0
    momentum = momentum0
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        b = fast_convolve(y)
        peak_k = int(np.argmax(b))
        grad = grad_autocorr_at(y, peak_k)
        
        # Nesterov update
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        s = x_new.sum()
        if s < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s
        
        # Momentum
        y = x_new + momentum * (x_new - x_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        y = y / s
        
        x_prev = x_new
        
        # Evaluate
        score = eval_obj(x_new)
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
        
        # Adapt learning rate
        if it % 100 == 0:
            lr = lr0 / (1 + it * 0.001)
            momentum = min(0.99, momentum0 + it * 0.00001)
        
        if it % 10000 == 0:
            print(f"  NSG iter={it}: score={best_score:.8f}, lr={lr:.6f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exponential gradient (mirror descent)
with Nesterov acceleration, multi-length exploration, LogSumExp L-BFGS refinement,
and intensive coordinate descent. Key innovations: direct optimization of the true
non-relaxed objective (no LP gap), adaptive step-size with backtracking, and
structured restarts to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_obj(a):
    """Fast local evaluation (no function call overhead)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Project to probability simplex."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_at_peak(a, k):
    """Gradient of (conv(a,a))[k] w.r.t. a. O(n) using shift."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return grad
    j_range = np.arange(lo, hi + 1)
    m_range = k - j_range
    grad[j_range] = 2.0 * a[m_range]
    return grad


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj(z, n, T):
    """Smoothed objective: (2n/T) * log(sum exp(b/T)) in log-domain."""
    a = softmax_stable(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def logsumexp_grad(z, n, T):
    """Gradient of smoothed objective in log-domain. Uses FFT for speed."""
    a = softmax_stable(z)
    b = fast_convolve(a)
    size = 2 * n - 1
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is a cross-correlation: compute via FFT
    significant = w > 1e-12
    n_sig = significant.sum()
    
    if n_sig * n < size * n * 0.3:
        # Sparse path
        grad_a = np.zeros(n)
        sig_idx = np.where(significant)[0]
        for k in sig_idx:
            wk = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]
    else:
        # Dense path via FFT
        w_rev = w[::-1]
        a_pad = np.zeros(size)
        a_pad[:n] = a
        fft_sz = 1 << (2 * size - 3 - 1).bit_length()
        Wr = rfft(w_rev, fft_sz)
        Ap = rfft(a_pad, fft_sz)
        conv = irfft(Wr * Ap, fft_sz)[:2 * size - 1]
        # grad_a[j] = 2 * conv[size-1-j] for j in [0, n-1]
        grad_a = 2.0 * conv[size - 1 : size - n - 1 : -1]
    
    # Log-domain gradient
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS refinement with LogSumExp smoothing in log-domain."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_eval):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return logsumexp_obj(z_eval, n, T)
        
        def jac(z_eval):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_eval)
            return logsumexp_grad(z_eval, n, T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',

        except Exception:

    """
    Exponential gradient (mirror descent) with optional Nesterov acceleration.
    Works in log-domain for natural handling of non-negativity.
    Minimizes max_k (conv(a,a))[k] subject to a in simplex.
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300)).copy()

        lr = 2.0 / np.sqrt(n)

    z_prev = z.copy()

            z_y = z + momentum * (z - z_prev)

def coordinate_descent(a0, time_limit, n_rounds=200000):
    """
    Intensive coordinate descent: find peak, reduce top contributors,
    redistribute to low-impact positions.

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()

        # Find top contributors to the peak
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Also consider the peak's neighbors (within a few positions)
        # For simultaneous reduction of multiple peaks
        
        improved = False
        # Strategy 1: Reduce top-1 contributor
        top_pos = idx[np.argmax(contribs)]
        if a[top_pos] > 1e-10:
            for factor in [0.999, 0.998, 0.997, 0.995, 0.99]:
                a_trial = a.copy()
                a_trial[top_pos] *= factor
                diff = a[top_pos] - a_trial[top_pos]
                if diff < 1e-15:
                    continue
                # Redistribute proportionally to all other elements
                mask = np.ones(n, dtype=bool)
                mask[top_pos] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = normalize(a_trial)
                score = eval_obj(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and len(idx) >= 2:
            top2 = idx[np.argsort(contribs)[::-1][:2]]

                a_trial = a.copy()
                for pos in top2:
                    if a_trial[pos] > 1e-10:

                a_trial = normalize(a_trial)

                    transfer = a[top_pos] * eps

                    score = eval_obj(a_trial)

def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                n_cur = len(best_a)
                x_lin = np.linspace(0, 2 * np.pi, n_cur)
                freq = rng.uniform(1, 8)
                amp = 0.02 / np.sqrt(no_improve_count)
                perturbation = amp * np.cos(freq * x_lin + rng.uniform(0, 2 * np.pi))
                best_a = normalize(best_a + perturbation)
                no_improve_count = 0
        
        phase_idx += 1
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with:
    1. Multi-length quick scan to find optimal length
    2. Deep optimization using exponential gradient + L-BFGS + coordinate descent
    3. Intensive local search with structured restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # --- Load parent sequence if available ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_obj(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_obj(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best_initial = len(best_seq)
    
    # === Phase 1: Quick multi-length exploration ===
    print(f"\n=== Phase 1: Multi-length exploration (best so far: {best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.20, 300.0)
    
    if t_phase1 > 60:
        candidate_lengths = [800, 1000, 1200, 1400, 1600, 1800, 2000]
        # Remove lengths too close to initial
        candidate_lengths = [n for n in candidate_lengths if abs(n - n_best_initial) > 100]
        # Add the initial length
        candidate_lengths.append(n_best_initial)
        candidate_lengths = sorted(set(candidate_lengths))
        
        per_length_time = t_phase1 / max(1, len(candidate_lengths))
        print(f"  Testing {len(candidate_lengths)} lengths, ~{per_length_time:.0f}s each")
        
        results = []
        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 100:
                break
            t_rem = deadline - time.monotonic()
            if t_rem < 20:
                break
            
            # Initialize: interpolate parent or use structured init
            if parent is not None and len(parent) > 50:
                init = interpolate_sequence(parent, n_try)
            else:
                init = structured_init(n_try, rng)
            
            t_alloc = min(per_length_time * 0.8, t_rem * 0.5)
            if t_alloc < 10:
                continue
            
            print(f"  Trying n={n_try} ({t_alloc:.0f}s)...")
            
            # Quick EG optimization
            a_quick, _ = exponential_gradient(init, t_alloc, use_nesterov=True)
            score = eval_obj(a_quick)
            results.append((score, n_try, a_quick))
            print(f"    n={n_try}: score={score:.8f}")
            
            if score < best_score:
                best_score = score
                best_seq = a_quick.copy()
        
        # Sort results and pick best
        results.sort(key=lambda x: x[0])
        if results:
            print(f"  Best from phase 1: n={results[0][1]}, score={results[0][0]:.8f}")
    
    # === Phase 2: Deep optimization on best length ===
    print(f"\n=== Phase 2: Deep optimization n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.45, 700.0)
    
    if t_phase2 > 60:
        print(f"  Budget: {t_phase2:.0f}s")
        a_opt, _ = multi_phase_optimize(best_seq, t_phase2)
        score = eval_obj(a_opt)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 2 result: {best_score:.8f}")
    
    # === Phase 3: Try second-best length ===
    rem = deadline - time.monotonic()
    if rem > 120 and len(results) > 1:
        # Try a different length from results
        for (sc, n_alt, a_alt) in results[1:4]:
            if time.monotonic() > deadline - 60:
                break
            if len(best_seq) == n_alt:
                continue
            t_rem = deadline - time.monotonic()
            if t_rem < 40:
                break
            t_alloc = min(t_rem * 0.3, 150.0)
            print(f"\n  Trying alt length n={n_alt} ({t_alloc:.0f}s)...")
            a_alt_opt, _ = multi_phase_optimize(a_alt, t_alloc)
            sc2 = eval_obj(a_alt_opt)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_alt_opt.copy()
                print(f"  NEW BEST at n={n_alt}: {sc2:.8f}")
    
    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # L-BFGS with very low temperature
        t_lbfs = min(rem * 0.3, 60.0)
        if t_lbfs > 15:
            T_final = [5.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.05, 0.03, 0.02]
            a_fin = lbgfs_refine(best_seq, t_lbfs, T_stages=T_final,
                                  maxiter_per_stage=4000)
            sc = eval_obj(a_fin)
            if sc < best_score:
                best_score = sc
                best_seq = a_fin.copy()
                print(f"  L-BFGS final: {sc:.8f}")
        
        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.4, 50.0)
            if t_cd > 10:
                a_cd, _ = coordinate_descent(best_seq, t_cd, n_rounds=50000)
                sc = eval_obj(a_cd)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_cd.copy()
                    print(f"  CD final: {sc:.8f}")
        
        # One more EG pass
        rem = deadline - time.monotonic()
        if rem > 15:
            t_eg = min(rem * 0.4, 30.0)
            if t_eg > 5:
                a_eg, _ = exponential_gradient(best_seq, t_eg, use_nesterov=True)
                sc = eval_obj(a_eg)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_eg.copy()
                    print(f"  EG final: {sc:.8f}")
    
    # === Final verification ===
    if best_seq is not None:
        final_score = eval_obj(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")
    
    # Return as list of floats
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
