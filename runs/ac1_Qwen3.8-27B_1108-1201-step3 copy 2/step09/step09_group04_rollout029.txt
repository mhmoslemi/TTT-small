For large τ, this concentrates on the argmax, and `∇f_τ ≈ ∇c_{argmax}`.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def logsumexp_obj_and_grad(z, tau):
    """Compute objective and gradient of 2n * LSE(tau*c)/tau where c=convolve(softmax(z),softmax(z))."""
    a = softmax(z)
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = tau * (c - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    obj = 2.0 * n * (c_max + log_sum / tau)
    # Softmax weights over c
    weights = exp_shifted / exp_shifted.sum()
    # grad_a = 2 * correlate(weights, a, 'valid')
    corr = np.correlate(weights, a, mode='valid')
    grad_a = 2.0 * corr
    # grad_z = 2n * a * (grad_a - dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - np.dot(grad_a, a))
    return obj, grad_z


def lbfgs_logsumexp(a0, time_limit, tau_start=20.0, tau_end=20000.0, n_stages=10):
    """L-BFGS optimization with increasing log-sum-exp temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    taus = np.geomspace(tau_start, tau_end, n_stages)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for si, tau in enumerate(taus):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 10:
            break
        # Allocate iterations proportional to remaining time, more for higher tau
        fraction = (si + 1) / n_stages
        iters = max(50, int(2000 * (1.0 + fraction)))

        def func(zv):
            val, _ = logsumexp_obj_and_grad(zv, tau)
            return val

        def jac(zv):
            _, g = logsumexp_obj_and_grad(zv, tau)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_cur = softmax(z)
        score = 2.0 * n * fast_convolve(a_cur).max()
        if score < best_score:
            best_score = score
            best_a = a_cur.copy()
        print(f"    LSE-LBFGS tau={tau:.0f} ({time.monotonic()-t0:.0f}s): score={score:.6f}")

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the standard simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_refine(a0, time_limit, n_top=5, n_max_iters=500000):
    """Top-k subgradient method with simplex projection and BB step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s

    best_a = a.copy()
    c = fast_convolve(a)
    best_score = 2.0 * n * c.max()
    t0 = time.monotonic()

    prev_grad = None
    prev_a = None
    step = 1.0 / np.sqrt(n)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        threshold = c_max * 0.95
        near_max = np.where(c >= threshold)[0]
        if len(near_max) == 0:
            near_max = np.array([np.argmax(c)])
        k_top = min(n_top, len(near_max))
        top_k = near_max[np.argsort(c[near_max])[::-1]][:k_top]

        # Compute subgradient: average of gradients at top-k positions
        grad = np.zeros(n)
        for k in top_k:
            # grad[j] += 2 * a[k-j] if 0 <= k-j < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                grad[js:je + 1] += 2.0 * a[k - js:k - je][-1] if (k - je) <= (k - js) else 0
                # More carefully: for j in [js, je], a[k-j] where k-j ranges from k-js to k-je
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * a[k - idx]

        grad /= k_top

        # BB step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            if sy > 1e-20:
                step = np.dot(s_vec, s_vec) / sy
                step = np.clip(step, 1e-10, 1.0)

        # Take step
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        # Check improvement
        c_new = fast_convolve(a_new)
        score_new = 2.0 * n * c_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        # If no improvement, reduce step
        if score_new > best_score * 1.001:
            step *= 0.5
            a_new = a - step * grad
            a_new = project_simplex(a_new)

        prev_grad = grad.copy()
        prev_a = a.copy()
        a = a_new

        if (it + 1) % 5000 == 0:
            print(f"    SubGrad iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    t0 = time.monotonic()

    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            a_i = a[i]
            if a_i < 1e-15:
                # Try small values
                candidates = [1e-4 / n, 1e-3 / n]
            else:
                # Golden section around current value
                lo = max(0.0, a_i * 0.3)
                hi = a_i * 2.5 + 1e-4 / n
                if hi <= lo:
                    hi = lo + 1e-4 / n

            # Evaluate at a few points (golden section ~ 12 evals)
            best_t = a_i
            best_c_max = c.max()

            # Store original c for restoration
            c_orig = c.copy()

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~ 0.618
            x1 = lo + (1 - gr) * (hi - lo)
            x2 = lo + gr * (hi - lo)

            def eval_at_t(t):
                """Evaluate max(c) when a[i] = t, with rest renormalized proportionally."""
                # Change a[i] to t, scale rest to maintain sum=1
                rest_sum = 1.0 - t
                if rest_sum < 1e-15:
                    return c_orig.max()  # degenerate
                # c_k changes: for affected k, delta = t - a_i
                # c_k_new = c_k + 2*(t-a_i)*a[k-i]*(rest_sum)/(1-a_i) + ... 
                # Actually, let's just compute directly for simplicity with incremental update
                delta = t - a_i
                if abs(delta) < 1e-18:
                    return c_orig.max()
                # Scale factor for other elements
                if a_i > 1e-15:
                    scale = rest_sum / (1.0 - a_i)
                else:
                    scale = 1.0  # a_i was ~0
                # c_k = sum_{p+q=k} a_p * a_q
                # New: a_i becomes t, others become a_j * scale (for j != i)
                # c_k_new = c_k + delta * a_{k-i} * scale + a_i * delta * scale + delta^2 
                #   Wait, this is getting complicated. Let me just recompute c for affected k.
                # 
                # c_k_new = c_k_old - (terms with old a_i) + (terms with new a_i and scaled others)
                # Terms with a_i: a_i*a_{k-i} (twice if k!=2i, once if k==2i as a_i^2)
                # New terms: t * (a_{k-i} * scale) + (a_{k-i} * scale) * t + t^2 * [k==2i]
                #   = 2*t*a_{k-i}*scale + t^2*[k==2i]  (if k-i != i)
                #   = 2*t*a_i*scale + t^2  (if k-i == i, i.e., k=2i) -- but a_{k-i} becomes t, not scaled
                # 
                # Actually let me just directly compute the new c_k for affected k:
                # c_k_new = sum_{p+q=k, p!=i, q!=i} (a_p*scale)*(a_q*scale) 
                #          + sum_{p=i} a_i_new * a_q_new + sum_{q=i} a_p_new * a_i_new
                # where a_i_new = t, and a_j_new = a_j * scale for j != i
                #
                # = scale^2 * (sum_{p+q=k, p!=i, q!=i} a_p*a_q) + t*a_{k-i}*scale*2 (if k-i!=i)
                #   + t^2 (if k==2i)
                #
                # sum_{p+q=k, p!=i, q!=i} a_p*a_q = c_k - 2*a_i*a_{k-i} + a_i^2*[k==2i]
                #   (subtracting the terms that involve a_i)
                #
                # So: c_k_new = scale^2 * (c_k - 2*a_i*a_{k-i} + a_i^2*[k==2i])
                #            + 2*t*a_{k-i}*scale * (1 - [k==2i]) + t^2 * [k==2i]
                
                # This is O(1) per k (for the affected k's). Let's do it for all affected k.
                # Affected k: those where a_{k-i} != 0, i.e., k in [i, i+n-1] intersect [0, 2n-2]
                ks_lo = max(0, i)
                ks_hi = min(2 * n - 2, i + n - 1)
                
                c_new = c_orig.copy()
                for k in range(ks_lo, ks_hi + 1):
                    a_ki = a[k - i] if 0 <= k - i < n else 0.0
                    is_self = (k == 2 * i)
                    without_i = c_orig[k] - 2.0 * a_i * a_ki + (a_i * a_i if is_self else 0.0)
                    c_new[k] = scale * scale * without_i
                    if is_self:
                        c_new[k] += t * t
                    else:
                        c_new[k] += 2.0 * t * a_ki * scale
                return c_new.max()

            # Do golden section: ~12 iterations
            for _ in range(12):
                f1 = eval_at_t(x1)
                f2 = eval_at_t(x2)
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    x1 = lo + (1 - gr) * (hi - lo)
                else:
                    lo = x1
                    x1 = x2
                    x2 = lo + gr * (hi - lo)

            best_t = (lo + hi) / 2.0
            f_best = eval_at_t(best_t)

            if f_best < best_c_max - 1e-15:
                # Apply the update
                rest_sum = 1.0 - best_t
                if rest_sum > 1e-15:
                    scale = rest_sum / (1.0 - a_i) if a_i > 1e-15 else 1.0
                    a_new = a * scale
                    a_new[i] = best_t
                    a = a_new
                    c = fast_convolve(a)
                    c_max = c.max()
                    if 2.0 * n * c_max < best_score:
                        best_score = 2.0 * n * c_max
                        best_a = a.copy()
                    improved = True

        if not improved:
            break
        if sweep % 10 == 0:
            print(f"    CoordDescent sweep={sweep} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label, verbose=True):
    """Try to update the best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def optimize_at_length(a0, time_limit, rng):
    """Full optimization pipeline for a fixed-length sequence."""
    best = a0.copy()
    bs = 2.0 * len(a0) * fast_convolve(a0).max()

    # Stage 1: L-BFGS with log-sum-exp
    t1 = time_limit * 0.40
    if t1 > 15:
        a1, s1 = lbfgs_logsumexp(best, t1, tau_start=20.0, tau_end=10000.0, n_stages=8)
        if s1 < bs:
            best, bs = a1, s1

    # Stage 2: Subgradient
    t2 = time_limit * 0.25
    if t2 > 10:
        a2, s2 = subgradient_refine(best, t2, n_top=5, n_max_iters=200000)
        if s2 < bs:
            best, bs = a2, s2

    # Stage 3: Coordinate descent
    t3 = time_limit * 0.25
    if t3 > 10:
        a3, s3 = coordinate_descent(best, t3, n_sweeps_max=100)
        if s3 < bs:
            best, bs = a3, s3

    # Stage 4: Another round of L-BFGS at high tau
    t4 = time_limit * 0.10
    if t4 > 5:
        a4, s4 = lbfgs_logsumexp(best, t4, tau_start=5000.0, tau_end=30000.0, n_stages=4)
        if s4 < bs:
            best, bs = a4, s4

    return best, bs


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining L-BFGS (log-sum-exp), subgradient,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=3)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: Intensive optimization at parent length ===
    print(f"\n=== Phase 1: Intensive optimization n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, s_opt = optimize_at_length(best_state['seq'], t_alloc, rng)
        try_update(best_state, a_opt, "phase1_opt")

    # === Phase 2: Length exploration ===
    print(f"\n=== Phase 2: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        candidate_lengths = []
        base = len(best_state['seq'])
        # Try nearby lengths
        for delta in [-128, -96, -64, -48, -32, -16, 16, 32, 48, 64, 96, 128]:
            alt_n = base + delta
            if 512 <= alt_n <= 2560:
                candidate_lengths.append(alt_n)
        # Also try some round numbers
        for rn in [768, 896, 1024, 1152, 1280, 1536]:
            if 512 <= rn <= 2560 and rn != base:
                candidate_lengths.append(rn)

        # Remove duplicates and sort by proximity to base
        candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - base))

        n_explored = 0
        for alt_n in candidate_lengths[:6]:
            rem = deadline - time.monotonic()
            if rem < 100 or n_explored >= 4:
                break
            n_explored += 1
            t_total = min(rem * 0.20, 200.0)
            if t_total < 50:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_state['seq'], alt_n)
            a_alt, s_alt = optimize_at_length(init_alt, t_total, rng)
            try_update(best_state, a_alt, f"len_{alt_n}")

    # === Phase 3: Random perturbation restarts ===
    print(f"\n=== Phase 3: Perturbation Restarts (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 0.01:
                continue
            perturbed = perturbed / ps

            t_sub = min(rem * 0.25, 150.0)
            if t_sub < 20:
                continue
            a_pert, s_pert = optimize_at_length(perturbed, t_sub, rng)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_final = min(rem * 0.5, 200.0)
        if t_final > 20:
            print(f"  Budget: {t_final:.0f}s")
            a_fin, s_fin = optimize_at_length(best_state['seq'], t_final, rng)
            try_update(best_state, a_fin, "final_polish")

    # === Phase 5: Very fine coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.4, 100.0)
        if t_cd > 10:
            a_cd, s_cd = coordinate_descent(best_state['seq'], t_cd, n_sweeps_max=200)
            try_update(best_state, a_cd, "final_cd")

    # === Phase 6: Another length check at best ===
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for delta in [16, -16, 32, -32]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            alt_n = n_cur + delta
            if alt_n < 256 or alt_n > 3000:
                continue
            init_alt = interpolate_sequence(best_state['seq'], alt_n)
            t_sub = min(rem * 0.4, 60.0)
            if t_sub < 15:
                continue
            a_alt, s_alt = optimize_at_length(init_alt, t_sub, rng)
            try_update(best_state, a_alt, f"final_len_{alt_n}")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
