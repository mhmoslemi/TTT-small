3. Multi-stage L-BFGS LogSumExp annealing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP best response and return (b_optimal, M_optimal, dual_y)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    ineq_constraints = [T @ b_var <= M_var]
    eq_constraint = [cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), ineq_constraints + eq_constraint)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            b_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            ineq_constraints = [T @ b_var <= M_var]
            eq_constraint = [cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), ineq_constraints + eq_constraint)
            try:
                prob.solve(solver=alt)
            except Exception:
                continue
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)

    # Extract dual variables from inequality constraints
    dual_y = None
    try:
        if ineq_constraints[0].dual_value is not None:
            dual_y = np.asarray(ineq_constraints[0].dual_value, dtype=np.float64)
            dual_y = np.maximum(dual_y, 0)
    except Exception:
        dual_y = None

    return b_val, M_val, dual_y


def compute_subgradient_from_dual(a, dual_y):
    """Compute subgradient using LP dual variables for weighting."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b

    if dual_y is not None and dual_y.sum() > 1e-15:
        # Use dual-weighted subgradient
        active_weight = dual_y / dual_y.sum()
    else:
        # Fallback: uniform over top 20%
        threshold = max_b - 0.2 * max_b
        active_mask = b >= threshold
        if active_mask.sum() == 0:
            active_mask = np.zeros(len(b), dtype=bool)
            active_mask[int(np.argmax(b))] = True
        active_weight = active_mask.astype(float)
        active_weight /= active_weight.sum()

    g = np.zeros(n)
    for k in range(len(b)):
        if active_weight[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += active_weight[k] * a[k - idx]
    return g, max_b


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=200000,
                        use_dual_perturb=True, verbose_every=500):
    """LP-Nesterov with dual-aware adaptive step size and perturbation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    rng = np.random.default_rng(42)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, dual_y = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Adaptive step size: Nesterov with gap-based adjustment
        alpha_nesterov = 2.0 / (it + 2)
        gap = best_score - f_lb
        if gap < 1e-8:
            break

        # Gap-based step (clamped)
        alpha_gap = gap / (4.0 * best_score + 1e-15)
        alpha = min(alpha_nesterov, max(alpha_gap, 0.01))

        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        # Nesterov extrapolation for a
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        score_a = 2.0 * n * fast_convolve(a_new).max()
        score_y = 2.0 * n * fast_convolve(y_new).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
            stall_count = 0

        y = y_new
        a = a_new

        # Periodic stall detection and perturbation
        if (it + 1) % 300 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score

            if stall_count >= 6:
                if use_dual_perturb and dual_y is not None and dual_y.sum() > 1e-10:
                    # Dual-guided perturbation: move mass from high-contribution
                    # positions to low-contribution positions
                    b_cur = fast_convolve(y)
                    peak_val = b_cur.max()
                    # Find positions contributing most to near-peak lags
                    threshold = peak_val - 0.05 * peak_val
                    near_peaks = np.where(b_cur >= threshold)[0]
                    contrib = np.zeros(n)
                    for k in near_peaks[:5]:
                        js = max(0, k - n + 1)
                        je = min(n - 1, k)
                        idx = np.arange(js, je + 1)
                        contrib[idx] += y[idx] * y[k - idx]
                    # Perturb: reduce high-contribution, increase low
                    top_idx = np.argsort(contrib)[::-1][:3]
                    low_idx = np.argsort(contrib)[:3]
                    for ti in top_idx:
                        for li in low_idx:
                            if ti != li and y[ti] > 1e-10:
                                amt = y[ti] * 0.03
                                y[ti] -= amt
                                y[li] += amt
                    y = np.maximum(y, 0)
                    sy2 = y.sum()
                    if sy2 > 1e-15:
                        y = y / sy2
                        a = y.copy()
                else:
                    # Random perturbation
                    noise = rng.random(n) * 0.02
                    y = y + noise
                    y = np.maximum(y, 0)
                    sy2 = y.sum()
                    if sy2 > 1e-15:
                        y = y / sy2
                        a = y.copy()
                stall_count = 0

        if (it + 1) % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} alpha={alpha:.4f}")

    return best_a, best_score


def mirror_descent_dual(a0, time_limit, solver=None, n_active_frac=0.08,
                        use_dual_weights=True):
    """Mirror descent with dual-weighted subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    b = fast_convolve(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    lr = 0.3
    no_improve = 0
    last_dual = None

    for step in range(300000):
        if time.monotonic() - t0 > time_limit:
            break

        # Periodically get dual from LP for better subgradient
        if use_dual_weights and solver is not None and step % 500 == 0:
            _, _, dual_y = solve_lp_best_response(a, solver)
            last_dual = dual_y

        if last_dual is not None and last_dual.sum() > 1e-10:
            g, max_b = compute_subgradient_from_dual(a, last_dual)
        else:
            g, max_b = compute_subgradient_from_dual(a, None)

        g_norm_inf = np.abs(g).max()
        if g_norm_inf < 1e-15:
            break

        gap = best_score - f_lb
        if gap < 1e-10:
            break

        polyak_lr = gap / (2.0 * g_norm_inf * best_score + 1e-15)
        alpha = min(lr, polyak_lr)
        alpha = max(alpha, 0.005)

        z_new = z - alpha * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            lr = min(lr * 1.03, 1.5)
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.75
                no_improve = 0
            lr = max(lr, 0.005)

        if step % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirDescDual {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.2, maxiter=10000):
    """L-BFGS on log-sum-exp smoothed objective with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S_stmt = np.dot(grad_a, a)
            return a * (grad_a - S_stmt)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def local_swap_search(a0, time_limit, n_steps=50000):
    """Local search via targeted mass transfers."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b >= threshold)[0]

        contrib = np.zeros(n)
        for k in near_peaks[:4]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        top_src = np.argsort(contrib)[::-1][:6]
        low_dst = np.argsort(contrib)[:6]

        improved = False
        for si in top_src:
            for di in low_dst:
                if si == di or a[si] < 1e-14:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    amount = a[si] * frac
                    a_trial = a.copy()
                    a_trial[si] -= amount
                    a_trial[di] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation when stuck
            if no_improve > 200:
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-14:
                        continue
                    amount = a[i] * 0.02
                    a_trial = a.copy()
                    a_trial[i] -= amount
                    a_trial[j] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:

                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break

        if no_improve > 3000:
            break

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)

    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_dual(best_seq, t_alloc, solver=solver,
                                      n_active_frac=0.06, use_dual_weights=True)
        try_update(a_md, "mir_desc_dual")

    # === Phase 3: Multi-start LP-Nesterov ===
    print(f"\n=== Phase 3: Multi-Start LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_starts = 3
    t_per_start = min(rem / (n_starts + 2), 120.0)
    if t_per_start > 20:
        print(f"  {n_starts} starts, {t_per_start:.0f}s each")
        for start_i in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            # Create perturbed start
            if start_i == 0:
                # Small random perturbation
                noise = rng.random(len(best_seq)) * 0.05
                perturbed = best_seq + noise
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
            elif start_i == 1:
                # Dual-guided perturbation
                _, _, dual_y = solve_lp_best_response(best_seq, solver)
                perturbed = best_seq.copy()
                if dual_y is not None and dual_y.sum() > 1e-10:
                    b_cur = fast_convolve(perturbed)
                    peak_val = b_cur.max()
                    near_peaks = np.where(b_cur >= peak_val * 0.95)[0]
                    contrib = np.zeros(len(perturbed))
                    for k in near_peaks[:5]:
                        js = max(0, k - len(perturbed) + 1)
                        je = min(len(perturbed) - 1, k)
                        idx = np.arange(js, je + 1)
                        contrib[idx] += perturbed[idx] * perturbed[k - idx]
                    top_idx = np.argsort(contrib)[::-1][:5]
                    low_idx = np.argsort(contrib)[:5]
                    for ti in top_idx:
                        for li in low_idx:
                            if ti != li:
                                amt = perturbed[ti] * 0.05
                                perturbed[ti] -= amt
                                perturbed[li] += amt
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
            else:
                # Moderate random perturbation
                noise = rng.random(len(best_seq)) * 0.1
                perturbed = best_seq + noise
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()

            sc_pert = evaluate_safe(perturbed)
            print(f"  Start {start_i}: score={sc_pert:.6f}")

            a_ms, _ = nesterov_lp_refine(perturbed, t_per_start, solver=solver,
                                         n_max_iters=50000, use_dual_perturb=True,
                                         verbose_every=200)
            try_update(a_ms, f"multistart_{start_i}")

    # === Phase 4: L-BFGS polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.1, maxiter=12000)
        try_update(a_lb, "lbgfs_polish")

    # === Phase 5: Alternative lengths ===
    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for candidate_n in [n_best + 64, n_best + 128, n_best - 64, n_best * 2, n_best // 2]:
        if 384 <= candidate_n <= 4000 and candidate_n != n_best:
            alt_lengths.append(candidate_n)
    alt_lengths = sorted(set(alt_lengths))[:2]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.30, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        print(f"  Interpolated score: {sc_alt:.6f}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()

        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver,
                                      n_max_iters=30000, use_dual_perturb=True,
                                      verbose_every=200)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_md = min(rem * 0.3, 40.0)
            if t_md > 10:
                a_alt2, _ = mirror_descent_dual(best_seq, t_md, solver=solver,
                                                n_active_frac=0.05, use_dual_weights=True)
                try_update(a_alt2, f"md_n{alt_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 60.0)
        print(f"  L-BFGS: {t_fin:.0f}s")
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.05, maxiter=6000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ls = min(rem * 0.25, 50.0)
        print(f"  LocalSwap: {t_ls:.0f}s")
        a_ls2, _ = local_swap_search(best_seq, t_ls, n_steps=30000)
        try_update(a_ls2, "final_swap")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.30, 50.0)
        print(f"  Final LP: {t_lp_fin:.0f}s")
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver,
                                         n_max_iters=20000, use_dual_perturb=True,
                                         verbose_every=200)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
