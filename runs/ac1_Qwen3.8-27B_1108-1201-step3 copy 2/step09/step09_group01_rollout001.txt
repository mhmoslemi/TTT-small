Actually, one more important thought: **the receding average in fictitious play**.

In standard fictitious play, the iterate at time `t` is the uniform average of all past best responses:
`ā_t = (1/t) * sum_{s=1}^{t} b_s`

In the receding average (or "sliding window"), we use:
`ā_t = (1/K) * sum_{s=t-K+1}^{t} b_s`

The receding average can be much better in practice because:
- Early iterates are far from the optimum and only add noise
- The sliding window adapts to the current region of the solution space

I'll use a receding average with `K` that grows over time (e.g., `K = min(t, 1000)`).

The Panaitiuk-Shahar algorithm:
- Maintain `x_t` (current iterate) and `y_t` (averaged iterate)
- Best response: `b_t = BR(x_t)`
- Update: `y_{t+1} = y_t + α_t (b_t - y_t)` where `α_t = 2/(t+2)`
- New iterate: `x_{t+1} = BR(y_{t+1})`

This gives `O(1/t)` convergence for the game value.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():

def solve_lp_best_response(a_fixed, T=None, solver=None):

    import cvxpy as cp

def spectral_gs_init(n, rng, n_iters=5000, verbose=False):
    """
    Gerchberg-Saxton style spectral initialization.
    Alternates between:
    - Time domain: enforce non-negativity, normalize, keep first n entries
    - Frequency domain: project |A|^2 toward target flat spectrum
    """
    N = 1 << (2 * n - 1).bit_length()  # Next power of 2 >= 2n-1
    
    # Target: flat convolution of value 1/(2n-1) for first 2n-1 entries
    target_time = np.zeros(N)
    target_time[:2*n-1] = 1.0 / (2*n - 1)
    target_freq = np.fft.fft(target_time)
    target_mag = np.abs(target_freq)
    
    # Initialize with random phases
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0  # DC = sum(a) = 1
    for k in range(1, N // 2 + 1):
        A[k] = target_mag[k] * np.exp(1j * rng.uniform(0, 2 * np.pi))
        A[N-k] = np.conj(A[k])
    
    # For real signal, we need Hermitian symmetry
    # Actually let's just work with the full complex FFT
    
    a = np.ones(n) / n
    for it in range(n_iters):
        # Forward: a -> A
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        
        # Frequency domain: project magnitude toward target
        current_mag = np.abs(A)
        phase = A / (current_mag + 1e-30)
        
        # Blend current and target magnitude
        target_mag_safe = np.maximum(target_mag, 1e-15)
        new_mag = current_mag * 0.7 + target_mag_safe * 0.3
        # Preserve DC
        new_mag[0] = current_mag[0]
        A = new_mag * phase
        
        # Inverse: A -> a
        a_padded = np.real(np.fft.ifft(A))
        a_new = a_padded[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = a.copy()  # Reset
        else:
            a = a_new / s
        a[0] = max(a[0], 1e-10)  # Ensure at least one nonzero
    
    return a

def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fast coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1].copy()
            a_ref = a.copy()

            best_delta = 0.0
            best_new_max = M

            # Try factor-based steps
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_ref
                if 0 <= 2 * i - k_start < len(new_affected):
                    new_affected = new_affected.copy()
                    new_affected[2 * i - k_start] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_ref
                if 0 <= 2 * i - k_start < len(changes):
                    changes = changes.copy()
                    changes[2 * i - k_start] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        else:
            no_improve_count = 0

    return best_a, best_score

def accelerated_fictitious_play(n, time_limit, initial=None, solver=None, 
                                  window_size=2000, verbose=True):

    """

        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    # Precompute T (will be updated as a changes)
    T = build_conv_matrix(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Receding average storage
    window = []
    window_max = window_size
    
    iters = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Solve LP best response
        b, M_val = solve_lp_best_response(a, T=T, solver=solver)
        if b is None:
            # Try rebuilding T
            T = build_conv_matrix(a)
            b, M_val = solve_lp_best_response(a, T=T, solver=solver)
            if b is None:
                break

        alpha = 2.0 / (iters + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b

        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

        T = build_conv_matrix(a)

        window.append(a.copy())
        if len(window) > window_max:
            window.pop(0)

        candidates = [a]
        if len(window) >= 2:
            avg = np.mean(window, axis=0)
            avg = np.maximum(avg, 0)
            sa = avg.sum()
            if sa > 1e-10:
                avg = avg / sa
                candidates.append(avg)

            recent = np.mean(window[-100:], axis=0)
            recent = np.maximum(recent, 0)
            sr = recent.sum()
            if sr > 1e-10:
                recent = recent / sr
                candidates.append(recent)

            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-15:
                best_score = score
                best_a = cand.copy()
        
        # Print progress
        now = time.monotonic()
        if verbose and now - last_print > 30:
            elapsed = now - t0
            print(f"    AFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f} M_lp={M_val:.8f}")
            last_print = now
    
    return best_a, best_score, iters

def standard_fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """Standard fictitious play with uniform average."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    T = build_conv_matrix(a)
    avg_a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        b, M_val = solve_lp_best_response(a, T=T, solver=solver)
        if b is None:
            T = build_conv_matrix(a)
            b, M_val = solve_lp_best_response(a, T=T, solver=solver)
            if b is None:
                break
        
        # Standard FP update
        a = (iters * a + b) / (iters + 1)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        avg_a = a  # In standard FP, the average IS a
        
        T = build_conv_matrix(a)
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
        
        now = time.monotonic()
        if verbose and now - last_print > 30:
            elapsed = now - t0
            print(f"    SFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
            last_print = now
    
    return best_a, best_score, iters
