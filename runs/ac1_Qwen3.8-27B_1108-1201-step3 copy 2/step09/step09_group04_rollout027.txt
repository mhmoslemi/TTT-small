    return irfft(A * A, fft_size)[:size]

    lse_val = c_max + tau * np.log(np.sum(np.exp((c_pos - c_max) / tau)))

def p_norm_objective_z(z, p):
    """Compute p-norm objective in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_gradient_z(z, p):
    """Gradient of p-norm objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))


def lse_lbfgs_refine(a0, time_limit, tau_start=0.1, tau_end=0.001, n_stages=10):
    """L-BFGS with adaptive log-sum-exp temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    temps = np.geomspace(tau_start, tau_end, n_stages)
    iters_per_stage = max(200, 10000 // n_stages)

    for stage_tau in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return lse_objective_z(z_val, stage_tau)

        def jac(z_val):
            return lse_gradient_z(z_val, stage_tau)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      LSE L-BFGS tau={stage_tau:.5f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)

        def func(z_val):
            return p_norm_objective_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)

        try:

                fun=func,

                jac=jac,
                method='L-BFGS-B',

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_stall_score - 1e-7:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 4:
                rng = np.random.default_rng(it + 42)
                noise_scale = 0.02 + 0.01 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_directed_refinement(a0, time_limit, n_steps=50000):

        peak_val = b_work[peak_idx]
        
        # Find all near-peak positions (within 2% of max)
        near_peak_threshold = peak_val * 0.98
        near_peaks = np.where(b_work >= near_peak_threshold)[0]
        
        improved = False

        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        
        # Sort by contribution
        top_idx = np.argsort(contribs)[::-1][:20]
        
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue

            for factor in [0.98, 0.96, 0.94, 0.92, 0.90, 0.85, 0.80]:
                a_trial = a.copy()

                a_trial[i] -= reduction
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    reduction = a[i]
                # Redistribute proportionally to other elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += reduction * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()

                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Pair adjustment at peak
        if not improved:
            for ci in top_idx[:10]:
                i = idx[ci]
                j = peak_idx - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.95, 0.93, 0.90]:
                    for f2 in [0.97, 0.95, 0.93, 0.90]:
                        a_trial = a.copy()

                        a_trial[j] *= f2

                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:

                    if improved:

                if improved:
                    break

        # Strategy 3: Multi-peak simultaneous reduction
        if not improved and len(near_peaks) >= 2:
            # Find elements that contribute to multiple peaks
            multi_contrib = np.zeros(n)
            for k in near_peaks[:5]:
                js_k = max(0, k - n + 1)
                je_k = min(n - 1, k)
                idx_k = np.arange(js_k, je_k + 1)
                if len(idx_k) > 0:
                    multi_contrib[idx_k] += a[idx_k] * a[k - idx_k]
            
            top_multi = np.argsort(multi_contrib)[::-1][:5]
            for i in top_multi:
                if a[i] < 1e-14:
                    continue
                for factor in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    reduction = a[i] * (1 - factor)
                    a_trial[i] -= reduction
                    if a_trial[i] < 0:

                        reduction = a[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break

                    score = 2.0 * n * fast_convolve(a_trial).max()

                        break

                    break

def structured_init(n, variant=0, rng=None):
    """Create structured initial sequence of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # Broad plateau
        x = np.linspace(0, 1, n)
        a = np.where(np.abs(x - 0.5) < 0.4, 1.0, 0.1)
    elif variant == 6:
        # Slightly modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:

    except Exception:
        return float('inf')


def compute_score(a):
    """Compute score directly."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:

    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s ** 2)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective_z(z, tau):
    """Log-sum-exp based objective in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 0)
    c_max = c_pos.max()
    if c_max < 1e-300:
        return 2.0 * n * c_max
    shifted = (c_pos - c_max) / tau
    lse_val = c_max + tau * np.log(np.sum(np.exp(shifted)))
    return 2.0 * n * lse_val


def lse_gradient_z(z, tau):
    """Gradient of LSE objective w.r.t. softmax parameterization."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    c_max = c_pos.max()
    logits = (c_pos - c_max) / tau
    exp_logits = np.exp(logits)
    softmax_w = exp_logits / exp_logits.sum()
    corr = np.correlate(softmax_w, a, mode='valid')
    g_a = 2.0 * n * corr
    return a * (g_a - np.dot(g_a, a))


def p_norm_objective_z(z, p):
    """Compute p-norm objective in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_gradient_z(z, p):
    """Gradient of p-norm objective w.r.t. z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    return a * (g_a - np.dot(g_a, a))

def lse_lbfgs_refine(a0, time_limit, tau_start=0.05, tau_end=0.001, n_stages=10):
    """L-BFGS with adaptive log-sum-exp temperature schedule."""
    global _g_lse_tau
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        _g_lse_tau = stage_tau

        try:
            res = minimize(
                fun=_lse_func,
                x0=z,
                jac=_lse_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxfun': iters_per_stage * 400, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0

    exp_z = np.exp(z - z_max)

    return a

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with adaptive step size and restarts."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score
    last_improve_iter = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            last_improve_iter = it
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            last_improve_iter = it

        if it % 300 == 0:

            if stall_count >= 5:

                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

        if (it + 1) % 3000 == 0:

def structured_init(n, variant=0):

        a = np.exp(-4.0 * x * (1.0 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)

    s = a.sum()

        return np.ones(n) / n

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """Evaluate sequence safely using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update_best(best_state, a, label):
    """Try to update the best state with a new candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LSE gradient, p-norm L-BFGS, Nesterov LP,
    peak-directed refinement, multi-length exploration, and perturbation.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(1.0, float(budget_s) - 15.0)

    best_state = {'score': float('inf'), 'seq': None}

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: LSE gradient optimization (primary) ===
    print(f"\n=== Phase 1: LSE L-BFGS n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, tau: 0.08 -> 0.001")
        a_lse = lse_lbfgs_refine(best_state['seq'], t_alloc, tau_start=0.08, tau_end=0.001, n_stages=12)
        try_update_best(best_state, a_lse, "lse_1")

    # === Phase 2: Nesterov LP (global convergence) ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=200000)
        try_update_best(best_state, a_lp, "nesterov_lp")

    # === Phase 3: p-norm L-BFGS at very high p ===
    print(f"\n=== Phase 3: p-norm high (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, p: 500 -> 5000")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=500.0, p_end=5000.0, n_stages=8)
        try_update_best(best_state, a_pn, "pn_very_high")

    # === Phase 4: Peak-directed refinement ===
    print(f"\n=== Phase 4: Peak Refinement (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        a_pk, _ = peak_directed_refinement(best_state['seq'], t_alloc, n_steps=100000)
        try_update_best(best_state, a_pk, "peak_1")

    # === Phase 5: LSE round 2 (lower tau) ===
    print(f"\n=== Phase 5: LSE r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 170.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s, tau: 0.005 -> 0.0003")
        a_lse2 = lse_lbfgs_refine(best_state['seq'], t_alloc, tau_start=0.005, tau_end=0.0003, n_stages=8)
        try_update_best(best_state, a_lse2, "lse_2")

    # === Phase 6: Nesterov LP round 2 ===
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 130.0)
    if t_alloc > 15:
        a_lp2, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=80000)
        try_update_best(best_state, a_lp2, "nesterov_lp_2")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-64, -48, -32, -16, 16, 32, 48, 64, 96]:
        alt_n = base + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.add(alt_n)
    for rn in [768, 832, 896, 928, 960, 992, 1024, 1088, 1152]:
        if 512 <= rn <= 2048:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base))

    n_explored = 0
    for alt_n in candidate_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 50 or n_explored >= 4:
            break
        n_explored += 1
        t_total = min(rem * 0.12, 120.0)
        if t_total < 25:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        t_lse = t_total * 0.6
        a_alt = lse_lbfgs_refine(init_alt, t_lse, tau_start=0.03, tau_end=0.002, n_stages=6)
        improved = try_update_best(best_state, a_alt, f"lse_n{alt_n}")

        if improved:
            rem = deadline - time.monotonic()
            if rem > 20:
                t_lp = min(rem * 0.08, 60.0)
                a_alt2, _ = nesterov_lp_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=20000)
                try_update_best(best_state, a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.03, 0.06, 0.12]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            pst = perturbed.sum()
            if pst > 0.01:
                perturbed = perturbed / pst
            t_sub = min(rem * 0.25, 40.0)
            a_pert = lse_lbfgs_refine(perturbed, t_sub, tau_start=0.01, tau_end=0.001, n_stages=5)
            try_update_best(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.30, 50.0)
        a_fin = lse_lbfgs_refine(best_state['seq'], t_fin, tau_start=0.003, tau_end=0.0002, n_stages=5)
        try_update_best(best_state, a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_pk = min(rem * 0.30, 30.0)
        a_pk2, _ = peak_directed_refinement(best_state['seq'], t_pk, n_steps=50000)
        try_update_best(best_state, a_pk2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 8:
        t_lp_fin = min(rem * 0.40, 25.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=10000)
        try_update_best(best_state, a_lp_fin, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - total_start
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
