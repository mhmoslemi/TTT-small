def lbgfs_refine(a0, time_limit, T_start, T_end, maxiter):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3)) * np.exp(-2 * (x - 0.7))
        a = a / a.max()
    else:
        a = np.ones(n) / n
        freqs = rng.integers(1, n // 2, size=3)
        for f in freqs:
            a += 0.1 * np.cos(2 * np.pi * f * np.arange(n) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def spectral_equalize(a0, alpha):
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(np.asarray(a0, dtype=np.float64), fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    # Blend magnitude towards flat (geometric mean of current and target)
    target_mag = np.ones_like(mag)
    target_mag = target_mag * mag.max()
    new_mag = mag ** (1 - alpha) * target_mag ** alpha
    new_A = new_mag * np.exp(1j * phase)
    a_new = irfft(new_A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_best = len(best_seq) if best_seq is not None else 960

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        n_best = n_init
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Phase 0: Spectral equalization of parent (quick, may help)
    print(f"\n=== Phase 0: Spectral Equalize (best={best_score:.6f}) ===")
    for alpha in [0.1, 0.2, 0.3]:
        a_spec = spectral_equalize(best_seq, alpha)
        try_update(a_spec, f"spec_{alpha}")

    # Phase 1: Quick multi-start screening
    print(f"\n=== Phase 1: Multi-Start Screening (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    candidates = [best_seq.tolist()]
    for k in range(5):
        init_k = structured_init(n_best, rng)
        candidates.append(init_k.tolist())
    # Also try interpolated versions
    for alt_n in [int(n_best * 0.85), int(n_best * 1.15)]:
        if 400 <= alt_n <= 2500:
            candidates.append(interpolate_sequence(best_seq, alt_n).tolist())

    screened = []
    for ci, cand in enumerate(candidates):
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        a_cand = np.asarray(cand, dtype=np.float64)
        a_cand = np.maximum(a_cand, 0)
        s = a_cand.sum()
        if s > 1e-10:
            a_cand = a_cand / s
        # Quick mirror descent
        t_quick = min(rem * 0.15, 30.0)
        a_screened, _ = mirror_descent_adaptive(a_cand, t_quick, n_active_frac=0.08)
        screened.append((evaluate_safe(a_screened), a_screened, len(a_screened)))

    # Sort by score and keep top 3
    screened.sort(key=lambda x: x[0])
    print(f"  Screened {len(screened)} candidates, best: {screened[0][0]:.6f}")

    # Use the best screened as working sequence
    if screened[0][0] < best_score:
        best_score = screened[0][0]
        best_seq = screened[0][1].copy()
        n_best = len(best_seq)
        print(f"  Updated best from screening: {best_score:.6f}")

    # Phase 2: Nesterov LP (main workhorse)
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=300000)
        try_update(a_lp, "nesterov_lp_1")

    # Phase 3: Coordinate descent (new method)
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_greedy(best_seq, t_alloc, n_passes=500000)
        try_update(a_cd, "coord_desc_1")

    # Phase 4: L-BFGS LogSumExp
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=150.0, T_end=0.1, maxiter=20000)
        try_update(a_lb, "lbgfs_1")

    # Phase 5: Second LP round (on improved sequence)
    print(f"\n=== Phase 5: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp2, "nesterov_lp_2")

    # Phase 6: Local swap search
    print(f"\n=== Phase 6: Local Swap (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_ls, _ = local_swap_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_ls, "local_swap")

    # Phase 7: Multi-length exploration
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [0.75, 0.9, 1.1, 1.25, 1.5]:
        alt_n = int(n_best * mult)
        if 384 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))[:3]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_total = min(rem * 0.3, 180.0)
        if t_total < 20:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolation improved: {sc_alt:.6f}")

        # LP at new length
        t_lp = t_total * 0.6
        a_alt, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=200000)
        try_update(a_alt, f"lp_n{alt_n}")

        # Coordinate descent at new length
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.3, 40.0)
            a_alt2, _ = coordinate_descent_greedy(best_seq, t_cd, n_passes=100000)
            try_update(a_alt2, f"cd_n{alt_n}")

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()

    # L-BFGS final
    if rem > 20:
        t_fin = min(rem * 0.2, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=8.0, T_end=0.05, maxiter=10000)
        try_update(a_fin, "final_lbgfs")

    # Mirror descent final with tight active set
    rem = deadline - time.monotonic()
    if rem > 15:
        t_md = min(rem * 0.2, 40.0)
        a_md = mirror_descent_adaptive(best_seq, t_md, n_active_frac=0.02)
        try_update(a_md, "final_md")

    # LP final
    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.3, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=50000)
        try_update(a_lp_fin, "final_lp")

    # Coordinate descent final
    rem = deadline - time.monotonic()
    if rem > 10:
        t_cd_fin = min(rem * 0.3, 30.0)
        a_cd_fin, _ = coordinate_descent_greedy(best_seq, t_cd_fin, n_passes=50000)
        try_update(a_cd_fin, "final_cd")

    # Perturbative restart: add small noise and re-optimize
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n  Perturbative restart...")
        rng_final = np.random.default_rng(seed + 999)
        for noise_scale in [0.005, 0.01, 0.02]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            a_noisy = best_seq + rng_final.standard_cauchy(len(best_seq)) * noise_scale
            a_noisy = np.maximum(a_noisy, 0)
            ss = a_noisy.sum()
            if ss > 1e-10:
                a_noisy = a_noisy / ss
                # Quick coordinate descent from noisy start
                t_restart = min(rem * 0.4, 25.0)
                a_restart, _ = coordinate_descent_greedy(a_noisy, t_restart, n_passes=50000)
                try_update(a_restart, f"restart_{noise_scale}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
