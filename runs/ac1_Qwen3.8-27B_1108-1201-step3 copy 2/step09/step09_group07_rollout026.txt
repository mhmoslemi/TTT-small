def normalize_v(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fft_autoconv(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    """Evaluate sequence using the actual evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.state not in ('optimal', 'optimal_inaccurate') and prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate') and prob.state not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b_var.value is None:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else score_normalized(b_val)
    return b_val, M_val


def mirror_descent_lse(a0, time_limit, T_start=5.0, T_end=300.0,
                       n_stages=10, steps_per_stage=2000):
    """Mirror descent with log-sum-exp smoothed objective and temperature annealing."""
    n = len(a0)
    a = normalize_v(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            b = fft_autoconv(a)
            Tb = T * b
            m = Tb.max()
            e = np.exp(Tb - m)
            w = e / e.sum()
            # Gradient: 2 * (a cross-correlated with w shifted)
            # grad[i] = 2 * sum_j a[j] * w[j+i]
            grad = fft_cross_corr(a, w)
            grad = 2.0 * grad
            # Mirror descent: a_i *= exp(-lr * grad_i)
            lr = 0.1 / (1.0 + 0.01 * step_count)
            a = a * np.exp(-lr * (grad - grad.mean()))
            a = normalize_v(a)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def mw_game_iteration(a0, time_limit, K_inner=30, eta_inner=1.0,
                      alpha_polyak=True, eval_interval=50,
                      lp_every=100, solver=None):
    """
    Multiplicative-weight game iteration.
    Inner loop: K MW steps to approximate best response b to a.
    Outer loop: update a toward b using Polyak/Nesterov averaging.
    """
    n = len(a0)
    a = normalize_v(a0.copy())
    b = normalize_v(np.ones(n) * 0.5 + 0.5 * a)
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    a_mom = a.copy()
    a_prev = a.copy()
    b_avg = b.copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    outer_step = 0

    while time.monotonic() - t0 < time_limit:
        # Inner MW loop: approximate best response b to fixed a
        for k in range(K_inner):
            # Compute p = T_a^T lam
            p = fft_cross_corr(a, lam)
            # Update b: b_i *= exp(-eta * p_i)
            centered_p = p - p.mean()
            b = b * np.exp(-eta_inner * centered_p)
            b = normalize_v(b)
            # Compute q = conv(a, b)
            q = fft_conv(a, b)
            # Update lam: lam_j *= exp(eta * q_j)
            centered_q = q - q.mean()
            lam = lam * np.exp(eta_inner * centered_q)
            lam = normalize_v(lam)

        # Outer update: a toward b
        outer_step += 1
        if alpha_polyak:
            alpha = 2.0 / (outer_step + 2)
        else:
            alpha = 0.05 / (1.0 + 0.001 * outer_step)

        # Nesterov momentum
        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * outer_step * outer_step) / 2.0
        t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (outer_step - 1) ** 2) / 2.0 if outer_step > 1 else 1.0
        beta = (t_prev - 1.0) / t_curr if outer_step > 1 else 0.0

        a_new = a_mom + beta * (a_mom - a_prev)
        a_new = (1.0 - alpha) * a_new + alpha * b
        a_new = normalize_v(a_new)

        a_prev = a.copy()
        a = a_new
        a_mom = a.copy()
        b_avg = (1.0 - alpha) * b_avg + alpha * b

        # Periodic evaluation
        if outer_step % eval_interval == 0:
            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            sc_b = score_normalized(b_avg)
            if sc_b < best_score:
                best_score = sc_b
                best_a = b_avg.copy()

        # Periodic LP correction
        if lp_every > 0 and outer_step % lp_every == 0 and solver is not None:
            b_lp, _ = solve_lp_best_response(a, solver)
            if b_lp is not None:
                b = b_lp.copy()
                lam = normalize_v(fft_conv(a, b))
                # Also try the LP result directly
                sc_lp = score_normalized(b_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_a = b_lp.copy()

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = normalize_v(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    t_curr = 1.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * it * it) / 2.0
        t_prev = 1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2) / 2.0 if it > 0 else 1.0
        beta = (t_prev - 1.0) / t_curr if it > 0 else 0.0

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize_v(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize_v(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it + 12345)
                y = y + 0.03 * rng.standard_normal(n)
                y = normalize_v(y)
                stall_count = 0

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Multi-peak shaving with coordinate adjustments."""
    n = len(a0)
    a = normalize_v(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fft_autoconv(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        if peak_val <= 0:
            break
        threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        # Find top contributors across near-peaks
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:15]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.96, 0.93, 0.89, 0.84]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = normalize_v(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction at main peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90]:
                    for f2 in [0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = normalize_v(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Try uniform small reduction on top coords
            a_trial = a.copy()
            for i in top_coords[:5]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = normalize_v(a_trial)
            score = score_normalized(a_trial)
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Try random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + 0.01 * rng.standard_normal(n)
                a_trial = normalize_v(a_trial)
                score = score_normalized(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, rng, mode=0):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    elif mode == 1:
        a = 1.0 + 0.4 * (2 * x - 1)
    elif mode == 2:
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
    elif mode == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif mode == 4:
        a = 1.0 / (1.0 + 2.0 * np.sin(2 * np.pi * x) ** 2)
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def fast_adaptive_eta(gradient, base_eta=1.0):
    """Adapt step size based on gradient magnitude."""
    g_max = np.abs(gradient).max()
    if g_max > 1e-10:
        return base_eta / (1.0 + g_max)
    return base_eta


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining MW game iteration, LP refinement,
    and peak shaving to minimize the evaluation function.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        best_seq = make_init(n_init, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: MW Game Iteration (main workhorse) ===
    print(f"\n=== Phase 1: MW Game n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, K_inner=30, eta=1.0")
        a_mw, _ = mw_game_iteration(best_seq, t_alloc, K_inner=30, eta_inner=1.0,
                                    alpha_polyak=True, eval_interval=30,
                                    lp_every=80, solver=solver)
        try_update(a_mw, "mw_main")

    # === Phase 2: LP Refinement ===
    print(f"\n=== Phase 2: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Second MW round with different params ===
    print(f"\n=== Phase 3: MW Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, K_inner=50, eta=0.7")
        a_mw2, _ = mw_game_iteration(best_seq, t_alloc, K_inner=50, eta_inner=0.7,
                                     alpha_polyak=False, eval_interval=20,
                                     lp_every=60, solver=solver)
        try_update(a_mw2, "mw_2")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Multi-start with diverse inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        modes = [1, 3, 4]
        for mode in modes:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 150.0)
            if t_sub < 30:
                break
            a_init = make_init(n_best, rng, mode)
            print(f"  mode={mode}, budget={t_sub:.0f}s")
            # Quick MW then polish
            a_sub, _ = mw_game_iteration(a_init, t_sub * 0.6, K_inner=20, eta_inner=1.2,
                                         alpha_polyak=True, eval_interval=30,
                                         lp_every=100, solver=solver)
            try_update(a_sub, f"init_m{mode}")
            rem = deadline - time.monotonic()
            if rem > 30 and t_sub * 0.3 > 15:
                a_sub2, _ = nesterov_lp_refine(a_sub, t_sub * 0.3, solver=solver, n_max_iters=5000)
                try_update(a_sub2, f"init_m{mode}_lp")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-64, -32, 32, 64, 96, 128]:
        alt_n = n_best + delta
        if 512 <= alt_n <= 4096:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.20, 150.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = mw_game_iteration(init_alt, t_total * 0.6, K_inner=25, eta_inner=1.0,
                                     alpha_polyak=True, eval_interval=30,
                                     lp_every=80, solver=solver)
        try_update(a_alt, f"mw_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.3, 40.0), solver=solver, n_max_iters=3000)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 90.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver=solver, n_max_iters=10000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.20, 30.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=10000)
        try_update(a_ps2, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
