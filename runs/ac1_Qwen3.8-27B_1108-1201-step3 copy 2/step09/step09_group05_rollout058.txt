2. Nesterov-accelerated LP best-response iteration (main optimization)
3. Adaptive-temperature L-BFGS with log-sum-exp smoothing (local refinement)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) for normalized (sum=1) sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


_lp_solver_cached = None


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.05, maxiter=8000, n_stages=10, adaptive=True):
    """L-BFGS with LogSumExp smoothing and adaptive temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    def lsfunc(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lsjac(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    if n_stages is None:
        n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Adaptive: if stuck, keep temperature; else proceed
        if adaptive and si > 0:
            # Check if we made progress
            pass  # Simplified: just proceed with schedule

        try:
            res = minimize(
                fun=lambda zi: lsfunc(zi, stage_T),
                x0=z,
                jac=lambda zi: lsjac(zi, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        current_a = softmax_from_z(z)
        current_score = score_normalized(current_a)
        if current_score < best_score:
            best_score = current_score
            best_a = current_a.copy()

    return best_a


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000, multi_peak=False):
    """Nesterov dual averaging LP refinement with adaptive perturbation."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = score_normalized(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_score - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(777 + it)
                noise_scale = 0.02 * (1 + stall_count * 0.5)
                noise = rng_local.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

    return best_a, best_score


def subgradient_polyak(a0, time_limit, n_max_iters=200000):
    """Direct subgradient descent with Polyak step size and Nesterov momentum."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Lower bound estimate (theoretical minimum is 2.0 for uniform, but we know better)
    lb = best_score - 0.5  # Conservative lower bound for Polyak step

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Subgradient: grad_i = 2 * y[k_star - i] for valid i
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * y[k_star - idx]

        # Also add contribution from second peak for robustness
        b2 = b.copy()
        b2[k_star] = -np.inf
        k2 = int(np.argmax(b2))
        if b[k2] > 0.9 * peak_val:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            g[idx2] += 0.5 * 2.0 * y[k2 - idx2]

        # Polyak step size
        current_val = 2.0 * n * peak_val
        gap = current_val - lb
        if gap < 1e-10:
            gap = 1e-10
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        eta = gap / g_norm_sq
        eta = min(eta, 1.0)  # Cap step size

        # Nesterov momentum
        alpha = 2.0 / (it + 2)
        y_candidate = y - eta * g
        y_candidate = np.maximum(y_candidate, 0)
        sc = y_candidate.sum()
        if sc < 1e-15:
            break
        y_candidate = y_candidate / sc

        # Projection to simplex is handled by normalization (since sum is preserved)
        # But we need to ensure we stay on the simplex properly
        # The gradient of the constraint sum(a)=1 is the all-ones vector
        # Projecting onto simplex: a_i = max(y_i - tau, 0) where tau is chosen so sum=1
        # For simplicity, just clip and normalize
        y_new = y_candidate

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * y_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        score_y = score_normalized(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            # Update lower bound
            lb = max(lb, best_score - 0.1)

    return best_a, best_score


def parametric_fourier_optimize(a0, time_limit):
    """Optimize a low-dimensional Fourier parametric model fitted to a0."""
    n = len(a0)
    # Fit: log(a_i) ~= c0 + sum_{k=1}^{K} (c_k * cos(2*pi*k*i/n) + d_k * sin(2*pi*k*i/n))
    # Use K harmonics
    K = 5
    x = np.arange(n)
    nparams = 1 + 2 * K
    # Initial fit via least squares
    A = np.ones((n, nparams))
    for k in range(1, K + 1):
        A[:, 2 * k - 1] = np.cos(2 * np.pi * k * x / n)
        A[:, 2 * k] = np.sin(2 * np.pi * k * x / n)
    log_a = np.log(np.maximum(a0, 1e-300))
    coeffs, _, _, _ = np.linalg.lstsq(A, log_a, rcond=None)

    t0 = time.monotonic()
    best_score = score_normalized(a0)
    best_a = a0.copy()

    def param_obj(params):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        c0 = params[0]
        shape = np.ones(n)
        for k in range(1, K + 1):
            ck = params[2 * k - 1]
            dk = params[2 * k]
            shape += ck * np.cos(2 * np.pi * k * x / n) + dk * np.sin(2 * np.pi * k * x / n)
        # Ensure positivity: use exp
        a = np.exp(shape)
        s = a.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 1e15
        a = a / s
        # Check non-negativity (always true with exp)
        return score_normalized(a)

    # Optimize with Nelder-Mead (robust to non-smoothness)
    try:
        res = minimize(
            fun=param_obj,
            x0=coeffs,
            method='Nelder-Mead',
            options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                     'adaptive': True, 'maxfev': 5000}
        )
        if time.monotonic() - t0 < time_limit:
            # Refine with BFGS
            res2 = minimize(
                fun=param_obj,
                x0=res.x,
                method='BFGS',
                options={'maxiter': 1000, 'gtol': 1e-10}
            )
            if param_obj(res2.x) < param_obj(res.x):
                res = res2
    except Exception:
        pass

    # Convert back to sequence
    c0 = res.x[0]
    shape = np.ones(n) + c0 - coeffs[0]  # Center around original
    for k in range(1, K + 1):
        shape += res.x[2 * k - 1] * np.cos(2 * np.pi * k * x / n) + res.x[2 * k] * np.sin(2 * np.pi * k * x / n)
    a_new = np.exp(shape)
    s = a_new.sum()
    if s > 1e-300 and np.isfinite(s):
        a_new = a_new / s
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, n_steps=30000, multi_peak_weight=0.3):
    """Frank-Wolfe with away steps and multi-peak awareness."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient from primary peak
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = a[k_star - idx]

        # Add secondary peaks
        b2 = b.copy()
        b2[k_star] = -np.inf
        for pk in range(3):
            k2 = int(np.argmax(b2))
            if b[k2] < 0.85 * b[k_star]:
                break
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            g[idx2] += multi_peak_weight * a[k2 - idx2]
            b2[k2] = -np.inf

        # FW toward step
        i_toward = int(np.argmin(g))
        best_trial = None
        best_trial_score = best_score

        for t_val in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]:
            a_trial = a.copy()
            a_trial[i_toward] += t_val * (1.0 - a[i_toward])
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score_trial = score_normalized(a_trial)
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial = a_trial.copy()

        # Away step
        mask = a > 1e-10
        if mask.sum() > 2:
            g_masked = g.copy()
            g_masked[~mask] = -np.inf
            i_away = int(np.argmax(g_masked))
            for gamma in [0.2, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[i_away] *= (1.0 - gamma)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score_trial = score_normalized(a_trial)
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=10000):
    """Targeted peak shaving with redistribution."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Ignore edges (they're naturally small)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[:2] = -np.inf
            b_work[-2:] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Find near-peaks
        threshold = peak_val * 0.95
        near_peaks = np.where(b_work >= threshold)[0]

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                # Redistribute to low-contributing positions
                low_coords = np.argsort(contribs)[:max(1, n // 5)]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    base = a_trial[low_mask].copy() + 1e-15
                    a_trial[low_mask] = base + diff * base / base.sum()
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = score_normalized(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair adjustment at the peak
            found = False
            for ci in top_idx[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.87]:
                    for f2 in [0.92, 0.87]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = score_normalized(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            if not improved:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Load parent + parametric Fourier optimization
    Phase 2: Nesterov LP (main global)
    Phase 3: Adaptive L-BFGS (local refinement)
    Phase 4: Subgradient Polyak (robust)
    Phase 5: Frank-Wolfe + Peak Shave (fine tuning)
    Phase 6: Multi-length exploration
    Phase 7: Final polish + return
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        # Try several structured inits
        inits = []
        x = np.linspace(0, 1, n_init)
        inits.append(np.cos(np.pi * x / 2) ** 2)
        inits.append(np.exp(-4 * x * (1 - x)))
        inits.append(0.5 + 0.4 * np.cos(2 * np.pi * x))
        inits.append(np.sin(np.pi * x) ** 1.5)
        best_init = None
        best_init_score = float('inf')
        for init in inits:
            init = norm_sequence(init)
            sc = evaluate_safe(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init
        best_seq = best_init
        best_score = best_init_score
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Parametric Fourier optimization ===
    print(f"\n=== Phase 1: Parametric Fourier (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_param, _ = parametric_fourier_optimize(best_seq, t_alloc)
        try_update(a_param, "fourier_param")

    # === Phase 2: Nesterov LP (main) ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Adaptive L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS Adaptive (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 80->0.03, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=80.0, T_end=0.03,
                           maxiter=12000, n_stages=14, adaptive=True)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Subgradient with Polyak ===
    print(f"\n=== Phase 4: Subgradient Polyak (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_polyak(best_seq, t_alloc, n_max_iters=300000)
        try_update(a_sg, "subgrad_polyak")

    # === Phase 5: Frank-Wolfe + Peak Shave ===
    print(f"\n=== Phase 5: FW + PeakShave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()

        a_fw, _ = frank_wolfe_refine(best_seq, t_fw, n_steps=30000, multi_peak_weight=0.3)
        try_update(a_fw, "frank_wolfe")

        a_ps, _ = peak_shave_local(best_seq, t_ps, n_steps=15000)
        try_update(a_ps, "peak_shave")

        try_update(a_lp2, "nesterov_lp_2")

    if t_lb2 > 15:
        a_lb2 = lbgfs_refine(best_seq, t_lb2, T_start=20.0, T_end=0.02,
                            maxiter=8000, n_stages=10, adaptive=True)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 1.33, 0.7]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.2, 150.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=15000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            t_altlb = min(rem * 0.3, 60.0)
            if t_altlb > 10:
                a_alt2 = lbgfs_refine(best_seq, t_altlb, T_start=30.0, T_end=0.05,
                                     maxiter=4000, n_stages=8, adaptive=True)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for scale in [0.05, 0.10, 0.03]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(rem * 0.3, 100.0)
            t_lp_pert = t_sub * 0.5
            a_pert, _ = nesterov_lp_refine(perturbed, t_lp_pert, solver=solver, n_max_iters=8000)
            try_update(a_pert, f"pert_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_lbp = min(rem * 0.3, t_sub * 0.4)
                if t_lbp > 10:
                    a_pert2 = lbgfs_refine(best_seq, t_lbp, T_start=10.0, T_end=0.05,
                                          maxiter=3000, n_stages=6, adaptive=True)
                    try_update(a_pert2, f"pert_lb_{scale}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.3, 50.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.02,
                            maxiter=5000, n_stages=8, adaptive=True)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_fw2 = min(rem * 0.2, 25.0)

        try_update(a_fw2, "final_fw")

            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
