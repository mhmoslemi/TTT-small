import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))

def gs_optimize(a_init, target_mag, n, N, max_iters, check_interval=200):
    """Gerchberg-Saxton optimization."""
    a_vec = np.zeros(N)
    a_vec[:n] = np.maximum(a_init, 0)
    s = a_vec[:n].sum()
    if s < 1e-15:
        a_vec[:n] = 1.0 / n
        s = 1.0
    a_vec[:n] /= s
    
    best_a = a_vec[:n].copy()
    best_score = float('inf')
    t0 = time.monotonic()
    
    for it in range(max_iters):
        # Frequency domain: enforce magnitude
        A = np.fft.rfft(a_vec)
        mag = np.abs(A)
        # Avoid division by zero
        safe_mag = np.maximum(mag, 1e-30)
        phase = A / safe_mag
        A_new = target_mag * phase
        # Time domain
        a_vec = np.fft.irfft(A_new, N)
        # Enforce support and non-negativity
        a_vec[n:] = 0
        a_vec[:n] = np.maximum(a_vec[:n], 0)
        s = a_vec[:n].sum()
        if s < 1e-15:
            break
        a_vec[:n] /= s
        
        if it % check_interval == 0:
            sc = evaluate_local(a_vec[:n])
            if sc < best_score:
                best_score = sc
                best_a = a_vec[:n].copy()
    
    # Final check
    sc = evaluate_local(a_vec[:n])
    if sc < best_score:
        best_score = sc
        best_a = a_vec[:n].copy()
    
    return best_a, best_score

def project_simplex(x):
    """Project x onto {a: sum(a)=1, a>=0}."""
    n = len(x)
    # Bisection for the threshold
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(x - mid, 0)
        s = projected.sum()
        if s > 1:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(x - mid, 0)
    s = projected.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return projected / s

def nesterov_subgrad(a_init, time_limit, norm_interval=500):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = float('inf')
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov momentum point
    a_prev = a.copy()
    beta = 1.0
    lr = 0.1
    no_improve = 0
    rng = np.random.default_rng(42)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        b = fast_autoconv(y)
        k_star = int(np.argmax(b))
        # Subgradient: g[m] = 2 * y[k_star - m] if 0 <= k_star - m < n
        g = np.zeros(n)
        # y[k_star - m] for m in range(n): valid when 0 <= k_star - m < n
        # i.e., k_star - n + 1 <= m <= k_star
        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        if m_lo <= m_hi:
            g[m_lo:m_hi+1] = 2.0 * y[k_star - m_hi:k_star - m_lo - 1:-1]
        
        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = a + (beta / beta_new - 1.0) * (a - a_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
            sy = 1.0
        y_new = y_new / sy
        
        # Gradient step
        x_new = y_new - lr * g
        a_new = project_simplex(x_new)
        
        # Evaluate
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a_prev = a
        a = a_new
        y = y_new
        beta = beta_new
        it += 1
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        if no_improve > 500:
            # Perturbation
            noise = rng.standard_normal(n) * (0.01 / np.sqrt(it))
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_prev = a.copy()
            y = a.copy()
            beta = 1.0
            lr = max(lr, 0.05)
            no_improve = 0
        
        if it % norm_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    """Compute log-sum-exp smoothed objective."""
    z_shifted = z - z.max()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated spectral (GS)
initialization, Nesterov subgradient descent, fast log-sum-exp L-BFGS with
FFT-based gradients, and multi-length exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global state for L-BFGS helper functions (avoid closures)
_G_LSE_N = 0
_G_LSE_T = 1.0


def fast_autoconv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def evaluate_local(a):
    """Compute the objective value locally (no function call overhead)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(x):
    """Project x onto the probability simplex {a: sum(a)=1, a>=0}."""
    n = len(x)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.maximum(x - mid, 0)
        s = projected.sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(x - mid, 0)
    s = projected.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return projected / s


def gs_optimize(a_init, target_mag, n, N, max_iters, check_interval=100):
    """Gerchberg-Saxton spectral optimization."""
    a_vec = np.zeros(N)
    a_vec[:n] = np.maximum(a_init, 0)
    s = a_vec[:n].sum()
    if s < 1e-15:
        a_vec[:n] = 1.0 / n
        s = 1.0
    a_vec[:n] /= s

    best_a = a_vec[:n].copy()
    best_score = evaluate_local(best_a)

    for it in range(max_iters):
        A = np.fft.rfft(a_vec)
        mag = np.abs(A)
        safe_mag = np.maximum(mag, 1e-300)
        phase = A / safe_mag
        A_new = target_mag * phase
        a_vec = np.fft.irfft(A_new, N)
        a_vec[n:] = 0
        a_vec[:n] = np.maximum(a_vec[:n], 0)
        s = a_vec[:n].sum()
        if s < 1e-15:
            break
        a_vec[:n] /= s

        if it % check_interval == 0:
            sc = evaluate_local(a_vec[:n])
            if sc < best_score:
                best_score = sc
                best_a = a_vec[:n].copy()

    sc = evaluate_local(a_vec[:n])
    if sc < best_score:
        best_score = sc
        best_a = a_vec[:n].copy()

    return best_a, best_score


def gs_multi_target(a_init, n, N, time_budget, rng):
    """Run GS with multiple target spectra and return best result."""
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = evaluate_local(a_init)

    # Target 1: perfectly flat (excluding DC)
    target_flat = np.full(N // 2 + 1, 1.0)
    target_flat[0] = np.sqrt(n)  # DC component elevated

    # Target 2: slightly U-shaped
    freqs = np.arange(N // 2 + 1) / (N // 2)
    target_u = 1.0 + 0.5 * np.cos(np.pi * freqs)
    target_u[0] = np.sqrt(n)

    # Target 3: elevated edges
    mask = np.ones(N // 2 + 1)
    mask[0] = np.sqrt(n)
    target_edge = mask * (1.0 + 0.3 * np.abs(freqs - 0.5) * 2.0)

    # Target 4: linearly decreasing
    target_lin = 1.0 + 0.5 * (1.0 - freqs)
    target_lin[0] = np.sqrt(n)

    targets = [target_flat, target_u, target_edge, target_lin]
    per_target_time = time_budget / len(targets)

    for i, target in enumerate(targets):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        alloc = min(per_target_time, remaining)
        if alloc < 2:
            break
        max_iters = int(alloc * 3000)  # ~3000 iters/sec
        max_iters = max(1000, min(max_iters, 200000))
        a_result, sc = gs_optimize(a_init, target, n, N, max_iters)
        if sc < best_score:
            best_score = sc
            best_a = a_result.copy()
            print(f"      GS target {i}: score={sc:.8f}")

    return best_a, best_score

    beta = 1.0

        b = irfft(A_y * A_y, N)[:2 * n - 1]
        k_star = int(np.argmax(b))

        # g[m] = 2 * y[k_star - m] for valid m
        g = np.zeros(n)
        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi + 1)
            g[idx] = 2.0 * y[k_star - idx]

        # Nesterov extrapolation
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = (beta / beta_new - 1.0)
        y_new = a + momentum * (a - a_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
            sy = 1.0
        y_new = y_new / sy

        # Gradient step + projection
        x_new = y_new - lr * g
        a_new = project_simplex(x_new)

        # Evaluate
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a
        a = a_new
        y = y_new
        beta = beta_new
        it += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr = max(lr * 0.85, 0.001)
            no_improve = 0

        # Periodic perturbation to escape local minima
        if no_improve > 1000:
            noise_scale = 0.02 / np.log(1 + it)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = np.ones(n) / n
                sa = 1.0
            a = a / sa
            a_prev = a.copy()
            y = a.copy()
            beta = 1.0
            lr = max(lr, 0.02)
            no_improve = 0

        if it % norm_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG iter={it} ({elapsed:.0f}s): "
                  f"best={best_score:.8f} lr={lr:.6f} no_imp={no_improve}")

    return best_a, best_score


def _lse_func(z):
    """Log-sum-exp objective (uses globals _G_LSE_N, _G_LSE_T)."""
    n = _G_LSE_N
    T = _G_LSE_T
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 1e15
    a = a / a_sum
    b = fast_autoconv(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    lse = T * (b_max / T + np.log(exp_b.sum()))
    return 2.0 * n * lse


def _lse_jac(z):
    """Log-sum-exp gradient via FFT cross-correlation."""
    n = _G_LSE_N
    T = _G_LSE_T
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return np.zeros(n)
    a = a / a_sum

    b = fast_autoconv(a)
    b_max = b.max()
    w = np.exp((b - b_max) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum

    # FFT-based cross-correlation: grad_a[m] = 2 * sum_j w[m+j] * a[j]
    N = 1 << (2 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    grad_a = irfft(np.conj(W) * A, N)[:n] * 2.0

    # Softmax chain rule
    S = float(np.dot(grad_a, a))
    return a * (grad_a - S) * 2.0 * n


def lbgfs_refine(a_init, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with log-sum-exp smoothing and temperature annealing.
    Uses FFT-accelerated gradients for O(N log N) per evaluation."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        _G_LSE_N = n
        _G_LSE_T = T

        try:
            res = minimize(
                fun=_lse_func, x0=z, jac=_lse_jac, method='L-BFGS-B',
                options={
                    'maxiter': maxiter_per_stage,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxfun': maxiter_per_stage * 100,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

        # Convert back and evaluate
        a_current = softmax_from_z(z)
        sc = evaluate_local(a_current)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={T:.3f} ({elapsed:.0f}s): score={sc:.8f}")

    a_final = softmax_from_z(z)
    return a_final


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def peak_coordinate_descent(a_init, time_limit, n_steps=50000):
    """Targeted coordinate descent on peak-contributing elements."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autoconv(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to the peak
        m_lo = max(0, peak_k - n + 1)
        m_hi = min(n - 1, peak_k)
        idx = np.arange(m_lo, m_hi + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-14:
                continue
            for factor in [0.997, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    break
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = int(rng.integers(0, 8))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x + rng.uniform(0, 6.28))
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.6 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.6 + 0.4 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Spectral GS initialization (multiple targets, multiple lengths)
    Phase 2: Nesterov subgradient descent
    Phase 3: L-BFGS with FFT-accelerated log-sum-exp gradients
    Phase 4: Coordinate descent / peak shaving
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    n_best = len(best_seq)

    # === Phase 1: Spectral GS Initialization (30% of time) ===
    print(f"\n=== Phase 1: Spectral GS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p1 = min(rem * 0.30, 500.0)
    if t_p1 > 20:
        print(f"  Budget: {t_p1:.0f}s")
        # Try current length first
        N_curr = 1 << (2 * n_best - 1).bit_length()
        a_gs, sc_gs = gs_multi_target(best_seq, n_best, N_curr, t_p1 * 0.5, rng)
        try_update(a_gs, "gs_curr")

        # Try alternative length
        rem = deadline - time.monotonic()
        if rem > 30:
            alt_n = int(n_best * 1.2)
            N_alt = 1 << (2 * alt_n - 1).bit_length()
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_gs_alt, sc_gs_alt = gs_multi_target(
                init_alt, alt_n, N_alt, rem * 0.4, rng)
            try_update(a_gs_alt, f"gs_n{alt_n}")

        # Another length
        rem = deadline - time.monotonic()
        if rem > 30:
            alt_n2 = int(n_best * 0.85)
            N_alt2 = 1 << (2 * alt_n2 - 1).bit_length()
            init_alt2 = interpolate_sequence(best_seq, alt_n2)
            a_gs_alt2, sc_gs_alt2 = gs_multi_target(
                init_alt2, alt_n2, N_alt2, rem * 0.4, rng)
            try_update(a_gs_alt2, f"gs_n{alt_n2}")

    # === Phase 2: Nesterov Subgradient (25% of time) ===
    print(f"\n=== Phase 2: Nesterov Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.25, 400.0)
    if t_p2 > 20:
        print(f"  Budget: {t_p2:.0f}s, n={len(best_seq)}")
        a_ng, sc_ng = nesterov_subgrad(best_seq, t_p2)
        try_update(a_ng, "nesterov_sg")

    # === Phase 3: L-BFGS with FFT gradients (30% of time) ===
    print(f"\n=== Phase 3: L-BFGS FFT (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.30, 500.0)
    if t_p3 > 20:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.02, 0.01]
        print(f"  T_stages: {T_stages}, budget={t_p3:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_p3, T_stages=T_stages, maxiter_per_stage=4000)
        try_update(a_lb, "lbgfs_fft")

    # === Phase 4: Coordinate Descent (10% of time) ===
    print(f"\n=== Phase 4: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.10, 150.0)
    if t_p4 > 10:
        a_cd, sc_cd = peak_coordinate_descent(best_seq, t_p4, n_steps=80000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Second L-BFGS pass at very low temperature (5% of time) ===
    print(f"\n=== Phase 5: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.08, 100.0)
    if t_p5 > 10:
        T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.03, 0.02, 0.01, 0.005]
        a_lb2 = lbgfs_refine(best_seq, t_p5, T_stages=T_stages2, maxiter_per_stage=5000)
        try_update(a_lb2, "lbgfs_final")

    # === Phase 6: Spectral perturbation + re-optimize (remaining time) ===
    print(f"\n=== Phase 6: Spectral Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        N_cur = 1 << (2 * n_cur - 1).bit_length()
        t_pert = min(rem * 0.5, 200.0)
        per_pert = t_pert / 3.0

        # Perturb in frequency domain
        a_base = best_seq.copy()
        for pert_i in range(3):
            if time.monotonic() > deadline - 30:
                break
            A = np.fft.rfft(np.concatenate([a_base, np.zeros(N_cur - n_cur)]))
            # Random phase perturbation
            phase_noise = rng.standard_normal(len(A)) * 0.1
            A_perturbed = A * np.exp(1j * phase_noise)
            a_pert = np.fft.irfft(A_perturbed, N_cur)[:n_cur]
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-15:
                continue
            a_pert = a_pert / sp

            # Quick GS refinement
            target = np.ones(N_cur // 2 + 1)
            target[0] = np.sqrt(n_cur)
            a_ref, sc_ref = gs_optimize(a_pert, target, n_cur, N_cur,
                                         max_iters=int(per_pert * 2000))
            try_update(a_ref, f"pert_gs_{pert_i}")

            # Quick L-BFGS
            rem2 = deadline - time.monotonic()
            if rem2 > 20:
                T_short = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]
                a_ref2 = lbgfs_refine(a_ref, min(rem2 * 0.3, 60.0),
                                      T_stages=T_short, maxiter_per_stage=2000)
                try_update(a_ref2, f"pert_lbgfs_{pert_i}")

    # === Phase 7: Final coordinate descent polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin, _ = peak_coordinate_descent(best_seq, min(rem * 0.4, 30.0), n_steps=20000)
        try_update(a_fin, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 10:
        T_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_fin2 = lbgfs_refine(best_seq, min(rem * 0.5, 20.0),
                              T_stages=T_final, maxiter_per_stage=3000)
        try_update(a_fin2, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
