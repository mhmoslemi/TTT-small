from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def score_raw(a):
    """Score for unnormalized sequence (matches evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_logsumexp(a, T):
    """Compute (1/T)*log(sum_k exp(T*b_k)) where b = conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    if m > 700:
        return b.max()
    e = np.exp(Tb - m)
    val = (m + np.log(e.sum())) / T
    return val


def grad_smooth_logsumexp(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j w[j+i] * a[j] for valid i,j
    # This is 2 * (reverse-correlate of a and w)
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # We need sum_j w[j] * a[j+i] for each i, which is cross-correlation
    corr = irfft(np.conj(A) * W, fft_size)[:len(w)]
    # grad[i] = 2 * sum_j a[j] * w[j+i]
    # Let's use: grad = 2 * (a * w) in the right sense
    # conv_a_w[k] = sum_j a[j]*w[k-j], we want sum_j a[j]*w[j+i] = conv_a_w[i+...] 
    # Actually: sum_j a[j] * w[j+i] for j=0..n-1 (where w[j+i] valid)
    # This is the cross-correlation of a and w at lag i
    # In FFT: ifft(conj(A) * W) gives sum_j a[j] * conj(w[j+k]) but for real it's sum_j a[j]*w[j+k]... 
    # Let me just do it directly with a different approach
    # grad[i] = 2 * sum_j a[j] * w[j+i] where 0<=j<n and 0<=j+i<len(w)
    # This equals 2 * (the (i)-th element of the cross-correlation of a with w)
    # Using FFT: cross_corr[i] = sum_j a[j] * w[j+i]
    # = ifft(fft(a) * fft(w)) [usize] -- actually this gives convolution not correlation
    # Let me use: if w_rev = w[::-1], then cross_corr = convolve(a, w_rev) appropriately
    
    # Simpler: use the fact that b = conv(a,a), db/da[i] = 2*sum_k w[k]*db_k/da[i]
    # db_k/da[i] = a[k-i] if 0<=k-i<n, i.e. the (k-i)-th element of a
    # So grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * (cross-correlation of w and a at lag i)
    # cross_corr[i] = sum_k w[k] * a[k-i]
    # In terms of convolution: if we reverse a, conv_a_rev[k] = sum_j a_rev[j]*w[k-j] = sum_j a[n-1-j]*w[k-j]
    # Let m = n-1-j: = sum_m a[m]*w[k-n+1+m]
    # This is getting complex. Let me just use a direct FFT approach.
    
    # The cleanest: grad = 2 * T * (a * w) where * denotes the appropriate convolution
    # b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = a[k-i] (if valid) + a[i-(k)] (if i>k, but by symmetry of auto-conv, 
    # actually db[k]/da[i] = a[k-i] if 0<=k-i<n else 0, times 2 (since a appears twice in b)
    # Wait no: b[k] = sum_j a[j]*a[k-j]. d/d(a[i]) = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    # = a[k-i] (if 0<=k-i<n) + a[k-i] (if k-i... wait)
    # = a[k-i] if 0<=k-i<n (from first term, j=i gives a[i]*a[k-i])
    # + a[k-i] if 0<=i<n and k-i ranges... hmm
    # Actually: d b[k]/d a[i] = a[k-i] (from d/d a_i of a_i * a[k-i], when j=i) 
    #                              + a[k-i] (from d/d a_i of a[j]*a[k-j], when k-j=i, i.e., j=k-i)
    # But these are the same term! When j=i, we get a[i]*a[k-i]. When j=k-i, we get a[k-i]*a[i].
    # If i != k-i (i.e., 2i != k), these are two separate terms, so the derivative is 2*a[k-i].
    # If i == k-i (i.e., 2i == k), it's a single term a[i]^2, derivative is 2*a[i].
    # Either way: d b[k]/d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    # (The factor of 2 comes from the chain rule on a product where both factors depend on a[i])
    
    # So grad[i] = sum_k w[k] * 2 * a[k-i] (for valid k-i, i.e., 0<=k-i<n)
    # = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]  (letting m = k-i, so k = i+m)
    # = 2 * (correlation of w with a, evaluated at lag i)
    
    # Using FFT: cross_corr[i] = sum_m a[m] * w[m+i]
    # This is ifft(fft(a) * fft(w)) at appropriate position? No.
    # sum_m a[m] * w[m+i] = (a * w_flipped)[i + n - 1]... let me just use numpy correlate
    # Actually for efficiency with FFT:
    # sum_m a[m] * w[m+i] for m=0..n-1 = sum_m a[m] * w_rev[w.size-1-m-i] ... 
    
    # Let me just compute it as: pad and use convolution
    # w has length 2n-1, a has length n
    # result[i] = sum_{m=0}^{n-1} a[m] * w[m+i] for i=0..n-1
    # = the first n elements of the "cross-correlation" of w and a
    # cross_correlation[i] = sum_m w[m+i]*a[m]
    # This equals: ifft(conj(fft(a, N)) * fft(w, N)) -- no that's not right either
    # Actually: ifft(fft(a) * fft(w))[k] = sum_j a[j] * w[k-j] (circular conv)
    # We want sum_m a[m] * w[m+i] = sum_m a[m] * w_rev[w.size-1-m-i]
    # Getting confused. Let me just do it with a single FFT multiply.
    
    # The cleanest approach: 
    # result = np.correlate(w, a, mode='valid') -- but w is longer than a
    # np.correlate(w, a)[i] = sum_j w[j+i] * a[j] for j=0..len(a)-1
    # That's exactly what we want! But it's O(n^2).
    
    # FFT approach: 
    # We want C[i] = sum_j a[j] * w[j+i] for j=0..n-1, i=0..n-1
    # Reverse w: wR = w[::-1], length 2n-1
    # C[i] = sum_j a[j] * w[j+i]
    # Let's compute conv(a, wR): conv(a,wR)[k] = sum_j a[j]*wR[k-j] = sum_j a[j]*w[2n-2-(k-j)] = sum_j a[j]*w[2n-2-k+j]
    # We want sum_j a[j]*w[j+i]. Let j' = j: sum_j a[j]*w[j+i]
    # From conv: sum_j a[j]*w[2n-2-k+j] = C[i] when 2n-2-k = i, i.e., k = 2n-2-i.
    # So C[i] = conv(a, wR)[2n-2-i]
    
    N = 1 << (n + 2*n - 1).bit_length()  # enough for linear conv of a (len n) and wR (len 2n-1)
    A = rfft(a, N)
    WR = rfft(w[::-1], N)
    conv_raw = irfft(A * WR, N)
    # C[i] = conv_raw[2n-2-i] for i=0..n-1
    C = conv_raw[2*n-2 : n-1 : -1]  # indices 2n-2, 2n-3, ..., n-1
    # Wait: C[i] = conv_raw[2n-2-i], so C[0] = conv_raw[2n-2], C[1] = conv_raw[2n-3], ...
    # C[n-1] = conv_raw[n-1]
    C = conv_raw[n-1 : 2*n-1]  # this gives conv_raw[n-1], ..., conv_raw[2n-2]
    # We need: C[i] = conv_raw[2n-2-i], so C = conv_raw[n-1:2n-1][::-1]
    C = conv_raw[n-1 : 2*n-1][::-1]
    
    return 2.0 * C


def grad_smooth_simple(a, T):
    """Simpler gradient computation for log-sum-exp smoothed max."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_{m=0}^{n-1} a[m] * w[m+i]
    # Use FFT: 
    N = 1 << (n + len(w)).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    # We need C[i] = sum_m a[m]*w[m+i] = cross-correlation
    # = ifft(conj(W) * A) at appropriate positions? 
    # ifft(conj(W)*A)[k] = sum_j w[j]*a[j+k] (circular)
    # For linear, we need N >= n + len(w) - 1, which we have.
    # So C[i] = ifft(conj(W)*A)[i] for i=0..n-1 (with appropriate handling)
    # Actually: ifft(conj(W)*A)[k] = sum_j conj(w[j])*a[j+k] = sum_j w[j]*a[j+k] (real case)
    # We want sum_m a[m]*w[m+i] = sum_m w[i+m]*a[m]
    # Let j = i+m: not quite the same form.
    # sum_m a[m]*w[m+i]: this is sum_m w[m+i]*a[m]
    # vs ifft(conj(W)*A)[i] = sum_j w[j]*a[j+i] = sum_j w[j]*a[j+i]
    # These are transposes. Let me use:
    # sum_m a[m]*w[m+i] = sum_m a[m]*w[i+m]
    # If I define w_shift[j] = w[i+j], this is a dot product of a and w_shift.
    # 
    # Alternative: use the fact that this is a "sliding dot product" = cross-correlation.
    # corr[i] = sum_m a[m]*w[m+i]
    # In FFT: if I compute ifft(fft(w) * conj(fft(a)))[i], I get sum_m w[m]*conj(a[m-i])
    # For real: sum_m w[m]*a[m-i] = sum_j w[j+i]*a[j] (let j=m-i)
    # That's sum_j w[j+i]*a[j] which is what we want!
    result = irfft(W * np.conj(A), N)
    C = result[:n]  # first n elements
    return 2.0 * C


def fista_smoothed(a0, time_limit, T_start=1.0, T_end=500.0, n_stages=15,
                    steps_per_stage=2000, mu_init=1.0):
    """FISTA with backtracking on log-sum-exp smoothed max."""
    n = len(a0)
    x = project_simplex(a0.copy())
    y = x.copy()
    z = x.copy()  # momentum variable
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    t_k = 1.0  # FISTA momentum parameter
    prev_obj = smooth_logsumexp(x, T_start)
    stall = 0
    
    for stage in range(n_stages):
        T = temps[stage]
        mu = mu_init  # Lipschitz estimate for backtracking
        t_curr = t_k
        
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # FISTA momentum update
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
            beta = (t_curr - 1.0) / t_next
            y = z + beta * (z - x)
            y_projected = project_simplex(y)
            
            # Gradient
            grad = grad_smooth_simple(y_projected, T)
            
            # Backtracking line search
            while True:
                trial = project_simplex(y_projected - y_projected * 0 + mu * (mu * (y_projected - grad / mu)))
                # Simplified: trial = project_simplex(y - grad/mu)
                nu = np.linalg.norm(y_projected - grad / mu)
                step_size = min(mu, 2.0 * np.max(np.abs(grad)))
                trial = project_simplex(y_projected - grad / step_size)
                obj_trial = smooth_logsumexp(trial, T)
                obj_y = smooth_logsumexp(y_projected, T)
                # Quadratic upper bound check (simplified: just check improvement)
                if obj_trial < obj_y + 1e-10 or step_size > 1e4:
                    x_new = trial
                    break
                step_size *= 1.5
            
            x_new = project_simplex(y_projected - grad / step_size)
            obj_new = smooth_logsumexp(x_new, T)
            
            # Restart if objective increased (helps with non-convexity)
            if obj_new > prev_obj + 0.01:
                t_next = 1.0
                x_new = x  # restart from previous
                obj_new = prev_obj
            
            t_k = t_next
            z = x
            x = x_new
            prev_obj = obj_new
            
            # Check actual score periodically
            if step % 20 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    stall = 0
                else:
                    stall += 1
                
                if stall > 200:
                    # Add perturbation and restart momentum
                    rng = np.random.default_rng(step)
                    x = x + 0.01 * rng.standard_normal(n)
                    x = project_simplex(x)
                    y = x.copy()
                    z = x.copy()
                    t_k = 1.0
                    stall = 0
            
            if step % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      FISTA stage={stage} step={step} ({elapsed:.0f}s): "
                      f"smooth={obj_new:.6f} T={T:.1f} score={best_score:.6f}")
    
    return best_a, best_score


def nesterov_hybrid(a0, time_limit, T_start=3.0, T_end=800.0,
                    n_stages=12, steps_per_stage=3000, lr=0.04):
    """Nesterov accelerated gradient with hybrid step size."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0
    best_smooth = smooth_logsumexp(x, T_start)
    stall = 0
    
    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev = 1.0 + np.sqrt(1.0 + 4.0 * max(0, step_count - 1) ** 2) / 2.0
            beta = (t_prev - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            
            # Adaptive learning rate
            grad = grad_smooth_simple(y, T)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            lr_adapt = lr / (1.0 + 0.002 * step_count)
            
            x_prev = x.copy()
            x = project_simplex(y - lr_adapt * grad / max(1.0, grad_norm))
            step_count += 1
            
            if step % 10 == 0:
                obj = smooth_logsumexp(x, T)
                if obj < best_smooth - 1e-6:
                    best_smooth = obj
                    stall = 0
                else:
                    stall += 1
                
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                
                # Restart on sustained stall
                if stall > 300:
                    rng = np.random.default_rng(step_count)
                    x = x + 0.03 * rng.standard_normal(n)
                    x = project_simplex(x)
                    x_prev = x.copy()
                    stall = 0
                    best_smooth = smooth_logsumexp(x, T)
            
            if step % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      NG stage={stage} step={step} ({elapsed:.0f}s): "
                      f"T={T:.1f} score={best_score:.6f}")
    
    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build sparse representation for conv matrix to use in LP."""
    n = len(a)
    size = 2 * n - 1
    # T @ b = convolve(a, b): T[k, j] = a[k-j] if 0<=k-j<n
    # For LP, we need T explicitly but it's (2n-1) x n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CBC, cp.GLPK, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                print(f"  LP solver: {solver_name}")
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, n_limit=5000):
    """LP: minimize M s.t. T@a b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    if n > n_limit:
        return None, None
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_refine(a0, time_limit, solver=None, n_iters=20000):
    """LP-based refinement with Nesterov dual averaging."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            b_new, _ = solve_lp_best_response(a, solver)
            if b_new is None:
                continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        
        # Nesterov average
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)
        
        if (it + 1) % 10 == 0:
            sc_a = score_normalized(a)
            sc_y = score_normalized(y)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            if sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
            
            # Perturbation on stall
            if it % 2000 == 0:
                rng = np.random.default_rng(it)
                y = y + 0.02 * rng.standard_normal(n)
                y = project_simplex(y)
    
    return best_a, best_score


def peak_shave_v2(a0, time_limit, n_steps=50000):
    """Improved peak shaving: targeted reduction of top contributors."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find top peaks (excluding boundaries)
        b_interior = b.copy()
        b_interior[0] = -np.inf
        b_interior[-1] = -np.inf
        peak_k = int(np.argmax(b_interior))
        peak_val = b[peak_k]
        
        if peak_val < 1e-15:
            break
        
        # Find contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_n = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        improved = False
        
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if j >= n:
                continue
            # Try reducing a[i]
            for f in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
            # Try reducing a[j] (if different from i)
            if j != i:
                for f in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[j] *= f
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try pair reduction
            for ci1 in top_idx[:4]:
                for ci2 in top_idx[:4]:
                    i1 = idx[ci1]
                    j1 = peak_k - i1
                    i2 = idx[ci2]
                    j2 = peak_k - i2
                    if j1 >= n or j2 >= n:
                        continue
                    candidates = set([i1, j1, i2, j2])
                    candidates.discard(i2)  # already tried single
                    for cand in sorted(candidates):
                        if a[cand] < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[cand] *= 0.92
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Broader move: reduce top 3 simultaneously
            a_trial = a.copy()
            for ci in top_idx[:3]:
                i = idx[ci]
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                # Try increasing a coordinate that doesn't contribute to peak
                non_contrib = np.full(n, False)
                for k_top in np.argsort(b)[::-1][:3]:
                    js2 = max(0, k_top - n + 1)
                    je2 = min(n - 1, k_top)
                    non_contrib[js2:je2+1] = True
                candidates = np.where(~non_contrib)[0]
                if len(candidates) > 0:
                    rng = np.random.default_rng(step)
                    pick = candidates[rng.integers(len(candidates))]
                    a_trial = a.copy()
                    a_trial[pick] += 0.05 * a.max()
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                    else:
                        break
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=10000):
    """Cyclic coordinate descent minimizing the max convolution peak."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # For each coordinate i, find the best value that reduces the peak
        # while keeping sum=1 and a>=0
        # This is expensive, so we use a cheaper heuristic:
        # Find the coordinate that most affects the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
            
        top_i = idx[np.argmax(contribs)]
        top_j = peak_k - top_i
        if top_j >= n:
            top_j = peak_k - top_i
            if top_j >= n:
                break
        
        # Try reducing the larger of a[top_i], a[top_j]
        if a[top_i] >= a[top_j]:
            target = top_i
        else:
            target = top_j
        
        for f in [0.98, 0.95, 0.90, 0.85, 0.80]:
            a_trial = a.copy()
            a_trial[target] *= f
            a_trial = project_simplex(a_trial)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break
        else:
            # No improvement, try a small random perturbation
            if rnd % 100 == 0:
                rng = np.random.default_rng(rnd)
                for _ in range(5):
                    i = int(rng.integers(n))
                    a_trial = a.copy()
                    a_trial[i] *= (0.9 + 0.2 * rng.random())
                    a_trial = project_simplex(a_trial)
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_init(n, mode, rng):
    """Create various initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Uniform with slight edge reduction
        a = 1.0 - 0.3 * np.abs(2*x - 1)**2
    elif mode == 1:
        # Sine shape
        a = 1.0 + 0.5 * np.sin(2*np.pi*x + 0.3)
    elif mode == 2:
        # Exponential decay
        a = np.exp(-1.5*x) + 0.3
    elif mode == 3:
        # Bimodal
        a = np.exp(-20*(x-0.3)**2) + np.exp(-20*(x-0.7)**2) + 0.1
    elif mode == 4:
        # Random smooth
        a = 1.0 + 0.4 * np.sin(4*np.pi*x + rng.random()*2) + 0.2*np.sin(7*np.pi*x)
    elif mode == 5:
        # Inverse U
        a = 0.5 + 0.5*np.cos(np.pi*x)**2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with FISTA, LP refinement, peak shaving,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(60.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass
    
    if parent is None:
        n_init = 960
        best_seq = make_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: FISTA on parent ===
    print(f"\n=== Phase 1: FISTA n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_f, _ = fista_smoothed(best_seq, t_alloc, T_start=3.0, T_end=600.0,
                                n_stages=14, steps_per_stage=2500)
        try_update(a_f, "fista1")
    
    # === Phase 2: Nesterov hybrid ===
    print(f"\n=== Phase 2: Nesterov Hybrid (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ng, _ = nesterov_hybrid(best_seq, t_alloc, T_start=5.0, T_end=1000.0,
                                  n_stages=12, steps_per_stage=3000, lr=0.06)
        try_update(a_ng, "nesterov_hybrid")
    
    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30 and n_best <= 2000:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_refine(best_seq, t_alloc, solver=solver, n_iters=30000)
        try_update(a_lp, "lp_main")
    
    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ps, _ = peak_shave_v2(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")
    
    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_rounds=20000)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Multi-start with different inits ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    modes_to_try = [1, 2, 3, 4, 5]
    for mode in modes_to_try:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.15, 120.0)
        if t_sub < 30:

        a_init = make_init(n_best, mode, rng)

        try_update(a_sub, f"init_m{mode}")

        if rem > 20 and n_best <= 2000:
            a_sub2, _ = lp_refine(a_sub, min(rem * 0.2, 30.0), solver=solver, n_iters=3000)
            try_update(a_sub2, f"init_m{mode}_lp")
    
    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for delta in [-100, -50, -20, 20, 50, 100, 200, 300, 500]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 4096:

    alt_lengths = sorted(set(alt_lengths))

            a_alt2, _ = lp_refine(a_alt, min(rem * 0.3, 40.0), solver=solver, n_iters=5000)
            try_update(a_alt2, f"lp_n{alt_n}")
    
    # === Phase 8: Final FISTA polish with high T ===
    print(f"\n=== Phase 8: Final FISTA Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.20, 80.0)
        a_fin, _ = fista_smoothed(best_seq, t_fin, T_start=50.0, T_end=2000.0,
                                   n_stages=8, steps_per_stage=2000)
        try_update(a_fin, "final_fista")
    
    # === Phase 9: Final LP polish ===
    rem = deadline - time.monotonic()
    if rem > 20 and len(best_seq) <= 2000:
        t_lp2 = min(rem * 0.20, 60.0)
        a_lp2, _ = lp_refine(best_seq, t_lp2, solver=solver, n_iters=10000)
        try_update(a_lp2, "final_lp")
    
    # === Phase 10: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps2 = min(rem * 0.15, 40.0)
        a_ps2, _ = peak_shave_v2(best_seq, t_ps2, n_steps=10000)
        try_update(a_ps2, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
