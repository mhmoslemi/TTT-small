```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_smoothed_grad(a, t):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    q = fast_convolve(a)  # length 2n-1
    # Softmax with temperature t
    q_shifted = t * (q - q.max())
    p = np.exp(np.clip(q_shifted, -50, 0))
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros_like(q)
        p[q.argmax()] = 1.0
    else:
        p = p / p_sum
    # Cross-correlation: grad[m] = 2 * sum_j p[m+j] * a[j]
    # = 2 * (rev(a) * p)[m + n - 1]
    ar = a[::-1]
    fft_size = 1 << (n + 2*n - 2 - 1).bit_length()  # >= len(ar) + len(p) - 1
    FA = rfft(ar, fft_size)
    FP = rfft(p, fft_size)
    conv_ar_p = irfft(FA * FP, fft_size)[:n + 2*n - 2]
    grad = 2.0 * conv_ar_p[n-1 : 2*n-1]  # length n
    return grad, q.max()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smoothed gradient descent (log-sum-exp approximation of max) via FFT
2. Multiplicative weights (mirror descent) with adaptive temperature
3. Coordinate descent polishing
4. Multi-length, multi-init exploration
5. Simulated annealing restarts for escaping local optima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_smoothed_grad(a, t):
    """
    Compute gradient of smooth approximation f_t(a) = (1/t)log(sum_k exp(t*q_k(a)))
    where q_k(a) = conv(a,a)[k].
    Returns (grad, max_q, active_weights).
    """
    n = len(a)
    q = fast_convolve(a)
    max_q = float(q.max())
    # Softmax with temperature t
    q_shifted = t * (q - max_q)
    p = np.exp(np.clip(q_shifted, -50, 0))
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros_like(q)
        p[int(q.argmax())] = 1.0
        p_sum = 1.0
    else:
        p = p / p_sum
    # Cross-correlation: grad[m] = 2 * sum_j p[m+j] * a[j]
    # = 2 * (rev(a) * p)[m + n - 1]
    ar = a[::-1]
    fft_size = 1 << (n + (2 * n - 1) - 1).bit_length()
    FA = rfft(ar, fft_size)
    FP = rfft(p, fft_size)
    conv_ar_p = irfft(FA * FP, fft_size)[:n + 2 * n - 2]
    grad = 2.0 * conv_ar_p[n - 1:2 * n - 1]
    return grad, max_q, p


def smoothed_gd(a0, time_limit, t_init=5.0, t_max=500.0, lr_init=0.5,
                use_mirror=True, verbose=False):
    """
    Smoothed gradient descent with adaptive temperature.
    Uses log-sum-exp to smooth the max of convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t = t_init
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 200

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, max_q, p = compute_smoothed_grad(a, t)

        if use_mirror:
            # Multiplicative weights update
            a_new = a * np.exp(-lr * grad)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-30:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s
        else:
            a_new = simplex_project(a - lr * grad)

        a = a_new
        score = 2.0 * n * max_q

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: increase when plateau
        if no_improve >= patience:
            t = min(t * 1.5, t_max)
            lr = max(lr * 0.9, 0.01)
            no_improve = 0

        # Slow increase of t
        if it % 100 == 0:
            t = min(t * 1.05, t_max)

        # Decay learning rate
        if it % 500 == 0:
            lr = max(lr * 0.95, 0.005)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): score={best_score:.6f} t={t:.1f} lr={lr:.4f}")

        if lr < 0.001 and t > t_max * 0.9:
            break

    return best_a, best_score


def coord_descent(a0, time_limit, n_trials=8, verbose=False):
    """
    Efficient coordinate descent for minimizing max(conv(a,a)) with sum(a)=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            # Affected conv positions: [i, i+n-1]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1].copy()

            best_delta = 0.0
            best_new_max = M

            # Try various fractions
            for f in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                for sign in [-1, 1]:
                    delta = sign * f * a[i]
                    new_val = a[i] + delta
                    if new_val < -1e-15:
                        continue

                    new_aff = affected + 2.0 * delta * a
                    # Correction for the diagonal term (k = 2i)
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff = new_aff.copy()
                        new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                    new_max = max(unaff_max, float(new_aff.max()))
                    if new_max < best_new_max - 1e-16:
                        best_new_max = new_max
                        best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.10f}")

    return best_a, best_score

def init_uniform(n):
    return np.ones(n) / n


def init_symmetric(n, rng):
    """Symmetric initialization with smooth profile."""
    half = (n + 1) // 2
    t = np.linspace(0, 1, half)
    n_comp = int(rng.integers(2, 7))
    profile = np.ones(half) * 0.5
    for j in range(n_comp):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    if n % 2 == 0:
        a = np.concatenate([profile[:-1], profile[::-1]])
    else:
        a = np.concatenate([profile, profile[-2::-1]])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w1 = n * rng.uniform(0.05, 0.15)
    w2 = n * rng.uniform(0.05, 0.15)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    """Cosine-based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * np.cos(np.pi * t / n)
    a += rng.uniform(0, 0.1) * np.random.standard_normal(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def init_triangular(n, rng):
    """Triangular (tent) shape."""
    t = np.arange(n, dtype=np.float64)
    center = n * rng.uniform(0.3, 0.7)
    a = 1.0 - np.abs(t - center) / (n * 0.6)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def init_periodic(n, rng):
    """Periodic structure."""
    period = int(n / rng.integers(3, 10))
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + rng.uniform(0.2, 0.6) * np.cos(2.0 * np.pi * t / period)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def anneal_restart(a, rng, strength=0.05):
    """Perturb a sequence for simulated annealing restart."""
    n = len(a)
    a_new = a * (1.0 + strength * rng.standard_normal(n))
    a_new = np.maximum(a_new, 0)
    # Zero out a small fraction
    frac_zero = int(n * 0.05)
    if frac_zero > 0:
        zero_idx = rng.choice(n, frac_zero, replace=False)
        a_new[zero_idx] = 0
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

                # Run smoothed GD from current
                a_gd, s_gd = smoothed_gd(current, t_alloc, t_init=10.0, t_max=300.0,
                                         lr_init=0.2, use_mirror=True)
                check_update(a_gd, f"anneal_{rnd}")

                # CD polish
                rem = deadline - time.monotonic() - 300
                if rem > 10:
                    a_cd, s_cd = coord_descent(a_gd, min(rem * 0.3, 40.0))
                    check_update(a_cd, f"anneal_cd_{rnd}")

                # Anneal restart
                if rnd < 6:
                    strength = 0.03 + 0.02 * rnd
                    current = anneal_restart(a_gd, rng, strength=strength)
                else:
                    current = best_seq.copy()

    # ============================================================
    # Phase 4: Multi-SGD with different temperatures
    # ============================================================
    print(f"\n=== Phase 4: Multi-temp SGD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for t_try in [3.0, 8.0, 20.0, 50.0]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 50.0)
            a_gd, s_gd = smoothed_gd(best_seq, t_alloc, t_init=t_try, t_max=t_try * 20,
                                     lr_init=0.4, use_mirror=True)
            check_update(a_gd, f"multit_{t_try}")

    # ============================================================
    # Phase 5: Fine coordinate descent polishing
    # ============================================================
    print(f"\n=== Phase 5: Fine CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_fine = min(rem * 0.5, 120.0)
            print(f"  FineCD budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine, verbose=True)
            check_update(a_fine, "fine_cd")

        # Another round
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_f2, s_f2 = fine_coord_descent(best_seq, min(rem * 0.6, 60.0))
            check_update(a_f2, "fine_cd_2")

    # ============================================================
    # Phase 6: Try alternative lengths
    # ============================================================
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alt_n in [n_best + 128, n_best + 256, n_best - 128, n_best + 384]:
            if alt_n < 128 or alt_n > 3072:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            init_alt = interpolate_seq(best_seq, alt_n)
            t_alloc = min(rem * 0.4, 40.0)
            a_gd, s_gd = smoothed_gd(init_alt, t_alloc, t_init=5.0, t_max=100.0)
            check_update(a_gd, f"alt_n{alt_n}")

            rem = deadline - time.monotonic() - 20
            if rem > 8:
                a_cd, s_cd = coord_descent(a_gd, min(rem * 0.5, 25.0))
                check_update(a_cd, f"alt_cd_n{alt_n}")

    # ============================================================
    # Phase 7: Final intensive CD
    # ============================================================
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_final, s_final = fine_coord_descent(best_seq, min(rem * 0.8, 60.0))
            check_update(a_final, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
