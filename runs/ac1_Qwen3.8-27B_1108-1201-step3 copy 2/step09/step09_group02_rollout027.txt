import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def log_sum_exp(x, T):
    """Stable computation of T * log(sum(exp(x/T)))."""
    x_shifted = x / T - x.max() / T
    return T * (x.max() / T + np.log(np.sum(np.exp(x_shifted))))

def softmax(x, T):
    """Compute softmax(x/T) stably."""
    x_shifted = x / T - x.max() / T
    e = np.exp(x_shifted)
    return e / e.sum()

def compute_gradient_smoothed(a, conv, T):
    """
    Compute the gradient of 2n * soft_max(conv, T) wrt a.
    soft_max(conv, T) = T * log(sum(exp(conv/T)))

                       = sum_k w[k] * 2*a[k-i] (for valid k-i)

    cor = irfft(np.conj(W) * A, fft_size)

    return grad_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with simplex projection and geometric annealing
2. Cross-correlation based gradient computation via FFT (O(n log n) per step)

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def softmax_stable(x, T):
    """Compute softmax(x/T) stably."""
    xs = x / T
    xs = xs - xs.max()
    e = np.exp(xs)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_smooth_gradient(a, conv, T):
    """
    Compute gradient of 2n * T * log(sum(exp(conv/T))) wrt a.
    = 2n * sum_k softmax(conv/T)[k] * d(conv[k])/da
    where d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n else 0.
    Result is cross-correlation of w=softmax(conv/T) with a, times 2n.
    """
    n = len(a)
    w = softmax_stable(conv, T)
    # cross-correlation: result[i] = sum_k w[k] * a[k-i] for valid k-i
    # Use FFT: IFFT(conj(FFT(w)) * FFT(a))
    fft_size = 1 << (len(conv) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cor = irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * n * cor[:n]
    return grad

    else:
        a = a / s

    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()

    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(a)
        cur_max = float(conv.max())

def peak_surgical_step(a, conv, n_peaks=5, alpha=0.01):
    """
    Directly reduce the top peaks by transferring mass from high-contribution
    positions to low-contribution positions.
    """
    n = len(a)
    a_new = a.copy()

    # Find top peaks
    top_k = min(n_peaks, 2 * n - 1)
    peak_indices = np.argsort(conv)[-top_k:]

    for k in peak_indices:
        k = int(k)
        if conv[k] <= float(conv.max()) * 0.7:
            continue

        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        indices = np.arange(i_start, i_end + 1)
        partners = k - indices
        products = a[indices] * a[partners]

        if len(indices) == 0:
            continue

        # Find the position with max product
        max_idx_local = np.argmax(products)
        i_max = indices[max_idx_local]
        p_max = partners[max_idx_local]

        # Find a "safe" position (low contribution to all top peaks)
        safe_scores = np.zeros(n)
        for kp in peak_indices:
            kp = int(kp)
            for i in range(max(0, kp - n + 1), min(n - 1, kp) + 1):
                j = kp - i
                if 0 <= j < n:
                    safe_scores[i] += a[j]

        # Pick a low-contribution position to receive mass
        # Exclude positions that already have high values
        candidate_mask = safe_scores < np.median(safe_scores)
        if not candidate_mask.any():
            continue
        safe_positions = np.where(candidate_mask)[0]
        if len(safe_positions) == 0:
            continue
        # Pick the one with lowest current value
        target = safe_positions[np.argmin(a[safe_positions])]

        # Transfer mass
        amt = alpha * a[i_max]
        if i_max == target:
            continue
        if i_max == p_max:
            # Would reduce the product by reducing i_max
            a_new[i_max] -= amt
            a_new[target] += amt
        else:
            # Reduce the larger of a[i_max], a[p_max]
            if a[i_max] >= a[p_max]:
                a_new[i_max] -= amt
                a_new[target] += amt
            else:
                a_new[p_max] -= amt
                a_new[target] += amt

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def golden_section_search_1d(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_gaussian(n, rng, asymmetric=False):
    """Gaussian bell initialization."""
    t = np.arange(n, dtype=np.float64)
    if asymmetric:
        center = n * rng.uniform(0.25, 0.75)
    else:
        center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.5)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 12)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.choice(np.arange(1, n // 4 + 1), size=min(n_harmonics, n // 4), replace=False)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(len(freqs)):
        a += amps[j] * np.cos(2.0 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0.0)
    s = a.sum()

        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def combined_optimize(a0, time_limit, verbose=False):
    """
    Combined optimization: smooth gradient + peak surgery + coordinate descent.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())

    # Phase A: Smooth gradient descent (60% of time)
    t_a = time_limit * 0.45
    a, sc = smooth_gradient_optimize(a, t_a, T_start=0.06, T_end=0.003,
                                      lr_init=0.02, verbose=verbose, label="grad")
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    # Phase B: Peak surgical refinement (15% of time)
    t_b = time_limit * 0.15
    a_cur = a.copy()
    t_b_start = time.monotonic()
    while time.monotonic() - t_b_start < t_b:
        conv = fast_convolve(a_cur)
        sc = 2.0 * n * float(conv.max())
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        a_new = peak_surgical_step(a_cur, conv, n_peaks=3, alpha=0.005)
        sc_new = 2.0 * n * float(fast_convolve(a_new).max())
        if sc_new < sc:
            a_cur = a_new
        else:
            # Try smaller step
            a_new2 = peak_surgical_step(a_cur, conv, n_peaks=3, alpha=0.001)
            sc_new2 = 2.0 * n * float(fast_convolve(a_new2).max())
            if sc_new2 < sc:
                a_cur = a_new2
            else:
                break

    # Phase C: Exact coordinate descent (30% of time)
    t_c = time_limit * 0.30
    a, sc = exact_coord_descent(a_cur, t_c, verbose=verbose, label="ecd")
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    # Phase D: Fine coordinate descent (remaining time)
    t_rem = time_limit - (time.monotonic() - t0)
    if t_rem > 5:
        a, sc = fine_coord_descent(a, t_rem, verbose=verbose, label="fine")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization search for best sequence:
    1. Diverse initialization with smooth gradient optimization
    2. Intensive coordinate descent on best candidates
    3. Basin hopping with structured perturbations
    4. Final fine polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(123)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        n_values = [n_parent]
        for alt in [n_parent - 128, n_parent + 128, n_parent - 256, n_parent + 256]:
            if 256 <= alt <= 2048 and alt not in n_values:
                n_values.append(alt)
        n_values.append(1024)
        n_values.append(1536)
    else:
        n_values = [512, 1024, 1536]

    # === Phase 1: Multi-start smooth gradient optimization ===
    print("\n=== Phase 1: Smooth Gradient Multi-Start ===")
    t1_start = time.monotonic()
    candidates = []

    for n in n_values[:4]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 300:
            break

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('flat_spec', init_flat_spectrum(n, rng)))
        inits.append(('gauss', init_gaussian(n, rng)))
        inits.append(('gauss_asym', init_gaussian(n, rng, asymmetric=True)))
        inits.append(('fourier', init_fourier_mix(n, rng)))
        inits.append(('flat_spec2', init_flat_spectrum(n, rng)))

        for itype, cand in inits:
            t_left = 300.0 - (time.monotonic() - t1_start)
            if t_left < 15:
                break
            t_alloc = min(t_left / 2.0, 50.0)
            try:
                a_opt, s_opt = smooth_gradient_optimize(cand, t_alloc,
                                                         T_start=0.06, T_end=0.003,
                                                         lr_init=0.02,
                                                         label=f"p1_n{n}_{itype}")
                candidates.append((s_opt, a_opt, n, itype))
            except Exception as e:
                print(f"    Error in {itype} n={n}: {e}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Combined optimization on best candidates ===
    print(f"\n=== Phase 2: Combined Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()

        # Run combined optimization on the current best
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  Combined opt n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = combined_optimize(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "combined_1")

        # Try another round with perturbation
        rem = deadline - time.monotonic() - 300
        if rem > 20 and best_score > 1.5030:
            perturbed = best_seq + rng.standard_normal(n_best) * 0.02
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_alloc = min(rem * 0.4, 100.0)
            a_opt, s_opt = combined_optimize(perturbed, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "combined_perturb")

    # === Phase 3: Basin hopping ===
    print(f"\n=== Phase 3: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            n_best = len(best_seq)
            t_bh = min(rem * 0.5, 150.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")

            a_cur = best_seq.copy()
            sc_cur = 2.0 * n_best * float(fast_convolve(a_cur).max())

            n_perturbs = max(3, int(t_bh / 20))
            t_each = t_bh / n_perturbs

            for p in range(n_perturbs):
                if time.monotonic() > deadline - 200:
                    break

                # Structured perturbation: target peaks
                conv_p = fast_convolve(a_cur)
                k_star = int(np.argmax(conv_p))
                i_start = max(0, k_star - n_best + 1)
                i_end = min(n_best - 1, k_star)
                indices = np.arange(i_start, i_end + 1)

                perturbed = a_cur.copy()
                # Jitter the peak-contributing positions
                n_jitter = max(1, len(indices) // 4)
                jitter_idx = rng.choice(indices, size=min(n_jitter, len(indices)), replace=False)
                perturbed[jitter_idx] *= (1.0 + rng.standard_normal(len(jitter_idx)) * 0.08)

                # Also add global noise
                perturbed += rng.standard_normal(n_best) * 0.01
                perturbed = np.maximum(perturbed, 0)
                ps = perturbed.sum()
                if ps < 1e-10:
                    continue
                perturbed = perturbed / ps

                # Optimize the perturbed version
                t_alloc = min(t_each, deadline - time.monotonic() - 200)
                if t_alloc < 5:
                    break
                a_opt, s_opt = combined_optimize(perturbed, t_alloc)
                if s_opt < best_score:
                    best_score = s_opt
                    best_seq = a_opt.copy()
                    a_cur = a_opt.copy()
                    sc_cur = s_opt
                    print(f"    BH p={p}: improved to {best_score:.8f}")
                else:
                    # Keep current as base for next perturbation
                    pass

    # === Phase 4: Final intensive polishing ===
    print(f"\n=== Phase 4: Final Polishing (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Exact coordinate descent
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            t_cd = min(rem * 0.5, 100.0)
            print(f"  ECD polish, budget={t_cd:.0f}s")
            a_opt, s_opt = exact_coord_descent(best_seq, t_cd, verbose=True, label="polish1")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "ecd_polish1")

        # Smooth gradient one more round
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_sg = min(rem * 0.4, 50.0)
            print(f"  SGO polish, budget={t_sg:.0f}s")
            a_opt, s_opt = smooth_gradient_optimize(best_seq, t_sg,
                                                     T_start=0.02, T_end=0.002,
                                                     lr_init=0.01, label="polish2")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "sgo_polish")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 5:
            t_fine = min(rem * 0.6, 40.0)
            a_opt, s_opt = fine_coord_descent(best_seq, t_fine, label="polish3")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "fine_polish")

        # Last exact CD
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_opt, s_opt = exact_coord_descent(best_seq, rem * 0.7, label="last")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, "last_ecd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
