For the reversal: conv[2n-2-k] = sum_i a[i]*a[2n-2-k-i]. Let i' = n-1-i. Then a[i] = a[n-1-i'] and a[2n-2-k-i] = a[2n-2-k-(n-1-i')] = a[n-1-k+i']. So conv[2n-2-k] = sum_i' a[n-1-i'] * a[n-1-k+i']. If a were symmetric (a[i] = a[n-1-i]), this would equal sum_i' a[i'] * a[n-1-k+i']. Hmm, this is getting complicated.

The key point is: conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]. This is always non-negative if a is non-negative, and the support is k in [0, 2n-2].

The maximum is often near k = n-1 (the center) for "peaked" distributions, but can be elsewhere for "spread out" distributions.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    M = fast_convolve(a).max()
    return 2.0 * n * M / (s * s)

                # Current value is already optimal (or close)
                continue

            for _ in range(15):

                m2 = hi - (hi - lo) / 3.0
                f1 = f_of_delta(m1)
                f2 = f_of_delta(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            new_M = f_of_delta(best_delta)
            
            if new_M < M - 1e-14:
                # Apply the update
                a[i] = max(v + best_delta, 0.0)
                new_v = a[i]
                
                # Update convolution
                for k in range(k_start, k_end + 1):
                    diff = k - i
                    if k == two_i:
                        conv[k] = conv[k] - v * v + new_v * new_v
                    else:
                        conv[k] = conv[k] - 2.0 * v * a[diff] + 2.0 * new_v * a[diff]
                
                M = new_M
                improved = True
        
        if not improved:
            break
    
    return a, conv, M

def fast_sweep(a, time_limit, verbose=False):
    """Run coordinate descent sweeps until time runs out."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    s = a.sum()
    t0 = time.monotonic()
    best_score = 2.0 * n * M / (s * s)
    best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        a, conv, M = coord_descent_exact(a, conv, max_sweeps=5, time_limit=time_limit - (time.monotonic() - t0))
        score = 2.0 * n * M / (s * s)  # s doesn't change since we're on simplex
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose:
            print(f"    sweep done: M={M:.10f}, score={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score

def perturb_peak_targeted(a, rng, strength=0.1):
    """Perturb coordinates that contribute to the convolution peak."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Pressure: for each coordinate i, how much does it contribute to the peak?
    # conv[k] = sum_i a[i]*a[k-i], so coordinate i contributes a[i]*a[k-i] to conv[k]
    # Total pressure on i = sum_k w(k) * a[i] * a[k-i] for k near peak
    peak_thresh = 0.9 * M
    peak_ks = np.where(conv >= peak_thresh)[0]
    
    pressure = np.zeros(n)
    for k in peak_ks:
        w = conv[k] / M  # weight by how close to peak
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            pressure[i] += w * a[i] * a[k - i]
    
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure /= p_max
    
    # Multiplicative update: reduce high-pressure coords, increase low-pressure
    noise = rng.standard_normal(n) * 0.1
    a_new = a * np.exp(-strength * pressure + noise * strength)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_perturb(a, rng, strength=0.1):
    """Random multiplicative perturbation."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def init_arc(n, rng):
    """Initialize with arcsine-like shape (high at edges, low in middle)."""
    t = np.linspace(1e-6, 1 - 1e-6, n)

    a += rng.standard_normal(n) * 0.1 * a.max()

Let me be more careful. When we change a[i] from v to new_v:
- For each k in [i, i+n-1] (clipped to [0, 2n-2]):

  - If k-i == i (i.e., k = 2i), these are the same term, so the total change is new_v^2 - v^2

    Wait no: original contribution is v * a[k-i] + a[k-i] * v = 2*v*a[k-i]
