"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based game algorithm with dual warm-starts and adaptive Nesterov
2. L-BFGS LogSumExp multi-stage annealing (very low final temp)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a_fixed, b):
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None

        try:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CBC, cp.GLPK, cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    # Try to get dual values for warm start info
    dual_vals = None
    try:
        dual_vals = np.asarray(prob.constraints[0].dual.value, dtype=np.float64)
    except Exception:
        pass
    return b_val, float(M_var.value), dual_vals


def lp_game_nesterov(a0, time_limit, solver=None, n_max_iters=500000):
    """LP game with Nesterov acceleration and adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_score = best_score
    gap_lower = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step size
        alpha = 2.0 / (it + 2.0)

        # Update y (Nesterov point)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update a
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Stall detection with restart
        if (it + 1) % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 10:
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                noise = rng.random(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            # Check convergence
            current_gap = best_score - gap_lower
            if current_gap < 0.005:
                break

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPGAME iter={it+1} ({elapsed:.0f}s): score={best_score:.7f} gap={best_score-gap_lower:.7f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=10, maxiter_per_stage=2000):
    """L-BFGS on LogSumExp with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def func_z(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        val = (2.0 * n / T_val) * log_sum_exp

        # Gradient
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return val, grad_z

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def fun_wrapper(z_in):
            v, g = func_z(z_in, stage_T)
            return v

        def jac_wrapper(z_in):
            v, g = func_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=fun_wrapper,
                x0=z,
                jac=jac_wrapper,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        a_check = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_check).max()
        print(f"    LBGFS stage {si+1}/{n_stages} T={stage_T:.3f} ({elapsed:.0f}s): score={sc:.7f}")

    return softmax_from_z(z)


def coordinate_descent(a0, time_limit, n_sweeps=50):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    def score_of(a_vec):
        return 2.0 * n * fast_convolve(a_vec).max()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Randomize coordinate order
        order = rng.permutation(n)
        improved = False

        for j in order:
            if a[j] < 1e-15:
                continue

            # We'll transfer from j to a "target" coordinate
            # Simple approach: try transferring to coordinate 0 (or a low-contribution coordinate)
            # Better: try the best destination
            # For efficiency, just try a few destinations

            # Compute current convolution
            conv_cur = fast_convolve(a)
            max_cur = conv_cur.max()

            # For coordinate j, find best (destination, amount)
            # Transfer t from j to i: a[j] -= t, a[i] += t
            # conv[k] changes by: -t*a[k-j] + t*a[k-i] (approximately, for k != 2j, k != 2i)
            # For simplicity, just try a few fractions

            best_t = 0
            best_dest = -1
            best_sc = best_score

            # Try transferring to a few random destinations
            candidates = [rng.integers(n) for _ in range(8)]
            # Also try the index with lowest contribution to the peak
            peak_k = int(np.argmax(conv_cur))
            # Indices that contribute most to conv[peak_k]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contrib_idx = np.arange(js, je + 1)
            contrib = a[contrib_idx] * a[peak_k - contrib_idx]
            # High contributors to peak (sources to reduce) and low contributors (destinations)
            top_contrib = contrib_idx[np.argsort(contrib)[::-1][:3]]
            low_contrib = contrib_idx[np.argsort(contrib)[:3]]

            for dest in candidates + list(top_contrib) + list(low_contrib):
                if dest == j:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    t = a[j] * frac
                    if t < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j] -= t
                    a_trial[dest] += t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = score_of(a_trial)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_t = t
                        best_dest = dest

            if best_dest >= 0:
                a[j] -= best_t
                a[best_dest] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_sc
                best_a = a.copy()
                improved = True

        if not improved and sweep > 5:
            break

        if sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    COORDDESC sweep={sweep} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 6)

        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    else:
        x = np.linspace(0, 2 * np.pi, n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_step(a, n_active_frac=0.08, alpha=0.5):
    """One subgradient step for the max-convolution problem."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return a
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    # Projected gradient step on simplex
    a_new = a - alpha * g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def subgradient_iterations(a0, time_limit, n_active_frac=0.05):
    """Polak-Howe subgradient method with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')

    t0 = time.monotonic()

    # Initial step size
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= max(len(active), 1)
    g_norm_sq = np.dot(g, g)
    if g_norm_sq < 1e-30:
        return a, best_score
    step = min(1.0 / np.sqrt(g_norm_sq), 2.0)

    no_improve = 0
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        max_b = b.max()
        threshold = max_b - n_active_frac * max_b

        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_alt = evaluate_safe(init_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = init_alt.copy()
            print(f"  Interpolated n={alt_n}: {sc_alt:.7f}")

        t_lp = t_total * 0.5
        a_alt, _ = lp_game_nesterov(best_seq, t_lp, solver=solver, n_max_iters=200000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.4, 80.0)
            if t_lb > 15:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=50.0, T_end=0.05,
                                      n_stages=5, maxiter_per_stage=1500)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.01,
                             n_stages=6, maxiter_per_stage=2000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.3, 50.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd, n_sweeps=50)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
