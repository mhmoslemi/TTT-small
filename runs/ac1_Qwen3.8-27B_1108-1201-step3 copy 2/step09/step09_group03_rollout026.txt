def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""

1. Smoothed gradient descent (log-sum-exp) with projected gradient steps
2. Exact coordinate descent with mass transfer

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def compute_gradient_smoothed(a, conv, tau, top_k=100):
    """
    Compute the gradient of log(sum exp(tau * c[k])) w.r.t. a.
    Uses top-k peaks for efficiency.
    Returns gradient of length n.
    """
    n = len(a)
    M = conv.max()
    shifted = conv - M
    exp_vals = np.exp(tau * shifted)
    total = exp_vals.sum()
    w = exp_vals / total  # softmax weights
    
    # Use top-k for efficiency
    if top_k < 2 * n - 1:
        peak_idx = np.argpartition(w, -top_k)[-top_k:]
        w_peaks = w[peak_idx]
    else:
        peak_idx = np.arange(2 * n - 1)
        w_peaks = w
    
    grad = np.zeros(n)
    for idx in range(len(peak_idx)):
        wi = w_peaks[idx]
        if wi < 1e-15:
            continue
        k = int(peak_idx[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # grad[i] += wi * 2 * a[k-i] for i in [i_start, i_end]
        # a[k-i] for i in [i_start, i_end]:
        # when i = i_start: a[k - i_start]
        # when i = i_end: a[k - i_end]
        # So the slice is a[k - i_end : k - i_start + 1][::-1]
        grad[i_start:i_end+1] += wi * 2.0 * a[k - i_end:k - i_start + 1][::-1]
    
    return grad

def smoothed_gradient_descent(a0, time_limit, tau_init=50.0, lr_init=0.005, 
                                top_k=80, momentum=0.5, verbose=True):
    """
    Smoothed gradient descent with log-sum-exp smoothing.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    tau = tau_init
    no_improve = 0
    v_mom = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        grad = compute_gradient_smoothed(a, conv, tau, top_k)
        
        # Momentum step
        v_mom = momentum * v_mom - lr * grad
        a_new = simplex_project(a + v_mom)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, try smaller step
            if no_improve > 50:
                a_new = simplex_project(a - 0.5 * lr * grad)
                conv_new = fast_convolve(a_new)
                score_new = 2.0 * n * conv_new.max()
                if score_new < best_score - 1e-12:
                    best_score = score_new
                    best_a = a_new.copy()
                no_improve = 0
        
        a = a_new
        conv = conv_new
        
        # Adaptive parameters
        if it % 200 == 0:
            tau *= 1.3  # Increase smoothness (sharper max)
            lr *= 0.95  # Slow down
        
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-12:
                break
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): score={best_score:.6f} tau={tau:.1f} lr={lr:.2e}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with fine-grained search for each coordinate.
    Tries to reduce each coordinate and redistribute mass.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    
    # Precompute for fast "unaffected max" queries
    # When we change a[i], only conv[k] for k in [i, i+n-1] changes
    # (specifically, k where 0 <= k-i < n, i.e., k in [i, i+n-1])
    # Plus k = 2i (quadratic term)
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        
        for i in range(n):
            v = a[i]
            if v < 1e-14:
                continue
            
            k_start = max(i, 0)
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max (positions not in [k_start, k_end])
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = 0.0
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end+1:].max()
            else:
                unaff_right = 0.0
            unaff_max = max(unaff_left, unaff_right)
            
            # Try several candidate values
            best_t = v
            best_new_max = 2.0 * n * max(unaff_max, conv[k_start:k_end+1].max())

            # Add a few more around the current value
            for factor in [0.3, 0.6, 1.3, 2.0]:
                candidates.append(v * factor)
            
            for t in candidates:
                if t < 0:
                    continue
                
                # Compute new affected convolution values
                # c[k](t) = conv[k] - 2*v*a[k-i] + 2*t*a[k-i] for k != 2i
                # c[2i](t) = conv[2i] - v^2 + t^2
                delta = t - v
                if abs(delta) < 1e-15:
                    continue
                
                # For efficiency, compute the affected range only
                affected = conv[k_start:k_end+1].copy()
                
                # Linear terms: for k in [k_start, k_end], k != 2i
                # a[k-i] for k in [k_start, k_end]:
                # k - i ranges from k_start-i to k_end-i
                ki_start = k_start - i  # = max(0, ...) = 0 (since k_start = i)
                ki_end = k_end - i      # = min(n-1, n-1) = n-1
                # So a[k-i] = a[0:n] = a (the whole vector!)
                # Wait: k_start = i, so k - i starts at 0
                # k_end = min(i+n-1, 2n-2), so k-i ends at min(n-1, n-2)
                # Actually k_end - i = min(i+n-1, 2n-2) - i = min(n-1, n-2)
                # For i < n-1: k_end - i = n-2... hmm let me just compute directly
                
                # For k in [i, i+n-1] (or [i, 2n-2]):
                # k - i ranges from 0 to min(n-1, 2n-2-i)
                max_ki = min(n - 1, 2*n - 2 - i)
                a_shifted = a[:max_ki+1]  # a[0], a[1], ..., a[max_ki]
                
                # Apply linear change to affected[k] for k where k-i is valid
                # affected[j] corresponds to conv[k_start + j] = conv[i + j]
                # k = i + j, so k - i = j, valid for j in [0, max_ki]
                if max_ki + 1 <= len(affected):
                    affected[:max_ki+1] += 2.0 * delta * a_shifted[:max_ki+1]
                
                # Fix quadratic term at k = 2i (if in range)
                if 2*i >= k_start and 2*i <= k_end:
                    idx = 2*i - k_start
                    # The linear change was applied: affected[idx] += 2*delta*a[i]
                    # But the correct change is: -v^2 + t^2 = -(v-t)(v+t) = -delta*(2v - delta)
                    # The linear approximation gave: 2*delta*a[i] = 2*delta*v
                    # The exact change: t^2 - v^2 = (t-v)(t+v) = delta*(2v-delta)
                    # Difference: delta*(2v-delta) - 2*delta*v = -delta^2
                    # So we need to subtract delta^2 from affected[idx]
                    affected[idx] -= delta * delta
                
                new_max = max(unaff_max, float(affected.max()))
                new_score = 2.0 * n * new_max
                
                if new_score < best_new_max:
                    best_new_max = new_score
                    best_t = t
            
            if best_t != v and best_new_max < 2.0 * n * max(unaff_max, conv[k_start:k_end+1].max()) - 1e-13:
                # Apply the update
                delta = best_t - v
                a[i] = max(best_t, 0.0)
                
                # Update convolution
                max_ki = min(n - 1, 2*n - 2 - i)
                a_shifted = a[:max_ki+1]
                # Need to save old a[i] for the quadratic fix
                old_ai = v
                
                if max_ki + 1 <= 2*n-1:
                    conv[i:i+max_ki+1] += 2.0 * delta * a_shifted[:max_ki+1]
                
                if 2*i < 2*n-1:
                    conv[2*i] -= delta * delta
                
                # Renormalize (since we changed the sum)
                s = a.sum()
                if s < 1e-15:
                    a = np.ones(n) / n
                    conv = fast_convolve(a)
                else:
                    # We need sum(a) = 1, but changing a[i] changes the sum
                    # For the score computation, normalization doesn't matter (scale invariant)
                    # So let's just track the score without re-normalizing
                    pass

def mass_transfer_cd(a0, time_limit, verbose=True, label=""):

    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    # Precompute: for each pair (i,j), the affected range of conv
    # When we change a[i] by +delta and a[j] by -delta:
    # conv[k] changes by:
    #   delta * (a[k-i] + a[k-i]) - delta * (a[k-j] + a[k-j])
    #   + special quadratic terms at k=2i and k=2j
    # = 2*delta*(a[k-i] - a[k-j]) + quadratic corrections
    
    # This is complex. Let's use a simpler approach:
    # Pick a random pair (i,j), try transferring mass from i to j (or vice versa)
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:

        i = rng.integers(0, n)
        j = rng.integers(0, n)

            continue
        
        v_i = a[i]
        v_j = a[j]
        
        if v_i < 1e-14:
            continue
        
        # Try moving a fraction of a[i] to a[j]

            a_try = a.copy()

            a_try[j] += delta
            if a_try[i] < 0:

            s_try = a_try.sum()
            # Scale invariant, so just compute max(conv) / sum^2
