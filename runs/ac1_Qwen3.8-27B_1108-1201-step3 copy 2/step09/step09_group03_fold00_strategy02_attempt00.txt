def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            for i in index_order:
                # Only if the current coefficient is not too small
                if curr_seq[i] < 1e-3:
                    continue  # Skip this index
                    
                # Compute the effect of increasing or decreasing this coordinate
                # This is an approximation: for each change, how to move mass to other coordinates to minimize the score.
                # But we might use a simple JSON storage for the best candidate.
                
                # We'll use a perturbation: first try increasing a little, then decreasing.
                # The best global might be updated with the current sequence if it improves.
                
                # Check against current_best
                base = curr_seq[i]
                factors = [0.99, 0.98, 0.97, 0.96, 1.01, 1.02, 1.03, 1.05]
                for factor in factors:
                    new_val = base * factor
                    if new_val > 1000:
                        continue  # Skip, because our candidate should not have coefficients above 1000
                    
                    new_seq = curr_seq.copy()
                    new_seq[i] = new_val
                    new_total = new_seq.sum()
                    if new_total < 0.01 or new_total > 1e10:
                        continue
                    # Normalize the new_seq but keep non-negative and <1000
                    norm_factor = (0.5 * n_val) / new_total  # arbitrary choice
                    new_seq = np.clip(new_seq * norm_factor, 0, 1000)
                    # Actually, we should normalize to 0.5*n_val or 1? No, the previous programs did not do this arbitrary scaling.
                    # Let's re-normalize to the simplex by setting new_seq = new_seq / new_seq.sum()?
                    # But the problem does not require the sum to be normalized, only that we normalize and then compute the score.
                    # However, the evaluator's adjustment might affect local optima detection.
                    
                    # We are allowed to return sequences of any sum, as long as they are non-negative and <1000.
                    # But the score is defined using the sum and max convolution.
                    # Let's do: evaluate_without_normalizing(new_seq) but the evaluator does normalize.
                    new_score = compute_score(new_seq)
                    if np.isfinite(new_score) and new_score < score_val:
                        # Only minor improvement, but we record
                        curr_seq = new_seq.copy()
                        score_val = new_score
                        break
                        
            # End for improvement in i
            
        # We run out of time
        
        return best_global
        
        # We are to return the refined sequence.
    except:
        return [0.5] * candidate_length if candidate_length else [0.5] * n_bucket1k
