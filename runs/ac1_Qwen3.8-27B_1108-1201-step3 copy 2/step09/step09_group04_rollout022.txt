"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-phase optimization:
- FFT-accelerated p-norm L-BFGS with aggressive p schedule
- Nesterov LP alternating with dual gap monitoring
- Subgradient mirror descent (exponentiated gradient)
- Multi-length exploration
- Coefficient-level local refinement
Key improvements over previous: higher p schedule, better time allocation,
dual gap monitoring, and coordinated length + optimization search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score assuming a is normalized (sum=1)."""
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max())


def eval_safe(a):
    """Safe evaluation using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def pnorm_val_grad_z(z, p):
    """
    Value and gradient of 2n * ||convolve(a,a)||_p where a = softmax(z).
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    S = np.sum(c_pos ** p)
    val = 2.0 * n * S ** (1.0 / p)
    if S < 1e-300:
        return val, np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # g_a[i] = 4n * factor * correlate(w, a, 'valid')[i]
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    g_z = a * (g_a - np.dot(g_a, a))
    return val, g_z


def pnorm_lbfgs(a0, time_budget, p_stages):
    """
    L-BFGS with p-norm smoothing.
    p_stages: list of (p, max_iter) pairs.
    """
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    
    for p, max_it in p_stages:
        if time.monotonic() - t0 > time_budget:
            break
        rem = time_budget - (time.monotonic() - t0)
        if rem < 5:
            break
        
        def _f(zv):
            v, _ = pnorm_val_grad_z(zv, p)
            return v
        
        def _g(zv):
            _, g = pnorm_val_grad_z(zv, p)
            return g
        
        try:
            res = minimize(_f, z, jac=_g, method='L-BFGS-B',
                          options={'maxiter': max_it, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': max_it * 200})
            z = res.x
        except Exception:
            pass
        
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        sc = compute_score(a)
        print(f"      p={p:.0f}: score={sc:.6f} t={time.monotonic()-t0:.0f}s")
    
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def build_T(a):
    """Build convolution matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_best_response(a_fixed, T, solver):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    cons = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val / s, float(M.value)


def nesterov_lp(a0, time_budget, solver, max_iters=100000):
    """Nesterov LP alternating minimization with dual gap tracking."""
    n = len(a0)
    T = build_T(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    best_dual = 0.0
    t0 = time.monotonic()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_budget:
            break
        
        b_new, M_val = solve_best_response(y, T, solver)
        if b_new is None:
            continue
        
        best_dual = max(best_dual, M_val)
        alpha = 2.0 / (it + 1)
        
        y = (1 - alpha) * y + alpha * b_new
        y = normalize(y)
        
        a = ((it - 1) * a + it * y) / it if it > 1 else y.copy()
        a = normalize(a)
        
        sc_a = compute_score(a)
        sc_y = compute_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if it % 2000 == 0:
            gap = best_score - 2*n*best_dual  # primal - dual (both with sum=1)
            print(f"      NL it={it}: primal={best_score:.6f} dual_bound={2*n*best_dual:.6f} gap={gap:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def local_refine(a0, time_budget, max_steps=50000):
    """Coordinate-wise local refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_budget:
            break
        
        c = fast_convolve(a)
        # Exclude boundary for peak selection
        c_inner = c.copy()
        if len(c_inner) > 2:
            c_inner[0] = -1
            c_inner[-1] = -1
        k_peak = c_inner.argmax()
        peak_val = c[k_peak]
        
        # Find top contributors to this peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[k_peak - idx]
        top = idx[np.argsort(contribs)[::-1][:10]]
        
        improved = False
        for i in top:
            if a[i] < 1e-12:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                reduce_amt = a[i] * (1 - factor)
                a_trial[i] -= reduce_amt
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    reduce_amt = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-12:
                    break
                a_trial[mask] += reduce_amt * a_trial[mask] / others_sum
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair reduction
            for ci in range(min(6, len(top))):
                i = top[ci]
                j = k_peak - i
                if i == j or j < 0 or j >= n or a[j] < 1e-12:
                    continue
                for f in [0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = normalize(a_trial)
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a, best_score
