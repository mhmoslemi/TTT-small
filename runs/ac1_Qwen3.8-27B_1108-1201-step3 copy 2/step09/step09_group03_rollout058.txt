import numpy as np

from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_structure(n):
    """Precompute sparse matrix structure for convolution constraint."""
    j_idx = np.arange(n)
    k_idx = np.arange(n)
    rows = (j_idx[:, None] + k_idx[None, :]).ravel()
    cols = np.repeat(j_idx, n)
    return rows, cols


def solve_lp_best_response(a, rows, cols, solver=None):
    """Solve LP: min t s.t. (a*b)_k <= t, b>=0, sum(b)=1.
    Returns (b, t_value, status)."""
    n = len(a)
    vals = np.tile(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [A @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    if solver is not None:
        prob.solve(solver=solver)
    else:
        prob.solve()

    status = prob.status
    if status not in ('optimal', 'optimal_inaccurate'):
        return None, np.inf, status

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, np.inf, status
    b_val = b_val / s
    t_val = float(t_var.value)
    return b_val, t_val, status

def lp_alternating_minimize(a_init, time_limit, solver=None, verbose=True, label=""):
    """LP-based alternating best-response minimization with averaging."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    rows, cols = build_conv_matrix_structure(n)

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0

    # Running average (for convergence guarantee)
    avg_a = a.copy().astype(np.float64)

    # Duality gap tracking
    prev_gap = np.inf

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: best response b to current a
        b, lp_val, status = solve_lp_best_response(a, rows, cols, solver=solver)
        if b is None:
            break

        # Current score
        cur_score = score_normalized(a)
        gap = cur_score - 2.0 * n * lp_val  # f(a) - v(a) scaled

        # Update running average
        avg_a = (it * avg_a + b) / (it + 1)

        # Check running average
        avg_score = score_normalized(avg_a)
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_a.copy()

        # Damped update towards b
        alpha = 2.0 / (it + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        new_score = score_normalized(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-ALT {label} it={it} ({elapsed:.0f}s): "
                  f"best={best_score:.6f} cur={new_score:.6f} gap={gap:.6f}")

        # Convergence check
        if it > 100 and gap < 1e-8:
            break

    return best_a, best_score

                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < best_new_M - threshold:
                        best_new_M = new_M
                        best_delta = delta
            else:
                base_step = max(a[i], 0.01 / n)
                for factor in factors:
                    delta = base_step * factor
                    new_val = a[i] + delta
                    if new_val < -1e-15:
                        continue
                    new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < best_new_M - threshold:
                        best_new_M = new_M
                        best_delta = delta

            if best_delta != 0.0 and best_new_M < M - threshold:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)

                # Update convolution incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > (30 if fine else 10):
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    inits = []

    if parent is not None:
        inits.append(parent.copy())
        # Noisy version
        noisy = parent + rng.standard_normal(n) * 0.02
        noisy = np.maximum(noisy, 0)
        noisy = noisy / noisy.sum()
        inits.append(noisy)
        # Reversed
        rev = parent[::-1].copy()
        inits.append(rev)
        # Fourier-scrambled: apply random phase in frequency domain
        A = rfft(parent)
        phases = rng.uniform(0, 2 * np.pi, len(A))
        A_scrambled = A * np.exp(1j * phases)
        scrambled = irfft(A_scrambled, n)
        scrambled = np.maximum(scrambled, 0)
        if scrambled.sum() > 1e-10:
            scrambled = scrambled / scrambled.sum()
            inits.append(scrambled)

    # Cosine window
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    for power in [0.5, 1.0, 2.0]:
        c = (0.5 * (1 + np.cos(np.pi * x))) ** power
        c = np.maximum(c, 0)
        if c.sum() > 1e-10:
            inits.append(c / c.sum())

    # Periodic patterns
    for period in [3, 4, 5, 7]:
        p = np.zeros(n)
        for i in range(n):
            p[i] = 0.5 + 0.4 * np.cos(2 * np.pi * (i % period) / period)
        p = np.maximum(p, 0)
        if p.sum() > 1e-10:
            inits.append(p / p.sum())

    # Two-hump
    c1 = n * 0.25
    c2 = n * 0.75
    w = n * 0.15
    h = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    h = np.maximum(h, 0)
    inits.append(h / h.sum())

    # U-shape (high at ends, low in middle)
    u = 0.3 + 0.7 * (2 * x) ** 2
    u = np.maximum(u, 0)
    inits.append(u / u.sum())

    # Inverted-U (high in middle)
    iu = 1.0 - 0.5 * (2 * x) ** 2
    iu = np.maximum(iu, 0)
    if iu.sum() > 1e-10:
        inits.append(iu / iu.sum())

    # Random
    r = np.abs(rng.standard_normal(n))
    r = r / r.sum()
    inits.append(r)

    # Sparse-like (most zeros, few large values)
    sp = np.zeros(n)
    idx = rng.choice(n, size=max(4, n // 16), replace=False)
    sp[idx] = 1.0
    sp = sp / sp.sum()
    inits.append(sp)

    return inits


def propose_candidate():
    """
    Multi-phase optimization:
    1. LP-based alternating best-response (primary optimizer)
    2. Coordinate descent polishing
    3. Fine coordinate descent
    4. Multi-size exploration
    """
    global _lp_solver_cache
    _lp_solver_cache = None

    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        n_main = len(best_seq)
    else:
        n_main = 1024
        best_seq = np.ones(n_main) / n_main
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent found. Using uniform n={n_main}, score={best_score:.6f}")

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")

    # === Phase 1: LP-based alternating minimization ===
    print(f"\n=== Phase 1: LP Alternating Minimization (best={best_score:.6f}) ===")
    t1_start = time.monotonic()

    solver = find_lp_solver()
    n = n_main
    rows, cols = build_conv_matrix_structure(n)

    # Generate initializations
    inits = generate_inits(n, parent, rng)

    best_lp = best_seq.copy()
    best_lp_score = best_score
    n_inits = len(inits)
    t_per_init = min(250.0, 1200.0 / max(n_inits, 1))

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic() - 500
        if remaining < 30:
            break
        t_alloc = min(t_per_init, remaining * 0.15 if idx > 0 else remaining * 0.25)
        if idx == 0:
            t_alloc = min(300.0, t_alloc + 50)  # Extra time for best init

        print(f"  LP-ALT init {idx}/{n_inits}, n={n}, budget={t_alloc:.0f}s")
        a_lp, s_lp = lp_alternating_minimize(
            init, t_alloc, solver=solver, verbose=True, label=f"init{idx}"
        )
        if s_lp < best_lp_score:
            best_lp_score = s_lp
            best_lp = a_lp.copy()
        update_best(a_lp, f"lp_{idx}")
        update_best(best_lp, f"lp_best_{idx}")

    # === Phase 2: Continued LP optimization on best found ===
    print(f"\n=== Phase 2: Extended LP on best (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 400
    if remaining > 60:
        t_ext = min(remaining * 0.5, 300.0)
        print(f"  Extended LP budget={t_ext:.0f}s")
        a_ext, s_ext = lp_alternating_minimize(
            best_seq, t_ext, solver=solver, verbose=True, label="ext"
        )
        update_best(a_ext, "lp_ext")

    # === Phase 3: Coordinate Descent Polishing ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 300
    if remaining > 60:
        t_cd = min(remaining * 0.5, 250.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_cd, s_cd = coord_descent_optimize(
            best_seq, t_cd, verbose=True, label="main", fine=False
        )
        update_best(a_cd, "cd_main")

        # Also polish the LP result
        remaining = deadline - time.monotonic() - 200
        if remaining > 60 and best_lp is not None:
            t_cd2 = min(remaining * 0.4, 150.0)
            a_cd2, s_cd2 = coord_descent_optimize(
                best_lp, t_cd2, verbose=True, label="lp_cd", fine=False
            )
            update_best(a_cd2, "cd_lp")

    # === Phase 4: Fine Polishing ===
    print(f"\n=== Phase 4: Fine Polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 30:
        t_fine = min(remaining * 0.5, 150.0)
        print(f"  Fine CD budget={t_fine:.0f}s")
        a_fine, s_fine = coord_descent_optimize(
            best_seq, t_fine, verbose=True, label="fine", fine=True
        )
        update_best(a_fine, "fine_cd")

    # === Phase 5: Try different n values ===
    print(f"\n=== Phase 5: Multi-Size Exploration (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 150
    if remaining > 100:
        # Try a few different sizes
        candidate_n_values = []
        for nc in [512, 768, 1536, 2048, 4096]:
            if nc != n_main:
                candidate_n_values.append(nc)

        # Sort by proximity to n_main (likely to be good)
        candidate_n_values.sort(key=lambda x: abs(x - n_main))

        t_per_n = min(80.0, remaining / max(len(candidate_n_values), 1) * 0.8)

        for nc in candidate_n_values[:4]:
            remaining = deadline - time.monotonic() - 100
            if remaining < t_per_n:
                break

            print(f"  Trying n={nc} (budget={t_per_n:.0f}s)")

            # Create initialization by interpolation or resampling of best
            if best_seq is not None:
                from scipy.interpolate import interp1d
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, nc)
                f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
                a_nc = np.maximum(f_interp(x_new), 0)
                a_nc = a_nc / a_nc.sum()
            else:
                a_nc = np.ones(nc) / nc

            # Brief LP optimization
            rows_nc, cols_nc = build_conv_matrix_structure(nc)
            a_res, s_res = lp_alternating_minimize(
                a_nc, t_per_n, solver=solver, verbose=False, label=f"n{nc}"
            )
            update_best(a_res, f"n{nc}_lp")

            # Quick CD polish
            remaining = deadline - time.monotonic() - 80
            if remaining > 20:
                a_cd_n, s_cd_n = coord_descent_optimize(
                    a_res, min(remaining * 0.5, 40.0), verbose=False,
                    label=f"n{nc}_cd", fine=True
                )
                update_best(a_cd_n, f"n{nc}_cd")

    # === Phase 6: Final Fine Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        t_final = min(remaining * 0.8, 60.0)
        a_fin, s_fin = coord_descent_optimize(
            best_seq, t_final, verbose=False, label="final", fine=True
        )
        update_best(a_fin, "final")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    # Ensure valid
    best_seq = np.maximum(best_seq, 0.0)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = np.ones(1024) / 1024
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
