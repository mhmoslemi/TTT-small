```python
def mirror_descent_step(a, eta, tolerance=0.0):
    """One mirror descent step."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Active set
    threshold = c_max - tolerance * c_max if tolerance > 0 else c_max - 1e-10
    S = np.where(c >= threshold)[0]
    
    # Compute subgradient via FFT cross-correlation
    S_ind = np.zeros(2 * n - 1)
    S_ind[S] = 1.0
    
    N = 1 << (3 * n - 2 - 1).bit_length()
    F_S = rfft(S_ind, N)
    F_a = rfft(a, N)
    s_full = irfft(np.conj(F_S) * F_a, N)
    s = s_full[:n] * (2.0 / len(S))
    
    # Mirror descent update: a[i] *= exp(-eta * s[i])
    a_new = a * np.exp(-eta * s)
    a_new = np.maximum(a_new, 0)
    total = a_new.sum()
    if total < 1e-300:
        return a, c_max
    a_new /= total
    return a_new, c_max
```

Actually, the condition for a Hankel matrix to be PSD is that the sequence is positive definite: sum_{i,j} c_i c_j p_{i+j} >= 0 for all c. This is equivalent to saying that the sequence p_k is the moment sequence of a positive measure on [0, ∞).

For a finite sequence p_0, ..., p_{2n-2}, the Hankel matrix of size n x n is:
H[i,j] = p_{i+j} for i,j = 0,...,n-1

This is PSD iff there exists a positive measure μ on [0,∞) such that p_k = ∫ x^k dμ(x) for k = 0,...,2n-2.

Since p is a probability vector (p_k >= 0, sum p_k = 1), we can always find such a μ: take μ = sum_{k=0}^{2n-2} p_k δ_{ε^k} for small enough ε > 0. But this is a bit artificial.

Actually, a simpler argument: the Hankel matrix H with H[i,j] = p_{i+j} where p >= 0 is NOT necessarily PSD. Consider p = [0, 1, 0]: H = [[0, 1, 0], [1, 0, 1], [0, 1, 0]] which has eigenvalues √2, 0, -√2. Not PSD!

So in general, A_p is NOT PSD, and the problem is non-convex. This confirms our earlier observation.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive mirror descent with
FFT-accelerated subgradients, LP-based Nesterov alternating refinement, p-norm
L-BFGS smoothing, local peak shaving, and multi-length exploration.
Key innovations: entropy-based mirror descent for fast O(n log n) iterations,
adaptive active-set tolerance, improved Nesterov LP with extrapolation,
and aggressive multi-scale length search.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def compute_subgradient(a, rel_tol=0.0):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using FFT cross-correlation.
    Returns (grad, c_max, c) where grad has same shape as a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    if rel_tol > 0:
        threshold = c_max * (1.0 - rel_tol)
    else:
        threshold = c_max - 1e-12 * max(1.0, c_max)
    S = np.where(c >= threshold)[0]
    if len(S) == 0:
        S = np.array([np.argmax(c)])
    S_ind = np.zeros(2 * n - 1)
    S_ind[S] = 1.0
    # Cross-correlation: s[i] = sum_k S_ind[k] * a[k-i]
    N = 1 << (3 * n - 2 - 1).bit_length()
    F_S = rfft(S_ind, N)
    F_a = rfft(a, N)
    s_full = irfft(np.conj(F_S) * F_a, N)
    grad = s_full[:n].real * (2.0 / len(S))
    return grad, c_max, c


def mirror_descent_optimize(a0, time_limit, n_stages=6, eta_start=0.5, eta_end=0.01):
    """Multi-stage mirror descent with adaptive step size and active set tightening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Stages with decreasing eta and tightening tolerance
    etas = np.geomspace(eta_start, eta_end, n_stages)
    tols = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001][:n_stages]
    # Pad tols if needed
    while len(tols) < n_stages:
        tols.append(0.001)

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        eta = etas[stage]
        tol = tols[stage]
        stage_iter = 0
        while time.monotonic() - t0 < time_limit:
            stage_iter += 1
            if stage_iter > 2000:
                break
            grad, c_max, c = compute_subgradient(a, rel_tol=tol)
            # Mirror descent: a[i] *= exp(-eta * grad[i])
            scale = np.exp(-eta * grad)
            a_new = a * scale
            a_new = np.maximum(a_new, 1e-300)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            sc = 2.0 * n * c_max  # use c_max from before update as proxy
            # Actual score
            c_new = fast_convolve(a_new)
            sc_new = 2.0 * n * c_new.max()
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            a = a_new
            if stage_iter % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"        MD stage {stage} iter {stage_iter} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def p_norm_objective(a, p):
    """Compute p-norm surrogate: 2n*(sum c_k^p)^(1/p) for c = conv(a,a)."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * n * val


def p_norm_grad_log(a, p):
    """Gradient of 2n*(sum c^p)^(1/p) w.r.t. log(a) parameterization."""
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_pow = c_pos ** p
    S = np.sum(p_pow)
    if S < 1e-300:
        return np.zeros_like(a)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    # Gradient w.r.t. a: 2n * factor * correlate(w, a, 'valid')
    corr = np.correlate(w, a, mode='valid')
    g_a = 2.0 * n * factor * corr
    # Gradient w.r.t. log(a): a * (g_a - dot(g_a, a)/sum(a)) ... 
    # Actually for the normalization-constrained problem, we use:
    # grad_log = a * g_a - (sum(a * g_a) / sum(a)) * a  (projected gradient)
    # But since we normalize after each step, just return g_a * a
    return a * g_a


def p_norm_lbfgs(a0, time_limit, p_start=30.0, p_end=500.0, n_stages=6):
    """L-BFGS in log-domain with p-norm smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(a)
    z = z - z.mean()  # center for stability
    t0 = time.monotonic()
    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per = max(50, 3000 // n_stages)

    for st_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            a_cur = np.exp(z_val - z_val.max())
            a_cur = a_cur / a_cur.sum()
            return p_norm_objective(a_cur, st_p)

        def jac(z_val):
            a_cur = np.exp(z_val - z_val.max())
            a_cur = a_cur / a_cur.sum()
            c = fast_convolve(a_cur)
            c_pos = np.maximum(c, 1e-300)
            p_pow = c_pos ** st_p
            S = np.sum(p_pow)
            if S < 1e-300:
                return np.zeros_like(a_cur)
            factor = S ** (1.0 / st_p - 1.0)
            w = c_pos ** (st_p - 1.0)
            corr = np.correlate(w, a_cur, mode='valid')
            g_a = 2.0 * len(a_cur) * factor * corr
            # Gradient w.r.t. z: for a = softmax(z), da/dz = a*(e_i - a_j)
            # grad_z = a * (g_a - a @ g_a / sum(a)) ... 
            # Actually: d/dz_i [f(softmax(z))] = a_i * (g_a_i - sum_j a_j * g_a_j / sum(a))
            # Since sum(a) = 1 after normalization:
            return a_cur * (g_a - np.dot(a_cur, g_a))

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = score_normalized(a_cur)
        elapsed = time.monotonic() - t0
        print(f"        p-norm LBFGS p={st_p:.0f} ({elapsed:.0f}s): score={sc:.6f}")

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final

def solve_lp(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix: T @ b where T[k, j] = a_fixed[k-j] if valid
    # Use sparse formulation
    import scipy.sparse as sp
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[k - j])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CBC, cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp(y, solver)
        if b_new is None:
            continue
        # Nesterov-style update
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Averaged iterate
        t_a = it + 1
        t_b = alpha * (it + 2)
        a = (t_a * a + t_b * b_new) / (t_a + t_b)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = score_normalized(a)
        sc_y = score_normalized(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 10:
                # Perturb to escape
                rng = np.random.default_rng(it + 123)
                noise = rng.standard_normal(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0
                prev_best = best_score

            elapsed = time.monotonic() - t0
            print(f"        NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def local_refinement(a0, time_limit, n_steps=20000):
    """Local refinement: peak shaving via coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_convolve(a)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        peak_idx = np.argsort(c_work)[::-1][:3]
        peak_val = c_work[peak_idx[0]]
        threshold = peak_val - 0.03 * peak_val
        near_peaks = [k for k in range(len(c_work)) if c_work[k] >= threshold][:6]

        improved = False
        # Compute contribution scores
        contrib = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            contrib[idx] += contribs

        # Try reducing top contributors
        top_coords = np.argsort(contrib)[::-1][:10]
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = max(0, a_trial[i] - diff)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask].sum()
                if others < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and step % 50 == 0:
            # Random perturbation
            rng = np.random.default_rng(step + 42)
            n_pert = rng.integers(2, 6)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.08, 0.08))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial /= st
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved and step > 0 and step % 500 == 0:
            # Check if we've stalled too long
            pass

        if not improved and step > 0:
            # Try harder: check consecutive non-improvement
            break

    return best_a, best_score


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        # U-shaped (bimodal at edges)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x2 = np.linspace(-1, 1, n)
        a = np.exp(-x2 ** 2 * 2)
    elif variant == 5:
        # Power law
        a = np.power(np.clip(x, 1e-6, 1), 0.7) + np.power(np.clip(1 - x, 1e-6, 1), 0.7)
    elif variant == 6:
        # Flat with slight dip
        a = 1.0 - 0.3 * np.sin(np.pi * x) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization combining mirror descent, Nesterov LP,
    p-norm L-BFGS, local refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_state = {'score': float('inf'), 'seq': None}
    solver = get_lp_solver()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    print(f"\nStarting optimization from n={n_best}, score={best_state['score']:.6f}")
    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Multi-start initialization (30s) ===
    print(f"\n=== Phase 1: Multi-start Init ===")
    t_p1 = 30
    rem = deadline - time.monotonic()
    t_alloc = min(t_p1, rem)
    if t_alloc > 5:
        per_init = t_alloc / 6
        for v in range(6):
            if time.monotonic() > deadline:
                break
            a_init = structured_init(n_best, variant=v)
            # Quick mirror descent
            a_opt = mirror_descent_optimize(a_init, per_init, n_stages=3, eta_start=0.3, eta_end=0.05)
            try_update(best_state, a_opt, f"init_v{v}")

    # === Phase 2: Mirror Descent (250s) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(250.0, rem)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md = mirror_descent_optimize(best_state['seq'], t_alloc, n_stages=8, eta_start=0.4, eta_end=0.005)
        try_update(best_state, a_md, "mirror_descent")

    # === Phase 3: Nesterov LP (400s) ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(400.0, rem)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        try_update(best_state, a_lp, "nesterov_lp_1")

    # === Phase 4: p-norm L-BFGS (250s) ===
    print(f"\n=== Phase 4: p-norm L-BFGS (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(250.0, rem)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pn = p_norm_lbfgs(best_state['seq'], t_alloc, p_start=20.0, p_end=400.0, n_stages=7)
        try_update(best_state, a_pn, "p_norm_lbfgs")

    # === Phase 5: Local Refinement (150s) ===
    print(f"\n=== Phase 5: Local Refinement (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(150.0, rem)
    if t_alloc > 10:
        a_lr, _ = local_refinement(best_state['seq'], t_alloc, n_steps=30000)
        try_update(best_state, a_lr, "local_refine")

    # === Phase 6: Nesterov LP round 2 (200s) ===
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(200.0, rem)
    if t_alloc > 20:
        a_lp2 = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=60000)
        try_update(best_state, a_lp2, "nesterov_lp_2")

    # === Phase 7: Length Exploration (250s) ===
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        candidate_lengths = set()
        base = len(best_state['seq'])
        for rn in [640, 768, 832, 896, 960, 1024, 1088, 1152, 1216, 1280, 1408, 1536]:
            if 400 <= rn <= 2048 and abs(rn - base) > 48:
                candidate_lengths.add(rn)
        # Also try nearby
        for d in [-64, -32, 32, 64]:
            alt = base + d
            if 400 <= alt <= 2048:
                candidate_lengths.add(alt)
        candidate_lengths = sorted(candidate_lengths, key=lambda x: abs(x - base))

        n_explored = 0
        for alt_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60 or n_explored >= 4:
                break
            n_explored += 1
            t_total = min(rem * 0.25, 150.0)
            if t_total < 30:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
            init_alt = interpolate_sequence(best_state['seq'], alt_n)
            # Quick mirror descent
            t_md = t_total * 0.4
            a_alt = mirror_descent_optimize(init_alt, t_md, n_stages=4, eta_start=0.3, eta_end=0.02)
            improved = try_update(best_state, a_alt, f"expl_n{alt_n}")
            # If improved, do LP
            if improved or t_total > 80:
                rem = deadline - time.monotonic()
                if rem > 20:
                    t_lp = min(rem * 0.25, 80.0)
                    a_alt2 = nesterov_lp_refine(a_alt if not improved else best_state['seq'], t_lp, solver=solver, n_max_iters=30000)
                    try_update(best_state, a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Perturbation + re-optimize (100s) ===
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_state['seq'])
        for noise_scale in [0.03, 0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed /= perturbed.sum()
            t_sub = min(rem * 0.3, 50.0)
            a_pert = mirror_descent_optimize(perturbed, t_sub, n_stages=3, eta_start=0.2, eta_end=0.02)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 9: Final Polish (80s) ===
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 40.0)
        a_fin = mirror_descent_optimize(best_state['seq'], t_fin, n_stages=3, eta_start=0.05, eta_end=0.001)
        try_update(best_state, a_fin, "final_md")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lr = min(rem * 0.4, 30.0)
        a_lr2, _ = local_refinement(best_state['seq'], t_lr, n_steps=10000)
        try_update(best_state, a_lr2, "final_local")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    total_time = time.monotonic() - total_start
    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}, time={total_time:.0f}s")
    return [float(x) for x in best_state['seq']]
