3. Multi-length exploration
4. Exact coordinate descent polishing

import numpy as np
from scipy.fft import rfft, irfft
import time

_DEADLINE = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None, b_init=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

def peak_crush(a, conv, n_top=5, strength=0.1):
    """Reduce the top peaks of the convolution by redistributing mass."""
    n = len(a)
    a = a.copy()
    
    # Find top peaks
    top_k = np.argsort(conv)[-n_top:]
    
    # For each top peak, find which a[j] contribute most
    for k in top_k:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_range = np.arange(i_start, i_end + 1)
        contributions = a[j_range] * a[k - j_range]
        
        # Find the index with the largest contribution
        max_idx = j_range[np.argmax(contributions)]
        max_contrib = contributions.max()
        
        # Reduce a[max_idx] slightly
        reduction = strength * max_contrib / max(a[max_idx], 1e-10)
        reduction = min(reduction, 0.5 * a[max_idx])  # Don't reduce by more than half
        
        a[max_idx] -= reduction
        
        # Redistribute to a random position that doesn't contribute to this peak
        safe_positions = [i for i in range(n) if i not in j_range or a[k-i] < 0.01]
        if safe_positions:
            target = np.random.choice(safe_positions)
            a[target] += reduction
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def mw_best_response(a_fixed, n_steps=50, lr=1.0):
    """Approximate best response using multiplicative weights."""
    n = len(a_fixed)
    b = np.ones(n) / n
    losses = np.zeros(n)
    
    for step in range(n_steps):
        # Compute current convolution a * b
        conv_ab = np.convolve(a_fixed, b)
        k_star = int(np.argmax(conv_ab))
        
        # Subgradient with respect to b at k_star
        # (a*b)[k] = sum_j a[j] * b[k-j]
        # d/d b[i] (a*b)[k_star] = a[k_star - i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a_fixed[j]
        
        # Update losses
        losses += grad
        
        # Multiplicative weights update
        weights = np.exp(-lr * losses)
        weights = np.maximum(weights, 1e-30)
        b = weights / weights.sum()
    
    return b

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def init_flat_spectrum(n, rng):
    """Non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()

    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

def init_fourier_mix(n, rng):
    """Mix of Fourier modes."""
    n_harmonics = rng.integers(3, 12)
    amps = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
    max_freq = max(2, n // 3)

    t = np.arange(n, dtype=np.float64)

        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with multiplicative factor search.
    For each coordinate, try multiple scale factors and pick the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    # Multiplicative factors to try (log-spaced around 1)
    factors = np.concatenate([
        np.arange(0.3, 0.95, 0.05),
        np.arange(1.05, 3.0, 0.1),
    ])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_x = v
            best_local_score = 2.0 * n * max(unaff_max, float(conv[k_lo:k_hi+1].max())) / (current_sum * current_sum)

            for factor in factors:
                x = v * factor
                if x < -1e-15:
                    continue
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = max(unaff_max, float(nv.max()))
                ns = current_sum + (x - v)
                if ns < 0.01:
                    continue
                sc = 2.0 * n * local_max / (ns * ns)
                if sc < best_local_score - 1e-15:
                    best_local_score = sc
                    best_x = x

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    factors = [0.999, 0.998, 0.996, 0.994, 0.992, 0.99, 0.985, 0.98,
               1.01, 1.012, 1.014, 1.016, 1.018, 1.02, 1.025, 1.03, 1.04]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a[partners[idx]]

            if has_quad:
                quad_idx = 2 * i - k_lo

            best_x = v
            best_local = float(conv[k_lo:k_hi+1].max())

            for factor in factors:
                x = v * factor
                nv = c_vals + 2.0 * x * a_partners
                if has_quad:
                    if not isinstance(nv, np.ndarray) or nv is c_vals:
                        nv = nv.copy()
                    nv[quad_idx] = c_vals[quad_idx] + x * x
                local_max = float(nv.max())
                if local_max < best_local - 1e-16:
                    best_local = local_max
                    best_x = x

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = max(a[i], 0.0)
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    if not isinstance(changes, np.ndarray):
                        changes = np.array(changes)
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, sweeps

def fictitious_play(a0, time_limit, solver=None, verbose=False):
    """
    Fictitious play with LP best responses.
    Uses time-averaged iterates and adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    a_avg = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Adaptive step size: larger early, smaller later
        alpha = 2.0 / (iters + 3.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        # Time average
        a_avg = (iters * a_avg + a_new) / (iters + 1.0)

        # Check candidates
        for cand in [a_new, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        a = a_new

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.8f}, M_br={M_val:.6f}")

    return best_a, best_score, iters

def peak_targeting(a0, n_peaks=5, rng=None):
    """
    Directed perturbation: explicitly target and reduce top convolution peaks.
    Returns a perturbed sequence.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    conv = fast_convolve(a)
    # Find top peaks
    top_k = np.argsort(conv)[-n_peaks:]

    for k in top_k:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_range = np.arange(i_start, i_end + 1)
        if len(j_range) == 0:
            continue
        partners = k - j_range
        contributions = a[j_range] * a[partners]
        # Pick the coordinate with largest contribution
        max_pos = np.argmax(contributions)
        idx = j_range[max_pos]
        reduction = 0.05 * a[idx] * rng.random()
        a[idx] -= reduction
        # Redistribute to a random position
        target = rng.integers(0, n)
        a[target] += reduction

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def basin_hopping(a0, time_limit, cd_func, rng, n_perturbs=6, verbose=False):
    """Basin hopping with multiple perturbation strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        t_alloc = min(rem * 0.65, 80.0)

        # Multiple perturbation strategies
        if p % 3 == 0:
            # Gaussian noise
            sigma = 0.015 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif p % 3 == 1:
            # Peak targeting
            perturbed = peak_targeting(a, n_peaks=8, rng=rng)
            perturbed += rng.standard_normal(n) * 0.01
        else:
            # Random coordinate swap
            perturbed = a.copy()
            n_swaps = rng.integers(n // 10, n // 5)
            for _ in range(min(n_swaps, 200)):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                delta = 0.02 * a[i1] * rng.random()
                perturbed[i1] -= delta
                perturbed[i2] += delta

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = cd_func(perturbed, t_alloc)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH p={p}: improved to {best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length diverse initialization + quick coordinate descent
    Phase 2: Intensive fictitious play with LP
    Phase 3: Coordinate descent polishing
    Phase 4: Basin hopping
    Phase 5: Fine coordinate descent
    Phase 6: Final verification
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Multi-length diverse initialization + quick CD ===
    print("\n=== Phase 1: Multi-length Init + Quick CD ===")
    t1_start = time.monotonic()

    n_values = [768, 1024, 1280, 1536]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
        # Add nearby lengths
        for delta in [-256, 256, 512]:
            alt = np_ + delta
            if 512 <= alt <= 2048 and alt not in n_values:
                n_values.append(alt)

    candidates = []

    for n in n_values[:5]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 240:
            break

        inits = []

        inits.append(('four', init_fourier_mix(n, rng)))
        inits.append(('flat2', init_flat_spectrum(n, rng)))

        for itype, cand in inits[:6]:
            t_left = 240.0 - (time.monotonic() - t1_start)

            t_alloc = max(4.0, min(15.0, t_left / 8.0))

            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best internal: {candidates[0][0]:.8f}")

    for s_c, a_c, n_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{n_c}_{it_c}")

    # === Phase 2: Fictitious Play with LP ===
    print(f"\n=== Phase 2: Fictitious Play (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try FP at the best length and one other
        fp_lengths = [n_best]
        for alt_n in [1024, 768, 1280, 1536]:
            if alt_n != n_best and len(fp_lengths) < 3:
                fp_lengths.append(alt_n)

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 400
            if rem < 60:

            t_alloc = min(rem * 0.45, 250.0)

            initial = interpolate_seq(best_seq, n_fp)

    # === Phase 5: Fine polishing ===
    print(f"\n=== Phase 5: Fine Polish (best={best_score:.8f}) ===")
    if best_seq is not None:

            print(f"  FineCD, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine,
                                                   verbose=True, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd1")

        # Another ECD round
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd2 = min(rem * 0.5, 40.0)
            print(f"  ECD-polish, budget={t_cd2:.0f}s")
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, t_cd2, label="polish1")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ecd_polish")

        # Final fine pass
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_fin, s_fin, _ = fine_coord_descent(best_seq, min(rem * 0.7, 25.0), label="last")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "last_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
