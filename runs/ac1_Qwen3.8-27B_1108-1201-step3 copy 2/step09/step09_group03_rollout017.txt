import numpy as np

import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return float(2.0 * n * fast_convolve(a).max())


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_alternating(a_init, time_limit, solver, damping=0.0, verbose=True):
    """
    LP-based block coordinate descent.
    If damping > 0, use damped updates: a_new = (1-d)*a + d*b_lp.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP for b given a
        b_new = solve_best_response_lp(a, solver)
        if b_new is None:
            break
        if damping > 0:
            b_new = (1 - damping) * b + damping * b_new
            b_new = np.maximum(b_new, 0)
            b_new /= b_new.sum()
        b = b_new

        # Solve LP for a given b
        a_new = solve_best_response_lp(b, solver)
        if a_new is None:
            break
        if damping > 0:
            a_new = (1 - damping) * a + damping * a_new
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        a = a_new

        # Evaluate
        sc_a = score_normalized(a)
        sc_b = score_normalized(b)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()
        # Also check the average

        avg /= avg.sum()
        sc_avg = score_normalized(avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg.copy()

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP alt it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    best_score = score_normalized(a)
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1

        for step in range(n_fw_per_round):
            k_star = int(np.argmax(ab_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = a[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            t_away = 0.0
            if use_away and step >= 10:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = b * grad_full
                i_away = int(np.argmax(away_scores))
                if b[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.4, b[i_away] * 0.5)
                    ks_s = i_away

                    ab_conv[ks_s:ks_e + 1] -= t_away * a
                    b[i_away] -= t_away
                    b[i_away] = max(b[i_away], 0.0)

            ab_conv = (1.0 - t) * ab_conv
            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            ab_conv[ks_s:ks_e + 1] += t * a
            b = (1.0 - t) * b
            b[j_star] += t

        bs = score_normalized(b)
        if bs < best_score:

            best_a = b.copy()

        # Optimize a given b
        ab_conv = convolve_ab(b, a)
        for step in range(n_fw_per_round):
            k_star = int(np.argmax(ab_conv))
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            grad_slice = b[k_star - i_end:k_star - i_start + 1][::-1]
            j_star = i_start + int(np.argmin(grad_slice))
            t = 2.0 / (step + 2.0)

            t_away = 0.0
            if use_away and step >= 10:
                grad_full = np.zeros(n)
                grad_full[i_start:i_end + 1] = grad_slice
                away_scores = a * grad_full
                i_away = int(np.argmax(away_scores))
                if a[i_away] > 1e-10 and grad_full[i_away] > 1e-10:
                    t_away = min(t * 0.4, a[i_away] * 0.5)
                    ks_s = i_away
                    ks_e = min(ks_s + n - 1, 2 * n - 2)
                    ab_conv[ks_s:ks_e + 1] -= t_away * b
                    a[i_away] -= t_away
                    a[i_away] = max(a[i_away], 0.0)

            ab_conv = (1.0 - t) * ab_conv
            ks_s = j_star
            ks_e = min(ks_s + n - 1, 2 * n - 2)
            ab_conv[ks_s:ks_e + 1] += t * b
            a = (1.0 - t) * a
            a[j_star] += t

        asc = score_normalized(a)
        if asc < best_score:
            best_score = asc
            best_a = a.copy()

        # Check average
        avg = 0.5 * (a + b)
        avg /= avg.sum()
        sc_avg = score_normalized(avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg.copy()

        if verbose and round_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW round={round_num} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a_init, time_limit, lr_init=0.03, top_k=30, verbose=True):
    """Multi-peak mirror descent."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        sc = float(2.0 * n * M)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-9:
                break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            sg[i_s:i_e + 1] += w[idx] * 2.0 * a[k - i_e:k - i_s + 1][::-1]

        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= sn
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6g}")

    return best_a, best_score


def cd_refine(a_init, time_limit, verbose=True):
    """Coordinate descent with adaptive steps."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = float(2.0 * n * M)
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if a[i] < 1e-14:
                continue
            k_s = i
            k_e = min(i + n - 1, 2 * n - 2)
            unaff = 0.0
            if k_s > 0:
                unaff = max(unaff, conv[:k_s].max())
            if k_e < 2 * n - 2:
                unaff = max(unaff, conv[k_e+1:].max())

            a_sh = a[k_s - i:k_e - i + 1]
            best_delta = 0.0
            best_M = M

            for factor in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]:
                delta = a[i] * factor
                nv = a[i] + delta
                if nv < -1e-15:
                    continue
                new_aff = conv[k_s:k_e+1].copy() + 2.0 * delta * a_sh
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx = 2 * i - k_s
                    new_aff[idx] = conv[2*i] + 2.0*delta*a[i] + delta*delta
                new_M = max(unaff, float(new_aff.max()))
                if new_M < best_M:
                    best_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_sh
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx = 2 * i - k_s
                    changes = changes.copy()
                    changes[idx] = 2.0*best_delta*(a[i]-best_delta) + best_delta*best_delta
                conv[k_s:k_e+1] += changes
                M = max(unaff, float(conv[k_s:k_e+1].max()))
                sc = float(2.0 * n * M)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_cd(a_init, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile improvement."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = float(2.0 * n * M)
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if a[i] < 1e-14:
                continue
            k_s = i
            k_e = min(i + n - 1, 2 * n - 2)
            unaff = 0.0
            if k_s > 0:
                unaff = max(unaff, conv[:k_s].max())
            if k_e < 2 * n - 2:
                unaff = max(unaff, conv[k_e+1:].max())

            a_sh = a[k_s - i:k_e - i + 1]
            best_delta = 0.0
            best_M = M

            for factor in [0.995, 0.99, 0.985, 0.9925, 1.005, 1.01, 1.015, 1.0075]:
                delta = a[i] * (factor - 1.0)
                nv = a[i] + delta
                if nv < -1e-15:
                    continue
                new_aff = conv[k_s:k_e+1].copy() + 2.0 * delta * a_sh
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx = 2 * i - k_s
                    new_aff[idx] = conv[2*i] + 2.0*delta*a[i] + delta*delta
                new_M = max(unaff, float(new_aff.max()))
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_sh
                if 2 * i >= k_s and 2 * i <= k_e:
                    idx = 2 * i - k_s
                    changes = changes.copy()
                    changes[idx] = 2.0*best_delta*(a[i]-best_delta) + best_delta*best_delta
                conv[k_s:k_e+1] += changes
                M = max(unaff, float(conv[k_s:k_e+1].max()))
                sc = float(2.0 * n * M)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 5:
            break
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def griffin_lim_init(n, n_iter=200, seed=42):
    """Spectral flattening via alternating projections."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    for _ in range(n_iter):
        A = rfft(a, 2 * n)
        mag = np.abs(A)
        target_mag = mag.mean()
        phase = np.exp(1j * np.angle(A))
        A_new = target_mag * phase
        a_new = irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        a = a_new
    return a


def cosine_init(n, power=1.0):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center
    a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def check_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        sc = float('inf')
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
    return best_seq, best_score
