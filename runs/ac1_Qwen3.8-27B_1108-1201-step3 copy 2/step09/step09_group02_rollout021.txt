import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def fast_convolve_raw(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def smooth_score_and_grad(a, temperature):
    """
    Compute smooth approximation of max(conv) and its gradient.
    Uses logsumexp with given temperature.
    Returns (smooth_max, gradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Scale for numerical stability
    conv_shifted = conv - conv.max()
    weights = np.exp(conv_shifted / temperature)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        weights = weights / w_sum

    smooth_max = float(conv.max() + temperature * np.log(w_sum))

    # Gradient: d/d(a_i) sum_k w_k * conv_k
    # d conv_k / d a_i = a_{k-i} (if 0 <= k-i < n and k != 2i)
    #                    = 2*a_i (if k = 2i)
    # So grad_i = sum_k w_k * (a_{k-i} + a_{k-i}*[k=2i])
    #           = sum_k w_k * a_{k-i} + sum_k w_k * a_i * [k=2i]
    # First term: cross-correlation of w with a
    # Second term: a_i * sum_{k=2i} w_k = a_i * w_{2i} (if 2i < 2n-1)

    # Cross-correlation: (w cross a)_i = sum_k w_k * a_{k-i} for valid k-i
    # This is: sum_{j=0}^{n-1} a_j * w_{i+j}
    # = convolution of reversed a with w, or directly via FFT

    # Use FFT: cross_corr = ifft(conj(FFT(a)) * FFT(w_padded))
    # Actually simpler: compute via direct convolution
    # grad_i = sum_{j=0}^{n-1} a[j] * w[i+j] (for i+j < 2n-1)
    # This is (a * reversed_w)[i + (2n-2-n+1)] ... let me just use correlation

    # np.correlate(w, a, mode='same') won't work directly due to different lengths
    # Let's use FFT approach
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_ft = rfft(a, fft_size)
    W_ft = rfft(weights, fft_size)
    # Cross-correlation: sum_k w_k * a_{k-i}
    # = ifft(conj(A_ft) * W_ft)
    grad = irfft(np.conj(A_ft) * W_ft, fft_size)[:n]

    # Add the diagonal correction: a_i * w_{2i}
    diagonal = np.zeros(n)
    for i in range(n):
        if 2 * i < 2 * n - 1:
            diagonal[i] = a[i] * weights[2 * i]
    grad += diagonal

    return smooth_max, grad

def smooth_grad_optimize(a0, time_limit, temperature=5.0, verbose=False, label=""):
    """
    Optimize using smooth gradient (logsumexp) with projected gradient + momentum.
    Works on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()

    # Momentum-based gradient descent on the simplex
    v_mom = np.zeros(n)
    lr = 0.01
    beta = 0.9
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Adaptive temperature: decrease over time
        T = max(temperature * 0.5 ** (it / 100), 0.5)

        smooth_max, grad = smooth_score_and_grad(a, T)

        # Momentum update
        v_mom = beta * v_mom + (1 - beta) * grad
        a_new = a - lr * v_mom

        # Project onto simplex
        a_new = simplex_project(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
                beta = min(beta + 0.005, 0.99)

        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGO {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} T={T:.2f} lr={lr:.4f}")

    return best_a, best_score

def l_bfgs_smooth(a0, temperature, max_iter=200, tol=1e-10):
    """
    Use scipy L-BFGS-B to optimize the smooth objective.
    Parameterization: a = exp(z) / sum(exp(z)) (softmax), so z is unconstrained.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15).copy()
    a_init = a_init / a_init.sum()
    # Initial z: log(a) (softmax parameterization)
    z_init = np.log(a_init) - np.log(a_init).mean()

    def objective(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        conv_shifted = conv - conv.max()
        logsumexp = float(conv.max() + np.log(np.exp(conv_shifted / temperature).sum()) * temperature)
        return 2.0 * n * logsumexp

    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        conv_shifted = conv - conv.max()
        weights = np.exp(conv_shifted / temperature)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w_sum = 1e-30
        weights = weights / w_sum

        # d logsumexp / d a_i = sum_k w_k * d(conv_k)/d(a_i)
        # = sum_k w_k * a_{k-i} + a_i * w_{2i} (diagonal)
        fft_size = 1 << (2 * n - 2).bit_length()
        A_ft = rfft(a, fft_size)
        W_ft = rfft(weights, fft_size)
        grad_a = irfft(np.conj(A_ft) * W_ft, fft_size)[:n]

        # Diagonal
        for i in range(n):
            if 2 * i < 2 * n - 1:
                grad_a[i] += a[i] * weights[2 * i]

        # Chain rule: d a_i / d z_i = a_i * (1 - a_i)
        # d a_j / d z_i = -a_i * a_j for j != i
        # So d obj / d z_i = grad_a_i * a_i * (1 - a_i) - a_i * sum_j grad_a_j * a_j
        za = (grad_a * a).sum()
        grad_z = grad_a * a * (1 - a) - a * za
        return 2.0 * n * grad_z

    result = minimize(objective, z_init, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8})

    z_opt = result.x
    a_opt = np.exp(z_opt - z_opt.max())
    a_opt = a_opt / a_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    a_opt = a_opt / a_opt.sum()

    conv_opt = fast_convolve(a_opt)
    score = 2.0 * n * float(conv_opt.max())
    return a_opt, score

def build_conv_matrix_sparse(a):
    """Build list of (k, indices, coeffs) for each lag k."""
    n = len(a)
    constraints = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        indices = list(range(i_start, i_end + 1))
        coeffs = [a[k - i] for i in indices]
        constraints.append((k, indices, coeffs))
    return constraints

def fictitious_play_cp(a0, time_limit, solver, verbose=False, label=""):
    """
    LP fictitious play with cutting planes and multiplicative weight updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    iters = 0
    a_ewma = a.copy()
    alpha_ewma = 2.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        # Solve LP best response with cutting planes
        b, M_val = lp_best_response_cutting_plane(a, solver, max_constraints=150, max_iters=20)
        if b is None:
            continue

        # Line search along a -> b direction
        best_t = 0.0
        best_ls_score = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_ls_score:
                best_ls_score = sc
                best_t = trial_t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            # EWM update
            alpha = alpha_ewma / (iters + alpha_ewma)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            a = a / a.sum()

        a_ewma = (iters * a_ewma + a) / (iters + 1)

        # Check candidates
        for cand, name in [(a, 'a'), (a_ewma, 'ewma'), (b, 'b')]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = 2.0 * n * float(fast_convolve(cand).max())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def golden_section_search(f, lo, hi, max_iter=35):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent with golden section 1D search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = -1e30
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            if has_quad:
                quad_idx = 2 * i - k_lo

            def compute_score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                              _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                              _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(2.0 * v, 3.0 / n)
            best_x = golden_section_search(compute_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-12:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def init_flat_spectrum(n, rng):
    """Generate non-negative sequence with approximately flat power spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cosine(n, rng):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping_smooth(a0, time_limit, rng, verbose=False, label=""):
    """
    Basin hopping: perturb, then optimize with smooth gradient + CD.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()

    n_perturbs = max(3, int(time_limit / 60))

    for p in range(n_perturbs):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.7, 80.0)
        t_cd_alloc = min(rem * 0.3, 40.0)

        # Multiple perturbation strategies
        strategy = p % 4
        if strategy == 0:
            # Gaussian noise
            sigma = 0.02 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif strategy == 1:
            # Peak-breaking: reduce values at positions contributing to max autocorrelation
            conv_p = fast_convolve(a)
            top_k = min(3, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            perturbed = a.copy()
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 4)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= max(0, 1.0 + rng.standard_normal() * 0.1)
        elif strategy == 2:
            # Block shift: shift a random block
            perturbed = a.copy()
            blen = rng.integers(n // 10, n // 3)
            start = rng.integers(0, max(1, n - blen))
            shift = rng.integers(-n // 5, n // 5 + 1)
            if shift != 0:
                new_start = max(0, min(n - blen, start + shift))
                perturbed[new_start:new_start + blen] = perturbed[start:start + blen]
                perturbed[start:start + blen] = 0
        else:
            # Random rescale subset
            perturbed = a.copy()
            n_subset = rng.integers(n // 5, n // 2)
            subset = rng.choice(n, size=n_subset, replace=False)
            factor = 1.0 + rng.standard_normal(n_subset) * 0.15
            perturbed[subset] *= factor

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Optimize: smooth gradient first, then CD
        a_sg, s_sg = smooth_grad_optimize(perturbed, t_alloc * 0.6, temperature=3.0)
        a_cd, s_cd, _ = exact_coord_descent(a_sg, t_cd_alloc)

        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"    BH {label} p={p} s={strategy}: improved to {best_score:.8f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluation function and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

            print(f"  ECD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc, verbose=True, label=f"ph3_r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ecd_r{rnd}")

            if best_score > 1.5035:
                # Perturb to escape local minimum
                perturbed = a_opt + rng.standard_normal(n_best) * 0.025
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                current = perturbed
            else:
                current = a_opt.copy()

    # === Phase 4: LP Fictitious Play with cutting planes ===
    print(f"\n=== Phase 4: FP with Cutting Planes (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768]:
            if alt_n != n_best:
                fp_lengths.append(alt_n)
        fp_lengths = list(set(fp_lengths))[:2]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 40:
                break
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp)
            a_fp, score_fp = fictitious_play_cp(initial, t_alloc, solver, verbose=True, label=f"n{n_fp}")
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

            # Follow up with CD
            rem = deadline - time.monotonic() - 200
            if rem > 20 and best_seq is not None:
                a_cd, s_cd, _ = exact_coord_descent(interpolate_seq(a_fp, len(best_seq)), min(rem * 0.4, 60.0))
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_after_fp_{n_fp}")

    # === Phase 5: Basin hopping ===
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_bh = min(rem * 0.6, 180.0)
            print(f"  Basin hopping, budget={t_bh:.0f}s")
            a_bh, s_bh = basin_hopping_smooth(best_seq, t_bh, rng, verbose=True, label="main")
            best_seq, best_score = check_and_update(a_bh, best_seq, best_score, "basin_hop")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Smooth gradient final pass
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            a_sg2, s_sg2 = smooth_grad_optimize(best_seq, min(rem * 0.4, 50.0), temperature=1.0, label="final_sg")
            best_seq, best_score = check_and_update(a_sg2, best_seq, best_score, "final_sg")

        # ECD polish
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_cd2, s_cd2, _ = exact_coord_descent(best_seq, min(rem * 0.6, 40.0), label="final_cd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "final_cd")

        # One more smooth pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_sg3, s_sg3 = smooth_grad_optimize(best_seq, min(rem * 0.7, 20.0), temperature=0.5, label="last_sg")
            best_seq, best_score = check_and_update(a_sg3, best_seq, best_score, "last_sg")

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_cd3, s_cd3, _ = exact_coord_descent(best_seq, min(rem * 0.8, 15.0), label="last_cd")
            best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, "last_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
