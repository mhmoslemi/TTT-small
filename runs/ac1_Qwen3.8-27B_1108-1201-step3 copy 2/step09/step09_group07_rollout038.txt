import numpy as np

from scipy.fft import rfft, irfft

# Helper functions at top level

def fast_conv(a, b=None):
    """Linear convolution of a with itself or with b."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_norm(a):
    """Score for sum-normalized sequence."""
    b = fast_conv(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_grad(a, T):

    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr

def nesterov_pg(a0, time_limit, T_start=2.0, T_end=200.0,
                 lr_init=0.05, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_norm(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            lr = lr_init / (1.0 + 0.01 * step_count)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count)) / 2.0
            if step_count > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1))) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = smooth_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 100 == 0:
                sc = score_norm(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
    return best_a, best_score

def eg_best_response(a, n_iters=5000, eta=0.5):
    """Exponentiated gradient best response to sequence a."""
    n = len(a)
    b = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    
    for it in range(n_iters):
        # Compute conv(a, b)
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:2*n-1]
        
        # Compute softmax weights (approximate subgradient)
        c_max = c.max()
        if c_max < 1e-15:
            break
        # Use a moderate temperature for smoothing
        T_smooth = 50.0
        wc = c * T_smooth
        wc -= wc.max()
        e = np.exp(wc)
        w = e / e.sum()
        
        # Gradient: correlate a with w
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]
        
        # Multiplicative update
        b *= np.exp(-eta * g / c_max)
        b = np.maximum(b, 1e-30)
        b /= b.sum()
    
    return b

def lp_best_response(a, solver=None):
    """LP-based best response."""
    if solver is None:
        solver = _get_solver()
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint: conv(a, b) <= M
    # conv(a, b)[k] = sum_j a[j] * b[k-j] for valid j
    # We can express this using the convolution matrix
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def alternating_eg(a0, time_limit, n_iters_per_side=2000, eta=0.5, alpha=0.3):
    """Alternating exponentiated gradient."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Best response for b given a
        b = eg_best_response(a, n_iters_per_side, eta)
        # Update a towards b
        a_new = project_simplex((1-alpha) * a + alpha * b)
        score = score_norm(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        iter_count += 1
        if iter_count % 10 == 0:
            print(f"    Alt-EG iter={iter_count}: score={best_score:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit):
    """Peak shaving coordinate descent."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        # Find the peak (exclude endpoints)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        top = np.argsort(contribs)[::-1][:8]
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if i >= n or j >= n:
                continue
            for factor in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                if i != j:
                    a_trial[i] *= factor
                    a_trial[j] *= factor
                else:
                    a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_norm(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try single coordinate
            for ci in top:
                i = idx[ci]
                for factor in [0.97, 0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = project_simplex(a_trial)
                    sc = score_norm(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return best_a, best_score

def make_init(n, rng, style=0):
    """Create various initializations."""
    if style == 0:  # Geometric
        a = np.geomspace(1.0, 0.1, n)
    elif style == 1:  # Exponential decay
        a = np.exp(-3.0 * np.arange(n) / n)
    elif style == 2:  # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20*(x-0.2)**2) + np.exp(-20*(x-0.8)**2)
    elif style == 3:  # Random
        a = rng.random(n)
    elif style == 4:  # Triangular (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2*x) + 0.1

        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2*np.pi*x - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

- Adaptive temperature annealing in the smoothed max for EG
- Dual averaging with diminishing step size for stable convergence
- Structured asymmetric initializations exploiting the non-symmetry
  of the optimal solution
- Multi-length exploration with interpolation-based transfer
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

_lp_solver_name = None


def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for s in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = s
                return s
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def fast_conv(a, b=None):
    """Linear convolution of a with b (or itself)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_norm(a):
    """Score for sum-normalized non-negative sequence."""
    b = fast_conv(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, T):
    """Gradient of (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_pg(a0, time_limit, T_start=2.0, T_end=200.0,
                 lr_init=0.06, n_stages=10, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_norm(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.02 * global_step)
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * step_count)) / 2.0
            if step_count > 0:
                t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1))) / 2.0
                beta = (t_prev - 1.0) / t_curr
            else:
                beta = 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)
            grad = smooth_max_grad(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1
            if step_count % 80 == 0:
                sc = score_norm(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

    final_sc = score_norm(x)
    if final_sc < best_score:
        best_score = final_sc
        best_a = x.copy()
    return best_a, best_score


def eg_best_response(a, n_iters=3000, eta=0.5, T_smooth_start=30.0, T_smooth_end=100.0):
    """
    Exponentiated gradient approximation to the best response problem:
    minimize max_k conv(a,b)_k subject to b >= 0, sum(b) = 1.
    Uses adaptive temperature for the smoothed max.
    """
    n = len(a)
    b = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)

    for it in range(n_iters):
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:2 * n - 1]
        c_max = c.max()
        if c_max < 1e-15:
            break

        # Adaptive temperature: start low (focus on peaks), increase to smooth
        progress = it / max(1, n_iters)
        T_smooth = T_smooth_start + (T_smooth_end - T_smooth_start) * progress

        wc = c * T_smooth
        wc -= wc.max()
        e = np.exp(wc)
        w = e / e.sum()

        # Gradient: correlation of a with w, truncated to length n
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        # Adaptive step size
        eta_t = eta / (1.0 + 0.1 * np.sqrt(it))

        # Multiplicative weight update
        b *= np.exp(-eta_t * g / c_max)
        b = np.maximum(b, 1e-30)
        b /= b.sum()

    return b


def alternating_eg(a0, time_limit, n_eg_iters=2000, eta=0.6, alpha_lr=0.25,
                   T_start=30.0, T_end=120.0, restart_threshold=50):
    """
    Alternating exponentiated gradient with adaptive mixing.
    Alternates: b = best_response(a); a = mix(a, b)
    With periodic restarts on stagnation.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    iter_count = 0
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b_br = eg_best_response(a, n_eg_iters, eta, T_start, T_end)
        alpha = alpha_lr / (1.0 + 0.05 * iter_count)
        a_new = project_simplex((1.0 - alpha) * a + alpha * b_br)
        sc = score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall_count = 0
        a = a_new
        iter_count += 1

        if iter_count % 20 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= restart_threshold:
                # Perturb and restart
                a = a + 0.03 * rng.standard_normal(n)
                a = project_simplex(a)
                stall_count = 0
                print(f"    Alt-EG restart at iter {iter_count}, score={best_score:.6f}")

        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Alt-EG iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_mat(n):
    """Precompute structure for convolution matrix (returns index info)."""
    # T has shape (2n-1, n), T[k,j] = a[k-j] if 0<=k-j<n
    return n


def lp_best_response(a, solver):
    """Solve the LP for best response to a."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def lp_refine(a0, time_limit, solver, n_max_iters=20000):
    """LP-based best response refinement with dual averaging."""
    n = len(a0)
    a = project_simplex(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_br = lp_best_response(y, solver)
        if b_br is None:
            b_br = eg_best_response(y, 2000, 0.5)

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_br
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_br) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        sc_a = score_norm(a)
        sc_y = score_norm(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                y = y + 0.02 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit):
    """Aggressive peak shaving with coordinate and pair updates."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        top = np.argsort(contribs)[::-1][:10]

        # Try pair reductions
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if i >= n or j >= n or i == j:
                continue
            if a[i] < 1e-14 or a[j] < 1e-14:
                continue
            for f in [0.95, 0.90, 0.85, 0.80, 0.75]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial[j] *= f
                a_trial = project_simplex(a_trial)
                sc = score_norm(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try single coordinate
        if not improved:
            for ci in top:
                i = idx[ci]
                for f in [0.97, 0.94, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial = project_simplex(a_trial)
                    sc = score_norm(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try multi-peak attack
            threshold = b_work[peak_k] * 0.8
            near_peaks = np.where(b_work >= threshold)[0]
            multi_improved = False
            for pk in near_peaks[:3]:
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                idx2 = np.arange(js2, je2 + 1)
                contribs2 = a[idx2] * a[pk - idx2]
                top2 = np.argsort(contribs2)[::-1][:3]
                for ci in top2:
                    i = idx2[ci]
                    a_trial = a.copy()
                    a_trial[i] *= 0.93
                    a_trial = project_simplex(a_trial)
                    sc = score_norm(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        multi_improved = True
                        break
                if multi_improved:
                    break
            if not multi_improved:
                break

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence through the official evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def make_init(n, rng, style=0):
    """Create diverse initializations exploiting asymmetry."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Strong geometric decay (mass at start)
        a = np.exp(-5.0 * x)
    elif style == 1:
        # Bimodal (mass at both ends)
        a = np.exp(-30.0 * (x - 0.15) ** 2) + np.exp(-30.0 * (x - 0.85) ** 2)
    elif style == 2:
        # Asymmetric cosine
        a = 0.5 + 0.45 * np.cos(2.0 * np.pi * x - 0.7)
    elif style == 3:
        # Power law
        a = (x + 0.01) ** 3
    elif style == 4:
        # Random with structure
        a = 0.5 + 0.5 * np.sin(3.0 * np.pi * x + rng.random())
    elif style == 5:
        # Pure random
        a = rng.random(n)
    elif style == 6:
        # Inverse geometric (mass at end)
        a = np.exp(5.0 * (x - 1.0))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def update_best(a, best_seq, best_score, label):
    """Try to update best sequence. Returns (updated_seq, updated_score)."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return a.copy(), sc
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Nesterov PG warmup (get into good basin)
    2. Alternating EG (many fast iterations)
    3. LP-based refinement (exact best responses)
    4. Peak shaving (local coordinate optimization)
    5. Multi-start exploration (diverse initializations)
    6. Alternative lengths
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    if best_seq is None:
        n_init = 960
        best_seq = make_init(n_init, rng, style=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: Nesterov PG Warmup ===
    print(f"\n=== Phase 1: NG Warmup (n={n_best}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 300.0)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a1, _ = nesterov_pg(best_seq, t1, T_start=2.0, T_end=250.0,
                            lr_init=0.07, n_stages=10, steps_per_stage=2500)
        best_seq, best_score = update_best(a1, best_seq, best_score, "ng1")

    # === Phase 2: Alternating EG (main workhorse) ===
    print(f"\n=== Phase 2: Alternating EG (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a2, _ = alternating_eg(best_seq, t2, n_eg_iters=1500, eta=0.6,
                               alpha_lr=0.25, T_start=20.0, T_end=150.0,
                               restart_threshold=40)
        best_seq, best_score = update_best(a2, best_seq, best_score, "alt_eg")

    # === Phase 3: LP Refinement ===
    print(f"\n=== Phase 3: LP Refinement (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a3, _ = lp_refine(best_seq, t3, solver, n_max_iters=50000)
        best_seq, best_score = update_best(a3, best_seq, best_score, "lp_main")

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        a4, _ = peak_shave(best_seq, t4)
        best_seq, best_score = update_best(a4, best_seq, best_score, "peak1")

    # === Phase 5: Second Alt-EG round (finer) ===
    print(f"\n=== Phase 5: Alt-EG Round 2 (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 180.0)
    if t5 > 15:
        print(f"  Budget: {t5:.0f}s")
        a5, _ = alternating_eg(best_seq, t5, n_eg_iters=2500, eta=0.4,
                               alpha_lr=0.15, T_start=50.0, T_end=300.0,
                               restart_threshold=60)
        best_seq, best_score = update_best(a5, best_seq, best_score, "alt_eg2")

    # === Phase 6: Multi-Start ===
    print(f"\n=== Phase 6: Multi-Start (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        styles = [0, 1, 3, 6]
        for si, style in enumerate(styles):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.30, 150.0)
            if t_sub < 25:
                break
            print(f"  Style {style}, budget={t_sub:.0f}s")
            init = make_init(n_best, rng, style=style)
            # Quick NG
            a_sub, _ = nesterov_pg(init, t_sub * 0.4, T_start=1.0, T_end=100.0,
                                   lr_init=0.08, n_stages=6, steps_per_stage=1500)
            best_seq, best_score = update_best(a_sub, best_seq, best_score, f"ms_s{style}_ng")
            # EG polish
            rem = deadline - time.monotonic()
            if rem > 20:
                a_sub2, _ = alternating_eg(a_sub, min(rem * 0.35, 50.0),
                                           n_eg_iters=1000, eta=0.5,
                                           alpha_lr=0.2, T_start=30.0, T_end=100.0,
                                           restart_threshold=30)
                best_seq, best_score = update_best(a_sub2, best_seq, best_score, f"ms_s{style}_eg")

    # === Phase 7: Alternative Lengths ===
    print(f"\n=== Phase 7: Alt Lengths (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        alt_ns = []
        for d in [-40, -20, 20, 40, 80]:
            alt_n = n_best + d
            if 512 <= alt_n <= 2048:
                alt_ns.append(alt_n)
        for alt_n in alt_ns[:3]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alt = min(rem * 0.30, 120.0)
            if t_alt < 20:
                break
            print(f"  n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, _ = nesterov_pg(init_alt, t_alt * 0.5, T_start=2.0, T_end=150.0,
                                   lr_init=0.06, n_stages=6, steps_per_stage=1000)
            best_seq, best_score = update_best(a_alt, best_seq, best_score, f"alt_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_alt2, _ = alternating_eg(a_alt, min(rem * 0.4, 30.0),
                                           n_eg_iters=800, eta=0.5,
                                           alpha_lr=0.2, T_start=30.0, T_end=100.0,
                                           restart_threshold=20)
                best_seq, best_score = update_best(a_alt2, best_seq, best_score, f"alt_n{alt_n}_eg")

    # === Phase 8: Final LP + Peak Shave ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 60.0)
        if t_fin > 10:
            a_fin, _ = lp_refine(best_seq, t_fin, solver, n_max_iters=10000)
            best_seq, best_score = update_best(a_fin, best_seq, best_score, "final_lp")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_ps, _ = peak_shave(best_seq, min(rem * 0.5, 30.0))
            best_seq, best_score = update_best(a_ps, best_seq, best_score, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
