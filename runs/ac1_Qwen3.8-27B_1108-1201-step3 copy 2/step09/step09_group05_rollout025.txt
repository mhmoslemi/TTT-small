def peak_directed_search(a0, time_limit, n_steps=30000):

    Multi-peak directed local search.
    Identifies top peaks and designs moves to reduce them simultaneously.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top K peaks
        K = min(5, len(b))
        peak_threshold = b.max() - 0.01 * b.max()
        peak_positions = np.where(b >= peak_threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = [np.argmax(b)]

        # Find contributions to peaks
        contrib = np.zeros(n)
        for k in peak_positions[:K]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            c = a[idx] * a[k - idx]
            contrib[idx] += c * (b[k] / b.max())

        # Top contributors (to reduce)
        top_reduce = np.argsort(contrib)[::-1][:10]
        # Best candidates to increase (low contribution)
        top_increase = np.argsort(contrib)[:n // 4]

        improved = False

        # Try reducing a top contributor and redistributing
        for i in top_reduce[:5]:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.85]:
                reduce_amt = a[i] * (1 - factor)
                if reduce_amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                # Redistribute to low-contribution positions
                inc_mask = np.zeros(n, dtype=bool)
                for j in top_increase:
                    if j != i:
                        inc_mask[j] = True
                if inc_mask.sum() == 0:
                    continue
                weights = a_trial[inc_mask] + 1e-12
                a_trial[inc_mask] = a_trial[inc_mask] + reduce_amt * weights / weights.sum()
                a_trial = norm_sequence(a_trial)
                sc = score_of(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            k_main = int(np.argmax(b))
            js = max(0, k_main - n + 1)
            je = min(n - 1, k_main)
            idx = np.arange(js, je + 1)
            c = a[idx] * a[k_main - idx]

                i = idx[ci]

                            found = True

                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                no_improve += 1
                if no_improve > 300:
                    break
        else:

def subgradient_method(a0, time_limit, n_steps=50000):
    """
    Subgradient method with Nesterov momentum for direct max optimization.
    Uses Polyak step size based on the theoretical lower bound.
    """
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0

    y = a.copy()  # Nesterov momentum variable
    # Theoretical lower bound on max_b (with sum=1): 1/(2n-1)
    lb = 1.0 / (2 * n - 1)
    # Objective lower bound: 2n * lb = 2n/(2n-1)
    obj_lb = 2.0 * n * lb

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        k_star = int(np.argmax(b))
        peak_val = b[k_star]

        # Subgradient at y
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * y[k_star - idx]
        if k_star % 2 == 0:
            pass  # Already handled: when i = k-i, g[i] = 2*y[i] which is correct

        # Projected subgradient step on simplex
        # Polyak step: alpha = (f(y) - f*) / ||g||^2
        cur_obj = 2.0 * n * peak_val
        gap = cur_obj - obj_lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        alpha = gap / (g_norm_sq * (step + 1))  # Summable step size
        alpha = min(alpha, 0.5)

        # Nesterov update
        beta = (1 + np.sqrt(1 + 4 * (step + 1) ** 2)) / 2
        a_new = y - alpha * (g - np.mean(g))  # Project subgradient
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s

        y = (1 + beta) * a_new - beta * y
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy

        a = a_new
        sc = score_of(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart if stuck
        if no_improve > 2000:
            rng_local = np.random.default_rng(step)
            noise = rng_local.random(n) * 0.05
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-initialization (cosine, Gaussian, frequency-domain, multi-modal)
    2. Fine-grained L-BFGS with adaptive temperature
    3. Peak-directed multi-peak local search
    4. Subgradient with momentum
    5. Low-T L-BFGS polish
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = init_cosine(n_init)
        best_score = score_of(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 0: Try diverse initializations (fast) ===
    print(f"\n=== Phase 0: Diversified Initialization ===")
    inits = []
    inits.append(('cosine', init_cosine(n_best)))
    inits.append(('gaussian', init_gaussian(n_best)))
    inits.append(('flat_top', init_flat_top(n_best)))
    inits.append(('freq1', init_freq_domain(n_best, np.random.default_rng(123))))
    inits.append(('freq2', init_freq_domain(n_best, np.random.default_rng(456))))
    inits.append(('freq_mixed', init_freq_mixed(n_best, np.random.default_rng(789))))
    inits.append(('multimodal', init_multimodal(n_best, np.random.default_rng(321))))
    inits.append(('riesz', init_riesz_nissin(n_best, np.random.default_rng(654))))

    # Score all inits
    init_scores = []
    for name, a_init in inits:
        sc = score_of(a_init)
        init_scores.append((sc, name, a_init))
        print(f"  {name}: {sc:.6f}")
    init_scores.sort()

    # Also include parent
    all_inits = [(best_score, 'parent', best_seq)] + init_scores
    all_inits.sort()

    # Pick top 3 diverse inits
    selected = all_inits[:3]
    for sc, name, a_init in selected:
        print(f"  Selected: {name} ({sc:.6f})")

    # === Phase 1: L-BFGS on best init ===
    print(f"\n=== Phase 1: L-BFGS Fine (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, T: 200->0.005, 30 stages")
        a_lb = lbgfs_refine_fine(best_seq, t_alloc, T_start=200.0, T_end=0.005, n_stages=30)
        try_update(a_lb, "lbgfs_fine_1")

    # === Phase 2: Peak-directed search ===
    print(f"\n=== Phase 2: Peak-Directed Search ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pd, _ = peak_directed_search(best_seq, t_alloc)
        try_update(a_pd, "peak_directed_1")

    # === Phase 3: L-BFGS polish at low T ===
    print(f"\n=== Phase 3: L-BFGS Low-T Polish ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 20->0.005, 20 stages")
        a_lb2 = lbgfs_refine_fine(best_seq, t_alloc, T_start=20.0, T_end=0.005, n_stages=20)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 4: Subgradient method ===
    print(f"\n=== Phase 4: Subgradient + Momentum ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_method(best_seq, t_alloc)
        try_update(a_sg, "subgradient")

    # === Phase 5: Second round of peak-directed + L-BFGS ===
    print(f"\n=== Phase 5: Second Refinement Round ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pd = min(rem * 0.15, 100.0)
        if t_pd > 15:
            a_pd2, _ = peak_directed_search(best_seq, t_pd)
            try_update(a_pd2, "peak_directed_2")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_lb3 = min(rem * 0.3, 150.0)
            if t_lb3 > 20:
                a_lb3 = lbgfs_refine_fine(best_seq, t_lb3, T_start=10.0, T_end=0.003, n_stages=15)
                try_update(a_lb3, "lbgfs_fine_2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 1.33, 2.0]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [512, 768, 1024, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        # Interpolate and also try fresh init
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_interp = score_of(init_alt)
        init_cos = init_cosine(alt_n)
        sc_cos = score_of(init_cos)
        init_fq = init_freq_domain(alt_n, rng)
        sc_fq = score_of(init_fq)

        # Pick best init for this length
        if sc_interp <= sc_cos and sc_interp <= sc_fq:
            start_alt = init_alt
        elif sc_cos <= sc_fq:
            start_alt = init_cos
        else:
            start_alt = init_fq

        # L-BFGS on best init
        t_lb_alt = t_total * 0.7
        a_alt = lbgfs_refine_fine(start_alt, t_lb_alt, T_start=100.0, T_end=0.01, n_stages=20)
        try_update(a_alt, f"lbgfs_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_pd_alt = min(rem * 0.5, 50.0)
            a_alt_pd, _ = peak_directed_search(best_seq, t_pd_alt)
            try_update(a_alt_pd, f"pd_n{alt_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 120.0)
        a_fin = lbgfs_refine_fine(best_seq, t_fin, T_start=5.0, T_end=0.002, n_stages=15)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_pd_fin = min(rem * 0.3, 60.0)
        a_pd_fin, _ = peak_directed_search(best_seq, t_pd_fin)
        try_update(a_pd_fin, "final_pd")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin2 = min(rem * 0.5, 40.0)
        a_fin2 = lbgfs_refine_fine(best_seq, t_fin2, T_start=3.0, T_end=0.001, n_stages=10)
        try_update(a_fin2, "final_lbgfs2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
