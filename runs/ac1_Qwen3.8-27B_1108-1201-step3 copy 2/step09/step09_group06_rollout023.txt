```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_conv_matrix_rows(a, indices):
    """Build rows of the convolution matrix T for given lag indices."""
    n = len(a)
    T = np.zeros((len(indices), n))
    for i, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            T[i, :] = 0
        else:
            idx = np.arange(js, je + 1)
            T[i, idx] = a[k - idx]
    return T


def solve_lp_best_response(a, constraint_indices=None, solver=None):
    """Solve the best response LP: min M s.t. T@g <= M, sum(g)=1, g>=0."""
    n = len(a)
    if constraint_indices is None:
        constraint_indices = list(range(2 * n - 1))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix for selected rows
    T_sub = build_conv_matrix_rows(a, constraint_indices)
    
    constraints = [T_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solver = get_lp_solver()

        prob.solve(solver=solver)
    except Exception:

                prob.solve(solver=alt)
                break

                continue

    exp_c = np.exp(c_scaled - c_max)
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
    A_eq[0, :n] = 1.0
```

```python
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    g_val = np.maximum(res.x[:n], 0)
    s = g_val.sum()
    if s < 1e-15:
        return None, None
    g_val = g_val / s
    return g_val, float(res.x[n])
```
