def fast_convolve(a):

    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov subgradient in softmax (log) parameterization with structured subgradients
2. Fast LP best-response via scipy HiGHS solver (much faster than cvxpy)
3. Peak shaving coordinate descent
4. Multi-start with perturbation and length exploration
Key innovations: log-space optimization avoids projection overhead; HiGHS LP solves in ms;
dual tracking focuses descent on active peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_autoconv(a, k_star):
    """
    Subgradient of max_k conv(a,a)_k w.r.t. a at the point where k_star is the argmax.
    grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0.
    """
    n = len(a)
    js = np.arange(n)
    idx = k_star - js
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad


def softmax_from_logits(theta):
    """Compute softmax from logits, numerically stable."""
    theta_shifted = theta - theta.max()
    a = np.exp(theta_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_logsubgrad(a0, time_limit, lr=0.5, max_steps=200000):
    """
    Nesterov accelerated subgradient in log (softmax) parameterization.
    The subgradient of the auto-convolution max is structured: just a shifted 2*a.
    """
    n = len(a0)
    theta = np.log(np.maximum(a0, 1e-30))
    theta_prev = theta.copy()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum in log-space
        if step > 0:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (step - 1) * (step - 1))) / (2.0 * step + 1.0)
            beta = (step - 1) / (step + 1.0)
            y = theta + beta * (theta - theta_prev)
        else:
            y = theta.copy()

        # Evaluate at y
        a_y = softmax_from_logits(y)
        b = fast_convolve(a_y)
        k_star = int(b.argmax())

        # Subgradient
        g = subgradient_autoconv(a_y, k_star)

        # Step size: eta = c / sqrt(t)
        eta = lr / np.sqrt(step + 1.0)

        # Update theta
        theta_new = y - eta * g
        theta_prev = theta.copy()
        theta = theta_new

        # Expand to track dual (which peaks to attend to) - weighted average of argmaxes
        if step % 100 == 0:
            a_check = softmax_from_logits(theta)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

    a_final = softmax_from_logits(theta)
    sc_final = score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def nesterov_logsubgrad_dual(a0, time_limit, lr=0.5, max_steps=100000):
    """
    Nesterov subgradient with dual tracking: maintain a distribution y over
    convolution indices to guide which peaks to attack.
    """
    n = len(a0)
    size = 2 * n - 1
    theta = np.log(np.maximum(a0, 1e-30))
    theta_prev = theta.copy()
    # Dual variable: distribution over convolution indices
    y_dual = np.ones(size) / size
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        if step > 0:
            beta = (step - 1) / (step + 1.0)
            y = theta + beta * (theta - theta_prev)
        else:
            y = theta.copy()

        a_y = softmax_from_logits(y)
        b = fast_convolve(a_y)

        # Update dual: emphasis on high-convolution indices
        T_dual = 5.0 + step * 0.01
        b_shifted = b - b.max()
        y_new = np.exp(T_dual * b_shifted)
        y_new /= y_new.sum()
        alpha = 2.0 / (step + 2.0)
        y_dual = (1 - alpha) * y_dual + alpha * y_new

        # Weighted subgradient: sum_k y_dual[k] * subgrad_k
        # This is the gradient of E_k~y_dual[conv(a,a)_k] = y_dual @ conv(a,a)
        # = y_dual @ (a * a) which has gradient 2 * (a * y_dual_reversed)_j
        # Actually: d/dx_j [sum_k y_k * sum_i a_i * a_{k-i}]
        # = sum_k y_k * 2*a_{k-j} (for valid k-j)
        # = 2 * sum_k y_k * a_{k-j}
        # This is 2 * (y_dual convolve a) evaluated at position... 
        # Let's compute it directly using FFT
        # h[j] = sum_k y_dual[k] * a[k-j] for valid k-j
        # This is (y_dual * a) at offset... it's a cross-correlation
        fft_size = 1 << (n + size - 1).bit_length()
        YD = rfft(y_dual, fft_size)
        A = rfft(a_y, fft_size)
        # h[j] = sum_k y_dual[k] * a_y[k-j] = cross-correlation
        # This is ifft(conj(YD) * A) at appropriate positions
        corr = irfft(np.conj(YD) * A, fft_size)[:n]
        g = 2.0 * corr

        eta = lr / np.sqrt(step + 1.0)
        theta_new = y - eta * g
        theta_prev = theta.copy()
        theta = theta_new

        if step % 200 == 0:
            a_check = softmax_from_logits(theta)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

    a_final = softmax_from_logits(theta)
    sc_final = score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def solve_lp_best_response(a_fixed, M_bound=None):
    """
    Given fixed a (sum=1, nonneg), find b (sum=1, b>=0) minimizing max_k (a*b)_k.
    Uses scipy linprog with HiGHS.
    Returns (b, M_opt) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1

    # Build constraint matrix: for each k in 0..2n-2, sum_j a[j]*b[k-j] <= M
    # Equivalently: (T @ b) - M * 1 <= 0
    # In standard form for linprog: A_ub @ x <= b_ub
    # x = [M, b_0, ..., b_{n-1}]
    # A_ub has shape (2n-1, n+1): row k is [-1, a[0]..a[n-1]] shifted appropriately

    # A_ub[k, 0] = -1 (coefficient of M)
    # A_ub[k, 1+j] = a[j] if 0 <= k-j < n, else 0

    A_ub = np.zeros((size, n + 1))
    A_ub[:, 0] = -1.0
    for j in range(n):
        # a[j] contributes to rows k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = j + n  # exclusive
        A_ub[k_start:k_end, 1 + j] = a_fixed[j]

    b_ub = np.zeros(size)

    # Equality: sum(b) = 1 → [0, 1, 1, ..., 1] @ x = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize M
    c = np.zeros(n + 1)
    c[0] = 1.0

    # Bounds: M >= 0, b_j >= 0
    bounds = [(0, None)] + [(0, None)] * n

    # Add M upper bound if provided (for warm-starting)
    if M_bound is not None:
        bounds[0] = (0, M_bound)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10, 'presolve': True})
    except Exception:
        return None, None

    if res.status not in (0, 3):  # 0=optimal, 3=iteration limit but feasible
        return None, None

    M_val = res.x[0]
    b_val = res.x[1:]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_val)


def nesterov_lp_refine(a0, time_limit, n_max_iters=5000):
    """
    Nesterov dual averaging LP refinement.
    Alternate between LP best-response and Nesterov averaging.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()  # Nesterov y-point
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Get current max as bound for LP
        b_conv = fast_convolve(y)
        M_current = float(b_conv.max())

        b_new, M_opt = solve_lp_best_response(y, M_bound=M_current + 0.01)
        if b_new is None:
            continue

        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        # Running average for the iterate
        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 2))
        a = project_simplex(a)

        # Check both iterates
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0

        if it % 20 == 0:
            if best_score >= prev_best - 1e-9:
                stall_count += 1
            else:
                stall_count = 0
            prev_best = best_score

            if stall_count >= 5:
                # Perturb to escape
                rng = np.random.default_rng(it * 7 + 3)
                y = y + 0.01 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}, M_opt={M_opt:.8f}")

    # Final check
    sc_a = score_normalized(a)
    if sc_a < best_score:
        best_score = sc_a
        best_a = a.copy()

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=50000):
    """
    Peak shaving: identify peaks in convolution, reduce top contributors.
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stagnation = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(b.argmax())
        peak_val = b[peak_k]

        # Find neighbors near the peak
        threshold = peak_val * 0.95
        near_peaks = np.where(b >= threshold)[0]

        # For the main peak, find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1]

        improved = False

        # Try reducing top contributors
        for ci in top_idx[:8]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            j_comp = peak_k - i
            # Try reducing i (and possibly j_comp)
            for factor in [0.98, 0.96, 0.93, 0.90]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = project_simplex(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in top_idx[:5]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94]:
                    for f2 in [0.97, 0.94]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = project_simplex(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):

    except Exception:
        return float('inf')

            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 15:
                break

            # Create perturbation
            noise_scale = [0.02, 0.05, 0.01, 0.03][trial]
            a_pert = best_seq + noise_scale * rng.standard_normal(n_best)
            a_pert = project_simplex(a_pert)

            print(f"  trial={trial}, noise={noise_scale}, budget={t_sub:.0f}s")
            a_sub, _ = nesterov_logsubgrad(a_pert, t_sub * 0.5, lr=0.5, max_steps=50000)
            try_update(a_sub, f"pert_{trial}")

            rem = deadline - time.monotonic()
            if rem > 15 and t_sub * 0.3 > 10:
                a_sub2, _ = nesterov_lp_refine(a_sub, min(rem * 0.3, 40.0), n_max_iters=2000)
                try_update(a_sub2, f"pert_{trial}_lp")

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for delta in [-64, -32, -16, 16, 32, 64, 128]:
        alt_n = n_best + delta
        if 256 <= alt_n <= 4096:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_total = min(rem * 0.20, 100.0)
        if t_total < 15:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = nesterov_logsubgrad(init_alt, t_total * 0.6, lr=0.8, max_steps=50000)
        try_update(a_alt, f"ng_n{alt_n}")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_alt2, _ = nesterov_lp_refine(a_alt, min(rem * 0.3, 30.0), n_max_iters=1000)
            try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 90.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, n_max_iters=5000)
        try_update(a_fin, "final_lp")

    # === Phase 9: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.15, 30.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
