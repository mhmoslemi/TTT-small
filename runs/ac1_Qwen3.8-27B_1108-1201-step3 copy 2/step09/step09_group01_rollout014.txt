**Key idea: Use the structure of the problem more directly.**

The problem is: minimize `max_k c[k]` where `c = conv(a,a)`, `a >= 0`, `sum(a) = 1`.

This is equivalent to: find `a >= 0`, `sum(a) = 1` such that `a^T T_k a <= M` for all k, where `T_k` is the kth Toeplitz matrix, and minimize M.

This is a **quatically constrained program** (QCP). While non-convex in general, we can use:
- **Semidefinite programming (SDP) relaxation**: Lift to the space of rank-1 PSD matrices. But this is expensive for large n.
- **Successive convex approximation (SCA)**: At each iteration, linearize the quadratic constraints around the current point and solve the resulting convex problem.
- **Burer-Monteiro factorization**: Write `a*a^T = X` and optimize over X with a rank-1 constraint (dropped for relaxation).

For n=1024, SDP is too expensive. But SCA might work:

At iteration t with current point `a_t`:
- Linearize each constraint `a^T T_k a <= M` at `a_t`:
  `(a - a_t)^T T_k (a + a_t) + a_t^T T_k a_t <= M + ...`
  Actually, the linearization of `a^T T_k a` at `a_t` is `2 * a_t^T T_k (a - a_t) + a_t^T T_k a_t`.

  `f(a) = a^T T_k a`. Gradient: `∇f(a) = 2 T_k a`.
  Linearization at `a_t`: `f(a) ≈ f(a_t) + ∇f(a_t)^T (a - a_t) = a_t^T T_k a_t + 2 * a_t^T T_k (a - a_t)`.

  Constraint: `a_t^T T_k a_t + 2 * a_t^T T_k (a - a_t) <= M`.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    global _lp_solver_cached

    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_curr)
    c_curr = fast_convolve(a_curr)
    M_curr = float(c_curr.max())
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, M_var >= 0]
    
    # Build constraints: for each k, sum_j a_curr[k-j]*a[j] - M/2 <= c_curr[k]/2
    # We only need to add constraints for k where c_curr[k] is close to max
    # Actually, we should add ALL constraints for correctness, but for speed
    # we can add only the top few and the ones that might become active.
    
    # For correctness, add all constraints
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:

        # Coefficient of a[j] is a_curr[k-j]
        idx = np.arange(j_start, j_end + 1)
        coeffs = a_curr[k - idx]

        idx_m = idx[mask]

            return a_val, M_val
        
        active_ks.append(worst_k)
        if len(active_ks) > 2 * n - 1:

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def solve_sca_lp(a_curr):
    """Solve the SCA LP subproblem using scipy linprog."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(a_curr)
    c_curr = fast_convolve(a_curr)
    
    # Build sparse constraint matrix
    # For each k (0 to 2n-2): sum_{j in S_k} a_curr[k-j] * a[j] - 0.5*M <= 0.5*c_curr[k]
    # Variables: a[0],...,a[n-1], M (total n+1)
    
    row_list = []
    col_list = []
    data_list = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            coeff = a_curr[k - j]
            if coeff > 1e-15:
                row_list.append(k)
                col_list.append(j)
                data_list.append(coeff)
        # M column
        row_list.append(k)
        col_list.append(n)
        data_list.append(-0.5)
    
    A_ub = csr_matrix((data_list, (row_list, col_list)), shape=(2 * n - 1, n + 1))
    b_ub = 0.5 * c_curr
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='hihps')

        return None, None, c_curr
    
    a_val = np.maximum(result.x[:n], 0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None, c_curr
    a_val = a_val / s
    M_val = float(result.x[n])
    return a_val, M_val, c_curr

def peak_redistribute(a, rng, strength=0.1):
    """Redistribute mass away from peak-contributing coordinates."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    
    # Find peak positions
    peak_threshold = M * 0.9
    peak_ks = np.where(c >= peak_threshold)[0]
    
    # Compute "pressure" on each coordinate
    pressure = np.zeros(n)
    for k in peak_ks:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            partner = k - j
            product = a[j] * a[partner]
            pressure[j] += product
            if partner != j:
                pressure[partner] += product * 0.5
    
    # Normalize
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    
    # Create new sequence: reduce high-pressure coords, increase low-pressure
    a_new = a * (1.0 - strength * pressure)
    a_new += strength * (1.0 / n)  # Add uniform mass
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
