"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive multi-stage L-BFGS
with log-sum-exp smoothing, LP-based best response, and multi-length exploration.
Key improvements: very low final temperatures, more iterations at low T,
multiple initializations, and focused time on the most promising candidates.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def make_lse_func(n, T_val):
    """Create log-sum-exp objective function."""
    def func(z_in):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp
    return func


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def make_lse_jac(n, T_val):
    """Create log-sum-exp Jacobian."""
    def jac(z_in):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad of sum_k w[k] * (a*a)[k] with respect to a
        # d/d a_i [(a*a)[k]] = a[k-i] (for valid indices)
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        # Chain rule through softmax: d z -> d a = a * (grad_a - sum(a*grad_a))
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    return jac


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.01, maxiter=30000, n_stages=None):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    if n_stages is None:
        n_stages = 15
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    # Weight iterations more heavily toward low temperatures
    # Use more iterations at lower temperatures
    total_iters = maxiter
    # Allocate iterations proportional to 1/T (more at low T)
    inv_T = 1.0 / temps
    iter_weights = inv_T / inv_T.sum()
    iters_per_stage = np.maximum(20, (iter_weights * total_iters).astype(int))
    
    for i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        n_iter = iters_per_stage[i]
        if remaining < 10:
            n_iter = min(n_iter, 50)
        
        try:
            res = minimize(
                fun=make_lse_func(n, stage_T),
                x0=z,
                jac=make_lse_jac(n, stage_T),
                method='L-BFGS-B',
                options={
                    'maxiter': n_iter,
                    'ftol': 1e-20,
                    'gtol': 1e-16,
                    'maxfun': n_iter * 500,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:

    T = build_conv_matrix(a_fixed)

        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

        sy = y.sum()
        if sy < 1e-15:

        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))

        if sa < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, n, T):
    """Log-sum-exp smoothed objective for given n and temperature T."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_gradient(z, n, T):
    """Gradient of log-sum-exp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:

        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)

        grad_a[m_start:m_end + 1] += 2.0 * wk * a[k - m_end:k - m_start + 1:-1]
    S = np.dot(grad_a, a)

def lbgfs_refine(a0, time_limit, T_start, T_end, maxiter, n_stages):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    # Allocate more iterations to lower temperatures
    inv_T = 1.0 / temps
    iter_weights = inv_T / inv_T.sum()
    iters_per_stage = np.maximum(30, (iter_weights * maxiter).astype(int))

    for i in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break
        stage_T = temps[i]
        n_iter = iters_per_stage[i]
        if remaining < 15:
            n_iter = min(n_iter, 100)
        try:
            res = minimize(
                fun=lse_objective,
                x0=z,
                jac=lse_gradient,
                args=(n, stage_T),
                method='L-BFGS-B',
                options={
                    'maxiter': n_iter,
                    'ftol': 1e-22,
                    'gtol': 1e-18,
                    'maxfun': n_iter * 1000,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if it % 200 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 5:
                rng_local = np.random.default_rng(777 + it)
                noise = rng_local.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, mode, rng):
    """Create structured initial sequence for given mode."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif mode == 6:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 - np.abs(x - n / 2) / (n / 2 + 1)
    elif mode == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.5) ** 4)
    elif mode == 8:
        # Raised cosine squared
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 3
    elif mode == 9:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def subgradient_polish(a0, time_limit, n_steps):
    """Direct subgradient method on the non-smooth objective as final polish."""
    n = len(a0)
    a = norm_sequence(a0).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        # Subgradient of max_k b[k] is the indicator of argmax set
        # grad_a[m] = 2*a[peak_k - m] for valid m
        m_start = max(0, peak_k - n + 1)
        m_end = min(n - 1, peak_k)
        grad = np.zeros(n)
        grad[m_start:m_end + 1] = 2.0 * a[peak_k - m_end:peak_k - m_start + 1:-1]
        # Projected gradient step with diminishing step size
        step_size = 0.05 / (1.0 + 0.01 * step)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-16:
            a = a_new
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: aggressive L-BFGS with log-sum-exp annealing
    at very low temperatures, LP-based global moves, multi-length exploration,
    and perturbation escape.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 960
        best_seq = structured_init(n_init, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Starting: n={n_best}, best={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Aggressive L-BFGS on parent (main workhorse) ===
    print(f"\n=== Phase 1: L-BFGS T=100->0.005, n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.45
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, 20 stages, maxiter=50000")
        a1 = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.005,
                          maxiter=50000, n_stages=20)
        improved1 = try_update(a1, "lbgfs_phase1")

    # === Phase 2: Second L-BFGS pass at even lower T ===
    print(f"\n=== Phase 2: L-BFGS T=5->0.003 polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.2
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, 12 stages, maxiter=30000")
        a2 = lbgfs_refine(best_seq, t_alloc, T_start=5.0, T_end=0.003,
                          maxiter=30000, n_stages=12)
        improved2 = try_update(a2, "lbgfs_phase2")

    # === Phase 3: LP-based global moves ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.12
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=30000)
        try_update(a3, "nesterov_lp")

    # === Phase 4: Post-LP L-BFGS refinement ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.15, 200.0)
        print(f"\n=== Phase 4: Post-LP L-BFGS (best={best_score:.6f}) ===")
        print(f"  Budget: {t_alloc:.0f}s")
        a4 = lbgfs_refine(best_seq, t_alloc, T_start=20.0, T_end=0.005,
                          maxiter=20000, n_stages=12)
        try_update(a4, "post_lp_lbgfs")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.125, 1.25, 1.5, 0.875, 1.375]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    for fixed_n in [768, 1024, 1152, 1280, 1536]:
        if 256 <= fixed_n <= 4096 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 90:
            break
        t_total = min(rem * 0.2, 250.0)
        if t_total < 40:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick L-BFGS
        t_lb = t_total * 0.7
        a_alt = lbgfs_refine(init_alt, t_lb, T_start=30.0, T_end=0.01,
                             maxiter=15000, n_stages=10)
        try_update(a_alt, f"lbgfs_n{alt_n}")
        # Subgradient polish
        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            a_sg, _ = subgradient_polish(best_seq, min(rem * 0.3, 40.0), n_steps=5000)
            try_update(a_sg, f"subgrad_n{alt_n}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        for scale in [0.02, 0.05, 0.1]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)
            t_sub = min(rem * 0.35, 120.0)
            if t_sub < 20:
                break
            a_pert = lbgfs_refine(perturbed, t_sub, T_start=10.0, T_end=0.01,
                                  maxiter=15000, n_stages=10)
            try_update(a_pert, f"pert_{scale}")

    # === Phase 7: Final ultra-low-T polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 100.0)
        if t_fin > 15:
            a_fin = lbgfs_refine(best_seq, t_fin, T_start=1.0, T_end=0.002,
                                 maxiter=20000, n_stages=10)
            try_update(a_fin, "final_ultra_low_T")

    # Subgradient final polish
    rem = deadline - time.monotonic()
    if rem > 20:
        a_sg, _ = subgradient_polish(best_seq, min(rem * 0.5, 50.0), n_steps=10000)
        try_update(a_sg, "final_subgrad")

    # One more L-BFGS at the very end
    rem = deadline - time.monotonic()
    if rem > 20:
        a_last = lbgfs_refine(best_seq, min(rem * 0.6, 60.0),
                              T_start=0.5, T_end=0.001,
                              maxiter=10000, n_stages=8)
        try_update(a_last, "last_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")

    if best_seq is None:
        # Fallback
        best_seq = np.ones(256) / 256.0
        best_score = evaluate_safe(best_seq)
        print(f"FALLBACK: score={best_score:.6f}")

    return [float(x) for x in best_seq]
