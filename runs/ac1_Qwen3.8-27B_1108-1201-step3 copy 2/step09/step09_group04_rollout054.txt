    n = len(a0)
    L = 1 << (2*n-1).bit_length()
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    
    # Target: flat autocorrelation
    box = np.zeros(L)
    box[:2*n-1] = 1.0/(2*n-1)

    target_mag = np.maximum(target_mag, 1e-15)
    
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = rfft(a_padded)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-15)
        phase = A / mag
        A_new = phase * target_mag
        a_new = irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if it % 5000 == 0:
            print(f"    GS iter={it}: score={best_score:.6f}")

    """Compute correlate(w, a, 'valid') via FFT. len(w) = 2n-1, len(a) = n."""
    n = len(a)
    size = len(w) + n - 1  # = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a[::-1], fft_size)  # flip for correlation
    result = irfft(W * A, fft_size)[:len(w)]  # full correlation

    # Valid has length len(w) - n + 1 = n
    # Valid starts at index n-1
    return result[n-1:n-1+n]

Spectral-Gradient Optimization for Minimizing 2n*max(convolve(a,a))/(sum(a))^2.
Combines: (1) GS spectral initialization, (2) LogSumExp L-BFGS with adaptive
temperature, (3) top-k peak gradient projected descent, (4) coordinate refinement,
(5) multi-length exploration. Key innovations: top-k hard-max gradient, adaptive
beta scheduling, and spectral multi-start for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n*max(conv)/sum^2 for non-negative a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value_and_weights(a, beta):
    """Compute LSE value and softmax weights of convolution."""
    c = fast_convolve(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Numerically stable logsumexp
    max_shifted = shifted.max()
    log_sum_exp = c_max + max_shifted / beta + np.log(np.sum(np.exp(shifted - max_shifted))) / beta
    weights = np.exp(shifted - max_shifted) / np.sum(np.exp(shifted - max_shifted))
    return 2.0 * len(a) * log_sum_exp, weights, c


def topk_weights(c, k):
    """Get weights for top-k peaks (uniform over top-k)."""
    if k >= len(c):
        return np.ones(len(c)) / len(c)
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    return w


def gradient_topk(a, k):
    """Gradient of 2n*max_k c[k] approximated by top-k average, w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    w = topk_weights(c, k)
    # d(c[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n
    # sum_k w[k] * 2*a[k-i] = 2 * correlate(w, a, 'valid')[i]
    corr = np.correlate(w, a, mode='valid')
    return 4.0 * n * corr


def lse_gradient_a(a, beta):
    """Gradient of 2n*LSE(beta*c) w.r.t. a."""
    n = len(a)
    _, weights, c = lse_value_and_weights(a, beta)
    corr = np.correlate(weights, a, mode='valid')
    return 4.0 * n * corr


def lse_func_z(z, beta):
    """LSE objective in log-domain (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    val, _, _ = lse_value_and_weights(a, beta)
    return val


def lse_jac_z(z, beta):
    """LSE gradient in log-domain."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    _, weights, c = lse_value_and_weights(a, beta)
    corr = np.correlate(weights, a, mode='valid')
    g_a = 4.0 * n * corr
    return a * (g_a - np.dot(g_a, a))


def lse_lbfgs(a0, time_limit, beta_start=10.0, beta_end=500.0, n_stages=8):
    """L-BFGS with adaptive LSE temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    betas = np.geomspace(beta_start, beta_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for stag_i, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return lse_func_z(z_val, beta)

        def jac(z_val):
            return lse_jac_z(z_val, beta)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        elapsed = time.monotonic() - t0
        if stag_i % 2 == 0:
            print(f"    LSE-LBFGS beta={beta:.1f} ({elapsed:.0f}s): score={best_score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def topk_projected_descent(a0, time_limit, k=3, n_iters=100000):
    """Projected gradient descent using top-k peak gradient."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Adaptive step size
    step = 0.1
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = gradient_topk(a, k)
        # Project gradient: remove component along (1,...,1) for sum constraint
        g = g - np.mean(g)
        
        # Backtracking line search
        for _ in range(20):
            a_trial = a - step * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / s
            sc = compute_score(a_trial)
            if sc < best_score * (1 + 1e-10):
                a = a_trial
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                break
            step *= 0.5
        else:
            step *= 0.5
            if step < 1e-10:
                # Try increasing step (stuck)
                step = 0.01
                continue
        
        if it % 5000 == 0:
            print(f"    TopK-PD iter={it} k={k} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def gs_spectral(a0, time_limit, n_iters=30000):
    """Gerchberg-Saxton spectral iteration for flat autocorrelation."""
    n = len(a0)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()

    box = np.zeros(L)
    box[:2 * n - 1] = 1.0 / (2 * n - 1)
    target_mag = np.abs(rfft(box))
    target_mag = np.maximum(target_mag, 1e-15)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = rfft(a_padded)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-15)
        phase = A / mag
        A_new = phase * target_mag
        a_new = irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0 and it > 0:
            print(f"    GS iter={it} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_refinement(a0, time_limit, n_steps=50000):
    """Aggressive coordinate descent on top peak contributors."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find top 3 peaks (excluding endpoints which are a[0]^2 and a[-1]^2)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        peak_idx = np.argsort(c_work)[::-1][:3]
        peak_val = c_work[peak_idx[0]]

        improved = False
        # For each peak, find top contributors and try reducing them
        for pk in peak_idx:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            top_i = idx[np.argmax(contribs)]
            
            # Try reducing top_i and redistributing
            for factor in [0.95, 0.90, 0.85, 0.80, 0.70]:
                a_trial = a.copy()
                diff = a_trial[top_i] * (1 - factor)
                if diff < 1e-15:
                    continue
                a_trial[top_i] -= diff
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Also try pair adjustments
        if not improved and len(peak_idx) > 0:
            pk = peak_idx[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top2 = np.argsort(contribs)[::-1][:4]
            for ci in top2:
                i = idx[ci]
                j = pk - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.85, 0.78]:
                    for f2 in [0.92, 0.85, 0.78]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        s = a_trial.sum()
                        if s < 0.01:
                            continue
                        a_trial /= s
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation every 500 steps
            if step % 500 == 0 and step > 0:
                rng = np.random.default_rng(step)
                n_pert = rng.integers(1, 8)
                indices = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx_p in indices:
                    a_trial[idx_p] *= (1.0 + rng.uniform(-0.08, 0.08))
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial /= s
                    sc = compute_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            if not improved:
                break

        if step % 10000 == 0 and step > 0:
            print(f"    CoordRef step={step} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, variant=0):
    """Create structured initial sequence."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x / 2)
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        # U-shaped: high at edges
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
    elif variant == 6:
        # Asymmetric: front-loaded
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update best state."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def pnorm_lbfgs(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with p-norm smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()
    t0 = time.monotonic()

    def pnorm_func(z_val, p_val):
        z_max = z_val.max()
        exp_z = np.exp(z_val - z_max)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        c_pos = np.maximum(c, 1e-300)
        val = (np.sum(c_pos ** p_val)) ** (1.0 / p_val)
        return 2.0 * n * val

    def pnorm_jac(z_val, p_val):
        z_max = z_val.max()
        exp_z = np.exp(z_val - z_max)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        c_pos = np.maximum(c, 1e-300)
        p_pow = c_pos ** p_val
        S = np.sum(p_pow)
        if S < 1e-300:
            return np.zeros_like(z_val)
        factor = S ** (1.0 / p_val - 1.0)
        w = c_pos ** (p_val - 1.0)
        corr = np.correlate(w, a, mode='valid')
        g_a = 4.0 * n * factor * corr
        return a * (g_a - np.dot(g_a, a))

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(80, 6000 // n_stages)

    for p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break

        def func(z_val):
            return pnorm_func(z_val, p)

        def jac(z_val):
            return pnorm_jac(z_val, p)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: GS spectral init, LSE/p-norm L-BFGS,
    top-k projected descent, coordinate refinement, multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_state = {'score': float('inf'), 'seq': None}

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=2)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])
    print(f"Starting optimization from n={n_best}, score={best_state['score']:.6f}")

    # === Phase 1: GS Spectral + P-norm L-BFGS (quick, 120s) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 100.0)
    if t1 > 20:
        print(f"\n=== Phase 1: GS + P-norm ({t1:.0f}s) ===")
        a_gs, _ = gs_spectral(best_state['seq'], min(t1 * 0.4, 40.0), n_iters=20000)
        try_update(best_state, a_gs, "gs_spectral")

        rem = deadline - time.monotonic()
        t1b = min(rem * 0.06, 80.0)
        if t1b > 15:
            a_pn = pnorm_lbfgs(best_state['seq'], t1b, p_start=15.0, p_end=200.0, n_stages=6)
            try_update(best_state, a_pn, "pnorm_1")

    # === Phase 2: LSE L-BFGS (main, 450s) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.28, 450.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LSE L-BFGS ({t2:.0f}s) ===")
        a_lse = lse_lbfgs(best_state['seq'], t2, beta_start=8.0, beta_end=400.0, n_stages=10)
        try_update(best_state, a_lse, "lse_lbfgs_1")

    # === Phase 3: Top-k Projected Descent (200s) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.13, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: TopK PD ({t3:.0f}s) ===")
        for k in [1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t3k = min(rem * 0.25, 60.0)
            if t3k < 5:
                continue
            a_k, _ = topk_projected_descent(best_state['seq'], t3k, k=k, n_iters=50000)
            try_update(best_state, a_k, f"topk_{k}")

    # === Phase 4: LSE L-BFGS round 2 (higher beta, 250s) ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LSE L-BFGS r2 ({t4:.0f}s) ===")
        a_lse2 = lse_lbfgs(best_state['seq'], t4, beta_start=100.0, beta_end=2000.0, n_stages=7)
        try_update(best_state, a_lse2, "lse_lbfgs_2")

    # === Phase 5: Coordinate refinement (150s) ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 10:
        print(f"\n=== Phase 5: CoordRefine ({t5:.0f}s) ===")
        a_cr, _ = coordinate_refinement(best_state['seq'], t5, n_steps=80000)
        try_update(best_state, a_cr, "coord_refine")

    # === Phase 6: Multi-length exploration (300s) ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 350.0)
    if t6 > 50:
        print(f"\n=== Phase 6: Multi-length ({t6:.0f}s) ===")
        n_cur = len(best_state['seq'])
        lengths = sorted(set([
            n_cur - 64, n_cur - 32, n_cur + 32, n_cur + 64,
            n_cur - 128, n_cur + 128,
            832, 896, 960, 1024, 1088, 1152
        ]))
        lengths = [l for l in lengths if 512 <= l <= 2048 and l != n_cur]
        lengths.sort(key=lambda x: abs(x - n_cur))

        n_explored = 0
        for alt_n in lengths[:6]:
            rem = deadline - time.monotonic()
            if rem < 40 or n_explored >= 4:
                break
            n_explored += 1
            t_alt = min(rem * 0.20, 120.0)
            if t_alt < 30:
                break
            print(f"  Exploring n={alt_n} ({t_alt:.0f}s)")

            init_alt = interpolate_seq(best_state['seq'], alt_n)

            # Quick optimization
            a_alt = lse_lbfgs(init_alt, min(t_alt * 0.5, 60.0),
                              beta_start=20.0, beta_end=300.0, n_stages=5)
            improved = try_update(best_state, a_alt, f"lse_n{alt_n}")

            if improved:
                rem = deadline - time.monotonic()
                if rem > 20:
                    a_alt2 = pnorm_lbfgs(best_state['seq'], min(rem * 0.15, 50.0),
                                         p_start=50.0, p_end=400.0, n_stages=4)
                    try_update(best_state, a_alt2, f"pnorm_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization (200s) ===
    rem = deadline - time.monotonic()
    if rem > 80:
        print(f"\n=== Phase 7: Perturbation ({rem:.0f}s remaining) ===")
        n_cur = len(best_state['seq'])
        for noise_scale in [0.02, 0.05, 0.10, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.25, 50.0)
            a_pert = lse_lbfgs(perturbed, t_sub, beta_start=30.0, beta_end=300.0, n_stages=5)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

            # Also try topk on perturbed
            rem = deadline - time.monotonic()
            if rem > 15:
                a_pert2, _ = topk_projected_descent(perturbed, min(rem * 0.15, 25.0), k=2, n_iters=20000)
                try_update(best_state, a_pert2, f"pert_topk_{noise_scale}")

    # === Phase 8: Final polish (100s) ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 8: Final Polish ({rem:.0f}s) ===")
        t_fin = min(rem * 0.4, 60.0)
        a_fin = lse_lbfgs(best_state['seq'], t_fin, beta_start=300.0, beta_end=3000.0, n_stages=4)
        try_update(best_state, a_fin, "final_lse")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin2, _ = coordinate_refinement(best_state['seq'], min(rem * 0.3, 30.0), n_steps=30000)
            try_update(best_state, a_fin2, "final_coord")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin3, _ = topk_projected_descent(best_state['seq'], min(rem * 0.3, 20.0), k=1, n_iters=20000)
            try_update(best_state, a_fin3, "_final_topk1")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
