def fast_convolve(a):

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            # (the change in conv[k] from delta is delta*(a[k-i] + a[i]) for k != 2i)
            # For k = 2i, there's an additional delta^2 term
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai  # a[k-i] + a[i]
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Generate candidate deltas
            candidates = [0.0, -ai]

            # Where individual lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / (c)
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            # Where top affected lines cross each other
            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            # Fractional moves
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.5 * ai)

            # Evaluate candidates
            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                # Apply update
                a[i] = max(ai + best_delta, 0.0)
                # Update conv
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        # Periodically recompute to avoid drift
        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 300:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in [0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 1.001, 1.002, 1.005, 1.01, 1.015, 1.02]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 30:
            break

    return best_a, best_score

def nesterov_smoothed_md(a0, time_limit, beta_init=20.0, lr_init=0.01, label=""):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov momentum
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y_proj = simplex_project(np.maximum(y, 0))

        # Compute smooth gradient via log-sum-exp
        conv_y = fast_convolve(y_proj)
        max_c = conv_y.max()
        scaled = beta * (conv_y - max_c)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient: grad[i] = 2 * sum_k weights[k] * y_proj[k-i] (for valid k-i)
        grad = np.zeros(n)
        active_k = np.where(weights > 1e-10)[0]
        for k in active_k:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 80:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            if lr < 1e-10:
                break
        elif it % 30 == 0:
            beta *= 1.01

        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve best response
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        b = b_new

        # Adaptive damping
        alpha = min(1.0, 10.0 / (iters + 5))
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Also try the LP solution directly
        score_b = 2.0 * n * fast_convolve(b).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

    return best_a, best_score, iters

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.2):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 5):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.2))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient method with Nesterov acceleration on the exact (non-smooth) objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        # Nesterov lookahead
        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        # Average over near-peak indices for a more stable subgradient
        threshold = M - 0.01 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        # Gradient step + projection
        a_new = a - step * grad
        a_new = simplex_project(a_new)

        # Nesterov average (for convergence guarantee)
        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        # Evaluate
        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Smart 1D line-search coordinate descent
    2. LP fictitious play with adaptive damping
    3. Nesterov smoothed mirror descent
    4. Subgradient with Nesterov acceleration
    5. Multi-length exploration
    6. Random restarts with structured perturbation
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(12345)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        # Fallback initialization
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = 2.0 * n_init
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    # === Phase 1: Smart coordinate descent on parent ===
    print(f"\n=== Phase 1: Smart CD (best={best_score:.6f}) ===")
    t1 = min(200.0, (deadline - time.monotonic()) * 0.15)
    print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
    a1, s1, sw1 = smart_coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_scd")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")

    # === Phase 3: Nesterov Smoothed Mirror Descent ===
    print(f"\n=== Phase 3: Nesterov MD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t3 = min(max(rem * 0.25, 30), 250.0)
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a3, s3 = nesterov_smoothed_md(best_seq, t3, beta_init=30.0, lr_init=0.008, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_nmd")

    # === Phase 4: Subgradient with Nesterov ===
    print(f"\n=== Phase 4: Subgradient Nesterov (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t4 = min(max(rem * 0.2, 30), 200.0)
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = subgradient_nesterov(best_seq, t4, label="p4")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_subgrad")

    # === Phase 5: Another round of smart CD + perturbation loop ===
    print(f"\n=== Phase 5: CD + Perturb Loop (best={best_score:.6f}) ===")
    if best_seq is not None:
        for rnd in range(4):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t5 = min(rem * 0.2, 100.0)
            a5, s5, _ = smart_coord_descent(best_seq, t5, label=f"p5r{rnd}")
            best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_scd_r{rnd}")

            # Perturb to escape
            if rnd < 3:
                if rnd % 2 == 0:
                    perturbed = peak_targeted_perturb(best_seq, rng, strength=0.03 + 0.01 * rnd)
                else:
                    perturbed = random_restructure(best_seq, rng, frac=0.1 + 0.1 * rnd)
                # Quick optimize the perturbed version
                t5b = min(rem * 0.1, 40.0)
                a5b, s5b, _ = smart_coord_descent(perturbed, t5b, label=f"p5p_r{rnd}")
                best_seq, best_score = check_and_update(a5b, best_seq, best_score, f"p5_pert_r{rnd}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best]
        for alt in [768, 1280, 1536, 2048]:
            if alt != n_best and 512 <= alt <= 2560:
                alt_lengths.append(alt)
        alt_lengths = alt_lengths[:4]

        for n_alt in alt_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 30:

            t6 = min(rem * 0.3, 80.0)
            if n_alt == n_best:

            a6, s6, _ = smart_coord_descent(init_alt, t6, label=f"p6_n{n_alt}")
            best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_n{n_alt}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t7 = min(rem * 0.5, 120.0)
            a7, s7 = fine_coord_descent(best_seq, t7, label="p7")
            best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_fine")

        # Another smart CD pass
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t7b = min(rem * 0.6, 60.0)
            a7b, s7b, _ = smart_coord_descent(best_seq, t7b, label="p7b")
            best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b_scd")

        # One more fine pass
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a7c, s7c = fine_coord_descent(best_seq, rem * 0.7, label="p7c")
            best_seq, best_score = check_and_update(a7c, best_seq, best_score, "p7c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
