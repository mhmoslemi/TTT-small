from scipy.fft import rfft, irfft
import scipy.sparse as sp
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build sparse matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    # T has shape (2n-1, n)

    rows = []
    cols = []

                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    return b_val / s

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

        # Damped update

        if s_new < 1e-15:

        a = a_new / s_new

        if M_direct < best_M - tol:
            best_M = M_direct
            best_a = b.copy()
            a = b.copy()

        # Check convergence
        if abs(M_b - best_M) < tol * best_M and it > 10:
            break
    
    return best_a, 2.0 * n * best_M

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact LP-based alternating minimization (fast convergence via exact best response)
2. Multi-length search to exploit larger-n theoretical improvements
3. Iterated local search with random restarts to escape local optima
4. Multi-peak subgradient descent for smooth refinement
5. Fast coordinate descent with incremental convolution updates for final polishing
"""

import numpy as np
from scipy.fft import rfft, irfft
import scipy.sparse as sp
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = conv(a, b). Shape: (2n-1, n)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        start_k = j
        end_k = min(j + n, 2 * n - 1)
        ks = np.arange(start_k, end_k)
        rows.extend(ks.tolist())
        cols.extend([j] * len(ks))
        vals.extend(a[ks - j].tolist())
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def find_lp_solver():
    """Find a working LP solver."""
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    _cached_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS, cp.SCTRL, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                print(f"  Using LP solver: {solver_name}")
                return _cached_solver
        except Exception:
            continue
    _cached_solver = cp.ECOS
    print(f"  Fallback LP solver: {_cached_solver}")
    return _cached_solver


def lp_best_response(a, solver, top_k=None):
    """
    Solve LP: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0.
    If top_k is specified, only use the top_k largest constraints (based on conv(a,a) peaks).
    """
    n = len(a)
    T = build_conv_matrix_sparse(a)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if top_k is not None and top_k < 2 * n - 1:
        # Use only top-k constraints
        conv_aa = fast_convolve(a)
        peak_indices = np.argpartition(conv_aa, -top_k)[-top_k:]
        constraints = [T[peak_indices] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def alternating_lp(a_init, time_limit, solver, verbose=True, use_approx=False, top_k=50):
    """
    Alternating LP minimization: fix one side, solve LP for the other.
    Much faster convergence than Frank-Wolfe.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    cur_score = 2.0 * n * best_M

    t0 = time.monotonic()
    it = 0
    prev_M = best_M

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: find best b given a
        if use_approx:
            b = lp_best_response(a, solver, top_k=top_k)
        else:
            b = lp_best_response(a, solver)

        if b is None:
            break

        M_b = float(fast_convolve(b).max())
        score_b = 2.0 * n * M_b

        if M_b < best_M:
            best_M = M_b
            best_a = b.copy()

        # Damped update with decreasing step
        alpha = 2.0 / (it + 2.0)
        # Also try full step
        if it > 5:
            a = b.copy()
        else:
            a_new = (1 - alpha) * a + alpha * b
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a = b.copy()
            else:
                a = a_new / s_new

        # Check convergence
        if abs(prev_M - best_M) < 1e-10 * max(best_M, 1e-10) and it > 20:
            if verbose:
                print(f"    ALP converged at it={it}")
            break

        prev_M = best_M

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALP it={it} ({elapsed:.0f}s): best_score={2*n*best_M:.6f}, M_b={M_b:.10f}")

    return best_a, 2.0 * n * best_M


def subgradient_descent(a_init, time_limit, lr_init=0.05, top_k=50, verbose=True):
    """
    Subgradient descent with multi-peak averaging and Polyak step size.
    Uses the subgradient of max_k conv(a,a)_k w.r.t. a.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_a = a.copy()
    best_M = float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    best_iter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())
        score = 2.0 * n * M

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            best_iter = it
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act == 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]
            temp = 0.01 * max(M, 1e-10)
            w = np.exp((pv - pv.max()) / temp)
            weights = w / w.sum()
            peaks = peaks[np.argsort(weights)[::-1]]
            weights = weights[np.argsort(weights)[::-1]]

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if weights[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # sg[i] += w * 2 * a[k-i] (the gradient of conv[k] w.r.t. a[i])
            # conv[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] + a[i]*a[k-i]... 
            # Actually d conv[k] / d a[i] = a[k-i] (if k-i in range) + a[i] (if i in range and k-i=i)
            # More carefully: conv[k] = sum_j a[j] * a[k-j]
            # d/d a[i]: a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i term)
            # But if k-i == i (i.e., k=2i), it's 2*a[i]
            # Actually: d/d a[i] sum_j a[j]*a[k-j] = a[k-i] + a[k-i] = 2*a[k-i] if both i and k-i are in [0,n-1]
            # Wait no. conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # d/d a[i]: the terms where j=i gives a[k-i], and where k-j=i (j=k-i) gives a[i]
            # So d conv[k] / d a[i] = a[k-i] + a[i] if k-i != i, or 2*a[i] if k-i == i
            # But since a[k-i] appears when j=i AND a[i] appears when j=k-i:
            # If i != k-i: derivative = a[k-i] + a[k-i] = 2*a[k-i]... no wait
            # d/d a[i] of a[j]*a[k-j]:
            #   = a[k-j] if j=i (i.e., the term a[i]*a[k-i])
            #   + a[j] if k-j=i (i.e., j=k-i, the term a[k-i]*a[i])
            # So if i != k-i: derivative = a[k-i] + a[k-i] = 2*a[k-i]
            # If i == k-i (k=2i): derivative = a[i] + a[i] = 2*a[i]
            # In both cases: derivative = 2*a[k-i] (when k-i is in range)
            # Wait, that's only if both i and k-i are valid indices.
            # If i is valid but k-i is not in [0,n-1], then only the "j=i" term contributes: a[k-i] doesn't exist
            # Actually if k-i is not in [0,n-1], then the term with j=i doesn't exist in the sum, so derivative is 0.
            
            if 0 <= k - i_end <= n - 1 and 0 <= k - i_start <= n - 1:
                a_contrib = a[k - i_end:k - i_start + 1][::-1]
                sg[i_start:i_end + 1] += weights[idx] * 2.0 * a_contrib

        # Projected gradient step (Euclidean projection onto simplex)
        a_new = a - lr * sg
        # Project onto {x >= 0, sum(x) = 1}
        a_new = simplex_project(a_new)

        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): score={2*n*M:.6f}, best={2*n*best_M:.6f}, lr={lr:.6f}")

    return best_a, 2.0 * n * best_M


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = np.nonzero(cond)[0][-1]
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def coord_descent(a_init, time_limit, verbose=True, label="", n_steps=2000):
    """
    Fast coordinate descent with O(n) incremental convolution updates.
    For each coordinate i, try multiple step sizes and pick the best.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    # Precompute the range of affected conv indices for each i
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            k_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Affected part: conv[k] += delta * (a[k-i] + a[k-i]) = delta * 2*a[k-i]
            # Plus quadratic correction at k=2i: += delta^2
            a_shift = a[k_start - i:k_end - i + 1]  # a[k-i] for k in [k_start, k_end]
            affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Try multiple step sizes
            base = a[i]
            for factor in [-0.5, -0.3, -0.1, -0.03, 0.03, 0.1, 0.3, 0.5, -0.7, 0.7]:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shift
                # Quadratic correction at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] += delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0:
                # Apply update
                old_val = a[i]
                a[i] = old_val + best_delta

                # Update conv incrementally
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): score={2*n*M:.6f}, best={best_score:.6f}")

    return best_a, best_score

def diverse_init(n, rng, method="random"):
    """Create diverse initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "cosine":
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(0.5, 4.0)
        a = (0.5 * (1 + np.cos(np.pi * np.clip(x, -1, 1)))) ** power
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    elif method == "two_hump":
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    elif method == "periodic":
        period = int(rng.integers(3, 12))
        a = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            a[i] = 0.5 + 0.4 * np.cos(2 * np.pi * phase + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    elif method == "fourier":
        # Random Fourier magnitude spectrum
        freqs = np.arange(n)
        mag = np.abs(rng.standard_normal(n)) + 0.1
        phase = rng.uniform(0, 2 * np.pi, n)
        spectrum = mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(spectrum))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    elif method == "noisy_uniform":
        a = np.ones(n) + rng.standard_normal(n) * rng.uniform(0.1, 0.5)
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n
    else:
        a = rng.random(n)
        return a / a.sum()


def peak_reducing_cd(a_init, time_limit, verbose=True):

    """

    if s < 1e-15:

    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())

    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(conv.max())

        k_peak = int(np.argmax(conv))

        i_end = min(n - 1, k_peak)

        # Pick the coordinate with the largest contribution
        if contributions.max() < 1e-20:
            break
            
        i_local = int(np.argmax(contributions))
        i = i_start + i_local
        
        # Try to reduce a[i]
        base = a[i]
        best_delta = 0.0
        best_new_M = M
        
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)

        if k_start > 0:
            unaff_max = max(unaff_max, float(conv[:k_start].max()))
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
        
        a_shift = a[k_start - i:k_end - i + 1]

        for factor in [-0.3, -0.15, -0.05, -0.02, 0.02, 0.05, 0.15, 0.3]:
            delta = base * factor
            new_val = a[i] + delta
            if new_val < -1e-15:
                continue
            new_affected = affected + 2.0 * delta * a_shift
            if 2 * i >= k_start and 2 * i <= k_end:
                idx = 2 * i - k_start
                new_affected = new_affected.copy()
                new_affected[idx] += delta * delta
            new_M = max(unaff_max, float(new_affected.max()))
            if new_M < best_new_M - 1e-15:
                best_new_M = new_M
                best_delta = delta
        
        if best_delta == 0.0:
            # Try a different coordinate (second largest contribution)
            if len(contributions) > 1:
                contributions[i_local] = -1
                i_local = int(np.argmax(contributions))
                i = i_start + i_local
                if contributions[i_local] < 1e-20:
                    break
            else:
                break
            
            # Redo for new i
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            a_shift = a[k_start - i:k_end - i + 1]
            affected = conv[k_start:k_end + 1]
            
            for factor in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                delta = base * factor
                if new_val := a[i] + delta < -1e-15:
                    continue
                new_affected = affected + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] += delta * delta
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta == 0.0:
                break
        
        # Apply
        a[i] += best_delta
        changes = 2.0 * best_delta * a_shift
        if 2 * i >= k_start and 2 * i <= k_end:
            idx = 2 * i - k_start
            changes = changes.copy()
            changes[idx] += best_delta * best_delta
        conv[k_start:k_end + 1] += changes
        
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD it={it} ({elapsed:.0f}s): score={2*n*M:.6f}, best={best_score:.6f}")

    return best_a, best_score


def iterated_local_search(a_init, time_limit, solver, rng, n_restarts=5, verbose=True):
    """
    Iterated local search: perturb the current best solution, run a short ALP,
    keep if improved.
    """
    n = len(a_init)
    best_a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        # Perturb
        a_pert = best_a.copy()
        # Random multiplicative perturbation on a subset of coordinates
        n_perturb = n // 4
        idx = rng.choice(n, size=n_perturb, replace=False)
        factors = 1.0 + rng.standard_normal(n_perturb) * 0.3
        a_pert[idx] *= factors
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s
        
        # Run short ALP
        t_alloc = min(remaining * 0.3, 120.0)
        a_res, s_res = alternating_lp(a_pert, t_alloc, solver, verbose=False)
        
        # Check score
        actual_score = 2.0 * n * fast_convolve(a_res).max()
        if actual_score < best_score:
            best_score = actual_score
            best_a = a_res.copy()
            if verbose:
                print(f"    ILS restart {r}: improved to {best_score:.6f}")
    
    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Load parent, evaluate
    2. ALP on parent (main workhorse)
    3. Try alternative lengths
    4. Subgradient descent refinement
    5. Iterated local search
    6. Coordinate descent polishing
    7. Fine polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(12345)

    # Find LP solver early
    solver = find_lp_solver()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = 2.0 * len(parent) * fast_convolve(parent).max()
    else:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = 2.0 * n_init * fast_convolve(best_seq).max()

    print(f"Starting score: {best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: ALP on parent ===
    n_parent = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alp = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: ALP on parent (n={n_parent}, budget={t_alp:.0f}s) ===")
    
    a_alp, s_alp = alternating_lp(best_seq, t_alp, solver, verbose=True)
    actual_s = 2.0 * n_parent * fast_convolve(a_alp).max()
    if actual_s < best_score:
        best_score = actual_s
        best_seq = a_alp.copy()
        print(f"  ALP improved: {best_score:.6f}")
    
    # === Phase 2: Try different lengths ===
    print(f"\n=== Phase 2: Multi-length search (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    
    lengths_to_try = [512, 1536, 2048, 3072]
    # Also consider slightly different length than parent
    if n_parent in [1024]:
        lengths_to_try = [512, 1536, 2048, 3072, 4096]
    
    for n_try in lengths_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        
        t_alloc = min(remaining * 0.2, 180.0)
        if t_alloc < 60:
            break
        
        # Create initialization by interpolating parent to new length
        if parent is not None:
            from numpy import interp
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n_try)
            a_interp = interp(x_new, x_old, parent)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s < 1e-10:
                a_interp = np.ones(n_try) / n_try
            else:
                a_interp = a_interp / s
        else:
            a_interp = np.ones(n_try) / n_try
        
        print(f"  Trying n={n_try} (budget={t_alloc:.0f}s)")
        a_res, s_res = alternating_lp(a_interp, t_alloc, solver, verbose=False)
        actual_s = 2.0 * n_try * fast_convolve(a_res).max()
        
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_res.copy()
            print(f"  NEW BEST at n={n_try}: {best_score:.6f}")
        else:
            print(f"  n={n_try}: score={actual_s:.6f} (no improvement)")

    # === Phase 3: Subgradient descent on best ===
    print(f"\n=== Phase 3: Subgradient descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 120:
        t_sg = min(remaining * 0.2, 200.0)
        print(f"  Budget: {t_sg:.0f}s")
        a_sg, s_sg = subgradient_descent(best_seq, t_sg, lr_init=0.02, top_k=50, verbose=True)
        actual_s = 2.0 * len(best_seq) * fast_convolve(a_sg).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_sg.copy()
            print(f"  SG improved: {best_score:.6f}")

    # === Phase 4: Iterated Local Search ===
    print(f"\n=== Phase 4: Iterated Local Search (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        t_ils = min(remaining * 0.25, 300.0)
        print(f"  Budget: {t_ils:.0f}s")
        a_ils, s_ils = iterated_local_search(best_seq, t_ils, solver, rng, n_restarts=5, verbose=True)
        actual_s = 2.0 * len(best_seq) * fast_convolve(a_ils).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_ils.copy()
            print(f"  ILS improved: {best_score:.6f}")

    # === Phase 5: Alternative ALP with different damping ===
    print(f"\n=== Phase 5: ALP with diverse inits (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    inits = [
        diverse_init(n_best, rng, "cosine"),
        diverse_init(n_best, rng, "fourier"),
        diverse_init(n_best, rng, "noisy_uniform"),
        diverse_init(n_best, rng, "two_hump"),
    ]
    
    # Also add perturbed parent
    pert = best_seq + rng.standard_normal(n_best) * 0.1
    pert = np.maximum(pert, 0)
    pert = pert / pert.sum()
    inits.append(pert)
    
    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_alloc = min((remaining - 100) * 0.25, 120.0)
        if t_alloc < 40:
            break
        
        print(f"  ALP init {idx} (budget={t_alloc:.0f}s)")
        a_res, s_res = alternating_lp(init, t_alloc, solver, verbose=False)
        actual_s = 2.0 * n_best * fast_convolve(a_res).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_res.copy()
            print(f"  NEW BEST from init {idx}: {best_score:.6f}")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_cd = min(remaining * 0.3, 300.0)
        print(f"  Budget: {t_cd:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_cd, verbose=True, label="main")
        actual_s = 2.0 * len(best_seq) * fast_convolve(a_cd).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_cd.copy()
            print(f"  CD improved: {best_score:.6f}")
        else:
            # Use the CD result anyway (it might have slightly different structure)
            best_seq = a_cd.copy()
            best_score = actual_s

    # === Phase 7: Fine Polishing ===
    print(f"\n=== Phase 7: Fine Polishing (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_fine = min(remaining * 0.4, 200.0)
        print(f"  Budget: {t_fine:.0f}s")
        a_fine, s_fine = fine_coord_descent(best_seq, t_fine, verbose=True)
        actual_s = 2.0 * len(best_seq) * fast_convolve(a_fine).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_fine.copy()
            print(f"  FineCD improved: {best_score:.6f}")

    # === Phase 8: Final peak-reducing CD ===
    print(f"\n=== Phase 8: Peak-Reducing CD (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_peak = min(remaining * 0.5, 100.0)
        a_pk, s_pk = peak_reducing_cd(best_seq, t_peak, verbose=True)
        actual_s = 2.0 * len(best_seq) * fast_convolve(a_pk).max()
        if actual_s < best_score:
            best_score = actual_s
            best_seq = a_pk.copy()

    # Final verification
    best_seq = np.maximum(best_seq, 0.0)
    s_final = best_seq.sum()
    if s_final < 1e-10:
        best_seq = np.ones(1024) / 1024
    else:
        best_seq = best_seq / s_final
    
    final_score = 2.0 * len(best_seq) * fast_convolve(best_seq).max()
    print(f"\nFinal computed score: {final_score:.6f} (n={len(best_seq)})")
    
    # Verify with evaluate_sequence if available
    try:
        eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
            best_score = eval_score
    except Exception as e:
        print(f"Verification failed: {e}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
