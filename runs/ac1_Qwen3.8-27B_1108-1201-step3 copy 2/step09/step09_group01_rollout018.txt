    best_score = 2.0 * n * float(conv.max())
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    prev_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        grad = smooth_max_gradient(conv, beta, a)

        # Multiplicative (exponential) update
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        conv = fast_convolve(a)
        cur_score = 2.0 * n * float(conv.max())

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()

        # Adaptive schedule
        if it % 50 == 0:
            beta *= 1.15
            lr *= 0.97
        if it % 200 == 0:
            # Occasionally reset lr to explore
            lr = max(0.01, lr * 1.5)

        if lr < 1e-12:
            break

    return best_a, best_score


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    LP fictitious play with Nesterov (Robbins-Monro) averaging.
    Alternately solves LP best-responses and averages the iterates.
    """
    t0 = time.monotonic()
    if solver is None:
        solver = get_lp_solver()

    if initial is None:
        a = np.ones(n) / n
        b = a.copy()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0).copy()
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        b = a.copy()

    # Nesterov sequents
    a_seq = a.copy()
    b_seq = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    iters = 0
    fails = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP: best response b to fixed a
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            fails += 1
            if fails > 10:
                break
            continue
        fails = 0

        # Solve LP: best response a to fixed b
        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            fails += 1
            if fails > 10:
                break
            continue
        fails = 0

        # Nesterov sequent update (gamma = 2/(t+2))
        gamma = 2.0 / (iters + 2.0)
        a_seq = (1.0 - gamma) * a_seq + gamma * a_new
        b_seq = (1.0 - gamma) * b_seq + gamma * b_new

        # Update running iterates
        a = a_new
        b = b_new

        # Evaluate candidates
        candidates = [a_new, b_new, a_seq, b_seq, 0.5 * (a_seq + b_seq)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * float(fast_convolve(cand).max())
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent for minimizing max auto-conv given sum=1.
    Vectorized peak tracking for efficiency.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # The affected part of the convolution
            # conv[k] += 2*delta*a[k-i] + delta^2 (if 2i == k)
            # For k in [i, i+n-1]: delta_a[k] = 2*delta*a[k-i]
            # Special: if k == 2i, extra delta^2

            a_part = a[0:n]  # a[k-i] for k = i..i+n-1 means a[0]..a[n-1]

            best_delta = 0.0
            best_new_max = M

            # Try several step sizes
            factors = [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3]
            for f in factors:
                delta = (f - 1.0) * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue

                # New affected conv values
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_part
                # Fix the self-contribution at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max_val = max(unaff_max, float(new_aff.max()))
                if new_max_val < best_new_max - 1e-15:
                    best_new_max = new_max_val
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)

                # Update conv
                changes = 2.0 * best_delta * a_part
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:

            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_part = a[0:n]
            best_delta = 0.0
            best_new_max = M

            factors = [0.995, 0.99, 0.985, 0.98, 0.97, 0.992, 0.998,
                      1.002, 1.005, 1.01, 1.015, 1.02]
            for f in factors:
                delta = (f - 1.0) * a[i]
                if a[i] + delta < -1e-15:
                    continue

                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_part
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max_val = max(unaff_max, float(new_aff.max()))
                if new_max_val < best_new_max - 1e-15:
                    best_new_max = new_max_val
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_part
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def shift_interpolate(f, new_n, shift_frac):
    """Interpolate with a circular shift to create a non-symmetric variant."""
    old_n = len(f)
    f_shifted = np.roll(f, int(shift_frac * old_n))
    return interpolate_seq(f_shifted, new_n)


def perturb_for_escape(a0, rng, strength=0.05):
    """Structured perturbation to escape local optima."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = float(conv.max())

    # Pressure: positions contributing most to peaks get reduced
    peak_threshold = M * 0.9
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: MWU from multiple diverse initializations
    Phase 2: LP fictitious play (main workhorse)
    Phase 3: Coordinate descent polishing
    Phase 4: Alternative lengths
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(123)
    solver = get_lp_solver()

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: MWU from multiple initializations ===
    print("\n=== Phase 1: MWU Optimization ===")
    t1_start = time.monotonic()

    n_values = [1024, 1152, 768, 896, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values[:5]:
        elapsed = time.monotonic() - t1_start
        if elapsed > 200:
            break

        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(("parent", interpolate_seq(parent, n)))
            inits.append(("shift01", shift_interpolate(parent, n, 0.1)))
            inits.append(("shift03", shift_interpolate(parent, n, 0.3)))
        inits.append(("uniform", np.ones(n) / n))
        # Non-symmetric initialization
        t = np.arange(n)
        inits.append(("asym_ramp", np.maximum(2.0 - t / (n * 0.7), 0.1)))
        inits.append(("asym_ramp2", np.maximum(1.5 - 0.5 * np.sin(np.pi * t / n), 0.1)))

        for name, init_a in inits:
            rem = 200.0 - (time.monotonic() - t1_start)
            if rem < 3:
                break
            t_alloc = max(2.0, min(8.0, rem / 4.0))
            a_opt, s_opt = mwu_optimize(init_a, t_alloc, beta_init=40.0, lr_init=0.04)
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, it_c in candidates[:10]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"mwu_{it_c}_n{n_c}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Try the best length and a couple alternatives
        lp_lengths = [n_best]
        for alt in [1024, 896, 1152]:
            if alt != n_best and 512 <= alt <= 1536:
                lp_lengths.append(alt)
        lp_lengths = lp_lengths[:3]

        for n_lp in lp_lengths:
            rem = deadline - time.monotonic() - 400
            if rem < 60:
                break
            t_alloc = min(rem * 0.5, 600.0)
            print(f"  LP-FP n={n_lp}, budget={t_alloc:.0f}s")

            if best_seq is not None:
                initial = interpolate_seq(best_seq, n_lp)
            else:
                initial = np.ones(n_lp) / n_lp

            a_fp, score_fp, iters = lp_fictitious_play(
                n_lp, t_alloc, initial=initial, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"lpfp_n{n_lp}")

    # === Phase 3: CD Polishing ===
    print(f"\n=== Phase 3: CD Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_polish = len(best_seq)

        # Fast CD
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_fcd = min(rem * 0.4, 180.0)
            print(f"  FCD n={n_polish}, budget={t_fcd:.0f}s")
            a_fcd, s_fcd, _ = fast_coord_descent(best_seq, t_fcd, label="fast1")
            best_seq, best_score = check_and_update(a_fcd, best_seq, best_score, "fcd1")

        # Fine CD
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_fine = min(rem * 0.5, 120.0)
            print(f"  FineCD n={n_polish}, budget={t_fine:.0f}s")
            a_fine, s_fine = fine_coord_descent(best_seq, t_fine)
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another MWU pass on the polished result
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_mwu2 = min(rem * 0.4, 80.0)
            a_mwu2, s_mwu2 = mwu_optimize(best_seq, t_mwu2, beta_init=100.0, lr_init=0.01)
            best_seq, best_score = check_and_update(a_mwu2, best_seq, best_score, "mwu_polish")

        # More fine CD
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_fine2, s_fine2 = fine_coord_descent(best_seq, min(rem * 0.6, 60.0))
            best_seq, best_score = check_and_update(a_fine2, best_seq, best_score, "fine_cd2")

    # === Phase 4: Try alternative lengths ===
    print(f"\n=== Phase 4: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        for alt_n in [n_cur + 128, n_cur - 128, n_cur + 256, 1024, 1280]:
            if alt_n <= 0 or alt_n > 2048 or alt_n == n_cur:
                continue
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alt = min(rem * 0.35, 60.0)
            if t_alt < 10:
                break
            print(f"  Alt n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_seq(best_seq, alt_n)
            # Quick MWU then fine CD
            a_tmp, _ = mwu_optimize(init_alt, t_alt * 0.4, beta_init=60.0, lr_init=0.02)
            a_alt, s_alt, _ = fast_coord_descent(a_tmp, t_alt * 0.5, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 5: Final intensive LP + CD ===
    print(f"\n=== Phase 5: Final LP+CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 60:
            n_fin = len(best_seq)
            t_lp = min(rem * 0.5, 200.0)
            print(f"  Final LP-FP n={n_fin}, budget={t_lp:.0f}s")
            a_fin, s_fin, _ = lp_fictitious_play(n_fin, t_lp, initial=best_seq, solver=solver)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_lp")

            rem = deadline - time.monotonic() - 5
            if rem > 20:
                a_fin2, s_fin2 = fine_coord_descent(best_seq, min(rem * 0.7, 60.0))
                best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
