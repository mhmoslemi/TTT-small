```python
def subgradient_step(a, step_size):
    n = len(a)
    c = np.convolve(a, a)
    max_val = c.max()
    max_indices = np.where(c >= max_val - 1e-10)[0]
    
    # Subgradient of max_k c_k
    grad = np.zeros(n)
    for k in max_indices:
        for i in range(n):
            if 0 <= k - i < n:
                if i == k - i:
                    grad[i] += 0.5 * a[i]
                else:
                    grad[i] += a[k-i]
    
    # Average over active k's
    grad /= len(max_indices)
    
    # Projected gradient step on {a >= 0, sum(a) = 1}
    a_new = a - step_size * grad
    # Project: set negatives to 0, renormalize
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new
```

```python
    """Project v onto {x >= 0 : sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1:
        # Already has sum <= 1, just need to handle negatives
        # Actually we need sum = 1 exactly
        pass
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
```

```python
    """Nesterov accelerated alternating LP with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n * fast_convolve(a0).max()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 * dampening / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 300 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1
            if stall_count >= 5:
                rng = np.random.default_rng(it + 12345)
                noise_scale = 0.03 + 0.02 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall_count = 0
                prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def mwu_polish(a0, time_limit, n_max_iters=100000):
    """Multiplicative weights update for peak shaving.
    Targets the maximum autocorrelation lag and reduces contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.05
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_val = c.max()
        max_k = int(np.argmax(c))

        # Find all elements contributing to c[max_k]
        js_start = max(0, max_k - n + 1)
        js_end = min(n - 1, max_k)
        if js_start > js_end:
            break

        js = np.arange(js_start, js_end + 1)
        partners = max_k - js  # a[js] * a[partners] contributes to c[max_k]
        contribs = a[js] * a[partners]
        total = contribs.sum()
        if total < 1e-15:
            break

        # Reduce elements with largest contribution
        # a[j] *= (1 - eta * a[partner] / max_val) approximately
        # More precisely: we want to reduce c[max_k] by reducing a[j]
        # d(c[max_k])/d(a[j]) = a[partner] (or 2*a[partner] if j == partner)
        # So reduce a[j] by eta * a[partner] / max_val
        
        reductions = a[partners] / (max_val + 1e-30)
        reductions = np.minimum(reductions, 0.5)  # cap reduction
        a[js] *= (1.0 - eta * reductions)
        a[js] = np.maximum(a[js], 0)

        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        score = 2.0 * n * c.max() if c.max() < max_val else 2.0 * n * fast_convolve(a).max()
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Adaptive eta: reduce if not improving
        if it % 1000 == 0:
            eta *= 0.95

        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"      MWU iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def subgradient_polish(a0, time_limit, n_max_iters=30000):
    """Projected subgradient with Nesterov-style averaging for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(y)
        max_val = c.max()
        # Active set: lags within 10% of max
        threshold = max_val * 0.9
        active = c >= threshold
        n_active = active.sum()
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Compute subgradient
        # grad[i] = (1/|K|) * sum_{k in K} (d c_k / d a_i)
        # d c_k / d a_i = a[k-i] if 0<=k-i<n (times 2 if i != k-i, times 1 if i == k-i)
        # Simplified: grad[i] = (1/|K|) * sum_{k in K, 0<=k-i<n} a[k-i]
        # (The factor of 2 vs 1 doesn't matter for the direction)
        
        active_idx = np.nonzero(active)[0]
        if len(active_idx) <= 30:
            grad = np.zeros(n)
            for k in active_idx:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    grad[js:je+1] += y[k - np.arange(js, je + 1)]
            grad /= len(active_idx)
        else:
            # FFT-based: grad[i] = sum_j y[j] * active[i+j] / n_active
            # = correlate(active, y, 'valid') / n_active
            active_f = active.astype(np.float64)
            grad = np.correlate(active_f, y, mode='valid')
            grad /= n_active

        # Step size
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            t = 0.01
        else:
            t = 2.0 / ((it + 1) * g_norm)

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * best_a
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a_new = y - t * grad
        a_new = project_simplex(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a


def coordinate_descent(a0, time_limit, n_steps=50000):
    """Aggressive coordinate descent: reduce peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Exclude first and last (always smaller)
        c_work = c.copy()
        if len(c_work) > 2:
            c_work[0] = -1
            c_work[-1] = -1
        peak_k = int(np.argmax(c_work))
        peak_val = c[peak_k]

        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = idx[np.argmax(contribs)]

        # Try reducing top contributor
        improved = False
        for factor in [0.99, 0.98, 0.96, 0.94, 0.90, 0.85]:
            a_trial = a.copy()
            reduction = a_trial[top_i] * (1 - factor)
            a_trial[top_i] -= reduction
            if a_trial[top_i] < 0:
                a_trial[top_i] = 0
                reduction = a[top_i]
            # Redistribute to minimize impact
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            # Redistribute to elements NOT contributing to peak
            partners = peak_k - idx
            peak_contrib_mask = np.zeros(n, dtype=bool)
            peak_contrib_mask[idx] = True
            peak_contrib_mask[partners] = True
            redistribute_mask = mask & ~peak_contrib_mask
            if redistribute_mask.sum() == 0:
                redistribute_mask = mask
            others_sum = a_trial[redistribute_mask].sum()
            if others_sum < 1e-15:
                a_trial[redistribute_mask] = 0
            else:
                a_trial[redistribute_mask] += reduction * a_trial[redistribute_mask] / others_sum
            st = a_trial.sum()
            if st < 0.01:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try pair reduction
            top2_idx = idx[np.argsort(contribs)[::-1][:3]]
            pair_improved = False
            for ci in top2_idx:
                i = ci
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.99, 0.97, 0.95]:
                    for f2 in [0.99, 0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            pair_improved = True
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break
            if not pair_improved:
                # Random perturbation
                if step % 200 == 0:
                    rng = np.random.default_rng(step + 999)
                    n_pert = rng.integers(2, 6)
                    indices = rng.choice(n, size=n_pert, replace=False)
                    a_trial = a.copy()
                    for i in indices:
                        a_trial[i] *= (1.0 + rng.uniform(-0.03, 0.03))
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 0.01:
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            pair_improved = True
                if not pair_improved:
                    break

    return best_a


def structured_init(n, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)  # Hamming
    elif variant == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))  # Hann
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)  # Blackman
    elif variant == 8:
        x = np.linspace(0, 1, n)
        a = np.ones(n)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2 / (0.15 ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label, verbose=True):
    """Try to update the best state."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: diverse initialization, Nesterov LP,
    MWU polishing, subgradient refinement, coordinate descent, and
    systematic length exploration.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: Multi-initialization scan (find best starting point) ===
    print(f"\n=== Phase 1: Multi-init scan (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        variants = [4, 0, 5, 6, 2, 7, 9]
        n_to_try = n_best
        t_per_init = t_alloc / (len(variants) + 1)

        for v in variants:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            init = structured_init(n_to_try, variant=v)
            # Quick Nesterov LP to see if this init is promising
            t_sub = min(t_per_init, 25.0)
            if t_sub > 5:
                a_try, sc_try = nesterov_lp(init, t_sub, solver=solver, n_max_iters=5000)
                try_update(best_state, a_try, f"init_v{v}", verbose=False)
                if sc_try < 2.0:
                    print(f"    variant {v}: LP score={sc_try:.6f}")

    # === Phase 2: Main Nesterov LP refinement ===
    print(f"\n=== Phase 2: Main Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_state['seq'], t_alloc, solver=solver, n_max_iters=200000)
        try_update(best_state, a_lp, "nesterov_lp_main")

    # === Phase 3: MWU polishing ===
    print(f"\n=== Phase 3: MWU Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mwu = mwu_polish(best_state['seq'], t_alloc, n_max_iters=200000)
        try_update(best_state, a_mwu, "mwu_polish_1")

    # === Phase 4: Nesterov LP round 2 ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp(best_state['seq'], t_alloc, solver=solver, n_max_iters=150000)
        try_update(best_state, a_lp2, "nesterov_lp_r2")

    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: Subgradient Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg = subgradient_polish(best_state['seq'], t_alloc, n_max_iters=50000)
        try_update(best_state, a_sg, "subgrad_polish")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd = coordinate_descent(best_state['seq'], t_alloc, n_steps=100000)
        try_update(best_state, a_cd, "coord_descent")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"  Budget: {rem:.0f}s remaining")
        n_cur = len(best_state['seq'])
        # Try lengths around current best
        candidate_lengths = []
        for delta in [-128, -96, -64, -48, -32, -16, 16, 32, 48, 64, 96, 128]:
            alt_n = n_cur + delta
            if 512 <= alt_n <= 2048:
                candidate_lengths.append(alt_n)
        # Also try some specific lengths
        for rn in [768, 832, 896, 1024, 1088, 1152, 1216, 1280]:
            if 512 <= rn <= 2048 and rn not in candidate_lengths:
                candidate_lengths.append(rn)

        candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_cur))

        n_explored = 0
        for alt_n in candidate_lengths[:6]:
            rem = deadline - time.monotonic()
            if rem < 50 or n_explored >= 4:
                break
            n_explored += 1
            t_total = min(rem * 0.15, 120.0)
            if t_total < 30:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_state['seq'], alt_n)
            # Nesterov LP at new length
            t_lp = t_total * 0.6
            a_alt, _ = nesterov_lp(init_alt, t_lp, solver=solver, n_max_iters=30000)
            improved = try_update(best_state, a_alt, f"lp_n{alt_n}", verbose=False)

            if improved:
                # Polish with MWU
                rem = deadline - time.monotonic()
                if rem > 20:
                    t_mwu = min(rem * 0.08, 40.0)
                    a_alt2 = mwu_polish(best_state['seq'], t_mwu, n_max_iters=50000)
                    try_update(best_state, a_alt2, f"mwu_n{alt_n}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"  Budget: {rem:.0f}s")
        n_cur = len(best_state['seq'])

        for noise_scale in [0.02, 0.05, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            psum = perturbed.sum()
            if psum < 0.01:
                continue
            perturbed = perturbed / psum

            t_sub = min(rem * 0.3, 60.0)
            if t_sub > 10:
                a_pert, _ = nesterov_lp(perturbed, t_sub, solver=solver, n_max_iters=20000)
                try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 9: Final polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.30, 60.0)
        if t_cd > 10:
            a_fin = coordinate_descent(best_state['seq'], t_cd, n_steps=50000)
            try_update(best_state, a_fin, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_mwu = min(rem * 0.40, 40.0)
        if t_mwu > 5:
            a_fin2 = mwu_polish(best_state['seq'], t_mwu, n_max_iters=100000)
            try_update(best_state, a_fin2, "final_mwu")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.50, 30.0)
        if t_lp_fin > 5:
            a_fin3, _ = nesterov_lp(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=10000)
            try_update(best_state, a_fin3, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
```
