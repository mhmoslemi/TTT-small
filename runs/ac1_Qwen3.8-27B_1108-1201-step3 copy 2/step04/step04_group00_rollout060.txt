def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(f, conv, top_k, temperature):
    """Compute softmax-weighted subgradient using top-k hottest lags.
    
    d(conv[k])/d f[i] = 2*f[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(f)
    conv_len = 2 * n - 1
    k = min(top_k, conv_len)
    
    if k >= conv_len:
        top_idx = np.arange(conv_len)
    else:
        top_idx = np.argpartition(conv, -k)[-k:]
    
    top_val = conv[top_idx]
    # Softmax weights with adaptive temperature
    t = max(temperature, (top_val.max() - top_val.min()) * 0.05 + 1e-12)
    scaled = top_val / t
    scaled -= scaled.max()
    weights = np.exp(scaled)
    ws = weights.sum()
    if ws < 1e-15:
        weights = np.zeros(k)
        weights[np.argmax(top_val)] = 1.0
    else:
        weights /= ws
    
    sg = np.zeros(n)
    for i in range(len(top_idx)):
        wi = weights[i]
        if wi < 1e-12:
            continue
        k_int = int(top_idx[i])
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            sg[idx] += wi * 2.0 * f[k_int - idx]
    
    return sg

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_window(n, window_type='hann'):
    """Create a windowed initialization."""
    x = np.linspace(0, 1, n)
    if window_type == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif window_type == 'hamming':
        w = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif window_type == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 'gaussian':
        w = np.exp(-2 * ((x - 0.5) * 3) ** 2)
    elif window_type == 'cosine':
        w = np.cos(np.pi * (x - 0.5))
        w = np.maximum(w, 0.01)
    else:
        w = np.ones(n)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        w = np.ones(n)
    else:
        w = w / s
    return w

    inits.append(('hamming', make_window(n_best, 'hamming')))
    inits.append(('gaussian', make_window(n_best, 'gaussian')))
    inits.append(('cosine', make_window(n_best, 'cosine')))
    
    # Blend with best
    for name, w in inits[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        # Blend: 70% best + 30% window
        f_init = 0.7 * best_seq + 0.3 * (len(best_seq) * w)
        f_init = project_simplex(f_init)
        
        t_alloc = min(rem * 0.25, 120)
        print(f"  Init '{name}' blend, time={t_alloc:.0f}s")
        a_opt, _ = accelerated_sg(f_init, time_limit=t_alloc,
                                   top_k=150, lr0=0.005, momentum=0.91,
                                   temp0=0.005, verbose=False)
        try_seq(a_opt, f"blend_{name}")
    
    # === Phase 4: Deep optimization on best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.35, 300)
        print(f"  Deep SG, time={t_alloc:.0f}s")
        a_opt, _ = accelerated_sg(best_seq, time_limit=t_alloc,
                                   top_k=300, lr0=0.003, momentum=0.95,
                                   temp0=0.001, verbose=False)
        try_seq(a_opt, "deep_sg")
    
    # === Phase 5: Peak shaving on best ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.3, 100)
        print(f"  Peak shaving, time={t_alloc:.0f}s")
        a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "final_peak")
    
    # === Phase 6: Local polish ===
    print(f"\n=== Phase 6: Local polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.5, 80)
        a_opt, _ = local_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "polish")
    
    # === Phase 7: Final tiny perturbations ===
    rem = deadline - time.monotonic()
    if rem > 10:
        for _ in range(5):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = accelerated_sg(f_pert, time_limit=min(rem * 0.4, 15),
                                       top_k=100, lr0=0.001, momentum=0.9,
                                       temp0=0.001)
            try_seq(a_opt, "tiny_perturb")
    
    # Fallback
    if best_seq is None:
        best_seq = make_window(384, 'hann')
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
