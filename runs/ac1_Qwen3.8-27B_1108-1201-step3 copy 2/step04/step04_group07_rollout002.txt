import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = full_convolution(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def obj_normalized(a):
    """Compute objective for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. C_a @ g <= M, g >= 0. Returns g."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        if g.sum() > 1e-10:
            return g
    return None

def coordinate_descent(a0, time_limit, n_iter_max=500):

def subgradient_step(a, conv, lr, momentum, v, top_k, temp):
    """Subgradient step with top-k peak emphasis."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]
    
    v = momentum * v - lr * sg
    a_new = np.maximum(a + v, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new, v

def resample(seq, new_n):
    """Resample sequence to new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_initializations(n, parent, rng):

    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['two_level'] = two_level / two_level.sum()

    inits['exp'] = exp_dec / exp_dec.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    return inits
