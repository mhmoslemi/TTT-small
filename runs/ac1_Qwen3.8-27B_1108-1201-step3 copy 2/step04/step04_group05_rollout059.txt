Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Nesterov momentum on the simplex
2. Linearized LP Frank-Wolfe for fine-tuning
3. Multi-start with diverse initializations including structured sequences
4. Multi-scale refinement (optimize small, upscale, repeat)
5. Adaptive temperature for the smoothing parameter

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """Compute gradient of tau * log(sum(exp(conv(a)/tau))) using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # Gradient: g_i = 2 * sum_k p_k * a_{k-i} for valid k-i
    # Computed as: g_i = 2 * (p * a_reversed)_{i + n - 1}
    a_rev = a[::-1]
    # Linear convolution of p (length 2n-1) and a_rev (length n) -> length 3n-2
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)
    # g_i = 2 * conv_pa[i + n - 1] for i = 0..n-1
    g = 2.0 * conv_pa[n-1:2*n-1]
    return g, conv

                          lr=0.01, momentum=0.9, verbose=False):
    """Nesterov-accelerated gradient descent on the simplex with LSE smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    y = a.copy()  # Nesterov extrapolation point
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    tau = tau0
    t0 = time.monotonic()
    it = 0
    lr_current = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at y (extrapolation point)
        y_clipped = np.maximum(y, 0.0)
        ys = y_clipped.sum()
        if ys < 1e-15:
            y = a.copy()
            continue
        y_norm = y_clipped / ys
        
        g, conv = lse_gradient(y_norm, tau)
        
        # Nesterov update
        if it % 2 == 0:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it-1)**2 * momentum**2)) / 2.0
        else:
            beta = momentum
        
        y_new = project_simplex(a + lr_current * (a - y) - lr_current * g)
        
        # Actually, let me use a simpler momentum scheme
        v = momentum * v - lr_current * g
        y_new = project_simplex(a + v)
        
        a = y_new
        obj = fast_convolve(a).max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay learning rate and temperature
        lr_current *= 0.9995
        tau = max(tau * 0.999, tau_min)
        
        # Reset if stuck
        if it % 200 == 0:
            v = v * 0.5
        
        if verbose and it % 500 == 0:
            print(f"    LSE-AD iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. Log-sum-exp smoothed gradient descent with momentum (dense gradient via FFT)
2. Linearized LP Frank-Wolfe for precision
3. Multi-scale refinement (coarse-to-fine)

The key improvement over previous: uses the full log-sum-exp gradient (not just top-k)
which provides much better directional information, combined with multi-scale approach.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """
    Compute gradient of tau * log(sum(exp(conv(a,a)/tau))).
    Returns (gradient, conv).
    The gradient is computed efficiently using FFT.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights (numerically stable)
    scaled = conv / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # Gradient: g_i = 2 * sum_k p_k * a_{k-i} (with a_j = 0 for j outside [0,n-1])
    # = 2 * (p * a_rev)_{i + n - 1} where a_rev = reverse(a)
    a_rev = a[::-1]
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)
    g = 2.0 * conv_pa[n-1:2*n-1]
    
    return g, conv


def lse_smoothed_optimize(a0, time_limit, tau_init=0.002, tau_min=0.00005,
                          lr_init=0.05, decay=0.999, verbose=False):
    """
    Smoothed gradient descent with momentum on the simplex.
    Uses log-sum-exp to get a dense, smooth gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(7777)
    
    while time.monotonic() - t0 < time_limit:

        g, conv = lse_gradient(a, tau)

            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # Still move (momentum helps)
            no_improve += 1

            a = project_simplex(best_a + noise)

    return best_a, best_obj


def build_conv_matrix(f):
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    """
    Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for selected k,
           sum(g) = 0, g >= -f.
    Returns direction g or None.
    """
    n = len(f)
    num_lags = 2 * n - 1
    
    if top_k is not None and top_k < num_lags:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peaks = np.arange(num_lags)
    
    B = build_conv_matrix(f)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def lp_frank_wolfe(a0, time_limit, top_k=None, verbose=False):
    """Linearized LP Frank-Wolfe."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        
        g = solve_linearized_lp(a, conv, top_k=top_k)
        if g is None:
            no_improve += 10
            if no_improve > 50:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        
        target = np.maximum(a + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        
        if np.max(np.abs(target - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in np.linspace(0.1, 1.0, 10):
            a_new = (1.0 - alpha) * a + alpha * target
            a_new = np.maximum(a_new, 0.0)

            a_new = a_new / ss
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

            a = (1.0 - best_alpha) * a + best_alpha * target

            a = a / a.sum()
            if best_val < best_obj - 1e-12:

def multi_scale_refine(a_fine_init, n_coarse, time_limit, verbose=False):
    """
    Multi-scale: optimize a coarse version, then use it to initialize the fine version.
    """
    n_fine = len(a_fine_init)
    
    # Downsample to coarse
    if n_coarse < n_fine:
        a_coarse = np.interp(np.linspace(0, 1, n_coarse), np.linspace(0, 1, n_fine), a_fine_init)
        a_coarse = np.maximum(a_coarse, 0)
        cs = a_coarse.sum()
        if cs > 1e-10:
            a_coarse = a_coarse / cs
    
        # Optimize coarse

        a_coarse_opt, _ = lse_smoothed_optimize(a_coarse, time_limit=t_coarse, 
                                                 tau_init=0.005, verbose=False)
        a_coarse_opt, _ = lp_frank_wolfe(a_coarse_opt, time_limit=t_coarse * 0.5,
                                           top_k=20, verbose=False)
        
        # Upsample back
        a_fine_init = np.interp(np.linspace(0, 1, n_fine), np.linspace(0, 1, n_coarse), a_coarse_opt)
        a_fine_init = np.maximum(a_fine_init, 0)
        fs = a_fine_init.sum()
        if fs > 1e-10:
            a_fine_init = a_fine_init / fs
    
    # Optimize fine
    a_fine_opt, _ = lse_smoothed_optimize(a_fine_init, time_limit=time_limit * 0.4,
                                           tau_init=0.002, verbose=verbose)
    a_fine_opt, _ = lp_frank_wolfe(a_fine_opt, time_limit=time_limit * 0.3,
                                     top_k=50, verbose=verbose)
    
    return a_fine_opt

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. Log-sum-exp (LSE) smoothed gradient descent with momentum (dense gradient via FFT)
2. Linearized LP Frank-Wolfe for precision
3. Multi-scale coarse-to-fine refinement
4. Diverse structured initializations + aggressive length exploration
Key improvement: LSE gradient targets ALL peaks simultaneously (not just top-k),
providing much richer directional information for escaping local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """Compute gradient of tau*log(sum(exp(conv(a)/tau))) using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    a_rev = a[::-1]
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_pa = np.fft.irfft(P * AR, fft_size)
    g = 2.0 * conv_pa[n - 1:2 * n - 1]
    return g, conv


def lse_optimize(a0, time_limit, tau_init=0.003, tau_min=0.00005,
                 lr_init=0.05, decay=0.9995, mom=0.9, verbose=False):
    """LSE smoothed gradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(9999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, conv = lse_gradient(a, tau)
        v = mom * v - lr * g
        a_new = project_simplex(a + v)
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
        lr = max(lr * decay, 0.0005)
        tau = max(tau * 0.9997, tau_min)
        if no_improve > 250:
            noise = rng.standard_normal(n) * (0.003 + 0.001 * min(no_improve - 250, 50))
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, 0.005)
            tau = max(tau_init * 0.4, tau_min * 5)
            no_improve = 0
        if verbose and it % 2000 == 0:
            print(f"    LSE iter {it}: best={best_obj:.10f}, tau={tau:.6f}, t={time.monotonic()-t0:.1f}s")
    return best_a, best_obj


def build_conv_matrix(f):
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    """Linearized LP: min t s.t. t >= conv[k] + 2*(B@g)[k], sum(g)=0, g>=-f."""
    n = len(f)
    num_lags = 2 * n - 1
    if top_k is not None and top_k < num_lags:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peaks = np.arange(num_lags)
    B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def lp_frank_wolfe(a0, time_limit, top_k=None, verbose=False):
    """Linearized LP Frank-Wolfe with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        g = solve_linearized_lp(a, conv, top_k=top_k)
        if g is None:
            no_improve += 10
            if no_improve > 60:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue
        target = np.maximum(a + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts
        if np.max(np.abs(target - a)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in np.linspace(0.05, 1.0, 12):
            a_new = (1.0 - alpha) * a + alpha * target
            a_new = np.maximum(a_new, 0.0)
            ss = a_new.sum()
            if ss < 1e-15:
                continue
            a_new = a_new / ss
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50:
            a = project_simplex(best_a + rng.standard_normal(n) * (0.002 + 0.0005 * (no_improve - 50)))
            no_improve = 0
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: best={best_obj:.10f}, t={time.monotonic()-t0:.1f}s")
    return best_a, best_obj


def multi_scale_refine(a_fine_init, n_coarse, time_limit, verbose=False):
    """Coarse-to-fine: optimize at coarse resolution, upscale, refine."""
    n_fine = len(a_fine_init)
    t0 = time.monotonic()
    if n_coarse < n_fine:
        a_coarse = np.interp(np.linspace(0, 1, n_coarse),
                             np.linspace(0, 1, n_fine), a_fine_init)
        a_coarse = np.maximum(a_coarse, 0)
        cs = a_coarse.sum()
        if cs > 1e-10:
            a_coarse = a_coarse / cs
        t_c = (time_limit * 0.3)
        a_coarse, _ = lse_optimize(a_coarse, time_limit=t_c, tau_init=0.01,
                                    tau_min=0.001, lr_init=0.1, verbose=False)
        rem_c = time_limit * 0.3 - (time.monotonic() - t0)
        if rem_c > 5:
            a_coarse, _ = lp_frank_wolfe(a_coarse, time_limit=rem_c, top_k=15)
        a_fine_init = np.interp(np.linspace(0, 1, n_fine),
                                np.linspace(0, 1, n_coarse), a_coarse)
        a_fine_init = np.maximum(a_fine_init, 0)
        fs = a_fine_init.sum()
        if fs > 1e-10:
            a_fine_init = a_fine_init / fs
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_fine, _ = lse_optimize(a_fine_init, time_limit=rem * 0.5,
                                  tau_init=0.002, tau_min=0.0001,
                                  lr_init=0.03, verbose=verbose)
        rem2 = rem - (time.monotonic() - t0)
        if rem2 > 10:
            a_fine, _ = lp_frank_wolfe(a_fine, time_limit=rem2, top_k=40)
    else:
        a_fine = a_fine_init
    return a_fine


def generate_inits(n, parent, rng):
    """Generate diverse initializations with design-theory-inspired structures."""
    inits = []
    x = np.linspace(0, 1, n)
    inits.append(np.ones(n) / n)
    # Parabolic
    p = 1.0 - (2 * x - 1) ** 2
    p = np.maximum(p, 0.01)
    inits.append(p / p.sum())
    # Cosine
    c = np.cos(np.pi * (x - 0.5))
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    # Edge-weighted
    e = (2 * x - 1) ** 2 + 0.05
    inits.append(e / e.sum())
    # Narrow Gaussian
    g = np.exp(-12 * (x - 0.5) ** 2)
    inits.append(g / g.sum())
    # Wide Gaussian
    g2 = np.exp(-2 * (x - 0.5) ** 2)
    inits.append(g2 / g2.sum())
    # Two bumps
    b = np.exp(-25 * (x - 0.3) ** 2) + np.exp(-25 * (x - 0.7) ** 2)
    inits.append(b / b.sum())
    # Sine modulated
    s = np.abs(np.sin(2 * np.pi * x * 2)) + 0.1
    inits.append(s / s.sum())
    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    # Symmetric random
    half_n = n // 2
    rh = rng.dirichlet(np.ones(half_n) * 0.5)
    a_sym = np.zeros(n)
    a_sym[:half_n] = rh
    a_sym[half_n:] = rh[::-1]
    inits.append(a_sym / a_sym.sum())
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        pp = px + rng.standard_normal(n) * 0.05
        pp = np.maximum(pp, 0)
        pps = pp.sum()
        if pps > 1e-10:
            inits.append(pp / pps)
    return inits
