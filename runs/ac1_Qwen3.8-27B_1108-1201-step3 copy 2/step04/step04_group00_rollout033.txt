import numpy as np
import time

def auto_conv(a):
    """Compute full linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n-1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def qp_grad(a, peaks, weights):
    """Compute gradient of a^T Q a where Q = sum w_i P_{k_i}."""
    n = len(a)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        k = peaks[idx]
        w = weights[idx]
        # (P_k * a)[i] = a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        j_vals = k - np.arange(i_start, i_end + 1)
        grad[i_start:i_end+1] += 2.0 * w * a[j_vals]
    return grad

def solve_qp(a_init, peaks, weights, n_iter=200, lr=0.05):
    """Solve min a^T Q a s.t. simplex using projected gradient."""
    a = a_init.copy()
    for i in range(n_iter):
        grad = qp_grad(a, peaks, weights)
        a_new = project_simplex(a - lr * grad)
        if np.max(np.abs(a_new - a)) < 1e-12:
            a = a_new
            break
        a = a_new
    return a

def mm_optimize(a_init, time_limit, verbose=False):
    """MM-QP optimization for flat auto-convolution."""
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_val = auto_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0
    threshold = 0.7  # fraction of max to include in peak set
    
    while time.monotonic() - t0 < time_limit:
        c = auto_conv(a)
        cur_max = c.max()
        
        # Find peaks
        thresh = cur_max * threshold
        peak_idx = np.where(c >= thresh)[0]
        peak_vals = c[peak_idx]
        weights = peak_vals / peak_vals.sum()
        
        # Solve QP
        a_new = solve_qp(a, peak_idx, weights, n_iter=200)
        
        # Line search
        best_on_line = cur_max
        best_t = 0.0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1-t) * a + t * a_new
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = auto_conv(a_trial).max()
            if val < best_on_line:
                best_on_line = val
                best_t = t
        
        if best_t > 0 and best_on_line < cur_max - 1e-14:

            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_on_line < best_val - 1e-14:
                best_val = best_on_line
                best_a = a.copy()
                no_improve = 0
                if verbose:

                threshold = 0.7
                no_improve = 0
