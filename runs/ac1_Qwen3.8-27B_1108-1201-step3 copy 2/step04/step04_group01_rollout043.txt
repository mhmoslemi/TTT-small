    p = np.ones(K) / K
    
    a = a0.copy() / a0.sum()

    best_obj = fast_convolve(a).max()

    a_sum = np.zeros(n)
    T = 0
    
    t0 = time.monotonic()

        # This is a Hankel-like matrix

        # Use projected gradient with a few iterations

        a_inner = a_inner_solve(Q_mat_fn=p, a0=a, n=n, iterations=20)
        
        # Compute b
        b = fast_convolve(a_inner)

        log_p -= eta * b
        log_p -= log_p.max()
        p = np.exp(log_p)
        p /= p.sum()
        
        # Update a_sum for averaging
        a_sum += a_inner
        T += 1
        
        # Compute objective of a_inner and averaged a
        obj_inner = b.max()
        a_avg = a_sum / T
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-15:

            obj_avg = fast_convolve(a_avg).max()
        
        if obj_inner < best_obj:
            best_obj = obj_inner
            best_a = a_inner.copy()
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = a_avg.copy()
        
        if T % 100 == 0:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA-LSE (Nesterov-accelerated Log-Sum-Exp) with adaptive beta
2. FW-LP refinement
3. High-beta LSE polishing
4. Top-k subgradient with variance reduction
5. Alternative length exploration
6. Final ultra-high-beta polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_full(weights, f):
    """Compute gradient of LSE: 2 * cross_correlation(weights, f).
    weights has length 2n-1, f has length n.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m]
    """
    n = len(f)
    # grad[i] = 2 * sum_m weights[i+m] * f[m]
    # This is a cross-correlation
    # Use FFT: pad to appropriate size
    L = 2 * n - 1 + n - 1  # = 3n - 2
    fft_size = 1 << (L - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n-1:n-1+n]


def lse_objective_and_grad(f, beta):
    """Compute LSE objective and gradient."""
    n = len(f)
    b = fast_convolve(f)
    bc = beta * b
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    obj = (bc_max + np.log(Z)) / beta
    w = exp_bc / Z
    grad = compute_lse_grad_full(w, f)
    return obj, grad, b, w

    t = 1.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA-LSE (Nesterov-accelerated Log-Sum-Exp) with adaptive beta and backtracking
2. FW-LP refinement with adaptive constraint relaxation
3. High-beta LSE polishing with reduced momentum
4. Top-k subgradient with variance reduction (SAGA-style)
5. Alternative length exploration
6. Final ultra-high-beta polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_full(weights, f):
    """Compute gradient: grad[i] = 2 * sum_{m} weights[i+m] * f[m].
    weights has length 2n-1, f has length n.
    """
    n = len(f)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_topk_grad(top_indices, weights, f):
    """Compute subgradient from top-k peaks.
    top_indices: array of peak indices (length k)
    weights: array of weights for each peak (length k)
    f: sequence of length n
    """
    n = len(f)
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        if weights[i] < 1e-14:
            continue
        k = int(top_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * f[k - idx]
    return grad


def lse_obj_and_grad(f, beta):
    """Compute LSE objective, gradient, and convolution."""
    n = len(f)
    b = fast_convolve(f)
    bc = beta * b
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    obj = (bc_max + np.log(Z)) / beta
    w = exp_bc / Z
    grad = compute_lse_grad_full(w, f)
    return obj, grad, b, w


def fista_lse(f0, time_limit, beta_start=30.0, beta_end=5000.0,

    """FISTA (Nesterov-accelerated) on LSE-smoothed objective with backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    beta = beta_start
    lr = lr_scale * 1.0 / (2.0 * beta + 1.0)
    x = f.copy()
    y = f.copy()
    t = 1.0

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta_target_increment = np.log(beta_end / beta_start) / max(1, time_limit)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y
        obj_y, grad_y, b_y, w_y = lse_obj_and_grad(y, beta)

        # Gradient step with backtracking
        x_new = y - lr * grad_y
        x_new = project_simplex(x_new)

        # Check if step is good (simplified backtracking)
        obj_new_full = fast_convolve(x_new).max()

        # Backtracking: if objective worsened significantly, reduce step
        if obj_new_full > obj_y + 1e-10:
            lr *= 0.5

                x_new = y - lr * grad_y
                x_new = project_simplex(x_new)
                obj_new_full = fast_convolve(x_new).max()

        # Nesterov extrapolation

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = project_simplex(y)
            t = t_new
        else:
            y = x_new.copy()

        x = x_new

        # Update best
        if obj_new_full < best_obj - 1e-15:
            best_obj = obj_new_full
            best_f = x.copy()

    return best_f, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            m_exp = min(m_exp * 1.03, 3.0)

            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            m_exp = 1.0

def topk_saga(f0, time_limit, top_k=12, lr_init=0.005, momentum=0.93, temp_init=0.0001):
    """Top-k subgradient descent with SAGA-style variance reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    v = np.zeros(n)
    cur_mom = momentum
    # SAGA table: store past gradients
    saga_table = np.zeros((top_k, n))
    saga_time = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Select top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Compute current subgradient
        sg = compute_topk_grad(peaks, w, f)

        # SAGA: replace old gradient for this mini-batch
        old_grad = saga_table[saga_time % top_k]
        avg_grad = saga_table.mean(axis=0)
        saga_table[saga_time % top_k] = sg
        saga_time += 1
        corrected_grad = sg - old_grad + avg_grad

        v = cur_mom * v - cur_lr * corrected_grad
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                cur_lr *= 0.94
            if no_improve > 250:
                rng = np.random.default_rng(time.monotonic() % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

        cur_temp *= 1.003

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FISTA-LSE (Nesterov-accelerated Log-Sum-Exp) with adaptive beta and backtracking
2. FW-LP refinement with adaptive constraint relaxation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_full(weights, f):
    """Compute gradient: grad[i] = 2 * sum_{m} weights[i+m] * f[m].
    weights has length 2n-1, f has length n.
    """
    n = len(f)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_topk_grad(top_indices, weights, f):
    """Compute subgradient from top-k peaks."""
    n = len(f)
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        if weights[i] < 1e-14:
            continue
        k = int(top_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * f[k - idx]
    return grad


def lse_obj_and_grad(f, beta):
    """Compute LSE objective, gradient, and convolution."""
    n = len(f)
    b = fast_convolve(f)
    bc = beta * b
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    Z = exp_bc.sum()
    obj = (bc_max + np.log(Z)) / beta
    w = exp_bc / Z
    grad = compute_lse_grad_full(w, f)
    return obj, grad, b, w

def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def topk_saga(f0, time_limit, top_k=12, lr_init=0.005, momentum=0.93, temp_init=0.0001):

    f = np.maximum(f0.astype(np.float64), 0.0)

    cur_temp = temp_init
    v = np.zeros(n)
    cur_mom = momentum
    table_size = max(top_k, 8)
    saga_table = np.zeros((table_size, n))
    saga_time = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = compute_topk_grad(peaks, w, f)

        slot = saga_time % table_size
        old_grad = saga_table[slot].copy()
        avg_grad = saga_table.mean(axis=0)
        saga_table[slot] = sg
        saga_time += 1
        corrected_grad = sg - old_grad + avg_grad

        v = cur_mom * v - cur_lr * corrected_grad
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0

                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
