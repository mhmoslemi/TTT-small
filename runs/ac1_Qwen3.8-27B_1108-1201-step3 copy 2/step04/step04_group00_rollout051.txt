def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
                no_improve += 1
                if no_improve > 20:
                    break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 30:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_obj = cur_peak

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center_frac=0.5, width_frac=0.3):
    """Create a Gaussian-shaped initialization on the simplex."""
    x = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def make_raised_cosine_init(n):
    """Raised cosine window initialization."""
    x = np.arange(n) / (n - 1)
    a = np.cos(np.pi * (x - 0.5)) ** 2
    # Actually use the standard raised cosine (Hanning-like)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    a /= s
    return a


def make_symmetric_gaussian_init(n, width_frac=0.25):
    """Symmetric Gaussian centered in the middle."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = width_frac * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining smoothed gradient, LP-FW,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = make_gaussian_init(768)

    if parent is None:
        parent = make_gaussian_init(768)

    # === Phase 1: Nesterov smoothed gradient on parent ===
    print(f"\n=== Phase 1: Nesterov SG on parent (n={len(parent)}, score={best_score:.6f}) ===")
    t1 = min(350, (deadline - time.monotonic()) * 0.25)
    if t1 > 30:
        print(f"  Time alloc: {t1:.0f}s")
        a_opt, obj_opt = nesterov_smoothed_optimize(parent, time_limit=t1,
                                                     T_init=0.08, T_decay=0.9997,
                                                     lr_init=0.04, verbose=True)
        try_seq(a_opt, "nesterov_parent")

    # === Phase 2: Multi-length diverse initializations ===
    print(f"\n=== Phase 2: Multi-length search (best={best_score:.6f}) ===")
    lengths_to_try = [512, 768, 1024, 1280]

    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            print("  Not enough time remaining, skipping longer lengths")
            break

        # Create initializations for this length
        inits = []
        init_labels = []

        # Initialize from current best via interpolation
        base = best_seq if best_seq is not None else parent
        f_interp = interpolate_sequence(base, target_n)
        inits.append(f_interp)
        init_labels.append("interp_best")

        # Gaussian
        inits.append(make_gaussian_init(target_n, width_frac=0.25))
        init_labels.append("gauss_025")
        inits.append(make_gaussian_init(target_n, width_frac=0.35))
        init_labels.append("gauss_035")

        # Raised cosine
        inits.append(make_raised_cosine_init(target_n))
        init_labels.append("raised_cos")

        # Symmetric narrow Gaussian
        inits.append(make_symmetric_gaussian_init(target_n, width_frac=0.2))
        init_labels.append("sym_gauss")

        # Random
        r = rng.random(target_n)
        r /= r.sum()
        inits.append(r)
        init_labels.append("random")

        # Allocate time per initialization
        t_per_init = min(rem / len(inits), 120)
        print(f"  n={target_n}, {len(inits)} inits, ~{t_per_init:.0f}s each")

        for i, (f_init, label) in enumerate(zip(inits, init_labels)):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_alloc = min(t_per_init, rem)
            if t_alloc < 15:
                break

            # Run Nesterov smoothed gradient
            a_opt, obj_opt = nesterov_smoothed_optimize(f_init, time_limit=t_alloc,
                                                         T_init=0.06, T_decay=0.9998,
                                                         lr_init=0.03)
            try_seq(a_opt, f"nesterov_n{target_n}_{label}")

            # Quick RIME polish
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                a_rime, _ = rime_search(a_opt, time_limit=min(rem, 30),
                                         pert_scale=0.005)
                try_seq(a_rime, f"rime_n{target_n}_{label}")

    # === Phase 3: LP-FW polishing on best ===
    print(f"\n=== Phase 3: LP-FW polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_lpfw = min(rem * 0.7, 200)
            print(f"  LP-FW time: {t_lpfw:.0f}s, n={len(best_seq)}")
            a_opt, obj_opt = lp_fw_optimize(best_seq, time_limit=t_lpfw, verbose=True,
                                             max_iters=100)
            try_seq(a_opt, "lpfw_polish")

    # === Phase 4: More Nesterov with different parameters ===
    print(f"\n=== Phase 4: Deep Nesterov (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            # Try with different T schedule
            t_alloc = min(rem * 0.5, 150)
            a_opt, _ = nesterov_smoothed_optimize(best_seq, time_limit=t_alloc,
                                                   T_init=0.03, T_decay=0.9999,
                                                   lr_init=0.02, verbose=True)
            try_seq(a_opt, "nesterov_deep_slow")

            # Try with higher initial T
            rem = deadline - time.monotonic() - 60
            if rem > 20:
                a_opt, _ = nesterov_smoothed_optimize(best_seq, time_limit=min(rem, 80),
                                                       T_init=0.15, T_decay=0.999,
                                                       lr_init=0.06, verbose=True)
                try_seq(a_opt, "nesterov_deep_broad")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_cd = min(rem * 0.8, 120)
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Final RIME + small perturbations ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # RIME with small perturbations
            a_opt, _ = rime_search(best_seq, time_limit=min(rem * 0.5, 60),
                                    pert_scale=0.003)
            try_seq(a_opt, "rime_final")

            # One more Nesterov with very low T
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_opt, _ = nesterov_smoothed_optimize(best_seq, time_limit=min(rem, 40),
                                                       T_init=0.01, T_decay=0.9999,
                                                       T_min=2e-6, lr_init=0.01)
                try_seq(a_opt, "nesterov_final_fine")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.6f}")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
