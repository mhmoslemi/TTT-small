import numpy as np
from scipy.optimize import linprog

def fast_convolve_fft(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(f, conv, temp):
    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(f)
    log_max = conv.max()
    shifted = temp * (conv - log_max)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        # Fallback: uniform over top few
        top = np.argpartition(conv, -32)[-32:]
        w = np.zeros(len(conv))
        w[top] = 1.0 / 32
    else:
        w /= wsum

    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * f[k - idx]
    return grad


def solve_lp_oracle(f, M, time_limit=15.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Geometric (exponential decay)
    for rate in [0.5, 0.7, 0.8, 0.9]:
        g = np.exp(-rate * x / n)
        inits.append(g / g.sum())
        g2 = np.exp(-rate * (n - 1 - x) / n)
        inits.append(g2 / g2.sum())

    # Power law
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        pw = (x + 1.0) ** (-p)
        inits.append(pw / pw.sum())
        pw2 = (n + 1.0 - x) ** (-p)
        inits.append(pw2 / pw2.sum())

    # Cosine
    cos = np.cos(np.pi * (x - c) / (n + 1)) ** 2 + 0.01
    inits.append(cos / cos.sum())

    # Triangle (peak in middle)
    tri = 1.0 - np.abs(x - c) / (c + 1)
    inits.append(tri / tri.sum())

    # V-shape (peak at edges)
    vs = np.abs(x - c) / (c + 1) + 0.1
    inits.append(vs / vs.sum())

    # Sine quarter
    sq = np.sin(np.pi * x / (2.0 * n)) + 0.05
    inits.append(sq / sq.sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    # Bimodal
    bimodal = np.exp(-((x - n * 0.25) ** 2) / (n * 0.05)) + np.exp(-((x - n * 0.75) ** 2) / (n * 0.05)) + 0.01
    inits.append(bimodal / bimodal.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses multi-phase optimization:
    Phase 1: Nesterov smoothed gradient on parent
    Phase 2: FW-LP on parent and best from Phase 1
    Phase 3: Explore alternative lengths/initializations
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01 or len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, sum={p.sum():.4f}", flush=True)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("Using default parent (uniform n=384)", flush=True)

    n_p = len(parent)

    # === Phase 1: Nesterov smoothed gradient on parent ===
    print("\n=== Phase 1: Nesterov Smoothed Gradient ===", flush=True)
    t1_budget = min(400, (deadline - time.monotonic()) * 0.35)
    if t1_budget > 30:
        print(f"  Optimize parent (n={n_p}), budget={t1_budget:.0f}s", flush=True)
        a_opt, obj = nesterov_smoothed_opt(
            parent, time_limit=t1_budget,
            init_temp=max(1.0, n_p / 40.0),
            max_temp=max(10.0, n_p / 5.0),
            temp_growth=1.003,
            init_lr=0.012,
            lr_decay=0.9996,
            lp_every=300,
            lp_time=10.0
        )
        try_seq(a_opt, "P1_nesterov_parent")

    # === Phase 2: FW-LP on parent and best ===
    print("\n=== Phase 2: Frank-Wolfe LP ===", flush=True)
    t2_budget = min(350, (deadline - time.monotonic()) * 0.30)
    if t2_budget > 20:
        # FW on parent
        t2a = min(t2_budget * 0.5, 150)
        if t2a > 15:
            print(f"  FW-LP on parent, t={t2a:.0f}s", flush=True)
            a_opt, obj = fw_lp_optimize(parent, time_limit=t2a, max_lp_time=15.0)
            try_seq(a_opt, "P2_fw_parent")

        # FW on best so far
        if best_seq is not None:
            t2b = min(t2_budget * 0.5, 150)
            if t2b > 15:
                print(f"  FW-LP on best (n={len(best_seq)}), t={t2b:.0f}s", flush=True)
                a_opt, obj = fw_lp_optimize(best_seq, time_limit=t2b, max_lp_time=15.0)
                try_seq(a_opt, "P2_fw_best")

    # === Phase 3: Alternative initializations ===
    print("\n=== Phase 3: Alternative Explorations ===", flush=True)
    t3_budget = (deadline - time.monotonic()) * 0.25
    if t3_budget > 20:
        # Try different lengths
        lengths_to_try = [n_p]
        for n_try in [256, 512, 192, 448]:
            if n_try != n_p:
                lengths_to_try.append(n_try)
        lengths_to_try = lengths_to_try[:4]

        for n_try in lengths_to_try:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            n_inits = 4
            t_per_init = max(10, rem / (n_inits * max(1, len(lengths_to_try) - 1)))
            t_per_init = min(t_per_init, 50)

            inits = generate_structured_inits(n_try, rng)
            # Pick best few initializations
            # Quick eval to rank
            init_objs = []
            for ini in inits:
                c = fast_convolve_fft(ini)
                init_objs.append(c.max())
            top_indices = np.argsort(init_objs)[:min(3, len(inits))]

            for idx in top_indices:
                rem = deadline - time.monotonic() - 40
                if rem < t_per_init:
                    break
                f_init = inits[idx]
                t_alloc = min(t_per_init, rem - 20)
                if t_alloc < 10:
                    break

                a_opt, obj = nesterov_smoothed_opt(
                    f_init, time_limit=t_alloc,
                    init_temp=max(1.0, n_try / 40.0),
                    max_temp=max(10.0, n_try / 5.0),
                    init_lr=0.01,
                    lp_every=200,
                    lp_time=8.0
                )
                try_seq(a_opt, f"P3_n{n_try}_i{idx}")

    # === Phase 4: Deep optimization and polish ===
    print("\n=== Phase 4: Deep Optimization & Polish ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"  Current best: {best_score:.8f} (n={n_best})", flush=True)

        # Nesterov on best with different params
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_alloc = min(rem * 0.4, 120)
            print(f"  Nesterov on best, t={t_alloc:.0f}s", flush=True)
            a_opt, obj = nesterov_smoothed_opt(
                best_seq, time_limit=t_alloc,
                init_temp=max(2.0, n_best / 20.0),
                max_temp=max(20.0, n_best / 2.0),
                temp_growth=1.005,
                init_lr=0.005,
                lr_decay=0.9997,
                lp_every=500,
                lp_time=10.0
            )
            try_seq(a_opt, "P4_nesterov_best")

        # FW-LP on perturbed best
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.3, 80)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            print(f"  FW-LP on perturbed best, t={t_alloc:.0f}s", flush=True)
            a_opt, obj = fw_lp_optimize(f_pert, time_limit=t_alloc, max_lp_time=12.0)
            try_seq(a_opt, "P4_fw_perturbed")

        # Fine polish
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.8, 60)
            print(f"  Fine polish, t={t_alloc:.0f}s", flush=True)
            a_opt, obj = fine_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "P4_polish")

    # === Phase 5: Final micro-optimizations ===
    print("\n=== Phase 5: Final Micro-Optimizations ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # Try a few micro-perturbations with quick polish
            for i in range(5):
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                scale = 0.003 / (i + 1)
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                t_alloc = min(rem - 2, 10)
                if t_alloc < 3:
                    break
                a_opt, obj = fine_polish(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"P5_micro{i}")

        # One final Nesterov with very sharp temperature
        rem = deadline - time.monotonic() - 3
        if rem > 15:
            a_opt, obj = nesterov_smoothed_opt(
                best_seq, time_limit=min(rem - 2, 30),
                init_temp=n_best / 2.0,
                max_temp=n_best,
                temp_growth=1.01,
                init_lr=0.002,
                lr_decay=0.999,
                lp_every=5000,
                lp_time=5.0
            )
            try_seq(a_opt, "P5_sharp")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
