3. Subgradient polish for final refinement

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp(f, M)

def lbfgs_smooth_optimize(f0, time_limit, tau_init=20.0, tau_max=500.0, verbose=False):
    """
    L-BFGS-B optimization using log-sum-exp smoothed objective.
    Parameters: b (log of a), softmax gives a.
    Anneals tau from tau_init to tau_max.
    """
    n = len(f0)
    a = np.maximum(f0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize b = log(a) (with floor for zeros)
    b = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Anneal tau
    tau_current = tau_init
    stages = 8
    
    for stage in range(stages):
        if time.monotonic() - t0 > time_limit:
            break
        tau_current = tau_init * (tau_max / tau_init) ** (stage / max(stages - 1, 1))
        
        # Define objective and gradient for L-BFGS-B
        def obj_grad(b):
            obj, grad = softmax_obj_grad(b, tau_current)
            return obj, grad
        
        # Run L-BFGS-B
        try:
            res = minimize(obj_grad, b, jac=True, method='L-BFGS-B',
                          options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500})
            a_new = np.exp(res.x - res.x.max())
            a_new /= a_new.sum()
            
            score = fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                b = res.x.copy()
                if verbose:
                    print(f"    tau={tau_current:.1f}: score={score:.8f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B failed at tau={tau_current:.1f}: {e}")
            # Try a smaller step
            a_new = a.copy()
            score = best_score
    
    return best_a, best_score


def subgradient_polish(f0, time_limit, lr_init=0.002, momentum=0.93, top_k=20):
    """Subgradient method with momentum for final polish."""
    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_min = max(0, n - top_k)
        peaks = np.where(conv >= np.sort(conv)[-top_k][0])[0]
        
        # Weight by peak height (sharper)
        pv = conv[peaks]
        max_pv = pv.max()
        w = np.exp(-(max_pv - pv) * 100)  # sharp weighting
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with FW-LP
2. L-BFGS-B optimization on log-sum-exp smoothed objective with annealed temperature (FFT-based gradient)
3. Subgradient polish with momentum
Key innovation: smooth optimization via log-sum-exp surrogate computed via FFT cross-correlation,
with temperature annealing to transition from global to local search, plus multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(b, tau):
    """
    Compute the log-sum-exp smoothed objective and gradient w.r.t. b,
    where a = softmax(b). Objective: (1/tau)*log(sum_k exp(tau*conv(a,a)_k))
    """
    n = len(b)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()

    conv = fast_convolve(a)

    tau_conv = tau * conv
    max_tc = tau_conv.max()
    exp_shifted = np.exp(tau_conv - max_tc)
    log_sum_exp = max_tc + np.log(exp_shifted.sum())
    obj = log_sum_exp / tau

    w = exp_shifted / exp_shifted.sum()

    # grad_a[m] = 2 * sum_l w[m+l] * a[l] = 2 * (w * a_rev)[m+n-1]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    ffs = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, ffs)
    AR = np.fft.rfft(a_rev, ffs)
    corr_full = np.fft.irfft(W * AR, ffs)[:conv_len]
    grad_a = 2.0 * corr_full[n - 1:2 * n - 1]

    db_grad = a * (grad_a - np.dot(grad_a, a))
    return obj, db_grad


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)

    best_obj = M

            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

def lbfgs_smooth_optimize(f0, time_limit, tau_init=20.0, tau_max=500.0, max_iter=300):
    """
    L-BFGS-B on log-sum-exp smoothed objective with tau annealing.
    Parameterizes a = softmax(b) to handle constraints automatically.
    """
    n = len(f0)
    a = np.maximum(f0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = np.log(np.maximum(a, 1e-25))
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    stages = 6

    for stage in range(stages):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(stages - 1, 1)
        tau_cur = tau_init * (tau_max / tau_init) ** frac

        def obj_grad(bb):
            return softmax_obj_grad(bb, tau_cur)

        try:
            res = minimize(obj_grad, b, jac=True, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'gtol': 1e-10, 'maxfun': 1000})
            a_new = np.exp(res.x - res.x.max())
            a_new /= a_new.sum()
            score = fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                b = res.x.copy()
                print(f"    tau={tau_cur:.1f}: smooth_score={score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def subgradient_polish(f0, time_limit, lr_init=0.002, momentum=0.93, top_k=20):
    """Subgradient method with momentum."""
    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        max_pv = pv.max()
        w = np.exp(-(max_pv - pv) * 50.0)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=15, temp_init=0.0004):

    f = np.maximum(f0, 0.0)

    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= cur_lr * 2.0 * w[i] * f[k - idx]

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.6, 0.01)
            cur_temp = temp_init * 2.0
            no_improve = 0

    return best_f, best_obj
