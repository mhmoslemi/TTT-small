"""

Uses diverse initialization, FW-LP for global structure, adaptive LSE gradient

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, w):
    """Compute subgradient 2*sum_k w[k]*a[k-i] for each i.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * result[:n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the FW LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend(f[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, A_cache):
    """Solve LP for FW direction: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=10000.0, top_k=16):
    """Adaptive LSE/log-sum-exp gradient descent with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.5 / n
    momentum = 0.92
    v = np.zeros(n)
    y = f.copy()  # Nesterov look-ahead
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov: evaluate gradient at look-ahead point
        conv = fast_convolve(y)
        bc = beta * conv

        # Top-k weighted softmax for stability at high beta
        if top_k < len(bc):
            peak_idx = np.argpartition(bc, -top_k)[-top_k:]
            peak_vals = bc[peak_idx]
            pv_max = peak_vals.max()
            exp_pv = np.exp(peak_vals - pv_max)
            w = np.zeros(2 * n - 1)
            w[peak_idx] = exp_pv
        else:
            bc_max = bc.max()
            w = np.exp(bc - bc_max)

        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(bc)] = 1.0
        else:
            w = w / w_sum

        grad = compute_subgrad(y, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 50
            if no_improve > 200:
                break
            continue
        grad = grad / grad_norm

        # Nesterov momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Update look-ahead for next iteration
        y = f_new + momentum * (f_new - f)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 5.0 / n)
        else:
            no_improve += 1
            # Backtracking: undo step and reduce lr
            if no_improve >= 5:
                f = best_f.copy()
                y = f.copy()
                v = np.zeros(n)
                lr *= 0.85
                momentum = max(0.85, momentum - 0.003)

            if no_improve > 100:
                # Restart with different beta
                f = best_f.copy()
                y = f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, lr * 0.5)
                beta = min(beta * 1.3, beta_end)
                momentum = 0.90
                no_improve = 0

        # Gradually increase beta
        if beta < beta_end:
            beta *= 1.002

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def kkt_refine(f0, time_limit, n_iter_max=200000):
    """KKT-based coordinate refinement.
    At optimum, for active positions a[i]>0, the subgradient should be constant.
    We iteratively identify violations and move mass to fix them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    step_size = 0.01

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()

        # Find peaks (top few)
        k_peaks = min(8, 2 * n - 1)
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_vals = conv[peak_indices]
        pv = np.maximum(peak_vals - M * 0.95, 0.0)
        if pv.sum() < 1e-15:
            pv = np.zeros(k_peaks)
            pv[np.argmax(peak_vals)] = 1.0
        else:
            pv = pv / pv.sum()

        # Build peak weights
        w = np.zeros(2 * n - 1)
        for i, pi in enumerate(peak_indices):
            w[int(pi)] = pv[i]

        # Compute subgradient
        g = compute_subgrad(f, w)

        # Active set
        active = f > 1e-12
        if active.sum() < 2:
            break

        # KKT: g should be constant on active set
        g_active = g[active]
        lambda_val = g_active.mean()

        # Find most violated: active positions with g < lambda (should reduce)
        # and inactive positions with g > lambda (should increase)
        violations_active = lambda_val - g_active  # positive means g[i] < lambda
        violations_inactive = g[~active] - lambda_val  # positive means g[i] > lambda

        if violations_active.max() < 1e-10 and violations_inactive.max() < 1e-10:
            step_size *= 0.9
            if step_size < 1e-6:
                break

        # Move mass from worst active to worst inactive
        active_idx = np.where(active)[0]
        inactive_idx = np.where(~active)[0]

        if len(inactive_idx) == 0:
            # All active: redistribute among actives
            worst_src = active_idx[np.argmax(violations_active)]
            best_dst = active_idx[np.argmin(g_active)]
            if worst_src == best_dst:
                break
            amt = min(f[worst_src] * 0.1, step_size / n * f[worst_src])
            f[worst_src] -= amt
            f[best_dst] += amt
        else:
            worst_src = active_idx[np.argmax(violations_active)]
            best_dst = inactive_idx[np.argmax(violations_inactive)]
            amt = f[worst_src] * min(0.05, step_size)
            f[worst_src] -= amt
            f[best_dst] += amt

        # Re-normalize
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            continue
        f = f / s

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT it={it}, obj={best_obj:.12f}, step={step_size:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def multi_peak_sg(f0, time_limit, top_k=12, temp_init=0.0001):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.002
    momentum = 0.93
    v = np.zeros(n)
    temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Top-k softmax
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_sum = w_peaks.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros(2 * n - 1)
            for i, pi in enumerate(peaks):
                w[int(pi)] = w_peaks[i] / w_sum

        # Subgradient
        g = compute_subgrad(f, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            no_improve += 100
            break
        g = g / g_norm

        v = momentum * v - lr * g
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.93
            if no_improve > 500:
                rng = np.random.default_rng(hash(int(time.monotonic() * 1e6)) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = max(0.001, lr * 0.6)
                no_improve = 0

        temp *= 1.003

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

            parent = p / p.sum()
            pscore = evaluate_sequence(parent.tolist())

    except Exception as e:

    if parent is None:

        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    print(f"\n=== Phase 1: FW-LP (n={n_parent}) ===")

    print(f"\n=== Phase 2: LSE Polishing ===")

    print(f"\n=== Phase 3: KKT Refinement ===")

    print(f"\n=== Phase 4: Final Polish ===")
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    n_best = len(current_best)

    t_alloc = min(120, deadline - time.monotonic() - 200)
    if t_alloc > 15:
        a_opt, _ = multi_peak_sg(current_best, time_limit=t_alloc,
                                  top_k=max(6, n_best // 10), temp_init=0.00003)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 100)
    if t_alloc > 20:
        a_opt, _ = lse_optimize(current_best, time_limit=t_alloc,
                                beta_start=100.0, beta_end=15000.0, top_k=8)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths ===")
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent

    alt_lengths = [384, 512, 576, 640]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)

    for n_alt in alt_lengths:

        if t_remaining < 60:

        x_old = np.linspace(0, 1, len(current_best), endpoint=False)
        x_new = np.linspace(0, 1, n_alt, endpoint=False)
        f_interp = np.interp(x_new, x_old, current_best)

        f_interp = f_interp / f_interp.sum()

                                        beta_start=20.0, beta_end=5000.0, top_k=10)
                try_sequence(a_opt, best_tracker)

    print(f"\n=== Phase 6: Final Aggressive Polish ===")

    if best_tracker[1] is not None:

            best_tracker[0] = final_score
    else:
        final_seq = np.ones(448) / 448
        best_tracker[0] = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_tracker[0]:.8f}, n={len(best_tracker[1])}")
    return [float(x) for x in best_tracker[1]]
