"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated gradient descent with Adam and softmax smoothing
2. LP-based polishing with perturbation escapes
3. Multi-scale initialization and hierarchical extension
4. Multi-restart strategy across different sequence lengths

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_grad(a, b, t, top_k=200):
    """
    Compute objective value and gradient using top-k softmax approximation.
    Returns (obj, grad) where obj = 2n * soft_max(b,t) / s^2
    and grad is the projected gradient (sum zero).
    """
    n = len(a)
    s = a.sum()
    
    # Top-k softmax
    if top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_sel = b[idx]
        scaled = b_sel / t
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        w_sel = exp_vals / exp_vals.sum()
        log_sum_exp = b_sel.max() / t + np.log(exp_vals.sum())
        soft_max = t * log_sum_exp
        
        # Build full w (mostly zeros)
        w = np.zeros(len(b))
        w[idx] = w_sel
    else:
        scaled = b / t
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        w = exp_vals / exp_vals.sum()
        soft_max = b.max() + t * np.log(exp_vals.sum())
    
    obj = 2.0 * n * soft_max / (s * s)
    
    # Gradient of soft_max w.r.t. a[i]:
    # d(soft_max)/da[i] = sum_k w[k] * 2*a[k-i] = 2 * sum_j a[j] * w[i+j]
    # Compute via FFT: conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    N = 1 << (3 * n).bit_length()
    conv_full = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(w_rev, N), N)
    grad_conv = conv_full[n-1:2*n-1][::-1]  # length n
    
    # Full gradient
    d_obj_da = (4.0 * n / (s * s)) * grad_conv - (4.0 * n * soft_max) / (s ** 3)
    
    # Project onto sum(a) = 1 (subtract mean)
    d_obj_da -= d_obj_da.mean()
    
    return obj, d_obj_da


def adam_optimize(a0, time_limit, lr=0.005, beta1=0.9, beta2=0.999, 
                  eps=1e-8, t_init=None, t_min=None, top_k=200,
                  t_decay=0.9995, no_improve_patience=500):
    """
    Adam optimizer for the smoothed objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    max_b = b.max()
    
    if t_init is None:
        t_init = max_b / 30.0
    if t_min is None:
        t_min = max_b / 5000.0
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1_t = 0
    beta2_t = 0
    
    t = t_init
    best_obj = 2.0 * n * max_b
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        obj, grad = softmax_grad(a, b, t, top_k=top_k)
        
        # True (non-smoothed) objective for tracking
        true_obj = 2.0 * n * b.max() / (a.sum() ** 2)
        
        beta1_t += 1
        beta2_t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** beta1_t)
        v_hat = v / (1 - beta2 ** beta2_t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a = a_new / s_new
        
        # Check improvement using true objective
        b = fast_convolve(a)
        cur_true = 2.0 * n * b.max()
        if cur_true < best_obj - 1e-10:
            best_obj = cur_true
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        t = max(t_min, t * t_decay)
        
        if no_improve >= no_improve_patience:
            break
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_optimize_fast(a0, time_limit, max_iters=500):

    Fast LP-based optimization.
    """

        a = np.ones(len(a)) / len(a)

        a = a / s

    max_no_improve = 60

    rng = np.random.default_rng(42)

        if time.monotonic() - t0 > time_limit:
            break

        b_ub = np.full(2 * n - 1, M * 1.0001)
        bounds = [(0, None)] * n

                           method='highs', options={'presolve': True,
                                                     'dual_feasibility_tolerance': 1e-9})
            if not result.success:
                no_improve += 1
                continue
        except Exception:
            no_improve += 1
            continue
        
        g = result.x
        g_sum = g.sum()

            no_improve += 1
            continue

        improved = False

                best_obj = obj_try

                C = build_conv_matrix(a)

            b = fast_convolve(a)
            M = b.max()
            C = build_conv_matrix(a)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Sine modulations
    for freq in [1, 2, 3]:
        key = f'sine{freq}'
        vals = np.sin(np.pi * freq * x / n) ** 2 + 0.01
        inits[key] = vals / vals.sum()
    
    # Cosine window
    cos_w = np.cos(np.pi * x / (2 * n)) ** 2 + 0.01
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext

        # Also try same length with different inits
        configs.insert(0, (len(parent), ['parent', 'sine2', 'alternating', 'random'], 60))
    
    phase1_results = []
    
    for n, init_names, t_per_init in configs:
        if time.monotonic() > deadline - 400:
            break
        
        inits = generate_initializations(n, parent, rng)
        available_inits = [nm for nm in init_names if nm in inits]
        if not available_inits:
            available_inits = list(inits.keys())[:3]
        
        for name in available_inits:
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 300)
            if t_alloc < 10:
                break
            
            a0 = inits[name]
            t = max_b_est = fast_convolve(a0).max()
            a_opt, obj = adam_optimize(
                a0, time_limit=t_alloc, lr=0.003, 
                beta1=0.9, beta2=0.999,
                t_init=t / 20.0, t_min=t / 2000.0,
                top_k=100, t_decay=0.9997
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Adam n={n}, {name}: {score:.6f}")
            else:
                print(f"  Adam n={n}, {name}: invalid (inf)")
    
    phase1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP-based deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Select top candidates (diverse n values)
    candidates = []
    seen_n = set()
    for score, n, name, a_opt in phase1_results:
        if n not in seen_n:
            candidates.append((n, a_opt))
            seen_n.add(n)
        if len(candidates) >= 4:
            break
    
    # Always include the best
    if best_sequence is not None and len(best_sequence) not in seen_n:
        candidates.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 200.0
    t_per_lp = max(30.0, remaining * 0.5 / max(1, len(candidates)))
    
    for i, (n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break
        
        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize_fast(a0, time_limit=t_alloc, max_iters=300)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Second round of Adam on the best (with lower temp)
    # =========================================================
    print("=== Phase 3: Adam Refinement ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        n_best = len(best_sequence)
        b = fast_convolve(best_sequence)
        max_b = b.max()
        
        # Multiple Adam runs with different parameters
        adam_configs = [
            {'lr': 0.002, 't_init': max_b / 50.0, 't_min': max_b / 5000.0, 'top_k': 50, 't_decay': 0.9998},
            {'lr': 0.001, 't_init': max_b / 100.0, 't_min': max_b / 10000.0, 'top_k': 30, 't_decay': 0.9999},
            {'lr': 0.005, 't_init': max_b / 10.0, 't_min': max_b / 1000.0, 'top_k': 200, 't_decay': 0.9995},
        ]
        
        for ci, cfg in enumerate(adam_configs):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(remaining / (len(adam_configs) - ci) * 0.8, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break
            
            # Slight perturbation for diversity
            if ci > 0:
                a0 = best_sequence + rng.standard_normal(n_best) * 0.001
                a0 = np.maximum(a0, 0)
                s0 = a0.sum()
                if s0 > 1e-15:
                    a0 = a0 / s0
                else:
                    a0 = best_sequence.copy()
            else:
                a0 = best_sequence.copy()
            
            a_opt, obj = adam_optimize(a0, time_limit=t_alloc, **cfg)
