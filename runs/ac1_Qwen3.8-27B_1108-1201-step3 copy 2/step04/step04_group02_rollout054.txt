def project_simplex(v, target_sum=1.0):

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj * target_sum

    scaled -= scaled.max()
    return T * (scaled.max() + np.log(np.exp(scaled).sum()))

def solve_lp(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: conv(f,g)[k] = sum_j f[max(0,k-n+1):min(n,k+1)][j - max(0,k-n+1)] * g[j]
    # For efficiency, use banded structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            f_new = f  # reject step if no improvement
        
        f = f_new
        
        # Anneal T
        progress = (time.monotonic() - t0) / time_limit
        T = max(T_min, T_init * (1 - progress) + T_min * progress)
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        
        if no_improve > 500:
            # Restart with noise
            rng = np.random.default_rng(t * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.005
            no_improve = 0
    
    return best_f, best_obj

def sg_descent_v2(f0, time_limit, initial_lr=0.005, top_k=128, T_init=0.005, T_min=0.0005):
    """Improved subgradient descent with Adam-style adaptation and T-annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    lr = initial_lr
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Adaptive T
        progress = (time.monotonic() - t0) / time_limit
        T = max(T_min, T_init * (1 - 0.9 * progress))
        
        # Smooth gradient
        grad = compute_smooth_grad(f, conv, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt learning rate
        if no_improve % 200 == 199:
            lr *= 0.9
        
        if no_improve > 1000:
            rng = np.random.default_rng(t * 11 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = initial_lr
            no_improve = 0
    
    return best_f, best_obj

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # Solve LP with current M (or slightly relaxed)
        M_use = M * m_exp
        g = solve_lp(f, M_use, time_limit=min(30, rem - 15))
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        alphas = np.linspace(0.1, 1.0, 18)
        for alpha in alphas:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    half = n / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangular (peak at center)
    tri = 1.0 - np.abs(x - half) / half
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Inverted triangle (peak at edges)
    itr = np.abs(x - half) / half + 0.01
    inits.append(itr / itr.sum())
    
    # Quadratic (U-shaped)
    q = ((x - half) / half) ** 2 + 0.1
    inits.append(q / q.sum())
    
    # Gaussian (centered)
    g = np.exp(-(x - half) ** 2 / (2 * (n / 4) ** 2))
    inits.append(g / g.sum())
    
    # Exponential decay from center
    e = np.exp(-np.abs(x - half) / (n / 3.0))
    inits.append(e / e.sum())
    
    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Sinc-like
    s = np.sinc((x - half) / (n / 6.0))
    s = np.maximum(s, 0.001)
    inits.append(s / s.sum())
    
    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.1)
    inits.append(r)
    
    # Random Dirichlet (spread)
    r2 = rng.dirichlet(np.ones(n) * 2.0)
    inits.append(r2)
    
    # Symmetric random
    half_sym = rng.dirichlet(np.ones(n // 2 + 1) * 0.5)
    full_sym = np.concatenate([half_sym[::-1], half_sym[1:]])
    inits.append(full_sym / full_sym.sum())
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) * target_sum / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    return proj * target_sum


def compute_smooth_grad(f, conv, T):
    """
    Compute gradient of T*log(sum(exp(conv/T))) w.r.t. f.
    grad_f[j] = (2/T) * sum_{l=0}^{n-1} w[j+l] * f[l]
    where w = softmax(conv/T)
    """
    n = len(f)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum

    # result[j] = sum_{l=0}^{n-1} f[l] * w[j+l]
    # Use cross-correlation via FFT: ifft(conj(FFT(f)) * FFT(w))[j]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(np.conj(F) * W, fft_size)
    result = cross[:n]

    return (2.0 / T) * result


def solve_lp(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def adam_smooth_descent(f0, time_limit, T_init=0.005, T_min=0.0003,
                        lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                        restart_every=800, noise_scale=0.003):
    """
    Adam optimization on smoothed objective with simplex projection.
    Uses T-annealing and periodic restarts.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # T-annealing: exponential decay
        progress = (time.monotonic() - t0) / time_limit
        T = max(T_min, T_init * np.exp(-3.0 * progress))

        # Compute smoothed gradient
        grad = compute_smooth_grad(f, conv, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate decay
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0

        # Periodic restart with noise
        if no_improve > restart_every:
            rng = np.random.default_rng(t * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(lr, 0.002)
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, max_lp_time=30):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < max_lp_time + 15:
            break

        M_use = M * m_exp
        g = solve_lp(f, M_use, time_limit=min(max_lp_time, rem - 20))
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp >= 5.0:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search: coarse grid
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for alpha in alphas:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    half = n / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (peak at center)
    tri = 1.0 - np.abs(x - half) / half
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Inverted triangle (peak at edges)
    itr = np.abs(x - half) / half + 0.05
    inits.append(itr / itr.sum())

    # Gaussian (centered, narrow)
    g1 = np.exp(-(x - half) ** 2 / (2 * (n / 5) ** 2))
    inits.append(g1 / g1.sum())

    # Gaussian (centered, wide)
    g2 = np.exp(-(x - half) ** 2 / (2 * (n / 2.5) ** 2))
    inits.append(g2 / g2.sum())

    # Exponential decay from center
    e = np.exp(-np.abs(x - half) / (n / 3.0))
    inits.append(e / e.sum())

    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Flat-top (high in middle, low at edges)
    ft = np.ones(n)
    edge = max(1, n // 8)
    ft[:edge] *= 0.5
    ft[-edge:] *= 0.5
    inits.append(ft / ft.sum())

    # Random Dirichlet (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.2)
    inits.append(r)

    # Symmetric random
    half_len = n // 2 + 1
    half_sym = rng.dirichlet(np.ones(half_len) * 0.5)
    full_sym = np.concatenate([half_sym[::-1], half_sym[1:]])
    inits.append(full_sym / full_sym.sum())

    # Parent-based interpolation
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with uniform
            blended = 0.7 * (px / ps) + 0.3 * (np.ones(n) / n)
            inits.append(blended / blended.sum())

    return inits

    if best_seq is not None:
        n_best = len(best_seq)
        print(f"  Best n={n_best}, score={best_score:.8f}")

        # Multiple FW-LP runs with perturbation
        for p in range(4):
            rem = deadline - time.monotonic() - 120
            if rem < 25:
                break
            t_alloc = min(rem // (4 - p + 1), 90)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            elif p == 2:
                f_init = project_simplex(0.92 * best_seq + 0.08 * best_seq[::-1])
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)

            print(f"  Deep FW-LP {p}, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, max_lp_time=30)
            try_seq(a_opt, f"deep_fw_{p}")

        # Multiple Adam-SG runs
        for p in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem // (4 - p + 1), 70)

            if p == 0:
                f_init = best_seq.copy()
                cfg = {"T_init": 0.002, "T_min": 0.0001, "lr": 0.002}
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                cfg = {"T_init": 0.003, "T_min": 0.0002, "lr": 0.003}
            elif p == 2:
                f_init = project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)
                cfg = {"T_init": 0.001, "T_min": 0.0001, "lr": 0.001}
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
                cfg = {"T_init": 0.004, "T_min": 0.0003, "lr": 0.004}

            print(f"  Deep SG {p}, t={t_alloc:.0f}s")
            a_opt, obj = adam_smooth_descent(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine Adam-SG
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            print(f"  Fine polish, t={min(rem,50):.0f}s")
            a_opt, _ = adam_smooth_descent(best_seq, time_limit=min(rem, 50),
                                           T_init=0.001, T_min=0.00005, lr=0.001,
                                           restart_every=500, noise_scale=0.001)
            try_seq(a_opt, "polish_fine")

        # FW-LP on best
        rem = deadline - time.monotonic() - 25
        if rem > 20:
            print(f"  Final FW-LP, t={min(rem,40):.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 40), max_lp_time=20)
            try_seq(a_opt, "polish_fw")

        # Micro perturbations
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            for _ in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)
                a_opt, _ = adam_smooth_descent(f_pert, time_limit=min(rem, 12),
                                               T_init=0.0005, T_min=0.00003, lr=0.0005,
                                               restart_every=300, noise_scale=0.0005)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK used: score={best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
