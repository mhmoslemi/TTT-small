def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_D(a, i, j):
    """Compute D[k] = a[k-i] - a[k-j] for the 2-coordinate swap analysis."""
    n = len(a)
    D = np.zeros(2 * n - 1)
    D[i:i + n] += a
    D[j:j + n] -= a
    return D


def find_optimal_delta(conv, D, a, i, j, top_k=40):
    """Find delta that minimizes max_k [conv[k] + 2*delta*D[k]].
    Uses pair enumeration on top-k active constraints."""
    n = len(a)
    conv_len = 2 * n - 1

    if top_k >= conv_len:
        active = np.arange(conv_len)
    else:
        active = np.argpartition(conv, -top_k)[-top_k:]

    conv_a = conv[active]
    D_a = D[active]

    # Find pairs (p, q) with D_a[p] > 0 > D_a[q]
    pos = np.where(D_a > 1e-15)[0]
    neg = np.where(D_a < -1e-15)[0]

    best_delta = 0.0
    best_val = np.max(conv_a)

    if len(pos) == 0 or len(neg) == 0:
        # All D values have the same sign; check boundary
        if len(pos) > 0:
            # Function increasing in delta; best is delta = 0 (or negative boundary)
            best_delta = 0.0
        elif len(neg) > 0:
            best_delta = 0.0
        return best_delta, best_val

    # Enumerate pairs
    # For efficiency, limit pairs
    max_pairs = 200
    idx = 0
    for p_idx in pos:
        for q_idx in neg:
            if idx >= max_pairs:
                break
            idx += 1
            dp = D_a[p_idx]
            dq = D_a[q_idx]
            if dp - dq < 1e-15:
                continue
            delta = (conv_a[q_idx] - conv_a[p_idx]) / (2.0 * (dp - dq))
            if delta < -0.5 or delta > 0.5:
                continue
            # Evaluate at this delta
            vals = conv_a + 2.0 * delta * D_a
            val = np.max(vals)
            if val < best_val:
                best_val = val
                best_delta = delta

    # Also check delta = 0 and small neighbors
    for d_candidate in [0.0, -0.001, 0.001, -0.005, 0.005, -0.01, 0.01]:
        vals = conv_a + 2.0 * d_candidate * D_a
        val = np.max(vals)
        if val < best_val:
            best_val = val
            best_delta = d_candidate

    # Clamp for non-negativity
    if best_delta > 0 and a[i] > 0:
        best_delta = min(best_delta, a[i] * 0.999)
    if best_delta < 0 and a[j] > 0:
        best_delta = max(best_delta, -a[j] * 0.999)

    return best_delta, best_val


def coord_descent_optimize(a0, time_limit, rng, verbose=False):
    """Coordinate descent with 2-coordinate swaps and exact line search."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    strategy = 'gradient'

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Choose strategy periodically
        if it % 500 == 0:
            strategy = 'gradient' if no_improve < 100 else 'random'

        if strategy == 'gradient' and it % 50 == 0:
            # Smart pair: high a[i] at peak-contributing position, low a[j] elsewhere
            peak_k = int(np.argmax(conv))
            # Gradient of peak w.r.t. a[i] is 2*a[peak_k - i] (if valid)
            # Find i that maximizes contribution to peak
            valid = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
            if len(valid) > 2:
                grad_at_peak = 2.0 * a[peak_k - valid]
                i = valid[int(np.argmax(grad_at_peak))]
                # Find j with smallest a value (far from i)
                dist = np.abs(np.arange(n) - i)
                j_candidates = np.where(dist > n // 4)[0]
                if len(j_candidates) > 0:
                    j = j_candidates[int(np.argmin(a[j_candidates]))]
                else:
                    j = (i + n // 2) % n
            else:
                i = int(np.argmax(a))
                j = int(np.argmin(a))
        elif strategy == 'gradient':
            # Semi-smart: random i, best j (minimum a among far positions)
            i = int(rng.integers(n))
            dist = np.abs(np.arange(n) - i)
            j_candidates = np.where(dist > n // 6)[0]
            if len(j_candidates) > 0:
                j = j_candidates[int(rng.integers(len(j_candidates)))]
            else:
                j = int(rng.integers(n))
                if j == i:
                    j = (i + 1) % n
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                j = (i + 1) % n

        # Compute D
        D = compute_D(a, i, j)

        # Find optimal delta
        delta, pred_val = find_optimal_delta(conv, D, a, i, j, top_k=30)

        if abs(delta) < 1e-14:
            no_improve += 1
            continue

        # Apply swap
        a[i] += delta
        a[j] -= delta

        # Verify with actual convolution
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    CD iter {it}: obj={best_obj:.10f}, t={elapsed:.0f}s")
        else:
            # Try a fraction of the step
            for frac in [0.5, 0.25, 0.1]:
                d2 = delta * frac
                a[i] -= delta
                a[j] += delta
                a[i] += d2
                a[j] -= d2
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    break
                else:
                    a[i] -= d2
                    a[j] += d2
            else:
                # Revert fully
                a[i] -= delta
                a[j] += delta
                no_improve += 1

        if no_improve > 300:
            # Restart
            scale = 0.005 + 0.003 * (no_improve // 300)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, rng, lr=0.002, mom=0.93, top_k=30, temp=0.0001):
    """Subgradient descent with Nesterov momentum and multi-peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead

    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        beta = (1.0 + np.sqrt(1.0 + 4.0 * cur_lr * cur_lr)) / 2.0
        y = (1.0 - beta * cur_lr) * a + beta * cur_lr * v

        conv_y = fast_convolve(np.maximum(y, 0.0))

        # Compute subgradient using top-k peaks with softmax weights
        if top_k >= len(conv_y):
            peaks = np.arange(len(conv_y))
        else:
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]

        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        v = (1.0 - cur_lr) * v - cur_lr * sg
        a = project_simplex(a - cur_lr * sg)

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.004

        if no_improve > 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            y = a.copy()
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_a, best_obj


def build_conv_matrix_slice(f, k_list):
    """Build B matrix rows only for specified lags k_list."""
    n = len(f)
    B = np.zeros((len(k_list), n))
    for idx, k in enumerate(k_list):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_range = np.arange(js, je + 1)
        B[idx, idx_range] = f[k - idx_range]
    return B


def solve_linearized_lp_reduced(f, conv, top_k=25):
    """Solve reduced linearized LP: min t s.t. t >= conv[k] + 2*(B*g)[k] for top k,
    g >= -f, sum(g)=0."""
    n = len(f)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peaks = np.sort(peaks)
    conv_peaks = conv[peaks]

    c = np.zeros(n + 1)
    c[0] = 1.0

    B = build_conv_matrix_slice(f, peaks)
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_peaks

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def lp_fw_optimize(a0, time_limit, rng, top_k=25, verbose=False):
    """Frank-Wolfe with linearized LP oracle (reduced constraints)."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        current_obj = conv.max()

        g = solve_linearized_lp_reduced(a, conv, top_k=top_k)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        target = a + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - alpha) * a + alpha * target
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new = a_new / sn
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0.0)
            sn = a.sum()
            if sn < 1e-15:
                a = best_a.copy()
            else:
                a = a / sn

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append((u"uniform", np.ones(n) / n))
    inits.append((u"random", rng.dirichlet(np.ones(n) * 0.3)))

    x = np.arange(n, dtype=np.float64)
    xc = 0.5 * (n - 1)

    # U-shapes with varying sharpness
    for sharp in [0.3, 0.5, 0.8, 1.2]:
        u = 1.0 / (1.0 + sharp * np.exp(-((x - xc) / (n * 0.15)) ** 2))
        inits.append((f"u_{sharp}", u / u.sum()))

    # Multi-modal
    for npeaks in [2, 3, 4]:
        centers = np.linspace(0, n - 1, npeaks + 2)[1:-1]
        mm = np.zeros(n)
        for c in centers:
            mm += np.exp(-0.5 * ((x - c) / (n * 0.08)) ** 2)
        inits.append((f"mm_{npeaks}", mm / mm.sum()))

    # Cosine-modulated (ensure non-negative)
    for freq in [2, 3, 5]:
        c = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits.append((f"cos_{freq}", c / c.sum()))

    # Beta-like
    for alpha_b, beta_b in [(0.5, 2.0), (2.0, 0.5), (0.5, 0.5)]:
        b = (x + 1) ** alpha_b * (n - x) ** beta_b
        b = np.maximum(b, 0.01)
        inits.append((f"beta_{alpha_b}_{beta_b}", b / b.sum()))

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append((u"parent_interp", px / ps))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: Diverse initialization with quick evaluation
    Phase 2: Coordinate descent (primary optimizer, fast iterations)
    Phase 3: Subgradient polishing
    Phase 4: LP-based Frank-Wolfe refinement
    Phase 5: Final local search and verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    # Determine lengths to try
    if parent is not None:
        n_p = len(parent)
    else:
        n_p = 384

    lengths_to_try = sorted(set([n_p, 256, 320, 384, 448, 512]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 640]

    all_candidates = []

    for n_try in lengths_to_try:
        inits = generate_inits(n_try, parent, rng)
        for label, init in inits:
            all_candidates.append((label, n_try, init))

    # Quick evaluation
    for label, n_try, cand in all_candidates:
        if time.monotonic() > deadline - 600:
            break
        try_seq(cand, f"init_{label}_n{n_try}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: Coordinate Descent ===
    print("\n=== Phase 2: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 300
    t_alloc_cd = min(remaining * 0.5, 900)

    if t_alloc_cd > 30:
        print(f"  Running coordinate descent (n={n_best}), budget={t_alloc_cd:.0f}s")
        a_opt, cd_obj = coord_descent_optimize(best_seq, time_limit=t_alloc_cd * 0.7,
                                               rng=rng, verbose=True)
        try_seq(a_opt, "coord_descent")

        # Run again on the improved result with more time
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_cd2 = min(rem * 0.5, 400)
            a_opt, cd_obj2 = coord_descent_optimize(best_seq, time_limit=t_cd2,
                                                    rng=rng, verbose=True)
            try_seq(a_opt, "coord_descent_2")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Subgradient Polish ===
    print("\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic() - 200
    t_alloc_sg = min(remaining * 0.3, 400)

    if t_alloc_sg > 20:
        print(f"  Running subgradient, budget={t_alloc_sg:.0f}s")
        t_sg1 = min(t_alloc_sg * 0.5, 200)
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg1, rng=rng,
                                        lr=0.002, mom=0.95, top_k=n_best // 4, temp=0.0001)
        try_seq(a_opt, "sg1")

        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_sg2 = min(rem * 0.5, 150)
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg2, rng=rng,
                                            lr=0.0008, mom=0.97, top_k=n_best // 2, temp=0.00003)
            try_seq(a_opt, "sg2_fine")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: LP-based Frank-Wolfe ===
    print("\n=== Phase 4: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 150
    t_alloc_lp = min(remaining * 0.4, 500)

    if t_alloc_lp > 30:
        print(f"  Running LP FW (n={n_best}), budget={t_alloc_lp:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(t_alloc_lp * 0.6, 300),
                                  rng=rng, top_k=20, verbose=True)
        try_seq(a_opt, "lp_fw")

        # Try on a perturbed version
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            noise = rng.standard_normal(n_best) * 0.008
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem * 0.5, 200),
                                      rng=rng, top_k=20)
            try_seq(a_opt, "lp_fw_pert")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Final Coordinate Descent + Local Search ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic() - 50
    t_final = min(remaining * 0.8, 300)

    if t_final > 15:
        print(f"  Final coordinate descent, budget={t_final:.0f}s")
        a_opt, _ = coord_descent_optimize(best_seq, time_limit=t_final * 0.6, rng=rng)
        try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # Very fine subgradient
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem * 0.5, 60),
                                            rng=rng, lr=0.0003, mom=0.98,
                                            top_k=n_best, temp=0.00001)
            try_seq(a_opt, "final_sg")

    # === Final: Try alternative lengths ===
    rem = deadline - time.monotonic() - 10
    if rem > 20 and best_seq is not None:
        for n_try in [n_best - 32, n_best + 32]:
            if n_try < 128 or rem < 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = coord_descent_optimize(px, time_limit=min(rem * 0.4, 30), rng=rng)
                try_seq(a_opt, f"len{n_try}_cd")
                rem = deadline - time.monotonic() - 8

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
