"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization strategy:
1. Trust-region QP for targeted peak reduction
2. Linearized LP Frank-Wolfe for larger exploratory jumps
3. Smoothed subgradient descent for fine polishing
4. Simulated annealing for escaping local minima
5. Local search for final fine-tuning
Combined with diverse initialization strategies and adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_convolution_matrix(f):
    """Build M matrix where (M@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    M = np.zeros((2 * n - 1, n))
    for j in range(n):
        M[j:j + n, j] = f
    return M


def compute_subgradient_fft(f, weights):
    """Compute subgradient using FFT. weights has length 2n-1."""
    n = len(f)
    # sg[m] = 2 * sum_k weights[k] * f[k-m] for valid k-m
    # This is 2 * cross-correlation of weights and f
    # = 2 * convolve(weights, f[::-1])[n-1 : n-1+n]
    conv = np.convolve(weights, f[::-1], mode='full')
    return 2.0 * conv[n-1:n-1+n]


def smoothed_subgradient(f, conv, top_k=30, temp=0.0005):
    """Compute smoothed subgradient of max(conv)."""
    n = len(f)
    size = len(conv)
    if top_k >= size:
        peaks = np.arange(size)
        w = np.ones(size) / size
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = (pv - pv.max()) / temp
        ew = np.exp(scaled)
        w_peaks = ew / ew.sum()
        w = np.zeros(size)
        w[peaks] = w_peaks
    
    sg = compute_subgradient_fft(f, w)
    return sg


def solve_linearized_lp(f, conv_f, B=None):
    """Solve the linearized LP to find a descent direction."""
    n = len(f)
    if B is None:
        B = build_convolution_matrix(f)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_trust_region_qp(f, conv_f, rho=0.1, M=None):
    """Solve the trust-region QP for a better descent direction."""
    n = len(f)
    if M is None:
        M = build_convolution_matrix(f)
    
    g = cp.Variable(n)
    t = cp.Variable()
    
    conv_fg = M @ g
    
    objective = cp.Minimize(t + (1.0 / rho) * cp.sum_squares(g))
    constraints = [
        t >= conv_f + 2 * conv_fg,
        cp.sum(g) == 0,
        g >= -f
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
        except Exception:
            return None
    
    if g.value is not None:
        g_val = np.maximum(g.value, -f)
        g_val = g_val - g_val.mean()
        return g_val
    return None


def line_search(f, direction, current_obj, max_alpha=1.0):
    """Adaptive line search along f + alpha*direction."""
    best_alpha = 0.0
    best_obj = current_obj

    for alpha in alphas:
        f_new = f + alpha * direction
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:

        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    # Refine
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.7)
        hi = min(max_alpha, best_alpha * 1.3)
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_new = f + mid * direction
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:

            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization strategy:
1. Diverse initialization including parametric families and parent interpolation
2. Trust-region QP (via CVXPY/ECOS) for targeted peak reduction
3. Linearized LP Frank-Wolfe for complementary directions
4. Smoothed subgradient descent with FFT-accelerated gradients
5. Simulated-annealing-style local search for fine-tuning
Key innovations: FFT-based subgradient computation, adaptive QP/LP switching,
and multi-timescale optimization with annealed perturbation restarts.

import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_convolution_matrix(f):
    """Build M matrix where (M@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    M = np.zeros((2 * n - 1, n))
    for j in range(n):
        M[j:j + n, j] = f
    return M


def compute_subgradient_fft(f, weights):
    """Compute weighted subgradient using FFT: sg[m] = 2*sum_k w[k]*f[k-m]."""
    n = len(f)
    conv = np.convolve(weights, f[::-1], mode='full')
    return 2.0 * conv[n - 1:n - 1 + n]


def smoothed_subgradient(f, conv, top_k=40, temp=0.0005):
    """Compute smoothed subgradient of max(conv) with softmax weighting."""
    n = len(f)
    size = len(conv)
    if top_k >= size:
        w = np.ones(size) / size
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = (pv - pv.max()) / temp
        ew = np.exp(scaled)
        w_peaks = ew / ew.sum()
        w = np.zeros(size)
        w[peaks] = w_peaks
    return compute_subgradient_fft(f, w)


def solve_linearized_lp(f, conv_f, B):
    """Solve: min t s.t. t >= conv_f + 2*B*g, sum(g)=0, g>=-f."""
    n = len(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_qp_direction(f, conv_f, rho, M):
    """Solve trust-region QP: min t + (1/rho)*||g||^2 s.t. t >= conv_f + 2*M@g."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    conv_fg = M @ g
    objective = cp.Minimize(t + (1.0 / rho) * cp.sum_squares(g))
    constraints = [
        t >= conv_f + 2.0 * conv_fg,
        cp.sum(g) == 0,
        g >= -f
    ]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, -f)
    g_val = g_val - g_val.mean()
    return g_val

    """Adaptive line search: find best alpha for f + alpha*direction."""
    best_alpha = 0.0
    best_obj = current_obj
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for alpha in alphas:
        f_new = f + alpha * direction
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.7)
        hi = min(1.0, best_alpha * 1.3)
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_new = f + mid * direction
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()

            if obj < best_obj:

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    n_base = len(parent) if parent is not None else 384
    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))
    for n_try in [n_base, n_base - 8, n_base + 8, 320, 448, 512]:
        if n_try < 64 or n_try > 600:
            continue
        inits = generate_inits(n_try, parent, rng)
        for i in range(min(4, len(inits))):
            candidates.append((f"n{n_try}_i{i}", inits[i]))
    for label, cand in candidates:
        if time.monotonic() > deadline - 500:
            break
        try_seq(cand, label)
    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")
    n_best = len(best_seq)
    print(f"Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: QP/LP Hybrid (primary optimization) ===
    print("\n=== Phase 2: QP/LP Hybrid ===")
    remaining = deadline - time.monotonic() - 250
    t_qp = min(remaining * 0.55, 700)
    if t_qp > 30:
        print(f"  QP/LP budget: {t_qp:.0f}s, n={n_best}")
        a_opt, _ = qp_lp_hybrid_optimize(best_seq, time_limit=t_qp * 0.65, rho_init=0.08, verbose=True)
        try_seq(a_opt, "qp_main")
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt, _ = qp_lp_hybrid_optimize(best_seq, time_limit=min(rem * 0.4, 250), rho_init=0.04, verbose=True)
            try_seq(a_opt, "qp_fine")
    print(f"Phase 2 best: {best_score:.8f}")

    # === Phase 3: Subgradient Polishing ===
    print("\n=== Phase 3: Subgradient ===")
    remaining = deadline - time.monotonic() - 120
    t_sg = min(remaining * 0.4, 280)
    if t_sg > 20:
        a_opt, _ = subgrad_optimize(best_seq, time_limit=t_sg * 0.5, lr=0.002, momentum=0.95, top_k=50, temp=0.0003)
        try_seq(a_opt, "sg_coarse")
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = subgrad_optimize(best_seq, time_limit=min(rem * 0.5, 130), lr=0.0008, momentum=0.97, top_k=80, temp=0.0001)
            try_seq(a_opt, "sg_fine")
    print(f"Phase 3 best: {best_score:.8f}")

    # === Phase 4: Local Search ===
    print("\n=== Phase 4: Local Search ===")
    remaining = deadline - time.monotonic() - 50
    t_ls = min(remaining * 0.6, 150)
    if t_ls > 10:
        a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng)
        try_seq(a_opt, "local")
    print(f"Phase 4 best: {best_score:.8f}")

    # === Phase 5: Length Exploration ===
    print("\n=== Phase 5: Length Exploration ===")
    rem = deadline - time.monotonic() - 25
    if rem > 20:
        for n_try in [n_best - 16, n_best + 16, n_best + 32]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = qp_lp_hybrid_optimize(px, time_limit=min(rem * 0.35, 40), rho_init=0.06)
                try_seq(a_opt, f"len{n_try}")
                rem = deadline - time.monotonic() - 20

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
