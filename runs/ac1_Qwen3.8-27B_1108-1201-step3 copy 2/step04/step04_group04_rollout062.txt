def fast_autoconv(a):

    n = len(a)

    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size, s=fft_size)[:size]
    return conv


def smooth_max(x, T):
    """Smooth approximation of max using log-sum-exp."""
    x_max = x.max()
    if x_max < -1e30:
        return 0.0
    shifted = (x - x_max) / T
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    return x_max + T * np.log(np.sum(np.exp(shifted)))


def smooth_max_grad(x, T):
    """Gradient of smooth max with respect to x."""
    x_max = x.max()
    shifted = (x - x_max) / T
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w /= w.sum()
    return w


def make_symmetric(a_half, n):
    """Create a symmetric sequence from the first half."""
    a = np.zeros(n)

        a[n//2:] = a_half[::-1]
    else:
        a[:n//2 + 1] = a_half
        a[n//2+1:] = a_half[-2::-1]
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) surrogate with adaptive temperature
2. Adam optimizer with cosine-decay learning rate
3. FFT-based gradient computation
4. Multi-length exploration (512, 768, 1024, 1536)
5. Multi-start with diverse initializations
6. Coordinate descent polishing
7. Peak-targeted local search
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def full_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]


def autoconv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = autoconv(a)
    return 2.0 * n * conv.max() / (s * s)


def smooth_max_and_grad(conv, T):
    """
    Compute smooth max of conv and its gradient (softmax weights).
    Returns (smooth_max_value, weights).
    """
    c = conv.max()
    shifted = (conv - c) / T
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    sm = c + T * np.log(exp_shifted.sum())
    return sm, w


def gradient_smooth_max(w, a):
    """
    Compute gradient of smooth_max(conv(a,a)) with respect to a.
    
    d(sm)/d(a[m]) = sum_k w[k] * d(conv[k])/d(a[m])
                   = sum_k w[k] * 2*a[k-m] (for valid k-m)
                   = 2 * sum_j a[j] * w[m+j]
    
    This is 2 * cross_correlation(a, w) where cross_corr[m] = sum_j a[j]*w[m+j].
    """
    n = len(a)
    # We want g[m] = sum_{j=0}^{n-1} a[j] * w[m+j] for m = 0, ..., n-1
    # where m+j < 2n-1 (length of w)
    # This is conv(a_reversed, w)[n-1 : 2n-1]
    a_rev = a[::-1].copy()
    # full convolution of a_rev (len n) and w (len 2n-1) has length 3n-2
    conv_aw = full_convolve(a_rev, w)
    g = conv_aw[n - 1 : 2 * n - 1]
    return 2.0 * g


def adam_step(params, grads, m, v, t, lr, beta1, beta2, eps):
    """One Adam step. Returns updated params, m, v."""
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * (grads ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    new_params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return new_params, m, v


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def optimize_smooth_max(a0, time_limit, T_schedule='cosine', lr_init=0.005,
                         beta1=0.9, beta2=0.999, eps=1e-8,
                         T_start=10.0, T_end=0.05, verbose=False):
    """
    Optimize using smooth max with Adam.
    a0: initial sequence (will be normalized)
    Returns (best_a, best_obj).
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    
    best_obj = compute_objective(a)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 300
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Check time
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break
        
        # Compute convolution and smooth max
        conv = autoconv(a)
        s = a.sum()
        
        # Temperature schedule
        progress = min(1.0, elapsed / time_limit)
        if T_schedule == 'cosine':
            T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        else:
            T = max(T_end, T_start * (1 - progress))
        
        # Learning rate schedule (cosine decay)
        lr = max(1e-7, lr_init * 0.5 * (1 + np.cos(np.pi * progress)))
        
        sm, w = smooth_max_and_grad(conv, T)
        grad = gradient_smooth_max(w, a)
        
        # Adam step
        t += 1
        a_new, m, v = adam_step(a, grad, m, v, t, lr, beta1, beta2, eps)
        a_new = normalize_nonneg(a_new)
        
        # Evaluate
        obj = compute_objective(a_new)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Still accept if not too bad (for exploration)
            if obj < best_obj * 1.01:
                a = a_new
            else:
                a = best_a.copy()
        
        if verbose and t % 1000 == 0:
            print(f"    iter {t}, obj={best_obj:.8f}, T={T:.4f}, lr={lr:.2e}, t={elapsed:.1f}s")
        
        if no_improve >= max_no_improve:
            # Perturb and reset
            rng = np.random.default_rng(t + int(elapsed * 1000))
            scale = 0.002 * (1 + no_improve / max_no_improve)
            a = best_a + rng.standard_normal(n) * scale
            a = normalize_nonneg(a)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            max_no_improve = min(1000, max_no_improve + 100)
    
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: try small adjustments to pairs of elements.
    Uses gradient information to guide which pairs to try.
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    conv = autoconv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    max_no_improve = 5000
    
    # Get peak locations
    n_peaks = min(10, 2 * n - 1)
    peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Try to move mass from positions contributing to peaks
        # to positions that don't contribute much
        for _ in range(max(1, int(200000 / n))):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # Pick a peak
            k = rng.choice(peak_indices)
            
            # The gradient at this peak: d(conv[k])/d(a[j]) = 2*a[k-j] for valid j
            # So positions with large a[k-j] contribute most to this peak
            # We want to reduce a[j] where a[k-j] is large
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            relevant = np.arange(j_start, j_end + 1)
            contributions = a[k - relevant]
            
            # Pick the position with highest contribution to reduce
            i_reduce = relevant[np.argmax(contributions)]
            
            # Pick a random position to increase
            i_increase = rng.integers(0, n)
            if i_increase == i_reduce:
                continue
            
            # Try a small transfer
            for delta in [0.0005, 0.001, 0.002, 0.005]:
                if a[i_reduce] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i_reduce] -= delta
                a_try[i_increase] += delta
                # Normalize (sum should still be 1)
                conv_try = autoconv(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    # Update peak indices
                    peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
                    break
            else:
                no_improve += 1
        
        if no_improve > max_no_improve:
            # Random restart from best
            scale = 0.001
            a = best_a + rng.standard_normal(n) * scale
            a = normalize_nonneg(a)
            conv = autoconv(a)
            best_obj = 2.0 * n * conv.max()
            peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
            no_improve = 0
            max_no_improve = min(20000, max_no_improve + 5000)
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    rng_seeded = np.random.default_rng(rng.integers(0, 2**31))
    inits['random'] = rng_seeded.dirichlet(np.ones(n) * 0.1)
    
    # Sine
    inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine'] /= inits['sine'].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # From parent (interpolated)
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) surrogate with adaptive temperature decay
2. Adam optimizer with cosine-decay learning rate and adaptive rest
3. FFT-based gradient computation (cross-correlation formula)
4. Multi-length exploration (512, 768, 1024, 1536)
5. Multi-start with diverse initializations including parent interpolation
6. Fast coordinate descent with first-order convolution update estimation
7. Peak-targeted local search with smart mass transfer
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def full_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]


def autoconv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = autoconv(a)
    return 2.0 * n * conv.max() / (s * s)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    c = conv.max()
    shifted = (conv - c) / T
    shifted = np.clip(shifted, -200, 200)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    sm = c + T * np.log(exp_s.sum())
    return sm, w


def gradient_from_weights(w, a):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    grad[m] = 2 * sum_j a[j] * w[m+j]
    Computed as 2 * full_convolve(a_rev, w)[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_aw = full_convolve(a_rev, w)
    g = conv_aw[n - 1:2 * n - 1]
    return 2.0 * g


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def adam_smooth_max_optimize(a0, time_limit, T_start=5.0, T_end=0.05,
                             lr_init=0.003, beta1=0.9, beta2=0.999,
                             eps=1e-8, max_no_improve=400, verbose=False):
    """
    Optimize using smooth max with Adam.
    Returns (best_a, best_obj).
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_obj = compute_objective(a)
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()
    log_interval = max(1, int(time_limit / 10))
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit - 0.5:
            break

        progress = min(1.0, elapsed / time_limit)

        # Temperature: cosine decay from T_start to T_end
        T = T_end + 0.5 * (T_start - T_end) * (1.0 + np.cos(np.pi * progress))
        T = max(T, 0.005)

        # Learning rate: cosine decay
        lr = max(1e-8, lr_init * 0.5 * (1.0 + np.cos(np.pi * progress)))

        # Compute convolution and smooth max
        conv = autoconv(a)
        _, w = smooth_max_and_weights(conv, T)
        grad = gradient_from_weights(w, a)

        # Adam step
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new)

        # Evaluate
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            no_improve += 1
            continue

        conv_new = autoconv(a_new)
        obj_new = 2.0 * n * conv_new.max() / (s_new * s_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Accept with some probability for exploration early on
            if no_improve < 100 and obj_new < best_obj * 1.005:
                a = a_new
            else:
                a = best_a.copy()

        iter_count += 1
        if verbose and iter_count % log_interval == 0:
            print(f"      iter={iter_count}, obj={best_obj:.8f}, T={T:.4f}, lr={lr:.2e}, t={elapsed:.1f}s")

        if no_improve >= max_no_improve:
            # Perturb from best
            rng_local = np.random.default_rng(t + 12345)
            scale = 0.002 * (1.0 + min(no_improve / max_no_improve, 3.0))
            a = best_a + rng_local.standard_normal(n) * scale
            a = normalize_nonneg(a)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            max_no_improve = min(2000, max_no_improve + 100)

    return best_a, best_obj


def fast_coord_polish(a0, time_limit, verbose=False):
    """
    Fast coordinate descent using first-order convolution update estimation.
    When a[i] changes by +d and a[j] changes by -d, the change in conv[k] is
    approximately d*(2*a[k-i] - 2*a[k-j]).
    """
    a = normalize_nonneg(a0.astype(np.float64))
    n = len(a)
    conv = autoconv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    max_no_improve = 10000

    # Track peak locations
    n_peaks = min(20, 2 * n - 1)
    peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]

    # Precompute: for each peak k, which positions contribute most
    # contribution of position i to peak k: a[k-i] (if valid)

    while time.monotonic() - t0 < time_limit - 0.5:
        if no_improve > max_no_improve:
            # Reset with perturbation
            scale = 0.001 * (1 + no_improve / max_no_improve)
            a = best_a + rng.standard_normal(n) * scale
            a = normalize_nonneg(a)
            conv = autoconv(a)
            best_obj = 2.0 * n * conv.max()
            peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
            no_improve = 0
            max_no_improve = min(50000, max_no_improve * 2)
            continue

        # Pick a peak
        k = int(rng.choice(peak_idx))

        # Positions contributing to this peak
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue

        relevant = np.arange(j_start, j_end + 1)
        # Contribution of a[i] to conv[k] is proportional to a[k-i]
        contribs = a[k - relevant]
        # Pick the position with highest contribution
        i_src = int(relevant[np.argmax(contribs)])

        # Destination: random position with low contribution to this peak
        i_dst = int(rng.integers(0, n))
        if i_dst == i_src:
            continue

        # Check that moving mass won't increase this peak too much
        # New contribution from i_dst to peak k: a[k - i_dst] (if valid)
        k_minus_dst = k - i_dst
        if 0 <= k_minus_dst < n:
            dst_contrib = a[k_minus_dst]
        else:
            dst_contrib = 0.0

        src_contrib_val = a[k - i_src] if 0 <= k - i_src < n else 0.0

        if dst_contrib >= src_contrib_val:
            # Moving mass here might not help this peak; skip sometimes
            if rng.random() > 0.3:
                no_improve += 1
                continue

        # Try small transfers
        for delta in [0.0005, 0.001, 0.002, 0.004]:
            if a[i_src] < delta:
                continue
            a_try = a.copy()
            a_try[i_src] -= delta
            a_try[i_dst] += delta

            # Fast check: estimate new max without full convolution
            # Only update affected conv values
            # conv changes at positions affected by i_src and i_dst
            new_conv = conv.copy()
            for ii in [i_src, i_dst]:
                d = delta if ii == i_dst else -delta
                for jj in range(n):
                    k_pos = ii + jj
                    if k_pos < 2 * n - 1:
                        new_conv[k_pos] += 2.0 * d * a[jj] + d * d  # approximate

            new_max = new_conv.max()
            new_obj_est = 2.0 * n * new_max

            if new_obj_est < best_obj * 0.999:
                # Full verification
                a_try_s = a_try.sum()
                conv_verify = autoconv(a_try)
                obj_verify = 2.0 * n * conv_verify.max() / (a_try_s * a_try_s)
                if obj_verify < best_obj - 1e-12:
                    a = a_try
                    conv = conv_verify
                    best_obj = obj_verify
                    best_a = a.copy()
                    peak_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
                    no_improve = 0
                    break
                else:
                    no_improve += 1
                    break
        else:
            no_improve += 1

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Random (Dirichlet)
    r = np.random.default_rng(rng.integers(0, 2 ** 31))
    inits['random'] = r.dirichlet(np.ones(n) * 0.05)

    # Sine envelope
    sinv = (np.sin(np.pi * x / n) ** 2 + 0.005)
    inits['sine'] = sinv / sinv.sum()

    # Bimodal (edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Add some middle mass
    mid = n // 4
    bi[n // 2 - mid:n // 2 + mid] += 0.5
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.12 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Triangular (peak in middle)
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / n
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            p = p / ps
            if len(p) == n:
                inits['parent'] = p.copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps

    return inits
