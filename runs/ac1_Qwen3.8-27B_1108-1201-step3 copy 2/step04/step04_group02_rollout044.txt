"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved Frank-Wolfe method
with Nesterov-style momentum, adaptive LP targets, peak equalization polishing,
and multi-scale exploration. Key improvements over prior work:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A @ g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'mip_rel_gap': 0.0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def line_search_smooth(f, g, M_current, max_iters=20):
    """Golden section + bisection line search for alpha in [0,1]."""
    def obj_at_alpha(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = obj_at_alpha(c_)
    fd = obj_at_alpha(d_)

    for _ in range(max_iters):
        if b - a < 1e-8:
            break
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = obj_at_alpha(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = obj_at_alpha(d_)

    best_alpha = (a + b) / 2.0
    # Fine bisection around best
    lo, hi = max(0, best_alpha - 0.02), min(1.0, best_alpha + 0.02)
    best_val = obj_at_alpha(best_alpha)
    for _ in range(12):
        if hi - lo < 1e-9:
            break
        mid = (lo + hi) / 2.0
        val = obj_at_alpha(mid)
        if val < best_val:
            best_alpha = mid
            best_val = val
            lo = mid
        else:
            hi = mid
    return best_alpha, best_val

def peak_equalize(f, n_peaks=5, n_iters=50):
    """Try to equalize the top n_peaks of the convolution via small adjustments."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    for it in range(n_iters):
        # Find top peaks
        top_idx = np.argsort(conv)[-n_peaks:]
        peak_vals = conv[top_idx]
        target = peak_vals.mean()

        # Compute subgradient pointing towards reducing peaks above target
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            if conv[k] <= target:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx]

        # Scale subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Try a few step sizes
        best_step = 0.0
        best_val = best_obj
        for step in [0.001, 0.002, 0.005, 0.01, 0.02]:
            ft = project_simplex(f - step * sg)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_step = step

        if best_step < 1e-8:
            break

        f = f - best_step * sg
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=200, temp=0.00005):
    """Fine subgradient polishing with very small steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Compute subgradient with softmax over top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 300:
            rng = np.random.default_rng(it + 200)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_f, best_obj


def two_bump_init(n, alpha=0.3):
    """Create a two-bump initialization: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - 0) ** 2 / (2 * alpha ** 2)) + np.exp(-(x - 1) ** 2 / (2 * alpha ** 2))
    a = a + 0.01  # avoid exact zeros
    return a / a.sum()


def cosine_init(n, power=2):
    """Cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** (2 * power) + 0.005
    return a / a.sum()


def edge_concentrated_init(n, beta=2.0):
    """Concentrate mass at edges."""
    x = np.linspace(0, 1, n)
    a = x ** beta + (1 - x) ** beta + 0.01
    return a / a.sum()


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Two-bump (various widths)
    for alpha in [0.15, 0.25, 0.35, 0.5]:
        inits.append(two_bump_init(n, alpha))

    # Cosine window
    for power in [1, 2, 3]:
        inits.append(cosine_init(n, power))

    # Edge concentrated
    for beta in [1.5, 2.0, 3.0]:
        inits.append(edge_concentrated_init(n, beta))

    # Linear ramps
    x = np.arange(n, dtype=np.float64)
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Parent-based
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with two-bump
            blended = 0.6 * (px / ps) + 0.4 * two_bump_init(n, 0.3)
            inits.append(blended / blended.sum())
            # Blend with cosine
            blended2 = 0.5 * (px / ps) + 0.5 * cosine_init(n, 2)
            inits.append(blended2 / blended2.sum())

    return inits

            a_eq2, obj_eq2 = peak_equalize(a_eq, n_peaks=5, n_iters=50)
            try_seq(a_eq2, "peak_eq2")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 60),
                                 lr=0.0008, mom=0.96, top_k=max(50, n_best // 4), temp=0.00003)
            try_seq(a_opt, "polish_sg")

        # FW polish with very tight M
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = fw_optimize(best_seq, time_limit=min(rem, 50), adaptive_m=False)
            try_seq(a_opt, "polish_fw")

        # Micro perturbations + quick polish
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            for p in range(5):
                rem = deadline - time.monotonic() - 8
                if rem < 6:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0008)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 10),
                                     lr=0.0004, mom=0.97, top_k=max(30, n_best // 5), temp=0.00001)
                try_seq(a_opt, f"micro_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK used: score={best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
