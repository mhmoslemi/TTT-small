import numpy as np
import time

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def normalize(a):

    s = a.sum()

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential decay (both directions)
    for beta in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Cosine
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(freq * np.pi * x / n)) + 0.1
        inits.append(c / c.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.15*n))**2)
    inits.append(g / g.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2*np.pi*x/n)) + 0.01
    inits.append(rc / rc.sum())
    
    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Parent (if provided and same length)
    if parent is not None and len(parent) == n:
        p = normalize(parent)
        inits.append(p)
        for scale in [0.005, 0.01, 0.02]:
            perturbed = p + rng.standard_normal(n) * scale
            inits.append(normalize(perturbed))
    
    return inits

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines FFT-based power spectrum flattening (Gerchberg-Saxton style),
exponentiated gradient with adaptive temperature/reheating, coordinate
descent polish, and multi-length multi-start exploration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Project to non-negative simplex."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    return sg


def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, verbose=False):
    """Exponentiated gradient with adaptive temperature and running average."""
    a = normalize(a0)
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1
        a = a_new
        c = c_new
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        eta *= eta_decay
        temp *= temp_decay
        if no_improve > 1500:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.001
            a = normalize(a)
            c = fast_convolve(a)
            no_improve = 0
            temp = max(temp, temp0 * 0.05)
            eta = eta0 * 0.05
            avg_a = a.copy()
            count = 0
    # Check running average one more time
    if count > 0:
        c_avg = fast_convolve(avg_a)
        if c_avg.max() < best_obj - 1e-12:
            best_obj = c_avg.max()
            best_a = avg_a.copy()
    return best_a.copy(), best_obj


def power_spectrum_optimize(a0, time_limit, alpha=0.05, verbose=False):
    """Gerchberg-Saxton style alternating between frequency flattening and time-domain projection."""
    a = normalize(a0)
    n = len(a)
    fft_size = 2 * n
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: geometric mean of non-DC power
        non_dc = power[1:]
        target = np.exp(np.log(non_dc + 1e-300).mean())
        # Compress dynamic range toward target
        scale = (target / (power + 1e-300)) ** alpha
        A_new = A * scale
        A_new[0] = A[0]  # Preserve DC
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        c = fast_convolve(a_new)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 200 == 0:
            print(f"    PS iter {it}: obj={best_obj:.10f}")
    return best_a.copy(), best_obj


def coordinate_descent_polish(a0, time_limit, max_rounds=200):
    """Coordinate-wise optimization trying multiplicative factors."""
    a = normalize(a0)
    n = len(a)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    no_improve = 0
    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.18, 1.43, 2.0, 0.6, 1.67]
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-12:
                for v in [1e-4, 5e-4, 1e-3, 5e-3]:
                    a[i] = v
                    a_norm = normalize(a)
                    c2 = fast_convolve(a_norm)
                    if c2.max() < best_obj - 1e-12:
                        best_obj = c2.max()
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        improved = True
                        break
                if not improved:
                    a[i] = 0.0
                continue
            for factor in factors:
                a[i] = old_val * factor
                a_norm = normalize(a)
                c2 = fast_convolve(a_norm)
                if c2.max() < best_obj - 1e-12:
                    best_obj = c2.max()
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                else:
                    a[i] = old_val
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
    return best_a.copy(), best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    for beta in [0.3, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(freq * np.pi * x / n)) + 0.05
        inits.append(c / c.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(rc / rc.sum())
    for alpha_d in [0.3, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha_d)
        inits.append(r)
    if parent is not None and len(parent) == n:
        p = normalize(parent)
        inits.append(p)
        for scale in [0.002, 0.005, 0.01, 0.02]:
            perturbed = p + rng.standard_normal(n) * scale
            inits.append(normalize(perturbed))
    return inits


def interpolate_sequence(parent, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(parent)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, parent)
    return normalize(a_new)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: combines power spectrum flattening, exponentiated gradient,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best_score={best_score:.8f}")

    def check_and_update(a, label):
        """Evaluate and update best if improved."""
        nonlocal best_seq, best_score
        if a is None:
            return
        a_clean = np.clip(a, 0.0, 1000.0)
        if a_clean.sum() < 1e-10:
            return
        try:
            score = float(evaluate_sequence(a_clean.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_clean.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_clean)})")

    # === Phase 1: Power spectrum flattening (15% time) ===
    print(f"\n=== Phase 1: Power Spectrum Flattening ===")
    rem = deadline - time.monotonic()
    t_alloc = min(180, rem * 0.15)
    if t_alloc > 15:
        print(f"  Alloc: {t_alloc:.0f}s")
        # Try different alpha values
        for alpha_val in [0.03, 0.05, 0.08]:
            rem2 = deadline - time.monotonic()
            t_sub = min(45, rem2 * 0.3)
            if t_sub < 5:
                break
            a_opt, _ = power_spectrum_optimize(parent, time_limit=t_sub, alpha=alpha_val, verbose=False)
            check_and_update(a_opt, f"ps_a{alpha_val}")
        # Also try from current best
        if best_seq is not None and best_seq is not parent:
            rem3 = deadline - time.monotonic()
            if rem3 > 10:
                a_opt, _ = power_spectrum_optimize(best_seq, time_limit=min(30, rem3 * 0.2), alpha=0.04, verbose=False)
                check_and_update(a_opt, "ps_best")

    # === Phase 2: EG from parent (25% time) ===
    print(f"\n=== Phase 2: EG from Parent ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    t_alloc = min(400, rem * 0.25)
    if t_alloc > 30:
        print(f"  Alloc: {t_alloc:.0f}s")
        # Main EG run with good parameters
        a_opt, _ = exp_gradient_optimize(parent, time_limit=t_alloc * 0.6,
                                         eta0=1.2, temp0=0.003,
                                         temp_decay=0.9996, eta_decay=0.99995,
                                         top_k_frac=0.2, verbose=True)
        check_and_update(a_opt, "eg_main")

        # Second EG with different params from current best
        rem2 = deadline - time.monotonic()
        t_sub = min(120, rem2 * 0.5)
        if t_sub > 15:
            start = best_seq if best_seq is not None else parent
            a_opt, _ = exp_gradient_optimize(start, time_limit=t_sub,
                                             eta0=0.5, temp0=0.001,
                                             temp_decay=0.9998, eta_decay=0.99997,
                                             top_k_frac=0.4, verbose=True)
            check_and_update(a_opt, "eg_main2")

    # === Phase 3: EG from PS result (15% time) ===
    print(f"\n=== Phase 3: EG from Power Spectrum Result ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    t_alloc = min(250, rem * 0.15)
    if t_alloc > 20:
        print(f"  Alloc: {t_alloc:.0f}s")
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc,
                                         eta0=0.8, temp0=0.002,
                                         temp_decay=0.9997, eta_decay=0.99995,
                                         top_k_frac=0.3, verbose=True)
        check_and_update(a_opt, "eg_ps")

    # === Phase 4: Multi-length exploration (20% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    t_total_phase4 = min(300, rem * 0.2)

    # Try lengths near parent and some others
    candidate_lengths = []
    for l in [n_parent - 64, n_parent - 32, n_parent + 32, n_parent + 64]:
        if 64 <= l <= 1024:
            candidate_lengths.append(l)
    candidate_lengths.extend([256, 320, 448, 512, 640])
    # Remove duplicates and keep unique
    seen = set()
    unique_lengths = []
    for l in candidate_lengths:
        if l not in seen and l != n_parent:
            seen.add(l)
            unique_lengths.append(l)
    unique_lengths.sort(key=lambda l: abs(l - n_parent))
    unique_lengths = unique_lengths[:4]

    for n_try in unique_lengths:
        rem2 = deadline - time.monotonic()
        t_alloc_l = min(60, t_total_phase4 / max(1, len(unique_lengths)))
        if rem2 < 10 or t_alloc_l < 8:
            break
        print(f"  Trying n={n_try}, alloc={t_alloc_l:.0f}s")
        a_init = interpolate_sequence(parent, n_try)
        a_opt, _ = exp_gradient_optimize(a_init, time_limit=t_alloc_l,
                                         eta0=1.0, temp0=0.005,
                                         temp_decay=0.9997, eta_decay=0.99995,
                                         top_k_frac=0.2, verbose=False)
        check_and_update(a_opt, f"len{n_try}")

        # Also try PS on this length
        rem3 = deadline - time.monotonic()
        if rem3 > 10:
            a_ps, _ = power_spectrum_optimize(a_init, time_limit=min(15, rem3 * 0.2), alpha=0.05, verbose=False)
            check_and_update(a_ps, f"ps_len{n_try}")

    # === Phase 5: Intensive refinement (15% time) ===
    print(f"\n=== Phase 5: Intensive Refinement ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    t_alloc = min(250, rem * 0.15)
    if t_alloc > 20:
        print(f"  Alloc: {t_alloc:.0f}s")
        # Very fine EG
        a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t_alloc * 0.5,
                                         eta0=0.2, temp0=0.0005,
                                         temp_decay=0.9999, eta_decay=0.99999,
                                         top_k_frac=0.5, verbose=True)
        check_and_update(a_opt, "eg_fine5")

        # Another PS pass on the refined result
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_ps, _ = power_spectrum_optimize(best_seq, time_limit=min(40, rem2 * 0.3), alpha=0.02, verbose=False)
            check_and_update(a_ps, "ps_fine5")

    # === Phase 6: Coordinate descent polish (10% time) ===
    print(f"\n=== Phase 6: CD Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    t_alloc = min(120, rem * 0.10)
    if t_alloc > 5:
        print(f"  Alloc: {t_alloc:.0f}s")
        a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, max_rounds=100)
        check_and_update(a_opt, "cd_p6")

    # === Phase 7: Final combined pass (remaining time) ===
    print(f"\n=== Phase 7: Final Pass ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 10:
        # One more fine EG
        t1 = min(60, rem * 0.5)
        if t1 > 5:
            a_opt, _ = exp_gradient_optimize(best_seq, time_limit=t1,
                                             eta0=0.1, temp0=0.0003,
                                             temp_decay=0.99995, eta_decay=0.99999,
                                             top_k_frac=0.5, verbose=False)
            check_and_update(a_opt, "eg_final7")

        # Final CD
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            t2 = min(30, rem2 * 0.5)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t2, max_rounds=50)
            check_and_update(a_opt, "cd_final7")

        # Final PS
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_ps, _ = power_spectrum_optimize(best_seq, time_limit=min(15, rem3 * 0.5), alpha=0.01, verbose=False)
            check_and_update(a_ps, "ps_final7")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
