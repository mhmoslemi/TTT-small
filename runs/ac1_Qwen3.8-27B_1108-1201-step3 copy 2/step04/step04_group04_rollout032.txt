```python
        peaks = [int(np.argmax(conv))]

        # Softmax weights
        peak_vals = conv[peak_idx]

            if 0 <= km < n:
```

```python
def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) > (cssv - 1)), 1, 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lp_direction(a, M, C=None, top_k_constraint=None):

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=5, temp_init=0.001):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj

def flatten_autocorrelation(a, time_limit, target_ratio=0.95):
    """
    Try to flatten the autocorrelation by reducing peaks and raising valleys.
    This is a post-processing step.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        # Find the peak
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Find a "valley" to borrow from
        # We want to move mass from positions that contribute to the peak
        # to positions that don't
        k = peak_idx
        # Positions that contribute to conv[k]: m such that 0 <= k-m < n
        contributing = set()
        for m in range(max(0, k-n+1), min(n-1, k)+1):
            contributing.add(m)
            if k - m != m:
                contributing.add(k - m)
        
        # Find positions NOT contributing
        non_contributing = [m for m in range(n) if m not in contributing]
        
        if not non_contributing:
            break
        
        # Try to move small mass from a contributing position to a non-contributing one
        contrib_list = list(contributing)
        if not contrib_list:
            break
        
        i = contrib_list[rng.integers(0, len(contrib_list))]
        j = non_contributing[rng.integers(0, len(non_contributing))]
        
        if a[i] < a[j] * 0.1:
            continue
        
        for delta_factors in [0.01, 0.005, 0.002]:
            delta = a[i] * delta_factors
            if delta < 1e-10:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0:
                continue
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-10:
                a = a_try.copy()
                best_obj = obj_try
                best_a = a.copy()
                conv = conv_try
                break
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-based
    for freq in [1, 2, 3]:
        key = f'sine{freq}'
        inits[key] = (np.sin(np.pi * freq * x / n) ** 2 + 0.01)
        inits[key] /= inits[key].sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n-1)) / (n - 1)
    inits['triangular'] = tri / tri.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits
```

```python
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            a_lp = result.x[:n]
            t_lp = result.x[n]
            return t_lp, a_lp
    except Exception:
        pass
    return None, None
```

This LP has (2n-1) + 1 = 2n constraints and n+1 variables. For n=768, that's 1537 constraints and 769 variables. Should be fast.

The key insight: if the LP solution a_lp is close to a feasible point of the original problem (i.e., conv(a_lp, a_lp).max() ≈ t_lp), then we're close to optimal. If not, we need to iterate (DCA / outer approximation).

**DCA (DC Algorithm) approach:**
1. Start with a^0
2. At iteration k:
   a. Solve LP: minimize t s.t. conv(a^k, a) <= t, sum(a)=1, a>=0 → get a^{k+1}, t^{k+1}
   b. Check: if conv(a^{k+1}, a^{k+1}).max() ≈ t^{k+1}, we're at (or near) a local optimum
   c. Otherwise, update a^k ← a^{k+1} (or a^{k+1} + α(a^k - a^{k+1}) for some α)
3. Return best solution found

This is essentially the "outer approximation" or "successive linear programming" method. It should converge to a local minimum of the original non-convex problem.

The advantage over the previous approach:
- More principled step sizing (take the full LP step, or a fraction of it)
- The LP bound gives a certificate of quality
- Can detect convergence
