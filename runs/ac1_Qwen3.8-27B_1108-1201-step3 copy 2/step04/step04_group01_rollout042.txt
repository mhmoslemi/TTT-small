```python
def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)

    """Compute (w *~ f)_i = sum_m w[i+m] * f[m]"""
    n = len(f)
    m = len(w)  # should be 2n-1
    total_len = m + n - 1
    fft_size = 1 << (total_len - 1).bit_length()

    conv = np.fft.irfft(W * F_rev, fft_size)
    return conv[n-1:n-1+n]

    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization
of a log-sum-exp smoothed objective, with multi-start strategy and
adaptive beta scheduling. Combines smooth gradient-based optimization
with Frank-Wolfe LP refinement for the non-smooth regime.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlation(w, f):
    """Compute (w *~ f)_i = sum_m w[i+m] * f[m] for i=0..n-1.
    w has length 2n-1, f has length n.
    """
    n = len(f)
    m = len(w)
    total_len = m + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F_rev = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F_rev, fft_size)
    return conv[n-1:n-1+n]


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_grad_x(x, beta, n):
    """
    LSE-smoothed objective and gradient w.r.t. log-parametrization x.
    f = softmax(x), J_smooth = 2n/beta * logsumexp(beta * conv(f))
    """
    f = softmax(x)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    w = exp_bc / w_sum
    lse_val = bc_max + np.log(w_sum)
    obj = 2.0 * n * lse_val / beta
    
    # Gradient w.r.t. f (treating f as free, sum f = 1)
    grad_f = 2.0 * n * cross_correlation(w, f)
    # Gradient w.r.t. x: since J is scale-invariant, grad_f · f = 0
    # So dJ/dx_i = f_i * grad_f_i
    grad_x = f * grad_f
    
    return obj, grad_x


def lbfgs_lse(f0, time_limit, beta_start=5.0, beta_end=2000.0,
              max_iter_per_beta=200, verbose=False, seed=42):
    """
    Optimize using L-BFGS on LSE-smoothed objective with increasing beta.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    x0 = np.log(np.maximum(f, 1e-30))
    
    t0 = time.monotonic()
    beta = beta_start
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit and beta <= beta_end:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        # Dynamically adjust max_iter based on remaining time
        iter_budget = max(30, min(max_iter_per_beta, int(remaining * 500)))
        
        def func_and_grad(x):
            return lse_obj_grad_x(x, beta, n)
        
        try:
            res = minimize(
                func_and_grad,
                x0,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': iter_budget, 'gtol': 1e-10, 'ftol': 1e-14,
                         'maxfun': iter_budget * 3}
            )
            x_opt = res.x
        except Exception:
            x_opt = x0
        
        f_opt = softmax(x_opt)
        f_opt = np.maximum(f_opt, 0.0)
        s_opt = f_opt.sum()
        if s_opt > 1e-15:
            f_opt = f_opt / s_opt
        
        conv = fast_convolve(f_opt)
        true_obj = 2.0 * n * conv.max()
        
        improved = False
        if true_obj < best_obj - 1e-12:
            best_obj = true_obj
            best_f = f_opt.copy()
            improved = True
            x0 = np.log(np.maximum(f_opt, 1e-30))
            if verbose:
                print(f"    LBFGS-LSE: obj={best_obj:.10f}, beta={beta:.1f}, "
                      f"iters={res.nit}, elapsed={elapsed:.1f}s")
        
        if not improved and beta > beta_start * 4:
            # Try a small perturbation
            x0 = x0 + rng.standard_normal(n) * 0.01
            f_tmp = softmax(x0)
            f_tmp = f_tmp / f_tmp.sum()
            obj_tmp = 2.0 * n * fast_convolve(f_tmp).max()
            if obj_tmp < best_obj:
                best_obj = obj_tmp
                best_f = f_tmp.copy()
        
        # Increase beta
        beta *= 1.8
        
    # Final: try the true max gradient (subgradient) for a few steps
    # This helps when beta is very large
    f = best_f.copy()
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(f)
        k_max = np.argmax(conv)
        # Subgradient of max(conv) at k_max
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        sg = np.zeros(n)
        for j in range(js, je + 1):
            sg[j] += f[k_max - j]
            if k_max - j != j:
                sg[k_max - j] += f[j]
        # Wait, d(conv_k)/d(f_i) = f_{k-i} + f_{k-i} = 2*f_{k-i} if 0 <= k-i < n
        # So subgradient of max(conv) is 2*f[k_max - i] (for valid i)
        sg = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                sg[i] = 2.0 * f[idx]
        
        # Projected subgradient step
        lr = 1.0 / (np.linalg.norm(sg) ** 2 + 1e-30) * 0.5
        f_new = f - lr * sg
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        obj_new = 2.0 * n * fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller step
            lr *= 0.5
            f_new = f - lr * sg
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new = f_new / s_new
                obj_new = 2.0 * n * fast_convolve(f_new).max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
    
    return best_f, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_step(f, M, A_cache, timeout=15):
    """Single FW-LP step."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
                return g
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, max_iters=100):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A_cache = build_lp_matrix(f)
        g = fw_lp_step(f, M, A_cache, timeout=10)
        if g is None:
            break
        
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        if best_alpha > 0:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj:
                best_obj = M
                best_f = f.copy()
    
    return best_f, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. L-BFGS on LSE-smoothed objective with continuation method (main driver)
2. Frank-Wolfe with LP oracle for non-smooth refinement
3. Multi-start with parametric families and random perturbations
4. Subgradient polish for final convergence
Key innovation: L-BFGS converges superlinearly vs. plain gradient descent,
enabling many more restarts and finer beta continuation within time budget.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlation(w, f):
    """Compute c[i] = sum_m w[i+m] * f[m] for i=0..n-1.
    w has length 2n-1, f has length n.
    """
    n = len(f)
    f_rev = f[::-1]
    L = len(w) + n - 1
    fft_size = 1 << (L - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return conv[n - 1:n - 1 + n]


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_grad_x(x, beta, n):
    """LSE-smoothed objective and gradient w.r.t. log-parametrization x."""
    f = softmax(x)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    w = exp_bc / w_sum
    lse_val = bc_max + np.log(w_sum)
    obj = 2.0 * n * lse_val / beta
    grad_f = 2.0 * n * cross_correlation(w, f)
    grad_x = f * grad_f
    return obj, grad_x


def true_obj(f):
    """Compute true objective 2*n*max(conv) for normalized f."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return float('inf')
    f_norm = f / s
    conv = fast_convolve(f_norm)
    return 2.0 * n * conv.max()


def lbfgs_lse(f0, time_limit, beta_start=5.0, beta_end=5000.0,
              max_iter_per_beta=200, beta_multiplier=1.7, verbose=False, seed=42):
    """L-BFGS on LSE-smoothed objective with continuation (increasing beta)."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()

    x0 = np.log(np.maximum(f, 1e-30))

    stagnant = 0

    while time.monotonic() - t0 < time_limit and beta <= beta_end:
        remaining = time_limit - (time.monotonic() - t0)
        iter_budget = max(30, min(max_iter_per_beta, int(remaining * 800)))

        def func_and_grad(x):
            return lse_obj_grad_x(x, beta, n)

        try:
            res = minimize(
                func_and_grad, x0, method='L-BFGS-B', jac=True,
                options={'maxiter': iter_budget, 'gtol': 1e-10,
                         'ftol': 1e-14, 'maxfun': iter_budget * 4}
            )
            x_opt = res.x
        except Exception:
            x_opt = x0

        f_opt = softmax(x_opt)

            stagnant = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      LBFGS-LSE: obj={best_obj:.10f}, beta={beta:.1f}, "
                      f"nit={res.nit}, t={elapsed:.1f}s")
        else:
            stagnant += 1
            if stagnant >= 3 and beta > beta_start * 3:
                # Perturb to escape local minimum

                stagnant = 0
                beta *= 1.3  # Increase beta more aggressively after stagnation
                continue

        beta *= beta_multiplier

    # Subgradient polish
    f = best_f.copy()
    for _ in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        n_local = len(f)
        conv = fast_convolve(f)
        k_max = np.argmax(conv)
        sg = np.zeros(n_local)
        for i in range(n_local):

            break

        f_new = f - lr * sg
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        obj_new = true_obj(f_new)

            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:

            f_new2 = f - lr * sg
            f_new2 = np.maximum(f_new2, 0.0)

                    best_f = f_new2.copy()
                    f = f_new2
            if lr < 1e-15:
                break

    return best_f, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fw_lp_optimize(f0, time_limit, max_iters=80):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A_cache = build_lp_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        g = None
        try:
            res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 12})
            if res.success:
                cand = np.maximum(res.x, 0.0)
                if cand.sum() > 1e-15:
                    g = cand / cand.sum()
        except Exception:
            pass

        if g is None:
            break
        if np.max(np.abs(g - f)) < 1e-11:
            break

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj


def parametric_init(n, params, family):
    """Generate a parametric initialization."""
    i = np.arange(n)
    if family == 'cosine':
        a, b, c = params
        f = 1.0 + a * np.cos(2 * np.pi * i / n) + b * np.cos(4 * np.pi * i / n) + c * np.cos(6 * np.pi * i / n)
    elif family == 'gaussian':
        alpha = params[0]
        f = np.exp(-alpha * ((i - (n - 1) / 2.0) / (n / 2.0)) ** 2)
    elif family == 'polynomial':
        p = params[0]
        x = 2.0 * i / (n - 1) - 1.0
        f = (1.0 - x ** 2) ** p
    elif family == 'lorentzian':
        alpha = params[0]
        f = 1.0 / (1.0 + alpha * np.abs(i - (n - 1) / 2.0) / n)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    return f
```
