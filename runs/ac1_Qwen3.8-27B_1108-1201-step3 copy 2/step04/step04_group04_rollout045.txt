from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) is already normalized."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M, C=None):
    """
    Solve LP: max sum(g) s.t. C@g <= M, g >= 0.
    Returns (g, g_sum, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success:
            g = np.maximum(result.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy().astype(np.float64), 1.0, False


def line_search_bisection(a, d, max_iters=40):
    """
    Find t in [0,1] minimizing max(conv(a + t*d, a + t*d)).
    Uses bisection on the derivative of the max function.
    Returns (best_t, best_conv).
    """
    # Precompute
    p = fast_convolve(a)  # conv(a,a)
    q = fast_cross_convolve(a, d)  # conv(a,d)
    r = fast_convolve(d)  # conv(d,d)

    # f_k(t) = p[k] + 2t*q[k] + t^2*r[k]
    def max_conv_at_t(t):
        vals = p + 2.0 * t * q + t * t * r
        return vals.max()

    # Coarse grid search first
    grid = np.linspace(0, 1, 33)
    best_t = 0.0
    best_val = max_conv_at_t(0)
    for t in grid[1:]:
        val = max_conv_at_t(t)
        if val < best_val:
            best_val = val
            best_t = t

    # Refine with golden section in neighborhood
    idx = int(best_t * 32)
    lo = max(0.0, grid[max(0, idx - 1)])
    hi = min(1.0, grid[min(32, idx + 1)])

    if hi - lo > 1e-6:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = max_conv_at_t(c1)
        f2 = max_conv_at_t(c2)
        for _ in range(max_iters):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = max_conv_at_t(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = max_conv_at_t(c2)
        best_t = (lo + hi) / 2.0
        best_val = max_conv_at_t(best_t)

    return best_t, p + 2.0 * best_t * q + best_t * best_t * r


def smooth_gradient(a, conv, top_k=50, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    # Use top_k peaks with softmax weighting
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad * 2.0  # factor of 2 from d/dda [a^T Q a]


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=50, temp_init=0.002, temp_rate=1.004):
    """Nesterov-accelerated gradient descent with adaptive log-sum-exp smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = smooth_gradient(y, conv_y, top_k=top_k, temp=temp)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates
        lr *= 0.9998
        temp *= temp_rate
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_step(a, conv, n):
    """
    One LP-based step with bisection line search.
    Returns (a_new, improved, new_conv).
    """
    M = conv.max()

    C = build_conv_matrix(a)
    g, g_sum, success = lp_direction(a, M * 1.00005, C)

    if not success or g_sum <= 1.0 + 1e-9:
        return a.copy(), False, conv

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False, conv

    # Bisection line search
    best_t, best_conv = line_search_bisection(a, d)

    if best_t > 0.01:
        a_new = a + best_t * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            return a.copy(), False, conv

        new_conv = fast_convolve(a_new)
        new_obj = 2.0 * n * new_conv.max()
        old_obj = 2.0 * n * M

        if new_obj < old_obj - 1e-12:
            return a_new, True, new_conv

    # Try a few fixed steps as fallback
    for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < old_obj - 1e-12:
            return a_try, True, conv_try

    return a.copy(), False, conv


def lp_optimize(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization with adaptive switching to gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        a_new, improved, conv = lp_step(a, conv, n)

        if improved:
            a = a_new
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            # Try gradient burst
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 10:
                t_grad = min(5.0, remaining * 0.1)
                a_tmp, obj_tmp = nesterov_optimize(best_a, time_limit=t_grad,
                                                   lr_init=0.008, momentum=0.93,
                                                   top_k=60, temp_init=0.001, temp_rate=1.005)
                if obj_tmp < best_obj:
                    best_obj = obj_tmp
                    best_a = a_tmp.copy()
                    a = best_a.copy()
                    conv = fast_convolve(a)
                no_improve = 0
                perturb_count += 1
            else:
                # Perturb
                rng = np.random.default_rng(iteration * 31 + perturb_count * 7)
                scale = 0.002 + 0.003 * min(perturb_count, 10)
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                else:
                    a = a / s
                conv = fast_convolve(a)
                no_improve = 0
                perturb_count += 1

            max_no_improve = min(200, 60 + 15 * perturb_count)
            if perturb_count > 50:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, window=30):
    """Randomized coordinate descent: try adjusting pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    deltas = [0.0005, 0.001, 0.002, 0.005, 0.01, -0.0005, -0.001, -0.002, -0.005, -0.01]

    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, int(300000 / n))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.3:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            found = False
            for delta in deltas:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_copy = a.copy()
                a_copy[i] += delta
                a_copy[j] -= delta
                s_copy = a_copy.sum()
                if s_copy < 1e-15:
                    continue
                # Quick check: only compute affected conv values
                # For efficiency, just do full conv (n is moderate)
                conv_try = fast_convolve(a_copy)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_copy
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1

        if no_improve > 150:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()

    # Triangle
    tri = 1.0 - np.abs(x - n / 2.0) / (n / 2.0)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + gradient + coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
            print(f"Parent: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick gradient scan across lengths
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_budget = 180.0

    lengths = [512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_initializations(n, parent, rng)

        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'random', 'alternating', 'gaussian']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        t_per = max(8.0, (phase1_budget - 5) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 600)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.91,
                                           top_k=40, temp_init=0.002, temp_rate=1.004)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP-based deep optimization
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))
            seen_lengths.add(len(best_sequence))

    # Sort by score
    top_configs_scored = []
    for n, a0 in top_configs:
        obj, _ = compute_objective(a0)
        top_configs_scored.append((obj, n, a0))
    top_configs_scored.sort(key=lambda x: x[0])

    remaining = deadline - time.monotonic() - 300.0
    n_candidates = max(1, len(top_configs_scored))
    t_per_lp = max(60.0, remaining * 0.65 / n_candidates)

    for i, (_, n, a0) in enumerate(top_configs_scored):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break

        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP polish with multiple perturbation seeds
    # =========================================================
    print("=== Phase 3: LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 60:
            for pi in range(4):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(remaining / (4 - pi) * 0.8, deadline - time.monotonic() - 100)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.001 * pi
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Coordinate descent + gradient fine-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        n_best = len(best_sequence)

        # Coordinate descent
        t_cd = min(remaining * 0.4, 80.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Short gradient runs with various configs
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 15:
            cfgs = [
                {'lr_init': 0.005, 'momentum': 0.94, 'top_k': 80, 'temp_init': 0.001, 'temp_rate': 1.006},
                {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 120, 'temp_init': 0.0005, 'temp_rate': 1.008},
                {'lr_init': 0.008, 'momentum': 0.90, 'top_k': 40, 'temp_init': 0.003, 'temp_rate': 1.005},
            ]
            for gi in range(len(cfgs)):
                if time.monotonic() > deadline - 20:
                    break
                t_sg = min(remaining / (len(cfgs) - gi) * 0.7, 40.0)
                if t_sg < 5:

                cfg = cfgs[gi]
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_final = min(25.0, deadline - time.monotonic() - 10)
        if t_final > 8:

            if np.isfinite(score) and score < best_score:

        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
