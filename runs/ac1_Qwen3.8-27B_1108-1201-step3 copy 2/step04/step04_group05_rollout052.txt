"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive soft-max gradient descent
with Adam optimizer, coordinate descent refinement, and multi-start exploration.
Key innovation: uses smooth soft-max approximation of the max function with
adaptive temperature, enabling fast gradient-based optimization that differs
fundamentally from the LP-based Frank-Wolfe approach.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_softmax_gradient(a, c, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*c)) w.r.t. a."""
    n = len(a)
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    w = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # Using FFT cross-correlation
    L = 1 << (3 * n - 2 - 1).bit_length()
    w_fft = np.fft.rfft(w, L)
    a_fft = np.fft.rfft(a, L)
    xcorr = np.fft.irfft(w_fft * np.conj(a_fft), L)
    return 2.0 * xcorr[:n]

def project_nonneg_sum1(a):
    """Project to non-negative with sum = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_gradient_optimize(a0, time_limit, tau_start=0.5, tau_end=100.0, lr=0.005, n_restarts=3):
    """Adaptive soft-max gradient descent with Adam."""
    a = project_nonneg_sum1(a0.astype(np.float64))
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    total_time = time_limit
    
    for restart in range(n_restarts):
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t = 0
        current_lr = lr
        no_improve = 0
        
        while time.monotonic() - t0 < total_time:
            t += 1
            c = fast_convolve(a)
            obj = c.max()
            
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive temperature
            progress = min(1.0, t / max(1, int(total_time * 200)))
            tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
            
            # Compute gradient
            grad = compute_softmax_gradient(a, c, tau)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            a = project_nonneg_sum1(a)
            
            # Decay learning rate
            current_lr *= 0.99995
            
            if no_improve > 500:
                break
        
        # Perturb for next restart
        if restart < n_restarts - 1:
            rng = np.random.default_rng(42 + restart * 1000)
            scale = 0.01 * (restart + 1)
            a = project_nonneg_sum1(best_a + rng.standard_normal(n) * scale)
            c = fast_convolve(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, rng, max_sweeps=100):
    """Coordinate descent for fine-tuning."""
    a = project_nonneg_sum1(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try several values for a[i]
            for delta in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]:
                a_new = a.copy()
                a_new[i] = max(0, a[i] + delta)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-15:
                    a = a_new
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and sweep > 10:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive soft-max gradient descent
with Adam optimizer, variance-minimization regularizer, coordinate descent refinement,
and multi-start exploration across different sequence lengths.
Key innovation: smooth soft-max approximates the non-smooth max, enabling fast
per-iteration gradient computation via FFT cross-correlation, with adaptive
temperature scheduling for progressive peak sharpening.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def xcorr_filtered(w, a):
    """Compute sum_j w[i+j]*a[j] for i=0..len(a)-1 using FFT cross-correlation."""
    n = len(a)
    L = 1 << (len(w) + n - 2).bit_length()
    w_fft = np.fft.rfft(w, L)
    a_fft = np.fft.rfft(a, L)
    xcorr = np.fft.irfft(w_fft * np.conj(a_fft), L)
    return xcorr[:n]


def softmax_gradient(a, c, tau):
    """Gradient of (1/tau)*log(sum exp(tau*c)) w.r.t. a."""
    n = len(a)
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    s = exp_c.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = exp_c / s
    return 2.0 * xcorr_filtered(w, a)


def variance_gradient(a, c):
    """Gradient of sum(c[k]^2) w.r.t. a (encourages flat convolution)."""
    return 4.0 * xcorr_filtered(c, a)


def project_nonneg_sum1(a):
    """Project to {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_adam_optimize(a0, time_limit, tau_start=0.5, tau_end=80.0,
                          lr=0.004, var_weight=0.0, n_restarts=3,
                          schedule='linear'):
    """Adaptive soft-max gradient descent with Adam optimizer."""
    a = project_nonneg_sum1(a0.astype(np.float64))
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    for restart in range(n_restarts):
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t = 0
        current_lr = lr
        no_improve = 0

        while time.monotonic() - t0 < time_limit:
            t += 1
            c = fast_convolve(a)
            obj = c.max()

            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adaptive temperature
            elapsed = time.monotonic() - t0
            if schedule == 'linear':
                progress = min(1.0, elapsed / time_limit)
            else:
                progress = min(1.0, t / max(1, int(time_limit * 100)))
            tau = tau_start + (tau_end - tau_start) * (progress ** 0.6)

            # Gradient
            grad = softmax_gradient(a, c, tau)
            if var_weight > 0:
                grad = grad + var_weight * variance_gradient(a, c)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad ** 2
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project
            a = project_nonneg_sum1(a)

            # Decay learning rate
            current_lr *= 0.99997

            if no_improve > 800 and tau > 10:
                break

        # Perturb for next restart
        if restart < n_restarts - 1:
            rng = np.random.default_rng(12345 + restart * 9999)
            scale = 0.008 * (restart + 1)
            a = project_nonneg_sum1(best_a + rng.standard_normal(n) * scale)
            c = fast_convolve(a)
            obj = c.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def coordinate_descent(a0, time_limit, rng, n_steps_per_sweep=200):
    """Randomized coordinate descent for fine-tuning."""
    a = project_nonneg_sum1(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01, -0.02]

    while time.monotonic() - t0 < time_limit:
        sweep_consumed = 0
        for i in rng.integers(0, n, size=min(n, n_steps_per_sweep)):
            if time.monotonic() - t0 > time_limit:

                a_new = a.copy()
                val = a[i] + ds
                if val < 0:

                a_new = project_nonneg_sum1(a_new)
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-14:
                    a = a_new
                    best_obj = obj
                    best_a = a.copy()
                    break
        sweep_consumed += 1
        if sweep_consumed > 50 and best_obj == fast_convolve(a).max():
            # Check if we're stuck
            pass

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    x = np.arange(n, dtype=np.float64)
    # Linear ramp
    ramp = (x + 1)
    inits.append(ramp / ramp.sum())
    # Reverse ramp
    inits.append(ramp[::-1].astype(np.float64) / ramp.sum())
    # Quadratic (bell)
    bell = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(bell / bell.sum())
    # Sine
    s = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(s / s.sum())
    # Exponential
    for alpha in [0.5, 1.0, 2.0]:
        e = np.exp(alpha * x / n)
        inits.append(e / e.sum())
    # Two peaks
    for frac in [0.2, 0.3, 0.4]:
        p1 = np.exp(-0.5 * ((x - frac * (n - 1)) / (0.1 * n)) ** 2)
        p2 = np.exp(-0.5 * ((x - (1 - frac) * (n - 1)) / (0.1 * n)) ** 2)
        two = p1 + p2
        inits.append(two / two.sum())
    # U-shape
    u = 1.0 / (np.abs(x - 0.5 * (n - 1)) / n + 0.05)
    inits.append(u / u.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Mirror of parent
        mx = parent[::-1].astype(np.float64)
        mxi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), mx)
        mxi = np.maximum(mxi, 0)
        ms = mxi.sum()
        if ms > 1e-10:
            inits.append(mxi / ms)

    return inits
