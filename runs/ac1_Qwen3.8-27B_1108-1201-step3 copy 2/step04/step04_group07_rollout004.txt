def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, conv, t):
    """Compute gradient of t*log(sum(exp(conv/t))) w.r.t. a.
    Uses the identity: grad[j] = 2 * sum_m p[j+m] * a[m]
    which is np.correlate(p, a, mode='valid').
    """
    conv_shifted = conv - conv.max()
    p = np.exp(conv_shifted / t)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p


def project_simplex(y):
    """Project y onto {x >= 0, sum(x) = 1} (Euclidean projection)."""
    n = len(y)
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)

def obj_value_normalized(a, n):
    """Compute objective and convolution for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Return normalized direction."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 5})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0

def coordinate_descent_polish(a0, time_limit, max_sweeps=50):
    """Coordinate descent: minimize max(conv) by adjusting one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:

    n = len(a)

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        improved = False
        # Random order
        perm = np.random.default_rng(sweep).permutation(n)

        for j in perm:
            if time.monotonic() - t0 > time_limit:
                break

            # Try changing a[j] while maintaining sum=1
            # We'll try a few candidate values for a[j]
            current_val = a[j]
            if current_val < 1e-12:
                continue

            # Compute current conv efficiently
            # conv[k] = sum_i a[i]*a[k-i]
            # Changing a[j] affects conv[k] for k where j is involved:
            # conv[j] += 2*da*a[j] + ... it's complex.
            # Simpler: just try a few values and recompute conv (expensive but reliable)
            # For efficiency, only try if this coordinate is "active" (near a peak)

            # Quick heuristic: only adjust coordinates that are larger than average
            if current_val < 1.0 / n:
                continue

            # Try reducing a[j] by distributing to neighbors
            for frac in [0.95, 0.9, 0.8]:
                da = current_val * (1 - frac)
                a[j] = current_val * frac
                # Distribute da to maintain sum
                # Add to a uniformly (cheap update)
                a += da / n
                # Re-normalize to be safe
                s_new = a.sum()
                a = a / s_new

                conv_test = fast_convolve(a)
                obj_test = 2.0 * n * conv_test.max()

                if obj_test < best_obj - 1e-12:
                    best_obj = obj_test
                    best_a = a.copy()
                    improved = True
                    break
            # Reset if no improvement from this coordinate
            if not improved:
                a = best_a.copy()

        sweep += 1
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0

    return best_a, best_obj


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # Various structured inits
    for freq, label in [(1, 'sine1'), (2, 'sine2'), (3, 'sine3'), (4, 'sine4')]:
        sig = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + rng_local.random() * 2 * np.pi)
        inits[label] = sig / sig.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Triangular
    tri = np.minimum(x + 1.0, n - x)
    inits['tri'] = tri / tri.sum()

    # Random Dirichlet with various concentrations
    for alpha, label in [(0.3, 'dir3'), (1.0, 'dir1'), (3.0, 'dir3b')]:
        inits[label] = rng_local.dirichlet(np.ones(n) * alpha)

    return inits


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Nesterov (n={n_parent}): {score:.6f}")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: LP optimization on best so far =====
    print("=== Phase 2: LP Optimization ===")
    t_start2 = time.monotonic()
    remaining = deadline - time.monotonic() - 200.0
    if remaining > 60 and best_sequence is not None:
        t_alloc = min(250.0, remaining * 0.6)
        a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP (n={len(a_opt)}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Nesterov GD with different hyperparams =====
    print("=== Phase 3: Nesterov GD (tuned) ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 30:
            configs = [
                {'t_init': 0.15, 't_min': 0.002, 't_decay': 0.997, 'lr_init': 0.002, 'momentum': 0.9},
                {'t_init': 0.35, 't_min': 0.005, 't_decay': 0.994, 'lr_init': 0.005, 'momentum': 0.8},
                {'t_init': 0.1, 't_min': 0.001, 't_decay': 0.998, 'lr_init': 0.003, 'momentum': 0.92},
            ]
            t_per = remaining / (len(configs) + 1)
            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 60)
                if t_alloc < 15:
                    break
                a_opt, obj_opt = nesterov_grad_descent(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Nesterov {ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Try different lengths =====
    print("=== Phase 4: Multi-length exploration ===")
    t_start4 = time.monotonic()
    n_best = len(best_sequence) if best_sequence is not None else 768
    alt_lengths = [l for l in [512, 768, 1024, 1536] if l != n_best]

    if best_sequence is not None and alt_lengths:
        remaining = deadline - time.monotonic() - 120.0
        if remaining > 40:
            for alt_n in alt_lengths[:2]:  # Try up to 2 alternative lengths
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(100.0, remaining * 0.4)
                if t_alloc < 20:
                    break

                # Resample parent to new length
                a_init = resample_sequence(best_sequence, alt_n)
                a_opt, obj_opt = nesterov_grad_descent(a_init, time_limit=t_alloc * 0.6,
                                                       t_init=0.2, t_min=0.003,
                                                       t_decay=0.996, lr_init=0.004, momentum=0.85)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Alt length n={alt_n}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 5: Final polish (LP + coordinate descent) =====
    print("=== Phase 5: Final Polish ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            # LP polish
            t_lp = min(60.0, remaining * 0.5)
            if t_lp > 15:
                a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f}")

            # Coordinate descent
            remaining2 = deadline - time.monotonic() - 15.0
            if remaining2 > 20:
                a_opt, obj_opt = coordinate_descent_polish(best_sequence, time_limit=remaining2,
                                                           max_sweeps=30)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Coord descent: {score:.6f}")

            # One more Nesterov pass
            remaining3 = deadline - time.monotonic() - 10.0
            if remaining3 > 15:
                a_opt, obj_opt = nesterov_grad_descent(best_sequence, time_limit=remaining3,
                                                       t_init=0.1, t_min=0.001,
                                                       t_decay=0.998, lr_init=0.002, momentum=0.95)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final Nesterov: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
