  But wait, if we change `a_j`, we need to renormalize. Or we can think of it as: fix the sum of all other coordinates, and vary `a_j`.

  Actually, for coordinate descent with the constraint `sum(a) = 1`, the standard approach is:

  - The constraint becomes: `a_j = 1 - sum_{i != j} a_i`... no, that's not right either, because we're allowing `a_j` to vary.
  
  The proper way: the constraint is `sum_i a_i = 1`. If we fix `a_i` for `i != j`, then `a_j` is determined: `a_j = 1 - sum_{i != j} a_i`. So there's only one degree of freedom, and coordinate descent doesn't make sense in this form.
  
  The proper coordinate descent for simplex-constrained problems is:
  - Pick a pair of coordinates `(j, l)`
  - Vary `a_j` and `a_l` subject to `a_j + a_l = const` (to maintain the sum constraint)
  - This is a 1D optimization problem
  
  For the objective `max_k conv(a,a)_k`, as we vary `a_j = t` and `a_l = c - t` (where `c = a_j + a_l` is fixed), the objective becomes `max_k Q_k(t)` where each `Q_k(t)` is a quadratic in `t`. """
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Multiplicative weights (exponentiated gradient) with adaptive temperature
2. 2D coordinate descent (pairwise L-BFGS-like exchanges) for polishing
3. Gradient descent with simplex projection as alternative optimizer
4. Multi-start exploration with diverse initializations
5. Length adaptation via interpolation
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, c, temperature, top_k=None):
    n = len(a)
    len_c = len(c)
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
    else:
        indices = np.arange(len_c)
    
    peek = c[indices]
    scale = peek.max()
    if scale < 1e-300:
        w = np.zeros(len(indices))
        w[0] = 1.0
    else:
        scaled = (peek - scale) / temperature
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je+1] += 2.0 * w[idx] * a[k-je:k-js+1][::-1]
    return sg

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0]) if v[0] > 0 else np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def multiplicative_weights_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                                     temp_decay=0.9997, eta_decay=0.99995,
                                     top_k_frac=0.25, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 5000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.01)
    
    return best_a.copy(), best_obj

def grad_descent_simplex(a0, time_limit, lr0=0.01, lr_decay=0.9999, top_k=50, verbose=False):
    """Gradient descent with simplex projection (alternative to multiplicative weights)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    lr = lr0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute subgradient at top-k peaks
        sg = compute_subgradient(a, c, temperature=0.01, top_k=top_k)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Gradient step
        a_grad = a - lr * sg / sg_norm
        # Project onto simplex
        a_new = simplex_project(a_grad)
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: reduce learning rate
            if no_improve % 100 == 0:
                lr *= 0.9
        
        a = a_new
        c = c_new
        lr *= lr_decay
        
        if no_improve > 10000:
            break
    
    return best_a.copy(), best_obj

def coord_descent_2d(a0, time_limit, max_rounds=500, verbose=False):
    """2D coordinate descent: pairwise exchanges on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        # Pick a random pair of coordinates
        j = rng.integers(n)
        l = rng.integers(n)
        if j == l:
            continue
        
        aj = a[j]
        al = a[l]
        c_sum = aj + al
        if c_sum < 1e-15:
            continue
        
        # Vary a_j = t, a_l = c_sum - t for t in [0, c_sum]
        # conv(a,a)_k = c_k_base + 2*t*a_{k-j} + 2*(c_sum-t)*a_{k-l} + corrections
        # More precisely:
        # Original contribution to conv_k from j and l:
        #   a_j*a_{k-j} + a_{k-j}*a_j + a_l*a_{k-l} + a_{k-l}*a_l
        #   (with special cases when k-j==l or k-l==j)
        # New contribution with a_j=t, a_l=c_sum-t:
        #   t*a_{k-j} + a_{k-j}*t + (c_sum-t)*a_{k-l} + a_{k-l}*(c_sum-t)
        
        # For efficiency, compute the "base" convolution (without j and l contributions)
        # and then add the variable part
        
        # Actually, let's just evaluate the objective at a few candidate t values
        # and pick the best one (since the objective is piecewise quadratic in t)
        
        # Find the current peak
        k_star = int(np.argmax(c))
        
        # The objective as function of t is max_k Q_k(t)
        # For the active peak k*, Q_{k*}(t) is quadratic
        # Try a few t values around the current t = aj
        
        candidates = []
        # Current value
        candidates.append(aj)
        # Midpoint
        candidates.append(c_sum / 2)
        # Equal split adjusted toward reducing peak
        # The peak is reduced if we move mass from high-convolution positions
        # Simple heuristic: try t = aj * factor for various factors
        for factor in [0.5, 0.7, 0.8, 1.25, 1.4, 2.0, 0.3, 0.1]:
            t = aj * factor
            if 0 <= t <= c_sum:
                candidates.append(t)
        
        # Also try to minimize the specific peak k*
        # Q_{k*}(t) = base + 2*t*a_{k*-j} + 2*(c_sum-t)*a_{k*-l} (approximately)
        # = base + 2*c_sum*a_{k*-l} + 2*t*(a_{k*-j} - a_{k*-l})
        # This is linear in t (ignoring the a_j^2 term when k*=2j)
        # Minimum is at one of the boundaries unless the coefficient is 0
        
        best_t = aj
        best_local_obj = np.inf
        
        for t in candidates:
            a[j] = t
            a[l] = c_sum - t
            c2 = fast_convolve(a)
            obj2 = c2.max()
            if obj2 < best_local_obj:
                best_local_obj = obj2
                best_t = t
        
        a[j] = best_t
        a[l] = c_sum - best_t
        
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = a.copy()
            improved = True
            no_improve = 0
        else:
            a = best_a.copy()
            no_improve += 1
        
        if no_improve > 50:
            break
    
    return best_a.copy(), best_obj

def generate_initializations(a0, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # From parent
    if len(a0) == n:
        inits.append(a0.copy())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Exponential decay (multiple rates)
    for beta in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n-1-x) / n)
        inits.append(e2 / e2.sum())
    
    # Edge-concentrated
    edge = 1.0 / (np.minimum(x, n-1-x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.15*n))**2)
    inits.append(g / g.sum())
    
    # Cosine variants
    for phase in [0, 0.5]:
        c1 = np.abs(np.cos(np.pi * (x + phase) / n)) + 0.01
        inits.append(c1 / c1.sum())
    
    # Random
    for _ in range(5):
        r = rng.dirichlet(np.ones(n) * 0.2)
        inits.append(r)
    
    # Perturbed parent
    if len(a0) == n:
        for scale in [0.001, 0.005, 0.01, 0.05]:
            p = a0 + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    return inits

def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    if n_old == n_new:
        return a_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new

If we change `a_j` by `delta` and `a_l` by `-delta` (to maintain sum = 1), then:
`conv(a',a')_k = conv(a,a)_k + 2*delta*a_{k-j} - 2*delta*a_{k-l} + delta^2*(delta_{2j,k} - 2*delta_{j+l,k} + delta_{2l,k})`

`conv(a',a')_k = conv(a,a)_k + conv(a, delta*e_j - delta*e_l)_k + conv(delta*e_j - delta*e_l, a)_k + conv(delta*e_j - delta*e_l, delta*e_j - delta*e_l)_k`

The cross terms: `conv(a, delta*e_j)_k = delta * a_{k-j}`, `conv(a, -delta*e_l)_k = -delta * a_{k-l}`.
Similarly for the other cross term.

The quadratic term: `conv(delta*e_j - delta*e_l, delta*e_j - delta*e_l)_k = delta^2 * (delta_{k,2j} - 2*delta_{k,j+l} + delta_{k,2l})`
(Here `delta_{k,m}` is 1 if `k==m` and the indices are valid, 0 otherwise.)

So: `conv(a',a')_k = conv(a,a)_k + 2*delta*(a_{k-j} - a_{k-l}) + delta^2*(delta_{2j,k} - 2*delta_{j+l,k} + delta_{2l,k})`

This is a quadratic in `delta`! So for each pair `(j,l)`, the change in `conv_k` is quadratic in `delta`, and the max over `k` is the max of quadratics, which is convex in `delta`. We can find the optimal `delta` by ternary search or by finding the "kink" points.

This is much more efficient than re-computing the full convolution! We can do this in O(n) per pair (just evaluating the quadratic at a few points), or even find the exact optimum.

But for the max of many quadratics, finding the exact minimum requires checking all pairs of "active" quadratics. A simpler approach is to do a few evaluations (ternary search) since the function is convex.
