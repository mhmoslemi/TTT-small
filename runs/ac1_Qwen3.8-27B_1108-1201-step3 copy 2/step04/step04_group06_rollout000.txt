import numpy as np

def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smoothed_subgrad(a, c, temp, top_k):

    n = len(a)
    len_c = len(c)

        indices = np.argpartition(c, -top_k)[-top_k:]

        indices = np.arange(len_c)
        peek = c.copy()

    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        wi = w[idx]
        if wi < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # sg[j] += 2*wi*a[k-j] for j in [js, je]
        sg[js:je + 1] += 2.0 * wi * a[k - je:k - js + 1][::-1]
    
    return sg

def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.01, temp_decay=0.9997, 
                eta_decay=0.99995, top_k=50, verbose=False):
    """Exponentiated gradient optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg = compute_smoothed_subgrad(a, c, temp, top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        c = fft_convolve(a)
        obj = c.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    EG {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1
        
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 5000:
            # Perturb and restart
            rng = np.random.default_rng(it)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fft_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.05)
    
    return best_a.copy(), best_obj

def lp_best_response(a, M):
    """Solve LP: maximize sum(b) s.t. (a*b)_k <= M, b >= 0.
    Returns normalized b (as a probability vector) or None on failure.
    """
    n = len(a)
    # Convolution matrix B: B[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1, n)
    # Use scipy linprog: minimize -sum(b) s.t. B@b <= M, b >= 0
    
    # Build B efficiently
    # B[k, j] = a_{k-j} for 0 <= k-j < n
    # For k=0: B[0,j] = a_{-j} = 0 for j>0, a_0 for j=0
    # For k=1: B[1,j] = a_{1-j}: a_1 for j=0, a_0 for j=1
    # etc.
    
    # Actually, let's think about this differently.
    # (a*b)_k = sum_j a_j * b_{k-j} for valid indices
    # = sum_{j: 0<=j<n, 0<=k-j<n} a_j * b_{k-j}
    # Let i = k-j, so j = k-i:
    # = sum_{i: 0<=i<n, 0<=k-i<n} a_{k-i} * b_i
    # = sum_i b_i * (a_{k-i} if 0 <= k-i < n, else 0)
    
    # So B[k, i] = a_{k-i} if 0 <= k-i < n, else 0
    # B is (2n-1) x n
    
    # For efficiency, we can use the fact that B is a convolution matrix (Toeplitz-like)
    # But for linprog, we need to pass it explicitly.
    
    if n > 1000:
        return None  # Too large
    
    # Build B
    k_vals = np.arange(2 * n - 1)
    i_vals = np.arange(n)
    # B[k, i] = a[k-i] if 0 <= k-i < n
    diff = k_vals[:, None] - i_vals[None, :]
    B = np.zeros((2 * n - 1, n))
    valid = (diff >= 0) & (diff < n)
    B[valid] = a[diff[valid]]
    
    # linprog: minimize c^T x s.t. A_ub @ x <= b_ub, x >= 0
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = B
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs')
    
    if res.success:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

def lp_iteration(a, time_limit, n_steps=100, t_lr=0.1, verbose=False):
    """LP-based best response iteration."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0
    
    while time.monotonic() - t0 < time_limit and step < n_steps:
        step += 1
        M = c.max()
        b = lp_best_response(a, M)
        if b is None:
            break
        
        # Interpolate
        a_new = (1 - t_lr) * a + t_lr * b
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        c_new = fft_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            if verbose and step % 10 == 0:
                print(f"    LP {step}: obj={best_obj:.10f}")
        else:
            # Try smaller step
            t_lr *= 0.5
            if t_lr < 1e-6:
                break
            a = best_a.copy()
            c = fft_convolve(a)
    
    return best_a.copy(), best_obj

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def projected_gradient_descent(a0, time_limit, lr0=0.01, top_k=50, temp=0.005):
    """Projected gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    lr = lr0
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg = compute_smoothed_subgrad(a, c, temp, top_k)
        a_new = simplex_project(a - lr * sg)
        
        c_new = fft_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
            a = best_a.copy()
            c = fft_convolve(a)
        
        lr = min(lr * 1.01, lr0)  # slow recovery
    
    return best_a.copy(), best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1
    inits.append(tri / tri.sum())
    
    # Gaussian center
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (sigma * n))**2)
        inits.append(g / g.sum())
    
    # Cosine window
    for beta in [1.0, 2.0, 0.5]:
        cw = np.cos(np.pi * x / n)**beta + 0.01
        inits.append(cw / cw.sum())
    
    # Exponential
    for rate in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-rate * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-rate * (n-1-x) / n)
        inits.append(e2 / e2.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)
    
    # Parent (if available and same length)
    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0)
        p = p / p.sum()
        inits.append(p)
        # Perturbed parent
        for scale in [0.001, 0.01, 0.05]:
            pp = p + rng.standard_normal(n) * scale
            pp = np.maximum(pp, 0)
            pp = pp / pp.sum()
            inits.append(pp)
    
    return inits

def local_search(a0, time_limit, max_rounds=50):
    """Local search with multiplicative perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        n_perturb = max(1, n // 10)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 1.0/0.85, 1.0/0.7, 0.5, 1.5, 0.9, 1.0/0.9]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fft_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a.copy(), best_obj
