Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using symmetric parameterization,
momentum subgradient descent with peak filtering, and LP polishing.
Key innovation: exploit symmetry a[j]=a[n-1-j] to halve the search space,
and use fast FFT-based convolution for efficient subgradient computation.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def subgradient_symmetric(a, conv, top_k, peak_weight):
    """
    Compute subgradient for symmetric parameterization.
    a is the full sequence (symmetric), returns gradient for first half.
    """
    n = len(a)
    half = n // 2
    m = len(conv)
    
    # Select top peaks
    if top_k < m:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(m)
    
    # Weight peaks by their values
    peak_vals = conv[peaks]
    w = peak_vals ** peak_weight
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / w_sum
    
    # Compute subgradient for full a
    grad_full = np.zeros(n)
    for i, k in enumerate(peaks):
        vi = w[i] * 2.0
        # dc[k]/da[j] = 2*a[k-j] for valid k-j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad_full[j] += vi * a[k - j]
    
    # Symmetric gradient: grad_sym[j] = grad_full[j] + grad_full[n-1-j]
    grad_sym = grad_full[:half] + grad_full[n-1-half:]
    
    return grad_sym


def symmetric_to_full(b_full, n):
    """Convert symmetric first-half to full sequence."""
    a = np.zeros(n)
    half = n // 2
    if n % 2 == 0:
        a[:half] = b_full
        a[half:] = b_full[::-1]
    else:
        a[:half] = b_full[:half]
        a[half] = b_full[half]  # middle element
        a[half+1:] = b_full[:half][::-1]
    return a


def full_to_symmetric(a, n):
    """Extract symmetric first-half from full sequence (averaging with reverse)."""
    half = n // 2
    if n % 2 == 0:
        return 0.5 * (a[:half] + a[half:][::-1])
    else:
        b = np.zeros(half + 1)
        b[:half] = 0.5 * (a[:half] + a[half-1:0:-1])
        b[half] = a[half]
        return b


def obj_from_conv(conv, n):

def subgradient_fft(a, conv, top_k=50, peak_power=2.0):

    O(n log n) time.

        gs = g.sum()
        if gs > 1e-10:

        cur_obj = 2.0 * n * conv.max()
        it += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= lr_decay

        # Periodic LP polish
        if lp_every > 0 and it % lp_every == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 3:
                M_lp = conv.max() * 0.9995
                g = solve_lp_fast(a, M_lp, timeout=min(3.0, remaining - 1.0))
                # Try mixing
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = np.maximum(a + t * (g - a), 0.0)
                    st = a_try.sum()
                    if st > 1e-15:
                        a_try = a_try / st
                        ct = fast_convolve(a_try)
                        ot = 2.0 * n * ct.max()
                        if ot < best_obj - 1e-12:
                            best_obj = ot
                            best_a = a_try.copy()
                            a = a_try.copy()
                            conv = ct
                            v = np.zeros(n)
                            break

        # Reset on stagnation
        if no_improve > 1000:
            scale = 0.003 * (1.0 + 0.1 * restart_count)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0
            lr = lr_init * 0.5
            restart_count += 1
            if restart_count > 30:
                break

        if verbose and it % 2000 == 0:
            el = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.8f}, t={el:.0f}s, lr={lr:.4f}")

    return best_a, best_obj


def make_symmetric(a):
    """Project onto symmetric subspace: a[j] = a[n-1-j]."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        # Resample parent to length n
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        p = np.interp(x_new, x_old, parent)
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 1e-15:
            inits['parent'] = p / s
        # Symmetric parent
        ps = make_symmetric(p)
        s2 = ps.sum()
        if s2 > 1e-15:
            inits['parent_sym'] = ps / s2

    # Sine-modulated
    for freq, name in [(1, 'sine1'), (2, 'sine2'), (3, 'sine3'), (4, 'sine4')]:
        s = 0.7 + 0.3 * np.sin(2 * np.pi * freq * x / n)
        inits[name] = s / s.sum()

    # Cosine lobes
    for freq, name in [(2, 'cos2'), (4, 'cos4'), (6, 'cos6')]:
        c = 0.6 + 0.4 * np.cos(2 * np.pi * freq * x / n)
        inits[name] = c / c.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Double Gaussian
    g2 = np.exp(-((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['dgauss'] = g2 / g2.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Walsh-like (alternating blocks)
    w = np.ones(n)
    block = max(1, n // 8)
    for i in range(1, n, 2 * block):
        w[i:i + block] = -0.5
    w = np.abs(w)
    inits['walsh'] = w / w.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Fast SG descent from multiple inits at n=768 (50% time)
    Phase 2: LP polishing on best (15% time)
    Phase 3: Try other lengths n=512,1024,1536 (25% time)
    Phase 4: Final refinement (10% time)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, obj={2*len(p)*fast_convolve(parent).max():.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = np.ones(128) / 128

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(seed)

    def try_update(a_opt):
        nonlocal best_sequence, best_score
        if a_opt is not None and len(a_opt) > 0:
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  NEW BEST: {score:.6f} (n={len(a_opt)})")

    # ===== Phase 1: Fast SG at n=768 =====
    print("=== Phase 1: SG Descent (n=768) ===")
    t1 = time.monotonic()
    p1_budget = 0.45 * (budget_s - 15.0)
    p1_deadline = t1 + p1_budget

    n_main = 768
    inits = get_initializations(n_main, parent, rng)

    # Priority: parent first, then structured
    priority = ['parent', 'parent_sym', 'cos4', 'sine2', 'gauss', 'dgauss', 'cos2', 'random']
    names = [nm for nm in priority if nm in inits]

    # Allocate time per init
    t_per_init = max(15.0, p1_budget / max(1, len(names)) * 0.7)

    best_from_p1 = None
    best_p1_obj = float('inf')

    for idx, name in enumerate(names):
        remaining = p1_deadline - time.monotonic()
        if remaining < 10:
            break
        t_alloc = min(t_per_init, remaining - 5.0)
        if t_alloc < 8:
            break

        a0 = inits[name]
        sym = (name.endswith('_sym'))
        if sym:
            a_opt, obj_opt = sg_descent(a0, t_alloc, top_k=60, peak_power=1.5,
                                        lr_init=0.4, momentum=0.93,
                                        lp_every=400, seed=seed + idx)
        else:
            a_opt, obj_opt = sg_descent(a0, t_alloc, top_k=50, peak_power=2.0,
                                        lr_init=0.3, momentum=0.92,
                                        lp_every=500, seed=seed + idx)

        # Also try symmetric version
        a_sym = make_symmetric(a_opt)
        s_sym = a_sym.sum()
        if s_sym > 1e-15:
            a_sym = a_sym / s_sym
            conv_sym = fast_convolve(a_sym)
            obj_sym = 2.0 * n_main * conv_sym.max()
            if obj_sym < obj_opt:
                a_opt = a_sym
                obj_opt = obj_sym

        if obj_opt < best_p1_obj:
            best_p1_obj = obj_opt
            best_from_p1 = a_opt.copy()

        print(f"  Init '{name}': obj={obj_opt:.6f} (alloc={t_alloc:.0f}s)")
        try_update(a_opt)

    # Keep best and continue refining
    if best_from_p1 is not None:
        remaining = p1_deadline - time.monotonic()
        if remaining > 10:
            print(f"  Continuing best with {remaining:.0f}s...")
            a_opt, obj_opt = sg_descent(best_from_p1, remaining, top_k=80,
                                        peak_power=1.5, lr_init=0.2,
                                        momentum=0.94, lp_every=300,
                                        seed=seed + 100)
            a_sym = make_symmetric(a_opt)
            s_sym = a_sym.sum()
            if s_sym > 1e-15:
                a_sym = a_sym / s_sym
                obj_sym = 2.0 * n_main * fast_convolve(a_sym).max()
                if obj_sym < obj_opt:
                    a_opt = a_sym
                    obj_opt = obj_sym
            if obj_opt < best_p1_obj:
                best_p1_obj = obj_opt
                best_from_p1 = a_opt.copy()
            try_update(a_opt)

    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best obj: {best_p1_obj:.6f}")

    # ===== Phase 2: LP Polish =====
    print("=== Phase 2: LP Polish ===")
    t2 = time.monotonic()
    p2_budget = 0.15 * (budget_s - 15.0)
    p2_deadline = t2 + p2_budget

    if best_sequence is not None:
        a_cur = best_sequence.astype(np.float64)
        a_cur = np.maximum(a_cur, 0)
        s = a_cur.sum()
        if s > 1e-15:
            a_cur = a_cur / s
        n = len(a_cur)

        # Iterative LP with tightening
        conv = fast_convolve(a_cur)
        M = conv.max()
        it = 0
        while time.monotonic() < p2_deadline:
            M_lp = M * 0.999
            g = solve_lp_fast(a_cur, M_lp, timeout=min(5.0, p2_deadline - time.monotonic() - 1))
            # Line search
            improved = False
            for t in [0.5, 0.7, 0.8, 0.9, 1.0]:
                a_try = np.maximum(a_cur + t * (g - a_cur), 0.0)
                st = a_try.sum()
                if st > 1e-15:
                    a_try = a_try / st
                    ct = fast_convolve(a_try)
                    ot = 2.0 * n * ct.max()
                    if ot < M - 1e-10:
                        a_cur = a_try
                        M = ct.max()
                        conv = ct
                        improved = True
                        try_update(a_cur)
                        break
            it += 1
            if not improved:
                # Try symmetric
                a_sym = make_symmetric(a_cur)
                s2 = a_sym.sum()
                if s2 > 1e-15:
                    a_sym = a_sym / s2
                    ct = fast_convolve(a_sym)
                    ot = 2.0 * n * ct.max()
                    if ot < M - 1e-8:
                        a_cur = a_sym
                        M = ct.max()
                        conv = ct
                        try_update(a_cur)
                    else:
                        break
        print(f"  LP polish: obj={2.0*n*M:.6f} ({time.monotonic()-t2:.0f}s, {it} iters)")

    # ===== Phase 3: Other Lengths =====
    print("=== Phase 3: Other Lengths ===")
    t3 = time.monotonic()
    p3_budget = 0.25 * (budget_s - 15.0)
    p3_deadline = t3 + p3_budget

    other_lengths = [512, 1024, 1536]
    t_per_len = max(10.0, p3_budget / len(other_lengths))

    for n_alt in other_lengths:
        remaining = p3_deadline - time.monotonic()
        if remaining < 15:
            break
        t_alloc = min(t_per_len * 0.9, remaining - 5.0)
        if t_alloc < 10:
            break

        # Resample best from phase 1/2
        if best_sequence is not None:
            x_old = np.linspace(0, 1, len(best_sequence))
            x_new = np.linspace(0, 1, n_alt)
            a0 = np.interp(x_new, x_old, best_sequence)
            a0 = np.maximum(a0, 0)
            s0 = a0.sum()
            if s0 > 1e-15:
                a0 = a0 / s0
        else:
            a0 = np.ones(n_alt) / n_alt

        a_opt, obj_opt = sg_descent(a0, t_alloc, top_k=60, peak_power=1.8,
                                    lr_init=0.3, momentum=0.92, lp_every=400,
                                    seed=seed + n_alt)
        a_sym = make_symmetric(a_opt)
        s_sym = a_sym.sum()
        if s_sym > 1e-15:
            a_sym = a_sym / s_sym
            obj_sym = 2.0 * n_alt * fast_convolve(a_sym).max()
            if obj_sym < obj_opt:
                a_opt = a_sym
                obj_opt = obj_sym

        print(f"  n={n_alt}: obj={obj_opt:.6f}")
        try_update(a_opt)

    # ===== Phase 4: Final Refinement =====
    print("=== Phase 4: Final Refinement ===")
    t4 = time.monotonic()
    remaining = deadline - time.monotonic()

    if best_sequence is not None and remaining > 10:
        a_cur = best_sequence.astype(np.float64)
        a_cur = np.maximum(a_cur, 0)
        s = a_cur.sum()
        if s > 1e-15:
            a_cur = a_cur / s
        n = len(a_cur)

        # Final SG with very small steps
        t_sg = min(0.5 * remaining, remaining - 5)
        if t_sg > 5:
            a_opt, obj_opt = sg_descent(a_cur, t_sg, top_k=100, peak_power=1.0,
                                        lr_init=0.1, momentum=0.95,
                                        lp_every=200, seed=seed + 999)
            a_sym = make_symmetric(a_opt)
            s_sym = a_sym.sum()
            if s_sym > 1e-15:
                a_sym = a_sym / s_sym
                obj_sym = 2.0 * n * fast_convolve(a_sym).max()
                if obj_sym < obj_opt:
                    a_opt = a_sym
                    obj_opt = obj_sym
            try_update(a_opt)

        # Final LP
        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            a_cur2 = best_sequence.astype(np.float64)
            a_cur2 = np.maximum(a_cur2, 0)
            s2 = a_cur2.sum()
            if s2 > 1e-15:
                a_cur2 = a_cur2 / s2
            n2 = len(a_cur2)
            conv2 = fast_convolve(a_cur2)
            M2 = conv2.max()
            g2 = solve_lp_fast(a_cur2, M2 * 0.999, timeout=min(5.0, remaining2 - 1))
            for t in [0.3, 0.5, 0.7, 1.0]:
                a_try = np.maximum(a_cur2 + t * (g2 - a_cur2), 0.0)
                st = a_try.sum()
                if st > 1e-15:
                    a_try = a_try / st
                    ct = fast_convolve(a_try)
                    ot = 2.0 * n2 * ct.max()
                    if ot < M2 - 1e-10:
                        try_update(a_try)
                        a_cur2 = a_try
                        M2 = ct.max()

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
