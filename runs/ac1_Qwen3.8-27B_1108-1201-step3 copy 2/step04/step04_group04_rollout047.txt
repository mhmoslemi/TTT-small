    """Solve min a^T W(a) a s.t. a >= 0, sum a = 1.
    W is the matrix with W[i,j] = w[i+j]."""
    # Build W
    W = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            W[i,j] = w[i+j]
    
    # Use active set method
    # Start with all indices active
    S = set(range(n))
    for iteration in range(100):
        # Solve for a on support S
        W_SS = W[np.array(list(S))[:, None], np.array(list(S))]
        # a[S] = W_SS^{-1} @ 1 / (1^T W_SS^{-1} @ 1)
        try:
            one = np.ones(len(S))
            sol = np.linalg.solve(W_SS, one)
            a_S = sol / sol.sum()
        except np.linalg.LinAlgError:

                sol = np.linalg.solve(W_SS + 1e-12 * np.eye(len(S)), one)
            else:
                sol = np.array([1.0])
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(W_SS, one, rcond=None)[0]
        
        # Check positivity
        if np.any(sol <= 1e-10):
            # Remove non-positive indices
            new_S = [S[i] for i in range(len(S)) if sol[i] > 1e-10]
            if len(new_S) == 0:
                return np.ones(n) / n
            if len(new_S) == len(S):
                # All slightly negative, add regularization
                sol = np.maximum(sol, 0)
                sol = sol / sol.sum()
                a = np.zeros(n)
                for i, idx in enumerate(S):
                    a[idx] = sol[i]
                return a
            S = new_S
            continue
        
        # Normalize
        sol = sol / sol.sum()
        
        # Build full a
        a = np.zeros(n)
        for i, idx in enumerate(S):
            a[idx] = sol[i]
        
        # Check KKT conditions: (Wa)_i >= mu/2 for i not in S
        Wa = mat_vec_w_a(w, a)
        mu_half = np.mean(Wa[S_arr])
        
        # Find violated indices
        violations = []
        for i in range(n):
            if i not in S and Wa[i] < mu_half - 1e-12:
                violations.append((Wa[i] - mu_half, i))
        
        if not violations:
            return a
        
        # Sort by violation severity and add top ones
        violations.sort()
        S = S + [v[1] for v in violations[:min(20, len(violations))]]
    
    # Fallback: use current best
    a = np.zeros(n)
    S_arr = np.array(S)
    W_SS = w[np.add.outer(S_arr, S_arr)]
    one = np.ones(len(S))
    try:
        sol = np.linalg.solve(W_SS + 1e-10 * np.eye(len(S)), one)
        sol = np.maximum(sol, 0)
        if sol.sum() > 1e-15:
            sol = sol / sol.sum()
            for i, idx in enumerate(S):
                a[idx] = sol[i]
        else:
            a = np.ones(n) / n
    except:
        a = np.ones(n) / n
    return a

def minimax_optimize(n, time_limit, verbose=False):
    """Minimax optimization using multiplicative weights + active set QP."""
    t0 = time.monotonic()
    w = np.ones(2*n-1) / (2*n-1)  # Uniform initial weights
    eta = 1.0  # Learning rate for weight updates
    
    best_a = np.ones(n) / n
    best_obj = 2.0 * n * fast_convolve(best_a).max()
    
    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve QP
        a = solve_qp_active_set(w, n)
        
        # Evaluate
        conv = fast_convolve(a)
        obj_val = 2.0 * n * conv.max()
        
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()
        
        # Update weights: multiplicative weights
        # Give more weight to high-convolution lags
        scale = eta * conv
        scale -= scale.max()
        w_new = w * np.exp(scale)
        w_sum = w_new.sum()
        if w_sum > 1e-300:
            w = w_new / w_sum
        else:
            w = np.exp(scale) / np.exp(scale).sum()
        
        # Adaptive eta: increase slowly to concentrate on peaks
        eta *= 1.01
        
        if verbose and iteration % 100 == 0:
            print(f"    MM iter {iteration}, obj={best_obj:.8f}, eta={eta:.2f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj

def softmax_gradient_descent(a0, time_limit, beta=50.0, lr=0.05, momentum=0.9):
    """Nesterov-accelerated gradient descent with softmax approximation to max."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(y)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()
        
        # Compute softmax gradient at y
        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv_y))
            w[int(np.argmax(conv_y))] = 1.0
        else:
            w = w / ws
        
        # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m]
        # Use the mat_vec trick: grad = 2 * mat_vec_w_a(w, y)
        grad = 2.0 * mat_vec_w_a(w, y)
        
        # Update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if iteration % 20 == 0:
            lr *= 0.995
            beta *= 1.02
            momentum = min(0.97, momentum + 0.0005)
        
        a = a_new if s_new > 1e-15 else best_a.copy()
        iteration += 1
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj

def coordinate_refine(a0, time_limit, rng):
    """Coordinate descent: try adjusting pairs of elements."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    # Sort indices by value (high value indices are more likely to need reduction)
    # Try moves from high to low
    while time.monotonic() - t0 < time_limit:
        batch = max(100, 100000 // n)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try several step sizes
            for frac in [0.01, 0.05, 0.1, 0.2, -0.01, -0.05, -0.1, -0.2]:
                delta = frac * a[i]
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                st = a_try.sum()
                if st < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try / st  # Re-normalize
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj
