"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient descent with Nesterov momentum as the
primary local optimizer, combined with LP-based Frank-Wolfe for larger jumps,
basin hopping for escaping local minima, and final local polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)_k))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = tau * conv - tau * conv.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # g[i] = sum_j a[j] * w[i+j]
    # Use direct computation (n is small enough)
    g = np.empty(n)
    for i in range(n):
        g[i] = np.dot(a, w[i:i+n])
    grad = (2.0 / tau) * g
    return grad, conv


def lse_gradient_descent(f0, time_limit, tau0=50.0, tau_end=5000.0,
                         lr0=0.005, mom0=0.9, verbose=False):
    """LSE smoothed gradient descent with Nesterov momentum on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr0
    mom = mom0
    no_improve = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau0 * (tau_end / tau0) ** progress
        
        # Nesterov lookahead
        f_look = project_simplex(f + mom * v)
        grad, conv_look = compute_lse_grad(f_look, tau)
        
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0

            no_improve += 1

def build_conv_matrix(f):
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    n = len(f)
    B = build_conv_matrix(f)
    
    if top_k is not None and top_k < 2*n-1:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
        rows = peaks
    else:
        rows = np.arange(len(conv_f))
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((len(rows), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[rows, :]
    b_ub = -conv_f[rows]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def subgrad_optimize(f0, time_limit, lr=0.002, mom=0.93, top_k=50, temp=0.0001):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        scaled -= scaled.max()

        ws = w.sum()

            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue

def local_refine(f0, time_limit, rng, num_rounds=10000):
    """Fine local search with decreasing perturbation scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    scale = 0.003
    
    for it in range(num_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        noise = rng.standard_normal(n) * scale
        f_pert = project_simplex(best_f + noise)
        obj = fast_convolve(f_pert).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_pert.copy()
        
        # Decay
        if it % 100 == 0:
            scale *= 0.97
    
    return best_f, best_obj


def symmetrize(a):
    """Try symmetrizing the sequence."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sigma * n)) ** 2)
        inits.append(g / g.sum())
    
    # U-shape
    u = np.exp(0.5 * ((x - mid) / (0.3 * n)) ** 2)
    inits.append(u / u.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())
    
    # Edge-heavy
    edge = 1.0 / (0.1 + np.abs(x - mid) / n)
    inits.append(edge / edge.sum())
    
    return inits

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed gradient descent with adaptive temperature as the primary
local optimizer, LP Frank-Wolfe for larger jumps, basin hopping for escaping
local minima, symmetrization, and final local refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)_k)) and the convolution."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = tau * conv - tau * conv.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # g[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    # Computed as cross-correlation of w and a
    g = np.empty(n)
    for i in range(n):
        g[i] = np.dot(a, w[i:i+n])
    grad = (2.0 / tau) * g
    return grad, conv


def lse_gradient_descent(f0, time_limit, tau0=50.0, tau_end=10000.0,
                         lr0=0.005, mom0=0.90, verbose=False):
    """LSE smoothed gradient descent with Nesterov momentum on the simplex.
    Adaptive temperature ramps from smooth to sharp."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr0
    mom = mom0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau0 * (tau_end / tau0) ** progress

        # Nesterov lookahead
        f_look = project_simplex(f + mom * v)
        grad, conv_look = compute_lse_grad(f_look, tau)

        v = mom * v - lr * grad
        f_new = project_simplex(f + v)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate: decay when stuck
        if no_improve > 500:
            lr *= 0.9
            if no_improve > 2000:
                # Restart with perturbation
                rng = np.random.default_rng((it * 7 + 3) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = lr0 * 0.5
                no_improve = 0

        # Slowly increase momentum
        mom = min(mom + 0.00001, 0.97)

        f = f_new

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD it={it}: best={best_obj:.10f} tau={tau:.0f} lr={lr:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def build_conv_matrix(f):
    """Build matrix B where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=None):
    """Solve linearized LP for Frank-Wolfe direction."""
    n = len(f)
    B = build_conv_matrix(f)

    if top_k is not None and top_k < 2 * n - 1:
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
        rows = peaks
    else:
        rows = np.arange(len(conv_f))

    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((len(rows), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[rows, :]
    b_ub = -conv_f[rows]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def fw_lp_optimize(f0, time_limit, top_k=25, verbose=False):
    """Frank-Wolfe with linearized LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42424)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        g = solve_linearized_lp(f, conv, top_k=top_k)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        target = np.maximum(f + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            obj = fast_convolve(f_new).max()
            if obj < best_val:
                best_val = obj
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                f_new = (1.0 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0.0)
                fs = f_new.sum()
                if fs < 1e-15:
                    break
                f_new = f_new / fs
                obj = fast_convolve(f_new).max()
                if obj < best_val:
                    best_val = obj
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()

            if best_val < best_obj - 1e-13:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_f, best_obj


def subgrad_optimize(f0, time_limit, lr=0.002, mom=0.93, top_k=50, temp=0.0001):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.002

        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def local_refine(f0, time_limit, rng, scale0=0.003):
    """Fine local search with geometrically decreasing perturbation scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    scale = scale0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        noise = rng.standard_normal(n) * scale
        f_pert = project_simplex(best_f + noise)
        obj = fast_convolve(f_pert).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_pert.copy()

        it2 = it + 1
        if it2 % 200 == 0:
            scale *= 0.95
            if scale < 1e-6:
                scale = 0.001

    return best_f, best_obj


def symmetrize(a):
    """Symmetrize: (a + reverse(a)) / 2, renormalized."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sigma * n)) ** 2)
        inits.append(g / g.sum())

    # U-shape (edge-heavy)
    u = np.exp(0.5 * ((x - mid) / (0.3 * n)) ** 2)
    inits.append(u / u.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Edge weight
    edge = 1.0 / (0.1 + np.abs(x - mid) / n)
    inits.append(edge / edge.sum())

    # W-shape
    w = 1.0 / (0.1 + np.abs(np.sin(2 * np.pi * x / n)))
    inits.append(w / w.sum())

    return inits
