"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed gradient descent,
LP-FW for periodic big steps, coordinate descent polishing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective_and_grad(a, t):
    """
    Compute the log-sum-exp objective and its gradient.
    Objective: (1/t) * log(sum(exp(t * (a*a)[k])))
    Gradient: computed via FFT-based cross-correlation.
    """
    n = len(a)
    conv = fast_convolve(a)
    lc = t * conv
    lc_max = lc.max()
    w = np.exp(lc - lc_max)
    w_sum = w.sum()
    obj = (lc_max + np.log(w_sum)) / t
    
    # Gradient via cross-correlation: grad[j] = (2/t) * sum_k w[k] * a[k-j]
    # Use FFT: IFFT(FFT(w_padded) * conj(FFT(a_padded)))
    fft_size = 1 << (4 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: IFFT(W * conj(A))
    # But we need sum_k w[k]*a[k-j], which is IFFT(W * conj(A))[j] for circular correlation
    # With sufficient zero-padding, circular = linear
    xcorr = np.fft.irfft(W * np.conj(A), fft_size)[:2 * n - 1]
    # xcorr[j] = sum_k w[k] * a[k-j] (for linear, with zero padding)
    grad = (2.0 / t) * xcorr[:n]
    
    return obj, grad


def smoothed_gradient_descent(a0, time_limit, t_init=10.0, t_final=200.0,

    s = a.sum()

    else:

    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    t = t_init

    total_time_est = time_limit
    lr = lr_init
    
    no_improve = 0
    while time.monotonic() - t0 < time_limit:

        # Increase temperature (sharper max approximation)

def solve_lp_direction(f, M, n, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 5:
            break
    
    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9995
        cur_temp *= 1.004
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def parametric_candidates(n):
    """Generate diverse parametric initial candidates."""
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Geometric (various ratios)
    for r in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
        a = r ** np.arange(n)
        a = np.maximum(a, 0)
        a = a / a.sum()
        candidates.append(a)
    
    # Sinusoidal modulation
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2 * np.pi * freq * np.arange(n) / n)
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            candidates.append(a)
    
    # Polynomial
    for p in [0.5, 1.0, 2.0, 3.0]:
        x = np.linspace(0, 1, n)
        a = x ** p
        a = a / a.sum()
        candidates.append(a)
        a = (1 - x) ** p
        a = a / a.sum()
        candidates.append(a)
    
    # Bimodal
    x = np.linspace(0, 1, n)
    for width in [0.1, 0.2, 0.3]:
        a = np.exp(-((x - 0.15) / width)**2) + np.exp(-((x - 0.85) / width)**2)
        a = np.maximum(a, 0)
        a = a / a.sum()
        candidates.append(a)
    
    return candidates
