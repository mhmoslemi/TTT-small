"""
Hybrid optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed subgradient descent, Gerchberg-Saxton spectral smoothing,
and LP-based direction finding with adaptive strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(m - 1, j + (n - 1))
        # C[k, j] = a[k - j] if 0 <= k-j < n
        # k - j ranges from 0 to n-1, so k ranges from j to j+n-1
        k_start = j
        k_end = min(m - 1, j + n - 1)
        C[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return C


def compute_subgradient(a, conv, top_k=15, temp=0.01):
    """Compute smoothed subgradient of max(conv) w.r.t. a."""
    n = len(a)
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peak_indices]

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peak_indices))
        w[0] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-14:
            continue
        k = int(peak_indices[idx])
        wi = w[idx]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        # d(conv[k])/d(a[j]) = 2*a[k-j] for j in [j_lo, j_hi]
        # a[k-j] for j from j_lo to j_hi = a[k-j_lo], a[k-j_lo-1], ..., a[k-j_hi]
        # = a[k-j_hi : k-j_lo+1][::-1]
        grad[j_lo:j_hi + 1] += 2.0 * wi * a[k - j_hi:k - j_lo + 1][::-1]

    return grad


def spectral_smooth(a, n_iters=3, flatten_power=0.5):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    a_work = a.copy()
    for _ in range(n_iters):
        A = np.fft.rfft(a_work, 2 * n)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        # Flatten: blend current magnitude with flat target
        target_mag = np.ones_like(mag) * np.sqrt(np.mean(mag ** 2))
        new_mag = mag ** (1 - flatten_power) * target_mag ** flatten_power
        A_new = new_mag * phase
        a_work = np.fft.irfft(A_new, 2 * n)[:n]
        a_work = np.maximum(a_work, 0.0)
        s = a_work.sum()
        if s > 1e-15:
            a_work = a_work / s
    return a_work


def objective(a):
    """Compute the objective value for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_direction(a, M=None):
    """
    Solve LP: min t s.t. conv(a,g)[k] <= t, sum(g)=1, g>=0.
    Returns (g, t_min, success).
    """
    n = len(a)
    m = 2 * n - 1
    C = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: C@g - t <= 0 for all k
    # => [C, -I_m] @ [g; t] <= 0
    A_ub = np.hstack([C, -np.eye(m)])
    b_ub = np.zeros(m)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x[:n]
            t_val = result.x[-1]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
            return g, t_val, True
    except Exception:
        pass
    return a.copy(), 0.0, False


def gradient_optimize(a0, time_limit, lr_init=0.02, top_k=20, temp_init=0.005,
                      momentum=0.85, spectral_every=50):
    """Smoothed subgradient with momentum and periodic spectral smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        grad = compute_subgradient(a, conv, top_k=top_k, temp=temp)

        # Momentum update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
        else:
            a = a_new / s_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        step_count += 1

        # Adaptive parameters
        lr *= 0.9997
        temp *= 1.002
        mom = min(0.95, mom + 0.0003)

        # Periodic spectral smoothing
        if step_count % spectral_every == 0 and no_improve > 20:
            a_spec = spectral_smooth(a, n_iters=2, flatten_power=0.3)
            conv_spec = fast_convolve(a_spec)
            obj_spec = 2.0 * n * conv_spec.max()
            if obj_spec < best_obj:
                a = a_spec
                conv = conv_spec
                best_obj = obj_spec
                best_a = a.copy()
            no_improve = max(0, no_improve - 10)
            v *= 0.5

        # Restart if stuck
        if no_improve > 300:
            break

    return best_a, best_obj


def lp_iterate(a0, time_limit, max_iters=5000):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    t_step = 0.3  # mixing parameter

    while time.monotonic() - t0 < time_limit and iteration < max_iters:
        g, t_val, success = lp_direction(a)

        if not success:
            no_improve += 5
        else:
            # Step: a_new = (1-t)*a + t*g
            a_try = (1 - t_step) * a + t_step * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()

                if obj_try < cur_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    cur_obj = obj_try
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    # Decrease step size slightly on success
                    t_step = max(0.05, t_step * 0.99)
                    continue
                else:
                    no_improve += 1
                    t_step = max(0.05, t_step * 0.9)
            else:
                no_improve += 1
                t_step = max(0.05, t_step * 0.9)

        iteration += 1

        # Handle stagnation
        if no_improve >= 50:
            # Try gradient-based escape
            esc_a, esc_obj = gradient_optimize(best_a, time_limit=3.0,
                                               lr_init=0.01, top_k=30,
                                               temp_init=0.003, spectral_every=20)
            if esc_obj < best_obj:
                best_obj = esc_obj
                best_a = esc_a.copy()
                a = best_a.copy()
                conv = fast_convolve(a)
                cur_obj = best_obj
            # Perturb
            rng = np.random.default_rng(iteration * 31 + 7)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0
            t_step = 0.2

            if no_improve > 200:
                break

        if iteration % 200 == 0:
            print(f"    LP iter {iteration}, best_obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj

def try_extend(a_short, n_target, rng):
    """Extend a sequence to a larger length."""
    n_s = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    # Interpolate
    x_old = np.linspace(0, 1, n_s)
    x_new = np.linspace(0, 1, n_target)
    a_ext = np.interp(x_new, x_old, a_norm)
    # Add small noise for diversity
    a_ext += rng.standard_normal(n_target) * 0.001
    a_ext = np.maximum(a_ext, 0)
    s_ext = a_ext.sum()
    if s_ext > 1e-15:
        a_ext = a_ext / s_ext
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase hybrid optimization combining subgradient,
    spectral smoothing, and LP-based direction finding.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def update_best(a):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST: {score:.6f} (n={len(a)})")
            return True
        return False

    # =========================================================
    # Phase 1: Exploration - try different n and methods (250s)
    # =========================================================
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 250.0

    candidates = []  # (score, n, a)

    # Start with parent
    if parent is not None:
        n_p = len(parent)
        a_p = np.maximum(parent, 0).astype(np.float64)
        s_p = a_p.sum()
        if s_p > 1e-10:
            a_p = a_p / s_p
            # Quick gradient optimization
            a_opt, obj = gradient_optimize(a_p, time_limit=30, lr_init=0.015,
                                           top_k=25, temp_init=0.004)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score):
                candidates.append((score, n_p, a_opt))
                print(f"  Parent grad: n={n_p}, score={score:.6f}")
                update_best(a_opt)

    # Try a few different n values with spectral initialization
    test_lengths = [512, 768, 1024, 1536]
    if parent is None:
        test_lengths = [512, 768, 1024]

    n_tested = 0
    for n in test_lengths:
        if time.monotonic() > deadline - (budget_s - 20) * 0.5:
            break
        if n == len(parent) if parent is not None else False:
            continue

        t_alloc = max(15.0, min(60.0, phase1_budget * 0.15))

        # Create initializations
        inits = []
        x = np.arange(n, dtype=np.float64)

        # Uniform
        inits.append(np.ones(n) / n)
        # Sine
        si = (1.0 + 0.5 * np.cos(2 * np.pi * x / n))
        inits.append(si / si.sum())
        # Bimodal edges
        edge = max(1, n // 4)
        bi = np.zeros(n)
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi[edge // 2:3 * edge // 2] = 0.5
        inits.append(bi / bi.sum())
        # Random
        inits.append(rng.dirichlet(np.ones(n) * 0.5))
        # From parent (interpolated)
        if parent is not None and len(parent) > 0:
            a_pn = np.maximum(parent, 0)
            sp = a_pn.sum()
            if sp > 1e-10:
                a_pn = a_pn / sp
                a_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(a_pn)), a_pn)
                a_interp = np.maximum(a_interp, 0)
                s_int = a_interp.sum()
                if s_int > 1e-10:
                    inits.append(a_interp / s_int)

        for i, a0 in enumerate(inits[:3]):
            if time.monotonic() > deadline - (budget_s - 20) * 0.4:
                break
            t_a = min(t_alloc, deadline - time.monotonic() - 300)
            if t_a < 5:
                break
            a_opt, obj = gradient_optimize(a0, time_limit=t_a, lr_init=0.015,
                                           top_k=20, temp_init=0.004)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score):
                candidates.append((score, n, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Explor n={n}, init{i}: {score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). "
          f"Best so far: {best_score:.6f}, {len(candidates)} candidates")

    # =========================================================
    # Phase 2: Hybrid optimization on top candidates (600s)
    # =========================================================
    print("=== Phase 2: Hybrid Optimization ===")
    t_start2 = time.monotonic()

    # Deduplicate by length, keep best per length
    seen = {}
    for score, n, a in candidates:
        if n not in seen or score < seen[n][0]:
            seen[n] = (score, a)
    top_candidates = sorted(seen.items(), key=lambda x: x[1][0])[:3]

    # Always include best
    if best_sequence is not None:
        n_best = len(best_sequence)
        if n_best not in seen:
            top_candidates.append((n_best, (best_score, best_sequence.copy())))

    remaining = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_candidates))
    t_per = max(60.0, remaining * 0.7 / n_cand)

    for i, (n_cand, (score_c, a_c)) in enumerate(top_candidates):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break
        print(f"  Hybrid candidate {i}: n={n_cand}, initial_score={score_c:.6f}, t={t_alloc:.0f}s")

        a_opt, obj = hybrid_optimize(a_c, time_limit=t_alloc)
        score = float(evaluate_sequence([float(x) for x in a_opt]))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Hybrid {i}: {score:.6f} *NEW BEST*")
            else:
                print(f"  Hybrid {i}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP deep refinement (400s)
    # =========================================================
    print("=== Phase 3: LP Refinement ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        a_best = best_sequence.copy()
        n_best = len(a_best)

        # Run LP on the best
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 50:
            t_lp = min(remaining * 0.6, 300.0)
            print(f"  LP on best: n={n_best}, t={t_lp:.0f}s")
            a_opt, obj = lp_iterate(a_best, time_limit=t_lp)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP best: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP best: {score:.6f}")

        # Try extending to larger n
        if time.monotonic() < deadline - 100 and n_best < 1536:
            n_ext = min(n_best * 2, 2048)
            a_ext = try_extend(best_sequence, n_ext, rng)
            # Quick optimize the extension
            t_ext = min(60.0, deadline - time.monotonic() - 80)
            if t_ext > 10:
                print(f"  Extending to n={n_ext}, t={t_ext:.0f}s")
                a_opt_ext, obj_ext = hybrid_optimize(a_ext, time_limit=t_ext)
                score_ext = float(evaluate_sequence([float(x) for x in a_opt_ext]))
                if np.isfinite(score_ext) and score_ext < best_score:
                    best_score = score_ext
                    best_sequence = a_opt_ext.copy()
                    print(f"  Extended: {score_ext:.6f} *NEW BEST*")
                else:
                    print(f"  Extended: {score_ext:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final polishing (remaining time)
    # =========================================================
    print("=== Phase 4: Polishing ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        n_best = len(best_sequence)

        # Multiple short hybrid runs with perturbation
        for pi in range(5):
            if time.monotonic() > deadline - 10:
                break
            t_pol = min(remaining / (5 - pi) * 0.7, 60.0)
            if t_pol < 5:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.002 * pi
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s0 = a0.sum()
                if s0 > 1e-15:
                    a0 = a0 / s0
                else:
                    a0 = best_sequence.copy()

            a_opt, obj = hybrid_optimize(a0, time_limit=t_pol)
            score = float(evaluate_sequence([float(x) for x in a_opt]))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {pi}: {score:.6f} *NEW BEST*")
            else:
                print(f"  Polish {pi}: {score:.6f}")

        # Final LP micro-step
        if time.monotonic() < deadline - 15:
            t_final = min(15.0, deadline - time.monotonic() - 5)
            if t_final > 5:
                a_opt, obj = lp_iterate(best_sequence, time_limit=t_final, max_iters=500)
                score = float(evaluate_sequence([float(x) for x in a_opt]))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence([float(x) for x in best_sequence]))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
