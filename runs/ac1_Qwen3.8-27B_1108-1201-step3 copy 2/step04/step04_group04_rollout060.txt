"""
Hybrid optimizer for minimizing 2*n*max(a*a)/(sum(a))^2:
1. Soft-gradient descent with Barzilai-Borwein adaptive step size
2. FFT-domain spectral projection (Gerchberg-Saxton style)
3. Multi-length search with diverse initializations
4. Coordinate descent polishing
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_full_gradient(a, conv, temp=0.01):
    """Compute gradient using softmax-weighted all peaks."""
    n = len(a)
    # Use softmax over ALL conv positions (not just top-k)
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    # For each position k with significant weight, add a + flip_shift(a, k)
    # Optimize: only process positions with w > threshold
    threshold = 1e-6
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        if wk < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += wk * a[k - idxs]
    return grad

def bb_step_size(grad, delta_a, delta_f):
    """Barzilai-Borwein step size."""
    dg = np.dot(delta_a, grad)
    gg = np.dot(grad, grad)
    if gg < 1e-30:
        return 0.01
    # BB1
    if abs(delta_f) < 1e-15:
        return 0.01
    bs1 = np.dot(delta_a, delta_a) / dg if abs(dg) > 1e-15 else 0.01
    return max(1e-6, min(bs1, 1.0))

def gradient_optimize(a0, time_limit, lr_init=0.01, temp_init=0.005, use_bb=True):
    """Gradient descent with soft gradient and optional BB step."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    prev_a = a.copy()
    prev_obj = best_obj
    grad_prev = None
    
    while time.monotonic() - t0 < time_limit:
        grad = compute_full_gradient(a, conv, temp=temp)
        
        if use_bb and grad_prev is not None:
            delta_a = a - prev_a
            delta_f = best_obj - prev_obj
            lr = bb_step_size(grad, delta_a, delta_f)
            lr = max(1e-6, min(lr, 0.1))
        
        # Gradient step with projection
        trial = a - lr * grad
        trial = np.maximum(trial, 0.0)
        s_trial = trial.sum()
        if s_trial < 1e-15:
            break
        trial = trial / s_trial
        
        conv_trial = fast_convolve(trial)
        obj_trial = 2.0 * n * conv_trial.max()
        
        if obj_trial < best_obj - 1e-12:
            # Accept
            grad_prev = grad.copy()
            prev_a = a.copy()
            prev_obj = best_obj
            a = trial
            conv = conv_trial
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
        else:
            # Reject, reduce step
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                scale = 0.01
                a = best_a + rng.standard_normal(n) * scale
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                else:
                    a = best_a.copy()
                conv = fast_convolve(a)
                no_improve = 0
                lr = lr_init
        
        # Gradually increase temperature (explore more peaks)
        temp *= 1.001
    
    return best_a, best_obj

def spectral_projection(a0, time_limit, n_iters=None):
    """
    Gerchberg-Saxton style: alternate between spectral flattening
    and time-domain non-negativity projection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    it = 0
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    # Target: cap the spectrum at some level
    # Start with a cap slightly above the current peak
    conv = fast_convolve(a)
    peak = conv.max()
    # The spectral cap corresponds to: |A[k]|^2 = N * c[k] for the peak
    # We want to cap c[k] at some level
    cap_fraction = 0.8  # Target: reduce peak by 20%
    
    while time.monotonic() - t0 < time_limit:
        if n_iters is not None and it >= n_iters:
            break
        it += 1
        
        # Zero-pad and FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        
        # Current spectral magnitudes
        mag = np.abs(A)
        
        # Cap the magnitudes (keep DC)
        # Target: all |A[k]| <= |A[0]| * cap_factor for k != 0
        # But we need to be careful: we can't just cap arbitrarily
        # because the inverse must give a valid signal.
        
        # Strategy: cap at a level that's a fraction of the current peak
        peak_mag = mag.max()
        dc = mag[0]
        # Cap: want max mag to be at most dc * (2n-1) / (2n) * peak_fraction
        # Actually, let's just cap at a fixed fraction of the max
        new_cap = peak_mag * 0.95  # Reduce peak by 5% each iteration
        # But don't go below DC (that would make the signal complex or negative)
        new_cap = max(new_cap, dc * 1.0)
        
        # Apply cap
        masked_mag = np.minimum(mag, new_cap)
        # Keep DC
        masked_mag[0] = dc
        # Maintain phase
        phase = np.angle(A)
        A_new = masked_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try less aggressive cap
            no_improve += 1
            if no_improve > 50:
                # Try a different strategy
                break
        
    return best_a, best_obj

def coordinate_descent(a0, time_limit):
    """Fast coordinate descent for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    peak_idx = int(np.argmax(conv))
    
    while time.monotonic() - t0 < time_limit:
        # Try to reduce the peak by adjusting elements that contribute to it
        # The peak is at position peak_idx
        # c[peak_idx] = sum_j a[j]*a[peak_idx-j]
        # To reduce this, decrease a[j] where a[j]*a[peak_idx-j] is large
        
        # Find the top contributors to the peak
        j_range = np.arange(max(0, peak_idx - n + 1), min(n - 1, peak_idx) + 1)
        products = a[j_range] * a[peak_idx - j_range]
        
        # Try reducing the largest contributor
        if len(j_range) == 0:
            break
        
        max_contrib = np.argmax(products)
        j = j_range[max_contrib]
        partner = peak_idx - j
        
        # Try reducing a[j] and a[partner]
        improvement = False
        for frac in [0.1, 0.05, 0.02, 0.01, 0.005]:
            for target in [j, partner, -1]:  # -1 means both
                a_try = a.copy()
                if target == j:
                    a_try[j] *= (1 - frac)
                elif target == partner:
                    if partner != j:
                        a_try[partner] *= (1 - frac)
                else:  # both
                    a_try[j] *= (1 - frac * 0.5)
                    if partner != j:
                        a_try[partner] *= (1 - frac * 0.5)
                
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    peak_idx = int(np.argmax(conv))
                    no_improve = 0
                    improvement = True
                    break
            if improvement:
                break
        
        if not improvement:
            no_improve += 1
            if no_improve > 500:
                break
            # Perturb
            mask = rng.random(n) < 0.1
            a[mask] *= (1 + rng.standard_normal(mask.sum()) * 0.05)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            peak_idx = int(np.argmax(conv))
    
    return best_a, best_obj

def generate_initializations(n, parent, rng):

    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Sine-based

"""
Hybrid LP + gradient optimizer for minimizing 2*n*max(a*a)/(sum(a))^2.
Key innovations over previous approach:
1. Direct LP formulation: min peak s.t. sum(g)=S, g>=0 (no rescaling needed)
2. Fast iteration with precomputed structure
3. Adaptive switching between LP and gradient steps
4. Multi-length exploration with diverse initializations
5. Spectral (FFT-domain) projection as auxiliary method
6. Aggressive convergence with periodic restarts
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (linear convolution, 2n-1 x n)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_min_peak(a, S, C=None):
    """
    Solve: min m s.t. C@g <= m*1, sum(g) = S, g >= 0.
    Returns (g_opt, m_opt, success).
    """
    n = len(a)
    m_dim = 2 * n - 1
    if C is None:
        C = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, m]  total n+1
    # Objective: minimize m -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: C@g - m <= 0 for each row -> [C, -1] @ [g; m] <= 0
    A_ub = np.hstack([C, -np.ones((m_dim, 1))])
    b_ub = np.zeros(m_dim)
    
    # Equality: sum(g) = S -> [1^T, 0] @ [g; m] = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g >= 0, m >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success:
            g_opt = result.x[:n]
            m_opt = result.x[-1]
            if m_opt < 1e-15:
                m_opt = 0.0
            return g_opt, m_opt, True
    except Exception:
        pass
    
    # Fallback: return current a
    conv = fast_convolve(a)
    return a.copy(), conv.max(), False

def _compute_gradient(a, conv, temp=0.002, top_k=None):
    """Compute soft-gradient of the peak convolution."""
    n = len(a)
    if top_k is None:
        top_k = min(50, len(conv))
    
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def gradient_optimize(a0, time_limit, lr_init=0.01, temp_init=0.003):
    """Nesterov-style gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = 0.9
    no_improve = 0
    prev_obj = best_obj
    prev_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        conv_y = fast_convolve(y)
        grad = _compute_gradient(y, conv_y, temp=temp, top_k=40)
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive: BB-inspired step
        delta = a - prev_a
        delta_f = prev_obj - cur_obj
        dg = np.dot(delta, grad)
        if abs(dg) > 1e-10:
            lr_new = np.dot(delta, delta) / dg
            lr = max(1e-5, min(lr_new, 0.05))
        lr *= 0.999
        temp *= 1.002
        mom = min(0.96, mom + 0.0001)
        
        prev_obj = cur_obj
        prev_a = a.copy()
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def spectral_projection(a0, time_limit):
    """FFT-domain iterative projection for flat spectrum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Cap the spectrum
        peak_mag = mag.max()
        dc = mag[0]
        # Target: reduce peak by a small amount
        cap = peak_mag * 0.97
        cap = max(cap, dc * 0.99)
        
        new_mag = np.minimum(mag, cap)
        new_mag[0] = dc
        A_new = new_mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.001)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.001)
    inits['sine2'] /= inits['sine2'].sum()
    
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    edge2 = max(1, n // 6)
    tri = np.zeros(n)
    tri[:edge2] = 1.0
    mid = n // 2
    tri[mid - edge2//2:mid + edge2//2] = 1.0
    tri[-edge2:] = 1.0
    inits['trimodal'] = tri / tri.sum()
    
    # Hamming
    hm = 0.54 - 0.46 * np.cos(2 * np.pi * x / max(1, n - 1))
    inits['hamming'] = hm / hm.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

    Main search: hybrid LP + gradient + spectral optimization.

    rng = np.random.default_rng(seed)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            score_p = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score_p):
                best_score = score_p
                best_sequence = p.copy()
                print(f"Parent: n={len(p)}, score={score_p:.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent error: {e}")
    
    if best_sequence is None:
        n0 = 512
        best_sequence = np.ones(n0) / n0
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Init uniform n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Quick gradient warmup (200s) ===
    print("=== Phase 1: Gradient Warmup ===")
    t1 = time.monotonic()
    p1_budget = 200.0
    
    n_main = len(best_sequence)
    lengths = [n_main]
    for alt in [512, 768, 1024]:
        if alt != n_main and alt not in lengths:
            lengths.append(alt)
    lengths.sort(key=lambda x: -1 if x == n_main else 0)
    
    for n in lengths[:3]:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_interp', 'bimodal', 'uniform', 'sine1', 'random']
        names = [nm for nm in priority if nm in inits][:3]
        
        t_per = max(15.0, p1_budget / (3 * 3))
        for name in names:
            if time.monotonic() > deadline - 550:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 500)
            if t_alloc < 8:
                break

            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc, lr_init=0.01, temp_init=0.003)

                else:
                    print(f"  P1 n={n} {name}: {score:.6f}")
    
    print(f"P1 done ({time.monotonic()-t1:.0f}s), best={best_score:.6f}")
    
    # === Phase 2: LP deep optimization (1000s) ===
