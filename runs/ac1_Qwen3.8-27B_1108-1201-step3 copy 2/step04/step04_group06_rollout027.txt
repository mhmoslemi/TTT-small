def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_obj_and_grad(x, t):
    """
    Compute the smoothed objective F_t(a) = 2*n*t*log(sum(exp(c/t)))
    and its gradient with respect to x (log-domain), where a = softmax(x).
    """
    # Compute a = softmax(x) for numerical stability
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    a = a / s
    n = len(a)
    
    # Compute convolution
    c = fast_convolve(a)
    
    # Compute log-sum-exp of c/t
    scaled = c / t
    scaled_max = scaled.max()
    log_sum_exp = scaled_max + np.log(np.exp(scaled - scaled_max).sum())
    obj = 2.0 * n * t * log_sum_exp
    
    # Compute probability distribution p
    p = np.exp(scaled - log_sum_exp)  # This is exp(c/t) / sum(exp(c/t))
    
    # Compute cross-correlation: corr[j] = sum_k p[k] * a[k-j] for valid k-j
    fft_size = 1 << (len(p) + n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    # Gradient of F wrt a: g = (4n/t) * corr
    g = (4.0 * n / t) * corr
    
    # Gradient of F wrt x: grad_x = a * (g - dot(g, a))
    mean_g = np.dot(g, a)
    grad_x = a * (g - mean_g)
    
    return obj, grad_x

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_obj_and_grad(x, t):
    """
    Compute smoothed objective F_t and gradient in log-domain.
    a = softmax(x), F_t = 2*n*t*logsumexp(c/t) where c = conv(a,a).
    """
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    a = a / s
    n = len(a)

    c = fast_convolve(a)

    scaled = c / t
    scaled_max = scaled.max()
    log_sum_exp = scaled_max + np.log(np.exp(scaled - scaled_max).sum())
    obj = 2.0 * n * t * log_sum_exp

    p = np.exp(scaled - log_sum_exp)

    fft_size = 1 << (len(p) + n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n]

    g = (4.0 * n / t) * corr
    mean_g = np.dot(g, a)
    grad_x = a * (g - mean_g)

    return obj, grad_x

def generate_inits(a0, rng, n):

    # Exponential decay

        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)

    # Perturbed best (same length)

                                           maxfun_per_phase=20000, n_phases=5)
            record(a_opt, "lbfgs_p3")

    # === Phase 4: Fine EG from L-BFGS result ===
    print(f"\n=== Phase 4: Fine EG ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        t4 = min(200, deadline - time.monotonic() - 150)
        if t4 > 15:
            a_opt, _ = eg_optimize(best_a, time_limit=t4,
                                   eta0=0.3, temp0=0.001,
                                   temp_decay=0.9998, eta_decay=0.99999,
                                   top_k_frac=0.5)
            record(a_opt, "eg_fine")

    # === Phase 5: Block coordinate descent ===
    print(f"\n=== Phase 5: Block Descent ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        t5 = min(100, deadline - time.monotonic() - 80)
        if t5 > 10:
            a_opt, _ = block_descent(best_a, time_limit=t5,
                                      block_size=24, max_rounds=300)
            record(a_opt, "block")

    # === Phase 6: Local search polish ===
    print(f"\n=== Phase 6: Local Search ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        t6 = min(60, deadline - time.monotonic() - 40)
        if t6 > 5:
            a_opt, _ = local_search(best_a, time_limit=t6, max_rounds=150)
            record(a_opt, "local")

    # === Phase 7: Multi-init exploration ===
    print(f"\n=== Phase 7: Multi-Init ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        inits = generate_inits(best_a, rng, n)
        for i, f_init in enumerate(inits[1:6]):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_alloc = min(40, rem / max(1, 6 - i))
            if t_alloc < 8:
                break
            # Quick L-BFGS from this init
            a_opt, _ = lbfgs_continuation(f_init, time_limit=t_alloc,
                                           maxfun_per_phase=10000, n_phases=3)
            record(a_opt, f"init{i}")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different Lengths ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        for n_new in [n // 2, int(n * 1.5), int(n * 0.75)]:
            if n_new < 32 or n_new > 2000:
                continue
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            # Interpolate
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_new)
            a_new = np.interp(x_new, x_old, best_a)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                t_alloc = min(30, rem / 2)
                a_opt, _ = lbfgs_continuation(a_new, time_limit=t_alloc,
                                               maxfun_per_phase=8000, n_phases=3)
                record(a_opt, f"len{n_new}")

    # === Phase 9: Final fine-tuning ===
    print(f"\n=== Phase 9: Final Polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_a is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = lbfgs_continuation(best_a, time_limit=min(rem, 60),
                                           maxfun_per_phase=15000, n_phases=3)
            record(a_opt, "final_lbfgs")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = local_search(best_a, time_limit=min(rem, 20), max_rounds=50)
            record(a_opt, "final_local")

    # Fallback
    if best_a is None:
        best_a = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
