    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smooth surrogate and temperature annealing
2. LP-based direction finding for refinement
3. Multi-scale length exploration with diverse initializations
4. Hybrid polish combining gradient, LP, and coordinate descent

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * fast_convolve(a).max() / (s * s)


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_smooth_gradient(w, t, top_k=100):
    """Compute gradient of 2n*softmax(conv(w,w),t)/(sum(w))^2."""
    n = len(w)
    s = w.sum()
    conv = fast_convolve(w)
    
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    
    peak_vals = conv[peaks]
    scaled = peak_vals / t
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    soft_max = peak_vals.max() + t * np.log(exp_scaled.sum())
    
    # dM_soft/dw[j] = 2 * sum_k p[k] * w[k-j] for valid k-j
    grad_msoft = np.zeros(n)
    for idx in range(len(peaks)):
        if p[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad_msoft[idxs] += 2.0 * p[idx] * w[k - idxs]
    
    # Full gradient: 2n/s^3 * (s * dM/dw[j] - 2*soft_max*w[j])
    grad = 2.0 * n * (s * grad_msoft - 2.0 * soft_max * w) / (s ** 3)
    obj = 2.0 * n * soft_max / (s * s)
    return obj, grad


def lsbfgs_optimize(a0, time_limit, t_init=0.05, t_final=0.002, top_k=100, max_anneal_steps=8):
    """
    L-BFGS optimization with log-sum-exp surrogate and temperature annealing.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / np.sqrt(n)
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    temperatures = np.geomspace(t_init, t_final, max_anneal_steps)
    time_per_anneal = time_limit / max_anneal_steps
    
    for step_i, t_val in enumerate(temperatures):
        if time.monotonic() - t0 > time_limit:
            break
        
        work = best_a.copy()
        
        def obj_grad(w):
            w = np.maximum(w, 0.0)
            ws = w.sum()
            if ws < 1e-15:
                return 1e10, np.zeros(n)
            return compute_smooth_gradient(w, t_val, top_k)
        
        max_iter = max(100, int(time_per_anneal * 200))
        
        try:
            result = minimize(
                obj_grad, work,
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': max_iter,
                    'maxfun': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-10,
                    'maxls': 30,
                    'iprint': -1,
                },
                bounds=[(0, 1000.0)] * n
            )
            
            if result.x is not None:
                cand = np.maximum(result.x, 0.0)
                cs = cand.sum()
                if cs > 1e-15:
                    obj = compute_obj(cand)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = cand.copy()
        except Exception:
            pass
        
        # Check time
        if time.monotonic() - t0 > time_limit - 1:
            break
    
    return best_a, best_obj


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns direction g."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, 1000.0)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > a.sum() * (1 + 1e-6):
                return g
    except Exception:
        pass
    return None


def lp_refine(a0, time_limit, max_iter=5000):
    """LP-based iterative refinement with adaptive step sizing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / np.sqrt(n)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M / (s * s)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    
    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M * 1.00001)
        
        if g is not None:
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d > 1e-10:
                # Line search
                best_t = 0.0
                best_local_obj = best_obj
                for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max() / (s_try * s_try)
                    if obj_try < best_local_obj - 1e-12:
                        best_local_obj = obj_try
                        best_t = t
                        break
                
                if best_t > 0:
                    a = np.maximum(a + best_t * d, 0.0)
                    conv = fast_convolve(a)
                    M = conv.max()
                    s = a.sum()
                    cur_obj = 2.0 * n * M / (s * s)
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Perturbation on stagnation
        if no_improve >= 50:
            rng = np.random.default_rng(int(time.monotonic() * 1000) + perturb_count)
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            if perturb_count > 60:
                break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Randomized coordinate descent: transfer mass between positions."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = len(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    conv_cached = fast_convolve(a)
    
    while time.monotonic() - t0 < time_limit:
        batch = max(1, min(50000, int(2e6 / n)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            improved = False
            for delta in [0.002, 0.01, 0.02, -0.002, -0.01, -0.02,
                          0.005, 0.05, -0.005, -0.05]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_copy = a.copy()
                a_copy[i] += delta
                a_copy[j] -= delta
                conv_try = fast_convolve(a_copy)
                obj_try = compute_obj(a_copy)
                if obj_try < best_obj - 1e-10:
                    a = a_copy
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        
        if no_improve > 300:
            break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):

    inits = {}

    inits['uniform'] = np.ones(n)
    
    # Hann window
    inits['hann'] = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits['hann'][inits['hann'] < 0.01] = 0.01
    
    # Blackman window
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    inits['blackman'] = a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) + a2 * np.cos(4 * np.pi * x / (n - 1))
    inits['blackman'] = np.maximum(inits['blackman'], 0.01)
    
    # Flat-top window (designed for minimal spectral leakage)
    b = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    ft = b[0] * np.ones(n)
    for k, bk in enumerate(b[1:], 1):
        ft += bk * np.cos(2 * np.pi * k * x / (n - 1))
    inits['flattop'] = np.maximum(ft, 0.01)
    
    # Bimodal (edges)

    bi[-edge_w:] = 1.0
    # Smooth the transitions
    ramp = max(1, edge_w // 4)
    bi[ramp:edge_w] = np.linspace(0, 1, edge_w - ramp)
    bi[-edge_w:-ramp] = np.linspace(1, 0, edge_w - ramp)
    inits['bimodal'] = bi + 0.01
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g + 0.01
    
    # Sine squared
    inits['sine'] = np.sin(np.pi * x / n) ** 2 + 0.01
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5) * n
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp * n
    
    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target)
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0.01 / n_target)
    return a_ext


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent if available
    parent = None
    try:
        p_raw = height_sequence_1
        p = np.array([float(x) for x in p_raw])
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
                print(f"Parent: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass
    
    # =========================================================
    # Phase 1: L-BFGS scan across lengths and initializations
    # =========================================================
    print("=== Phase 1: L-BFGS Scan ===")
    t_start = time.monotonic()
    
    candidate_lengths = [512, 768, 1024, 1536, 2048]
    # Prioritize longer lengths and parent-based inits
    scan_configs = []
    for n in candidate_lengths:
        for init_name in ['parent', 'hann', 'bimodal', 'uniform', 'flattop']:
            scan_configs.append((n, init_name))
    
    # Filter to configs we can actually generate
    valid_configs = []
    for n, init_name in scan_configs:
        inits_tmp = generate_initializations(n, parent, rng)
        if init_name in inits_tmp:
            valid_configs.append((n, init_name, inits_tmp[init_name]))
    
    # Limit to ~12 configs for the scan
    valid_configs = valid_configs[:12]
    
    t_per_scan = max(15.0, (250.0) / len(valid_configs))
    scan_results = []
    
    for n, init_name, a0 in valid_configs:
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(t_per_scan, deadline - time.monotonic() - 400)
        if t_alloc < 10:
            break
        
        a_opt, obj = lsbfgs_optimize(a0, time_limit=t_alloc,
                                      t_init=0.08, t_final=0.003,
                                      top_k=80, max_anneal_steps=6)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, init_name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  L-BFGS n={n}, {init_name}: {score:.6f} *NEW BEST*")
            else:
                print(f"  L-BFGS n={n}, {init_name}: {score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP deep optimization on best candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Select top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break
    
    if best_sequence is not None and len(best_sequence) not in seen_lengths:
        top_configs.append((len(best_sequence), best_sequence.copy()))
        seen_lengths.add(len(best_sequence))
    
    remaining = deadline - time.monotonic() - 300.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.55 / n_cand)
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 15:
            break
        
        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_refine(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Extended L-BFGS + LP hybrid on the best
    # =========================================================
    print("=== Phase 3: Hybrid Refinement ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        
        # Multiple rounds of alternating L-BFGS and LP
        n_rounds = 5
        t_per_round = remaining / n_rounds
        
        for rnd in range(n_rounds):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per_round, deadline - time.monotonic() - 80)
            if t_alloc < 10:
                break
            
            if rnd % 2 == 0:
                # L-BFGS round
                a_opt, obj = lsbfgs_optimize(best_sequence, time_limit=t_alloc,
                                              t_init=0.03, t_final=0.001,
                                              top_k=120, max_anneal_steps=4)
            else:
                # LP round
                a_opt, obj = lp_refine(best_sequence, time_limit=t_alloc)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Hybrid round {rnd}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Hybrid round {rnd}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Coordinate descent + fine L-BFGS
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        
        # Coordinate descent
        t_cd = min(remaining * 0.3, 80.0)
        if t_cd > 10:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
        
        # Fine L-BFGS with very low temperature
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            a_opt, obj = lsbfgs_optimize(best_sequence, time_limit=remaining * 0.6,
                                          t_init=0.01, t_final=0.0005,
                                          top_k=200, max_anneal_steps=3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FineBFGS: {score:.6f} *NEW BEST*")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(30.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lp_refine(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Ensure valid output
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256)
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
