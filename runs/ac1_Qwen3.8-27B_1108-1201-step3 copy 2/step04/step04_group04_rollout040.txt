import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)

def compute_conv_and_grad(a, p=None):
    """Compute conv(a,a) and gradient of log-sum-exp smoothing."""
    n = len(a)
    conv = fast_convolve(a)
    if p is None:
        # Use uniform p (corresponds to very large mu)
        p = np.ones(2 * n - 1) / (2 * n - 1)
    # grad[i] = 2 * sum_{l=0}^{n-1} p[i+l] * a[l] (p[k]=0 for k >= 2n-1)
    grad = np.zeros(n)
    for i in range(n):
        # l ranges from 0 to min(n-1, 2n-2-i)
        l_end = min(n - 1, 2 * n - 2 - i)
        if l_end >= 0:
            grad[i] = 2.0 * np.dot(p[i:i + l_end + 1], a[:l_end + 1])
    return conv, grad

def fista_optimize(a0, mu, time_limit, L=4.0, tol=1e-10):
    """Run FISTA for a fixed smoothing parameter mu."""
    a = simplex_project(a0.copy().astype(np.float64))
    n = len(a)
    a_prev = a.copy()
    t = 1.0
    step = 1.0 / L
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        alpha = (t - 1.0) / (t + 1.0)
        y = a + alpha * (a - a_prev)
        y = simplex_project(y)
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = conv_y / mu
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        p = w / w_sum
        
        n_y = len(y)
        grad = np.zeros(n_y)
        for i in range(n_y):
            l_end = min(n_y - 1, 2 * n_y - 2 - i)
            if l_end >= 0:
                grad[i] = 2.0 * np.dot(p[i:i + l_end + 1], y[:l_end + 1])
        
        # Gradient step + projection
        a_new = simplex_project(y - step * grad)
        
        # Update
        a_prev = a
        a = a_new
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        
        # Check convergence
        if iteration % 10 == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = iteration
        
        iteration += 1
        
        if iteration - last_improve > 500:
            break
    
    return best_a, best_obj
