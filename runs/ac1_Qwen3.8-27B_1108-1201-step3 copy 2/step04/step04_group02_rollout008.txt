import numpy as np

from scipy.optimize import minimize_scalar

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, conv, tau):
    """Compute gradient of tau*log(sum(exp(conv/tau))) w.r.t. a."""
    n = len(a)
    # Softmax weights over convolution
    scaled = conv / tau
    scaled -= scaled.max()

    # w has length 2n-1, a has length n, output has length n
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def fast_convolve(a):

    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, conv, tau):
    """Compute gradient of tau*log(sum(exp(conv/tau))) w.r.t. a using correlation."""
    n = len(a)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    # Gradient: 2 * correlate(w, a, 'valid')
    # w has length 2n-1, a has length n, output length n
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def fista_optimize(a0, tau, time_limit, L_init=None):
    """FISTA on smooth max approximation with simplex projection. No backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if L_init is None:
        conv0 = fast_convolve(a)
        L_init = 4.0 * n * max(1.0, conv0.max() / tau)

    L = L_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    zk = a.copy()
    tk = 1.0
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        conv_zk = fast_convolve(zk)
        gk = compute_smooth_gradient(zk, conv_zk, tau)

        yk = project_simplex(zk - gk / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        zk = yk + (tk - 1.0) / t_new * (yk - a)
        zk_sum = zk.sum()
        if zk_sum > 1e-15:
            zk = zk / zk_sum
        else:
            zk = yk.copy()
        a = yk.copy()
        tk = t_new

        if it % 5 == 0:
            conv_a = fast_convolve(a)
            obj = conv_a.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                # Slightly reduce L when making progress
                L *= 0.98
            else:
                no_improve += 1
                if no_improve > 200:
                    L *= 0.95
                    no_improve = 0

    # Final evaluation
    conv_a = fast_convolve(a)
    if conv_a.max() < best_obj:
        best_obj = conv_a.max()
        best_a = a.copy()

    return best_a, best_obj


def adam_optimize(a0, time_limit, tau=0.01, lr=0.005):
    """Adam optimizer on the smooth max approximation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        conv = fast_convolve(a)
        g = compute_smooth_gradient(a, conv, tau)

        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        lr_t = lr * (1.0 + 0.5 * np.exp(-it / 200.0))
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)

        if it % 10 == 0:
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def coord_descent_fast(a0, time_limit, max_sweeps=200):
    """
    Fast coordinate descent with incremental convolution update.
    For each coordinate, performs 1D bounded minimization of the full objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a).copy()
    best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    t0 = time.monotonic()

    def obj_with_change(j, t, conv_arr, sa):
        """Evaluate objective after setting a[j]=t using cached conv."""
        old = a[j]
        if abs(t - old) < 1e-15:
            return 2.0 * n * conv_arr.max() / (sa * sa)
        new_sa = sa + (t - old)
        if new_sa < 1e-10:
            return 1e10
        # Compute max of updated convolution without modifying conv_arr
        # Positions affected: k in [j, j+n-1]
        start_k = j
        end_k = min(j + n - 1, 2 * n - 2)
        delta = t - old
        local_max = conv_arr[start_k:end_k + 1].copy()
        for idx in range(len(local_max)):
            k = start_k + idx
            if k - j == j:  # k == 2j
                local_max[idx] += t * t - old * old
            else:
                local_max[idx] += 2.0 * delta * a[k - j]
        # The max is max of (max of unmodified parts, max of modified parts)
        if start_k > 0:
            unmod_before = conv_arr[:start_k].max()
        else:
            unmod_before = 0.0
        if end_k < 2 * n - 2:
            unmod_after = conv_arr[end_k + 1:].max()
        else:
            unmod_after = 0.0
        unmod_max = max(unmod_before, unmod_after)
        max_conv = max(unmod_max, local_max.max())
        return 2.0 * n * max_conv / (new_sa * new_sa)

    sweep = 0
    while time.monotonic() - t0 < time_limit - 2 and sweep < max_sweeps:
        sweep += 1
        any_improved = False
        sa = a.sum()

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break

            old_val = a[j]
            if old_val < 1e-12:
                # Try setting to a small positive value
                candidates = [old_val, 1e-4, 1e-3, 0.01, sa / n]
                best_t = old_val
                best_local = obj_with_change(j, old_val, conv, sa)
                for c in candidates:
                    if c <= 0:
                        continue
                    obj_c = obj_with_change(j, c, conv, sa)
                    if obj_c < best_local:
                        best_local = obj_c
                        best_t = c
            else:
                # Bounded 1D minimization
                t_lo = 0.0
                t_hi = max(old_val * 3.0, sa * 0.1)

                def obj_1d(t):
                    return obj_with_change(j, t, conv, sa)

                try:
                    res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                          options={'xatol': 1e-8, 'maxiter': 40})
                    if res.success:
                        best_t = res.x
                        best_local = res.fun
                    else:
                        best_t = old_val
                        best_local = obj_with_change(j, old_val, conv, sa)
                except Exception:
                    # Fallback: try a few values
                    best_t = old_val
                    best_local = obj_with_change(j, old_val, conv, sa)
                    for frac in [0.5, 0.7, 1.3, 1.5, 2.0]:
                        c = old_val * frac
                        if c > 0:
                            obj_c = obj_with_change(j, c, conv, sa)
                            if obj_c < best_local:
                                best_local = obj_c
                                best_t = c

            if best_local < best_obj - 1e-12:
                # Update a[j] and conv
                old = a[j]
                new = max(0.0, best_t)
                delta = new - old
                a[j] = new
                start_k = j
                end_k = min(j + n - 1, 2 * n - 2)
                for k in range(start_k, end_k + 1):
                    if k - j == j:
                        conv[k] += new * new - old * old
                    else:
                        conv[k] += 2.0 * delta * a[k - j]
                best_obj = best_local
                best_a = a.copy()
                any_improved = True

        if not any_improved:
            break

    return best_a, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(12345)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    two_bump = np.exp(-((x - n * 0.25) / (n * 0.08)) ** 2) + np.exp(-((x - n * 0.75) / (n * 0.08)) ** 2) + 0.01
    inits.append(two_bump / two_bump.sum())

    tri = 1.0 - np.abs(x - n / 2) / (n / 2 + 1e-10)
    inits.append(tri / tri.sum())

    # Power law
    pw = (x + 1.0) ** 0.3
    inits.append(pw / pw.sum())

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.7 * (px / ps) + 0.3 * (np.ones(n) / n)
            inits.append(blended / blended.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses FISTA + coordinate descent + Adam hybrid approach.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            # Scale to get the actual sequence values (for evaluate_sequence)
            parent_raw = p * (p.sum() / np.ones(1))  # just use as-is
            parent_raw = np.clip(p, 0, 1000)
            print(f"Parent loaded: n={len(parent)}, sum={p.sum():.4f}")
            # Try the raw parent first
            score = try_seq(parent_raw, "parent_raw")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: FISTA on parent ===
    print("\n=== Phase 1: FISTA Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t_remaining = deadline - time.monotonic() - 600
        t1 = min(200, max(30, t_remaining))
        print(f"  FISTA on parent (n={n_p}), t={t1:.0f}s")

        for tau in [0.01, 0.005, 0.002]:
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(rem / 3.0, 80)
            a_opt, obj = fista_optimize(parent, tau, time_limit=t_alloc)
            try_seq(a_opt, f"fista_tau{tau}")
            if best_seq is not None and len(best_seq) == n_p:
                parent = best_seq / best_seq.sum()

    # === Phase 2: Coordinate Descent Polishing (main workhorse) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    if parent is not None:
        n_p = len(parent)
        t_remaining = deadline - time.monotonic() - 300
        t2 = min(600, max(60, t_remaining))
        print(f"  Coord descent (n={n_p}), t={t2:.0f}s")
        a_opt, obj = coord_descent_fast(parent, time_limit=t2)
        try_seq(a_opt, "coord1")

        # Second round from improved point
        if best_seq is not None and len(best_seq) == n_p:
            parent = best_seq / best_seq.sum()
            rem = deadline - time.monotonic() - 200
            if rem > 30:
                t_alloc = min(rem, 300)
                print(f"  Coord descent round 2, t={t_alloc:.0f}s")
                a_opt2, obj2 = coord_descent_fast(parent, time_limit=t_alloc)
                try_seq(a_opt2, "coord2")

    # === Phase 3: FISTA + Coord Descent iteration ===
    print("\n=== Phase 3: Iterative Refinement ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for round_i in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break

            # FISTA step
            t_fw = min(rem * 0.3, 60)
            if t_fw > 15:
                tau = 0.003 * (0.5 ** round_i)
                a_opt, obj = fista_optimize(best_seq / best_seq.sum(), tau, time_limit=t_fw)
                try_seq(a_opt, f"iter_fw_{round_i}")

            # Adam step
            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_adam = min(rem * 0.2, 40)
                if t_adam > 10:
                    a_opt, obj = adam_optimize(best_seq / best_seq.sum(), time_limit=t_adam,
                                               tau=0.005, lr=0.003)
                    try_seq(a_opt, f"iter_adam_{round_i}")

            # Coord descent
            rem = deadline - time.monotonic() - 60
            if rem > 20:
                t_cd = min(rem * 0.5, 120)
                if t_cd > 15:
                    a_opt, obj = coord_descent_fast(best_seq / best_seq.sum(), time_limit=t_cd)
                    try_seq(a_opt, f"iter_cd_{round_i}")

    # === Phase 4: Explore other lengths ===
    print("\n=== Phase 4: Length Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        other_lengths = [256, 512, 640, 768]
        other_lengths = [l for l in other_lengths if l != n_best]

        for n_try in other_lengths:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break

            # Create initial guess by interpolation
            x_new = np.arange(n_try, dtype=np.float64)
            f_init = np.interp(x_new, np.linspace(0, 1, n_best), best_seq / best_seq.sum())
            f_init = np.maximum(f_init, 0)
            f_init = f_init / f_init.sum()

            t_alloc = min(rem * 0.4, 120)
            if t_alloc < 20:
                continue

            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            # Quick FISTA
            t_fw = t_alloc * 0.4
            a_opt, obj = fista_optimize(f_init, 0.005, time_limit=t_fw)
            try_seq(a_opt, f"explore_fw_n{n_try}")

            # Coord descent
            rem = deadline - time.monotonic() - 60
            if rem > 20:
                t_cd = min(rem * 0.5, t_alloc * 0.5)
                a_opt, obj = coord_descent_fast(a_opt, time_limit=t_cd)
                try_seq(a_opt, f"explore_cd_n{n_try}")

    # === Phase 5: Final intensive polish ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"  Current best: {best_score:.8f} (n={n_best})")

        # Multiple coord descent rounds with different random restarts
        for r in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break

            if r == 0:
                f_init = best_seq / best_seq.sum()
            else:
                noise = rng.standard_normal(n_best) * 0.002 * (1.0 + r * 0.5)
                f_init = project_simplex(best_seq / best_seq.sum() + noise)

            t_alloc = min(rem, 80)
            a_opt, obj = coord_descent_fast(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"polish_cd_{r}")

            # Quick FISTA after coord descent
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                a_opt, obj = fista_optimize(a_opt, 0.001, time_limit=min(rem, 20))
                try_seq(a_opt, f"polish_fw_{r}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
