Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with exact
1D optimization per coordinate, combined with smoothed subgradient polishing
and multi-start diverse initializations across multiple sequence lengths.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    s = a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on a log-sum-exp smoothed
objective (via softmax parameterization), combined with exact coordinate descent
polishing, smoothed subgradient, and multi-start diverse initializations.
Key innovation: closed-form gradient of the smoothed auto-convolution peak,
enabling fast L-BFGS iterations that outperform LP-based Frank-Wolfe.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2 * n * conv.max() / (s * s))


def smooth_objective_and_grad(x, tau, n):
    """Compute smoothed objective and gradient for L-BFGS."""
    a = softmax(x)
    conv = fast_convolve(a)
    cv = conv / tau
    cv_shifted = cv - cv.max()
    w = np.exp(cv_shifted)
    w /= w.sum()
    # Objective: 2*n*tau*logsumexp(conv/tau)
    obj = 2.0 * n * tau * (cv.max() + np.log(w.sum()))
    # Gradient: 4*n*a_i*(T_i - V)
    T = np.correlate(w, a, 'valid')
    V = np.dot(w, conv)
    grad = 4.0 * n * a * (T - V)
    return obj, grad


def lgbfs_optimize(a0, time_limit, tau_start=0.05, tau_end=0.0005, n_phases=12):
    """L-BFGS with log-sum-exp smoothing and tau annealing."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15).astype(np.float64)
    a0 = a0 / a0.sum()
    x_current = np.log(a0)
    
    best_a = a0.copy()
    best_obj = objective_value(a0)
    
    t0 = time.monotonic()
    taus = np.geomspace(tau_start, tau_end, n_phases)
    
    for phase, tau in enumerate(taus):
        if time.monotonic() - t0 > time_limit:
            break
        
        phase_time = (time_limit - (time.monotonic() - t0)) / (n_phases - phase)
        max_iter = max(50, int(phase_time * 20))
        
        def obj_fn(x):
            a = softmax(x)
            conv = fast_convolve(a)
            cv = conv / tau
            return 2.0 * n * tau * (cv.max() + np.log(np.sum(np.exp(cv - cv.max()))))
        
        def grad_fn(x):
            a = softmax(x)
            conv = fast_convolve(a)
            cv = conv / tau
            cv_s = cv - cv.max()
            w = np.exp(cv_s)
            w /= w.sum()
            T = np.correlate(w, a, 'valid')
            V = np.dot(w, conv)
            return 4.0 * n * a * (T - V)
        
        res = minimize(obj_fn, grad_fn, x_current, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-7, 'ftol': 1e-10})
        
        if res.x is not None:
            a_new = softmax(res.x)
            obj_val = objective_value(a_new)
            if obj_val < best_obj:
                best_obj = obj_val
                best_a = a_new.copy()
                print(f"    L-BFGS phase {phase}: tau={tau:.6f}, obj={obj_val:.10f}")
            x_current = res.x
    
    return best_a, best_obj


def coord_descent_polish(a0, time_limit, max_passes=50000):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    best_obj = objective_value(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    for it in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        pass_improved = False
        
        # Randomize coordinate order for better mixing
        order = rng.permutation(n)
        
        for j in order:
            aj = a[j]
            
            # Compute c, p, q for this coordinate
            # (a*a)_k = c_k + x*p_k + q_k*x^2 where x is new value of a_j
            c = conv.copy()
            p = np.zeros(2 * n - 1)
            q = np.zeros(2 * n - 1)
            
            k2j = 2 * j
            if k2j < 2 * n - 1:
                q[k2j] = 1.0
                c[k2j] -= aj * aj
            
            # p[k] = 2*a[k-j] for valid k-j, k != 2j
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            mask = ks != k2j
            valid_ks = ks[mask]
            if len(valid_ks) > 0:
                idx = valid_ks - j
                p[valid_ks] = 2.0 * a[idx]
                c[valid_ks] -= aj * p[valid_ks]
            
            # Now minimize f(x) = max_k(c_k + x*p_k + q_k*x^2) / (1-aj+x)^2 over x >= 0
            # Try several candidate values
            candidates = [0.0, aj, aj * 0.5, aj * 1.5, 1.0 / n, 2.0 / n, 5.0 / n]
            
            # Also try the optimal for the currently dominant lags
            num_at_aj = c + aj * p + q * aj * aj
            top_lags = np.argsort(num_at_aj)[-3:]
            for k in top_lags:
                if q[k] > 0:
                    # Optimal x = c_k / (1-aj)
                    if 1.0 - aj > 1e-10:
                        x_opt = c[k] / (1.0 - aj)
                        if x_opt > 0:
                            candidates.append(x_opt)
                else:
                    # Optimal x = (2*c_k - p_k*(1-aj)) / p_k
                    if p[k] > 1e-15:
                        x_opt = (2.0 * c[k] - p[k] * (1.0 - aj)) / p[k]
                        if x_opt > 0:
                            candidates.append(x_opt)
            
            candidates = np.array(candidates)
            candidates = candidates[(candidates >= 0) & (candidates < 1.0)]
            if len(candidates) == 0:
                candidates = np.array([aj])
            
            # Evaluate and pick best
            best_x = aj
            best_val = best_obj  # upper bound
            
            for x in candidates:
                denom = (1.0 - aj + x) ** 2
                if denom < 1e-30:
                    continue
                num = c + x * p + q * x * x
                val = num.max() / denom
                if val < best_val:
                    best_val = val
                    best_x = x
            
            if abs(best_x - aj) > 1e-14:
                a[j] = best_x
                pass_improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        cur_obj = objective_value(a)
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Add small perturbation to escape
            a = project_simplex(a + rng.standard_normal(n) * 0.001)
            no_improve = 0
    
    return a, best_obj


def subgradient_optimize(a0, time_limit, lr=0.002, momentum=0.92, top_k=30, temp=0.0001):
    """Smoothed subgradient descent with multi-peak targeting."""
    n = len(a0)
    f = np.maximum(a0, 0.0).astype(np.float64)
    f = f / f.sum()
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9995
        cur_temp *= 1.003
        
        if no_improve > 200:
            rng = np.random.default_rng((int(time.monotonic() * 1000) % 100000))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def maxsum_lp_step(f, M_expand=1.05, time_per_lp=2.0):
    """One Frank-Wolfe step using max-sum LP oracle."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max() * M_expand
    
    # Build constraint matrix
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    
    c = -np.ones(n)
    A_ub = B
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': time_per_lp})
    
    if res.success and res.x is not None:
        g = res.x
        g_sum = g.sum()
        if g_sum > 1e-15:
            g = g / g_sum
            return g, M
    return None, M


def multi_length_init(n_target, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n_target) / n_target)
    
    # If parent available, interpolate
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Various structured inits
    x = np.arange(n_target, dtype=np.float64)
    
    # Gaussian
    for sigma_frac in [0.2, 0.35, 0.5]:
        sigma = sigma_frac * n_target
        g = np.exp(-0.5 * ((x - 0.5 * (n_target - 1)) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n_target)) + 0.1
    inits.append(c / c.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n_target) * 0.3))
    inits.append(rng.dirichlet(np.ones(n_target) * 1.0))
    
    # U-shape
    edge = 1.0 / (0.3 * np.abs(x - 0.5 * (n_target - 1)) / n_target + 0.2)
    inits.append(edge / edge.sum())
    
    return inits
