    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-resolution initialization (coarse-to-fine)
2. Nesterov-accelerated mirror descent (multiplicative weights) with adaptive LSE
3. LP-based Frank-Wolfe for global structure
4. Fast pairwise local search
5. Multi-length exploration with zero-trimming

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE-smoothed objective: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = 2 * n - 1 + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_subgrad(f, k):
    """Subgradient at lag k: 2 * f[k-j] for j=0..n-1 (padded)."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    sg[idx] = 2.0 * f[k - idx]
    return sg


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, time_limit=20):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def nesterov_mirror_descent(f0, time_limit, beta_start=30.0, beta_end=5000.0,
                            eta_init=None, verbose=True):
    """
    Nesterov-accelerated mirror descent with LSE smoothing.
    Uses multiplicative weights (KL projection) on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if eta_init is None:
        eta_init = 2.0 / (beta_start * n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Nesterov state
    x = f.copy()  # main iterate
    y = f.copy()  # extrapolated point

    # Momentum parameters (Nesterov schedule)
    mu = 1.0  # will be updated as (sqrt(1+8*t)-1)/2

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation: y = x + (mu_old/(mu_old+1)) * (x - x_prev)
        # Simplified: use standard Nesterov with t-based coefficients
        t_curr = it
        mu_curr = (np.sqrt(1.0 + 8.0 * t_curr) - 1.0) / 2.0
        mu_prev = (np.sqrt(1.0 + 8.0 * (t_curr - 1)) - 1.0) / 2.0

        theta = (1.0 - mu_prev) / (1.0 + mu_curr) if mu_curr > 0 else 1.0
        y = theta * x + (1.0 - theta) * y

        # Ensure y is valid
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = x.copy()
            y_sum = 1.0
        elif abs(y_sum - 1.0) > 1e-10:
            y = y / y_sum

        # Compute gradient at y using LSE
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, y)

        # Mirror descent update: x_new = x * exp(-eta * grad), then normalize
        # Clip to avoid overflow
        grad_clipped = np.clip(grad, -100.0, 100.0)
        x_new = x * np.exp(-eta * grad_clipped)
        x_new = np.maximum(x_new, 0.0)
        x_sum = x_new.sum()
        if x_sum < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / x_sum

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = x_new.copy()
            x = x_new
            no_improve = 0
        else:
            # Revert and reduce step size
            no_improve += 1
            if no_improve >= 15:
                eta *= 0.85
            if no_improve > 200:
                # Restart from best
                x = best_f.copy()
                y = best_f.copy()
                eta = max(eta_init / beta * beta_start, 1e-6)
                no_improve = 0
                it = max(1, it // 3)  # Reset Nesterov schedule partially

        # Ramp beta
        if beta < beta_end:
            beta *= 1.004
            eta = 2.0 / (beta * n)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD it={it}, obj={best_obj:.12f}, beta={beta:.1f}, eta={eta:.8f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, beta_start=40.0, beta_end=8000.0,
                         lr_init=None, momentum=0.92, verbose=True):
    """
    Standard gradient descent with Euclidean projection and LSE smoothing.
    Includes Nesterov momentum and adaptive learning rate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 3.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, f)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break

        # Nesterov momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.93
            if no_improve > 150:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 1.0 / n)
                beta = min(beta * 1.8, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.8f}, elapsed={elapsed:.0f}s")
        it += 1

    return best_f, best_obj

            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:

                    hi = mid

            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:

                best_f = f.copy()
                no_improve = 0

                no_improve += 1

        if verbose and it % 5 == 0:

def pairwise_local_search(f0, time_limit, verbose=True):
    """
    Fast local search: repeatedly transfer mass between the worst and best
    gradient positions. Very fast (O(n) per step).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    step = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)
        # Find the max lag (active constraint)
        k_max = int(np.argmax(conv))

        # Subgradient at k_max
        sg = compute_subgrad(f, k_max)

        # Best move: decrease f at argmax(sg), increase f at argmin(sg)
        i_out = int(np.argmax(sg))
        i_in = int(np.argmin(sg))

        if i_out == i_in:
            no_improve += 50
            break

        # Try several step sizes
        best_step = 0.0
        best_val = conv.max()
        for trial_step in [step * 0.1, step * 0.3, step * 0.5, step, step * 2.0, step * 5.0]:
            delta = min(trial_step, f[i_out])
            if delta < 1e-12:
                continue
            f_trial = f.copy()
            f_trial[i_out] -= delta
            f_trial[i_in] += delta
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_step = delta

        if best_step > 0:
            f[i_out] -= best_step
            f[i_in] += best_step
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-15:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
                step = min(step * 1.01, 0.01)
            else:
                f = best_f.copy()
                no_improve += 1
                step *= 0.8
        else:
            no_improve += 1
            step *= 0.9

        if no_improve > 500:
            # Try random pairs
            rng = np.random.default_rng(it * 13)
            for _ in range(100):
                i1, i2 = rng.choice(n, size=2, replace=False)
                delta = min(0.001, f[i1], f[i2])
                if delta < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[i1] -= delta
                f_trial[i2] += delta
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_f = f_trial.copy()
                    f = f_trial
                    no_improve = 0
                    break
            else:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS it={it}, obj={best_obj:.12f}, step={step:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def trim_zeros(f, threshold=1e-10):
    """Remove leading and trailing zeros, and truly zero interior elements."""
    f = np.array(f, dtype=np.float64)
    # Trim leading zeros
    start = 0
    while start < len(f) - 1 and f[start] < threshold:
        start += 1
    # Trim trailing zeros
    end = len(f) - 1
    while end > start and f[end] < threshold:
        end -= 1
    f = f[start:end + 1]
    # Zero out interior small values
    f[f < threshold] = 0.0
    return f


def multi_resolution_init(n, base_pattern, coarse_n=16):
    """Create a multi-resolution initialization."""
    # Optimize at coarse level
    coarse = base_pattern[:coarse_n].copy()
    if coarse.sum() < 1e-15:
        coarse = np.ones(coarse_n) / coarse_n
    else:
        coarse = coarse / coarse.sum()

    # Up-sample to full resolution using interpolation
    x_old = np.linspace(0, 1, coarse_n)
    x_new = np.linspace(0, 1, n)
    full = np.interp(x_new, x_old, coarse)
    full = np.maximum(full, 0.0)
    if full.sum() > 1e-15:
        full = full / full.sum()
    return full

    t_alloc = min(150.0, deadline - time.monotonic() - 250.0)
    if t_alloc > 15:
        print(f"  Pairwise search t={t_alloc:.0f}s")
        a_opt, obj = pairwise_local_search(current_best, time_limit=t_alloc, verbose=False)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    print(f"  Phase 4 best: {best_tracker[0]:.8f}")

    # ============ Phase 5: Multi-length exploration ============
    print("\n=== Phase 5: Multi-length ===")
    n_best = len(current_best)

    # Try nearby lengths
    alt_lengths = []
    for dl in [-64, -32, 32, 64, 96, -96, 128, -128]:
        nl = n_best + dl
        if 128 <= nl <= 800 and nl not in alt_lengths:
            alt_lengths.append(nl)

    # Also try some specific lengths
    for nl in [256, 320, 384, 512, 576, 640]:
        if nl not in alt_lengths and 128 <= nl <= 800:
            alt_lengths.append(nl)

    t_per_alt = 60.0
    max_alts = min(len(alt_lengths), max(1, int((deadline - time.monotonic() - 100.0) / t_per_alt)))
    alt_lengths = alt_lengths[:max_alts]

    for n_alt in alt_lengths:
        t_remaining = deadline - time.monotonic() - 60.0
        if t_remaining < 40:
            break
        t_alloc = min(t_per_alt, t_remaining / max(1, len(alt_lengths) - alt_lengths.index(n_alt)))
        if t_alloc < 25:
            break

        print(f"  Trying n={n_alt}, t={t_alloc:.0f}s")

        # Interpolate current best to new length
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick optimization
        t1 = t_alloc * 0.6
        t2 = t_alloc * 0.4
        a_opt, _ = lse_gradient_descent(f_interp, time_limit=t1,
                                          beta_start=30.0, beta_end=4000.0,
                                          momentum=0.93, verbose=False)
        try_sequence(a_opt, best_tracker)

        curr = best_tracker[1] if best_tracker[1] is not None else f_interp
        if len(curr) == n_alt:
            a_opt, _ = lse_gradient_descent(curr, time_limit=t2,
                                             beta_start=100.0, beta_end=8000.0,
                                             momentum=0.95, verbose=False)
            try_sequence(a_opt, best_tracker)

    print(f"  Phase 5 best: {best_tracker[0]:.8f}")

    # ============ Phase 6: Final polish and trim ============
    print("\n=== Phase 6: Final Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # Final LSE with very high beta
    t_alloc = min(100.0, deadline - time.monotonic() - 30.0)
    if t_alloc > 15:
        a_opt, _ = lse_gradient_descent(current_best, time_limit=t_alloc,
                                          beta_start=200.0, beta_end=15000.0,
                                          momentum=0.96, verbose=False)
        try_sequence(a_opt, best_tracker)

    # Trim zeros and re-evaluate
    if best_tracker[1] is not None:
        trimmed = trim_zeros(best_tracker[1], threshold=1e-12)
        if len(trimmed) > 10:
            try_sequence(trimmed, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
