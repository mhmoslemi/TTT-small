import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_objective(a, T):
    """Log-sum-exp approximation of max(conv(a,a))"""
    c = fast_convolve(a)
    c_scaled = c / T
    c_max = c_scaled.max()
    log_sum_exp = c_max + np.log(np.sum(np.exp(c_scaled - c_max)))
    return T * log_sum_exp

def smooth_gradient(a, T):

    c = fast_convolve(a)
    c_scaled = c / T
    c_max = c_scaled.max()
    exps = np.exp(c_scaled - c_max)

    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_conv(a, T):
    """Smooth approximation of max(conv(a,a)) via log-sum-exp."""
    c = fast_convolve(a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, c
    scaled = (c - c_max) / T
    exps = np.exp(scaled)
    val = c_max + T * np.log(np.sum(exps))
    return val, c


def grads_from_weights(w, a):
    """Compute grad_a = 2 * cross_correlation(w, a)[:n] via FFT."""
    n = len(a)
    L = 1 << (2 * n - 1 + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(W * np.conj(A), L)
    return 2.0 * corr[:n]


def softmax_obj_grad(z, T):
    """Objective and gradient for softmax-parameterized smooth problem."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    n = len(a)
    val, c = smooth_max_conv(a, T)
    if val < 1e-30:
        return 0.0, np.zeros(n)
    c_max = c.max()
    scaled = (c - c_max) / T
    exps = np.exp(scaled)
    w = exps / exps.sum()
    grad_a = grads_from_weights(w, a)
    s = np.dot(grad_a, a)
    grad_z = a * (grad_a - s)
    obj = 2.0 * n * val
    grad = 2.0 * n * grad_z
    return obj, grad


def adam_smooth(z0, T_schedule, n_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                time_limit=100.0, verbose=False):
    """Adam optimizer with smooth log-sum-exp objective and temperature schedule."""
    z = z0.copy().astype(np.float64)
    n = len(z)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = 2.0 * n
    best_z = z.copy()
    
    t0 = time.monotonic()
    
    while t < n_steps and (time.monotonic() - t0) < time_limit:
        T = T_schedule[min(t, len(T_schedule) - 1)]
        obj, grad = softmax_obj_grad(z, T)
        
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_z = z.copy()
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 1000 == 0:
            print(f"    Adam step {t}, T={T:.6f}, obj={obj:.8f}, best={best_obj:.8f}")
    
    return best_z, best_obj


def lwfgs_smooth(z0, T_schedule, time_limit=100.0, maxiter=200, verbose=False):
    """L-BFGS-B on the smooth objective with temperature annealing."""
    z = z0.copy().astype(np.float64)
    n = len(z)
    best_obj = 2.0 * n
    best_z = z.copy()
    t0 = time.monotonic()
    
    for i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        state = {'T': T}
        
        def func_wrapper(z_in, state=state):
            obj, _ = softmax_obj_grad(z_in, state['T'])
            return obj
        
        def grad_wrapper(z_in, state=state):
            _, grad = softmax_obj_grad(z_in, state['T'])
            return grad
        
        result = minimize(func_wrapper, z, jac=grad_wrapper, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'gtol': 1e-6, 'ftol': 1e-10})
        
        z = result.x
        obj, _ = softmax_obj_grad(z, T)
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_z = z.copy()
        
        if verbose:
            print(f"    L-BFGS T={T:.6f}: obj={obj:.8f}, best={best_obj:.8f}, iters={result.nit if hasattr(result, 'nit') else '?'}")
    
    return best_z, best_obj


def block_cd_smooth(a0, block_size=16, n_passes=10, T=0.001, time_limit=200.0, verbose=False):
    """Block coordinate descent with smooth objective.
    For each block of consecutive coordinates, optimize jointly using L-BFGS."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for pas in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            bsz = end - start
            
            # Define sub-problem: optimize block coordinates
            def block_obj(block_vals, a_full=a, st=start, en=end, T=T):
                af = a_full.copy()
                af[st:en] = np.maximum(block_vals, 0.0)
                sf = af.sum()
                if sf < 1e-15:
                    return 1e10
                af = af / sf
                val, _ = smooth_max_conv(af, T)
                return 2.0 * len(af) * val
            
            def block_grad(block_vals, a_full=a, st=start, en=end, T=T):
                af = a_full.copy()
                af[st:en] = np.maximum(block_vals, 0.0)
                sf = af.sum()
                if sf < 1e-15:
                    return np.zeros(bsz)
                af = af / sf
                val, c = smooth_max_conv(af, T)
                c_max = c.max()
                if c_max < 1e-30:
                    return np.zeros(bsz)
                scaled = (c - c_max) / T
                exps = np.exp(scaled)
                w = exps / exps.sum()
                grad_a = grads_from_weights(w, af)
                # Only return gradient for the block
                return 2.0 * len(af) * grad_a[st:en]
            
            x0 = a[start:end]
            res = minimize(block_obj, x0, jac=block_grad, method='L-BFGS-B',
                           options={'maxiter': 30, 'gtol': 1e-5, 'ftol': 1e-8})
            
            new_block = np.maximum(res.x, 0.0)
            new_a = a.copy()
            new_a[start:end] = new_block
            new_a = new_a / new_a.sum()
            
            new_score = 2.0 * n * fast_convolve(new_a).max()
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                a = new_a
    
    return best_a, best_score


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle (from previous best, kept for combination)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 5})
        except Exception:
            break
        
        if not res.success or res.x is None:
            if no_improve > 20:
                break
            no_improve += 1
            continue
        
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def subgradient_polish(a0, time_limit, lr=0.002, mom=0.93, top_k=20, verbose=False):
    """Subgradient method with momentum targeting top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        tw = weights.sum()
        if tw > 0:
            weights /= tw
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            valid = (other >= 0) & (other < n)
            sg[idx[valid]] += weights[i] * 2.0 * a[other[valid]]
        
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        new_score = 2.0 * n * fast_convolve(a_new).max()
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = a_new
        cur_lr *= 0.9998
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0003)
            no_improve = 0
    
    return best_a, best_score


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    x = np.arange(n, dtype=np.float64)
    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(cw / cw.sum())
    # Gaussian
    g = np.exp(-((x - n / 2) ** 2) / (2 * (n / 4) ** 2)) + 0.01
    inits.append(g / g.sum())
    # Edge emphasis
    ew = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(ew / ew.sum())
    # Multi-frequency
    mf = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x / n) + 0.2 * np.cos(2 * np.pi * 7 * x / n)
    mf = np.maximum(mf, 0.01)
    inits.append(mf / mf.sum())
    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.5, size=5)
    sm = 1.0
    for fi, ph, am in zip(freqs, phases, amps):
        sm += am * np.cos(2 * np.pi * fi * x / n + ph)
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Explore multiple n with diverse inits using smooth L-BFGS
    Phase 2: Adam optimizer with temperature annealing
    Phase 3: Block coordinate descent
    Phase 4: FW-LP and subgradient polish
    Phase 5: Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    # === Phase 1: Exploration with smooth L-BFGS (~400s) ===
    print("\n=== Phase 1: Exploration (smooth L-BFGS) ===")
    target_ns = [384, 512, 768]
    
    for n_target in target_ns:
        rem = deadline - time.monotonic() - 900
        if rem < 20:
            break
        t_alloc = min(rem, 150)
        print(f"  n={n_target}, t={t_alloc:.0f}s")
        
        inits = generate_diverse_inits(n_target, parent, rng)
        
        for ii, init in enumerate(inits[:5]):
            rem = deadline - time.monotonic() - 850
            if rem < 10:
                break
            z0 = np.log(init + 1e-20) - np.log(init + 1e-20).max()
            # Temperature schedule: start high, end low
            T_sched = [0.01, 0.005, 0.002, 0.001, 0.0005]
            t_per = min(rem, t_alloc / len(inits))
            z_best, obj_best = lwfgs_smooth(z0, T_sched, time_limit=t_per, maxiter=100)
            a_best = np.exp(z_best - z_best.max())
            a_best = a_best / a_best.sum()
            try_seq(a_best, f"lbfgs_n{n_target}_i{ii}")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Adam optimization (~400s) ===
    print("\n=== Phase 2: Adam Optimizer ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        z0 = np.log(best_seq + 1e-20) - np.log(best_seq + 1e-20).max()
        
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            # Multiple Adam runs with different hyperparameters
            adam_configs = [
                {'lr': 0.005, 'n_steps': 30000, 'T_start': 0.005, 'T_end': 0.0001},
                {'lr': 0.01, 'n_steps': 20000, 'T_start': 0.003, 'T_end': 0.00005},
                {'lr': 0.003, 'n_steps': 50000, 'T_start': 0.01, 'T_end': 0.0002},
            ]
            t_per = max(10, rem / len(adam_configs))
            
            for ci, cfg in enumerate(adam_configs):
                rem = deadline - time.monotonic() - 400
                if rem < 10:
                    break
                # Create temperature schedule
                n_steps = cfg['n_steps']
                T_sched = np.geomspace(cfg['T_start'], cfg['T_end'], n_steps)
                t_alloc = min(t_per, rem)
                print(f"  Adam config {ci} (lr={cfg['lr']}), t={t_alloc:.0f}s")
                z_best, obj_best = adam_smooth(z0, T_sched, n_steps=n_steps,
                                               lr=cfg['lr'], time_limit=t_alloc)
                a_best = np.exp(z_best - z_best.max())
                a_best = a_best / a_best.sum()
                try_seq(a_best, f"adam_{ci}")
                
                # Update z0 from best found so far
                if best_seq is not None:
                    z0 = np.log(best_seq + 1e-20) - np.log(best_seq + 1e-20).max()
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Block Coordinate Descent (~300s) ===
    print("\n=== Phase 3: Block Coordinate Descent ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            for bs in [16, 32]:
                rem = deadline - time.monotonic() - 200
                if rem < 10:
                    break
                t_alloc = min(rem, 120)
                print(f"  Block CD (bs={bs}), t={t_alloc:.0f}s")
                a_opt, sc = block_cd_smooth(best_seq, block_size=bs, n_passes=5,
                                            T=0.0005, time_limit=t_alloc)
                try_seq(a_opt, f"bcd_bs{bs}")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: FW-LP + Subgradient (~250s) ===
    print("\n=== Phase 4: FW-LP + Subgradient ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_fw = min(rem // 2, 100)
            print(f"  FW-LP, t={t_fw:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_fw)
            try_seq(a_opt, "fw_lp")
        
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_sg = min(rem, 100)
            print(f"  Subgradient, t={t_sg:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_sg, lr=0.002,
                                          mom=0.93, top_k=20)
            try_seq(a_opt, "subgrad")
    
    print(f"  Phase 4 best: {best_score:.8f}")
    
    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Low-temperature Adam
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            z0 = np.log(best_seq + 1e-20) - np.log(best_seq + 1e-20).max()
            T_sched = np.geomspace(0.0003, 0.00001, 20000)
            t_alloc = min(rem, 60)
            print(f"  Low-T Adam, t={t_alloc:.0f}s")
            z_best, _ = adam_smooth(z0, T_sched, n_steps=20000, lr=0.002,
                                    time_limit=t_alloc)
            a_best = np.exp(z_best - z_best.max())
            a_best = a_best / a_best.sum()
            try_seq(a_best, "lowT_adam")
        
        # Fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 40)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_alloc, lr=0.0005,
                                          mom=0.95, top_k=10)
            try_seq(a_opt, "fine_sg")
        
        # Perturbation + re-optimize
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            for _ in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                a_pert = project_simplex(best_seq + noise)
                z0 = np.log(a_pert + 1e-20) - np.log(a_pert + 1e-20).max()
                T_sched = np.geomspace(0.001, 0.0001, 10000)
                t_alloc = min(rem, 15)
                z_best, _ = adam_smooth(z0, T_sched, n_steps=10000, lr=0.003,
                                        time_limit=t_alloc)
                a_best = np.exp(z_best - z_best.max())
                a_best = a_best / a_best.sum()
                try_seq(a_best, "perturb_adam")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(384) / 384
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
