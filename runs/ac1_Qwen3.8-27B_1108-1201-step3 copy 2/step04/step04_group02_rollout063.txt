from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f*g)[k]"""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je+1] = f[js:je+1]
    return A

def solve_lp_fixed_f(f):
    """Solve: minimize M s.t. (f@g)[k] <= M, sum(g)=1, g>=0"""
    n = len(f)
    A = build_conv_matrix(f)
    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.ones(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:

    return None, None

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alternating_lp(f0, time_limit, ...):
    f = f0.copy() / f0.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP direction finding
with Adam smoothed gradient polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f):
    """Solve LP: minimize M s.t. (f@g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(f)
    # Build constraint matrix: A[k, j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]
    
    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.ones(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(f0, time_limit, alpha=0.5, verbose=False):
    """Alternating LP: minimize max(f*g) by alternating LPs for f and g."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    g = f.copy()
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Solve for g given f
        g_new, M_g = solve_lp_direction(f)
        if g_new is None:
            break
        g_new = np.maximum(g_new, 0)
        gs = g_new.sum()
        if gs < 1e-15:
            break
        g_new = g_new / gs
        
        # Solve for f given g
        f_new, M_f = solve_lp_direction(g_new)
        if f_new is None:
            break
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            break
        f_new = f_new / fs
        
        # Update (smoothed)
        g = (1 - alpha) * g + alpha * g_new
        g = g / g.sum()
        f = (1 - alpha) * f + alpha * f_new
        f = f / f.sum()
        
        # Evaluate on the diagonal (f*f)
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 100 == 0:
            print(f"    ALT-LP iter {it}: obj={obj:.8f}, M_f={M_f:.6f}, M_g={M_g:.6f}")
        
        if no_improve > 200:
            # Perturb to escape
            rng = np.random.default_rng(it)
            f = project_simplex(f + rng.standard_normal(n) * 0.01)
            g = f.copy()
            no_improve = 0
    
    return best_f, best_obj

    """Adam optimizer on smoothed max of auto-convolution."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    cur_temp = temp_init
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Compute smoothed gradient
        if top_k is not None and top_k < 2 * n - 1:
            # Only use top-k peaks for efficiency
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            conv_peaks = conv[peak_idx].copy()
            scaled = conv_peaks / cur_temp
            scaled -= scaled.max()
            w_peaks = np.exp(scaled)
            w_peaks /= w_peaks.sum()
            
            grad = np.zeros(n)
            for idx in range(len(peak_idx)):
                if w_peaks[idx] < 1e-15:
                    continue
                k = int(peak_idx[idx])
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                grad[i_start:i_end + 1] += 2.0 * w_peaks[idx] * f[k - np.arange(i_start, i_end + 1)]
        else:
            grad = compute_smoothed_grad(f, conv, cur_temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + 1e-8))
        
        conv = fast_convolve(f_new)
        obj = conv.max()

            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            f_new = f  # don't update if no improvement
        
        f = f_new
        cur_temp = max(temp_min, cur_temp * 0.999)
        
        if no_improve > 500:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.5)
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP direction finding
combined with Adam-optimized smoothed gradient descent and FFT-accelerated gradients.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(f, conv, top_k=128, temp=0.01):
    """Compute gradient of smoothed max of auto-convolution using FFT."""
    n = len(f)
    conv_len = 2 * n - 1
    
    if top_k is not None and top_k < conv_len:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx]
        scaled = pv / temp
        scaled -= scaled.max()
        sw = np.exp(scaled)
        sw /= sw.sum()
        w = np.zeros(conv_len)
        w[peak_idx] = sw
    else:
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    
    # corr[i] = sum_j w[j+i] * f[j] for i=0..n-1
    N = 1 << (n + conv_len - 1 - 1).bit_length()
    F = np.fft.fft(f, N)
    W = np.fft.fft(w, N)
    corr = np.fft.ifft(W * np.conj(F)).real[:n]
    
    return 2.0 * corr


def solve_lp_direction(f):
    """Solve LP: min M s.t. (f@g)[k] <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.ones(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(f0, time_limit, alpha=0.7, verbose=False):
    """Alternating LP optimization."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    g = f.copy()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_alpha = alpha
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        g_new, M_g = solve_lp_direction(f)
        if g_new is None:
            break
        
        f_new, M_f = solve_lp_direction(g_new)
        if f_new is None:
            break
        
        g = (1 - cur_alpha) * g + cur_alpha * g_new
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs
        
        f = (1 - cur_alpha) * f + cur_alpha * f_new
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-15:
            break
        f /= fs
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay alpha
        cur_alpha = max(0.3, alpha * 0.999)
        
        if verbose and it % 50 == 0:
            print(f"    ALT it={it} obj={obj:.8f} M_f={M_f:.6f}")
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            g = f.copy()
            cur_alpha = alpha
            no_improve = 0
    
    return best_f, best_obj


def adam_optimize(f0, time_limit, lr=0.05, temp_init=0.05, temp_decay=0.9995,
                  top_k=128, beta1=0.9, beta2=0.999, verbose=False):
    """Adam optimizer on smoothed max of auto-convolution."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    cur_temp = temp_init
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        grad = compute_grad_fft(f, conv, top_k=top_k, temp=cur_temp)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_candidate = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + 1e-8))
        
        conv_cand = fast_convolve(f_candidate)
        obj_cand = conv_cand.max()
        
        if obj_cand < best_obj - 1e-13:
            best_obj = obj_cand
            best_f = f_candidate.copy()
            no_improve = 0
            f = f_candidate
            conv = conv_cand
        else:
            no_improve += 1
        
        cur_temp = max(0.0001, cur_temp * temp_decay)
        
        if verbose and t % 200 == 0:
            print(f"    ADAM t={t} obj={best_obj:.8f} temp={cur_temp:.6f}")
        
        if no_improve > 1000:
            rng = np.random.default_rng(t * 3 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            cur_temp = temp_init * 0.5
            no_improve = 0
            lr = max(0.005, lr * 0.7)
    
    return best_f, best_obj


def coord_descent_polish(f0, time_limit, n_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 3:
                    break
            
            # Try reducing f[i] and redistributing to others
            old_val = f[i]
            if old_val < 1e-12:
                continue
            
            # Try several new values
            for frac in [0.5, 0.7, 0.85, 0.95, 0.3]:
                f_trial = f.copy()
                f_trial[i] = old_val * frac
                # Redistribute the difference proportionally to others
                diff = old_val * (1 - frac)
                others = np.ones(n)
                others[i] = 0
                if others.sum() > 0:
                    f_trial += diff * others / others.sum()
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                conv_t = fast_convolve(f_trial)
                obj_t = conv_t.max()
                
                if obj_t < best_obj - 1e-14:
                    best_obj = obj_t
                    best_f = f_trial.copy()
                    f = f_trial
                    conv = conv_t
                    break
        
        if best_obj < 1.503:
            break
    
    return best_f, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(rc / rc.sum())
    
    # Exponential from center
    ec = np.exp(-((x - n/2) ** 2) / (2 * (n/4) ** 2))
    inits.append(ec / ec.sum())
    
    # Asymmetric exponential
    ea = np.exp(-x / (n/3))
    inits.append(ea / ea.sum())
    
    # Power law
    pw = (x + 1) ** (-0.3)
    inits.append(pw / pw.sum())
    
    # Sine window
    sw = np.sin(np.pi * x / n) ** 2 + 0.05
    inits.append(sw / sw.sum())
    
    # Hamming-like
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1)) + 0.01
    inits.append(ham / ham.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random with structure
    base = np.ones(n) / n
    for _ in range(3):
        center = rng.integers(0, n)
        width = rng.integers(n//10, n//3)
        amplitude = rng.uniform(0.5, 2.0)
        base += amplitude * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    inits.append(np.maximum(base, 0) / np.maximum(base, 0).sum())
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend
            for blend in [0.3, 0.5, 0.7]:
                b = blend * (px / ps) + (1 - blend) * (np.ones(n) / n)
                inits.append(b / b.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (exact subproblem solves for direction finding)
2. Adam optimizer on log-sum-exp smoothed max with FFT-accelerated gradients
3. Diverse structured initializations across multiple sequence lengths
4. Coordinate descent and micro-perturbation polishing
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(f, conv, top_k=128, temp=0.01):
    """Compute gradient of smoothed max of auto-convolution using FFT cross-correlation."""
    n = len(f)
    conv_len = 2 * n - 1

    if top_k is not None and top_k < conv_len:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peak_idx]
        scaled = pv / max(temp, 1e-12)
        scaled -= scaled.max()
        sw = np.exp(scaled)
        sw_sum = sw.sum()
        if sw_sum < 1e-30:
            sw = np.zeros(len(peak_idx))
            sw[np.argmax(pv)] = 1.0
        else:
            sw /= sw_sum
        w = np.zeros(conv_len)
        w[peak_idx] = sw
    else:
        scaled = conv / max(temp, 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

    # corr[i] = sum_j w[j+i] * f[j] for i=0..n-1 via FFT
    N = 1 << (n + conv_len - 1 - 1).bit_length()
    F = np.fft.fft(f, N)
    W = np.fft.fft(w, N)
    corr = np.fft.ifft(W * np.conj(F)).real[:n]

    return 2.0 * corr


def solve_lp_direction(f):
    """Solve LP: minimize M s.t. (f@g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(f)
    # Build constraint matrix A where (A@g)[k] = (f*g)[k]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]

    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.ones(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def alternating_lp(f0, time_limit, alpha_init=0.8, alpha_min=0.3, verbose=False):
    """Alternating LP: alternate solving exact LPs for f and g."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    g = f.copy()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_alpha = alpha_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Solve for best g given current f
        g_new, M_g = solve_lp_direction(f)
        if g_new is None:
            break

        # Solve for best f given new g
        f_new, M_f = solve_lp_direction(g_new)
        if f_new is None:
            break

        # Smoothed update
        g = (1 - cur_alpha) * g + cur_alpha * g_new
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs

        f = (1 - cur_alpha) * f + cur_alpha * f_new
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-15:
            break
        f /= fs

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_alpha = max(alpha_min, cur_alpha * 0.998)

        if verbose and it % 100 == 0:
            print(f"    ALT-LP it={it} obj={obj:.8f} M_f={M_f:.6f} M_g={M_g:.6f}")

        if no_improve > 400:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            g = f.copy()
            cur_alpha = alpha_init
            no_improve = 0

    return best_f, best_obj


def adam_optimize(f0, time_limit, lr=0.05, temp_init=0.05, temp_decay=0.9995,
                  temp_min=0.0001, top_k=128, beta1=0.9, beta2=0.999, verbose=False):
    """Adam optimizer on log-sum-exp smoothed max of auto-convolution."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    cur_temp = temp_init

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        grad = compute_grad_fft(f, conv, top_k=top_k, temp=cur_temp)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_candidate = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + 1e-8))

        conv_cand = fast_convolve(f_candidate)
        obj_cand = conv_cand.max()

        if obj_cand < best_obj - 1e-14:
            best_obj = obj_cand
            best_f = f_candidate.copy()
            no_improve = 0
            f = f_candidate
            conv = conv_cand
        else:
            no_improve += 1

        cur_temp = max(temp_min, cur_temp * temp_decay)

        if verbose and t % 500 == 0:
            print(f"    ADAM t={t} best={best_obj:.8f} temp={cur_temp:.6f} no_improve={no_improve}")

        if no_improve > 1500:
            rng = np.random.default_rng(t * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            cur_temp = max(temp_init * 0.5, temp_min * 10)
            no_improve = 0
            lr = max(0.005, lr * 0.6)

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, max_sweeps=100):
    """Coordinate descent: try adjusting individual coordinates."""
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 3:
                break

            old_val = f[i]
            if old_val < 1e-13:
                continue

            best_local_obj = conv.max()
            best_local_f = f.copy()

            for frac in [0.3, 0.5, 0.7, 0.85, 0.95]:
                f_trial = f.copy()
                f_trial[i] = old_val * frac
                diff = old_val * (1.0 - frac)
                # Redistribute to other coordinates proportionally
                if n > 1:
                    others_mask = np.ones(n)
                    others_mask[i] = 0.0
                    osum = f.sum() - f[i]
                    if osum > 1e-15:
                        f_trial += diff * (f / osum) * others_mask
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                conv_t = fast_convolve(f_trial)
                obj_t = conv_t.max()

                if obj_t < best_local_obj - 1e-14:
                    best_local_obj = obj_t
                    best_local_f = f_trial.copy()

            if best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                best_f = best_local_f.copy()
                f = best_local_f
                conv = fast_convolve(f)

        if best_obj < 1.502:
            break

    return best_f, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n)) + 0.05
    inits.append(rc / rc.sum())

    # Gaussian centered
    gc = np.exp(-((x - n / 2.0) ** 2) / (2.0 * (n / 4.0) ** 2))
    inits.append(gc / gc.sum())

    # Asymmetric exponential (decay from left)
    ea = np.exp(-x / (n / 3.0))
    inits.append(ea / ea.sum())

    # Asymmetric exponential (decay from right)
    eb = np.exp(-x[::-1] / (n / 3.0))
    inits.append(eb / eb.sum())

    # Power law
    pw = (x + 1.0) ** (-0.3)
    inits.append(pw / pw.sum())

    # Sine window
    sw = np.sin(np.pi * x / n) ** 2 + 0.02
    inits.append(sw / sw.sum())

    # Hamming-like
    ham = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / max(n - 1, 1)) + 0.01
    inits.append(ham / ham.sum())

    # Tukey (flat top with tapered edges)
    r = 0.3
    tk = np.ones(n)
    edge = int(r * n // 2)
    if edge > 0:
        for i in range(edge):
            t = (i + 0.5) / edge
            tk[i] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 2) / (2 - 2)))
            tk[n - 1 - i] = tk[i]
    inits.append(tk / tk.sum())

    # Doubly peaked
    dp = np.exp(-((x - n * 0.25) ** 2) / (2 * (n / 8) ** 2)) + \
         np.exp(-((x - n * 0.75) ** 2) / (2 * (n / 8) ** 2)) + 0.01
    inits.append(dp / dp.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Random Gaussian bumps
    base = np.ones(n) / n
    for _ in range(rng.integers(2, 5)):
        center = rng.uniform(0, n)
        width = rng.uniform(n / 12, n / 4)
        amplitude = rng.uniform(0.5, 3.0)
        base += amplitude * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    base = np.maximum(base, 0)
    inits.append(base / base.sum())

    # Frequency-domain inspired: IFFT of concentrated spectrum
    freq = np.zeros(n // 2 + 1)
    freq[rng.integers(0, 3)] = 1.0
    freq[rng.integers(0, 5)] = 0.5
    td = np.fft.irfft(freq * np.exp(1j * rng.uniform(0, 2 * np.pi, n // 2 + 1)), n)
    td = np.maximum(td, 0)
    if td.sum() > 1e-10:
        inits.append(td / td.sum())

    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            for blend in [0.3, 0.5, 0.7]:
                b = blend * (px / ps) + (1 - blend) * (np.ones(n) / n)
                inits.append(b / b.sum())
                b2 = blend * (px / ps) + (1 - blend) * (rc / rc.sum())
                inits.append(b2 / b2.sum())

    return inits

        n_inits = len(inits)

        rem = deadline - time.monotonic() - 200
        t_per = max(25, rem / max(1, n_inits * len(lengths)))
        t_per = min(t_per, 70)

        print(f"  n={n_len}, {n_inits} inits, ~{t_per:.0f}s each")

        best_for_n = None
        best_for_n_score = float('inf')

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break

            t_alloc = min(t_per, deadline - time.monotonic() - 100)
            if t_alloc < 15:
                break

            # Hybrid: LP then Adam
            t_lp = t_alloc * 0.4
            t_adam = t_alloc * 0.55
            f1, _ = alternating_lp(f_init, time_limit=t_lp, alpha_init=0.8)
            f2, _ = adam_optimize(f1, time_limit=t_adam,
                                   lr=0.05, temp_init=0.05, top_k=64)
            sc = try_seq(f2, f"n{n_len}_i{i}")

            if sc < best_for_n_score:
                best_for_n_score = sc
                best_for_n = f2.copy()

        # If we found something good for this length, do more optimization
        if best_for_n is not None and best_for_n_score < best_score + 0.01:
            rem = deadline - time.monotonic() - 120
            if rem > 30:
                f_deep, _ = adam_optimize(best_for_n, time_limit=min(rem, 80),
                                           lr=0.03, temp_init=0.03, top_k=48)
                try_seq(f_deep, f"n{n_len}_deep")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on global best ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Deep optimization ===")
        n_best = len(best_seq)

        # Multiple Adam configs
        configs = [
            {"lr": 0.02, "temp_init": 0.02, "top_k": 48, "temp_decay": 0.9997},
            {"lr": 0.05, "temp_init": 0.06, "top_k": 100, "temp_decay": 0.9995},
            {"lr": 0.01, "temp_init": 0.01, "top_k": 32, "temp_decay": 0.9998},
            {"lr": 0.08, "temp_init": 0.08, "top_k": 160, "temp_decay": 0.9993},
        ]

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (len(configs) - ci), 100)

            if ci == 0:
                f_init = best_seq.copy()
            elif ci == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
            elif ci == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)

            f_opt, _ = adam_optimize(f_init, time_limit=t_alloc, **cfg)
            try_seq(f_opt, f"deep_adam_{ci}")

        # LP deep
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 25:
                break
            t_alloc = min(rem // (3 - p), 70)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(0.85 * best_seq + 0.15 * best_seq[::-1])
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)

            f_opt, _ = alternating_lp(f_init, time_limit=t_alloc, alpha_init=0.9)
            try_seq(f_opt, f"deep_lp_{p}")

        # Adam after LP
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            f_opt, _ = adam_optimize(best_seq, time_limit=min(rem, 60),
                                      lr=0.015, temp_init=0.015, top_k=40, temp_decay=0.9999)
            try_seq(f_opt, "deep_adam_lp")

    # === Phase 4: Final polishing ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Final polishing ===")
        print(f"  Current best: {best_score:.8f}")
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            f_pol, _ = coord_descent_polish(best_seq, time_limit=min(rem, 80))
            try_seq(f_pol, "coord")

        # Very fine Adam
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            f_pol, _ = adam_optimize(best_seq, time_limit=min(rem, 40),
                                      lr=0.003, temp_init=0.003, top_k=24,
                                      temp_decay=0.99999)
            try_seq(f_pol, "fine_adam")

        # Micro perturbations with quick optimization
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            for mi in range(4):
                rem = deadline - time.monotonic() - 4
                if rem < 3:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                f_pol, _ = adam_optimize(f_pert, time_limit=min(rem, 5),
                                          lr=0.008, temp_init=0.008, top_k=24)
                try_seq(f_pol, f"micro_{mi}")

        # Quick LP polish
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            f_pol, _ = alternating_lp(best_seq, time_limit=min(rem, 20), alpha_init=0.95)
            try_seq(f_pol, "lp_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
