Minimizes 2*n*max(auto_conv(a))/(sum(a))^2 using:
1. L-BFGS optimization in log-parameter space with log-sum-exp smoothing
2. Projected subgradient refinement on exact max objective
3. Coordinate descent polish

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time


def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size)[:size])


def fast_cross_corr_pa(p, a):
    """Compute s_i = sum_k p[k] * a[k-i] for i=0,...,n-1.
    p has length 2n-1, a has length n."""
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    result = np.real(ifft(P * np.conj(A), N)[:n])
    return result


def smooth_obj_grad(w, beta):
    """Smoothed objective and gradient in log-parameter space.
    w: log-parameters (length n)
    Returns (F, grad) where F = (1/beta)*log(sum exp(beta*C_k)), grad = dF/dw"""
    a = w - w.max()
    ea = np.exp(a)
    a = ea / ea.sum()
    n = len(a)
    
    C = fast_autoconv(a)
    
    # Log-sum-exp
    maxC = C.max()
    exps = np.exp(beta * (C - maxC))
    log_sum = maxC + np.log(exps.sum()) / beta  # This is wrong
    # Correct: F = (1/beta) * log(sum_k exp(beta * C_k))
    #         = (1/beta) * log(exp(beta*maxC) * sum_k exp(beta*(C_k - maxC)))
    #         = (1/beta) * (beta*maxC + log(sum_k exp(beta*(C_k - maxC))))
    #         = maxC + (1/beta) * log(sum_k exp(beta*(C_k - maxC)))
    F = maxC + np.log(exps.sum()) / beta
    
    # Softmax weights
    p = exps / exps.sum()
    
    t = np.dot(p, C)
    s = fast_cross_corr_pa(p, a)
    grad = 2.0 * a * (s - t) / beta
    
    return F, grad


def lbfgs_smoothed(w0, beta_schedule, max_iter=200, time_limit=300):
    """Run L-BFGS with increasing beta."""
    w = w0.copy()
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        res = minimize(smooth_obj_grad, w, args=(beta,), method='L-BFGS-B',
                       jac=True, options={'maxiter': max_iter, 'gtol': 1e-8,
                                          'maxls': 50, 'maxfun': 500})
        if res.success or res.x is not None:
            w = res.x
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
    
    # Convert back to probability vector
    a = w - w.max()
    ea = np.exp(a)
    return ea / ea.sum()

    best_obj = C.max()
    best_a = a.copy()

    momentum = 0.9
    v = np.zeros(n)

        C = fast_autoconv(a)
        cur_obj = C.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top-k peaks
        if top_k < len(C):
            peaks = np.argpartition(C, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(C))
        
        pv = C[peaks]
        # Softmax weighting
        scale = 50.0
        p = np.exp(scale * (pv - pv.max()))
        p /= p.sum()
        
        # Subgradient: sum_k p_k * 2*a_{k-i} (simplified, ignoring the -t part for direction)
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if p[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * p[i] * a[k - j]
        
        # Update with momentum
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a = best_a.copy()
            continue
        a = a_new / ss
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.5
            v *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break
        elif no_improve == 0:
            lr = min(lr * 1.01, 0.05)
    
    return best_a


def coordinate_descent(a0, time_limit=30, passes=10):
    """Coordinate descent: optimize one coordinate at a time."""
    a = a0.copy()
    n = len(a)
    t0 = time.monotonic()
    
    rng = np.random.default_rng(42)
    order = np.arange(n)
    
    for _ in range(passes):
        if time.monotonic() - t0 > time_limit:
            break
        rng.shuffle(order)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            # Optimize a[i] while keeping sum = 1
            # a[i] + sum_{j≠i} a[j] = 1, so a[i] = 1 - sum_{j≠i} a[j]
            # Let s_i = sum_{j≠i} a[j] = 1 - a[i]
            # C_k = C_k^(-i) + a[i]*a[k-i] (where C_k^(-i) is C_k with a[i]=0)
            # We want to minimize max_k C_k over a[i] in [0, s_i_min]
            # where s_i_min is constrained by a[j] >= 0 for all j (already satisfied since we only change a[i])
            # Actually a[i] can range from 0 to 1 (since others sum to 1-a[i] and are non-negative)
            
            # Compute C_k^(-i)
            a_save = a[i]
            a[i] = 0
            C_base = fast_autoconv(a)
            a[i] = a_save
            
            # C_k = C_base[k] + a_i * a[k-i]
            # where a[k-i] is the original value (0 if out of range)
            a_ki = np.zeros(2*n-1)
            for k in range(2*n-1):
                if 0 <= k-i < n:
                    a_ki[k] = a[k-i]
            
            # C_k(x) = C_base[k] + x * a_ki[k] for x = a[i]
            # Minimize max_k (C_base[k] + x * a_ki[k]) for x in [0, 1]
            # This is a 1D problem: minimize f(x) = max_k (c_k + d_k * x)
            # where c_k = C_base[k], d_k = a_ki[k]
            
            # The function f(x) is the max of linear functions, so it's convex in x.
            # We can find the minimum by bisection on the subgradient.
            
            c = C_base
            d = a_ki
            
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2
                vals = c + mid * d
                # Subgradient at mid: weighted sum of active d_k
                max_val = vals.max()
                active = d[vals >= max_val - 1e-12]
                if len(active) == 0:
                    break
                subgrad = active.mean()
                if subgrad > 0:
                    hi = mid
                elif subgrad < 0:
                    lo = mid
                else:
                    break
            
            x_opt = (lo + hi) / 2
            a[i] = x_opt
    
    # Normalize
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss
    return a

    """Evaluate a sequence, returning inf on failure."""

    except Exception:
