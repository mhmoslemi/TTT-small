def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # Then c[i] = sum_j w[i+j] f[j]
    # Define g[m] = w[m] for m=0..2n-2
    # c[i] = sum_{j=0}^{n-1} g[i+j] f[j]
    # This is the "valid" cross-correlation of g and f, but only for lags 0 to n-1.
    
    # Actually, the simplest efficient approach:
    N = 1 << (3 * n - 1).bit_length()  # FFT size
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    # We want sum_j w[i+j] * f[j]
    # = sum_j w[i+j] f[j]
    # Let's define h = flip(w) where h[m] = w[2n-2-m]
    # Then sum_j w[i+j] f[j] = sum_j h[2n-2-i-j] f[j]
    # = (h * f)[2n-2-i] (convolution)
    # So c[i] = (h*f)[2n-2-i]
    
    h = w[::-1]  # h[m] = w[2n-2-m]
    H = np.fft.rfft(h, N)
    conv_hf = np.fft.irfft(H * F, N)
    # c[i] = conv_hf[2n-2-i]
    c = conv_hf[2*n-2 : 2*n-2-n:-1]  # c[i] = conv_hf[2n-2-i] for i=0..n-1
    
    grad = 4.0 * n * c
    return grad


def grad_optimize(f0, time_limit, beta_init=100.0, beta_max=10000.0, 
                  lr_init=0.01, beta_growth=1.02, seed=0):
    """
    Gradient descent with Adam on the simplex, using log-sum-exp smooth max.
    Uses projection onto simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    rng = np.random.default_rng(seed)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    beta = beta_init
    lr = lr_init
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        current_obj = conv.max()
        
        # Compute gradient
        grad = compute_gradient_smooth_max(f, conv, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adapt beta (increase to make smooth max closer to true max)
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_max)
        
        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
            if lr < 0.0001:
                # Restart
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = beta_init * 5
                no_improve = 0
                t = max(t // 2, 10)
        
        # Periodic check with exact objective
        if t % 200 == 0:
            exact_obj = fast_convolve(f).max()
            if exact_obj < best_obj - 1e-12:
                best_obj = exact_obj
                best_f = f.copy()
    
    return best_f, best_obj


def solve_qp_direction(f, conv, M, n_constraints=None):
    """
    Solve: min ||g - f||^2 s.t. (f*g)[k] <= M for top lags, g >= 0, sum(g) = 1.
    Uses scipy.optimize.minimize with SLSQP.
    """
    n = len(f)
    
    if n_constraints is None:
        n_constraints = min(len(conv), 50)
    
    # Get top lags
    peak_indices = np.argpartition(conv, -n_constraints)[-n_constraints:]
    peak_indices = np.sort(peak_indices)
    
    def objective(g):
        return np.sum((g - f) ** 2)
    
    def objective_grad(g):
        return 2.0 * (g - f)
    
    # Constraints: (f*g)[k] <= M for k in peak_indices
    # (f*g)[k] = sum_j f[j] * g[k-j] = sum_j f[k-j] * g[j] (reindexing)
    # Let's define: for each k, constraint is sum_j f[k-j] * g[j] <= M
    # where f[k-j] = 0 if k-j < 0 or k-j >= n
    
    constraints = []
    for k in peak_indices:
        # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        coeff = f[js:je + 1][::-1]  # f[k-je], f[k-je+1], ..., f[k-js]
        
        # Actually: (f*g)[k] = sum_j f[j] * g[k-j]
        # Let m = k-j, so j = k-m: sum_m f[k-m] * g[m]
        # m ranges from max(0, k-n+1) to min(n-1, k)
        # So coeff[m] = f[k-m] for m in that range
        # In terms of g: sum_{m=js}^{je} f[k-m] * g[m]
        
        def make_constraint(m):
            c = np.zeros(n)
            c[js:je+1] = f[k-m]  # Wait, I'm getting confused with variable names
            
            return c
        
        # Let me redo this more carefully
        # (f*g)[k] = sum_{j=0}^{n-1} f[j] * g[k-j] where g[m]=0 if m not in [0,n-1]
        # = sum_{j: 0<=k-j<=n-1} f[j] * g[k-j]
        # = sum_{j: k-n+1 <= j <= k, 0<=j<=n-1} f[j] * g[k-j]
        
        # As a function of g: = sum_{m: 0<=m<=n-1, k-n+1<=k-m<=k} f[k-m] * g[m]
        # = sum_{m: max(0,k-n+1) <= m <= min(n-1,k)} f[k-m] * g[m]
        
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        coeff = np.zeros(n)
        if lo <= hi:
            coeff[lo:hi+1] = f[k - lo : k - hi + 1][::-1]
            # Wait: coeff[m] = f[k-m] for m in [lo, hi]
            # So coeff[lo] = f[k-lo], coeff[lo+1] = f[k-lo-1], ...
            # coeff[m] = f[k-m]
            coeff[lo:hi+1] = [f[k-m] for m in range(lo, hi+1)]
        
        # Constraint: coeff @ g <= M
        # In scipy form: fun(g) = coeff @ g - M <= 0
        constraints.append({'type': 'ineq', 'fun': lambda g, coeff=coeff: M - coeff @ g,
                           'jac': lambda g, coeff=coeff: -coeff})
    
    bounds = [(0, None)] * n
    # Equality: sum(g) = 1
    constraints.append({'type': 'eq', 'fun': lambda g: g.sum() - 1,
                        'jac': lambda g: np.ones(n)})
    
    result = minimize(objective, f.copy(), jac=objective_grad,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-10})
    
    if result.success:
        return result.x
    return None


def qp_frank_wolfe(f0, time_limit, verbose=False):
    """Frank-Wolfe with QP subproblem."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Solve QP direction
        g = solve_qp_direction(f, conv, M * 0.95, n_constraints=min(30, 2*n-1))
        
        if g is None:
            no_improve += 5
            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            no_improve += 1
            continue
        g = g / gs
        
        # Check if it's different from current
        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    QP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=0):
    """
    Coordinate descent: iterate over coordinates, finding optimal value for each.
    For coordinate i, the objective as a function of f_i (with others fixed) involves
    the convolution values that depend on f_i.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    
    t0 = time.monotonic()
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Save old value
            old_val = f[i]
            
            # Compute the effect of f_i on the convolution
            # (f*f)[k] = base[k] + f_i * (f[k-i] + f[i-k] if i+k != 2i... )
            # More precisely: (f*f)[k] with f_i changed to v:
            # delta(k) = v * f[k-i] + f[k-i] * v - 2*old_val*f[k-i] + (v^2 - old_val^2 if k=2i)
            # Actually: (f_new*f_new)[k] - (f*f)[k]
            # = (f_new[k-i] - f[k-i]) * f_new[i] + f_new[k-i] * (f_new[i] - f[i]) ... 
            # This is getting complicated. Let me just do it numerically.
            
            # The convolution values that depend on f_i are those where k-i is in [0,n-1]
            # or i-k is in [0,n-1], i.e., k in [i, i+n-1] (intersected with [0, 2n-2])
            
            # For a given f_i = v, the new convolution at lag k is:
            # new_conv[k] = old_conv[k] - old_val * (f[k-i] + (f[i-k] if 2i==k else 0)) ... 
            # Actually let me just compute it directly.
            
            # (f*f)[k] = sum_j f[j] f[k-j]
            # When we change f[i] from old_val to v:
            # The change is: for each k, the terms involving f[i] are:
            #   f[i]*f[k-i] (when j=i) and f[k-i]*f[i] (when k-j=i, i.e., j=k-i)
            #   If i = k-i (i.e., k=2i), these are the same term, so f[i]^2 appears once.
            #   
            # So: new_conv[k] = old_conv[k] - old_val*f[k-i] - f[k-i]*old_val + v*f[k-i] + f[k-i]*v
            #   (with the understanding that f[m]=0 if m not in [0,n-1])
            #   If k=2i: new_conv[2i] = old_conv[2i] - old_val^2 + v^2
            #   If k!=2i: new_conv[k] = old_conv[k] + 2*(v - old_val)*f[k-i]
            
            # For the objective 2n * max(conv) / (sum)^2, after normalization sum=1:
            # We want to minimize max(new_conv)
            
            # Since changing f_i changes the sum, we need to renormalize.
            # Let's work with the unnormalized version for simplicity:
            # Actually, let's just evaluate the full objective for candidate values of v.
            
            # Try a few candidate values
            # The optimal v should be near old_val (for fine tuning)
            # Let's try: old_val * [0.5, 0.75, 0.9, 0.95, 1.0, 1.05, 1.1, 1.25, 1.5]
            candidates = old_val * np.array([0.0, 0.3, 0.5, 0.7, 0.85, 0.92, 0.97, 1.0, 1.03, 1.07, 1.15, 1.3, 1.6, 2.0])
            candidates = np.unique(candidates)
            
            base_rest = f.copy()
            base_rest[i] = 0.0
            rest_sum = base_rest.sum()
            
            best_v = old_val
            best_obj_local = fast_convolve(f).max()
            
            for v in candidates:
                f_trial = base_rest.copy()
                f_trial[i] = v
                trial_sum = f_trial.sum()
                if trial_sum < 1e-15:
                    continue
                f_trial = f_trial / trial_sum
                obj_val = fast_convolve(f_trial).max()
                if obj_val < best_obj_local - 1e-14:
                    best_obj_local = obj_val
                    best_v = v
            
            if best_v != old_val:
                f[i] = best_v
                f = f / f.sum()
                improved = True
        
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
        elif not improved:
            # No improvement in a full pass
            break
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    
    # Gaussian with various widths
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine raised to power
    for p in [0.5, 1.0, 2.0]:
        c = np.abs(np.cos(np.pi * x / n)) ** p + 0.01
        inits.append(c / c.sum())
    
    # U-shape (edges favored)
    edge_w = 1.0 / (0.3 * np.abs(x - center) / n + 0.05)
    inits.append(edge_w / edge_w.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - center) / n
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining gradient-based optimization,
    QP Frank-Wolfe, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass
    
    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    
    n_base = len(parent) if parent is not None else 384
    
    candidates = []

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, conv, beta):
    """
    Gradient of (2n/beta)*log(sum(exp(beta*(f*f)[k]))).
    Returns gradient w.r.t. f (before normalization).
    """
    n = len(f)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum

    # grad[i] = 4n * sum_k w[k] * f[k-i] (where 0<=k-i<n)
    # = 4n * cross_correlation(w, f)[i]
    # Compute via FFT
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    # We need c[i] = sum_j w[i+j] * f[j] for j=0..n-1
    # This equals (flip(w) * f)[2n-2-i]
    h = w[::-1]
    H = np.fft.rfft(h, N)
    conv_hf = np.fft.irfft(H * F, N)
    c = conv_hf[2 * n - 2 - np.arange(n)]

    return 4.0 * n * c

def peak_shave_optimize(f0, time_limit, seed=0):
    """
    Direct peak-shaving: iteratively identify the max lag and redistribute
    mass from peak-contributing coordinates to non-peak coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()

        # Find all lags near the max
        threshold = current_max * 0.95
        active_lags = np.where(conv >= threshold)[0]

        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])

        # For each active lag k, find which coordinates contribute most
        # Contribution of f[i] to (f*f)[k] is: 2*f[i]*f[k-i] (roughly)
        # We want to reduce f[i] for i that are "hot" (contribute to many active lags)

        hotness = np.zeros(n)
        for k in active_lags:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    hotness[i] += f[i] * f[j]

        # Reduce the hottest coordinates, increase the coldest
        f_max_idx = np.argmax(hotness)
        f_min_idx = np.argmin(hotness)

        if hotness[f_max_idx] - hotness[f_min_idx] < 1e-10:
            no_improve += 10
        else:
            # Redistribution: move mass from hot to cold
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                delta = f[f_max_idx] * frac
                if delta < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[f_max_idx] -= delta
                f_trial[f_min_idx] += delta
                f_trial = np.maximum(f_trial, 0)
                ts = f_trial.sum()
                if ts < 1e-15:
                    continue
                f_trial = f_trial / ts
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < current_max - 1e-13:
                    f = f_trial
                    break

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Try a more aggressive perturbation
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            no_improve = 0
            if no_improve > 500:
                break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=0, num_rounds=5):
    """
    Coordinate descent with golden-section search on each coordinate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    order = np.arange(n)
    golden = 0.618033988749895

    for rnd in range(num_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]

            # Search range for f[i]
            lo = 0.0
            hi = old_val * 3.0 + 0.1 / n

            if hi < 1e-15:
                hi = 1.0 / n * 10

            # Golden section search
            c = hi - golden * (hi - lo)
            d = lo + golden * (hi - lo)

            base = f.copy()
            base[i] = 0.0
            rest_sum = base.sum()

            def eval_at(val):
                f_trial = base.copy()
                f_trial[i] = val
                ts = f_trial.sum()
                if ts < 1e-15:
                    return float('inf')
                f_trial = f_trial / ts
                return fast_convolve(f_trial).max()

            fc = eval_at(c)
            fd = eval_at(d)

            for _ in range(8):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - golden * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + golden * (hi - lo)
                    fd = eval_at(d)

            best_cand = c if fc < fd else d
            best_val = min(fc, fd)

            # Also check the original
            if old_val > lo and old_val < hi:
                fo = eval_at(old_val)
                if fo < best_val:
                    best_cand = old_val
                    best_val = fo

            if best_val < fast_convolve(f).max() - 1e-13:
                f[i] = best_cand
                f = np.maximum(f, 0)
                f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            print(f"      Coord desc round {rnd+1}: obj={best_obj:.10f}")
        else:
            break

    return best_f, best_obj

def spectral_flatness_init(n, rng):
    """
    Generate initialization by optimizing spectral flatness in Fourier domain.
    Start with random phase and iteratively enforce non-negativity.
    """
    # Random initial spectrum with flat magnitude
    fft_size = 1 << (2 * n - 1).bit_length()
    # Flat magnitude, random phase
    magnitude = np.ones(fft_size // 2 + 1)
    phase = rng.standard_normal(fft_size // 2 + 1)
    # Make it real-valued (Hermitian symmetry)
    F = np.zeros(fft_size // 2 + 1, dtype=complex)
    F[0] = n  # DC = sum = n (before normalization)
    for k in range(1, fft_size // 2 + 1):
        if k > fft_size - k:
            break
        F[k] = magnitude[k] * (1 + 1j * phase[k])
        F[-k if -k >= -(fft_size//2) else k] = F[k].conjugate()

    # Compute IFFT and project
    f = np.fft.irfft(F, fft_size)[:n]
    f = np.maximum(f, 0)
    fs = f.sum()
    if fs < 1e-10:
        return np.ones(n) / n
    return f / fs

    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())

    for p in [0.5, 1.0, 2.0, 3.0]:
        c = np.abs(np.cos(np.pi * x / n)) ** p + 0.01
        inits.append(c / c.sum())

    edge_w = 1.0 / (0.3 * np.abs(x - center) / n + 0.05)
    inits.append(edge_w / edge_w.sum())

    # Spectral flatness inits
    for _ in range(3):
        inits.append(spectral_flatness_init(n, rng))

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        pp = np.maximum(parent, 0)
        pp = pp / pp.sum()
        for scale in [0.01, 0.03, 0.05]:
            pert = project_simplex(pp + rng.standard_normal(n) * scale)
            inits.append(pert)

    return inits


def smooth_subgrad_optimize(f0, time_limit, lr=0.005, mom=0.9,
                            top_k=50, temp=0.0001, seed=0):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            sg[indices] += 2.0 * w[idx] * f[k - indices]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 20 == 0:
            cur_lr *= 0.999
            cur_temp *= 1.005

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_f, best_obj
