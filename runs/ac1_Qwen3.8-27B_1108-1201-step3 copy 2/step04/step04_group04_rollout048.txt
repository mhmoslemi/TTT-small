"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid spectral + subgradient + LP approach.
Key innovations:
1. Gerchberg-Saxton spectral initialization for flat convolution
2. Nesterov-accelerated projected subgradient with backtracking line search
3. LP-based direction finding for stuck detection
4. Stochastic coordinate descent for fine-tuning
5. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_subgradient(a, conv, top_k=5, temp=0.001):
    """
    Compute smoothed subgradient of max(conv).
    Uses softmax weighting over top-k peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Find top-k peaks
    idx = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[idx]
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for t in range(len(idx)):
        if w[t] < 1e-15:
            continue
        k = int(idx[t])
        # d(a*a)[k]/da[i] = 2*a[k-i] for valid i
        # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi]
            # When i goes from i_lo to i_hi, k-i goes from k-i_lo to k-i_hi
            # k-i_lo = k - max(0,k-n+1), k-i_hi = k - min(n-1,k)
            j_lo = k - i_hi  # = max(0, k-n+1) ... let me just compute
            j_hi = k - i_lo
            grad[i_lo:i_hi+1] += w[t] * 2.0 * a[j_lo:j_hi+1][::-1]
    
    return grad


def spectral_init(n, max_iters=200, alpha=0.5, sum_target=1.0):
    """
    Gerchberg-Saxton style initialization.
    Zero-pad to 2n, FFT, flatten magnitude, IFFT, project.
    """
    a = np.ones(n) / n
    fft_len = 2 * n  # Next power of 2 for efficiency
    fft_len = 1 << (2 * n - 1).bit_length()
    
    for it in range(max_iters):
        # Zero-pad
        a_pad = np.zeros(fft_len)
        a_pad[:n] = a
        
        # FFT
        A = np.fft.rfft(a_pad)
        
        # Flatten magnitude
        mag = np.abs(A)
        # alpha < 1 flattens, alpha > 1 sharpens
        mag_new = mag ** alpha
        # Keep a floor to avoid zeroing
        mag_new = np.maximum(mag_new, mag.min() * 0.1 + 1e-10)
        A_new = mag_new * np.exp(1j * np.angle(A))
        
        # IFFT
        a_pad_new = np.fft.irfft(A_new, fft_len)
        
        # Project: take first n elements, clip to non-negative
        a = a_pad_new[:n].copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a * (sum_target / s)
    
    return a


def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=10, temp_init=0.001):
    """
    Nesterov-accelerated projected subgradient with backtracking.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)
        
        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
        else:
            a = a_new / s_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve % 20 == 0:
            lr *= 0.95
            temp *= 1.05
            mom = min(0.98, mom + 0.001)
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives the full convolution a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_direction(a, M, C=None):
    """
    Solve LP: max sum(g) s.t. C@g <= M, g >= 0.
    Returns direction d = g - a (if g has larger sum than a).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > a.sum() * (1 + 1e-6):
                return g
    except Exception:
        pass
    return None


def lp_step(a, conv, n):
    """One LP-based step with line search."""
    M = conv.max()
    C = build_conv_matrix(a)
    g = lp_direction(a, M * 1.001, C)
    
    if g is None:
        return a.copy(), False
    
    d = g - a
    if np.linalg.norm(d) < 1e-12:
        return a.copy(), False
    
    # Line search
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            break
    
    if best_obj < 2.0 * n * conv.max() - 1e-10:
        return best_a, True
    return a.copy(), False


def stochastic_local_search(a0, time_limit, n_trials_per_sec=5000):
    """
    Stochastic coordinate descent: randomly move mass between pairs.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    no_improve = 0
    T = 0.01  # "temperature" for move size
    
    while time.monotonic() - t0 < time_limit:
        # Batch of moves
        batch = max(1, int(n_trials_per_sec * 0.1))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Move small amount from i to j
            delta = rng.uniform(0.001, 0.01) * T
            if a[i] < delta:
                continue
            
            a[i] -= delta
            a[j] += delta
            
            # Quick check: only recompute if potentially beneficial
            # (heuristic: only check every few moves for speed)
            if _ % 5 == 0:
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    T = max(0.001, T * 0.99)
                else:
                    no_improve += 1
                    # Revert if too bad
                    if obj > best_obj * 1.01:
                        a[i] += delta
                        a[j] -= delta
            
            if no_improve > 10000:
                T = max(0.001, T * 0.9)
                if T < 0.001:
                    break
        
        if T < 0.001:
            break
    
    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Edge-concentrated
    edge_w = np.exp(-0.5 * (x / (0.3 * n)) ** 2) + np.exp(-0.5 * ((x - n + 1) / (0.3 * n)) ** 2)
    inits['edge'] = edge_w / edge_w.sum()
    
    # Center-concentrated
    center_w = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['center'] = center_w / center_w.sum()
    
    # Linear ramp
    ramp = (x + 1.0) / (n + 1.0)
    inits['ramp'] = ramp / ramp.sum()
    
    # V-shape (high at edges, low in middle)
    v = np.abs(x - 0.5 * (n - 1))
    inits['vshape'] = v / v.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

def _lp_optimize_full(a0, time_limit, max_no_improve=100):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            # Rebuild C periodically
            if no_improve == 1:
                C = build_conv_matrix(a)
        else:
            no_improve += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(no_improve * 31 + perturb_count)
            scale = 0.002 + 0.003 * min(perturb_count, 5)
            noise = rng.standard_normal(n) * scale

                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(300, 100 + 30 * perturb_count)
            
            if perturb_count > 50:

Actually, let me reconsider. The linear convolution `a*a` has length 2n-1. If we zero-pad `a` to length L >= 2n-1, then the circular convolution of the zero-padded sequence with itself equals the linear convolution. So:

`circular_conv(a_pad, a_pad) = linear_conv(a, a)` (padded to length L)

In the frequency domain: `IFFT(FFT(a_pad)^2) = circular_conv(a_pad, a_pad)`

So if we want `linear_conv(a,a)` to be flat, we want `FFT(a_pad)^2` to have a flat IFFT, which means `FFT(a_pad)^2` should be a constant (in the frequency domain), which means `|FFT(a_pad)|` should be constant.

So the GS approach IS correct! If we can make `|FFT(a_pad)|` constant, then the linear convolution will be flat. The issue is that we also need `a_pad` to be non-negative in the first n positions and zero in the rest.

The GS iteration alternates between:
1. Enforcing the time-domain constraint (non-negative, zero-padded)
2. Enforcing the frequency-domain constraint (flat magnitude)

def spectral_init(n, max_iters=200, alpha=0.5, sum_target=1.0):
    a = np.ones(n) / n
    fft_len = 1 << (2 * n - 1).bit_length()
    
    for it in range(max_iters):
        a_pad = np.zeros(fft_len)
        a_pad[:n] = a
        
        A = np.fft.rfft(a_pad)
        mag = np.abs(A)
        
        # Flatten magnitude (keep DC)
        # For k >= 1: mag_new[k] = mag[k]^alpha
        # This makes the spectrum flatter for alpha < 1
        mag_new = mag.copy()
        mag_new[1:] = (mag[1:] ** alpha)
        # Add a small floor
        floor_val = mag_new[1:].min() * 0.01 + 1e-10
        mag_new[1:] = np.maximum(mag_new[1:], floor_val)
        
        A_new = mag_new * np.exp(1j * np.angle(A))
        a_pad_new = np.fft.irfft(A_new, fft_len)
        
        a = np.maximum(a_pad_new[:n], 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (sum_target / s)
        else:
            a = np.ones(n) / n
    
    return a
