import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, mu, conv_cache=None):
    """
    Compute gradient of the log-sum-exp smooth approximation of max(conv(f,f)).
    F_mu(f) = mu * log(sum_k exp(conv(f,f)[k] / mu))
    grad[F_mu][j] = 2 * sum_k p[k] * f[k-j] * 1_{0<=k-j<n}
    where p[k] = exp(conv[k]/mu) / sum_l exp(conv[l]/mu)
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Compute softmax weights
    scaled = conv / mu
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient: for each j, sum over valid k of p[k] * 2 * f[k-j]
    # This is a cross-correlation of p (length 2n-1) with f (length n)
    # grad[j] = 2 * sum_{k} p[k] * f[k-j] * 1_{0<=k-j<n}
    #         = 2 * sum_{d=0}^{n-1} f[d] * p[j+d] * 1_{0<=j+d<2n-1}
    
    # Use FFT for efficiency
    pad_size = 1 << (3 * n - 1).bit_length()
    F = np.fft.rfft(f, pad_size)
    P = np.fft.rfft(p, pad_size)
    # We need correlation: sum_d f[d] * p[j+d]
    # This is the correlation of f and p, which is ifft(conj(F) * P) shifted
    corr = np.fft.irfft(np.conj(F) * P, pad_size)
    # corr[m] = sum_d f[d] * p[m+d]
    # We want grad[j] = 2 * corr[j] for j in [0, n-1]
    grad = 2.0 * corr[:n]
    
    return grad, conv, p


def nesterov_smooth(f0, time_limit, mu_init=0.001, mu_min=1e-6, mu_decay=0.9997,
                     lr_init=0.002, use_entropic=True, verbose=False):

    Nesterov-accelerated smooth subgradient on the simplex.
    Uses log-sum-exp smoothing with adaptive temperature.

    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    f_prev = f.copy()
    y = f.copy()
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mu = mu_init
    lr = lr_init
    t_nest = 1
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        t_nest = it
        
        # Compute smooth gradient at lookahead point y
        grad, conv_y, p = smooth_grad(y, mu)
        
        # Line search: try a few step sizes
        best_f_new = f.copy()
        best_val = best_obj
        best_alpha = 0.0
        
        alpha_candidates = [lr, lr * 0.5, lr * 2.0, lr * 0.25, lr * 4.0]
        for alpha in alpha_candidates:
            if use_entropic:
                # Multiplicative update (entropic mirror)
                f_new = f * np.exp(-alpha * grad)
                f_new = np.maximum(f_new, 1e-30)
                ss = f_new.sum()
                if ss > 1e-15:
                    f_new = f_new / ss
                else:
                    continue
            else:
                # Additive update (Euclidean mirror)
                f_new = project_simplex(f - alpha * grad)
            
            conv_new = fast_convolve(f_new)
            val = conv_new.max()
            if val < best_val:
                best_val = val
                best_f_new = f_new.copy()
                best_alpha = alpha
        
        if best_alpha > 0:
            f_new = best_f_new
        else:
            # Fallback: tiny step
            if use_entropic:
                f_new = f * np.exp(-lr * 0.1 * grad)
                f_new = np.maximum(f_new, 1e-30)
                ss = f_new.sum()
                if ss > 1e-15:
                    f_new = f_new / ss
                else:
                    f_new = f.copy()
            else:
                f_new = project_simplex(f - lr * 0.1 * grad)
        
        # Nesterov update
        beta = (t_nest - 1.0) / (t_nest + 2.0)
        y = f_new + beta * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = f_new.copy()
        
        f_prev = f
        f = f_new
        
        # Update best
        conv_f = fast_convolve(f)
        obj = conv_f.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive mu: cool down slowly, reheat if stuck
        if no_improve > 50:
            mu = min(mu * 1.5, mu_init * 5.0)
            lr = lr * 0.7
            if no_improve > 200:
                # Restart from best
                f = best_f.copy()
                y = best_f.copy()
                mu = mu_init
                lr = lr_init
                t_nest = 1
                no_improve = 0
        elif no_improve > 20:

        else:
            mu = max(mu * mu_decay, mu_min)

def entropic_mirror_descent(f0, time_limit, lr_init=0.01, mu_init=0.002,
                            mu_decay=0.9995, verbose=False):
    """
    Mirror descent with entropic regularization and log-sum-exp smoothing.
    Simpler than Nesterov but more stable.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mu = mu_init
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv_f, p = smooth_grad(f, mu)
        
        # Entropic mirror update: f_new[j] proportional to f[j] * exp(-lr * grad[j])
        log_update = -lr * grad
        log_update -= log_update.max()  # numerical stability
        f_new = f * np.exp(log_update)
        f_new = np.maximum(f_new, 1e-30)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss
        
        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Schedule
        mu = max(mu * mu_decay, 1e-7)
        lr = max(lr * 0.9998, lr_init * 0.2)
        
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            f = best_f * np.exp(rng.standard_normal(n) * 0.003)
            f = f / f.sum()
            mu = mu_init
            lr = lr_init
            no_improve = 0
    
    return best_f, best_obj


def targeted_peak_break(f0, time_limit, verbose=False):
    """
    Targeted peak-breaking: identify the dominant peak and apply
    a structured reduction to break it, then re-optimize.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        n_peaks = min(10, len(conv))
        peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[peak_indices]
        
        # Weight peaks by their "excess" over the median
        median_conv = np.median(conv)
        excess = peak_vals - median_conv
        if excess.max() > 0:
            w = excess / excess.sum()
        else:
            w = np.ones(len(peak_indices)) / len(peak_indices)
        
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-14:
                continue
            k = int(peak_indices[i])
            # d/d(f[j]) (f*f)[k] = 2*f[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += 2.0 * w[i] * f[k - idx]
        
        # Apply reduction: decrease f[j] proportionally to pressure
        # But preserve the "structure" by using a soft update
        eta = 0.005
        f_new = f * (1.0 - eta * pressure / (pressure.max() + 1e-15))
        f_new = np.maximum(f_new, 1e-30)
        ss = f_new.sum()
        if ss > 1e-15:
            f_new = f_new / ss
        else:
            f_new = f.copy()
        
        # Also try a "symmetric averaging" step: blend with reversed f
        f_rev = f[::-1].copy()
        t_sym = 0.1
        f_sym = (1 - t_sym) * f_new + t_sym * f_rev
        f_sym = np.maximum(f_sym, 0)
        f_sym = f_sym / f_sym.sum()
        
        # Pick the better of f_new and f_sym
        obj_new = fast_convolve(f_new).max()
        obj_sym = fast_convolve(f_sym).max()
        
        if obj_sym < obj_new:
            f = f_sym
            obj = obj_sym
        else:
            f = f_new
            obj = obj_new
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
        
        # Decay eta
        eta = max(eta * 0.999, 0.0005)
    
    return best_f, best_obj


def nesterov_projected(f0, time_limit, mu_init=0.001, lr_init=0.001,
                       mu_decay=0.9998, verbose=False):
    """Nesterov accelerated method with Euclidean projection (additive updates)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    f_prev = f.copy()
    y = f.copy()
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mu = mu_init
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, _, _ = smooth_grad(y, mu)
        
        # Projected gradient step
        f_new = project_simplex(y - lr * grad)
        
        # Nesterov momentum
        beta = it / (it + 3.0)
        y = f_new + beta * (f_new - f)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = f_new.copy()
        
        f_prev = f
        f = f_new
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        mu = max(mu * mu_decay, 1e-7)
        lr = lr * 0.9999
        
        if no_improve > 200:
            f = best_f.copy()
            y = best_f.copy()
            mu = mu_init
            lr = lr_init * 0.8
            no_improve = 0
    
    return best_f, best_obj


def interleaved_initialization(f_ref, n_target, rng):
    """Create an initialization of length n_target from a reference sequence."""
    if len(f_ref) == n_target:
        return f_ref.copy()
    # Interpolate
    n_ref = len(f_ref)
    x_ref = np.linspace(0, 1, n_ref)
    x_tgt = np.linspace(0, 1, n_target)
    f_tgt = np.interp(x_tgt, x_ref, f_ref)
    f_tgt = np.maximum(f_tgt, 0)
    ss = f_tgt.sum()
    if ss > 1e-15:
        f_tgt = f_tgt / ss
    else:
        f_tgt = np.ones(n_target) / n_target
    return f_tgt


def diverse_initializations(n, f_ref, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Interpolated reference
    if f_ref is not None:
        inits.append(interleaved_initialization(f_ref, n, rng))
        # Perturbed reference
        f_pert = interleaved_initialization(f_ref, n, rng)
        f_pert = f_pert * np.exp(rng.standard_normal(n) * 0.01)
        f_pert = np.maximum(f_pert, 0)
        f_pert = f_pert / f_pert.sum()
        inits.append(f_pert)
    
    # Edge-weighted (more mass near edges)
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())
    
    # Center-weighted (more mass in center)
    center_w = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits.append(center_w / center_w.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Two-bump
    b1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    two_bump = b1 + b2
    inits.append(two_bump / two_bump.sum())
    
    return inits
