def subgradient_step(a, conv, step_size, top_k=10):
    """One subgradient step with clipping to non-negativity."""
    n = len(a)
    # Get active set (top k peaks)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    max_val = conv.max()
    tol = max_val * 1e-8
    active = peaks[conv[peaks] >= max_val - tol]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += a[k - idxs]
    
    # Step
    s = a.sum()
    a_new = a - step_size * grad / (s * s)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    return a_new / s_new

def lp_repair(a, M, max_iter=50):
    """Try to find a nearby point with conv <= M using LP relaxation."""
    n = len(a)
    # Build the convolution matrix
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = a
    
    # Solve: max sum(g) s.t. C@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except:
        pass
    return a.copy()

def anneal_optimize(a0, time_limit, rng, seed_offset=0):
    """Simulated annealing with subgradient steps."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    T_min = 1e-6

            j_end = min(k, n - 1)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += a[k - idxs]
        
        # Step size adaptive
        step = T * n / max(grad.sum(), 1e-10)
        
        a_try = a - step * grad
        # Add noise

        a_try = a_try + noise

        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        # Accept if better or with probability exp(-delta/T)
        if obj_try < cur_obj:

The convolution `c[k] = sum_j a[j]*a[k-j]` for non-negative `a` is always non-negative. The sum of all `c[k]` is `(sum a)^2 = 1` (when normalized). There are `2n-1` values.

To reduce the peak, we want to decrease `a[i]` for `i` where `a[k*-i]` is large, and increase `a[i]` for `i` where `a[k*-i]` is small.

This is like an "anti-correlation" step: move mass from positions `i` where the "mirror position" `k*-i` has high value, to positions where the mirror position has low value.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a, conv=None):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    if conv is None:
        conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_subgradient(a, conv, top_k=10, sharpness=20):
    """
    Compute weighted subgradient of max(conv) w.r.t. a.
    Uses top-k peaks with exponential weighting for smoothness.
    """
    n = len(a)
    if top_k >= n:
        top_k = n
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    max_val = peak_vals.max()
    
    # Exponential weighting
    scaled = (peak_vals / max_val) ** sharpness
    w_sum = scaled.sum()
    if w_sum < 1e-15:
        scaled = np.zeros(len(peaks))
        scaled[np.argmax(peak_vals)] = 1.0
    else:
        scaled /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if scaled[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += scaled[idx] * a[k - idxs]
    return grad


def backtracking_line_search(a, grad, conv, n, max_iter=30):
    """
    Backtracking line search for a - step * grad.
    Returns (a_new, step_used, improved).
    """
    s = a.sum()
    cur_obj = 2.0 * n * conv.max() / (s * s)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a.copy(), 0.0, False
    
    # Initial step size
    step = 1.0 / grad_norm
    best_a = a.copy()
    best_obj = cur_obj
    best_step = 0.0
    
    for _ in range(max_iter):
        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            step *= 0.5
            continue
        a_try = a_try / s_try
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        if obj_try < best_obj - 1e-12:
            best_a = a_try.copy()
            best_obj = obj_try
            best_step = step
            break
        step *= 0.6
    
    return best_a, best_step, best_step > 0

def lp_direction(a, M, n):
    """
    Find direction g that maximizes sum(g) subject to a*g <= M, g >= 0.
    Returns normalized g or None on failure.
    """
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True,
                                                  'time_limit': 10})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10 and g_sum > np.sum(a) * (1 + 1e-6):
                return g / g_sum
    except Exception:
        pass
    return None

        sig = (n - x + 1.0) ** pw
        inits.append((f'rpw{pw}', sig / sig.sum()))

            sig = np.exp(-0.5 * ((x - center * n) / (width * n)) ** 2)
            inits.append((f'gau{center}_{width}', sig / sig.sum()))
    
    # Bimodal
    for ratio in [0.3, 0.5, 0.7]:
        bi = np.zeros(n)
        half = n // 2
        bi[:int(half * ratio)] = 1.0
        bi[int(half + half * (1 - ratio)):] = 1.0
        if bi.sum() > 0:
            inits.append((f'bi{ratio}', bi / bi.sum()))
    
    # V-shape (high at edges)
    v = np.abs(x - n / 2) / (n / 2)
    v = v ** 2 + 0.01
    inits.append(('vshape', v / v.sum()))
    
    # Random

                    pps = pp.sum()
