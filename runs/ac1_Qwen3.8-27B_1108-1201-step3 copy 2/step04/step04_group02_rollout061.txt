def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}"""
    v = np.array(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)] * 1) * (target_sum / max(v[0], 1e-30))
    # Standard simplex projection for target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * target_sum / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # Second term: for fixed j, k ranges from 0 to min(j, size-1)

    #       second term at k=j gives w[j]*f[0]

    #

    # (for real signals, conj(FFT(w)) = FFT(w) reversed, but let's use the general formula)

    grad = T1 + T2

Now I have an efficient gradient computation. Let me design the full optimization:

**Algorithm: Smoothed Gradient Descent with Nesterov Momentum**

1. Initialize f on simplex (sum=1, f>=0)
2. Loop:
   a. Compute conv = f*f (FFT)
   b. Compute weights w = softmax(β * conv) / sum(softmax(β*conv)) [this is the smoothed max gradient weights]
   c. Compute grad = 2 * cross_corr(w, f)
   d. Nesterov step: f_new = f - lr * grad, project to simplex
   e. Adaptive step size (Barzilai-Borwein or backtracking)
   f. Update β (increase over time to sharpen the max)
3. Return best f found

This should be much more effective than the previous subgradient approach because:
- We use the true gradient of the smooth approximation (not a subgradient)
- Nesterov momentum gives faster convergence
- The softmax approximation to max becomes exact as β → ∞

Let me also add the spectral flattening as a complementary method:

**Spectral Flattening:**
1. Compute P = |FFT(f)|^2 (power spectrum)
2. Target: make P as flat as possible
3. Update: FFT(f_new) = FFT(f) * (P_mean / P)^α for α in (0, 1]
4. f_new = real(IFFT(FFT(f_new)))[:n]
5. Project to simplex

This works because flattening the power spectrum reduces the variance of the auto-convolution (by Parseval's identity relating L4 norm of spectrum to L2 norm of convolution).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (softmax approximation to max) with Nesterov momentum
2. Spectral flattening (iterative power spectrum equalization)
3. Multi-scale exploration with diverse initializations
4. LP-based Frank-Wolfe for global direction finding
5. Adaptive time allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(f, weights):
    """Compute gradient of sum_k weights[k] * (f*f)[k] w.r.t. f.
    
    grad = 2 * cross_correlation(weights, f)
    where cross_correlation[w,f][j] = sum_k w[k+j]*f[k]
    """
    n = len(f)
    size = 2 * n - 1
    # Need N >= size + n - 1 to avoid circular aliasing in cross-correlation
    N = 1 << (size + n - 2).bit_length()
    W = np.fft.rfft(weights, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(np.conj(W) * F, N)
    return 2.0 * cross[:n]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0) * s / max(abs(v[0]), 1e-30)])
    # Standard algorithm for projection onto simplex with target sum s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.ones(n) * s / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_smooth_max(conv, beta):
    """Compute softmax weights for smoothed max of conv."""
    shifted = beta * conv - beta * conv.max()
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    return w


def spectral_flatten(f, alpha=0.3, n_iter=10):
    """Iteratively flatten the power spectrum of f."""
    n = len(f)
    f = np.maximum(f, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for _ in range(n_iter):
        A = np.fft.rfft(f, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        # Avoid division by zero
        P_safe = np.maximum(P, P_mean * 0.05)
        # Flatten: reduce peaks, boost valleys
        A_new = A * (P_mean / P_safe) ** alpha
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f = project_simplex(f_new)
    
    return f


def sg_smooth_descent(f0, time_limit, beta_init=100.0, beta_max=10000.0,
                      lr_init=0.005, mom_init=0.84, verbose=False):
    """Smoothed gradient descent with Nesterov momentum and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Nesterov state
    y = f.copy()  # look-ahead point
    v = np.zeros(n)  # velocity
    mom = mom_init
    lr = lr_init
    beta = beta_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_grad = None
    prev_f = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Nesterov: compute gradient at look-ahead point y
        # For simplicity, use gradient at current f (standard momentum)
        w = softmax_smooth_max(conv, beta)
        grad = compute_gradient(f, w)
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            s_vec = f - prev_f
            y_vec = grad - prev_grad
            num = np.dot(s_vec, s_vec)
            den = np.dot(s_vec, y_vec)
            if den > 1e-12:
                bb_lr = num / den
                if 1e-6 < bb_lr < 0.1:
                    lr = 0.5 * lr + 0.5 * bb_lr
        
        prev_f = f.copy()
        prev_grad = grad.copy()
        
        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        
        # Adaptive parameters
        beta = min(beta * 1.005, beta_max)
        lr *= 0.9998
        mom = min(mom * 1.0001, 0.96)
        
        # Escape from local minima
        if no_improve > 200:
            # Try spectral flattening
            f_spec = spectral_flatten(f, alpha=0.5, n_iter=5)
            conv_spec = fast_convolve(f_spec)
            if conv_spec.max() < best_obj:
                best_obj = conv_spec.max()
                best_f = f_spec.copy()
                f = f_spec
                conv = conv_spec
            
            # Perturb
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            lr = lr_init * 0.5
            beta = beta_init
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_quick(f, M, time_limit=10):
    """Quick LP solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] <= M for all k
    # (f*g)[k] = sum_j f[j] * g[k-j] (valid j)
    # This is a (2n-1) x n matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_escape(f, best_obj, time_limit=15):
    """Use LP to find an escape direction."""
    n = len(f)
    # Try slightly larger M to find a feasible direction
    for expansion in [1.01, 1.05, 1.1, 1.2, 1.5]:
        M = best_obj * expansion
        g = solve_lp_quick(f, M, time_limit=time_limit)
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            # Line search
            best_alpha = 0.0
            best_val = best_obj
            for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                ft = project_simplex((1 - alpha) * f + alpha * g)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            if best_alpha > 0:
                f_new = project_simplex((1 - best_alpha) * f + best_alpha * g)
                return f_new, best_val
    return f, best_obj


def nesterov_polish(f0, time_limit, beta=5000.0, lr=0.001, mom=0.94):
    """Fine polishing with very high beta (nearly exact max) and small steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        it += 1
        
        w = softmax_smooth_max(conv, beta)
        grad = compute_gradient(f, w)
        
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        lr *= 0.9999
        
        if no_improve > 300:
            rng = np.random.default_rng(it + 50)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            conv = fast_convolve(f)
            v = np.zeros(n)
            lr = 0.0005
            no_improve = 0
    
    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Triangle (peaked in middle)
    tri = 1.0 - 2.0 * np.abs(x - (n-1)/2.0) / (n-1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())
    
    # Inverse triangle (peaked at edges)
    itr = 2.0 * np.abs(x - (n-1)/2.0) / (n-1) + 0.1
    inits.append(itr / itr.sum())
    
    # Cosine window (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())
    
    # Flat-top approximation
    flat = np.ones(n)
    for i, c in enumerate([0.2156, 0.4160, 0.2742, 0.0811]):
        flat += c * np.cos(2 * np.pi * i * (x + 0.5) / n)
    flat = np.maximum(flat, 0.01)
    inits.append(flat / flat.sum())
    
    # Exponential (centered)
    exp_c = np.exp(-0.5 * ((x - (n-1)/2.0) / (n/4.0)) ** 2)
    inits.append(exp_c / exp_c.sum())
    
    # Two-peak
    two = np.exp(-0.5 * ((x - n/4.0) / (n/10.0)) ** 2) + np.exp(-0.5 * ((x - 3*n/4.0) / (n/10.0)) ** 2)
    inits.append(two / two.sum())
    
    # Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # Square wave (alternating high/low)
    sq = np.where(x % 2 == 0, 1.0, 0.3)
    inits.append(sq / sq.sum())
    
    # Random smooth (Gaussian kernel random)
    base = rng.standard_normal(n)
    # Smooth it
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(base, kernel, mode='same')
    smooth = np.maximum(smooth, 0) + 0.01
    inits.append(smooth / smooth.sum())
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with various
            inits.append((0.7 * px/ps + 0.3 * np.ones(n)/n) )
            inits.append((0.7 * px/ps + 0.3 * (hann/hann.sum())))
    
    # Normalize all
    inits = [a / a.sum() for a in inits if a.sum() > 1e-10]
    return inits
