def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve_pair(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_gradient(f, w, n):
    """Compute grad[i] = 2 * sum_j f[j] * w[i+j] where w has length 2n-1."""
    # grad[i] = sum_j f[j] * w[i+j] = convolve(f, w_rev)[2n-2-i]
    w_rev = w[::-1]
    size = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:size]
    # grad[i] = 2 * conv[2n-2-i] for i=0..n-1
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    return grad

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    if not cond.any():

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def smoothed_max_and_weights(conv, tau):
    """Compute smoothed max and softmax weights."""
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    smooth_max = conv.max() + tau * np.log(ws)  # Actually: tau * log(sum(exp(conv/tau)))
    return smooth_max, w

def generate_inits(n, parent, rng):

        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / sigma)**2)
        inits.append(g / g.sum())
    
    # Cosine

    # Beta-like
    for alpha, beta in [(2,2), (3,1.5), (4,2)]:

    # U-shape (edges elevated)

    # Random

    # From parent

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:

            ps2 = pp.sum()

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization on the probability
simplex with exact gradient of the smoothed L-infinity norm of auto-convolution.
Key innovations: (1) exact gradient via FFT-based cross-correlation, (2) adaptive
temperature scheduling with multi-scale tau averaging, (3) peak-targeting mode,
(4) multi-length search with warm-starting, (5) local search polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_pair(a, b):
    """Linear cross-convolution of a (len n) and b (len m) via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_gradient_fft(f, w, n):
    """
    Compute grad[i] = 2 * sum_j f[j] * w[i+j] where w has length 2n-1.
    Uses FFT-based convolution with reversal trick.
    """
    w_rev = w[::-1]
    size = n + (2 * n - 1) - 1  # 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:size]
    # grad[i] = 2 * conv[2n-2-i] for i=0..n-1
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, tau):
    """Compute softmax weights of conv/tau."""
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    return w


def adam_simplex_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                          eps=1e-8, tau_init=0.0005, tau_min=1e-5,
                          tau_decay=0.9995, multi_tau=False, verbose=False):
    """
    Adam optimization on the probability simplex to minimize max auto-conv.
    Uses exact gradient of the smoothed (log-sum-exp) objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    best_tau = tau

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1

        conv = fast_convolve(f)

        if multi_tau:
            # Average gradient over multiple tau values
            tau_list = [tau, tau * 2, tau * 4]
            grad = np.zeros(n)
            count = 0
            for t_i in tau_list:
                w = softmax_weights(conv, t_i)
                grad += compute_gradient_fft(f, w, n)
                count += 1
            grad /= count
        else:
            w = softmax_weights(conv, tau)
            grad = compute_gradient_fft(f, w, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            best_tau = tau
        else:
            no_improve += 1

        f = f_new
        tau = max(tau_min, tau * tau_decay)

        # Adaptive restart
        if no_improve > 400:
            f = best_f.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = max(tau_min, best_tau * 0.7)
            lr = max(lr * 0.7, 0.0001)
            no_improve = 0

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t}, obj={best_obj:.10f}, tau={tau:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj

    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    lr = lr_init
    no_improve = 0

    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_max = conv.max()

        kk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -kk)[-kk:]
        pv = conv[peaks]
        # Softmax-like weights over peaks
        scaled = pv - pv.max()
        ew = np.exp(scaled / (pv.max() * 0.1 + 1e-10))
        wsum = ew.sum()
        if wsum < 1e-300:
            ew = np.zeros(kk)
            ew[0] = 1.0
        else:
            ew /= wsum

        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(kk):
            if ew[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            grad[indices] += 2.0 * ew[i] * f[k - indices]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        f_new = f - lr * grad / gnorm
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.02)
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 300:
                f = best_f.copy()
                no_improve = 0
                lr = max(lr_init * 0.5, 0.0005)

        f = f_new

        if verbose and it % 2000 == 0:
            print(f"      PeakTarget it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def momentum_subgrad_optimize(f0, time_limit, lr=0.003, momentum=0.93,
                              top_k=20, temp=0.0001, verbose=False):
    """
    Momentum-based subgradient with softmax-weighted multi-peak targeting.
    More aggressive than Adam for this non-smooth objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        kk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -kk)[-kk:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(kk)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(kk):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            indices = np.arange(js, je + 1)
            grad[indices] += 2.0 * w[i] * f[k - indices]

        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0

    return best_f, best_obj


def local_search(f0, time_limit, rng, base_scale=0.003):
    """Random perturbation local search on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    scale = base_scale

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        for _ in range(300):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()

        scale *= 0.94
        if scale < 1e-5:
            scale = base_scale * 0.3
            f_jump = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            obj = fast_convolve(f_jump).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_jump.copy()

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_rounds=100):
    """
    Coordinate descent: for each coordinate i, find the optimal value
    that minimizes the max conv, keeping others fixed.
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    round_num = 0
    while time.monotonic() - t0 < time_limit and round_num < n_rounds:
        round_num += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Try a few values for f[i]
            old_val = f[i]
            # Normalize: if we change f[i] by delta, others scale
            # f_new[j] = f[j] * (1 - delta/f[i]) for j != i, f_new[i] = f[i] - delta + 1
            # Actually simpler: parametrize f[i] = alpha, rest scaled to sum to 1-alpha
            rest_sum = 1.0 - old_val
            if rest_sum < 1e-10:
                continue

            # Try alpha values: old, old*0.8, old*1.2, old +/- small
            candidates = [old_val,
                          max(0.0, old_val * 0.7),
                          max(0.0, old_val * 1.3),
                          max(0.0, old_val - 0.001),
                          max(0.0, old_val + 0.001)]
            candidates = [c for c in candidates if c < 0.999]

            best_alpha = old_val
            best_cov_max = best_obj

            for alpha in candidates:
                if alpha >= 1.0 or alpha < 0.0:
                    continue
                f_trial = f.copy()
                f_trial[i] = alpha
                rest = 1.0 - alpha
                if rest < 1e-10:
                    continue
                # Scale rest proportionally
                for j in range(n):
                    if j != i:
                        f_trial[j] = f[j] * rest / rest_sum
                obj = fast_convolve(f_trial).max()
                if obj < best_cov_max:
                    best_cov_max = obj
                    best_alpha = alpha
                    improved = True

            if best_alpha != old_val:
                f[i] = best_alpha
                rest = 1.0 - best_alpha
                if rest > 1e-10:
                    for j in range(n):
                        if j != i:
                            f[j] = f[j] * rest / rest_sum

        if improved:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
        else:
            if round_num > 5:
                break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    # Gaussians with various widths
    for sigma_frac in [0.12, 0.18, 0.25, 0.35, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine hump
    c = np.cos(np.pi * (x - mid) / n) ** 2 + 0.01
    inits.append(c / c.sum())

    # Beta-like
    for alpha, beta in [(2.0, 2.0), (3.0, 3.0), (4.0, 2.0), (1.5, 1.5)]:
        b = (x + 1) ** alpha * (n - x) ** beta
        inits.append(b / b.sum())

    # U-shape
    edge = 1.0 / (0.5 * np.abs(x - mid) / n + 0.08)
    inits.append(edge / edge.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.7))

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        for scale in [0.005, 0.01, 0.02]:
            pp = np.maximum(px / ps + rng.standard_normal(n) * scale, 0)
            ps2 = pp.sum()
            if ps2 > 1e-10:
                inits.append(pp / ps2)

    return inits
