def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, g):
    """Compute cross-correlation: (a * g)[k] = sum_i a[i] * g[k-i]"""
    n = len(a)  # assume len(a) == len(g)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]

def smooth_obj_grad(a, beta):
    """Compute 2n * smax_beta(conv(a)) and its gradient, where smax is log-sum-exp."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    
    # Objective: 2n * (max_c + log(sum_exp)/beta)
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)
    
    # Gradient: 4n * correlate(w, a, 'valid')
    grad = 4.0 * n * np.correlate(w, a, mode='valid')
    
    return obj, grad

def lp_direction(a, M, top_k=200):

    n = len(a)
    conv = fast_convolve(a)
    # Get top-k indices

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_obj_grad(a, beta):
    """Compute smooth approximation 2n*smax_beta(conv(a)) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)
    grad = 4.0 * n * np.correlate(w, a, mode='valid')
    return obj, grad


def smooth_optimize(a0, time_limit, beta_init=200.0, beta_final=5000.0,
                    lr_init=0.01, momentum=0.85, use_mult=True):
    """
    Smooth max optimization with adaptive beta and multiplicative/projected updates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    beta = beta_init
    no_improve = 0
    step_count = 0

        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        obj_smooth, grad = smooth_obj_grad(a, beta)

            # Multiplicative (exponentiated gradient) update
            v = momentum * v + (1 - momentum) * grad
            a_new = a * np.exp(-lr * v)
            a_new = np.maximum(a_new, 1e-30)
        else:
            # Projected gradient update
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)

def lp_direction_search(a, M, top_k=300):

        m_end = min(n - 1, k)
        if m_start <= m_end:

            A_ub[idx, m_arr] = a[k - m_arr]

    b_ub = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n

        if len(parent) != n:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            parent_resampled = np.interp(x_new, x_old, parent)
        else:
            parent_resampled = parent.copy()
        s = parent_resampled.sum()
        if s > 1e-15:
            inits['parent'] = parent_resampled / s

    # Alternating pattern (key structural insight)
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alt'] = alt / alt.sum()

    # Strong alternating
    alt2 = np.ones(n)
    alt2[::2] = 1.6
    alt2[1::2] = 0.4
    inits['alt2'] = alt2 / alt2.sum()

    # Sine modulated
    sine = 0.7 + 0.3 * np.sin(4.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()

    # Two-lobe
    lobe = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = np.maximum(lobe, 0.01)
    inits['tlobe'] = inits['tlobe'] / inits['tlobe'].sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    return inits


def resample_to_length(seq, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Quick exploration across lengths and initializations (LP + smooth)
    Phase 2: Deep optimization on best candidates (combined method)
    Phase 3: Intensive polishing (alternating smooth/LP with fine params)
    Phase 4: Final micro-perturbations and verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score≈{2*len(p)*fast_convolve(parent).max():.6f}")
    except Exception:
        pass

    # ===== Phase 1: Quick Exploration =====
    print("=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 300.0
    phase1_deadline = t_start + phase1_budget

    lengths = [640, 768, 896, 1024]
    results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_initializations(n, parent, rng)
        # Priority: parent > alt > alt2 > uniform > sine > tlobe > others
        priority = ['parent', 'alt', 'alt2', 'uniform', 'sine', 'tlobe', 'gauss', 'tri', 'random']
        init_names = [nm for nm in priority if nm in inits][:5]

        t_per = max(12.0, (phase1_budget - 40.0) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]

            # Alternate between LP and smooth for diversity
            if name in ['parent', 'uniform', 'sine']:
                a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc, top_k=200)
            else:
                a_opt, obj_opt = smooth_optimize(a0, time_limit=t_alloc,
                                                  beta_init=200.0, beta_final=5000.0,
                                                  lr_init=0.01, momentum=0.85)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep Optimization =====
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    # Get top candidates (distinct)
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, name, a_opt.copy()))
            seen.add(key)
        if len(top_candidates) >= 5:
            break

    # Always include the overall best
    if best_sequence is not None:
        best_n = len(best_sequence)
        if (best_n, 'overall_best') not in seen:
            top_candidates.append((best_n, 'overall_best', best_sequence.copy()))

    # Allocate time: reserve 450s for phases 3+4
    remaining_for_p2 = deadline - time.monotonic() - 450.0
    if remaining_for_p2 < 60:
        remaining_for_p2 = 60
    t_per_deep = max(25.0, remaining_for_p2 / max(1, len(top_candidates)))

    for i, (n, name, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 440:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 420)
        if t_alloc < 10:
            break

        a_opt, obj_opt = combined_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i} (n={n}, {name}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Intensive Polishing =====
    print("=== Phase 3: Intensive Polishing ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 180.0
        if remaining > 30:
            # Multiple polishing passes with different params
            polish_configs = [
                {'beta_init': 400.0, 'beta_final': 15000.0, 'lr_init': 0.004, 'momentum': 0.93, 'use_mult': True},
                {'beta_init': 800.0, 'beta_final': 25000.0, 'lr_init': 0.003, 'momentum': 0.95, 'use_mult': False},
                {'beta_init': 2000.0, 'beta_final': 40000.0, 'lr_init': 0.002, 'momentum': 0.90, 'use_mult': True},
            ]

            t_per_polish = remaining / (len(polish_configs) + 1.0)

            for ci, cfg in enumerate(polish_configs):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(t_per_polish, deadline - time.monotonic() - 120)
                if t_alloc < 8:
                    break
                a_opt, obj_opt = smooth_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {ci}: {score:.6f}")

            # LP polish after smooth
            if time.monotonic() < deadline - 60:
                t_alloc = min(45.0, deadline - time.monotonic() - 40)
                if t_alloc > 10:
                    a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc, top_k=400)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP polish: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Refinement =====
    print("=== Phase 4: Final Refinement ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_trials = max(1, int(remaining / 20))
            t_per = remaining / n_trials

            for pi in range(n_trials):
                if time.monotonic() > deadline - 15:
                    break
                if t_per < 5:
                    break

                # Diverse perturbation strategies
                strategy = pi % 4
                if strategy == 0:
                    # Random noise
                    scale = 0.001 + 0.0005 * pi
                    a_pert = best_sequence + rng.standard_normal(n_best) * scale
                elif strategy == 1:
                    # Smooth perturbation
                    freq = 3 + rng.integers(0, 5)
                    phase = rng.uniform(0, 2 * np.pi)
                    perturb = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase) * 0.002
                    a_pert = best_sequence + perturb
                elif strategy == 2:
                    # Alternating perturbation
                    alt_pert = np.ones(n_best) * 0.001
                    alt_pert[::2] *= 1.5
                    alt_pert[1::2] *= 0.5
                    a_pert = best_sequence * (1 + (alt_pert - alt_pert.mean()) * 5)
                else:
                    # Local random changes
                    a_pert = best_sequence.copy()
                    n_changes = rng.integers(5, 20)
                    idx = rng.choice(n_best, size=n_changes, replace=False)
                    a_pert[idx] *= (1 + rng.standard_normal(n_changes) * 0.05)

                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 5)
                if t_alloc < 3:
                    break

                # Use the best-performing method for this perturbation
                if pi % 2 == 0:
                    a_opt, obj_opt = smooth_optimize(a_pert, time_limit=t_alloc,
                                                      beta_init=3000.0, beta_final=50000.0,
                                                      lr_init=0.002, momentum=0.93)
                else:
                    a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, top_k=350)

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        if parent is not None:
            best_sequence = parent
        else:
            best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
