import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve_auto(a):
    """Auto-convolution via FFT, returns full linear convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=128, temp=0.001):
    """Subgradient of max(conv) using softmax over top peaks."""
    n = len(f)
    L = len(conv)
    if top_k >= L:
        peaks = np.arange(L)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def solve_lp_direction(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix efficiently using banded structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def subgradient_optimize(f0, time_limit, initial_lr=0.01, mom=0.9,
                         top_k=128, temp=0.001, lp_interval=200,
                         bb_blend=0.3):
    """
    Subgradient descent with Nesterov momentum, BB step size,
    and periodic LP-based global moves.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve_auto(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp

    # BB state
    s_prev = None
    y_prev = None
    f_prev = None
    g_prev = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)

        # BB step size
        if s_prev is not None and y_prev is not None and g_prev is not None and it > 1:
            s_diff = f - f_prev
            y_diff = sg - g_prev
            num = np.dot(s_diff, s_diff)
            den = np.dot(s_diff, y_diff)
            if den > 1e-14:
                bb_lr = num / den
                if 1e-6 < bb_lr < 2.0:
                    cur_lr = (1.0 - bb_blend) * cur_lr + bb_blend * bb_lr

        f_prev = f.copy()
        g_prev = sg.copy()

        # Nesterov step
        f_new = project_simplex(f - cur_lr * sg + mom * v)
        v = f_new - f
        s_prev = f_new - f
        y_prev = sg - g_prev
        f = f_new

        conv = fft_convolve_auto(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_temp *= 1.005
        cur_lr *= 0.9998

        # LP escape
        if lp_interval > 0 and no_improve >= lp_interval and rem > 30:
            g = solve_lp_direction(f, best_obj * 1.05, time_limit=min(15, rem // 4))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        ft = project_simplex((1 - alpha) * f + alpha * g)
                        c_try = fft_convolve_auto(ft).max()
                        if c_try < best_obj:
                            best_obj = c_try
                            best_f = ft.copy()
                            f = ft
                            conv = fft_convolve_auto(f)
                            break
            no_improve = 0
            cur_lr = initial_lr * 0.6
            cur_temp = max(0.0001, temp * 0.5)
            v = np.zeros(n)

        # Random restart
        if no_improve > 800:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fft_convolve_auto(f)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.5
            cur_temp = temp
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve_auto(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, M * 1.01, time_limit=min(20, rem // 3))
        if g is None:
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            M *= 1.01
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 20)
        for alpha in alphas:
            ft = project_simplex((1.0 - alpha) * f + alpha * g)
            val = fft_convolve_auto(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) / 2.0
                ft = project_simplex((1.0 - mid) * f + mid * g)
                val = fft_convolve_auto(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = project_simplex((1.0 - best_alpha) * f + best_alpha * g)
            conv = fft_convolve_auto(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            M *= 1.02

        if no_improve > 50:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fft_convolve_auto(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def fine_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=200, temp=1e-5):
    """Very fine subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve_auto(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        sg = compute_subgradient(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fft_convolve_auto(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.002

        if no_improve > 300:
            rng = np.random.default_rng(it + 200)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            conv = fft_convolve_auto(f)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_f, best_obj


def make_init_geometric(n, ratio, rng=None):
    """Geometric init: a[i] = ratio^i, normalized."""
    a = np.power(ratio, np.arange(n, dtype=np.float64))
    return a / a.sum()


def make_init_asymmetric_triangle(n, peak_frac, rng=None):
    """Asymmetric triangle: peaks at peak_frac * n."""
    x = np.arange(n, dtype=np.float64)
    peak = peak_frac * n
    a = np.maximum(0, 1 - np.abs(x - peak) / (n * 0.9))
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    return a / a.sum()


def make_init_blend(f1, f2, alpha, rng=None):
    """Blend two normalized sequences."""
    return project_simplex(alpha * f1 + (1 - alpha) * f2)


def generate_diverse_inits(n, parent=None, rng=None):
    """Generate diverse initializations for optimization."""
    if rng is None:
        rng = np.random.default_rng(12345)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Geometric (various ratios)
    for r in [0.97, 0.95, 0.9, 0.8]:
        inits.append(make_init_geometric(n, r))

    # Asymmetric triangles
    for pf in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        inits.append(make_init_asymmetric_triangle(n, pf))

    # Sine-based asymmetric
    for phase in [0.3, 0.5, 0.7]:
        s = np.sin(np.pi * (x / n + phase)) + 0.1
        s = np.maximum(s, 0)
        if s.sum() > 1e-10:
            inits.append(s / s.sum())

    # Power law asymmetric
    for p in [0.5, 1.0, 2.0]:
        pw = (n - x + 1) ** p
        inits.append(pw / pw.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Reversed
            inits.append(px[::-1] / ps)
            # Blend with geometric
            geo = make_init_geometric(n, 0.95)
            inits.append(project_simplex(0.5 * px / ps + 0.5 * geo))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining subgradient descent,
    LP Frank-Wolfe, and fine polishing with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick parent optimization ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(120, max(30, deadline - time.monotonic() - 1200))
        if t1 > 20:
            print(f"  SG on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = subgradient_optimize(parent, time_limit=t1,
                                              initial_lr=0.008, top_k=max(64, n_p // 6),
                                              temp=0.0005, lp_interval=150)
            try_seq(a_opt, "parent_sg")

    # === Phase 2: Broad exploration across lengths ===
    print("\n=== Phase 2: Broad Exploration ===")

    # Determine which lengths to try
    n_candidates = [384, 512, 768, 1024]
    if parent is not None:
        n_p = len(parent)
        if n_p not in n_candidates:
            n_candidates.insert(0, n_p)
        else:
            n_candidates.remove(n_p)
            n_candidates.insert(0, n_p)

    for n in n_candidates:
        if time.monotonic() > deadline - 300:
            break

        inits = generate_diverse_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 200
        t_per_init = max(10, rem // (len(inits) * max(1, len(n_candidates) - 1)))
        t_per_init = min(t_per_init, 45)

        print(f"  n={n}: {len(inits)} inits, ~{t_per_init:.0f}s each")

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 180:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 150)
            if t_alloc < 8:
                break

            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc,
                                              initial_lr=0.012,
                                              top_k=max(64, n // 8),
                                              temp=0.001,
                                              lp_interval=250)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best few ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # FW-LP deep runs (most powerful for global moves)
        n_fw_runs = 3
        for p in range(n_fw_runs):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(rem // (n_fw_runs - p), 90)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (best_seq[::-1]))

            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")

        # SG deep runs with different parameters
        n_sg_runs = 4
        for p in range(n_sg_runs):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem // (n_sg_runs - p), 70)

            if p == 0:
                f_init = best_seq.copy()
                cfg = {"initial_lr": 0.004, "top_k": max(100, n_best // 4), "temp": 0.0003, "lp_interval": 200}
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                cfg = {"initial_lr": 0.006, "top_k": max(80, n_best // 5), "temp": 0.0006, "lp_interval": 300}
            elif p == 2:
                f_init = project_simplex(0.85 * best_seq + 0.15 * make_init_geometric(n_best, 0.93))
                cfg = {"initial_lr": 0.005, "top_k": max(100, n_best // 4), "temp": 0.0004, "lp_interval": 250}
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
                cfg = {"initial_lr": 0.008, "top_k": max(64, n_best // 6), "temp": 0.001, "lp_interval": 150}

            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")

    # === Phase 4: Try different n from best structure ===
    print(f"\n=== Phase 4: Length Transfer ===")
    if best_seq is not None:
        # Try to transfer to nearby lengths
        for target_n in [best_seq.size + 64, best_seq.size - 64, best_seq.size + 128]:
            if target_n < 128 or target_n > 2048:
                continue
            if target_n == best_seq.size:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break

            # Interpolate best_seq to target_n
            x_new = np.arange(target_n, dtype=np.float64) * (len(best_seq) - 1) / (target_n - 1)
            f_interp = np.interp(x_new, np.arange(len(best_seq)), best_seq)
            f_interp = np.maximum(f_interp, 0)
            f_interp = f_interp / f_interp.sum()

            t_alloc = min(60, rem // 2)
            a_opt, obj = subgradient_optimize(f_interp, time_limit=t_alloc,
                                              initial_lr=0.008,
                                              top_k=max(64, target_n // 8),
                                              temp=0.0008,
                                              lp_interval=200)
            try_seq(a_opt, f"transfer_n{target_n}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG polish
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            a_opt, _ = fine_polish(best_seq, time_limit=min(rem, 60),
                                   lr=0.0008, mom=0.96,
                                   top_k=max(100, n_best // 3), temp=5e-6)
            try_seq(a_opt, "polish1")

        # Another round of FW-LP
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "polish_fw")

        # Micro perturbations + polish
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            for m in range(3):
                rem = deadline - time.monotonic() - 12
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0008)
                a_opt, _ = fine_polish(f_pert, time_limit=min(rem, 15),
                                       lr=0.0004, mom=0.97,
                                       top_k=max(100, n_best // 3), temp=1e-6)
                try_seq(a_opt, f"micro_{m}")

        # One more FW on the very best
        rem = deadline - time.monotonic() - 10
        if rem > 10 and best_seq is not None:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
