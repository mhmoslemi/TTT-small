            # Shape: reduce high AC bins, boost low AC bins
            P_new = P.copy()
            P_new[0] = dc_power
            for k in range(1, len(P)):
                ratio = P[k] / ac_mean
                # Gently push toward 1
                P_new[k] = P[k] - alpha * (P[k] - ac_mean)
                P_new[k] = max(P_new[k], 0)

            # Reconstruct: keep phase of A, use new amplitude
            phase = np.angle(A)
            A_new = np.sqrt(P_new) * np.exp(1j * phase)

            # Inverse DFT
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a = a_new / s_new

    return a


def optimize_gradient(a0, time_limit, lr_init=0.01, momentum=0.92,
                      top_k=50, temp_init=0.005, use_spectral=False):
    """Nesterov gradient descent with adaptive parameters and optional spectral shaping."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)

        # Periodic spectral shaping
        if use_spectral and iter_count % 200 == 0 and iter_count > 0:
            y_shaped = spectral_shaping(y, n_iters=10, alpha=0.3)
            # Blend: 50% shaped
            y = 0.5 * y + 0.5 * y_shaped
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            conv_y = fast_convolve(y)

        # Compute gradient
        grad = compute_gradient_soft(y, conv_y, top_k=top_k, temp=temp)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates
        lr *= 0.9997
        temp *= 1.004
        mom = min(0.98, mom + 0.0001)

        iter_count += 1
        if no_improve >= 400:
            break

    return best_a, best_obj


def optimize_lp(a0, time_limit, max_no_improve=60):
    """LP-based optimization with perturbation restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        C = build_conv_matrix(a)

        c = -np.ones(n)
        b = np.full(2 * n - 1, M * 1.00003)
        bounds = [(0, None)] * n

        success = False
        remaining_lp = time_limit - (time.monotonic() - t0)
        if remaining_lp < 2:
            break

        try:
            lp_time = max(0.5, min(remaining_lp * 0.3, 30.0))
            result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': lp_time})
            if result.success:
                g = result.x
                g_sum = g.sum()
                if g_sum > a.sum() * (1 + 1e-7):
                    d = g - a
                    # Try step sizes from large to small
                    for t in [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        conv_try = fast_convolve(a_try)
                        obj_try = 2.0 * n * conv_try.max()
                        if obj_try < best_obj - 1e-10:
                            a = a_try
                            conv = conv_try
                            best_obj = obj_try
                            best_a = a.copy()
                            no_improve = 0
                            success = True
                            break
        except Exception:
            pass

        iteration += 1
        if not success:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Perturb with structured moves
                perturb_count += 1
                scale = 0.003 + 0.002 * min(perturb_count, 15)

                # Mix of random and structured perturbation
                a = best_a.copy()
                move_type = rng.integers(0, 3)
                if move_type == 0:
                    # Random noise
                    noise = rng.standard_normal(n) * scale
                    a += noise
                elif move_type == 1:
                    # Shift move: move mass between two random positions
                    for _ in range(max(1, n // 50)):
                        i = rng.integers(0, n)
                        j = rng.integers(0, n)
                        if i != j:
                            amt = a[i] * scale * 2
                            a[i] -= amt
                            a[j] += amt
                else:
                    # Block blend: smooth a random block
                    block_len = max(2, n // 20)
                    start = rng.integers(0, max(1, n - block_len))
                    end = start + block_len
                    if end <= n:
                        block = a[start:end]
                        kernel = np.ones(block_len) / block_len
                        smoothed = np.convolve(block, kernel, mode='same')
                        a[start:end] = (1 - scale * 5) * block + (scale * 5) * smoothed

                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                else:
                    a = a / s
                conv = fast_convolve(a)
                no_improve = 0
                max_no_improve = min(200, 60 + 20 * perturb_count)

                if perturb_count > 60:
                    break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, rng, n_iterations=None):
    """
    Simulated annealing with structured moves.
    Moves: shift mass, mirror segment, blend with smoothed version.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()

    t0 = time.monotonic()
    if n_iterations is None:
        n_iterations = 2000

    temp_start = 0.01
    temp_end = 0.0001
    cooling = (temp_end / temp_start) ** (1.0 / max(1, n_iterations))

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break

        temp = temp_start * (cooling ** it)
        a_new = a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Shift: move small amount from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = a[i] * rng.uniform(0.01, 0.1)
                a_new[i] -= amt
                a_new[j] += amt
        elif move_type == 1:
            # Mirror a segment
            seg_len = rng.integers(max(2, n // 20), max(3, n // 5))
            start = rng.integers(0, max(1, n - seg_len))
            end = min(start + seg_len, n)
            seg = a_new[start:end].copy()
            a_new[start:end] = seg[::-1]
        elif move_type == 2:
            # Blend with smoothed version in a window
            win = rng.integers(max(3, n // 10), max(4, n // 3))
            start = rng.integers(0, max(1, n - win))
            end = min(start + win, n)
            block = a_new[start:end]
            kernel = np.ones(win) / win
            smoothed = np.convolve(block, kernel, mode='same')
            blend = rng.uniform(0.1, 0.5)
            a_new[start:end] = (1 - blend) * block + blend * smoothed
        else:
            # Gaussian perturbation
            noise = rng.standard_normal(n) * temp * 0.5
            a_new += noise

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        delta = obj_new - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Sine-squared (periodic structure)
    for freq in [1, 2, 3]:
        key = f'sine{freq}'
        v = (np.sin(np.pi * freq * x / n) ** 2 + 0.01)
        inits[key] = v / v.sum()

    # Cosine window (concentrated at center)
    cw = np.cos(np.pi * x / (2 * n)) ** 2
    inits['cos_window'] = cw / cw.sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)

    # Random periodic
    period = rng.integers(max(2, n // 4), max(3, n // 2))
    base = rng.uniform(0.5, 1.5, size=period)
    periodic = np.tile(base, n // period + 1)[:n]
    inits['periodic'] = periodic / periodic.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid optimization combining gradient,
    spectral shaping, LP, and simulated annealing across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick scan across multiple lengths and initializations
    # =========================================================
    print("=== Phase 1: Multi-length Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = min(300.0, (deadline - time.monotonic()) * 0.2)

    # Lengths to try - include larger sizes
    lengths = [384, 512, 768, 1024, 1536]
    # Filter: if parent exists, prioritize its length and nearby
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(min(2, len(lengths)), pn)
        lengths = sorted(set(lengths))[:6]

    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 900:
            break

        inits = generate_initializations(n, parent, rng)
        # Prioritize diverse inits
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'cos_window',
                    'bimodal', 'random', 'alternating', 'gaussian', 'sine2']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        # Limit to top 3 per length for speed
        init_names = init_names[:3]

        t_per = max(8.0, (phase1_limit - 5) / (len(lengths) * 3))

        for name in init_names:
            if time.monotonic() > deadline - 800:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 600)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj = optimize_gradient(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.92,
                                           top_k=40, temp_init=0.004,
                                           use_spectral=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan n={n}, {name}: {score:.6f} *BEST*")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Pick top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    # Always include best overall
    if best_sequence is not None:
        blen = len(best_sequence)
        if blen not in seen_lengths:
            top_configs.append((blen, best_sequence.copy()))

    remaining = deadline - time.monotonic() - 400.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(30.0, remaining * 0.55 / n_cands)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 250)
        if t_alloc < 15:
            break

        print(f"  LP cand {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = optimize_lp(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Spectral shaping + gradient on best
    # =========================================================
    print("=== Phase 3: Spectral + Gradient Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 300.0
        if remaining > 50:
            # Multiple spectral shaping passes with different parameters
            for si in range(3):
                if time.monotonic() > deadline - 250:
                    break
                t_alloc = min(remaining / (3 - si) * 0.7, deadline - time.monotonic() - 200)
                if t_alloc < 10:
                    break

                # Apply spectral shaping first
                a_shaped = spectral_shaping(best_sequence, n_iters=30 + 20 * si,
                                            alpha=0.3 + 0.1 * si)
                # Then gradient polish
                a_opt, obj = optimize_gradient(a_shaped, time_limit=t_alloc,
                                               lr_init=0.008, momentum=0.93,
                                               top_k=60, temp_init=0.002,
                                               use_spectral=(si > 0))
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Spectral+Grad {si}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  Spectral+Grad {si}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: LP polish (more iterations, different restarts)
    # =========================================================
    print("=== Phase 4: LP Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 40:
            for pi in range(3):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(remaining / (3 - pi) * 0.75, deadline - time.monotonic() - 120)
                if t_alloc < 15:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                elif pi == 1:
                    noise = rng.standard_normal(len(best_sequence)) * 0.002
                    a0 = np.maximum(best_sequence + noise, 0)
                    a0 /= a0.sum()
                else:
                    a0 = spectral_shaping(best_sequence, n_iters=15, alpha=0.4)

                a_opt, obj = optimize_lp(a0, time_limit=t_alloc, max_no_improve=80)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP Polish {pi}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  LP Polish {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Simulated annealing with structured moves
    # =========================================================
    print("=== Phase 5: Simulated Annealing ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            for ai in range(3):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(remaining / (3 - ai) * 0.7, deadline - time.monotonic() - 60)
                if t_alloc < 10:
                    break

                rng_sa = np.random.default_rng(seed + 1000 + ai * 7)
                a_init = best_sequence.copy()
                if ai > 0:
                    noise = rng_sa.standard_normal(len(a_init)) * 0.003 * ai
                    a_init = np.maximum(a_init + noise, 0)
                    a_init /= a_init.sum()

                a_opt, obj = simulated_annealing(a_init, time_limit=t_alloc, rng=rng_sa)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  SA {ai}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  SA {ai}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Final combined polish (gradient + short LP)
    # =========================================================
    print("=== Phase 6: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0

        # Short gradient
        t_g = min(remaining * 0.4, 60.0)
        if t_g > 5:
            a_opt, obj = optimize_gradient(best_sequence, time_limit=t_g,
                                           lr_init=0.004, momentum=0.94,
                                           top_k=80, temp_init=0.001)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final Grad: {score:.6f} *NEW BEST*")

        # Very short LP
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            a_opt, obj = optimize_lp(best_sequence, time_limit=remaining, max_no_improve=30)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
