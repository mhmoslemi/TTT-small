def project_simplex(v):
    """Euclidean projection onto the probability simplex."""

        return np.array([max(v[0], 0.0)])

        g_sum = g.sum()
        if g_sum < 1e-15:

        # Line search
        best_alpha = 0.0
        best_val = fast_convolve(f).max()
        
        # Coarse search

            if val < best_val:

                best_alpha = alpha

            lo = max(0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):

                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0)

                if s < 1e-15:

                ft = ft / s
                val = fast_convolve(ft).max()
                if val < best_val:

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = f / f.sum()
            conv = fast_convolve(f)

                no_improve = 0

Now let me think about the LSE phase more carefully. The key is the beta schedule and learning rate.

The LSE-smoothed objective is:
`L(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))`

As beta → ∞, L(a) → max_k (a*a)[k].

The gradient is: `∇L(a) = sum_k w_k * ∇((a*a)[k])` where `w_k = exp(beta*(a*a)[k]) / sum_j exp(beta*(a*a)[j])`.

For large beta, `w_k` concentrates on the argmax, and the gradient approximates the subgradient of the max.

The Hessian (approximately, for large beta) is: `H ≈ sum_k w_k * 2*M_k + beta * sum_{k,l} w_k w_l * (∇_k)(∇_l)^T` where `M_k` is the matrix for the quadratic form `(a*a)[k]`.

For the CG-based Newton step, I'll use the Gauss-Newton approximation: `H_GN ≈ J^T J` where `J[k,j] = 2*f[k-j]` (Jacobian of the vector-valued function `g(a) = [(a*a)[0], ..., (a*a)[2n-2]]`).

`J^T J` is a Toeplitz matrix: `(J^T J)[i,j] = 4 * sum_k f[k-i] * f[k-j]` (with valid indices).

For a Toeplitz matrix, matrix-vector products can be done in O(n log n) using FFT.

def solve_cg_newton(f, grad, max_iter=50):
    """Solve H @ delta = -grad using CG, where H = J^T J (Gauss-Newton).
    H is Toeplitz with H[i,j] = 4*r[i-j]."""
    n = len(f)
    
    # Compute auto-correlation of f
    size = 2 * n - 1
    fft_sz = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_sz)
    acf = np.fft.irfft(F * F, fft_sz)[:size]
    r = acf[n-1:n-1+n]  # r[d] = sum_k f[k] * f[k+d]
    
    # Build H (n x n Toeplitz)
    # H[i,j] = 4 * r[|i-j|]
    idx = np.abs(np.arange(n)[:, None] - np.arange(n)[None, :])
    H = 4.0 * r[idx]
    
    # CG to solve H @ delta = -grad
    b = -grad
    x = np.zeros(n)
    r_vec = b - H @ x  # = b
    p = r_vec.copy()
    r_choose_r = r_vec @ r_vec
    
    for _ in range(max_iter):
        Hp = H @ p
        p_choose_Hp = p @ Hp
        if p_choose_Hp < 1e-15:

        alpha = r_choose_r / p_choose_Hp

        r_vec -= alpha * Hp
        r_new = r_vec @ r_vec
        if np.sqrt(r_new) < 1e-10 * np.sqrt(r_choose_r):
            break
        p = r_vec + (r_new / r_choose_r) * p
        r_choose_r = r_new
    
    return x

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses:
1. FW with correct LP oracle (min t s.t. A@g<=t, sum(g)=1, g>=0)
2. LSE gradient descent with adaptive beta and momentum
3. Spectral shaping (GS-like amplitude manipulation)
4. CG-based quasi-Newton polishing
5. Multi-length exploration with interpolation

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    n = len(f)
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_A_matrix(f):
    """Build A (m x n) where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi+1] = f[k - np.arange(j_lo, j_hi+1)]
    return A


def solve_lp_oracle(f):
    """Solve: min t s.t. A@g <= t*1, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(f)
    m = 2 * n - 1
    A = build_A_matrix(f)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def phase_lse(f0, time_limit, beta_start=10.0, beta_end=5000.0):
    """Phase 2: LSE gradient descent with adaptive parameters."""
    f = f0.copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 >= time_limit:
            break
        
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        grad = compute_lse_grad(w, f)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            no_improve += 100
            break
        grad = grad / gn

        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0/n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.9
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0/n, 0.001)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.005
        
        if it % 500 == 0:
            print(f"    LSE: it={it}, obj={best_obj:.10f}, beta={beta:.0f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def phase_spectral(f0, time_limit, lambda_start=0.5):
