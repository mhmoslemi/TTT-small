def fast_convolve(a):

def build_conv_matrix(a):

    n = len(a)

def compute_multi_peak_gradient(a, conv, top_k=10, temp=0.005):
    """
    Compute gradient of 2*n*logsumexp(conv[top_k]/temp) for the top-k peaks.
    Returns (smooth_obj, gradient) where gradient is d(smooth_obj)/d(a) projected
    to the simplex tangent space.
    """
    n = len(a)
    # Get top-k peak indices
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peak_indices]

    # Soft weights over top-k peaks
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(top_k)
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Smooth objective: 2*n*temp*log(sum(exp(peak_vals/temp)))
    lse = peak_vals.max() + temp * np.log(w_sum)
    smooth_obj = 2.0 * n * lse

    # Gradient: d(smooth_obj)/d(a_i) = 2*n * sum_k w[k] * 2 * a[peak_k - i] (valid)
    grad = np.zeros(n)
    for idx in range(top_k):

        k = int(peak_indices[idx])

        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    grad *= 4.0 * n

    # Project to simplex tangent space (subtract mean)
    grad = grad - grad.mean()

    return smooth_obj, grad

            if high_mask.sum() > 0:
                a[idxs[high_mask]] *= (1 - alpha * frac[high_mask])

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            conv = fast_convolve(a)
            continue
        a = a / s

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert if no improvement for a while
            if no_improve > 50:
                a = best_a.copy()
                conv = fast_convolve(a)
                no_improve = 0
                break

        iteration += 1
        if iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq iter {iteration}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a, best_obj


def lp_optimize(a0, time_limit, seed=42):
    """
    LP-based optimization with multiple constraint levels and backtracking.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Try multiple constraint levels
        found_improvement = False
        for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:
            M_target = M * (1 + delta)
            C = build_conv_matrix(a)
            c_obj = -np.ones(n)
            b_vec = np.full(2 * n - 1, M_target)
            bounds = [(0, None)] * n

            try:
                result = linprog(c_obj, A_ub=C, b_ub=b_vec, bounds=bounds,
                                 method='highs', options={'time_limit': 5.0})
                if not result.success:
                    continue
                g = result.x
                g_sum = g.sum()
                if g_sum <= s * (1 + 1e-6):
                    continue
            except Exception:
                continue

            d = g - a
            # Line search
            best_t = 0
            best_try_obj = 2.0 * n * M
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_try = a + t * d
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_try_obj - 1e-12:
                    best_try_obj = obj_try
                    best_t = t
                    break

            if best_t > 0:
                a = a + best_t * d
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                no_improve = 0
                found_improvement = True
                break  # Move to next iteration

        if not found_improvement:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                scale = 0.002 + 0.003 * min(no_improve // 30, 5)
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s_r = a.sum()
                if s_r > 1e-15:
                    a = a / s_r
                else:
                    a = best_a.copy()
                no_improve = 0

                if no_improve > 200:
                    break

        iteration += 1
        if iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a, best_obj


def generate_variants(n, parent, rng):
    """Generate diverse initializations for length n."""
    variants = []
    x = np.arange(n, dtype=np.float64)

    # Parent interpolation
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                variants.append(('parent', p / ps))
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    variants.append(('parent_interp', pp / pps))

    # Uniform
    variants.append(('uniform', np.ones(n) / n))

    # Gaussian (varying widths)
    for width_frac in [0.2, 0.3, 0.4]:
        sigma = width_frac * n
        g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
        gs = g.sum()
        if gs > 1e-10:
            variants.append((f'gauss_{width_frac}', g / gs))

    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cws = cw.sum()
    if cws > 1e-10:
        variants.append(('cosine', cw / cws))

    # Raised cosine squared
    rcs = (0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))) ** 2
    rcsv = rcs.sum()
    if rcsv > 1e-10:
        variants.append(('cosine2', rcs / rcsv))

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    variants.append(('random', r))

    return variants


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid optimization combining Adam log-sum-exp gradient,
    peak equalization, and LP polishing with adaptive scheduling.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
            print(f"Parent: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass

    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Initialized uniform n=256, score={best_score:.6f}")

    mid = time.monotonic()
    total_elapsed = time.monotonic() - mid
    time_remaining = float(budget_s) - 30.0 - total_elapsed
    print(f"Time remaining: {time_remaining:.0f}s")

    # =========================================================
    # Phase 1: Multi-start Adam gradient (50% of time)
    # =========================================================
    print("=== Phase 1: Adam LogSumExp Gradient ===")
    t_phase1 = time_remaining * 0.50
    n_main = len(best_sequence)

    # Try parent and a few variants at the main length
    t_per_start = t_phase1 / 6.0
    starts = [
        ('parent', best_sequence.copy()),
    ]

    # Add a couple of different-shape starts
    if parent is not None:
        variants = generate_variants(n_main, parent, rng)
        for name, v in variants:
            if name in ['gauss_0.3', 'cosine']:
                starts.append((name, v))
                break

    # Add a random start
    starts.append(('random', rng.dirichlet(np.ones(n_main) * 0.5)))

    for i, (name, a0) in enumerate(starts):
        t_left = deadline - time.monotonic()
        if t_left < 10:
            break
        t_alloc = min(t_per_start, t_left - 30)
        if t_alloc < 10:
            break

        print(f"  Start {i} ({name}), t_alloc={t_alloc:.0f}s")
        a_opt, obj = adam_logsumexp_optimize(
            a0, time_limit=t_alloc, lr=0.004, temp_init=0.01,
            top_k=10, seed=seed + i * 100
        )
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Start {i} ({name}): {score:.6f} *** NEW BEST ***")
        else:
            print(f"  Start {i} ({name}): obj={obj:.6f}, eval={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Peak Equalization (20% of time)
    # =========================================================
    print("=== Phase 2: Peak Equalization ===")
    t_left = deadline - time.monotonic()
    t_phase2 = min(t_left * 0.25, 120.0)
    if t_phase2 > 10:
        a_opt, obj = peak_equalization(
            best_sequence, time_limit=t_phase2, n_peaks=5,
            alpha=0.03, seed=seed + 500
        )
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  PeakEq: {score:.6f} *** NEW BEST ***")
        else:
            print(f"  PeakEq: obj={obj:.6f}, eval={score:.6f}")

    # =========================================================
    # Phase 3: LP Optimization (25% of time)
    # =========================================================
    print("=== Phase 3: LP Optimization ===")
    t_left = deadline - time.monotonic()
    t_phase3 = min(t_left * 0.40, 400.0)
    if t_phase3 > 15:
        a_opt, obj = lp_optimize(
            best_sequence, time_limit=t_phase3, seed=seed + 900
        )
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP: {score:.6f} *** NEW BEST ***")
        else:
            print(f"  LP: obj={obj:.6f}, eval={score:.6f}")

    # =========================================================
    # Phase 4: Fine gradient polish (remaining time)
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    t_left = deadline - time.monotonic()
    if t_left > 15:
        # Fine gradient with low temperature
        t_fine = min(t_left * 0.6, 120.0)
        if t_fine > 10:
            a_opt, obj = adam_logsumexp_optimize(
                best_sequence, time_limit=t_fine, lr=0.002,
                temp_init=0.003, top_k=5, seed=seed + 2000
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Fine Grad: {score:.6f} *** NEW BEST ***")

        # Final LP micro-polish
        t_left = deadline - time.monotonic()
        if t_left > 10:
            a_opt, obj = lp_optimize(
                best_sequence, time_limit=min(t_left - 5, 60.0), seed=seed + 3000
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *** NEW BEST ***")

    # =========================================================
    # Phase 5: Try alternative lengths briefly
    # =========================================================
    t_left = deadline - time.monotonic()
    if t_left > 30:
        print("=== Phase 5: Alternative Lengths ===")
        n_main = len(best_sequence)
        alt_lengths = [n_main - 64, n_main + 64]
        alt_lengths = [l for l in alt_lengths if 128 <= l <= 2048 and l != n_main]

        for alt_n in alt_lengths:
            t_left = deadline - time.monotonic()
            if t_left < 20:
                break
            t_alt = min(t_left * 0.4, 60.0)
            if t_alt < 15:
                continue

            # Interpolate parent to new length
            p = np.maximum(best_sequence, 0).astype(np.float64)
            ps = p.sum()
            p_norm = p / ps
            a_init = np.interp(np.linspace(0, 1, alt_n),
                               np.linspace(0, 1, len(p_norm)), p_norm)
            a_init = np.maximum(a_init, 0)
            ais = a_init.sum()
            if ais < 1e-10:
                continue
            a_init = a_init / ais

            a_opt, obj = adam_logsumexp_optimize(
                a_init, time_limit=t_alt, lr=0.005,
                temp_init=0.01, top_k=8, seed=seed + alt_n
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Alt n={alt_n}: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  Alt n={alt_n}: obj={obj:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
