"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast LP-based Frank-Wolfe steps, Nesterov-accelerated subgradient,
evolutionary refinement, and adaptive perturbation. Explores multiple lengths
and uses structured crossover/mutation to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def obj_val(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_direction(a, M, C):
    """Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def nesterov_sg_step(a, v, conv, lr, momentum, top_k, temp, n):
    """Nesterov-accelerated subgradient step with softmax-weighted peaks."""
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    # Nesterov update
    y = a + momentum * v
    v_new = momentum * v - lr * sg
    a_new = y + v_new
    a_new = normalize(a_new)
    return a_new, v_new


def line_search(a, d, current_obj, n, max_evals=14):
    """Adaptive line search: coarse scan + golden section."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0

    candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_val(a_try, n)
        evals += 1
        if obj_try < best_obj - 1e-15:
            best_obj = obj_try
            best_t = t

    # Golden section refinement
    if 0.03 < best_t < 0.97 and evals < max_evals - 4:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.01, best_t - 0.1)
        hi = min(0.99, best_t + 0.1)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)

        for _ in range(3):
            if evals >= max_evals:
                break
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            obj1 = obj_val(a1 / s1, n)[0] if s1 > 1e-15 else current_obj
            evals += 1

            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            obj2 = obj_val(a2 / s2, n)[0] if s2 > 1e-15 else current_obj
            evals += 1

            if obj1 < obj2:
                hi, d2, obj2 = d2, d1, obj1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo, d1, obj1 = d1, d2, obj2
                d2 = lo + gr * (hi - lo)
            if abs(hi - lo) < 0.005:
                break

        mid = (lo + hi) / 2.0
        a_mid = np.maximum(a + mid * d, 0.0)
        s_mid = a_mid.sum()
        if s_mid > 1e-15:
            a_mid = a_mid / s_mid
            obj_mid, _ = obj_val(a_mid, n)
            evals += 1
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = mid

    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False):

    n = len(a)
    _, conv = obj_val(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M

    it = 0
    no_improve = 0
    perturb_count = 0
    tightness = 0.9995

        M_lp = M * tightness
        g, g_sum = lp_direction(a, M_lp, C)

        else:

            else:

                    _, conv = obj_val(a, n)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.999, tightness - 0.0003)
                    if verbose and it % 30 == 0:
                        print(f"      LP it={it} obj={best_obj:.8f} t={time.monotonic()-t0:.0f}s")

            no_improve = 0
            perturb_count += 1
            tightness = 1.001

def nesterov_optimize(a0, time_limit, lr_init=0.008, momentum=0.93, top_k=80, temp=0.0002):
    """Nesterov-accelerated subgradient optimization."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    v = np.zeros(n)
    _, conv = obj_val(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        a, v = nesterov_sg_step(a, v, conv, lr, momentum, top_k, temp, n)
        _, conv = obj_val(a, n)
        cur_obj = 2.0 * n * conv.max()
        step += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 40 == 0:
            lr *= 0.997
            temp *= 1.003

        if no_improve >= 350:
            rng = np.random.default_rng(step * 5 + 3)
            a = normalize(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            _, conv = obj_val(a, n)
            no_improve = 0
            lr = lr_init * 0.8

    return best_a, best_obj


def resample(seq, new_n):
    """Resample to different length via interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return normalize(np.interp(x_new, x_old, seq))


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = resample(parent, n)

    # Various structured patterns
    inits['sine1'] = normalize(0.7 + 0.3 * np.sin(2 * np.pi * x / n))
    inits['sine2'] = normalize(0.6 + 0.4 * np.cos(4 * np.pi * x / n))
    inits['sine3'] = normalize(0.7 + 0.3 * np.cos(6 * np.pi * x / n))
    # Bimodal (mass at edges)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()
    # Anti-tri (mass in center)
    anti = n - np.minimum(x, n - 1 - x)
    inits['anti'] = normalize(anti)
    # Gaussian
    inits['gauss'] = normalize(np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2))
    # Random
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.5)
    # Two blocks
    block = np.ones(n) * 0.3
    block[:n // 4] = 1.0
    block[3 * n // 4:] = 1.0
    inits['blocks'] = normalize(block)

    return inits


def evolutionary_phase(pop, time_limit, n, rng):
    """Evolutionary refinement: crossover + mutation + selection."""
    t0 = time.monotonic()
    pop_size = len(pop)
    gen = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        # Generate offspring
        offspring = []
        n_offspring = max(3, pop_size)
        gen_deadline = time.monotonic() + min(30.0, time_limit / max(1, pop_size))

        for oi in range(n_offspring):
            if time.monotonic() > gen_deadline:
                break
            # Crossover: interpolate between two random parents
            i1, i2 = rng.integers(0, pop_size), rng.integers(0, pop_size)
            alpha = rng.uniform(0.3, 0.7)
            if i1 == i2:
                child = pop[i1][0].copy()
            else:
                child = alpha * pop[i1][0] + (1 - alpha) * pop[i2][0]

            # Mutation
            mut_scale = 0.002 + 0.002 * (gen % 10) / 10
            child = child + rng.standard_normal(n) * mut_scale
            # Occasional larger structured perturbation
            if gen % 5 == 0:
                scale = 0.005
                smooth = np.sin(np.pi * np.arange(n) / n) * scale
                child = child + smooth
            child = normalize(child)
            offspring.append((child, float('inf')))

        # Evaluate offspring with short optimization
        for idx in range(len(offspring)):
            if time.monotonic() > time_limit + t0 - 5:
                break
            child = offspring[idx][0]
            t_alloc = min(8.0, time_limit + t0 - time.monotonic() - 5)
            if t_alloc < 3:
                break

            # Quick SG burst
            a_opt, obj_opt = nesterov_optimize(child, time_limit=t_alloc,
                                                lr_init=0.005, momentum=0.94,
                                                top_k=60, temp=0.0003)
            offspring[idx] = (a_opt, obj_opt)

        # Selection: merge population and keep best
        all_candidates = [(s, a) for a, s in pop] + [(s, a) for a, s in offspring if s < float('inf')]
        all_candidates.sort(key=lambda x: x[0])
        pop = [(a, s) for s, a in all_candidates[:pop_size]]

        gen += 1

    return pop[0][0], pop[0][1]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP, Nesterov SG, and evolutionary search.
    Explores multiple sequence lengths and uses structured perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score target improvement needed")
    except Exception:
        pass

    # ===== Phase 1: Multi-length LP Scout =====
    print("=== Phase 1: Multi-length LP Scout ===")
    t_start = time.monotonic()
    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.25)

    lengths = [768, 1024]
    if parent is not None and 512 not in lengths:
        lengths.insert(0, 512)
    if (deadline - time.monotonic()) > 1200:
        lengths.append(1536)

    results = []
    init_priority = ['parent', 'uniform', 'sine1', 'blocks', 'tri', 'gauss']

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_inits(n, parent, rng)
        chosen = [nm for nm in init_priority if nm in inits][:4]

        t_per = max(20.0, phase1_budget / (len(lengths) * 4))

        for name in chosen:
            if time.monotonic() > deadline - 380:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 370)
            if t_alloc < 10:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n} {name}: {score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep LP + Nesterov alternating =====
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates from different lengths
    top_cands = []
    seen_lens = set()
    for score, n, a in results:
        if n not in seen_lens:
            top_cands.append((n, a.copy()))
            seen_lens.add(n)
        if len(top_cands) >= 3:
            break
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lens:
            top_cands.append((bn, best_sequence.copy()))

    remaining_p2 = deadline - time.monotonic() - 300.0
    if remaining_p2 < 60:
        remaining_p2 = 60
    t_per_deep = max(25.0, remaining_p2 / max(1, len(top_cands)))

    for ci, (n, a0) in enumerate(top_cands):
        if time.monotonic() > deadline - 280:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 270)
        if t_alloc < 15:
            break

        # Alternate LP and Nesterov
        a_cur = a0.copy()
        half = t_alloc / 2

        a_lp, _ = lp_optimize(a_cur, time_limit=half)
        a_ng, _ = nesterov_optimize(a_lp, time_limit=max(5.0, t_alloc - half - 5),
                                     lr_init=0.006, momentum=0.94, top_k=70, temp=0.00015)
        a_final, _ = lp_optimize(a_ng, time_limit=max(5.0, t_alloc - half - max(5.0, t_alloc - half - 5)))

        score = float(evaluate_sequence(a_final.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_final.copy()
            print(f"  Phase2 cand{ci} n={n}: {score:.6f} (t={time.monotonic()-t_start2:.0f}s)")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Evolutionary Refinement =====
    print("=== Phase 3: Evolutionary Refinement ===")
    t_start3 = time.monotonic()
    remaining = deadline - time.monotonic() - 200.0
    if remaining > 60 and best_sequence is not None:
        n_best = len(best_sequence)
        # Build initial population
        pop = [(best_sequence.copy(), best_score)]
        # Add varied members
        for pi in range(4):
            if time.monotonic() > deadline - 190:
                break
            if pi < len(results) - 1:
                pop.append((results[pi + 1][2].copy(), results[pi + 1][0]))
            else:
                a_var = normalize(best_sequence + rng.standard_normal(n_best) * 0.005)
                a_opt, obj_opt = nesterov_optimize(a_var, time_limit=10,
                                                    lr_init=0.005, momentum=0.94,
                                                    top_k=60, temp=0.0003)
                pop.append((a_opt, obj_opt))

        pop_ev, score_ev = evolutionary_phase(pop, time_limit=remaining, n=n_best, rng=rng)
        if score_ev < best_score:
            best_score = score_ev
            best_sequence = pop_ev.copy()
            print(f"  Phase3 evolution: {score_ev:.6f} (t={time.monotonic()-t_start3:.0f}s)")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Intensive Polish =====
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            # Intensive LP + SG alternation
            a_cur = best_sequence.copy()
            n_best = len(a_cur)

            # Multiple short bursts with different hyperparams
            configs = [
                {'type': 'lp', 't': remaining * 0.3},
                {'type': 'sg', 't': remaining * 0.25, 'lr': 0.004, 'mom': 0.95, 'tk': 100, 'tp': 0.0001},
                {'type': 'lp', 't': remaining * 0.2},
                {'type': 'sg', 't': remaining * 0.15, 'lr': 0.003, 'mom': 0.96, 'tk': 120, 'tp': 0.00008},
                {'type': 'lp', 't': remaining * 0.1},
            ]

            for cfg in configs:
                if time.monotonic() > deadline - 8:
                    break
                t_alloc = min(cfg['t'], deadline - time.monotonic() - 5)
                if t_alloc < 5:
                    break
                if cfg['type'] == 'lp':
                    a_cur, _ = lp_optimize(a_cur, time_limit=t_alloc)
                else:
                    a_cur, _ = nesterov_optimize(a_cur, time_limit=t_alloc,
                                                  lr_init=cfg['lr'], momentum=cfg['mom'],
                                                  top_k=cfg['tk'], temp=cfg['tp'])
                score = float(evaluate_sequence(a_cur.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cur.copy()
                    print(f"  Phase4 {'LP' if cfg['type']=='lp' else 'SG'}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 5: Try different length if time remains =====
    print("=== Phase 5: Cross-length check ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_cur = len(best_sequence)
        # Try a slightly different length
        n_alt = n_cur + 128 if n_cur <= 1024 else n_cur - 128
        n_alt = max(256, min(2048, n_alt))
        if n_alt != n_cur:
            a_alt = normalize(best_sequence + rng.standard_normal(n_alt) * 0.01)
            t_alloc = min(40.0, deadline - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, _ = lp_optimize(a_alt, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase5 n={n_alt}: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
