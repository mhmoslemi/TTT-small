```python
from scipy.sparse import lil_matrix

def solve_lp_direction_sparse(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    # Build sparse constraint matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def project_simplex(v):
```
```python
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
```

```python
def subgradient_at_peaks(f, conv, top_k=30, temp=0.001):
    """Compute weighted subgradient at peak lags."""
    n = len(f)

    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            cur_obj = 2.0 * n * M
            m_relax = 1.0
            f = f_new

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.1, 3.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 137 + 31)
            scale = 0.003 + 0.003 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj / (2.0 * n)


def sg_polish(f0, time_limit, lr=0.005, mom=0.9, top_k=50, temp=0.0001, bb=False):
    """Subgradient polishing with optional Barzilai-Borwein step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    # BB variables
    prev_f = f.copy()
    prev_sg = None
    bb_lr = lr

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        sg = subgradient_at_peaks(f, conv, top_k=top_k, temp=cur_temp)

        if bb and prev_sg is not None:
            sf = f - prev_f
            y = sg - prev_sg
            sy = np.dot(s, y)
            ss = np.dot(sf, sf)
            yy = np.dot(y, y)
            if sy > 1e-14 and yy > 1e-14 and ss > 1e-14:
                bb_lr = max(0.0001, min(0.1, sy / yy))
                bb_lr = max(0.0001, min(0.1, ss / sy))

        step_lr = bb_lr if bb else cur_lr
        v = mom * v - step_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        cur_obj = 2.0 * n * co

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if bb:
            prev_f = f.copy()
            prev_sg = sg.copy()
        else:
            cur_lr *= 0.9995
            cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            if not bb:
                cur_lr = max(lr * 0.5, 0.001)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj / (2.0 * n)


def pairwise_search(f0, time_limit, verbose=False):
    """
    Pairwise swap local search: try moving mass from coordinate i to j.
    Uses exact line search for each swap.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    # Precompute: for each coordinate i, which lags does it affect?
    # Changing f[i] affects conv[k] for k in [i, i+n-1] (clamped to [0, 2n-2])

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Identify coordinates with high subgradient (contribute to peaks)
        sg = subgradient_at_peaks(f, conv, top_k=min(20, n // 3), temp=0.0001)

        # Sort by subgradient: high sg = "bad" coordinates, low sg = "good"
        order = np.argsort(sg)
        bad_coords = order[-min(10, n // 4):][::-1]  # highest sg
        good_coords = order[:min(10, n // 4)]  # lowest sg

        improved = False
        for i in bad_coords:
            if time.monotonic() - t0 > time_limit - 1:
                break
            if f[i] < 1e-8:
                continue

            for j in good_coords:
                if i == j:
                    continue
                if f[i] < 1e-8:
                    continue

                # Compute max delta for swap i -> j
                # conv'_k = conv_k + delta*(f[k-j] - f[k-i]) approximately
                # Need conv'_k <= M for all k, and f[i]-delta >= 0
                delta_max = f[i] * 0.95  # can't move more than we have

                # Check constraints at affected lags
                # i affects k in [max(0, i), min(2n-2, i+n-1)]
                # j affects k in [max(0, j), min(2n-2, j+n-1)]
                k_lo = max(0, min(i, j))
                k_hi = min(2 * n - 2, max(i, j) + n - 1)

                for k in range(k_lo, k_hi + 1):
                    # Contribution from i: f[k-i] if 0 <= k-i < n
                    ci = f[k - i] if (0 <= k - i < n) else 0.0
                    # Contribution from j: f[k-j] if 0 <= k-j < n
                    cj = f[k - j] if (0 <= k - j < n) else 0.0

                    diff = cj - ci
                    if diff > 1e-15:
                        # conv increases, need delta <= (M - conv[k]) / (2*diff)
                        slack = M - conv[k]
                        if slack < 0:
                            delta_max = 0
                            break
                        delta_candidate = slack / (2.0 * diff)
                        if delta_candidate < delta_max:
                            delta_max = delta_candidate

                if delta_max < 1e-10:
                    continue

                # Try the swap
                f[i] -= delta_max
                f[j] += delta_max
                conv_new = fast_convolve(f)
                M_new = conv_new.max()

                if M_new < M - 1e-14:
                    conv = conv_new
                    M = M_new
                    improved = True
                    if 2.0 * n * M < best_obj - 1e-12:
                        best_obj = 2.0 * n * M
                        best_f = f.copy()
                        no_improve = 0
                    # Continue trying more swaps from the improved point
                else:
                    # Revert
                    f[i] += delta_max
                    f[j] -= delta_max

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
            # Perturb to escape
            rng2 = np.random.default_rng(no_improve * 7919)
            f = project_simplex(best_f + rng2.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if verbose and best_obj / (2.0 * n) < 0.754:
            print(f"    Pairwise: obj={best_obj:.10f}")

    return best_f, best_obj / (2.0 * n)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_simple(f0, time_limit, verbose=False):
    """Simple coordinate descent: try adjusting one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved = False
        order = np.random.default_rng(int(time.monotonic() * 1000) % 999999).permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = M

            for mult in [0.5, 0.7, 1.3, 1.7]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                if 2.0 * n * M < best_obj - 1e-12:
                    best_obj = 2.0 * n * M
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj / (2.0 * n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining sparse LP-FW, subgradient polishing,
    pairwise swap search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: LP-FW on parent ===
    print("=== Phase 1: LP-FW on parent ===")
    n_parent = len(parent)
    t1 = min(250, deadline - time.monotonic() - 800)
    if t1 > 30:
        print(f"  n={n_parent}, time={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                  use_reduced_lp=(n_parent > 300), tol=0.08)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Try longer sequences ===
    print("\n=== Phase 2: Longer sequences ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [1024, 1536, 512]:
        rem = deadline - time.monotonic() - 600
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 150)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False,
                                  use_reduced_lp=True, tol=0.1)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 3: Deep LP-FW on best ===
    print(f"\n=== Phase 3: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 200)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            else:
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            print(f"  Round {p+1}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0),
                                      use_reduced_lp=(n_best > 300), tol=0.06)
            try_seq(a_opt, f"deep_{p}")

    # === Phase 4: Subgradient + Pairwise ===
    print(f"\n=== Phase 4: SG + Pairwise (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_sg = min(rem * 0.4, 120)
            print(f"  SG t={t_sg:.0f}s")
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg, lr=0.004, mom=0.92,
                                  top_k=max(20, n_best // 6), temp=0.00005, bb=True)
            try_seq(a_opt, "sg_bb")

        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_pw = min(rem * 0.5, 100)
            print(f"  Pairwise t={t_pw:.0f}s")
            a_opt, _ = pairwise_search(best_seq, time_limit=t_pw, verbose=False)
            try_seq(a_opt, "pairwise")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd = min(rem * 0.5, 100)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_simple(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_fine = min(rem * 0.4, 50)
            a_opt, _ = sg_polish(best_seq, time_limit=t_fine, lr=0.001, mom=0.95,
                                  top_k=max(10, n_best // 4), temp=0.00001, bb=True)
            try_seq(a_opt, "sg_fine")

        # Final LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_final = min(rem * 0.5, 40)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                       use_reduced_lp=True, tol=0.04)
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations + polish
        for p in range(2):
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.6, 15)
            a_opt, _ = sg_polish(f_pert, time_limit=t_pert, lr=0.0008, mom=0.95,
                                  top_k=max(10, n_best // 4), temp=0.00001, bb=True)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
