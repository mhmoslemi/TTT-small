def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(weights, f):
    """
    Compute gradient of LSE objective w.r.t. f.
    weights has length 2n-1, f has length n.
    grad[i] = 2 * sum_m weights[i+m] * f[m] (cross-correlation)
    """
    n = len(f)
    f_rev = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_subgrad(conv, f, top_k=20):
    """Compute subgradient at top-k peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = 10.0 * (pv - pv.min())
        weights = np.exp(scaled - scaled.max())
        weights /= weights.sum()

    sg = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        w = weights[i]
        if w < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * f[k - idx]
    return sg

        for direction in [-1, 1]:
            for delta in [0.001, 0.003, 0.01, 0.02]:
                f_trial = f.copy()
                f_trial[i] += direction * delta
                f_trial[j] -= direction * delta
                if f_trial[i] < 0 or f_trial[j] < 0:
                    continue
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_trial.copy()
                    f = f.copy()

    return best_f, best_obj


def parametric_tilt_optimize(n, time_limit):
    """Optimize over parametric family: a(i) = (1 + alpha * t(i)) / Z
    where t(i) is a tilt function. Try several tilt shapes."""
    best_f = None
    best_obj = np.inf
    t0 = time.monotonic()

    # Define several tilt functions
    i = np.arange(n, dtype=np.float64)
    tilts = {
        'linear': (2.0 * i / (n - 1) - 1.0),
        'quadratic': ((i / (n - 1) - 0.5) ** 2 - 0.25) * 4.0,
        'ramp': i / (n - 1),
        'bump': np.sin(np.pi * i / (n - 1)) - 0.5,
        'edge': np.abs(2.0 * i / (n - 1) - 1.0) - 0.5,
    }

    for name, t in tilts.items():
        if time.monotonic() - t0 > time_limit:
            break

        # Optimize alpha for this tilt
        best_alpha = 0.0
        best_obj_t = np.inf

        for alpha in np.linspace(-2.0, 2.0, 81):
            a = 1.0 + alpha * t
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s
            obj = fast_convolve(a).max()
            if obj < best_obj_t:
                best_obj_t = obj
                best_alpha = alpha

        # Refine alpha
        lo, hi = best_alpha - 0.05, best_alpha + 0.05
        for _ in range(30):
            mid = (lo + hi) / 2.0
            a = np.maximum(1.0 + mid * t, 0.0)
            s = a.sum()
            if s < 1e-10:
                lo = mid
                continue
            a = a / s
            obj = fast_convolve(a).max()
            if obj < best_obj_t:
                best_obj_t = obj
                hi = mid
            else:
                lo = mid

        a = np.maximum(1.0 + best_alpha * t, 0.0)
        a = a / a.sum()
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_f = a.copy()
            print(f"    Tilt '{name}': alpha={best_alpha:.4f}, obj={obj:.10f}")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"    NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization exploiting non-symmetric structure:
    Phase 1: Parametric tilt exploration for diverse starting points
    Phase 2: LSE optimization from best starts
    Phase 3: FW-LP for global structure
    Phase 4: Multi-start LSE with diversity
    Phase 5: Final high-beta polish + coordinate refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    n_primary = n_parent  # Main working length

    # === Phase 1: Parametric tilt exploration ===
    print(f"\n=== Phase 1: Parametric Tilt Exploration (n={n_primary}) ===")
    t1_end = time.monotonic() + 60
    tilt_f, tilt_obj = parametric_tilt_optimize(n_primary, time_limit=60)
    if tilt_f is not None:
        try_sequence(tilt_f, best_tracker)

    # Also try from parent
    try_sequence(parent, best_tracker)

    current_best = best_tracker[1].copy()
    print(f"  After Phase 1: best={best_tracker[0]:.8f}")

    # === Phase 2: LSE from multiple starts ===
    print(f"\n=== Phase 2: LSE Optimization ===")

    # Build list of starting points
    starts = [current_best.copy()]
    if tilt_f is not None:
        starts.append(tilt_f.copy())

    # Add some random perturbations
    for _ in range(2):
        pert = current_best + rng.standard_normal(n_primary) * 0.01
        starts.append(project_simplex(pert))

    # LSE on each start
    for idx, start in enumerate(starts):
        remaining = deadline - time.monotonic()
        t_alloc = min(200, remaining / (len(starts) - idx))
        if t_alloc < 30:
            break
        print(f"  LSE start {idx+1}/{len(starts)}, alloc={t_alloc:.0f}s")
        a_opt, obj = lse_optimize(start, time_limit=t_alloc,
                                  beta_start=15.0, beta_end=5000.0,
                                  lr_scale=2.5)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    print(f"  After Phase 2: best={best_tracker[0]:.8f}")

    # === Phase 3: FW-LP for global structure ===
    print(f"\n=== Phase 3: FW-LP Global Structure ===")
    remaining = deadline - time.monotonic()
    t_alloc = min(250, remaining * 0.3)
    if t_alloc > 60:
        print(f"  Allocating {t_alloc:.0f}s for FW-LP")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc, lp_timeout=40)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    print(f"  After Phase 3: best={best_tracker[0]:.8f}")

    # === Phase 4: Multi-start LSE with diversity ===
    print(f"\n=== Phase 4: Multi-Start LSE ===")

    n_starts = 4
    for s_idx in range(n_starts):
        remaining = deadline - time.monotonic()
        t_alloc = min(150, remaining / (n_starts - s_idx))
        if t_alloc < 40:
            break

        # Create diverse start
        if s_idx == 0:
            start = current_best.copy()
        else:
            scale = 0.005 + 0.01 * s_idx
            pert = current_best + rng.standard_normal(n_primary) * scale
            start = project_simplex(pert)

        beta_start = 30.0 + 50.0 * s_idx
        beta_end = 5000.0 + 2000.0 * s_idx
        print(f"  Multi-start {s_idx+1}/{n_starts}: beta {beta_start:.0f}->{beta_end:.0f}, "
              f"alloc={t_alloc:.0f}s")

        a_opt, obj = lse_optimize(start, time_limit=t_alloc,
                                  beta_start=beta_start, beta_end=beta_end,
                                  lr_scale=2.0, restart_every=60)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    print(f"  After Phase 4: best={best_tracker[0]:.8f}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")

    # High-beta LSE
    remaining = deadline - time.monotonic()
    t_alloc = min(150, remaining * 0.4)
    if t_alloc > 30:
        print(f"  High-beta LSE polish, {t_alloc:.0f}s")
        a_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=200.0, beta_end=20000.0,
                                  lr_scale=1.5, momentum_init=0.85,
                                  restart_every=100)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    # Coordinate refinement
    remaining = deadline - time.monotonic()
    t_alloc = min(60, remaining * 0.3)
    if t_alloc > 10:
        print(f"  Coordinate refinement, {t_alloc:.0f}s")
        a_opt, obj = coordinate_refine(current_best, time_limit=t_alloc,
                                       n_trials=500)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    # Final FW-LP sprint
    remaining = deadline - time.monotonic()
    t_alloc = min(60, remaining * 0.4)
    if t_alloc > 20:
        print(f"  Final FW-LP sprint, {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc, lp_timeout=30)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy()

    # Last LSE micro-polish
    remaining = deadline - time.monotonic()
    t_alloc = min(40, remaining * 0.5)
    if t_alloc > 10:
        print(f"  Final LSE micro-polish, {t_alloc:.0f}s")
        a_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=500.0, beta_end=30000.0,
                                  lr_scale=1.0, momentum_init=0.80)
        try_sequence(a_opt, best_tracker)

    # === Try alternative length ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Alternative Length ===")
        # Try n = n_primary + 16 or n_parent - 16
        for n_alt in [n_primary + 32, n_primary - 32, n_primary + 64]:
            if n_alt < 64 or remaining < 50:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            # Interpolate
            x_old = np.linspace(0, 1, n_primary)
            x_new = np.linspace(0, 1, n_alt)
            f_interp = np.interp(x_new, x_old, current_best)
            f_interp = np.maximum(f_interp, 0.0)
            if f_interp.sum() > 1e-10:
                f_interp = f_interp / f_interp.sum()

            t_alloc = min(60, remaining - 30)
            if t_alloc < 25:
                continue

            a_opt, obj = lse_optimize(f_interp, time_limit=t_alloc,
                                      beta_start=20.0, beta_end=5000.0)
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
