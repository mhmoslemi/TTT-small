Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max approximation with L-BFGS in softmax parameterization
2. Active set local optimality test
3. Coordinate descent polish
4. Random restarts for escaping local minima

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_obj_grad(x, tau):
    """
    Objective and gradient for smooth-max in softmax parameterization.
    a = softmax(x), minimize tau * log(sum exp(conv(a)/tau))
    """
    n = len(x)
    # Softmax
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a_sum = a.sum()
    a = a / a_sum
    
    # Convolution
    conv = fast_convolve(a)
    
    # Softmax over conv values
    c = conv / tau
    c_max = c.max()
    exp_c = np.exp(c - c_max)
    p = exp_c / exp_c.sum()
    
    # Objective
    obj = tau * (np.log(exp_c.sum()) + c_max)
    
    # Gradient of h w.r.t. a
    r = np.correlate(p, a, mode='valid')
    dh_da = 2.0 * r
    
    # Gradient w.r.t. x (softmax parameterization)
    # dh/dx_i = dh/da_i * a_i - a_i * sum_j dh/da_j * a_j
    s = np.dot(dh_da, a)
    dh_dx = dh_da * a - a * s
    
    return obj, dh_dx

    n = len(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = fast_convolve(a)
    c = conv / tau
    c_max = c.max()
    return tau * (np.log(np.exp(c - c_max).sum()) + c_max)


def lbfgs_smooth_max(a0, tau_list, max_iter_per_tau=200):
    """
    Run L-BFGS with decreasing tau schedule.
    a0: initial point on simplex
    tau_list: list of tau values (decreasing)
    """
    n = len(a0)
    # Convert to log-parameterization
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    x = np.log(a)
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    for tau in tau_list:
        res = minimize(
            fun=lambda x: smooth_obj_grad(x, tau)[0],
            x0=x,
            jac=lambda x: smooth_obj_grad(x, tau)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_tau, 'gtol': 1e-10, 'ftol': 1e-14}
        )
        x = res.x
        a = np.exp(x - x.max())
        a = a / a.sum()
        
        conv_max = fast_convolve(a).max()
        if conv_max < best_conv_max:
            best_conv_max = conv_max
            best_a = a.copy()
    
    return best_a, best_conv_max


def active_set_test(a, tol=1e-6):
    """
    Check if a is a local optimum using the active set method.
    Returns (is_optimal, active_set, lambda_values)
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    # Active set: peaks within tol of max
    S = set(np.where(conv >= M - tol)[0])
    S_list = sorted(S)
    
    if len(S_list) == 0:
        return True, S, np.array([])
    
    # Active indices (where a_i > 0)
    active_i = np.where(a > 1e-10)[0]
    
    # Build system: for each active i, sum_{k in S} lambda_k * 2 * a_{k-i} = mu
    # Plus: sum lambda = 1
    # Unknowns: lambda (|S|) and mu (1)
    
    m = len(S_list)
    # Matrix A of size (len(active_i)+1) x (m+1)
    # Rows 0..len(active_i)-1: sum_k A[i,k] * lambda_k - mu = 0
    # Row len(active_i): sum_k lambda_k = 1
    
    A = np.zeros((len(active_i) + 1, m + 1))
    for ii, i in enumerate(active_i):
        for kk, k in enumerate(S_list):
            j = k - i
            if 0 <= j < n:
                A[ii, kk] = 2.0 * a[j]
        A[ii, m] = -1.0  # coefficient of mu
    
    # Normalization row
    A[len(active_i), :m] = 1.0
    A[len(active_i), m] = 0.0
    
    b = np.zeros(len(active_i) + 1)
    b[len(active_i)] = 1.0
    
    # Solve least squares (system may be overdetermined)
    sol, residuals, rank, sv = np.linalg.lstsq(A, b, rcond=None)
    lambda_vals = sol[:m]
    mu = sol[m]
    
    # Check if all lambda >= -tol
    is_optimal = np.all(lambda_vals >= -tol)
    
    return is_optimal, S, lambda_vals


def active_set_optimize(a0, time_limit=60):
    """
    Optimize using active set method.
    At each step, solve for the optimal a given the active set S,
    then check KKT conditions.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    # Initial active set
    conv = fast_convolve(a)
    M = conv.max()
    S = set(np.where(conv >= M * 0.95)[0])
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        S_list = sorted(S)
        if len(S_list) == 0:
            break
        
        # Solve: minimize max_{k in S} conv_k(a) s.t. sum(a)=1, a>=0
        # Using L-BFGS on the smooth version with small tau
        tau = 0.001 * M
        x = np.log(np.maximum(a, 1e-15))
        
        # Custom objective: only consider active peaks
        def active_obj_grad(x):
            x_shift = x - x.max()
            a_cur = np.exp(x_shift)
            a_cur = a_cur / a_cur.sum()
            conv_cur = fast_convolve(a_cur)
            
            # Only active peaks
            c = conv_cur[S_list] / tau
            c_max = c.max()
            exp_c = np.exp(c - c_max)
            p_act = exp_c / exp_c.sum()
            
            obj = tau * (np.log(exp_c.sum()) + c_max)
            
            # Gradient: only from active peaks
            # dh/da_i = 2 * sum_{k in S} p_k * a_{k-i}
            dh_da = np.zeros(n)
            p_full = np.zeros(2*n-1)
            for ii, k in enumerate(S_list):
                p_full[k] = p_act[ii]
            r = np.correlate(p_full, a_cur, mode='valid')
            dh_da = 2.0 * r
            
            s = np.dot(dh_da, a_cur)
            dh_dx = dh_da * a_cur - a_cur * s
            return obj, dh_dx
        
        res = minimize(
            fun=lambda x: active_obj_grad(x)[0],
            x0=x,
            jac=lambda x: active_obj_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 100, 'gtol': 1e-8}
        )
        
        x = res.x
        a_new = np.exp(x - x.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_convolve(a_new).max()
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Update active set
        conv = fast_convolve(a)
        M = conv.max()
        # Expand: any peak within 1% of current max
        S = set(np.where(conv >= M * 0.99)[0])
        
        # Check: if a new peak has appeared that's higher than our current max
        # Expand S to include it
        S = set(np.where(conv >= M * 0.98)[0])
        
        if no_improve > 10:
            break
    
    return best_a, best_M


def coordinate_descent(a0, time_limit=60, n_sweeps=50):
    """
    Cyclic coordinate descent on the simplex.
    For each coordinate i, find the optimal value given all others fixed.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        sweep += 1
        improved = False
        
        for i in range(n):
            # Save current value
            a_i_old = a[i]
            a_rest_sum = 1.0 - a_i_old
            
            # conv_k as function of a_i (with a_j fixed for j != i):
            # conv_k = (contrib from a_i) + (const from other terms)
            # Terms involving a_i: a_i^2 (if k=2i) + a_i * 2 * a_{k-i} (if k-i != i)
            
            # For each k, conv_k(a) = c_k * a_i^2 + d_k * a_i + e_k
            # where c_k = 1 if k == 2i else 0
            #       d_k = 2 * a_{k-i} if k-i != i and 0 <= k-i < n else (a_{k-i} if k-i == i, which is covered by c_k)
            #       e_k = conv_k with a_i removed
            
            # Actually: conv_k = sum_j a_j * a_{k-j}
            # Terms with a_i: j=i gives a_i * a_{k-i}; j=k-i gives a_{k-i} * a_i
            # If k-i != i and both valid: contribution = 2 * a_i * a_{k-i}
            # If k-i == i (k=2i): contribution = a_i^2
            # If k-i out of range: only one valid term
            
            # Let's compute for a range of a_i values
            # a_i in [0, 1] (since sum must be 1 and all a_j >= 0)
            
            # Actually, a_i can be at most a_i_old + a_rest_sum (if we redistribute)
            # but in coordinate descent, we fix the sum: a_i + sum_{j!=i} a_j = 1
            # So a_i is the only free variable, and a_j for j != i are fixed.
            # Thus a_i is determined: a_i = 1 - sum_{j!=i} a_j = a_i_old. 
            # Wait, that's circular. In coordinate descent on the simplex,
            # we fix all coordinates except i, and optimize a_i.
            # But the constraint sum(a) = 1 means a_i = 1 - sum_{j!=i} a_j,
            # which is already fixed! So coordinate descent on the simplex
            # doesn't work the usual way.
            
            # Instead, we can use the "pair exchange" approach:
            # exchange mass between coordinate i and some other coordinate j.
            
            # Or: use the parameterization a_i = alpha * b_i where b is fixed
            # and alpha is the free parameter. But this doesn't quite work either.
            
            # Better approach: use the "push" method.
            # Increase a_i by delta and decrease some other coordinate by delta.
            
            # Actually, for the simplex constraint, the standard coordinate
            # descent approach is:
            # For coordinate i, the feasible range is [0, 1 - sum_{j!=i} a_j] = [0, 0]
            # since sum(a) = 1 forces a_i = 1 - sum_{j!=i} a_j.
            
            # So pure coordinate descent on the simplex is degenerate.
            # Instead, let's use "coordinate pairs": increase a_i and decrease a_j
            # by the same amount.
            
            # For simplicity, let's skip pure coordinate descent and use
            # the "pair" approach in a different function.
            
            pass
        
        break  # Skip for now, use pair approach instead
    
    return best_a, best_M


def pair_descent(a0, time_limit=60):
    """
    Pair coordinate descent: for pairs (i,j), optimize the transfer of mass
    from i to j (or vice versa).
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Random pair
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        a_i = a[i]
        a_j = a[j]
        
        # Transfer: a_i' = a_i + t, a_j' = a_j - t
        # t in [-a_j, a_i]
        
        def obj_1d(t):
            a[i] = a_i + t
            a[j] = a_j - t
            return fast_convolve(a).max()
        
        # Restore
        a[i] = a_i
        a[j] = a_j
        
        t_lo = -a_j
        t_hi = a_i
        
        res = minimize_scalar(
            fun=obj_1d,
            bounds=(t_lo, t_hi),
            method='bounded',
            options={'maxiter': 50, 'xatol': 1e-12}
        )
        
        if res.success:
            t_opt = res.x
            a[i] = a_i + t_opt
            a[j] = a_j - t_opt
            a[i] = max(0, a[i])
            a[j] = max(0, a[j])
            a = a / a.sum()
            
            M_new = fast_convolve(a).max()
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_a = a.copy()
    
    return best_a, best_M


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Two-hump
    h1 = np.exp(-0.5 * ((x - 0.25 * (n-1)) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * (n-1)) / (0.1 * n)) ** 2)
    two_hump = (h1 + h2)
    inits.append(two_hump / two_hump.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Smooth-max L-BFGS with tau schedule (main workhorse)
    Phase 2: Active set refinement
    Phase 3: Pair descent polish
    Phase 4: Final fine L-BFGS
    Phase 5: Random restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_parent = 384
        parent = np.ones(n_parent) / n_parent
    
    n_base = len(parent)
    
    # === Phase 1: Smooth-max L-BFGS on parent (~600s) ===
    print("\n=== Phase 1: Smooth-max L-BFGS ===")
    
    tau_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_per_tau = rem / len(tau_list)
        print(f"  Running L-BFGS with tau schedule, {len(tau_list)} stages")
        
        a_cur = parent.copy()
        x = np.log(np.maximum(a_cur, 1e-15))
        
        for ti, tau in enumerate(tau_list):
            rem = deadline - time.monotonic() - (900 - ti * 60)
            if rem < 5:
                break
            max_iter = int(rem / 0.05)  # estimate iterations based on time
            max_iter = max(20, min(max_iter, 5000))
            
            res = minimize(
                fun=lambda x, t=tau: smooth_obj_grad(x, t)[0],
                x0=x,
                jac=lambda x, t=tau: smooth_obj_grad(x, t)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-12, 'ftol': 1e-15}
            )
            x = res.x
            a_cur = np.exp(x - x.max())
            a_cur = a_cur / a_cur.sum()
            a_cur = np.maximum(a_cur, 0)
            a_cur = a_cur / a_cur.sum()
            
            conv_max = fast_convolve(a_cur).max()
            print(f"  tau={tau:.5f}: conv_max={conv_max:.8f}, score≈{2*n_base*conv_max:.8f}")
            try_seq(a_cur, f"lbfgs_tau{tau}")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Active set refinement (~400s) ===
    print("\n=== Phase 2: Active set refinement ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 400)
            print(f"  Active set optimize, t={t_alloc:.0f}s")
            a_opt, M_opt = active_set_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "active_set")
    
    print(f"  Phase 2 best: {best_score:.8f}")
    
    # === Phase 3: Pair descent (~300s) ===
    print("\n=== Phase 3: Pair descent ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 300)
            print(f"  Pair descent, t={t_alloc:.0f}s")
            a_opt, M_opt = pair_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pair_descent")
    
    print(f"  Phase 3 best: {best_score:.8f}")
    
    # === Phase 4: Fine L-BFGS with very small tau (~200s) ===
    print("\n=== Phase 4: Fine L-BFGS ===")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:

            tau_fine = [0.0003, 0.0001, 0.00005]
            a_cur = best_seq.copy()
            x = np.log(np.maximum(a_cur, 1e-15))
            
            for tau in tau_fine:
                rem = deadline - time.monotonic() - 50
                if rem < 5:
                    break
                max_iter = max(20, int(rem / 0.03))
                
                res = minimize(
                    fun=lambda x, t=tau: smooth_obj_grad(x, t)[0],
                    x0=x,
                    jac=lambda x, t=tau: smooth_obj_grad(x, t)[1],
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'gtol': 1e-13, 'ftol': 1e-16}
                )
                x = res.x
                a_cur = np.exp(x - x.max())
