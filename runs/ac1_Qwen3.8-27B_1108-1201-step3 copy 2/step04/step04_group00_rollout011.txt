import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, m_relax_start=1.0, m_relax_max=3.0):
    """LP-based Frank-Wolfe with adaptive M relaxation and bisection line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = m_relax_start

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.15, m_relax_max)
            if m_exp > m_relax_max:
                m_exp = m_relax_start
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            m_exp = min(m_exp * 1.03, m_relax_max)
            continue

        # Coarse-to-fine line search
        best_alpha = 0.0
        best_val = M

        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            if best_alpha == alphas[0]:
                lo, hi = 0.0, alphas[1]
            elif best_alpha == alphas[-1]:
                lo, hi = alphas[-2], 1.0
            else:
                idx = alphas.index(best_alpha)
                lo, hi = alphas[idx - 1], alphas[idx + 1]

            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                m_exp = m_relax_start
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, m_relax_max)

        if no_improve > 60:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.001 + 0.003 * min(no_improve // 50, 8)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_relax_start
            no_improve = 0

    return best_f, best_obj


def compute_smooth_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a.
    weights has 2n-1 elements (normalized to sum to 1).
    """
    n = len(a)
    # grad[i] = 2 * sum_j a[j] * weights[i+j] for valid i+j
    # This is a cross-correlation: (a *correlate* weights)[i]
    # Use FFT for efficiency
    fft_size = 1 << (4 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    # We need: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Let's define w_ext of length 2n-1 (already have this)
    # Cross-correlation: c[i] = sum_j a[j] * w[i+j]
    # This equals the convolution of a_rev with w at index (n-1-i) + (n-1)
    # Actually: let's just use the definition directly with FFT
    # c[i] = sum_j a[j] * w[i+j]
    # Define w2 of length 4n-2: w2[0:2n-1] = w, rest = 0
    # Then c[i] for i=0..n-1: sum_j a[j]*w[i+j]
    # This is (a * w2) but with a specific indexing...
    # Let's use: c = ifft(fft(a_ext) * conj(fft(w_ext))) -- no that's not right either
    # 
    # Simplest correct approach: 
    # c[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Let w_rev = w[::-1] (length 2n-1)
    # Then c[i] = sum_j a[j] * w[i+j]
    # Define: full_conv = convolve(a, w, mode='full') has length n + 2n-2 = 3n-2
    # full_conv[k] = sum_j a[j] * w[k-j] for valid j, k-j
    # We want sum_j a[j] * w[i+j]. Let k = i + 2*(n-1) - j... this is getting complicated.
    # 
    # Let me just use the direct O(n^2) approach since n <= 1024 means ~1M ops:
    grad = np.zeros(n)
    w = weights  # length 2n-1
    for i in range(n):
        # grad[i] = 2.0 * sum_{j=0}^{n-1} a[j] * w[i+j]
        # i+j ranges from i to i+n-1, all < 2n-1 since i < n
        grad[i] = 2.0 * np.dot(a, w[i:i+n])
    return grad


def fista_smooth(a0, time_limit, tau_init=50.0, tau_end=500.0, L=2.2, verbose=False):
    """
    FISTA with log-sum-exp smoothing of the max auto-correlation objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    x_prev = a.copy()
    x_curr = a.copy()
    y = a.copy()
    t_counter = 1.0

    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive tau: increase over time
        progress = min((time.monotonic() - t0) / total_time, 1.0)
        tau = tau_init * (tau_end / tau_init) ** progress

        # Compute weights (softmax of conv values)
        conv = fast_convolve(y)
        max_conv = conv.max()
        scaled = tau * (conv - max_conv)
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 50)
        exp_vals = np.exp(scaled)
        weights = exp_vals / exp_vals.sum()

        # Compute gradient
        grad = compute_smooth_gradient(y, weights)

        # FISTA update
        x_new = project_simplex(y - (1.0 / L) * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_counter ** 2)) / 2.0
        y_new = x_new + ((t_counter - 1.0) / t_new) * (x_new - x_prev)
        y_new = project_simplex(y_new)

        x_prev = x_curr
        x_curr = x_new
        y = y_new
        t_counter = t_new

        # Check improvement
        conv_curr = fast_convolve(x_curr)
        obj = conv_curr.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x_curr.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    FISTA iter {it}: obj={best_obj:.12f}, tau={tau:.1f}")
        else:
            no_improve += 1

        if no_improve > 500:
            rng = np.random.default_rng(it * 7 + 13)
            y = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            x_prev = y.copy()
            x_curr = y.copy()
            t_counter = 1.0
            no_improve = 0

    return best_a, best_obj


def peak_shave(a0, time_limit, verbose=False, start_lr=0.01, decay=0.9999):
    """
    Aggressive peak shaving: directly target the worst lag(s) with adaptive step size.
    Uses top-k peaks with softmax weighting and backtracking line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = start_lr
    momentum = np.zeros(n)
    beta = 0.9
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Identify top peaks
        conv = fast_convolve(a)
        k_peaks = min(20, 2 * n - 1)
        peak_indices = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # Softmax weights over peaks
        pv = conv[peak_indices]
        temp = 0.001
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k_peaks)
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        # For each peak at lag k, gradient of conv(a,a)[k] w.r.t. a:
        #   d/da[i] conv(a,a)[k] = a[k-i] + a[k-i] (if 0<=k-i<n) = 2*a[k-i]
        # More precisely: if k-i is in [0,n-1], contribution is 2*a[k-i], else 0
        sg = np.zeros(n)
        for ii in range(k_peaks):
            if w[ii] < 1e-12:
                continue
            k = int(peak_indices[ii])
            # For each i in [0,n-1]: contribution is 2*a[k-i] if 0<=k-i<n
            js = np.arange(n)
            ks = k - js
            valid = (ks >= 0) & (ks < n)
            if valid.any():
                sg[js[valid]] += 2.0 * w[ii] * a[ks[valid]]

        # Nesterov momentum update
        momentum = beta * momentum - lr * sg
        a_new = project_simplex(a + momentum)

        # Check if improvement; if not, backtrack
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj + 1e-10:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
                if verbose and it % 200 == 0:
                    print(f"    PeakShave iter {it}: obj={best_obj:.12f}, lr={lr:.6f}")
            else:
                no_improve += 1
        else:
            # Backtrack
            lr *= 0.7
            momentum *= 0.5
            no_improve += 1

        lr *= decay
        lr = max(lr, 0.0001)

        if no_improve > 100:
            rng = np.random.default_rng(it * 3 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            lr = max(start_lr * 0.5, 0.002)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_val = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-15 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 5:
            break

    return best_f, best_obj


def make_raised_cosine(n):
    """Create a raised cosine (Hann) window sequence."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_window(n, sigma_ratio=0.3):
    """Create a Gaussian window sequence."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_ratio * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_hamming_window(n):
    """Create a Hamming window sequence."""
    x = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1)) if n > 1 else np.array([1.0])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_raised_cosine(512)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(200, deadline - time.monotonic() - 600)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Try analytical initializations ===
    print("\n=== Phase 2: Analytical initializations ===")
    n_parent = len(parent)
    init_shapes = {
        "raised_cos": make_raised_cosine(n_parent),
        "gaussian": make_gaussian_window(n_parent, 0.3),
        "hamming": make_hamming_window(n_parent),
        "broad_gauss": make_gaussian_window(n_parent, 0.45),
    }

    for name, init in init_shapes.items():
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(60, rem * 0.25)
        print(f"  {name}: LP-FW t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(init, time_limit=t_alloc)
        try_seq(a_opt, f"lpfw_{name}")

    # === Phase 3: FISTA refinement on best ===
    print(f"\n=== Phase 3: FISTA (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_fista = min(rem * 0.5, 200)
            print(f"  FISTA t={t_fista:.0f}s")
            a_opt, fista_obj = fista_smooth(best_seq, time_limit=t_fista,
                                            tau_init=30.0, tau_end=300.0,
                                            L=2.2, verbose=True)
            try_seq(a_opt, "fista")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_ps = min(rem * 0.5, 180)
            print(f"  PeakShave t={t_ps:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_ps,
                                  start_lr=0.008, decay=0.99995, verbose=True)
            try_seq(a_opt, "peak_shave")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Different lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        for target_n in [640, 896, 1024]:
            if len(best_seq) == target_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 120)
            print(f"  n={target_n}: LP-FW t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"length_{target_n}")

            if len(a_opt) != len(best_seq):
                rem2 = deadline - time.monotonic() - 150
                if rem2 > 15:
                    a_opt2, _ = fista_smooth(a_opt, time_limit=min(rem2 * 0.4, 60),
                                             tau_init=50.0, tau_end=400.0, L=2.2)
                    try_seq(a_opt2, f"fista_n{target_n}")

    # === Phase 6: Deep LP-FW restarts ===
    print(f"\n=== Phase 6: Deep LP-FW restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            else:
                # Blend with uniform
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            print(f"  Restart {p+1}: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{p}")

    # === Phase 7: FISTA on the best found ===
    print(f"\n=== Phase 7: Final FISTA (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_f2 = min(rem * 0.6, 100)
            print(f"  FISTA t={t_f2:.0f}s")
            a_opt, _ = fista_smooth(best_seq, time_limit=t_f2,
                                    tau_init=100.0, tau_end=1000.0, L=2.2, verbose=True)
            try_seq(a_opt, "fista_final")

    # === Phase 8: Peak shave final ===
    print(f"\n=== Phase 8: Final Peak Shave (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_ps2 = min(rem * 0.6, 60)
            a_opt, _ = peak_shave(best_seq, time_limit=t_ps2,
                                  start_lr=0.003, decay=0.9999, verbose=True)
            try_seq(a_opt, "peak_shave_final")

    # === Phase 9: Coordinate descent polish ===
    print(f"\n=== Phase 9: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem * 0.7, 60)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_final")

    # === Fallback ===
    if best_seq is None:
        best_seq = make_raised_cosine(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
