"""

2. Nesterov gradient with adaptive softmax peak weighting and backtracking line search

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute the objective value and convolution."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), conv
    obj = 2.0 * n * conv.max() / (s * s)
    return obj, conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def softmax_gradient(a, conv, top_k=30, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive parameters and backtracking."""

    conv = fast_convolve(a)

    best_a = a.copy()

    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y * n

        conv_y = fast_convolve(y)
        grad = softmax_gradient(y, conv_y, top_k=top_k, temp=temp)

        a_try = a + v
        a_try = np.maximum(a_try, 0.0)

def lp_direction(a, conv, top_k=5):
    """
    LP: maximize sum(g) subject to conv(a, g) <= M for top-k peaks, g >= 0.
    Returns direction d = g - a (normalized).
    """
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    M = conv.max()

    # Only constrain the top-k peaks (and a few neighbors for robustness)
    row_indices = set()
    for p in peaks:
        for offset in range(-2, 3):
            if 0 <= p + offset < 2 * n - 1:
                row_indices.add(p + offset)
    row_indices = sorted(row_indices)

    C_full = build_conv_matrix(a)
    C_sub = C_full[row_indices, :]
    b_sub = np.full(len(row_indices), M * 1.0001)

    c_vec = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_vec, A_ub=C_sub, b_ub=b_sub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9,
                                                   'time_limit': 5})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_step(a, conv, n):
    """One LP-based step with adaptive step sizing."""
    M = conv.max()
    g = lp_direction(a, conv, top_k=5)

    if g is None:
        return None, False

    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, False

    # Normalize g to same sum as a
    a_sum = a.sum()
    g = g / g_sum * a_sum

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return None, False

    # Adaptive step sizing: try multiple candidates
    best_a = a.copy()
    best_obj = 2.0 * n * M
    best_t = 0.0

    candidates = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s * n
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t
            # Don't break; try to find even better step in next iteration
            # but for efficiency, accept first good improvement
            break

    if best_t > 0:
        return best_a, True
    return None, False

def coordinate_descent(a0, time_limit):
    """
    Coordinate descent with golden-section 1D minimization.
    For each coordinate, find the optimal value by minimizing the objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s * n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    gauss_inv = (1.0 + np.sqrt(5.0)) / 2.0  # golden ratio conjugate

    def obj_at(a_test):
        s_t = a_test.sum()
        if s_t < 1e-15:
            return float('inf')
        conv_t = fast_convolve(a_test)
        return 2.0 * n * conv_t.max() / (s_t * s_t)

    # Precompute: for coordinate i, conv[k] depends on a[i] as:
    # conv[k] = sum_j a[j]*a[k-j], and a[i] appears when j=i or k-j=i
    # So conv[k] = (terms not involving a[i]) + a[i]*a[k-i] + a[i]*a[i] (if k=2i and i valid)
    # Actually: conv[k] includes a[i]*a[k-i] (if 0<=k-i<n) and a[k-i]*a[i] (same term)
    # For the full conv: conv[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j]
    # a[i] appears in conv[k] when: j=i (need 0<=k-i<n) or k-j=i (need 0<=k-i<n)
    # Both conditions are the same: 0 <= k-i < n, i.e., i <= k <= i+n-1
    # In that case, a[i] appears as a[i]*a[k-i] (once if i != k-i, twice if i == k-i)

    while time.monotonic() - t0 < time_limit:
        # Random permutation of coordinates
        perm = rng.permutation(n)
        improved_round = False

        for i in perm:
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Find valid k range where a[i] appears in conv[k]
            k_lo = max(0, i)  # k >= i (from j=i, need k-i >= 0)
            k_hi = min(2 * n - 2, i + n - 1)  # k <= i+n-1 (from j=i, need k-i < n)
            # Also from k-j=i: j = k-i, need 0 <= j < n, so i <= k <= i+n-1
            # So k range is [max(0, 0), min(2n-2, i+n-1)] but also k >= 0
            # Actually k must satisfy: there exists j in [0,n-1] such that k-j=i AND k-i=j
            # j = k-i must be in [0, n-1], so k in [i, i+n-1]
            # AND k-j = i means j = k-i, same thing
            # Also need k-j valid: 0 <= k-j < n, i.e., 0 <= i < n (always true)
            # And j valid: 0 <= j < n, i.e., 0 <= k-i < n
            # So k in [i, i+n-1] intersect [0, 2n-2]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            if k_lo > k_hi:
                continue

            # Current value
            a_i_old = a[i]

            # 1D optimization: minimize max(conv[k] for k in active set) as function of a[i]
            # conv[k](a[i]) = base[k] + a[i] * a[k-i] + (a[i]^2 if k == 2*i and valid)
            # Let's compute: for each k in [k_lo, k_hi]:
            #   conv[k] = sum_{j != i} a[j]*a[k-j] + a[i]*a[k-i] (if valid)
            #   But if k-i == i (i.e., k == 2i), then a[i]*a[k-i] = a[i]^2
            #   And the term a[k-i]*a[i] is the same as a[i]*a[k-i]
            #   Wait, in the sum sum_j a[j]*a[k-j], when j=i we get a[i]*a[k-i]
            #   When k-j=i, i.e., j=k-i, we get a[k-i]*a[i] = a[i]*a[k-i]
            #   These are the same term! So a[i] appears only once in the sum (at j=i)
            #   UNLESS j=i and j=k-i are different indices, which means i != k-i, i.e., k != 2i
            #   If k == 2i, then j=i gives a[i]*a[i] = a[i]^2
            #   So conv[k] = (sum over j != i) + a[i]*a[k-i] if k != 2i
            #              = (sum over j != i) + a[i]^2 if k == 2i

            # For 1D minimization, we need conv[k] as a function of x = a[i]:
            # if k != 2i: conv[k](x) = base_k + x * a[k-i]  (linear in x)
            # if k == 2i: conv[k](x) = base_k + x^2  (quadratic in x)
            # where base_k = sum_{j != i, valid} a[j]*a[k-j]

            # This is a "max of linear and one quadratic" problem in 1D!
            # We can solve it exactly or approximately.

            # For efficiency, use golden section search over a reasonable range
            # Range: [0, 2 * a_i_old] (or wider if a_i_old is 0)
            lo = 0.0
            hi = max(2.0 * a_i_old, 0.01) if a_i_old > 0 else (a.mean() * 2 + 0.01)

            # Evaluate at golden section points
            invphi2 = (3.0 - np.sqrt(5.0)) / 2.0
            invphi = (np.sqrt(5.0) - 1.0) / 2.0

            x1 = lo + invphi2 * (hi - lo)
            x2 = lo + invphi * (hi - lo)

            a[i] = x1
            f1 = obj_at(a)
            a[i] = x2
            f2 = obj_at(a)

            # Golden section iterations
            for _ in range(8):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = lo + invphi2 * (hi - lo)
                    a[i] = x1
                    f1 = obj_at(a)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + invphi * (hi - lo)
                    a[i] = x2
                    f2 = obj_at(a)

            # Take the better
            if f1 < f2:
                a[i] = x1
                obj_new = f1
            else:
                a[i] = x2
                obj_new = f2

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                improved_round = True
            else:
                # Revert if no improvement
                a[i] = a_i_old

        if not improved_round:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_a, best_obj

    inits['sine_half'] = 1.0 + 0.5 * np.sin(np.pi * x / n)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5) * n

    # Bimodal (two peaks)
    bi = np.exp(-0.5 * ((x - 0.2 * n) / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - 0.8 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = bi + 0.05

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps * n
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps * n)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps * n

    # Asymmetric combination
    asym = 1.0 + 0.3 * np.sin(2 * np.pi * x / n) + 0.2 * np.cos(4 * np.pi * x / n)
    inits['asym'] = np.maximum(asym, 0.1)

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target)
    a_norm = a_short / s * n_target
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses * n_target
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid optimization with multi-length exploration,
    Nesterov gradient, LP direction finding, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
                print(f"Parent: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Multi-length gradient scan
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 180.0

    lengths = [768, 1024, 1280]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 900:
            break

        inits = generate_inits(n, parent, rng)
        # Prioritize diverse inits
        priority = ['parent', 'parent_interp', 'u_shape', 'mono_dec', 'sine_half', 'random', 'asym']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]  # Limit to 5 per length

        t_per = max(8.0, (phase1_limit - 5) / max(1, len(lengths) * len(init_names)))

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 400)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, obj_local = nesterov_optimize(a0, time_limit=t_alloc,
                                                  lr_init=0.01, momentum=0.9,
                                                  top_k=40, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (up to 4, diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt, score))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    # Always include the global best if not already there
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy(), best_score))
            seen_lengths.add(len(best_sequence))

    remaining = deadline - time.monotonic() - 400.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(20.0, remaining * 0.55 / n_cand)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break

        print(f"  LP candidate {i}: n={n}, init_score={sc:.6f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Coordinate descent polishing
    # =========================================================
    print("=== Phase 3: Coordinate Descent ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 30:
            t_cd = min(remaining * 0.6, 150.0)
            if t_cd > 10:
                n_best = len(best_sequence)
                # For longer sequences, do fewer passes but wider search
                a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordDesc (n={n_best}): {score:.6f} *NEW BEST*")
                else:
                    print(f"  CoordDesc (n={n_best}): {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final gradient + LP micro-polish
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0

        # Short gradient runs with different configs
        if remaining > 20:
            t_g = min(remaining * 0.3, 60.0)
            if t_g > 5:
                configs = [
                    {'lr_init': 0.003, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.0005},
                    {'lr_init': 0.008, 'momentum': 0.88, 'top_k': 30, 'temp_init': 0.003},
                ]
                for ci, cfg in enumerate(configs):
                    if time.monotonic() > deadline - 20:
                        break
                    t_alloc = min(t_g / (len(configs) - ci), remaining / 2)
                    if t_alloc < 3:
                        break
                    a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc, **cfg)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final Grad {ci}: {score:.6f} *NEW BEST*")

        # Final LP
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            t_lp = min(remaining * 0.8, 40.0)
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
