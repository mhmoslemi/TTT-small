import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_value(a):
    """Compute objective: 2n * max(conv(a,a)) / (sum(a))^2. Assumes sum(a)=1."""
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf'), conv
    return 2.0 * len(a) * conv.max() / (s * s), conv


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns normalized direction."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def line_search(a, d, current_obj, n, max_evals=14):
    """Adaptive line search: coarse scan + golden section."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0

    for t in [0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 1.0]:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_value(a_try)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    if 0.1 < best_t < 0.9 and evals < max_evals - 4:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.01, best_t - 0.06)
        hi = min(0.99, best_t + 0.06)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)

        for _ in range(4):
            if evals >= max_evals:
                break
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            obj1 = obj_value(a1 / s1)[0] if s1 > 1e-15 else current_obj
            evals += 1

            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            obj2 = obj_value(a2 / s2)[0] if s2 > 1e-15 else current_obj
            evals += 1

            if obj1 < obj2:
                hi = d2
                d2 = d1
                obj2 = obj1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo = d1
                d1 = d2
                obj1 = obj2
                d2 = lo + gr * (hi - lo)

            if abs(hi - lo) < 0.003:
                break

        mid = (lo + hi) / 2.0
        a_mid = np.maximum(a + mid * d, 0.0)
        s_mid = a_mid.sum()
        if s_mid > 1e-15:
            obj_mid, _ = obj_value(a_mid / s_mid)
            evals += 1
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = mid

    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive constraint tightness."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    tightness = 0.9995

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * tightness
        g, g_sum = solve_lp_direction(a, M_lp, A_ub)

        if g_sum <= 1.0 + 1e-7:
            tightness = min(1.003, tightness * 1.004)
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)
                if obj_best < best_obj - 1e-13:
                    a_new = a + t_best * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    _, conv = obj_value(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.9985, tightness - 0.00015)
                    if verbose and iteration % 30 == 0:
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, "
                              f"t={time.monotonic()-t0:.1f}s")
                else:
                    no_improve += 1

        iteration += 1

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13 + perturb_count * 101)
            scale = 0.0015 + 0.001 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a.copy()
            else:
                a = a / s_new
            _, conv = obj_value(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)
            tightness = 1.001
            if perturb_count > 40:
                break

    return best_a, best_obj


def logsumexp_gradient_step(a, conv, beta, lr, v, momentum, nesterov_coeff=0.9):
    """One Nesterov-momentum step using log-sum-exp smoothed gradient."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()

    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += p[k] * 2.0 * a[k - idx]

    v = nesterov_coeff * v - lr * grad
    a_new = a + v
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new, v


def smooth_gradient_optimize(a0, time_limit, beta_init=50, lr_init=0.003,
                             momentum=0.92, beta_growth=1.001):
    """Log-sum-exp smooth gradient descent with Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        a, v = logsumexp_gradient_step(a, conv, beta, lr, v, momentum)
        _, conv = obj_value(a)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            beta = min(beta * beta_growth, 500.0)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.995
                beta = max(10.0, beta * 0.98)

        if no_improve >= 300:
            rng = np.random.default_rng(step_count * 3 + 7)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a / s_new
            v = np.zeros(n)
            _, conv = obj_value(a)
            no_improve = 0

    return best_a, best_obj


def peak_pair_descent(a0, time_limit, n_max_pairs=50):
    """Coordinate descent: directly reduce peak by adjusting contributing pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Find contributing indices
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        indices = np.arange(j_start, j_end + 1)

        if len(indices) < 2:
            break

        # Compute contribution of each index to the peak
        contributions = np.array([a[i] * a[k_star - i] for i in indices])
        total_contrib = contributions.sum()

        if total_contrib < 1e-15:
            break

        # Find non-peak indices to increase
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[indices] = True
        non_peak = np.where(~peak_mask)[0]

        if len(non_peak) == 0:
            break

        # Choose top contributors to decrease and random non-peak to increase
        n_decrease = min(5, len(indices))
        top_contrib_idx = indices[np.argsort(contributions)[-n_decrease:]]

        # Small adjustment amount
        adj = 0.001 * M / max(total_contrib, 1e-15)

        # Decrease peak contributors, increase non-peak elements
        a_new = a.copy()
        for idx in top_contrib_idx:
            a_new[idx] -= adj

        # Distribute increase to random non-peak elements
        n_incr = min(n_decrease, len(non_peak))
        incr_indices = np.random.default_rng(iteration).choice(non_peak, size=n_incr, replace=False)
        for idx in incr_indices:
            a_new[idx] += adj

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            _, conv_new = obj_value(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                conv = conv_new
                iteration += 1
            else:
                iteration += 1

        # Every 20 iterations, try a larger step
        if iteration % 20 == 0:
            adj_large = 0.005 * M / max(total_contrib, 1e-15)
            a_try = a.copy()
            for idx in top_contrib_idx:
                a_try[idx] -= adj_large
            for idx in incr_indices:
                a_try[idx] += adj_large
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                obj_try, conv_try = obj_value(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    conv = conv_try

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # Cosine bell
    cos_bell = 0.5 + 0.5 * np.cos(np.pi * x / n)
    inits['cosbell'] = cos_bell / cos_bell.sum()

    # Sine-modulated
    sine = 0.7 + 0.3 * np.sin(2.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()

    # Two-lobe cosine
    lobe = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()

    # Four-lobe
    lobe4 = 0.7 + 0.3 * np.cos(8.0 * np.pi * x / n)
    inits['flobe'] = lobe4 / lobe4.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random Dirichlet (concentrated)
    inits['random'] = rng_local.dirichlet(np.ones(n) * 0.3)

    # Flat with edges boosted
    edge = np.ones(n) * 0.8
    edge[:n//10] += 0.4
    edge[-n//10:] += 0.4
    inits['edge'] = edge / edge.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: (1) Multi-length quick scan, (2) Deep LP + smooth gradient,
    (3) Peak-pair descent polish, (4) Final structured perturbation refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    # ===== Phase 1: Multi-length Quick Scan =====
    print("=== Phase 1: Multi-length Quick Scan ===")
    t_start = time.monotonic()
    phase1_budget = 250.0
    phase1_deadline = t_start + phase1_budget

    lengths = [768, 1024, 1280, 1536]
    scan_results = []
    priority = ['parent', 'uniform', 'cosbell', 'tlobe', 'gauss', 'random', 'flobe', 'sine', 'edge']

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = get_inits(n, parent, rng)
        init_names = [nm for nm in priority if nm in inits][:4]

        t_per_init = max(15.0, (phase1_budget - 20.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep Optimization (LP + Smooth Gradient alternating) =====
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates from different lengths
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt.copy()))
            seen_lengths.add(n)
        if len(top_configs) >= 4:
            break

    if best_sequence is not None:
        best_n = len(best_sequence)
        if best_n not in seen_lengths:
            top_configs.append((best_n, best_sequence.copy()))

    # Reserve time for phases 3+4
    reserve_p34 = 400.0
    remaining_for_p2 = deadline - time.monotonic() - reserve_p34
    if remaining_for_p2 < 60:
        remaining_for_p2 = 60

    t_per_config = max(25.0, remaining_for_p2 / max(1, len(top_configs)))

    for ci, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - reserve_p34:
            break
        t_alloc = min(t_per_config, deadline - time.monotonic() - reserve_p34 + 30)
        if t_alloc < 12:
            break

        # Alternate LP and smooth gradient
        a_curr = a0.copy()
        phase_deadline = time.monotonic() + t_alloc

        while time.monotonic() < phase_deadline:
            # LP phase
            t_lp = min(40.0, phase_deadline - time.monotonic() - 20)
            if t_lp > 10:
                a_opt, _ = lp_optimize(a_curr, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P2 cfg{ci} LP (n={n}): {score:.6f}")
                a_curr = a_opt

            # Smooth gradient phase
            t_sg = min(25.0, phase_deadline - time.monotonic() - 10)
            if t_sg > 8:
                a_opt, _ = smooth_gradient_optimize(
                    a_curr, time_limit=t_sg,
                    beta_init=80, lr_init=0.004, momentum=0.93, beta_growth=1.002)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P2 cfg{ci} SG (n={n}): {score:.6f}")
                a_curr = a_opt

            # Small perturbation to escape
            if time.monotonic() < phase_deadline - 15:
                noise = rng.standard_normal(n) * 0.002
                a_pert = a_curr + noise
                a_pert = np.maximum(a_pert, 0.0)
                a_curr = a_pert / a_pert.sum()

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Peak-Pair Descent + Fine LP Polish =====
    print("=== Phase 3: Peak-Pair + Fine Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 40:
            # Peak-pair descent
            t_ppd = min(150.0, remaining * 0.5)
            if t_ppd > 20:
                a_opt, _ = peak_pair_descent(best_sequence, time_limit=t_ppd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P3 PPD: {score:.6f}")

            # Fine LP with very tight constraints
            t_lp = min(120.0, remaining - t_ppd - 20)
            if t_lp > 15:
                a_opt, _ = lp_optimize(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P3 Fine LP: {score:.6f}")

            # Smooth gradient with high beta
            t_sg = min(80.0, remaining - t_ppd - t_lp - 15)
            if t_sg > 10:
                a_opt, _ = smooth_gradient_optimize(
                    best_sequence, time_limit=t_sg,
                    beta_init=200, lr_init=0.002, momentum=0.95, beta_growth=1.0005)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P3 Fine SG: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Structured Perturbation + Refinement =====
    print("=== Phase 4: Final Refinement ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_rounds = max(1, int(remaining / 25))
            t_per = remaining / n_rounds

            for pi in range(n_rounds):
                if time.monotonic() > deadline - 12:
                    break
                if t_per < 5:
                    break

                # Structured perturbation: smooth + random
                scale = 0.001 * (1.0 + 0.2 * pi)
                x_freq = np.arange(n_best)
                smooth = np.sin(2 * np.pi * x_freq * rng.integers(1, 5) / n_best) * scale * 0.6
                smooth += np.cos(2 * np.pi * x_freq * rng.integers(1, 5) / n_best) * scale * 0.3
                noise = rng.standard_normal(n_best) * scale * 0.5

                a_pert = best_sequence + noise + smooth
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert / s_pert
                else:
                    a_pert = best_sequence.copy()

                t_alloc = min(t_per, deadline - time.monotonic() - 8)
                if t_alloc < 4:
                    break

                # Alternate methods
                if pi % 3 == 0:
                    a_opt, _ = lp_optimize(a_pert, time_limit=t_alloc)
                elif pi % 3 == 1:
                    a_opt, _ = smooth_gradient_optimize(
                        a_pert, time_limit=t_alloc,
                        beta_init=150, lr_init=0.003, momentum=0.94, beta_growth=1.001)
                else:
                    a_opt, _ = peak_pair_descent(a_pert, time_limit=t_alloc)

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4 round {pi} (n={n_best}): {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
