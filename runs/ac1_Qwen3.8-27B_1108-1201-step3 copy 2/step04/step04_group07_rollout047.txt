from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def obj_from_conv(a, conv):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)

def solve_minmax_lp(a, M_scale=1.0):

    Solve: min t
           s.t. C_a @ g <= t
                sum(g) = 1
                g >= 0
    Returns (g, t_opt) or None.

    n = len(a)
    C = build_conv_matrix(a)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_conv_a_g(a, g):
    """Compute a * g (full linear convolution)."""
    return np.convolve(a, g, mode='full')

        C[i:i + n, i] = a
    return C

For `i = n-1`, `(B@a)[n-1] = sum_k λ[k] * a[k-(n-1)]`. The valid terms are `k-(n-1) in [0, n-1]`, i.e., `k in [n-1, 2n-2]`. So it's `sum_{k=n-1}^{2n-2} λ[k] * a[k-n+1]`.

OK so the FFT approach should work. Let me implement it:

```python
def compute_B_a(lambda_norm, a, fft_size):
    """Compute B(λ) @ a using FFT."""
    n = len(a)
    # λ has length 2n-1, a has length n
    lam = np.zeros(fft_size)
    lam[:2*n-1] = lambda_norm
    ap = np.zeros(fft_size)
    ap[:n] = a
    # Cross-correlation: ifft(fft(λ) * conj(fft(a)))
    corr = np.fft.ifft(np.fft.fft(lam) * np.conj(np.fft.fft(ap))).real
    return corr[:n]  # We only need indices 0 to n-1
```
