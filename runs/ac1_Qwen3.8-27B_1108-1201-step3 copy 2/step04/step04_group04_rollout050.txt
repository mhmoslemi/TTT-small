import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = full_convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()

        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s), conv


def lp_solve_direction(a, M, conv_matrix=None):
    """

        conv_matrix = build_conv_matrix(a)

    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=conv_matrix, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            gs = g.sum()
            if gs > 1e-10:

    except Exception:
        pass
    return None, 0.0

def subgradient_optimize(a0, time_limit, num_averaging=50):
    """
    Subgradient method with Nesterov's averaging.
    Uses the exact subgradient of max_k b_k(a) = a[k*-i] pattern.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s
    n = len(a)

    best_obj, conv = compute_objective(a)
    best_a = a.copy()

    # Nesterov averaging: maintain a running average
    a_avg = a.copy()
    t0 = time.monotonic()
    t_iter = 1
    lr0 = 1.0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Compute subgradient at current point
        peak_k = int(np.argmax(conv))
        # Subgradient of b_{peak_k} w.r.t. a_i is a_{peak_k - i} (shifted/reversed)
        sg = np.zeros(n)
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if 0 <= j < n:
                sg[i] += a[j]

        # Project to tangent space of simplex: subtract mean
        sg = sg - sg.mean()

        # Learning rate (Nesterov's O(1/t) with averaging)
        lr = lr0 / np.sqrt(t_iter)

        # Gradient step
        a_new = a - lr * sg
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Update running average (Nesterov averaging)
        weight = 1.0 / np.sqrt(t_iter)
        a_avg = (a_avg * (np.sqrt(t_iter) - 1) + a_new * weight) / np.sqrt(t_iter) if t_iter > 1 else a_new.copy()
        a_avg = np.maximum(a_avg, 0)
        s_avg = a_avg.sum()
        if s_avg > 1e-10:
            a_avg = a_avg / s_avg

        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Also check the averaged point
        obj_avg, _ = compute_objective(a_avg)
        if obj_avg < best_obj - 1e-12:
            best_obj = obj_avg
            best_a = a_avg.copy()

        t_iter += 1

    return best_a, best_obj


def exp_gradient_optimize(a0, time_limit, lr_init=0.1, momentum=0.9):
    """
    Gradient descent in exponential parametrization: a_i = exp(theta_i) / Z.
    This automatically maintains a_i > 0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), np.inf
    a = a / s
    n = len(a)

    # Initialize theta
    theta = np.log(np.maximum(a, 1e-15))
    v = np.zeros(n)

    best_obj, conv = compute_objective(a)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = 0.01
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Compute a from theta
        theta_max = theta.max()
        a = np.exp(theta - theta_max)
        a = a / a.sum()

        conv = fast_convolve(a)

        # Soft-max weighted gradient
        top_k = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute d(conv)/d(a) for each peak, weighted
        dconv_da = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idxs = np.arange(i_start, i_end + 1)
            dconv_da[idxs] += w[idx] * a[k - idxs]

        # d(a)/d(theta_i) = a_i (a_i - a_i) ... actually:
        # a_i = exp(theta_i) / Z
        # d a_j / d theta_i = a_j (delta_{ij} - a_i)
        # So d(f)/d(theta_i) = sum_j (df/da_j)(a_j)(delta_{ij} - a_i)
        # = a_i * df/da_i - a_i * sum_j (df/da_j) a_j
        # = a_i * (dconv_da_i - dot(dconv_da, a))

        dot_term = np.dot(dconv_da, a)
        grad_theta = a * (dconv_da - dot_term)

        # Momentum update
        v = momentum * v - lr * grad_theta
        theta = theta + v

        # Periodic re-centering to prevent overflow
        if np.any(np.abs(theta) > 50):
            theta = theta - theta.mean()

        # Evaluate
        a_new = np.exp(theta - theta.max())
        a_new = a_new / a_new.sum()
        obj_new, _ = compute_objective(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        lr *= 0.999
        temp *= 1.002
        if no_improve >= 100:
            break

    return best_a, best_obj

                    continue
                # Pick a target: position that doesn't contribute to peaks
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta

                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()

                if obj_try < best_obj - 1e-10:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve >= max_no_improve:
            break

    return best_a, best_obj


def generate_initialization(n, parent, rng, init_type='uniform'):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)

    if init_type == 'uniform':
        a = np.ones(n) / n
    elif init_type == 'random':
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif init_type == 'sine':
        a = np.sin(np.pi * x / n) ** 2 + 0.01
        a = a / a.sum()
    elif init_type == 'bimodal':
        edge = max(1, n // 6)
        a = np.zeros(n)
        a[:edge] = 1.0
        a[-edge:] = 1.0
        a = a / a.sum()
    elif init_type == 'gaussian':
        a = np.exp(-0.5 * ((x - n * 0.5) / (n * 0.2)) ** 2)
        a = a / a.sum()
    elif init_type == 'alternating':
        a = np.ones(n)
        a[::2] = 1.3
        a[1::2] = 0.7
        a = a / a.sum()
    elif init_type == 'parent':
        if parent is not None and len(parent) > 0:
            p = np.maximum(parent, 0).astype(np.float64)
            ps = p.sum()
            if ps > 1e-10:
                if len(p) == n:
                    a = p / ps
                else:
                    a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
            else:
                a = np.ones(n) / n
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n

    return np.maximum(a, 0)


def interpolate_to_length(a, n_target):
    """Interpolate a sequence to a different length."""
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a / s
    a_new = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, len(a)), a_norm)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn > 1e-15:
        a_new = a_new / sn
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + subgradient + exponential gradient + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
            print(f"Parent: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass

    if best_sequence is None:
        # No parent available, use uniform
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Default: n=512, score={best_score:.6f}")

    # =========================================================
    # Phase 1: Quick exploration (200s)
    # =========================================================
    print("\n=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()

    # Try parent at its length and a few nearby lengths
    n_parent = len(best_sequence)
    lengths_to_try = sorted(set([n_parent, n_parent - 64, n_parent + 64, n_parent - 128, n_parent + 128]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048]

    t_per_init = max(15.0, 180.0 / (len(lengths_to_try) * 2))

    for n in lengths_to_try:
        if time.monotonic() > deadline - 1200:
            break

        # Try parent-based and one diverse init
        inits = []
        if parent is not None:
            inits.append(generate_initialization(n, parent, rng, 'parent'))
        inits.append(generate_initialization(n, parent, rng, 'random'))

        for init_a in inits:
            if time.monotonic() > deadline - 1100:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 1000)
            if t_alloc < 5:
                break

            # Quick subgradient
            a_opt, obj = subgradient_optimize(init_a, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SubGrad n={n}: {score:.6f} *NEW BEST*")

            # Quick exp-gradient
            if time.monotonic() < deadline - 1050:
                t_alloc2 = min(t_per_init * 0.7, deadline - time.monotonic() - 1000)
                if t_alloc2 > 5:
                    a_opt2, obj2 = exp_gradient_optimize(a_opt, time_limit=t_alloc2, lr_init=0.05, momentum=0.85)
                    score2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score2) and score2 < best_score:
                        best_score = score2
                        best_sequence = a_opt2.copy()
                        print(f"  ExpGrad n={n}: {score2:.6f} *NEW BEST*")

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LP optimization (900s)
    # =========================================================
    print("\n=== Phase 2: Deep LP Optimization ===")
    t_start2 = time.monotonic()

    # Run LP on the best sequence with multiple strategies
    n_best = len(best_sequence)

    # Strategy A: Direct LP from best
    t_alloc = min(400.0, deadline - time.monotonic() - 500)
    if t_alloc > 30:
        print(f"  LP direct: n={n_best}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP direct: {score:.6f} *NEW BEST*")
        else:
            print(f"  LP direct: {score:.6f}")

    # Strategy B: LP from a perturbed version
    if time.monotonic() < deadline - 450:
        t_alloc = min(250.0, deadline - time.monotonic() - 350)
        if t_alloc > 20:
            noise = rng.standard_normal(n_best) * 0.003
            a_perturbed = np.maximum(best_sequence + noise, 0)
            a_perturbed = a_perturbed / a_perturbed.sum()
            print(f"  LP perturbed: t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(a_perturbed, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP perturbed: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP perturbed: {score:.6f}")

    # Strategy C: LP from a slightly different length
    n_alt = n_best + 32
    if time.monotonic() < deadline - 400:
        t_alloc = min(200.0, deadline - time.monotonic() - 300)
        if t_alloc > 20:
            a_alt = interpolate_to_length(best_sequence, n_alt)
            print(f"  LP alt-length n={n_alt}: t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(a_alt, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP alt-length: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP alt-length: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Subgradient + Exp-Gradient refinement (300s)
    # =========================================================
    print("\n=== Phase 3: Gradient Refinement ===")
    t_start3 = time.monotonic()

    if time.monotonic() < deadline - 200:
        t_alloc = min(150.0, deadline - time.monotonic() - 150)
        if t_alloc > 10:
            # Subgradient with different lr
            a_opt, obj = subgradient_optimize(best_sequence, time_limit=t_alloc * 0.5, lr0_sim=2.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SubGrad refine: {score:.6f} *NEW BEST*")

        if time.monotonic() < deadline - 150:
            t_alloc = min(100.0, deadline - time.monotonic() - 100)
            if t_alloc > 10:
                a_opt, obj = exp_gradient_optimize(best_sequence, time_limit=t_alloc, lr_init=0.03, momentum=0.92)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  ExpGrad refine: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: LP micro-optimization rounds (200s)
    # =========================================================
    print("\n=== Phase 4: LP Micro-Optimization ===")
    t_start4 = time.monotonic()

    # Multiple short LP runs with small perturbations
    for round_i in range(5):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(40.0, (deadline - time.monotonic() - 80) / (5 - round_i))
        if t_alloc < 10:
            break

        # Small perturbation for diversity
        scale = 0.001 * (round_i + 1)
        noise = rng.standard_normal(len(best_sequence)) * scale
        a_start = np.maximum(best_sequence + noise, 0)
        a_start = a_start / a_start.sum()

        a_opt, obj = lp_optimize(a_start, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Micro LP {round_i}: {score:.6f} *NEW BEST*")
        else:
            print(f"  Micro LP {round_i}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Coordinate descent polish (remaining time)
    # =========================================================
    print("\n=== Phase 5: Coordinate Descent Polish ===")
    t_start5 = time.monotonic()

    if time.monotonic() < deadline - 20:
        t_alloc = min(80.0, deadline - time.monotonic() - 15)
        if t_alloc > 10:
            a_opt, obj = coordinate_descent_polish(best_sequence, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
            else:
                print(f"  CoordDesc: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Final: try one more LP pass if time allows
    # =========================================================
    if time.monotonic() < deadline - 15 and best_sequence is not None:
        t_final = min(15.0, deadline - time.monotonic() - 5)
        if t_final > 8:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
