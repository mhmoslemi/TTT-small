    """Coordinate descent with exact 1D line search."""
    a = a.copy()
    conv = conv.copy()
    best_obj, _ = compute_obj(a)
    t0 = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time_limit and time.monotonic() - t0 > time_limit:
            break
        
        # Random order
        order = np.random.permutation(n)
        
        improved = False

            # Find optimal delta for coordinate j
            # Affected k range: [j, j+n-1]
            k_start = j
            k_end = min(j + n - 1, 2*n - 2)

            if k_start > 0 or k_end < 2*n - 2:
                unaffected = np.concatenate([conv[:k_start], conv[k_end+1:]])
                max_unaff = np.max(unaffected) if len(unaffected) > 0 else 0.0

            # and conv[2j] + 2*delta*a[j] + delta^2 (if 2j in range)

            # The optimal delta is bounded: we want new a[j] >= 0, so delta >= -a[j]
            # And we don't want delta too large.
            
            s = a.sum()

            # Let's search in [-a[j]*0.5, a[j]*2] or so
            
            lo = -a[j] + 1e-15
            hi = a[j] * 3.0 + 1e-6

            gr = (np.sqrt(5) - 1) / 2

                if d < lo:
                    return float('inf')
                new_s = s + d
                if new_s <= 0:
                    return float('inf')
                new_max = max_unaff
                for k in range(k_start, k_end + 1):
                    if k == 2 * j and 2 * j <= 2 * n - 2:

                    else:
                        val = conv[k] + 2 * d * a[k - j]

                        new_max = val

            d1 = lo + (1 - gr) * (hi - lo)
            d2 = lo + gr * (hi - lo)
            f1 = obj_delta(d1)
            f2 = obj_delta(d2)

                    hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = lo + (1 - gr) * (hi - lo)
                    f1 = obj_delta(d1)
                else:
                    lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = lo + gr * (hi - lo)
                    f2 = obj_delta(d2)
                if abs(hi - lo) < 1e-10 * max(1, a[j]):
                    break
            
            best_d = (lo + hi) / 2
            best_f = obj_delta(best_d)
            
            cur_f = 2.0 * n * conv.max() / (s * s)

"""
Multi-method optimizer for minimizing 2n*max(autoconv(a))/(sum(a))^2.
Combines Adam-subgradient with Nesterov acceleration, exact coordinate descent,
and LP direction search. Uses multi-length exploration and adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    conv = fast_convolve(a)
    n = len(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub):
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def adam_subgrad_step(a, conv, n, m, v, beta1, beta2, t, lr, eps, top_k, temp):
    """One Adam subgradient step."""
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    m = beta1 * m + (1 - beta1) * sg
    v = beta2 * v + (1 - beta2) * sg ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new, m, v


def nesterov_subgrad_step(a, y, conv_y, n, v_mom, momentum, lr, top_k, temp):
    """Nesterov-accelerated subgradient step."""
    # Compute subgradient at y
    top_k = min(top_k, len(conv_y))
    peaks = np.argpartition(conv_y, -top_k)[-top_k:]
    peak_vals = conv_y[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * y[k - idx]

    # Nesterov update
    new_x = a - lr * sg
    new_x = np.maximum(new_x, 0.0)
    s_new = new_x.sum()
    if s_new > 1e-15:
        new_x = new_x / s_new
    return new_x


def adam_optimize(a0, time_limit, lr_init=0.003, beta1=0.9, beta2=0.999,
                  top_k=80, temp=0.0001, verbose=False):
    """Adam-based subgradient optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    _, conv = compute_obj(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = 0
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        a, m, v = adam_subgrad_step(a, conv, n, m, v, beta1, beta2, t + 1, lr, 1e-8, top_k, temp)
        _, conv = compute_obj(a)
        cur_obj = 2.0 * n * conv.max()
        t += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.95
            temp *= 1.005

        if no_improve >= 500:
            rng_local = np.random.default_rng(t * 7 + 3)
            noise = rng_local.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            m = np.zeros(n)
            v = np.zeros(n)
            _, conv = compute_obj(a)
            no_improve = 0

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum_init=0.5,
                      top_k=80, temp=0.0001, verbose=False):
    """Nesterov-accelerated subgradient optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    _, conv_y = compute_obj(y)
    best_obj = 2.0 * n * conv_y.max()
    best_a = a.copy()
    momentum = momentum_init

    t0 = time.monotonic()
    t = 0
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        a_new, _ = nesterov_subgrad_step(a, y, conv_y, n, a, momentum, lr, top_k, temp)
        # Update y for next iteration
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t += 1

        _, conv_y = compute_obj(y)
        _, conv_a = compute_obj(a)
        cur_obj = 2.0 * n * conv_a.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay learning rate
        lr = lr_init / np.sqrt(1 + t * 0.01)
        momentum = min(0.99, momentum_init + 0.001 * t)

        if no_improve % 200 == 0 and no_improve > 0:
            temp *= 1.01

        if no_improve >= 400:
            rng_local = np.random.default_rng(t * 11 + 5)
            noise = rng_local.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            y = a.copy()
            _, conv_y = compute_obj(y)
            no_improve = 0
            momentum = momentum_init

        if verbose and t % 500 == 0:
            print(f"    Nest iter {t}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def coord_descent_step(a, conv, n, j):
    """
    Perform one coordinate descent step for coordinate j.
    Returns updated a[j], updated conv, and improvement flag.
    """
    s = a.sum()
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)

    # Compute max of unaffected coefficients
    if k_start > 0 and k_end < 2 * n - 2:
        max_unaff = max(np.max(conv[:k_start]), np.max(conv[k_end + 1:]))
    elif k_start > 0:
        max_unaff = np.max(conv[:k_start])
    elif k_end < 2 * n - 2:
        max_unaff = np.max(conv[k_end + 1:])
    else:
        max_unaff = 0.0

    # Precompute affected values
    k_vals = np.arange(k_start, k_end + 1)
    a_idx = k_vals - j
    base_vals = conv[k_vals].copy()
    coeff_vals = a[a_idx].copy()

    # For k=2j, the formula is different (has delta^2 term)
    idx_2j = np.where(k_vals == 2 * j)[0]

    def obj_at_delta(d):
        new_s = s + d
        if new_s <= 1e-15:
            return float('inf')
        new_vals = base_vals + 2.0 * d * coeff_vals
        if len(idx_2j) > 0:
            new_vals[idx_2j] += d * d
        new_max = max(max_unaff, float(np.max(new_vals)))
        return 2.0 * n * new_max / (new_s * new_s)

    # Golden section search
    cur_f = 2.0 * n * conv.max() / (s * s)
    lo = -a[j] + 1e-12
    hi = a[j] * 2.5 + 1e-6

    gr = (np.sqrt(5) - 1) / 2
    d1 = lo + (1 - gr) * (hi - lo)
    d2 = lo + gr * (hi - lo)
    f1 = obj_at_delta(d1)
    f2 = obj_at_delta(d2)

    for _ in range(25):
        if f1 < f2:
            hi = d2
            d2 = d1
            f2 = f1
            d1 = lo + (1 - gr) * (hi - lo)
            f1 = obj_at_delta(d1)
        else:
            lo = d1
            d1 = d2
            f1 = f2
            d2 = lo + gr * (hi - lo)
            f2 = obj_at_delta(d2)
        if abs(hi - lo) < 1e-10 * max(1.0, a[j]):
            break

    best_d = (lo + hi) / 2.0
    best_f = obj_at_delta(best_d)

    if best_f < cur_f - 1e-15 and best_d > -a[j] + 1e-12:
        # Apply update
        delta = best_d
        for idx in range(len(k_vals)):
            k = k_vals[idx]
            if k == 2 * j:
                conv[k] += 2.0 * delta * coeff_vals[idx] + delta * delta
            else:
                conv[k] += 2.0 * delta * coeff_vals[idx]
        a[j] += delta
        return True
    return False


def coord_descent_optimize(a0, time_limit, max_sweeps=500, verbose=False):
    """Full coordinate descent optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng_local.permutation(n)
        improved = False

        for j in order:
            if coord_descent_step(a, conv, n, j):
                improved = True

        # Periodic recompute to avoid drift
        if (sweep + 1) % 5 == 0:
            conv = fast_convolve(a)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                conv = fast_convolve(a)
            else:
                a = a / s
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        if verbose and (sweep + 1) % 10 == 0:
            cur_obj = 2.0 * n * conv.max()
            print(f"    CD sweep {sweep+1}, obj={best_obj:.8f}, cur={cur_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if not improved:
            break

    # Final recompute
    conv = fast_convolve(a)
    final_obj = 2.0 * n * conv.max()
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = compute_obj(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    tightness = 0.9995
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * tightness
        g, g_sum = solve_lp_direction(a, M_lp, A_ub)

        if g_sum <= 1.0 + 1e-6:
            tightness = min(1.002, tightness * 1.003)
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Line search
                best_t = 0.0
                best_obj_ls = 2.0 * n * M
                for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try, _ = compute_obj(a_try)
                    if obj_try < best_obj_ls - 1e-14:
                        best_obj_ls = obj_try
                        best_t = t

                if best_t > 0:
                    a_new = a + best_t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    _, conv = compute_obj(a)
                    M = conv.max()
                    if 2.0 * n * M < best_obj:
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        no_improve = 0
                        tightness = max(0.999, tightness - 0.0001)
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        iteration += 1

        if no_improve >= 60:

            scale = 0.001 + 0.001 * min(perturb_count, 10)

            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv = compute_obj(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            tightness = 1.001
            if perturb_count > 25:
                break

    return best_a, best_obj


def resample_sequence(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_initializations(n, parent, seed):
    """Generate diverse initializations."""
    rng_local = np.random.default_rng(seed)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)

    # Sine modulated
    sine = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()

    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(np.pi * (x / n - 0.5) * 2)
    inits['raised'] = rc / rc.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Two Gaussian lobes
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    two_g = g1 + g2
    inits['two_gauss'] = two_g / two_g.sum()

    # Random smooth (B-spline like)
    base = rng_local.dirichlet(np.ones(n) * 0.5)
    # Smooth it a few times
    for _ in range(3):
        base = np.convolve(base, np.ones(5)/5, mode='same')
        base = np.maximum(base, 0)
        base = base / base.sum()
    inits['smooth_rand'] = base

    return inits

    print("=== Phase 3: Deep Refinement ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining_p3 = deadline - time.monotonic() - 100.0
        if remaining_p3 > 50:
            # LP deep on best
            t_lp = min(120.0, remaining_p3 * 0.3)
            if t_lp > 15:
                a_opt, _ = lp_optimize(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP deep: {score:.6f}")

            # CD deep
            remaining_p3 = deadline - time.monotonic() - 80.0
            t_cd = min(150.0, remaining_p3 * 0.4)
            if t_cd > 15:
                a_opt, _ = coord_descent_optimize(best_sequence, time_limit=t_cd)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CD deep: {score:.6f}")

            # Nesterov
            remaining_p3 = deadline - time.monotonic() - 60.0
            t_ne = min(100.0, remaining_p3 * 0.3)
            if t_ne > 10:
                a_opt, _ = nesterov_optimize(best_sequence, time_limit=t_ne,
                                             lr_init=0.003, momentum_init=0.7,
                                             top_k=100, temp=0.00005)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Nesterov: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Polish with Perturbation =====
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            n_best = len(best_sequence)
            # Several rounds of small perturbation + CD
            n_rounds = max(1, int(remaining / 25))
            t_per = remaining / n_rounds

            for ri in range(n_rounds):
                if time.monotonic() > deadline - 12:
                    break
                if t_per < 8:
                    break

                # Small structured perturbation
                scale = 0.001 * (1.0 + 0.5 * ri)
                noise = rng.standard_normal(n_best) * scale
                # Add smooth component
                smooth = np.sin(np.pi * np.arange(n_best) / n_best) * scale * 0.3
                a_pert = best_sequence + noise + smooth
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 8)
                if t_alloc < 5:
                    break

                # Use CD for final polish
                a_opt, _ = coord_descent_optimize(a_pert, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {ri}: {score:.6f}")

                # Also try LP on the perturbed
                if time.monotonic() < deadline - 15:
                    t_alloc2 = min(15.0, deadline - time.monotonic() - 8)
                    if t_alloc2 > 5:
                        a_opt2, _ = lp_optimize(a_pert, time_limit=t_alloc2)
                        score2 = float(evaluate_sequence(a_opt2.tolist()))
                        if np.isfinite(score2) and score2 < best_score:
                            best_score = score2
                            best_sequence = a_opt2.copy()
                            print(f"  LP-polish {ri}: {score2:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # ===== Final verification =====
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"\nFALLBACK: score={best_score:.6f}")

    return [float(x) for x in best_sequence]
