def evaluate_sequence(sequence: list[float]) -> float:
    """

    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""

    C = np.zeros((2 * n - 1, n))

    return C


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0."""

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        dual = None
        try:
            if result.ineqlin is not None:
                dual = result.ineqlin.marginals
        except Exception:
            pass
        if g_sum > 1e-10:
            return g / g_sum, g_sum, dual
    return a.copy(), 1.0, None

def sg_step(a, conv, lr, momentum, v, top_k, temp):
    """One momentum subgradient step with softmax-weighted peak selection."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    v = momentum * v - lr * sg
    a_new = a + v
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    return a_new, v

def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # Sine-modulated
    sine = 0.7 + 0.3 * np.sin(2.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()

    # Two-lobe cosine
    lobe = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()

    # Four-lobe
    lobe4 = 0.7 + 0.3 * np.cos(8.0 * np.pi * x / n)
    inits['flobe'] = lobe4 / lobe4.sum()

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng_local.dirichlet(np.ones(n) * 0.4)

    return inits
