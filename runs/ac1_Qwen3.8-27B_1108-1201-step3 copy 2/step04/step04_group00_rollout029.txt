import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad(a, conv, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv))) w.r.t. a."""
    n = len(a)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # Cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)[:n]

    idx = np.arange(js, je + 1)
    grad[idx] = a[k_star - idx]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp smoothing,
adaptive temperature annealing, multi-length exploration, and subgradient polishing.
Key improvement over LP-FW: FISTA is O(n log n) per iteration vs O(n^3) for LP,
allowing far more iterations within the time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, conv, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau * conv_k)) w.r.t. a.
    The gradient is the cross-correlation of softmax(tau*conv) with a.
    """
    n = len(a)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # Cross-correlation via FFT: (w ⊛ a)[:n]
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N)[:n]


def exact_subgradient(a, conv):
    """Exact subgradient at the argmax position of conv."""
    n = len(a)
    k_star = int(np.argmax(conv))
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    grad = np.zeros(n)
    idx = np.arange(js, je + 1)
    grad[idx] = a[k_star - idx]
    return grad


def fista_optimize(a0, time_limit, tau_start=5.0, tau_end=300.0,
                   step_init=0.5, verbose=False, seed_offset=0):
    """
    FISTA with log-sum-exp smoothing and adaptive temperature.
    Optimizes the smoothed approximation to min max_k (a*a)_k on the simplex.
    """
    rng = np.random.default_rng(12345 + seed_offset)
    a = project_simplex(np.maximum(a0, 0.0).astype(np.float64).copy())
    n = len(a)

    y = a.copy()
    t_k = 1.0
    tau = tau_start
    step = step_init

    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(y)
        peak = conv.max()

        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = y.copy()
            stagnation = 0
            if verbose and it % 500 == 0:
                print(f"      FISTA it={it}, peak={peak:.10f}, tau={tau:.1f}, step={step:.4f}")
        else:
            stagnation += 1

        # Compute smooth gradient
        grad = smooth_gradient(y, conv, tau)
        grad_norm = np.linalg.norm(grad)

        if grad_norm < 1e-14:
            break

        # FISTA update with backtracking
        a_new = project_simplex(y - grad / step)

        if not np.all(np.isfinite(a_new)):
            step *= 0.5
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y_new = project_simplex(y_new)

        # Check if step was too large (function increased significantly)
        conv_new = fast_convolve(y_new)
        if conv_new.max() > peak + 0.01 * step * grad_norm:
            step *= 0.7
            # Don't update, retry with smaller step
            if it % 500 == 0:
                print(f"      FISTA: step reduced to {step:.4f}")
            continue
        elif conv_new.max() < peak - 0.001:
            step = min(step * 1.05, 10.0)

        y = y_new
        a = a_new
        t_k = t_new

        # Temperature annealing
        if it % 100 == 0:
            tau = min(tau * 1.04, tau_end)

        # Stagnation handling: restart from best with perturbation
        if stagnation > 2000:
            pert = rng.standard_normal(n) * 0.003
            y = project_simplex(best_a + pert)
            a = y.copy()
            t_k = 1.0
            step = step_init
            tau = max(tau * 0.7, tau_start)
            stagnation = 0
            if verbose:
                print(f"      FISTA: restart at it={it}, tau={tau:.1f}")

        if it > 200000:
            break

    return best_a, best_peak


def sg_polish(a0, time_limit, lr=0.003, mom=0.93, top_k=20, tau=30.0,
              verbose=False, seed_offset=0):
    """
    Subgradient polishing with top-k peaks and momentum.
    Uses softmax-weighted top-k peaks for a smoother subgradient than exact argmax.
    """
    rng = np.random.default_rng(67890 + seed_offset)
    a = project_simplex(np.maximum(a0, 0.0).astype(np.float64).copy())
    n = len(a)

    v = np.zeros(n)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    cur_lr = lr
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peak = conv.max()

        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = a.copy()
            stagnation = 0
            if verbose and it % 1000 == 0:
                print(f"      SG it={it}, peak={best_peak:.10f}, lr={cur_lr:.6f}")
        else:
            stagnation += 1

        # Weighted subgradient using top-k
        k_actual = min(top_k, len(conv))
        peaks_idx = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks_idx]
        scaled = tau * (pv - pv.min())
        weights = np.exp(scaled)
        weights /= weights.sum()

        grad = np.zeros(n)
        for i in range(len(peaks_idx)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * a[k - idx]

        v = mom * v - cur_lr * grad
        a = project_simplex(a + v)

        cur_lr *= 0.99985
        if cur_lr < 1e-7:
            break

        # Restart on stagnation
        if stagnation > 3000:
            pert = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + pert)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0005)
            stagnation = 0

        if it > 300000:
            break

    return best_a, best_peak


def interp_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_init(n, kind, rng=None):
    """Generate an initial sequence of length n with the given shape."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        sigma = n / 5.0
        a = np.exp(-0.5 * (x - center) ** 2 / (sigma * sigma))
    elif kind == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * x / max(n - 1, 1)))
    elif kind == 'hann':
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    elif kind == 'bump':
        a = np.exp(-0.5 * (x - center) ** 2 / (n / 7.0) ** 2)
        a += 0.5 * np.exp(-0.5 * (x - center) ** 2 / (n / 2.5) ** 2)
    elif kind == 'plateau':
        half = n // 4
        a = np.ones(n)
        for i in range(n):
            d = abs(i - center)
            if d > half:
                a[i] = max(0.0, 1.0 - (d - half) / half)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a
