import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_direction(a, M, C=None):
    """Solve LP to find best direction."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x, result.x.sum(), True
    return None, 0, False

def gradient_step(a, conv, lr, top_k=50, temp=0.005):
    """One gradient step using softmax-weighted peaks."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = (peak_vals - peak_vals.max()) / temp
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if w[idx] < 1e-12:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def coordinate_descent(a0, time_limit, n_trials_per_batch=500):
    """Random coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(n_trials_per_batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            improved = False
            for delta in [0.005, 0.01, 0.02, 0.002, -0.005, -0.01, -0.02, -0.002]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_solve_direction(a, M, C):

        peak_vals = conv_y[peaks]
        
        # Softmax weighting
        max_pv = peak_vals.max()
        scaled = (peak_vals - max_pv) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum
        
        # Accumulate gradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]
        
        # Momentum update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9996
        temp *= 1.005
        mom = min(0.97, mom + 0.0003)
        
        if no_improve >= 250:
            break
    
    return best_a, best_obj


def optimize_annealing(a0, time_limit, n_restarts=5):
    """Simulated annealing for escaping local minima."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    total_time = time_limit
    temp_start = 0.05
    temp_end = 0.001
    
    prev_obj = cur_obj
    no_improve = 0
    
    while time.monotonic() - t0 < total_time:
        progress = (time.monotonic() - t0) / total_time
        temp = temp_start * (temp_end / temp_start) ** progress
        
        # Generate perturbation
        rng_local = np.random.default_rng()
        scale = temp * 0.1
        i = rng_local.integers(0, n)
        j = rng_local.integers(0, n)
        if i == j:
            continue
        
        delta = rng_local.uniform(-scale, scale)
        if a[i] + delta < 0 or a[j] - delta < 0:
            continue
        
        a_try = a.copy()
        a_try[i] += delta
        a_try[j] -= delta
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        
        # Accept/reject
        if obj_try < prev_obj or rng_local.random() < np.exp(-(obj_try - prev_obj) / temp):
            a = a_try
            prev_obj = obj_try
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_a, best_obj


def spectral_init(n, rng):
    """
    Generate initialization using spectral method:
    Create a flat power spectrum, then derive a non-negative sequence.
    """
    # Create a "flat" complex spectrum with random phases
    fft_size = 1 << (2 * n - 1).bit_length()
    freqs = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    freqs[0] = np.sqrt(n)  # DC component
    for k in range(1, fft_size // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        freqs[k] = np.exp(1j * phase)
    
    # Inverse FFT to get time-domain signal
    sig = np.fft.irfft(freqs, fft_size)
    
    # Take the window and make it non-negative
    a = sig[:n].copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum < n * 0.1:
        # Fallback: use absolute value
        a = np.abs(sig[:n])
        a_sum = a.sum()
    if a_sum < 1e-10:
        return np.ones(n) / n
    return a / a_sum


def structured_init(n, name, rng, parent=None):
    """Generate structured initializations."""
    x = np.arange(n, dtype=np.float64)
    
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'random':
        return rng.dirichlet(np.ones(n) * 0.5)
    elif name == 'sine1':
        v = np.sin(np.pi * x / n) ** 2 + 0.05
        return v / v.sum()
    elif name == 'sine2':
        v = np.sin(2 * np.pi * x / n) ** 2 + 0.05
        return v / v.sum()
    elif name == 'sine3':
        v = np.sin(3 * np.pi * x / n) ** 2 + 0.05
        return v / v.sum()
    elif name == 'bimodal':
        edge = max(1, n // 4)
        v = np.zeros(n)
        v[:edge] = 1.0
        v[-edge:] = 1.0
        return v / v.sum()
    elif name == 'gaussian':
        v = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
        return v / v.sum()
    elif name == 'gaussian_wide':
        v = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
        return v / v.sum()
    elif name == 'alternating':
        v = np.ones(n)
        v[::2] = 1.3
        v[1::2] = 0.7
        return v / v.sum()
    elif name == 'alternating2':
        v = np.ones(n)
        v[::3] = 1.5
        v[1::3] = 0.8
        v[2::3] = 0.5
        return v / v.sum()
    elif name == 'triangular':
        mid = n / 2.0
        v = 1.0 - np.abs(x - mid) / mid
        v = np.maximum(v, 0.05)
        return v / v.sum()
    elif name == 'parent':
        if parent is not None and len(parent) > 0:
            if len(parent) == n:
                v = parent.copy()
            else:
                v = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            v = np.maximum(v, 0)
            s = v.sum()
            if s > 1e-10:
                return v / s
        return np.ones(n) / n
    elif name == 'parent_perturbed':
        if parent is not None and len(parent) > 0:
            if len(parent) == n:
                v = parent.copy()
            else:
                v = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            v = np.maximum(v, 0)
            s = v.sum()
            if s > 1e-10:
                v = v / s
                v += rng.standard_normal(n) * 0.02
                v = np.maximum(v, 0)
                v /= v.sum()
                return v
        return rng.dirichlet(np.ones(n) * 0.5)
    elif name == 'spectral':
        return spectral_init(n, rng)
    else:
        return np.ones(n) / n


def coordinate_descent_polish(a0, time_limit):
    """Random pair coordinate descent for fine polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    
    deltas = [0.002, 0.005, 0.01, 0.02, 0.04]
    
    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(200000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            improved = False
            for d in deltas:
                for sign in [1, -1]:
                    delta = sign * d
                    if a[i] + delta < 0 or a[j] - delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] += delta
                    a_try[j] -= delta
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        conv = conv_try
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                no_improve += 1
        
        if no_improve > 1000:
            break
    
    return best_a, best_obj


# Global deadline for time management
deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Hybrid LP + gradient + annealing + coordinate descent.
    
    Strategy:
    - Phase 1: Quick multi-length, multi-init gradient scan (300s)
    - Phase 2: LP deep optimization on best candidates (800s)
    - Phase 3: Alternating gradient/LP on the best (300s)
    - Phase 4: Annealing + coordinate descent polishing (200s)
    - Phase 5: Final micro-polish (100s)
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    deadline_global = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    def update_best(a, label):
        nonlocal best_score, best_sequence
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  *** NEW BEST: {score:.6f} ({label}, n={len(a)}) ***")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
                print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
    except Exception:
        pass
    
    # =========================================================
    # Phase 1: Quick exploration (300s)
    # =========================================================
    print("\n=== Phase 1: Multi-length Gradient Exploration ===")
    t_p1 = time.monotonic()
    phase1_limit = 280.0
    
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for l in [512, 768, 1024, 1536]:
        if l not in lengths:
            lengths.append(l)
    lengths = [l for l in lengths if 128 <= l <= 2048]
    
    init_names_pool = ['parent', 'parent_perturbed', 'uniform', 'spectral', 'random', 'sine1', 'alternating', 'bimodal']
    
    results = []
    
    for n in lengths:
        remaining = deadline_global - time.monotonic()
        if remaining < 400:
            break
        
        t_per_init = max(8.0, min(40.0, phase1_limit / (len(lengths) * 3)))
        inits_done = 0
        
        for name in init_names_pool:
            if time.monotonic() - t_p1 > phase1_limit:
                break
            if inits_done >= 3:
                break
            remaining = deadline_global - time.monotonic()
            if remaining < 350:
                break
            
            t_alloc = min(t_per_init, remaining - 350)
            if t_alloc < 5:
                break
            
            a0 = structured_init(n, name, rng, parent)
            a0 = np.maximum(a0, 0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            inits_done += 1
            
            a_opt, obj = optimize_nesterov(a0, time_limit=t_alloc,
                                            lr_init=0.012, momentum=0.91,
                                            top_k=40, temp_init=0.003)
            score = obj
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                update_best(a_opt, f"P1 grad n={n} {name}")
    
    # Sort results
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_p1:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP Deep Optimization (800s)
    # =========================================================
    print("\n=== Phase 2: LP Deep Optimization ===")
    t_p2 = time.monotonic()
    phase2_limit = 750.0
    
    if best_sequence is not None:
        n_best = len(best_sequence)
        
        # Choose lengths for LP (prioritize best length, then try one larger)
        lp_targets = [n_best]
        for l in [1024, 1536, 2048]:
            if l not in lp_targets and l != n_best:
                lp_targets.append(l)
            if len(lp_targets) >= 3:
                break
        
        t_per_lp = max(60.0, phase2_limit / len(lp_targets) * 0.9)
        
        for n_lp in lp_targets:
            remaining = deadline_global - time.monotonic()
            if remaining < 200:
                break
            t_alloc = min(t_per_lp, remaining - 150)
            if t_alloc < 30:
                break
            
            if n_lp == n_best:
                a0 = best_sequence.copy()
            else:
                a0 = structured_init(n_lp, 'parent', rng, best_sequence)
            
            print(f"  LP: n={n_lp}, time={t_alloc:.0f}s")
            a_opt, obj = optimize_lp(a0, time_limit=t_alloc, verbose=True)
            update_best(a_opt, f"P2 LP n={n_lp}")
    
    print(f"Phase 2 done ({time.monotonic()-t_p2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Alternating Gradient + LP (300s)
    # =========================================================
    print("\n=== Phase 3: Alternating Refinement ===")
    t_p3 = time.monotonic()
    phase3_limit = 250.0
    
    if best_sequence is not None:
        n_cycles = 6
        t_per_cycle = max(10.0, phase3_limit / n_cycles)
        
        gradient_cfgs = [
            {'lr_init': 0.008, 'momentum': 0.92, 'top_k': 60, 'temp_init': 0.002},
            {'lr_init': 0.005, 'momentum': 0.94, 'top_k': 100, 'temp_init': 0.001},
            {'lr_init': 0.015, 'momentum': 0.89, 'top_k': 30, 'temp_init': 0.004},
        ]
        
        for i in range(n_cycles):
            remaining = deadline_global - time.monotonic()
            if remaining < 80:
                break
            t_alloc = min(t_per_cycle, remaining - 60)
            if t_alloc < 5:
                break
            
            if i % 2 == 0:
                cfg = gradient_cfgs[i % len(gradient_cfgs)]
                a_opt, obj = optimize_nesterov(best_sequence, time_limit=t_alloc, **cfg)
                update_best(a_opt, f"P3 grad{i}")
            else:
                a_opt, obj = optimize_lp(best_sequence, time_limit=t_alloc)
                update_best(a_opt, f"P3 LP{i}")
    
    print(f"Phase 3 done ({time.monotonic()-t_p3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Annealing + Coordinate Descent (150s)
    # =========================================================
    print("\n=== Phase 4: Annealing + CoordDescent ===")
    t_p4 = time.monotonic()
    phase4_limit = 130.0
    
    if best_sequence is not None:
        # Try annealing from best and perturbed
        t_anneal = min(40.0, phase4_limit * 0.4)
        if t_anneal > 5:
            a_opt, obj = optimize_annealing(best_sequence, time_limit=t_anneal)
            update_best(a_opt, "P4 anneal")
        
        # Coordinate descent
        remaining = deadline_global - time.monotonic()
        t_cd = min(50.0, remaining - 60)
        if t_cd > 5:
            a_opt, obj = coordinate_descent_polish(best_sequence, time_limit=t_cd)
            update_best(a_opt, "P4 coord")
        
        # Another anneal round if time permits
        remaining = deadline_global - time.monotonic()
        if remaining > 40:
            t_anneal2 = min(30.0, remaining - 30)
            if t_anneal2 > 5:
                a_opt, obj = optimize_annealing(best_sequence, time_limit=t_anneal2)
                update_best(a_opt, "P4 anneal2")
    
    print(f"Phase 4 done ({time.monotonic()-t_p4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Final Polish (remaining time)
    # =========================================================
    print("\n=== Phase 5: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline_global - time.monotonic()
        
        # Final LP micro-run
        t_final_lp = min(40.0, remaining * 0.4)
        if t_final_lp > 5:
            a_opt, obj = optimize_lp(best_sequence, time_limit=t_final_lp)
            update_best(a_opt, "P5 final_lp")
        
        # Final gradient micro-run
        remaining = deadline_global - time.monotonic()
        t_final_grad = min(20.0, remaining * 0.4)
        if t_final_grad > 3:
            a_opt, obj = optimize_nesterov(best_sequence, time_limit=t_final_grad,
                                            lr_init=0.003, momentum=0.95,
                                            top_k=200, temp_init=0.0005)
            update_best(a_opt, "P5 final_grad")
        
        # Final coord descent
        remaining = deadline_global - time.monotonic()
        t_final_cd = min(15.0, remaining * 0.5)
        if t_final_cd > 3:
            a_opt, obj = coordinate_descent_polish(best_sequence, time_limit=t_final_cd)
            update_best(a_opt, "P5 final_cd")
    
    # Final verification
    if best_sequence is not None:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    else:
        # Fallback
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"\nFALLBACK: score={best_score:.6f}, n=256")
    
    return [float(x) for x in best_sequence]
