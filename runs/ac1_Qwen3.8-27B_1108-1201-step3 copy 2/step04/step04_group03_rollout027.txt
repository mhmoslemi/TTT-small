    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Anneal beta
    betas = np.geomspace(beta_init, beta_end, n_beta_steps)
    
    for beta in betas:
        # Work in log-space
        x = np.log(a + 1e-30)  # log a
        # The actual a is softmax(x) = exp(x) / sum(exp(x))
        # But since sum(a)=1, we have a = exp(x - log(Z)) where Z = sum(exp(x))
        # For the gradient, we use the natural gradient formula
        
        def obj_and_grad(x):
            a_x = np.exp(x - np.log(np.sum(np.exp(x))))  # softmax
            a_x = np.maximum(a_x, 1e-30)
            conv = fast_convolve(a_x)
            scaled = beta * conv
            scaled -= scaled.max()
            lse = np.log(np.sum(np.exp(scaled)))
            obj = (scaled.max() + lse) / beta
            p = np.exp(scaled) / np.exp(lse)
            # Gradient w.r.t. a
            grad_a = compute_softmax_gradient(a_x, p)
            # Natural gradient: dF/dx_j = a_j * (grad_a_j - sum_k a_k * grad_a_k)
            mean_grad = np.dot(a_x, grad_a)
            grad_x = a_x * (grad_a - mean_grad)
            return obj, grad_x
        
        # Use L-BFGS-B on x
        from scipy.optimize import minimize as sp_minimize
        x0 = np.log(a + 1e-30)
        
        # Run L-BFGS with time limit
        result = sp_minimize(obj_and_grad, x0, method='L-BFGS-B',
                            jac=None,  # let it compute from func
                            options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 10000})
        
        if result.fun < obj_and_grad(x0)[0]:
            a_new = np.exp(result.x - np.log(np.sum(np.exp(result.x))))
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a_new.copy()
                a = a_new.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max L-BFGS with annealed temperature
2. Exact coordinate descent (1D convex subproblems)
3. Iterative refinement with increasing precision

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize as sp_minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, p):
    """Compute gradient of soft-max w.r.t. a. Returns 2*sum_k p[k]*a[k-j]."""
    n = len(a)
    nconv = len(p)
    # g[j] = sum_i a[i] * p[i+j] = (a * p_rev)[2n-2-j]
    p_rev = p[::-1]
    size = 1 << (nconv + n - 2).bit_length()
    A = rfft(a, size)
    P = rfft(p_rev, size)
    conv_result = irfft(A * P, size)
    # g[j] = conv_result[2n-2-j] for j=0,...,n-1
    g = conv_result[2 * n - 2 : n - 2 : -1]
    return 2.0 * g


def softmax_l_bfgs(a0, time_limit, beta_start=50, beta_end=5000, n_rounds=8, max_iter=300):
    """Optimize using soft-max approximation with L-BFGS on log-parametrization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = np.maximum(a, 1e-30) / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_rounds)
    
    for round_idx, beta in enumerate(betas):
        x = np.log(a)  # log parametrization
        
        def func_and_grad(x):
            # Softmax to get a from x
            x_shifted = x - x.max()
            a_x = np.exp(x_shifted)
            a_x = a_x / a_x.sum()
            a_x = np.maximum(a_x, 1e-30)
            
            conv = fast_convolve(a_x)
            scaled = beta * conv
            scaled -= scaled.max()
            lse = np.log(np.sum(np.exp(scaled)))
            obj = (scaled.max() + lse) / beta
            
            p = np.exp(scaled) / np.exp(lse)
            grad_a = compute_grad_a(a_x, p)
            # Natural gradient: dF/dx_j = a_j * (grad_a_j - dot(a, grad_a))
            mean_g = np.dot(a_x, grad_a)
            grad_x = a_x * (grad_a - mean_g)
            
            return obj, grad_x
        
        try:
            result = sp_minimize(func_and_grad, x, method='L-BFGS-B',
                               jac=None,
                               options={'maxiter': max_iter, 'gtol': 1e-10, 'maxfun': 5000})
            
            # Convert back to a
            x_new = result.x
            x_new = x_new - x_new.max()
            a_new = np.exp(x_new)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            
            print(f"    L-BFGS beta={beta:.0f}: obj={obj_new:.10f}, best={best_obj:.10f}")
            
        except Exception as e:
            print(f"    L-BFGS beta={beta:.0f} failed: {e}")
        
        if time.monotonic() - t0 > time_limit * 0.8:
            break
    
    return best_a, best_obj


def coord_descent(a0, n_passes=50, time_limit=None):
    """Exact coordinate descent for the normalized objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    conv = fast_convolve(a)
    S = a.sum()
    obj = conv.max() / (S * S)
    best_a = a.copy()
    best_obj = obj
    
    t0 = time.monotonic()
    nconv = 2 * n - 1
    
    def objective_delta(delta, conv, d, e, S):
        new_conv = conv + d * delta + e * delta * delta
        new_S = S + delta
        if new_S <= 1e-15:
            return np.inf
        return np.max(new_conv) / (new_S * new_S)
    
    for pass_idx in range(n_passes):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        for j in range(n):
            # Compute d and e for coordinate j
            d = np.zeros(nconv)
            k_min = j
            k_max = min(nconv - 1, j + n - 1)
            d[k_min:k_max + 1] = 2.0 * a[0:k_max + 1 - k_min]
            
            e = np.zeros(nconv)
            if 2 * j < nconv:
                e[2 * j] = 1.0
            
            # 1D optimization
            lo = -a[j] + 1e-12
            hi = max(2.0 * S, 0.1)
            
            res = minimize_scalar(objective_delta, bounds=(lo, hi), method='bounded',
                                args=(conv, d, e, S), options={'xatol': 1e-12})
            
            delta = res.x
            a[j] += delta
            
            # Recompute convolution (could be done incrementally, but FFT is fast enough)
            conv = fast_convolve(a)
            S = a.sum()
            obj = conv.max() / (S * S)
        
        # Check improvement
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Early stopping if no improvement
        if abs(obj - best_obj) < 1e-15:
            break
    
    return best_a, best_obj

def compute_grad_a(a, p):
    n = len(a)
    nconv = len(p)
    p_rev = p[::-1]
    size = 1 << (nconv + n - 2).bit_length()
    # Wait, I need size >= nconv + n - 1 for the linear convolution
    # nconv = 2n-1, so size >= 3n-2
    size = 1 << (3*n - 3).bit_length()
    A = rfft(a, size)
    P = rfft(p_rev, size)
    result = irfft(A * P, size)
    g = result[2*n-2 : n-2 : -1]
    return 2.0 * g
