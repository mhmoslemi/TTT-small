def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase approach:
1. FW-LP with warm-started LP and adaptive constraint relaxation
2. Top-k peak LSE gradient descent with adaptive beta and momentum

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cross_conv(weights, f):
    """Compute cross-correlation: result[i] = sum_m weights[i+m] * f[m]."""
    n = len(f)
    f_flip = f[::-1]
    total = (2 * n - 1) + (n - 1)
    fft_size = 1 << (total - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flip, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return conv[n - 1:n - 1 + n]


def build_A(f):
    """Build sparse constraint matrix for FW LP. A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    row_parts = []
    col_parts = []
    data_parts = []
    for j in range(n):
        r = np.arange(j, min(j + n, 2 * n - 1))
        length = len(r)
        row_parts.append(r)
        col_parts.append(np.full(length, j))
        data_parts.append(f[:length])
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_fw_lp(f, M_val, A_mat, timeout=20):
    """Solve LP: max sum(g) s.t. A@g <= M_val, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle, adaptive M multiplier, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A_mat = build_A(f)
        g = solve_fw_lp(f, M * m_mult, A_mat, timeout=lp_timeout)
        if g is None:
            m_mult = min(m_mult * 1.15, 8.0)
            if m_mult >= 8.0:
                m_mult = 1.0
                no_improve += 1
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_mult = min(m_mult * 1.05, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_mult = min(m_mult * 1.08, 6.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 50) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            no_improve = 0
            if it % 20 == 0:
                print(f"    FW restart {no_improve//50}, obj={best_obj:.12f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0, top_k=None):
    """Log-sum-exp smoothed gradient descent with top-k peak selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    if top_k is None:
        top_k = max(16, n // 4)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Top-k peak selection with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = beta * (pv - pv.max())
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # Build weight vector for full convolution
        w_full = np.zeros(len(conv))
        for i, k in enumerate(peaks):
            w_full[k] = w[i]

        # Gradient via FFT
        grad = compute_cross_conv(w_full, f)
        grad *= 2.0

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Adam-like momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.93
                momentum = max(0.87, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0 / n, 0.001)
                beta = min(beta * 1.3, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, no_impr={no_improve}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def peak_target_optimize(f0, time_limit, n_peaks=8):
    """Direct peak-targeting: iteratively reduce the highest convolution peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 1.0 / n
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Select top peaks
        if n_peaks >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        # Compute subgradient: weighted sum of shifted f
        pv = conv[peaks]
        # Use temperature-based weighting
        temp = 0.001 + 0.002 * (1.0 + np.cos(np.pi * min(it / 500.0, 1.0)))
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            idx = np.arange(j_lo, j_hi + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f_new = project_simplex(f - lr * sg)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 3.0 / n)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(it * 13 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = max(1.0 / n, 0.0005)
                no_improve = 0

        if it % 300 == 0:
            print(f"    Peak it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def symmetrize(f):
    """Make sequence symmetric: f[i] = (f[i] + f[n-1-i]) / 2."""
    n = len(f)
    f_sym = 0.5 * (f + f[::-1])
    f_sym = np.maximum(f_sym, 0.0)
    s = f_sym.sum()
    if s > 1e-15:
        f_sym = f_sym / s
    return f_sym


def coord_refine(f0, time_limit, n_steps=500):
    """Coordinate descent: try moving mass between pairs of positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving epsilon from i to j (and vice versa)
        for direction in [1, -1]:
            eps_candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
            for eps in eps_candidates:
                if f[i] < eps and direction == 1:
                    continue
                if f[j] < eps and direction == -1:
                    continue
                f_trial = f.copy()
                f_trial[i] -= direction * eps
                f_trial[j] += direction * eps
                if f_trial[i] < -1e-12 or f_trial[j] < -1e-12:
                    continue
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_f = f_trial.copy()
                    f = f_trial
                    break

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    return f_new

def fast_convolve(a):

def project_simplex(v):

    n = len(v)

        print(f"  Coord refine: {t_alloc:.0f}s")
        a_opt, obj = coord_refine(current_best, time_limit=t_alloc, n_steps=200)
        try_sequence(a_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(60, deadline - time.monotonic() - 20)
    if t_alloc > 10:
        a_opt, obj = lse_optimize(current_best, time_limit=t_alloc,
                                  beta_start=500.0, beta_end=8000.0,
                                  top_k=max(4, len(current_best) // 15))
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
