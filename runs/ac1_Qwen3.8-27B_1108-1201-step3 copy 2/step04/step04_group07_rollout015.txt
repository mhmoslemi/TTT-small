            )
            x = result.x
            a = softmax_from_x(x)
            obj = true_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        except Exception:
            pass
    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample sequence to different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def subgradient_polish(a0, time_limit, lr_init=0.004, momentum=0.95, top_k=60, temp=0.001):
    """Subgradient descent with momentum for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        top_k_eff = min(top_k, len(c))
        peaks = np.argpartition(c, -top_k_eff)[-top_k_eff:]
        peak_vals = c[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
        c = fast_convolve(a)
        cur_obj = 2.0 * n * c.max()
        step_count += 1
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve % 80 == 0:
            lr *= 0.98
        if no_improve >= 400:
            rng = np.random.default_rng(step_count * 3 + 7)
            noise = rng.standard_normal(n) * 0.0015
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            c = fast_convolve(a)
            no_improve = 0
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0, sum(g)=1."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def line_search(a, d, current_obj, n, max_evals=15):
    """Adaptive line search: coarse scan then golden section."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0
    candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try = true_objective(a_try)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    if 0.03 < best_t < 0.97 and evals < max_evals - 6:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = max(0.005, best_t - 0.06)
        hi = min(0.995, best_t + 0.06)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)
        obj1, obj2 = current_obj, current_obj
        for _ in range(5):
            if evals >= max_evals:
                break
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            obj1 = true_objective(a1 / s1) if s1 > 1e-15 else current_obj
            evals += 1
            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            obj2 = true_objective(a2 / s2) if s2 > 1e-15 else current_obj
            evals += 1
            if obj1 < obj2:
                hi, d2, obj2 = d2, d1, obj1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo, d1, obj1 = d1, d2, obj2
                d2 = lo + gr * (hi - lo)
            if abs(hi - lo) < 0.003:
                break
    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive constraint tightness."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    tightness = 0.9995

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2.0:
            break
        A_ub = build_conv_matrix(a)
        g, g_sum = solve_lp_direction(a, M * tightness, A_ub)
        if g_sum <= 1.0 + 1e-7:
            tightness = min(1.001, tightness * 1.003)
            no_improve += 1
        else:
            d = g - a
            if np.linalg.norm(d) < 1e-13:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)
                if obj_best < best_obj - 1e-13:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    c = fast_convolve(a)
                    M = c.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.999, tightness - 0.0001)
                else:
                    no_improve += 1
        iteration += 1
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + perturb_count * 101)
            scale = 0.002 + 0.001 * min(perturb_count, 5)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            c = fast_convolve(a)
            M = c.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)
            tightness = 1.0005
            if perturb_count > 25:
                break
    return best_a, best_obj


def get_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)
    sine = 0.7 + 0.3 * np.sin(2.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()
    lobe = 0.6 + 0.4 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()
    lobe4 = 0.7 + 0.3 * np.cos(8.0 * np.pi * x / n)
    inits['flobe'] = lobe4 / lobe4.sum()
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    # U-shaped (high at edges)
    u = 0.5 + 0.5 * (2.0 * x / (n - 1) - 1.0) ** 2
    inits['u_shape'] = u / u.sum()
    # Sine squared
    sq = 0.5 + 0.5 * np.sin(np.pi * x / n) ** 2
    inits['sine_sq'] = sq / sq.sum()
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Fast L-BFGS-B smooth optimization on diverse inits
    Phase 2: Deep LP refinement on best candidates
    Phase 3: Subgradient + LP alternating polish
    Phase 4: Final perturbation and refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # ===== Phase 1: Fast L-BFGS-B smooth optimization =====
    print("=== Phase 1: L-BFGS-B Smooth Optimization ===")
    t_start = time.monotonic()
    phase1_budget = 200.0
    phase1_deadline = t_start + phase1_budget

    lengths = [512, 768, 1024, 1536]
    T_schedule = [10, 25, 50, 100, 200, 400]
    all_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break
        inits = get_inits(n, parent, rng)
        priority = ['parent', 'uniform', 'sine', 'tlobe', 'flobe', 'gauss', 'u_shape', 'sine_sq', 'tri', 'random']
        init_names = [nm for nm in priority if nm in inits]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 5:
                break
            a0 = inits[name]
            a_opt, obj_opt = lbfgs_smooth_optimize(a0, T_schedule)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f} ({time.monotonic()-t_start:.1f}s)")

    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.1f}s). Best: {best_score:.6f}")

    # ===== Phase 2: L-BFGS-B with more temperatures + local search =====
    print("=== Phase 2: Enhanced L-BFGS-B + Local Search ===")
    t_start2 = time.monotonic()
    phase2_budget = 200.0
    phase2_deadline = t_start2 + phase2_budget

    # Take top results and try harder
    top_indices = list(range(min(10, len(all_results))))
    # Ensure we have diverse lengths
    seen = set()
    for idx in sorted(top_indices, key=lambda i: all_results[i][0]):
        if all_results[idx][1] not in seen:
            seen.add(all_results[idx][1])
        if len(seen) >= 4:
            break

    enhanced_T = [5, 15, 30, 60, 120, 250, 500, 1000]
    for idx in top_indices:
        if time.monotonic() > phase2_deadline - 10:
            break
        score, n, name, a_opt = all_results[idx]
        # Try with enhanced schedule
        a_enh, obj_enh = lbfgs_smooth_optimize(a_opt, enhanced_T, max_iter=500)
        score_enh = float(evaluate_sequence(a_enh.tolist()))
        if np.isfinite(score_enh) and score_enh < best_score:
            best_score = score_enh
            best_sequence = a_enh.copy()
            print(f"  Enhanced L-BFGS n={n}, {name}: {score_enh:.6f}")

        # Local search: perturb in x-space and re-optimize
        if time.monotonic() < phase2_deadline - 15:
            x = np.log(np.maximum(a_opt, 1e-25))
            for trial in range(3):
                if time.monotonic() > phase2_deadline - 5:
                    break
                pert_r = 0.1 / (trial + 1)
                x_pert = x + rng.standard_normal(n) * pert_r
                a_pert = softmax_from_x(x_pert)
                a_ls, obj_ls = lbfgs_smooth_optimize(a_pert, enhanced_T[:5], max_iter=200)
                score_ls = float(evaluate_sequence(a_ls.tolist()))
                if np.isfinite(score_ls) and score_ls < best_score:
                    best_score = score_ls
                    best_sequence = a_ls.copy()
                    print(f"  Local search n={n}, {name} trial{trial}: {score_ls:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.1f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Deep LP refinement =====
    print("=== Phase 3: Deep LP Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining_p3 = deadline - time.monotonic() - 300.0
        if remaining_p3 < 50:
            remaining_p3 = 50
        t_per_lp = max(20.0, remaining_p3 / 4.0)

        # LP on best sequence
        n_best = len(best_sequence)
        a_lp, obj_lp = lp_optimize(best_sequence, time_limit=min(t_per_lp, remaining_p3 - 10), verbose=True)
        score_lp = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score_lp) and score_lp < best_score:
            best_score = score_lp
            best_sequence = a_lp.copy()
            print(f"  LP on best (n={n_best}): {score_lp:.6f}")

        # LP on other top results with different lengths
        for idx in range(len(all_results)):
            if time.monotonic() > deadline - 300:
                break
            score_r, n_r, name_r, a_r = all_results[idx]
            if n_r == n_best:
                continue
            if score_r > best_score * 1.01:
                continue
            t_alloc = min(t_per_lp * 0.5, deadline - time.monotonic() - 280)
            if t_alloc < 10:
                break
            a_lp2, _ = lp_optimize(a_r, time_limit=t_alloc)
            score_lp2 = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score_lp2) and score_lp2 < best_score:
                best_score = score_lp2
                best_sequence = a_lp2.copy()
                print(f"  LP n={n_r}, {name_r}: {score_lp2:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.1f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Subgradient + L-BFGS-B alternating polish =====
    print("=== Phase 4: Alternating Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # SG configs
        sg_configs = [
            {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 80, 'temp': 0.0008},
            {'lr_init': 0.005, 'momentum': 0.93, 'top_k': 50, 'temp': 0.002},
            {'lr_init': 0.002, 'momentum': 0.97, 'top_k': 120, 'temp': 0.0005},
        ]

        t_per = remaining / (len(sg_configs) + 3.0)

        for ci, cfg in enumerate(sg_configs):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 25)
            if t_alloc < 8:
                break
            a_sg, _ = subgradient_polish(best_sequence, time_limit=t_alloc, **cfg)
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                print(f"  SG {ci}: {score_sg:.6f}")

            # Follow up with L-BFGS-B
            if time.monotonic() < deadline - 20:
                a_lb, _ = lbfgs_smooth_optimize(best_sequence, [50, 100, 200, 500], max_iter=300)
                score_lb = float(evaluate_sequence(a_lb.tolist()))
                if np.isfinite(score_lb) and score_lb < best_score:
                    best_score = score_lb
                    best_sequence = a_lb.copy()
                    print(f"  L-BFGS after SG {ci}: {score_lb:.6f}")

        # Short LP polish
        if time.monotonic() < deadline - 15:
            t_alloc = min(30.0, deadline - time.monotonic() - 10)
            if t_alloc > 8:
                a_lp3, _ = lp_optimize(best_sequence, time_limit=t_alloc)
                score_lp3 = float(evaluate_sequence(a_lp3.tolist()))
                if np.isfinite(score_lp3) and score_lp3 < best_score:
                    best_score = score_lp3
                    best_sequence = a_lp3.copy()
                    print(f"  Final LP: {score_lp3:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.1f}s). Best: {best_score:.6f}")

    # ===== Phase 5: Final perturbation attempts =====
    print("=== Phase 5: Final Perturbations ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            n_best = len(best_sequence)
            n_attempts = max(1, int(remaining / 15))
            t_per = remaining / n_attempts

            for pi in range(n_attempts):
                if time.monotonic() > deadline - 5:
                    break
                if t_per < 5:
                    break
                # Diverse perturbations
                if pi % 3 == 0:
                    noise = rng.standard_normal(n_best) * 0.002
                    a_pert = best_sequence + noise
                elif pi % 3 == 1:
                    # Smooth perturbation
                    freq = rng.uniform(1, 5)
                    phase = rng.uniform(0, 2 * np.pi)
                    smooth = np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase) * 0.003
                    a_pert = best_sequence + smooth
                else:
                    # Block perturbation
                    block = max(1, n_best // 8)
                    start = rng.integers(0, n_best - block)
                    a_pert = best_sequence.copy()
                    a_pert[start:start + block] *= (1 + rng.uniform(-0.05, 0.05))

                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                # Quick L-BFGS-B
                a_final, _ = lbfgs_smooth_optimize(a_pert, [30, 80, 200, 500], max_iter=200)
                score_f = float(evaluate_sequence(a_final.tolist()))
                if np.isfinite(score_f) and score_f < best_score:
                    best_score = score_f
                    best_sequence = a_final.copy()
                    print(f"  Final pert {pi}: {score_f:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.1f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
