    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(b, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*b_k))) w.r.t. a."""
    # This returns the weights w_k = exp(beta*b_k)/sum(exp(beta*b_l))
    shifted = beta * b - beta * b.max()
    w = np.exp(shifted)
    w /= w.sum()
    return w

def compute_grad(a, w):
    """Compute gradient of smooth max w.r.t. a, given weights w on b_k."""
    n = len(a)
    m = len(w)  # 2n-1
    grad = np.zeros(n)

        je = min(n - 1, k)
        for j in range(js, je + 1):

    return grad

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fast(a, w):
    """
    Compute gradient of smooth max at a, given softmax weights w on b_k.
    grad[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l] (for j+l < 2n-1)
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    m = len(w)
    N = 1 << (m + n - 1 - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    h = np.fft.ifft(W * np.conj(A)).real[:n]
    return 2.0 * h


def smooth_obj(b, beta):
    """Compute (1/beta)*log(sum(exp(beta*b_k))) - beta*b_max (numerically stable)."""
    shifted = beta * (b - b.max())
    return (beta * b.max() + np.log(np.sum(np.exp(shifted)))) / beta


def freq_domain_deflation(a, eta=0.3, n_peaks=3):
    """
    Frequency-domain peak deflation: reduce |A(f)| at frequencies that
    constructively interfere at the peak positions of a*a.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(A * A, fft_size)[:size]

    # Find top peaks
    if n_peaks >= size:
        peak_indices = np.arange(size)
    else:
        peak_indices = np.argpartition(b, -n_peaks)[-n_peaks:]

    # For each peak position, compute the "constructive" frequency mask
    reduction = np.zeros(fft_size // 2 + 1)
    for k in peak_indices:
        # Contribution of freq f to b[k]: Re(|A[f]|^2 * exp(2i*angle(A[f])) * exp(2*pi*i*f*k/fft_size))
        # = |A[f]|^2 * cos(2*angle(A[f]) + 2*pi*f*k/fft_size)
        angles = 2.0 * np.angle(A)
        phases = 2.0 * np.pi * np.arange(fft_size // 2 + 1, dtype=np.float64) * k / fft_size
        cos_val = np.cos(angles + phases)
        # Only reduce where constructive (positive contribution)
        reduction += np.maximum(cos_val, 0.0)

    # Normalize reduction
    rmax = reduction.max()
    if rmax < 1e-15:
        return a.copy()
    reduction /= rmax

    # Apply: reduce |A| where reduction is high
    scale = 1.0 - eta * reduction
    scale = np.maximum(scale, 0.1)  # Don't reduce to zero
    A_new = A * scale

    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def coordinate_descent_step(a, b, n_iter_per_coord=2, top_k_coords=50):
    """
    Coordinate descent: for each of the top-k most important coordinates,
    find the optimal scalar value that minimizes the max autocorrelation.
    """
    n = len(a)
    # Identify coordinates that contribute most to the peak
    peak_pos = np.argmax(b)
    # Coordinates contributing to peak: j and peak_pos - j for valid j
    contrib = np.zeros(n)
    js = max(0, peak_pos - n + 1)
    je = min(n - 1, peak_pos)
    for j in range(js, je + 1):
        contrib[j] = a[peak_pos - j]
        contrib[peak_pos - j] += a[j]

    # Select top coordinates to optimize
    top_coords = np.argsort(contrib)[-min(top_k_coords, n):]

    for j in top_coords:
        # Binary search for optimal a[j]
        best_val = a[j]
        best_obj = b.max()
        # Try a range of values
        for scale in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.3]:
            trial = a.copy()
            trial[j] = a[j] * scale
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            b_trial = fast_convolve(trial)
            obj = b_trial.max()
            if obj < best_obj:
                best_obj = obj
                best_val = a[j] * scale
        a[j] = best_val / a.sum() * a.sum()  # Keep sum consistent
        a = a / a.sum()

    return a


def smooth_sg_nesterov(a0, time_limit, beta_init=50.0, beta_end=500.0,
                        lr_init=0.02, momentum=0.9, deflation_interval=20):
    """
    Smooth-subgradient with Nesterov momentum and periodic frequency-domain deflation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    it = 0
    cur_beta = beta_init
    cur_lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()

        # Update beta (anneal)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        cur_beta = beta_init * (beta_end / beta_init) ** progress

        # Compute softmax weights
        shifted = cur_beta * (b - b.max())
        w = np.exp(shifted)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= wsum

        # Compute gradient
        grad = compute_grad_fast(a, w)

        # Nesterov momentum step
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        # Periodic frequency-domain deflation
        if it % deflation_interval == 0:
            a_deflated = freq_domain_deflation(a_new, eta=0.2 + 0.1 * progress, n_peaks=5)
            # Blend
            a_new = 0.7 * a_new + 0.3 * a_deflated
            a_new = project_simplex(a_new)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        cur_lr *= 0.9997

        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(it * 7919 + 13)
            if no_improve < 200:
                # Small perturbation
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(best_a + noise)
            else:
                # Larger perturbation with freq deflation
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(best_a + noise)
                a = freq_domain_deflation(a, eta=0.5, n_peaks=3)
            v = np.zeros(n)
            cur_lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj

def multiplicative_weight_step(a0, time_limit, lr_init=0.03, top_k=15, temp_init=0.0004):
    """Multiplicative weight (mirror descent) optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)

        k = min(top_k, len(b))
        peaks = np.argpartition(b, -k)[-k:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-14:
                continue
            k_idx = int(peaks[i])
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * a[k_idx - idx]
            log_update[idx] -= cur_lr * grad_contrib

        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()

        b = fast_convolve(a)
        obj = b.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 120:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            a = best_a * np.exp(rng.standard_normal(n) * 0.005)
            a = a / a.sum()
            cur_lr = max(lr_init * 0.6, 0.01)
            cur_temp = temp_init * 2.0
            no_improve = 0

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.03
    inits.append(c1 / c1.sum())

    # Double peak
    dp = np.exp(-((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(dp / dp.sum())

    # Sine squared
    ss = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(ss / ss.sum())

    # Alternating (chirp-like)
    chirp = 1.0 + 0.3 * np.cos(4 * np.pi * x / n)
    inits.append(chirp / chirp.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

    Main entry point. Multi-phase search:
    Phase 1: Multi-length, multi-start exploration
    Phase 2: Intensive smooth-SG + frequency deflation on best candidate
    Phase 3: MW polish
    Phase 4: Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Multi-start exploration (~400s) ===
    print("\n=== Phase 1: Multi-Start Exploration ===")

    lengths_to_try = [384]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
    # Add some alternative lengths
    for alt_n in [320, 448, 512]:
        if alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)

    # Deduplicate and sort
    lengths_to_try = sorted(set(lengths_to_try))

    time_per_start = 60
    max_starts = 8

    start_count = 0
    for n in lengths_to_try:
        if start_count >= max_starts:
            break
        rem = deadline - time.monotonic() - 800
        if rem < 30:
            break

        inits = generate_diverse_inits(n, parent, rng)
        # Try up to 2 initializations per length
        for ii in range(min(2, len(inits))):
            start_count += 1
            rem = deadline - time.monotonic() - 700
            if rem < 20:
                break
            t_alloc = min(time_per_start, rem)
            init = inits[ii]
            print(f"  n={n}, init={ii}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_sg_nesterov(init, time_limit=t_alloc,
                                           beta_init=30.0, beta_end=300.0,
                                           lr_init=0.015, momentum=0.9,
                                           deflation_interval=15)
            try_seq(a_opt, f"SG_n{n}_i{ii}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Intensive optimization on best (~800s) ===
    print("\n=== Phase 2: Intensive Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Long smooth-SG with deflation
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 500)
            print(f"  Intensive SG+deflation, n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_sg_nesterov(best_seq, time_limit=t_alloc,
                                           beta_init=80.0, beta_end=800.0,
                                           lr_init=0.01, momentum=0.92,
                                           deflation_interval=12)
            try_seq(a_opt, "intensive_sg")

        # Try perturbed version
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_alloc = min(rem, 200)
            noise = rng.standard_normal(n_best) * 0.008
            perturbed = project_simplex(best_seq + noise)
            perturbed = freq_domain_deflation(perturbed, eta=0.4, n_peaks=5)
            print(f"  Perturbed SG+deflation, t={t_alloc:.0f}s")
            a_opt, _ = smooth_sg_nesterov(perturbed, time_limit=t_alloc,
                                           beta_init=50.0, beta_end=500.0,
                                           lr_init=0.008, momentum=0.91,
                                           deflation_interval=10)
            try_seq(a_opt, "perturbed_sg")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multiplicative weight polish (~250s) ===
    print("\n=== Phase 3: MW Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        mw_configs = [
            {'lr_init': 0.02, 'top_k': 12, 'temp_init': 0.0003},
            {'lr_init': 0.04, 'top_k': 20, 'temp_init': 0.0005},
            {'lr_init': 0.01, 'top_k': 8, 'temp_init': 0.0002},
        ]
        for ci, cfg in enumerate(mw_configs):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(80, rem)
            print(f"  MW config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weight_step(
                best_seq, time_limit=t_alloc,
                lr_init=cfg['lr_init'], top_k=cfg['top_k'],
                temp_init=cfg['temp_init'])
            try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final fine polish (~150s) ===
    print("\n=== Phase 4: Final Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine smooth-SG
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = smooth_sg_simple(best_seq, time_limit=t_alloc,
                                          beta=500.0, lr_init=0.003, momentum=0.95)
            try_seq(a_opt, "fine_sg")

        # Frequency deflation + fine SG
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            deflated = freq_domain_deflation(best_seq, eta=0.15, n_peaks=5)
            a_opt, _ = smooth_sg_simple(deflated, time_limit=min(rem, 30),
                                          beta=400.0, lr_init=0.002, momentum=0.94)
            try_seq(a_opt, "deflated_fine")

        # Tiny perturbations with quick optimization
        rem = deadline - time.monotonic() - 20
        for pi in range(4):
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            pert = project_simplex(best_seq + noise)
            t_alloc = min(15, rem - 5)
            if t_alloc < 5:
                break
            a_opt, _ = smooth_sg_simple(pert, time_limit=t_alloc,
                                          beta=600.0, lr_init=0.001, momentum=0.96)
            try_seq(a_opt, f"micro_pert_{pi}")
            rem = deadline - time.monotonic() - 20

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
