```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak shaving
with exact quadratic step sizing and adaptive relaxation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Full linear convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_direction(a, M, C=None):
    """
    Solve: max sum(g) s.t. C@g <= M, g >= 0.
    Returns (g, success).
    """
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            return g, True
    except Exception:
        pass
    return a.copy(), False


def exact_step_size(A, B, C, M, gamma=0.95):
    """
    Compute the maximum step size t such that A + 2t*B + t^2*C <= M elementwise.
    A = conv(a,a), B = cross_conv(a,d), C = conv(d,d), M = max(A).
    Returns t_max * gamma.
    """
    # For each k, find the largest t >= 0 such that C[k]*t^2 + 2*B[k]*t + (A[k]-M) <= 0
    A_k_minus_M = A - M  # <= 0 for all k
    t_candidates = []
    
    for k in range(len(A)):
        ck = C[k]
        bk = B[k]
        ak = A_k_minus_M[k]
        
        if ck < 1e-15:
            # Linear: 2*bk*t + ak <= 0
            if bk < -1e-15:
                # Always satisfied for t >= 0
                continue
            elif bk > 1e-15:
                t_k = -ak / (2 * bk)
                if t_k > 0:
                    t_candidates.append(t_k)
            # else bk ≈ 0, ak <= 0, always satisfied
        else:
            # Quadratic: ck*t^2 + 2*bk*t + ak <= 0
            # Discriminant: 4*bk^2 - 4*ck*ak = 4*(bk^2 - ck*ak)
            # Since ak <= 0 and ck > 0: bk^2 - ck*ak = bk^2 + ck*|ak| > 0
            disc = bk * bk - ck * ak
            if disc < 0:
                continue
            sqrt_disc = np.sqrt(disc)
            # Roots: (-2*bk ± 2*sqrt_disc) / (2*ck) = (-bk ± sqrt_disc) / ck
            t_plus = (-bk + sqrt_disc) / ck
            if t_plus > 0:
                t_candidates.append(t_plus)
    
    if not t_candidates:
        return 1.0  # No constraint binding, can take full step
    
    t_max = min(t_candidates)
    return min(t_max * gamma, 1.0)


def lp_step_exact(a, C=None, gamma=0.95):
    """
    One LP step with exact quadratic step sizing.
    Returns (a_new, improved, M_new).
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    
    if C is None:
        C = build_conv_matrix(a)
    
    g, success = solve_lp_direction(a, M, C)
    if not success:
        return a.copy(), False, M
    
    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-10:
        return a.copy(), False, M
    
    # Compute cross-convolutions
    B = cross_convolve(a, d)  # a * d
    C_conv = fast_convolve(d)  # d * d
    
    # Ensure B and C_conv have the right length (2n-1)
    target_len = 2 * n - 1
    if len(B) < target_len:
        B = np.concatenate([B, np.zeros(target_len - len(B))])
    if len(C_conv) < target_len:
        C_conv = np.concatenate([C_conv, np.zeros(target_len - len(C_conv))])
    
    # Compute exact step size
    t = exact_step_size(conv_a, B, C_conv, M, gamma)
    
    # Take the step
    a_new = a + t * d
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False, M
    a_new = a_new / s * a.sum()  # Preserve sum
    
    # Evaluate
    conv_new = fast_convolve(a_new)
    M_new = conv_new.max()
    
    if M_new < M - 1e-12:
        return a_new, True, M_new
    else:
        # Try smaller step
        for gamma_try in [0.5, 0.3, 0.1, 0.05]:
            a_try = a + (t * gamma_try / gamma) * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try * a.sum()
            conv_try = fast_convolve(a_try)
            if conv_try.max() < M - 1e-12:
                return a_try, True, conv_try.max()
        return a.copy(), False, M


def gradient_step(a, lr, top_k=20, temp=0.005):
    """One gradient descent step using softmax-weighted peak gradient."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights over top k peaks
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # grad[j] += w[idx] * (a[k-j] + a[j-k]) (with bounds)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += w[idx] * a[kj]
            jk = j - k
            if 0 <= jk < n:
                grad[j] += w[idx] * a[jk]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s * a.sum()
    return a_new
```

```python
        if w[idx] < 1e-12:
            continue

        # a[k-j] for j=0..n-1 (with padding)
        # This is a reversed, shifted a
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        grad[js] += w[idx] * a[k - js]
        
        # a[j-k] for j=0..n-1 (with padding)
        # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1, intersected with [0, n-1]
        j_start2 = max(0, k)
        j_end2 = min(n - 1, k + n - 1)
        js2 = np.arange(j_start2, j_end2 + 1)
        grad[js2] += w[idx] * a[js2 - k]
    
    return grad
```

The main challenge is: how to efficiently minimize the peak of the convolution. The LP approach is powerful but can get stuck. I need a robust strategy.

Here's my refined plan:

**Algorithm: LP-PeakShave with Adaptive Relaxation**

1. Initialize a (from parent or good heuristic)
2. Build the convolution matrix C (once)
3. Loop:
   a. Compute conv = convolve(a, a), M = max(conv)
   b. Solve LP: max sum(g) s.t. C@g <= M_relaxed, g >= 0
      where M_relaxed = M * (1 + epsilon) for small epsilon (adaptive)
   c. If g ≈ a (LP gives no improvement):
      - Increase epsilon (relax more)
      - Or perturb a
      - Or switch to gradient step
   d. Compute d = g - a
   e. Compute exact step size using quadratic bound
   f. Take step, project to feasible set
   g. If M decreased: update best
   h. If no improvement in K iterations: perturb and restart

The key parameter is epsilon (relaxation). Starting with epsilon=0 (strict) and gradually increasing when stuck.

Actually, I realize there might be an issue with the LP. If M is the current maximum, the constraint C@g <= M means that at the peak position k*, we need (C@g)[k*] = sum_i a[i]*g[k*-i] <= M = (C@a)[k*] = sum_i a[i]*a[k*-i]. This means g at positions k*-i (weighted by a[i]) must be "less than" a at those positions. The LP will try to maximize sum(g) subject to this, which means it will try to move mass from peak-contributing positions to non-peak positions.

This is exactly the right behavior! The LP finds the optimal "redistribution" that doesn't increase the peak (in the first-order sense).

Let me now think about what happens when the LP gets stuck. The LP will give g = a (or very close) when:
- All the constraints are "tight" in a way that prevents any improvement
- The current a is at a "vertex" of the feasible region defined by C@g <= M

In this case, we need to either:
1. Relax M (allow the peak to increase slightly in exchange for a better structure)
2. Perturb a to escape the vertex
3. Use a different direction-finding method (e.g., gradient)

I think the most effective strategy is:
- When LP is stuck, take a gradient step (which can move in directions that the LP can't)
- Then re-solve the LP from the new point

```python
    Solve: min t s.t. C@g <= t, sum(g) = sum(a), g >= 0.
    Returns (g, t_opt, success).

    sum_a = a.sum()
    if C is None:
        C = build_conv_matrix(a)

    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1
    
    # Constraints:
    # C@g - t <= 0 for each k -> [C_row, -1] @ [g, t] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, -1] = -1
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = sum_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([sum_a])
    
    # Bounds: g_i >= 0, t free (but t >= 0 implicitly)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x[:n], 0)
            t_opt = result.x[-1]
            return g, t_opt, True
    except Exception:
        pass
    return a.copy(), C @ a).max(), False
```

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_min_peak(a, C):

    n = len(a)
    sum_a = a.sum()

    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))

    bounds = [(0, None)] * (n + 1)
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.success:

            return g, True
    except Exception:

    # Fallback: original LP
    M = (C @ a).max()
    c2 = -np.ones(n)

    bounds2 = [(0, None)] * n
    try:

                         options={'presolve': True})

def compute_max_step(A, B, Cq, M, gamma=0.95):
    """Max step t s.t. A + 2tB + t^2 Cq <= M elementwise."""
    diff = A - M  # <= 0
    t_max = np.inf
    for k in range(len(A)):
        ck = Cq[k]
        bk = B[k]
        ak = diff[k]
        if ck < 1e-14:
            if bk > 1e-14:
                tk = -ak / (2.0 * bk)
                if 0 < tk < t_max:
                    t_max = tk
        else:
            disc = bk * bk - ck * ak
            if disc > 0:
                tk = (-bk + np.sqrt(disc)) / ck
                if tk > 0 and tk < t_max:
                    t_max = tk
    if np.isinf(t_max):
        t_max = 1.0
    return min(t_max * gamma, 1.0)
