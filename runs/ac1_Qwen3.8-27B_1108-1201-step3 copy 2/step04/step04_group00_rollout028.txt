"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-gradient descent
with momentum, simplex clipping, frequency-domain flattening, and
multi-length multi-initialization search.
"""

import numpy as np
import time

def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_sg(a, top_k=32, temp=0.01):
    """Compute subgradient of max(conv(a,a)) using softmax smoothing."""
    n = len(a)
    c = fft_convolve(a)
    # Find top-k peaks
    k_actual = min(top_k, len(c))
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    # Softmax weights
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute gradient: sg[i] = 2 * sum_k w[k] * a[k-i]
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        k = peaks[idx]
        wi = w[idx]
        if wi < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]
        # = a[k-i_end : k-i_start+1]
        sg[i_start:i_end+1] += 2.0 * wi * a[k-i_end:k-i_start+1]
    
    return sg, c.max()

def optimize_grad(f0, time_limit, lr=0.01, momentum=0.9, top_k=32, temp=0.01, 
                  decay_temp=1.001, verbose=False, seed=42):
    """Gradient descent with momentum on the simplex (clipping projection)."""
    n = len(f0)
    a = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fft_convolve(a).max()
    t0 = time.monotonic()
    cur_lr = lr
    cur_temp = temp
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        sg, cur_max = compute_sg(a, top_k=top_k, temp=cur_temp)
        
        # Normalize gradient by the sum of a (since we're on the simplex)
        # Actually, let's just use it directly
        v = momentum * v - cur_lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        new_max = fft_convolve(a_new).max()
        
        if new_max < best_obj - 1e-14:
            best_obj = new_max
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Don't update a if it didn't help
            # But do keep the momentum
            a = a_new  # Still move, just track best separately
        
        # Adaptive learning rate
        if no_improve > 30:
            cur_lr *= 0.9
            if no_improve > 100:
                # Restart with perturbation
                noise = rng.standard_normal(n) * 0.005
                a = np.maximum(best_a + noise, 0)
                a = a / a.sum()
                v = np.zeros(n)
                cur_lr = lr * 0.5
                cur_temp = temp * 2
                no_improve = 0
        
        # Animate temperature
        cur_temp *= decay_temp
        cur_temp = min(cur_temp, 1.0)
        
        if verbose:
            elapsed = time.monotonic() - t0
            if no_improve % 50 == 0:
                print(f"    GD: t={elapsed:.0f}s, obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.4f}")
    
    return best_a, best_obj

def optimize_adam(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999, top_k=64, 
                  temp=0.005, verbose=False, seed=42):

    v = np.zeros(n)  # Second moment
    best_a = a.copy()
    best_obj = fft_convolve(a).max()

    no_improve = 0
    rng = np.random.default_rng(seed)
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, cur_max = compute_sg(a, top_k=top_k, temp=cur_temp)

        v = beta2 * v + (1 - beta2) * sg**2

def freq_flatten(a, time_limit, iterations=100, verbose=False):
    """Frequency-domain flattening: try to equalize |FFT(a)|^2."""
    n = len(a)

    best_obj = fft_convolve(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2*n-1).bit_length() - 1
    fft_size = 1 << (2*n-1 - 1).bit_length()
    
    for it in range(iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        |A| = np.abs(A)
        # Target: make |A| as flat as possible
        # Compute the "excess" at each frequency
        mean_A = np.mean(|A|)
        # Gradient: reduce high-amplitude frequencies, increase low-amplitude
        # In the time domain, this means:
        # a_new = ifft(A / |A|^alpha) for some alpha in (0,1)
        # This is like a "water-filling" approach
        
        alpha = 0.5
        A_new = A / (np.abs(A) + 1e-10) ** alpha
        # Make it real
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj = fft_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj

def interpolate_seq(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-gradient descent with
continuation (annealing) on temperature, Adam optimizer, multi-length search,
and diverse initializations. Uses exact subgradient computation via FFT-based
convolution and top-K peak extraction.
"""

import numpy as np
import time


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, top_k=64, temp=0.01):
    """Compute smoothed subgradient of max(conv(a,a)) using top-K softmax."""
    n = len(a)
    c = fft_convolve(a)
    k_actual = min(top_k, len(c))
    if k_actual == len(c):
        peaks = np.arange(len(c))
        pv = c
    else:
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
    # Softmax weights
    scaled = pv / max(temp, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= w_sum

    # Gradient: sg[i] = 2 * sum_k w[k] * a[k-i] for valid (k-i)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end] = a[k-i_end : k-i_start+1]
        sg[i_start:i_end + 1] += 2.0 * wi * a[k - i_end:k - i_start + 1]

    return sg, c.max()


def optimize_continuation(f0, time_limit, lr_init=0.05, lr_min=0.002,
                          mom=0.90, top_k=64, temp_init=0.5, temp_min=0.002,
                          temp_decay=0.997, seed=42, verbose=False):
    """
    Gradient descent with continuation: starts with high temperature (smooth
    objective) and anneals down to reveal the true max. Uses momentum and
    clipping projection onto simplex.
    """
    n = len(f0)
    a = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fft_convolve(a).max()
    t0 = time.monotonic()
    cur_temp = temp_init
    cur_lr = lr_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, cur_max = compute_subgradient(a, top_k=top_k, temp=cur_temp)

        # Momentum update
        v = mom * v - cur_lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a
        else:
            a_new = a_new / ss

        new_max = fft_convolve(a_new).max()

        if new_max < best_obj - 1e-14:
            best_obj = new_max
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Anneal temperature
        cur_temp = max(cur_temp * temp_decay, temp_min)

        # Adaptive learning rate
        if no_improve > 50:
            cur_lr = max(cur_lr * 0.92, lr_min)
        if no_improve > 200:
            # Restart with perturbation
            scale = 0.003 + 0.005 * min(no_improve // 400, 3)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0)
            a = a / a.sum()
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.3, lr_min)
            cur_temp = max(cur_temp * 3, temp_min * 5)
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Cont-GD: it={it}, t={elapsed:.0f}s, obj={best_obj:.10f}, "
                  f"temp={cur_temp:.5f}, lr={cur_lr:.6f}", flush=True)

    return best_a, best_obj


def optimize_adam(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,

def interpolate_seq(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new


def make_initial(n, pattern="hann", rng=None):
    """Create initial sequence with various patterns."""
    if pattern == "uniform":
        a = np.ones(n)
    elif pattern == "hann":
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif pattern == "raised_cos":
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif pattern == "bell":
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif pattern == "asym":
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.35) ** 2) + 0.3 * np.exp(-3 * (x - 0.7) ** 2)
    elif pattern == "plateau":
        a = np.zeros(n)
        m = max(1, n // 4)
        a[m:n - m] = 1.0
    else:
        a = np.ones(n)

    if rng is not None:
        a = a + rng.standard_normal(n) * 0.02 * a.mean()
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a
