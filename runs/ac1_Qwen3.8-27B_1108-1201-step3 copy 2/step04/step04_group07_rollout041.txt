def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(a):
    """Compute objective for a with sum=1: 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_conv_matrix(a, n=None):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    if n is None:
        n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def subgradient_direction(a, conv, threshold_frac=0.85):
    """
    Compute subgradient of max(conv) using all peaks above threshold.
    Returns gradient vector of length n.
    """
    n = len(a)
    max_val = conv.max()
    threshold = threshold_frac * max_val
    active_peaks = np.where(conv >= threshold)[0]
    
    # Weight peaks by their value (normalized)
    weights = conv[active_peaks] - threshold
    weights = np.maximum(weights, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active_peaks))
    else:
        weights = weights / weights.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for weight, k in zip(weights, active_peaks):
        if weight < 1e-12:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += weight * 2.0 * a[k - idx]
    
    return sg

def spectral_flatten(a0, time_limit, verbose=False):
    """
    Try to flatten the autoconvolution by working in the spectral domain.
    The idea: minimize the variance of the convolution values while keeping
    the mean fixed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj, best_conv = obj_normalized(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = best_conv
        # Compute "flatness gradient": push high values down, low values up
        mean_b = conv.mean()
        # Gradient of (b - mean)^2 w.r.t. a
        # For simplicity, use the peaks
        max_b = conv.max()
        min_b = conv.min()
        
        if max_b - min_b < 1e-10:
            break
        
        # Subgradient focusing on peaks
        threshold = min_b + 0.7 * (max_b - min_b)
        active = np.where(conv > threshold)[0]
        
        if len(active) == 0:
            break
        
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * a[k - idx]
        
        sg /= len(active)
        norm_sg = np.linalg.norm(sg)
        if norm_sg < 1e-15:
            break
        
        # Step
        a_try = a - lr * sg / norm_sg
        a_try = np.maximum(a_try, 0.0)
        s_new = a_try.sum()
        if s_new < 1e-15:
            break
        a_try = a_try / s_new
        
        obj_try, conv_try = obj_normalized(a_try)
        
        if obj_try < best_obj - 1e-14:
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            best_conv = conv_try
            no_improve = 0
            lr = min(0.02, lr * 1.01)
        else:
            no_improve += 1
            lr *= 0.95
        
        if no_improve >= 200:
            rng = np.random.default_rng(step * 3 + 1)
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            lr = 0.005
            no_improve = 0
        
        step += 1
        if verbose and step % 200 == 0:
            print(f"    Spectral step {step}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smoothed L-BFGS-B optimization (log-sum-exp approximation of max)
2. LP-based direction finding with adaptive margins
3. Multi-peak subgradient with backtracking
4. Length exploration via resampling
5. Final alternating polish
Key improvement: smooth continuous optimization via L-BFGS-B on logsumexp
approximation, combined with combinatorial LP directions.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_general(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def obj_normalized(a):
    """Compute objective for a with sum=1: 2*n*max(conv). Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def smooth_obj_and_grad(x, tau):
    """
    Smoothed objective using logsumexp and its gradient.
    Objective: 2n * tau * logsumexp(conv/tau) / (sum(x))^2
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)

    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = max_c + tau * (max_shifted + np.log(exp_shifted.sum()))

    obj = 2.0 * n * log_sum_exp / (s * s)

    # probs[k] proportional to exp((conv[k]-max_c)/tau)
    probs = exp_shifted / exp_shifted.sum()

    # grad_lse[i] = 2 * sum_j probs[i+j] * a[j] for valid indices
    # Build the index matrix and do a matrix-vector product
    idx_i = np.arange(n)[:, None]
    idx_j = np.arange(n)[None, :]
    idx_k = idx_i + idx_j
    valid = idx_k < (2 * n - 1)
    M = np.where(valid, probs[idx_k], 0.0)
    grad_lse = 2.0 * (M @ a)

    # Full gradient via chain rule
    grad = 2.0 * n * (grad_lse / (s * s) - 2.0 * log_sum_exp / (s ** 3))

    # Zero out gradient where x is at the bound (x=0)
    mask = x > 1e-10
    grad[~mask] = 0.0

    return obj, grad


def lbfgs_smoothed(a0, time_limit, tau=0.005, max_iter=500, verbose=False):
    """Optimize using L-BFGS-B on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj, _ = obj_normalized(a)
    best_a = a.copy()

    t0 = time.monotonic()

    def fun(x):
        return smooth_obj_and_grad(x, tau)[0]

    def fun_and_grad(x):
        return smooth_obj_and_grad(x, tau)

    result = minimize(fun_and_grad, a, method='L-BFGS-B',
                      bounds=[(0.0, None)] * n,
                      options={'maxiter': max_iter, 'gtol': 1e-8,
                               'maxfun': 2000, 'disp': False})

    if result.x is not None:
        a_new = np.maximum(result.x, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            obj_new, _ = obj_normalized(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    if verbose:
        print(f"    L-BFGS-B: obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a, n=None):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    if n is None:
        n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns (g_normalized, g_sum)."""
    n = len(a)
    C = build_conv_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 20})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def backtrack_line_search(a, d, current_obj, n, c1=1e-4, rho=0.5, max_iter=25):
    """Armijo backtracking line search with extension."""
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-15:
        return 0.0, current_obj

    best_t = 0.0
    best_obj = current_obj

    t = min(0.3, 0.05 / (norm_d + 1e-8))
    for _ in range(max_iter):
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= rho
            continue
        a_try = a_try / s
        obj_try, _ = obj_normalized(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_t = t
            # Try extending
            t_ext = t * 2.0
            a_ext = np.maximum(a + t_ext * d, 0.0)
            s_ext = a_ext.sum()
            if s_ext > 1e-15:
                a_ext = a_ext / s_ext
                obj_ext, _ = obj_normalized(a_ext)
                if obj_ext < best_obj - 1e-12:
                    best_obj = obj_ext
                    best_t = t_ext
            return best_t, best_obj
        else:
            t *= rho

    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_normalized(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        M_lp = M * 0.9995
        g, g_sum = lp_direction(a, M_lp)

        if g_sum > 1.0 + 1e-8:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-12:
                t_best, obj_best = backtrack_line_search(a, d, best_obj, n)
                if obj_best < best_obj - 1e-13:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    _, conv = obj_normalized(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        if no_improve >= 40:
            rng = np.random.default_rng(iteration * 13 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 40, 10)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            _, conv = obj_normalized(a)
            M = conv.max()
            no_improve = 0
            if no_improve > 500:
                break

        if verbose and iteration % 10 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.006, momentum=0.92,
                         threshold_frac=0.80, verbose=False):
    """Multi-peak subgradient with backtracking and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = obj_normalized(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        # Compute multi-peak subgradient
        max_val = conv.max()
        threshold = threshold_frac * max_val
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            break

        # Weight by excess above threshold
        excess = conv[active] - threshold
        w_sum = excess.sum()
        if w_sum < 1e-15:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = excess / w_sum

        sg = np.zeros(n)
        for w, k in zip(weights, active):
            if w < 1e-12:
                continue
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w * 2.0 * a[k - idx]

        norm_sg = np.linalg.norm(sg)
        if norm_sg < 1e-15:
            break

        # Momentum
        v = momentum * v - lr * sg / norm_sg
        d = v

        # Backtracking
        t_best, obj_best = backtrack_line_search(a, d, best_obj, n, max_iter=20)

        if t_best > 1e-8 and obj_best < best_obj - 1e-13:
            a_new = np.maximum(a + t_best * d, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            best_obj = obj_best
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            momentum = min(0.98, momentum + 0.003)

        if no_improve >= 150:
            rng = np.random.default_rng(step * 5 + 11)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            lr = lr_init
            momentum = 0.92
            _, conv = obj_normalized(a)
            no_improve = 0

        _, conv = obj_normalized(a)
        step += 1

        if verbose and step % 50 == 0:
            print(f"    SG step {step}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq

            nl = n_best + delta
            if 128 <= nl <= 2048 and nl != n_best:
                candidate_lengths.append(nl)
        candidate_lengths = sorted(set(candidate_lengths))

        t_per_len = max(25.0, p4_budget / max(1, len(candidate_lengths)))

        for nl in candidate_lengths[:4]:
            if time.monotonic() > p4_deadline - 20:
                break
            t_alloc = min(t_per_len, p4_deadline - time.monotonic() - 15)
            if t_alloc < 15:
                break

            a_res = resample_sequence(best_sequence, nl)
            # Quick L-BFGS
            t_lbfgs = t_alloc * 0.5
            a_opt, _ = lbfgs_smoothed(a_res, time_limit=t_lbfgs, tau=0.01,
                                      max_iter=max(30, int(t_lbfgs / 2)))
            update_best(a_opt, f"Len-{nl}")

            # Quick LP
            t_left = p4_deadline - time.monotonic() - 15
            if t_left > 15 and best_sequence is not None:
                a_opt2, _ = lp_optimize(a_opt, time_limit=min(t_alloc * 0.4, t_left))
                update_best(a_opt2, f"Len-{nl}-LP")

    print(f"Phase 4 done ({time.monotonic()-t0:.0f}s), best={best_score:.8f}")

    # === Phase 5: Final alternating polish ===
    print("\n=== Phase 5: Final polish ===")
    t0 = time.monotonic()
    p5_budget = deadline - time.monotonic() - 10
    if best_sequence is not None and p5_budget > 20:
        a = best_sequence.copy()
        n_cyc = max(2, int(p5_budget / 40))
        t_per = p5_budget / n_cyc

        for ci in range(n_cyc):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break

            if ci % 3 == 0:
                a_opt, _ = lbfgs_smoothed(a, time_limit=t_alloc, tau=0.003,
                                          max_iter=max(30, int(t_alloc / 2)))
            elif ci % 3 == 1:
                a_opt, _ = lp_optimize(a, time_limit=t_alloc)
            else:
                a_opt, _ = subgradient_optimize(a, time_limit=t_alloc,
                                                 lr_init=0.003, momentum=0.95,
                                                 threshold_frac=0.70)
            update_best(a_opt, f"Polish-{ci}")
            a = best_sequence.copy()

    print(f"Phase 5 done ({time.monotonic()-t0:.0f}s), best={best_score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = parent.copy()
        best_score = float(evaluate_sequence([float(x) for x in best_sequence]))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_score = float('inf')
        best_sequence = parent.copy()
        final_score = float(evaluate_sequence([float(x) for x in best_sequence]))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
