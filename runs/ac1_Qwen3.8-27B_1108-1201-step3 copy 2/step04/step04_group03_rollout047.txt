def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_conv_seq(n):
    """Generate the 'flat convolution' sequence: a[k] = binom(2k,k)/4^k normalized."""
    a = np.zeros(n)
    for k in range(n):
        if k == 0:
            a[k] = 1.0
        else:
            a[k] = a[k-1] * (2*k - 1) / (2*k)  # ratio: binom(2k,k)/4^k / (binom(2k-2,k-1)/4^{k-1}) = (2k-1)/(2k) ... 
            # Actually: binom(2k,k)/binom(2k-2,k-1) = (2k)!/(k!k!) * ((k-1)!^2)/(2k-2)! = (2k)(2k-1)/(k^2)
            # So binom(2k,k)/4^k / (binom(2k-2,k-1)/4^{k-1}) = (2k)(2k-1)/(k^2 * 4) = (2k-1)/(2k)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: row k has f[j] at position j for valid j

    for j in range(n):

    b_ub = np.full(2 * n - 1, M)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
