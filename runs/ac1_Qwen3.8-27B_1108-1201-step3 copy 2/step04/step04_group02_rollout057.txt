def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Cyclic Coordinate Descent (CCD) with exact 1D subproblem solving via critical point enumeration
2. Subgradient descent with softmax smoothing for polishing
3. LP-based Frank-Wolfe for global direction finding
4. Multi-scale diverse initializations (analytical windows + parent)
5. Adaptive time allocation across phases
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def ccd_optimize(a0, time_limit, seed=42, verbose=False):
    """
    Cyclic Coordinate Descent with exact 1D minimization.
    For each coordinate, finds optimal value by enumerating critical points.
    Works on normalized (sum=1) sequences, minimizing 2*n*max(conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    best_sweep_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        sweep_start_obj = best_obj

        for j in order:
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            aj = a[j]

            # Compute b_conv = convolution of a with a[j] set to 0
            b_conv = conv.copy()
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                b_conv[k_lo:k_hi + 1] -= 2.0 * aj * a[k_lo - j:k_hi - j + 1]
            if 2 * j <= 2 * n - 2:
                b_conv[2 * j] += aj * aj

            s_b = s - aj

            if s_b < 1e-12:
                # Edge case: all mass at a[j]
                continue

            # Collect critical points for t (new value of a[j])
            t_cands = [0.0, aj]

            # Linear critical points: t = b_conv[k]/a[k-j] - s_b
            if k_lo <= k_hi:
                idx = np.arange(k_lo, k_hi + 1)
                # Exclude k=2j (handled by quadratic)
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    positions = np.where(idx == 2 * j)[0]
                    if len(positions) > 0:
                        idx = np.delete(idx, positions)
                if len(idx) > 0:
                    akj = a[idx - j]
                    pos = akj > 1e-12
                    if np.any(pos):
                        t_lin = b_conv[idx[pos]] / akj[pos] - s_b
                        t_cands.extend(t_lin[t_lin > 0].tolist())

            # Quadratic critical point: t = b_conv[2j] / s_b
            if 2 * j <= 2 * n - 2:
                tq = b_conv[2 * j] / s_b
                if tq > 0:
                    t_cands.append(tq)

            t_cands = np.array(t_cands)
            t_cands = np.maximum(t_cands, 0.0)
            t_cands = np.unique(t_cands)

            # Evaluate F(t) = 2*n * max_k(profile) / (s_b+t)^2 for each candidate
            if len(t_cands) > 0:
                best_t = t_cands[0]
                best_F = float('inf')

                # Vectorize: compute max for each t
                # Profile: linear part p[k] = 2*a[k-j] for k in [k_lo, k_hi], k != 2j
                #          quadratic part at k=2j: t^2

                # For efficiency, compute the linear part max using matrix ops
                # linear_part[i, k] = b_conv[k] + 2 * t_cands[i] * a[k-j]
                if k_lo <= k_hi:
                    profile_lin = a[k_lo - j:k_hi - j + 1]  # shape (n_lin,)
                    base = b_conv[k_lo:k_hi + 1]  # shape (n_lin,)
                    # max_val_lin[i] = max(base + 2*t_cands[i]*profile_lin)
                    # = max over k of base[k] + 2*t_i*profile_lin[k]
                    # This is the support function of the set {(profile_lin[k], base[k])}
                    # Just compute directly for small n
                    mat = base[None, :] + 2.0 * t_cands[:, None] * profile_lin[None, :]
                    max_lin = mat.max(axis=1)
                else:
                    max_lin = np.zeros(len(t_cands))

                # Quadratic part
                if 2 * j <= 2 * n - 2:
                    quad_vals = b_conv[2 * j] + t_cands ** 2
                    max_vals = np.maximum(max_lin, quad_vals)
                else:
                    max_vals = max_lin

                denoms = (s_b + t_cands) ** 2
                F_vals = 2.0 * n * max_vals / np.maximum(denoms, 1e-20)

                valid = np.isfinite(F_vals)
                if np.any(valid):
                    best_i = np.argmin(np.where(valid, F_vals, np.inf))
                    best_t = t_cands[best_i]
                    best_F = F_vals[best_i]

            if best_F < best_obj - 1e-14 and abs(best_t - aj) > 1e-12:
                a[j] = best_t
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                    s = 1.0
                    conv = fast_convolve(a)
                    continue
                # Don't renormalize here to keep CCD clean
                conv = fast_convolve(a)
                cur_obj = float(2.0 * n * conv.max() / s * s)  # = 2*n*max(conv) since we track unscaled
                # Actually since a is not normalized, let's just track the ratio properly
                cur_obj_full = float(2.0 * len(a) * conv.max() / (s * s))
                if cur_obj_full < best_obj / (best_obj / (2.0 * n * fast_convolve(best_a).max())) - 1e-15:
                    pass
                # Simplify: just track normalized objective
                a_norm = a / s
                conv_norm = fast_convolve(a_norm)
                obj_norm = float(2.0 * n * conv_norm.max())
                if obj_norm < best_obj - 1e-14:
                    best_obj = obj_norm
                    best_a = a_norm.copy()
                    a = a_norm
                    s = 1.0
                    conv = conv_norm

        # End of sweep
        if best_obj < sweep_start_obj - 1e-14:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 40:
            # Perturb from best
            noise_scale = 0.003 * (1.0 + 0.5 * np.sin(sweep * 0.1))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            cur_obj = float(2.0 * n * conv.max())
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0

        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CCD sweep {sweep}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    # Final normalize
    a = best_a / best_a.sum()
    return a, float(2.0 * len(a) * fast_convolve(a).max())


def sg_polish(a0, time_limit, lr=0.003, mom=0.95, top_k=64, temp=0.0005, seed=0):
    """Subgradient polishing with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    cur_lr = lr
    cur_temp = temp
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Compute softmax-weighted subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Nesterov momentum step
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = float(2.0 * n * conv_new.max())

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        # Adaptive parameters
        cur_lr *= 0.9995
        cur_temp *= 1.002

        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_obj = float(2.0 * n * M)
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Solve LP: max sum(g) s.t. (a*g)[k] <= M*m_factor, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_factor)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 10})
            if not res.success:
                m_factor = min(m_factor * 1.3, 5.0)
                continue
        except Exception:
            m_factor = min(m_factor * 1.3, 5.0)
            continue

        g = res.x
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            m_factor = min(m_factor * 1.02, 3.0)
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = float(2.0 * n * M)
        alphas = np.linspace(0.05, 1.0, 12)
        for alpha in alphas:
            at = (1.0 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = float(2.0 * n * fast_convolve(at).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = float(conv.max())
            m_factor = 1.0

            obj = float(2.0 * n * M)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.1, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it + 42)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            M = float(conv.max())
            m_factor = 1.0
            no_improve = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP iter {it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    # Hann window (raised cosine)
    hann = 0.5 * (1.0 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    blackman = np.maximum(blackman, 0)
    inits.append(blackman / blackman.sum())

    # Tukey (flat-top) window
    w = 0.5
    tukey = np.ones(n)
    edge = int(w * n // 2)
    if edge > 0:
        for i in range(edge):
            val = 0.5 * (1 + np.cos(np.pi * (2 * i / (w * n) - 1)))
            tukey[i] = val
            tukey[n - 1 - i] = val
    inits.append(tukey / tukey.sum())

    # Exponential decay (symmetric)
    center = n / 2.0
    exp_sym = np.exp(-0.5 * ((x - center) / (n / 4.0)) ** 2)
    inits.append(exp_sym / exp_sym.sum())

    # Linear ramp decreasing
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    # Power-law decay
    pl = 1.0 / (x + 1.0) ** 0.7
    inits.append(pl / pl.sum())

    # Sinc-like
    sinc = np.sinc((x - (n - 1) / 2.0) / (n / 6.0))
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 1e-10:
        inits.append(sinc / sinc.sum())

    # Cosine to the 4th power
    c4 = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 4 + 0.01
    inits.append(c4 / c4.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Parent-based
    if parent is not None and len(parent) > 0:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            px = px / ps
            inits.append(px)
            # Blend with Hann
            hann = 0.5 * (1.0 + np.cos(2 * np.pi * x / (n - 1)))
            blend = 0.5 * px + 0.5 * hann / hann.sum()
            inits.append(blend / blend.sum())
            # Blend with uniform
            blend2 = 0.7 * px + 0.3 * (np.ones(n) / n)
            inits.append(blend2 / blend2.sum())

    return inits
