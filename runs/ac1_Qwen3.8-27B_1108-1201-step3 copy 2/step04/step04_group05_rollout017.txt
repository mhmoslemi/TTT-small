"""

exploration. Combines fast gradient iterations with occasional LP-based

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, conv, peaks, weights):
    """Compute weighted subgradient of max conv at given peaks."""
    n = len(f)
    grad = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        w = 2.0 * weights[i]
        if w < 1e-15:
            continue
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start > m_end:
            continue
        m_range = np.arange(m_start, m_end + 1)
        f_idx = k - m_range
        grad[m_range] += w * f[f_idx]
    return grad


def softmax_weights(values, temperature):
    """Compute softmax weights for peak targeting."""
    scaled = values / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-15:
        w = np.zeros(len(values))
        w[0] = 1.0
    else:
        w /= s
    return w


def gradient_optimize(f0, time_limit, lr=0.01, momentum=0.9, top_k=20,
                      temperature=0.001, decay_lr=0.999, decay_temp=1.002,
                      verbose=False, rng=None):
    """
    Projected gradient descent with momentum and peak targeting.
    Minimizes max(conv(f,f)) subject to f>=0, sum(f)=1.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fft_convolve(f)
        cur_obj = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        w = softmax_weights(pv, cur_temp)
        grad = compute_subgrad(f, conv, peaks, w)

        # Momentum update
        v = momentum * v - cur_lr * grad
        f_new = project_simplex(f + v)

        # Evaluate
        conv_new = fft_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < cur_obj:
            f = f_new
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"      GradOpt it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            # Reject step, reduce momentum
            v *= 0.5
            no_improve += 1

        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 200:
            # Restart with perturbation
            scale = min(0.01, 0.002 + no_improve * 0.00005)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def lp_fw_step(f, top_k=30):
    """
    Solve reduced linearized LP for Frank-Wolfe direction.
    Min t s.t. t >= 2*(f*g)[k] - conv_f[k] for top-k peaks,
    g >= -f, sum(g) = 0.
    """
    n = len(f)
    conv_f = fft_convolve(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks_sorted = np.sort(peaks)

    # Build constraint matrix for selected peaks
    # (f*g)[k] = sum_j f[k-j]*g[j]
    A_rows = len(peaks_sorted)
    A_ub = np.zeros((A_rows * 2, n + 1))
    b_ub = np.zeros(A_rows * 2)

    for i, k in enumerate(peaks_sorted):
        k = int(k)
        # t - 2*(f*g)[k] >= -conv_f[k]  =>  -t + 2*(f*g)[k] <= conv_f[k]
        # (f*g)[k] = sum_j g[j] * f[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # Row for this constraint: -1*t + 2*sum_{j=js}^{je} f[k-j]*g[j] <= conv_f[k]
        A_ub[i, 0] = -1.0
        for j in range(js, je + 1):
            A_ub[i, j + 1] = 2.0 * f[k - j]
        b_ub[i] = conv_f[k]

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5})
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_fw_step(f, top_k=top_k)
        if g is None:
            no_improve += 3
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = fft_convolve(f).max()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fft_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            val = fft_convolve(f).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0

    return best_f, best_obj


def nesterov_gradient_optimize(f0, time_limit, lr=0.005, top_k=40,
                                temperature=0.0005, verbose=False, rng=None):
    """Nesterov accelerated gradient for finer polishing."""
    if rng is None:
        rng = np.random.default_rng(2)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    y = f.copy()
    n = len(f)

    conv = fft_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = 0.9
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fft_convolve(y)
        cur_obj = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        w = softmax_weights(pv, cur_temp)
        grad = compute_subgrad(y, conv, peaks, w)

        # Nesterov step
        t_k = 2.0 / (it + 1.5)
        f_new = project_simplex(y - cur_lr * grad)
        y = f_new + (t_k - 1.0) / (t_k + 1.0) * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv_new = fft_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      Nesterov it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.003

        if no_improve > 300:
            scale = 0.001 + no_improve * 0.00001
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            cur_lr = max(lr * 0.7, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, rng=None):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce peak.
    """
    if rng is None:
        rng = np.random.default_rng(3)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving mass from i to j
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = frac * f[i]
            if delta > f[i]:
                continue
            f_trial = f.copy()
            f_trial[i] -= delta
            f_trial[j] += delta
            # Recompute convolution efficiently (only changed parts)
            # For simplicity, use full FFT
            obj = fft_convolve(f_trial).max()
            if obj < best_obj - 1e-15:
                f = f_trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                break
        else:
            no_improve += 1

        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    inits.append(rng.dirichlet(np.ones(n) * 3.0))

    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    # Gaussians
    for sig in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - center) / (sig * n)) ** 2)
        inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Two bumps
    c2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(c2 / c2.sum())

    # Edges
    e = 1.0 / (np.abs(x - center) / n + 0.05)
    inits.append(e / e.sum())

    # Linear ramp
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())

    if parent is not None and len(parent) != n:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    elif parent is not None:
        inits.append(parent.copy())

    return inits


def quick_optimize(f0, time_limit, rng=None):
    """Quick optimization using gradient method with moderate settings."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(f0)
    top_k = max(10, n // 10)
    return gradient_optimize(f0, time_limit, lr=0.02, momentum=0.85,
                             top_k=top_k, temperature=0.005,
                             decay_lr=0.998, decay_temp=1.005, rng=rng)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Diverse initialization and quick exploration
    Phase 2: Primary gradient optimization
    Phase 3: Frank-Wolfe with LP oracle
    Phase 4: Nesterov polishing
    Phase 5: Final coordinate descent and verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Initialization and quick exploration ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    n_main = 384 if parent is None else len(parent)
    lengths_to_try = [n_main]
    for nl in [256, 320, 448, 512]:
        if nl != n_main and nl >= 128:
            lengths_to_try.append(nl)

    phase1_time = min(200.0, (deadline - time.monotonic()) * 0.12)
    t_phase1 = time.monotonic() + phase1_time

    candidates = []
    for n_try in lengths_to_try:
        inits = generate_inits(n_try, parent, rng)
        for i, ini in enumerate(inits[:6]):  # Limit inits per length
            candidates.append((f"n{n_try}_i{i}", ini))

    # Quick optimize each candidate
    for label, cand in candidates:
        if time.monotonic() > t_phase1:
            break
        t_each = min(15.0, t_phase1 - time.monotonic())
        if t_each < 2.0:
            break
        a_opt, _ = quick_optimize(cand, t_each * 0.8, rng=rng)
        try_seq(a_opt, f"p1_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 done. Best: {best_score:.8f} (n={len(best_seq)}, t={time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Primary gradient optimization ===
    print("\n=== Phase 2: Primary Gradient Optimization ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    phase2_time = min(remaining * 0.4, 600.0)
    t_phase2 = time.monotonic() + phase2_time

    # Run with different hyperparameters
    configs = [
        {"lr": 0.015, "momentum": 0.9, "top_k": n_best // 8, "temperature": 0.003},
        {"lr": 0.008, "momentum": 0.93, "top_k": n_best // 5, "temperature": 0.001},
        {"lr": 0.02, "momentum": 0.85, "top_k": n_best // 12, "temperature": 0.01},
    ]

    f_current = best_seq.copy()
    for ci, cfg in enumerate(configs):
        t_each = min(phase2_time / (len(configs) - ci), t_phase2 - time.monotonic())
        if t_each < 10.0:
            break
        print(f"  Config {ci}: lr={cfg['lr']}, mom={cfg['momentum']}, "
              f"top_k={cfg['top_k']}, temp={cfg['temperature']}, t={t_each:.0f}s")
        a_opt, _ = gradient_optimize(f_current, t_each * 0.9,
                                     lr=cfg["lr"], momentum=cfg["momentum"],
                                     top_k=cfg["top_k"], temperature=cfg["temperature"],
                                     decay_lr=0.999, decay_temp=1.003,
                                     verbose=True, rng=rng)
        try_seq(a_opt, f"p2_cfg{ci}")
        f_current = best_seq.copy()

    print(f"  Phase 2 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Frank-Wolfe with LP ===
    print("\n=== Phase 3: Frank-Wolfe (LP) ===")
    remaining = deadline - time.monotonic()
    phase3_time = min(remaining * 0.25, 350.0)
    t_phase3 = time.monotonic() + phase3_time

    if phase3_time > 20.0:
        # Run FW on best
        t_fw = min(phase3_time * 0.6, t_phase3 - time.monotonic())
        if t_fw > 10.0:
            print(f"  Running FW (top_k=20), t={t_fw:.0f}s")
            a_opt, _ = fw_optimize(best_seq, t_fw, top_k=20, verbose=True, rng=rng)
            try_seq(a_opt, "p3_fw")

        # Run FW on perturbed version
        rem = t_phase3 - time.monotonic()
        if rem > 10.0:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            a_opt, _ = fw_optimize(f_pert, rem * 0.9, top_k=15, rng=rng)
            try_seq(a_opt, "p3_fw_pert")

    print(f"  Phase 3 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Nesterov polishing ===
    print("\n=== Phase 4: Nesterov Polish ===")
    remaining = deadline - time.monotonic()
    phase4_time = min(remaining * 0.2, 300.0)
    t_phase4 = time.monotonic() + phase4_time

    if phase4_time > 15.0:
        configs_nest = [
            {"lr": 0.003, "top_k": n_best // 4, "temperature": 0.0005},
            {"lr": 0.001, "top_k": n_best // 2, "temperature": 0.0002},
        ]
        for ci, cfg in enumerate(configs_nest):
            t_each = min(phase4_time / (len(configs_nest) - ci), t_phase4 - time.monotonic())
            if t_each < 8.0:
                break
            print(f"  Nesterov {ci}: lr={cfg['lr']}, top_k={cfg['top_k']}, t={t_each:.0f}s")
            a_opt, _ = nesterov_gradient_optimize(
                best_seq, t_each * 0.9,
                lr=cfg["lr"], top_k=cfg["top_k"],
                temperature=cfg["temperature"], verbose=True, rng=rng)
            try_seq(a_opt, f"p4_nest{ci}")

    print(f"  Phase 4 done. Best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final polishing and diverse attempts ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    phase5_time = min(remaining * 0.9, (deadline - time.monotonic()) - 20.0)
    t_phase5 = time.monotonic() + max(0, phase5_time)

    if phase5_time > 10.0:
        # Multiple short gradient runs with different random restarts
        n_attempts = min(6, int(phase5_time / 15))
        for ai in range(n_attempts):
            t_each = min(t_phase5 - time.monotonic(), 20.0)
            if t_each < 5.0:
                break
            noise_scale = 0.001 + 0.002 * ai
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            a_opt, _ = gradient_optimize(
                f_pert, t_each * 0.85,
                lr=0.005, momentum=0.95, top_k=n_best // 3,
                temperature=0.0005, decay_lr=0.9995, decay_temp=1.002,
                rng=rng)
            try_seq(a_opt, f"p5_{ai}")

        # Try a different length
        rem = t_phase5 - time.monotonic()
        if rem > 15.0 and best_seq is not None:
            for n_try in [n_best - 32, n_best + 32, n_best + 64]:
                if n_try < 64 or n_try > 1024 or rem < 8.0:
                    continue
                px = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_best), best_seq)
                px = np.maximum(px, 0)
                ps = px.sum()
                if ps > 1e-10:
                    px = px / ps
                    a_opt, _ = quick_optimize(px, min(rem * 0.4, 20.0), rng=rng)
                    try_seq(a_opt, f"p5_len{n_try}")
                    rem = t_phase5 - time.monotonic()

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
