def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_objective_sum_sq(a):
    """Compute objective using sum(a^2) as the max (valid for non-negative a)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    return 2.0 * n * np.sum(a * a) / (s * s)


def projected_gradient_descent(f0, time_limit=60.0, lr=0.01, patience=200):
    """Projected gradient descent on the simplex for minimizing 2*n*sum(a^2)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        # Gradient of sum(a^2) subject to sum(a)=1, a>=0
        # d/da_i sum(a^2) = 2*a_i
        grad = 2.0 * f
        # Remove mean component (tangent to simplex)
        grad = grad - grad.mean()

        f_new = project_simplex(f - cur_lr * grad)
        obj = compute_objective(f_new)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_lr *= 0.9995

        if no_improve > patience:
            break

    return best_f, best_obj

def scipy_optimize(f0, time_limit=60.0):
    """Use scipy minimize with SLSQP for a more systematic optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    def objective(x):
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e10
        return 2.0 * n * np.sum(x * x) / (sx * sx)

    def gradient(x):
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            return np.zeros_like(x)
        return 4.0 * n * x / (sx * sx) - 4.0 * n * np.sum(x * x) / (sx * sx * sx)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        result = minimize(objective, f, jac=gradient, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-12, 'time_limit': max(1, time_limit - (time.monotonic() - t0))})
        if result.success:
            f_opt = np.maximum(result.x, 0)
            f_opt = f_opt / f_opt.sum()
            return f_opt, compute_objective(f_opt)
    except Exception:
        pass

    return f, compute_objective(f)

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.15, 0.25, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        t_alloc = min(rem * 0.5, 100)
        f_opt, obj_opt = multi_scale_search(best_seq, time_limit=t_alloc, rng=rng)
        try_seq(f_opt, "multiscale")

        # Try longer sequences by upscaling
        for n_try in [n_p * 2, n_p * 3]:
            if n_try > 8192 or time.monotonic() > deadline - 20:
                break
            f_up = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_p), best_seq)
            f_up = np.maximum(f_up, 0)
            f_up = f_up / f_up.sum()
            try_seq(f_up, f"upscale_{n_try}")

    print(f"  Phase 5 best: {best_score:.10f}")

    # === Phase 6: Fine-tuning with scipy ===
    print("\n=== Phase 6: SciPy Fine-Tuning ===")
    if time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic() - 15
        t_alloc = min(rem * 0.6, 60)
        f_opt, obj_opt = scipy_optimize(best_seq, time_limit=t_alloc)
        try_seq(f_opt, "scipy")

        # Also try from uniform
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            f_opt2, obj_opt2 = scipy_optimize(np.ones(n_p) / n_p, time_limit=min(rem * 0.5, 30))
            try_seq(f_opt2, "scipy_uniform")

    print(f"  Phase 6 best: {best_score:.10f}")

    # === Phase 7: Final micro-optimization ===
    print("\n=== Phase 7: Final Polish ===")
    if time.monotonic() < deadline - 5:
        rem = deadline - time.monotonic() - 5
        t_alloc = min(rem * 0.8, 30)
        rng2 = np.random.default_rng(seed + 1000)
        f_opt, obj_opt = local_search(best_seq, time_limit=t_alloc, rng=rng2)
        try_seq(f_opt, "final_polish")

        # Very small perturbations
        for scale in [0.0001, 0.0003, 0.001]:
            if time.monotonic() > deadline - 5:
                break
            for _ in range(200):
                if time.monotonic() > deadline - 5:
                    break
                noise = rng2.standard_normal(len(best_seq)) * scale
                f_pert = project_simplex(best_seq + noise)
                score = try_seq(f_pert, f"micro_{scale}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
