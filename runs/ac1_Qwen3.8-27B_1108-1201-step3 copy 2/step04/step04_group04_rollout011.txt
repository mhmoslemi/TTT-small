import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgrad_max(a, conv):
    """Subgradient of max_k conv[k] w.r.t. a. grad[m] = 2*a[k-m] for k=argmax, 0<=k-m<n."""
    n = len(a)
    k = int(np.argmax(conv))
    grad = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start <= m_end:
        grad[m_start:m_end+1] = 2.0 * a[k - np.arange(m_start, m_end+1)]
    return grad


def softmax_grad(a, conv, tau, top_k=100):
    """Smooth gradient using top-k softmax approximation."""
    n = len(a)
    if top_k >= 2*n-1:
        top_k = 2*n-1

    conv_sub = conv[indices]
    scaled = tau * conv_sub
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()

        w = np.zeros(top_k)
        w[np.argmax(conv_sub)] = 1.0
    else:
        w /= w_sum

    for idx, w_val in zip(indices, w):

        k = int(idx)
        m_start = max(0, k - n + 1)

    FISTA with varying-temperature softmax gradient.
    a0: initial sequence (will be projected to sum=1, nonneg)
    """

    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()
    
    elapsed_check = 0.0
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit:

        tau = tau_start * (tau_end / tau_start) ** progress
        
        conv_y = fast_convolve(y)

def coordinate_descent(a0, time_limit, rng=None):
    """
    Coordinate descent: try adjusting pairs of elements.
    More efficient than the previous version by batching.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    a = project_nonneg_sum1(a0.astype(np.float64))
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve_count = 0
    
    # Precompute which convolve indices are affected by each position
    # Changing a[i] affects conv[k] for k in [i, i+n-1]
    
    while time.monotonic() - t0 < time_limit:
        # Try a batch of moves
        batch = max(100, min(5000, int(2e6 / n)))
        
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving mass from i to j
            for frac in [0.01, 0.05, 0.1, 0.005, 0.02]:
                delta = frac * a[i]
                if delta < 1e-10:
                    continue
                a[i] -= delta
                a[j] += delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve_count = 0
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            else:
                no_improve_count += 1
        
        if no_improve_count > 500:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with softmax-approximated max (varying temperature, backtracking)
2. Adam optimizer with subgradient
3. Multi-length exploration (768, 1024, 2048)
4. Coordinate descent polishing
5. "Peak reduction" targeted moves
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_grad(a, conv, tau, top_k=100):
    """Compute gradient of (1/tau)*logsumexp(tau*conv) w.r.t. a using top-k approximation.
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices."""
    n = len(a)
    size = len(conv)
    if top_k >= size:
        top_k = size
    indices = np.argpartition(conv, -top_k)[-top_k:]
    conv_sub = conv[indices]
    scaled = tau * conv_sub
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(top_k)
        w[np.argmax(conv_sub)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx, w_val in zip(indices, w):
        if w_val < 1e-15:
            continue
        k = int(idx)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            grad[m_start:m_end + 1] += 2.0 * w_val * a[k - np.arange(m_start, m_end + 1)]
    return grad


def subgrad_max_only(a, conv):
    """Exact subgradient at the argmax."""
    n = len(a)
    k = int(np.argmax(conv))
    grad = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start <= m_end:
        grad[m_start:m_end + 1] = 2.0 * a[k - np.arange(m_start, m_end + 1)]
    return grad


def fista_bt(a0, time_limit, tau_start=50.0, tau_end=1000.0, lr0=0.01, top_k=100,
             bt_factor=0.5, bt_max_iter=20):
    """
    FISTA with backtracking line search and varying-temperature softmax gradient.
    """
    a = project(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 1000
    step = lr0
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)

        conv_y = fast_convolve(y)
        obj_y = conv_y.max()
        grad = softmax_grad(y, conv_y, tau, top_k=top_k)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

        # Backtracking line search
        best_local = None
        best_local_obj = obj_y
        s = step
        for _ in range(bt_max_iter):
            y_try = y - s * grad
            y_try = project(y_try)
            y_try = y_try + ((t_k - 1.0) / t_new) * (y_try - y)
            y_try = project(y_try)
            conv_try = fast_convolve(y_try)
            obj_try = conv_try.max()
            if obj_try < best_local_obj - 1e-14:
                best_local = y_try
                best_local_obj = obj_try
            s *= bt_factor
            if s < 1e-10:
                break

        if best_local is not None:
            a = best_local
            conv = fast_convolve(a)
            cur_obj = conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            y = a
            step = min(step * 2.0, lr0 * 2)
        else:
            no_improve += 1
            step *= bt_factor

        t_k = t_new
        iterations += 1

        if no_improve >= max_no_improve:
            break

    return best_a, 2.0 * n * best_obj


def adam_opt(a0, time_limit, lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8,
             top_k=100, tau_base=100.0):
    """Adam with softmax-approximated gradient."""
    a = project(a0.astype(np.float64))
    n = len(a)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    count = 0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_base * (1.0 + 3.0 * progress)

        conv_a = fast_convolve(a)
        grad = softmax_grad(a, conv_a, tau, top_k=top_k)

        count += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad * grad
        m_hat = m_v / (1 - beta1 ** count)
        v_hat = v_v / (1 - beta2 ** count)

        a = project(a - lr * m_hat / (np.sqrt(v_hat) + eps))

        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 3000:
            break

    return best_a, 2.0 * n * best_obj


def peak_reduction_step(a, conv, n, rng):
    """Targeted move: identify peak lag, shift mass from high-contributing positions."""
    peak_k = int(np.argmax(conv))
    # The peak at lag k is influenced by pairs (i, k-i)
    # To reduce it, move mass from positions with high a values in the "correlation window"
    m_start = max(0, peak_k - n + 1)
    m_end = min(n - 1, peak_k)
    window = a[m_start:m_end + 1].copy()
    partners = a[peak_k - np.arange(m_start, m_end + 1)]

    # Find position in window with max contribution (a[i] * a[k-i])
    contributions = window * partners
    if contributions.sum() < 1e-15:
        return a, False

    # Move mass from the highest contributor to a random low position
    src_local = int(np.argmax(contributions))
    src = m_start + src_local

    # Find a target position with low value
    candidates = np.where(a < 0.5 * a.mean())[0]
    if len(candidates) == 0:
        candidates = np.where(a < a.mean())[0]
    if len(candidates) == 0:
        return a, False

    dst = int(rng.choice(candidates))
    if dst == src:
        return a, False

    # Try moving a fraction of mass
    for frac in [0.02, 0.05, 0.01, 0.1]:
        delta = frac * a[src]
        if delta < 1e-10:
            continue
        a[src] -= delta
        a[dst] += delta
        conv_try = fast_convolve(a)
        if conv_try.max() < conv.max() - 1e-12:
            return a, True
        a[src] += delta
        a[dst] -= delta

    return a, False


def coord_descent(a0, time_limit, rng, n_moves_per_sec=500):
    """Coordinate descent with moves from high to low positions."""
    a = project(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        deadline_batch = time.monotonic() + min(1.0, (time_limit - (time.monotonic() - t0)))
        while time.monotonic() < deadline_batch:
            # Random pair move
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue

            for frac in [0.01, 0.03, 0.05, 0.005]:
                delta = frac * a[i]
                if delta < 1e-10:
                    break
                a[i] -= delta
                a[j] += delta
                conv_try = fast_convolve(a)
                if conv_try.max() < best_obj - 1e-12:
                    best_obj = conv_try.max()
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                a[i] += delta
                a[j] -= delta
            else:
                no_improve += 1

        if no_improve > 2000:
            break

    return best_a, 2.0 * n * best_obj


def nesterov_subgrad(a0, time_limit, lr0=0.005, momentum=0.9):
    """Nesterov-accelerated subgradient method on the exact max."""
    a = project(a0.astype(np.float64))
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    t_k = 1.0
    t0 = time.monotonic()
    no_improve = 0

    conv_a = fast_convolve(a)
    best_obj = conv_a.max()
    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        progress = (time.monotonic() - t0) / time_limit
        lr = lr0 * (1.0 + 2.0 * progress) ** (-0.5)

        conv_y = fast_convolve(y)
        grad = subgrad_max_only(y, conv_y)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        v = momentum * v - lr * grad
        y_new = y + v
        y_new = project(y_new)
        y_new = y_new + ((t_k - 1.0) / t_new) * (y_new - y)
        y_new = project(y_new)
        y = y_new
        t_k = t_new

        conv = fast_convolve(y)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 2000:
            break

    return best_a, 2.0 * n * best_obj

                    print(f"  CoordDesc: {score:.6f} *NEW BEST*")
                else:
                    print(f"  CoordDesc: {score:.6f}")

    # === Phase 5: Try larger n ===
    print("\n=== Phase 5: Larger lengths ===")
    for n_new in [1024, 2048]:
        if time.monotonic() > deadline - 60:
            break
        a_interp = np.interp(
            np.linspace(0, 1, n_new),
            np.linspace(0, 1, len(parent)),
            parent
        )
        a_interp = project(a_interp)
        t_alloc = min(80.0, deadline - time.monotonic() - 40)
        if t_alloc < 15:
            break
        a_opt, obj = fista_bt(a_interp, time_limit=t_alloc, tau_start=50, tau_end=500, lr0=0.008, top_k=100)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  n={n_new}: {score:.6f} *NEW BEST*")
        else:
            print(f"  n={n_new}: {score:.6f}")

    # === Phase 6: Final micro-polish ===
    print("\n=== Phase 6: Final polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_fin = deadline - time.monotonic() - 3
        if t_fin > 5:
            a_opt, obj = fista_bt(best_sequence, time_limit=t_fin,
                                   tau_start=500, tau_end=10000, lr0=0.002, top_k=50)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in best_sequence]
