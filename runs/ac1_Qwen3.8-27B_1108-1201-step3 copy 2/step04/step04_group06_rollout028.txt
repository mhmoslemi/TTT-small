Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
exact 1D subproblem solving, combined with multiplicative gradient updates
and multi-scale exploration. The key innovation is solving each coordinate
update exactly via 1D optimization (golden section search), which avoids

polishing phase.

import numpy as np

from scipy.optimize import minimize_scalar

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def coord_descent_pass(a, conv, s, rng, n_subsets=1):
    """One pass of coordinate descent over all coordinates in random order."""
    n = len(a)
    order = rng.permutation(n)
    
    for i in order:
        v_old = a[i]
        if v_old < 1e-18:
            # Try setting it to a small positive value
            # Just skip for now
            continue
        
        # Compute C = conv with a_i removed
        # conv[k] includes: for k in [i, i+n-1] (clamped), the terms involving a[i]
        k_start = max(0, i)
        k_end = min(2*n-1, i+n)  # exclusive
        
        C = conv.copy()
        # Remove contribution of a[i]
        for k in range(k_start, k_end):
            if k == 2*i:
                C[k] -= v_old * v_old
            else:
                C[k] -= 2.0 * v_old * a[k - i]
        
        # Precompute for fast 1D eval
        ks = np.arange(k_start, k_end)
        mask_not_2i = ks != 2*i
        a_shifted = a[ks - i]
        
        C_max_outside = 0.0
        if k_start > 0:
            C_max_outside = max(C_max_outside, C[:k_start].max())
        if k_end < 2*n-1:
            C_max_outside = max(C_max_outside, C[k_end:].max())
        
        # 1D objective
        def f1d(v):
            if v < 0:
                return np.inf
            vals = C[ks[mask_not_2i]] + 2.0 * v * a_shifted[mask_not_2i]
            local_max = vals.max()
            if 2*i < 2*n-1:
                local_max = max(local_max, C[2*i] + v*v)
            max_cv = max(C_max_outside, local_max)
            s_new = s - v_old + v
            if s_new < 1e-15:
                return np.inf
            return 2.0 * n * max_cv / (s_new * s_new)
        
        # Bracket
        f_old = f1d(v_old)
        v_lo = 0.0
        v_hi = max(v_old * 3, 1e-6)
        
        # Expand bracket
        for _ in range(30):
            if f1d(v_hi) < f_old:
                v_hi *= 2.0
            else:
                break
        
        # Also check smaller values
        v_test = v_old * 0.1
        if v_test > 0 and f1d(v_test) < f_old:
            v_lo = 0.0
            v_hi = v_old
        
        # Minimize
        result = minimize_scalar(f1d, bounds=(v_lo, v_hi), method='bounded',
                                  options={'xatol': 1e-12})
        v_new = max(0.0, result.x)
        
        # Update
        a[i] = v_new
        # Update conv incrementally
        delta = v_new - v_old
        if abs(delta) > 1e-18:

            s += delta
    
    return a, conv, s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact 1D coordinate descent
with golden-section refinement as the primary optimizer, combined with
multiplicative gradient updates for exploration, oscillatory initializations,
and multi-length search. The key innovation over prior approaches is solving
each coordinate update exactly via 1D optimization (exploiting the structure
that the convolution is linear/quadratic in each individual coordinate),
which avoids subgradient approximation errors and converges much faster.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_internal_score(a):
    """Compute 2*n*max(conv)/(sum)^2 internally."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def f1d_eval(v, n, conv_kr, a_shifted, mask_not_2i, outside_max, two_i, v_old, conv_2i, s):
    """
    1D objective for coordinate descent at coordinate i.
    v: candidate value for a[i]
    n: sequence length
    conv_kr: conv values at indices in k_range (array)
    a_shifted: a[k-i] values for k in k_range (array)
    mask_not_2i: boolean mask where k != 2i
    outside_max: max of conv outside k_range
    two_i: value of 2*i
    v_old: old value of a[i]
    conv_2i: conv[2*i] current value
    s: current sum
    """
    if v < 0.0:
        return np.inf
    d = v - v_old
    if mask_not_2i.any():
        inside_vals = conv_kr[mask_not_2i] + 2.0 * d * a_shifted[mask_not_2i]
        inside_max = float(inside_vals.max())
    else:
        inside_max = 0.0
    two_i_val = conv_2i + v * v - v_old * v_old
    max_cv = max(outside_max, inside_max, two_i_val)
    s_new = s + d
    if s_new < 1e-15:
        return np.inf
    return 2.0 * n * max_cv / (s_new * s_new)


def coord_descent_pass(a, conv, s, n, order, max_expand=40):
    """One full pass of coordinate descent over coordinates in given order."""
    for i in order:
        v_old = a[i]
        if v_old < 1e-16:
            v_old = 0.0

        k_start = i
        k_end = min(2 * n - 1, i + n)

        if k_start >= k_end:
            continue

        k_range = np.arange(k_start, k_end)
        two_i = 2 * i
        mask_not_2i = k_range != two_i
        a_shifted = a[k_range - i]

        # Compute outside max (conv values not in k_range)
        outside_max = 0.0
        if k_start > 0:
            outside_max = max(outside_max, float(conv[:k_start].max()))
        if k_end <= 2 * n - 2:
            outside_max = max(outside_max, float(conv[k_end:].max()))

        conv_2i = float(conv[two_i]) if two_i < 2 * n - 1 else 0.0

        # Evaluate at current value
        f_current = f1d_eval(v_old, n, conv[k_range], a_shifted, mask_not_2i,
                             outside_max, two_i, v_old, conv_2i, s)

        # Find bracket
        v_lo = 0.0
        v_hi = max(v_old * 5.0, s / n * 3.0, 1e-8)

        # Check if decreasing at v_old
        v_probe = v_old * 0.5
        if v_probe > 0:
            f_probe = f1d_eval(v_probe, n, conv[k_range], a_shifted, mask_not_2i,
                               outside_max, two_i, v_old, conv_2i, s)
            if f_probe < f_current:
                v_hi = v_old
                v_lo = 0.0
            else:
                v_hi = max(v_old * 3.0, s / n)
        else:
            v_hi = max(s / n * 2.0, 1e-6)

        # Expand upper bound if needed
        for _ in range(max_expand):
            f_hi = f1d_eval(v_hi, n, conv[k_range], a_shifted, mask_not_2i,
                            outside_max, two_i, v_old, conv_2i, s)
            if f_hi < f_current - 1e-15:
                v_hi *= 2.0
            else:
                break

        # Shrink lower bound: check if v=0 is better
        f_lo = f1d_eval(0.0, n, conv[k_range], a_shifted, mask_not_2i,
                        outside_max, two_i, v_old, conv_2i, s)

        # Only minimize if there's potential improvement
        if min(f_lo, f_current) >= f_current - 1e-14:
            # Check v=0 specifically
            if f_lo < f_current - 1e-14:
                v_new = 0.0
            else:
                continue
        else:
            # Use bounded scalar minimization
            try:
                result = minimize_scalar(
                    f1d_eval,
                    args=(n, conv[k_range], a_shifted, mask_not_2i,
                          outside_max, two_i, v_old, conv_2i, s),
                    bounds=(v_lo, v_hi),
                    method='bounded',
                    options={'xatol': 1e-14}
                )
                v_new = max(0.0, float(result.x))
            except Exception:
                v_new = v_old

        # Apply update
        delta = v_new - v_old
        if abs(delta) > 1e-18:
            a[i] = v_new
            # Update convolution incrementally
            for k in range(k_start, k_end):
                if k == two_i:
                    conv[k] += delta * (v_new + v_old)
                else:
                    conv[k] += 2.0 * delta * a[k - i]
            s += delta

    return a, conv, s


def coord_descent_optimize(a0, time_limit, verbose=True, seed=42):
    """Full coordinate descent optimization with restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    else:
        a = a / s
        s = 1.0
    n = len(a)
    conv = fast_convolve(a)

    best_obj = 2.0 * n * float(conv.max()) / (s * s)
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve_passes = 0
    pass_count = 0
    avg_a = a.copy()
    avg_count = 0

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        a, conv, s = coord_descent_pass(a, conv, s, n, order)

        pass_count += 1
        obj = 2.0 * n * float(conv.max()) / (s * s)

        # Running average
        avg_count += 1
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve_passes = 0
            if verbose and pass_count % 50 == 0:
                print(f"      CD pass {pass_count}: obj={best_obj:.10f}")
        else:
            no_improve_passes += 1

        # Check running average periodically
        if pass_count % 20 == 0:
            avg_s = float(avg_a.sum())
            if avg_s > 1e-10:
                avg_norm = avg_a / avg_s
                avg_conv = fast_convolve(avg_norm)
                avg_obj = 2.0 * n * float(avg_conv.max()) / 1.0
                if avg_obj < best_obj - 1e-13:
                    best_obj = avg_obj
                    best_a = avg_norm.copy()
                    a = avg_norm.copy()
                    s = 1.0
                    conv = avg_conv

        # Stagnation: perturb and restart
        if no_improve_passes >= 30:
            perturb_scale = min(0.05, 0.001 * no_improve_passes)
            a = best_a + rng.standard_normal(n) * perturb_scale
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s > 1e-10:
                a = a / s
                s = 1.0
                conv = fast_convolve(a)
            else:
                a = best_a.copy()
                s = 1.0
                conv = fast_convolve(a)
            no_improve_passes = 0
            avg_a = a.copy()
            avg_count = 0
            if verbose:
                print(f"      CD restart at pass {pass_count}, perturb={perturb_scale:.5f}")

    return best_a, best_obj


def multiplicative_gradient(a0, time_limit, eta0=1.0, temp0=0.003,
                            temp_decay=0.9997, eta_decay=0.99995,
                            top_k_frac=0.3, seed=0, verbose=False):
    """Multiplicative weights / exponentiated gradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()

    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Softmax-weighted subgradient
        if top_k < 2 * n - 1:
            indices = np.argpartition(conv, -top_k)[-top_k:]
            peek = conv[indices]
        else:
            indices = np.arange(2 * n - 1)
            peek = conv.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-14:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = float(conv_new.max())

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"      EG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        eta *= eta_decay
        temp = max(temp * temp_decay, 1e-6)

        if no_improve > 2000:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.1)

    return best_a, best_obj


def generate_oscillatory_init(n, rng, freq_range=(2, 12)):
    """Generate oscillatory non-negative initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    for freq in np.linspace(freq_range[0], freq_range[1], 8):
        # Rectified cosine
        for phase in [0.0, 0.5, 1.0]:
            sig = np.cos(2.0 * np.pi * freq * x / n + phase)
            a = np.maximum(sig, 0.0) + 0.01
            a = a / a.sum()
            inits.append(a)

        # Exponentiated cosine (always positive)
        a = np.exp(0.5 * np.cos(2.0 * np.pi * freq * x / n))
        a = a / a.sum()
        inits.append(a)

    # Alternating high-low
    ratio = rng.uniform(0.3, 0.7)
    a = np.where(np.arange(n) % 2 == 0, 1.0, ratio)
    inits.append(a / a.sum())

    # Edge-concentrated
    edge_w = np.minimum(x, n - 1 - x).astype(float)
    a = 1.0 / (edge_w + 1.0)
    inits.append(a / a.sum())

    # Gaussian center
    a = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(a / a.sum())

    # Random Dirichlet
    for _ in range(5):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.dirichlet(np.ones(n) * alpha)
        inits.append(a)

    return inits

        a_opt, _ = coord_descent_optimize(f_init, time_limit=t_alloc,
                                           verbose=False, seed=seed + 200 + i)
        try_seq(a_opt, f"osc_init{i}")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Different lengths ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        alt_lengths = []
        n_cur = len(best_seq)
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0]:
            n_alt = int(n_cur * factor)
            if 64 <= n_alt <= 1024 and n_alt != n_cur:
                alt_lengths.append(n_alt)

        for li, n_alt in enumerate(alt_lengths[:4]):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_alloc = min(rem / max(1, len(alt_lengths[:4]) - li), 90)
            if t_alloc < 10:
                break
            # Interpolate to new length
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_seq)
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()

            a_opt, _ = coord_descent_optimize(a_alt, time_limit=t_alloc,
                                               verbose=False, seed=seed + 300 + li)
            try_seq(a_opt, f"len{n_alt}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coord_descent_optimize(best_seq, time_limit=min(rem, 120),
                                               verbose=False, seed=seed + 500)
            try_seq(a_opt, "final_cd")

        # One more EG pass with small learning rate
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_opt, _ = multiplicative_gradient(best_seq, time_limit=min(rem, 30),
                                                eta0=0.3, temp0=0.001,
                                                temp_decay=0.99999, eta_decay=0.99999,
                                                top_k_frac=0.5, seed=seed + 600)
            try_seq(a_opt, "final_eg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
