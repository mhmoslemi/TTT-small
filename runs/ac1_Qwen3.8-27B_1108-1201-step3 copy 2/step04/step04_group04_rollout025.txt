Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP with line search,
multiplicative gradient descent, and multi-scale search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def active_set_lp_step(a, conv, active_frac=0.05, max_active=50):
    """
    Solve active-set LP to find descent direction.
    Returns (direction, success).
    """
    n = len(a)
    M = conv.max()
    
    # Find active set
    threshold = M * (1 - active_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        top_idx = np.argsort(conv[active])[-max_active:]
        active = active[top_idx]
    
    # Build LP: min M' s.t. c[k] + 2*(A_k @ d) <= M', sum(d)=0, a+d>=0

def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

            # Adaptive restart
            lr = max(0.01, lr * 3)
            temp = max(0.001, temp * 0.5)
            no_improve = 100
    
    return best_a, best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.002):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        if y.sum() > 1e-15:
            y = y / y.sum()
        
        conv_y = fast_convolve(y)
        grad = compute_gradient(y, conv_y, top_k=top_k, temp=temp)
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_trials_per_batch=50000):
    """Random coordinate descent polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    trial = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of random moves
        batch = max(1, min(n_trials_per_batch, int(2e7 / max(1, n))))
        
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            # Random pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try several delta values
            base = min(a[i], a[j])
            deltas = [0.002, 0.005, 0.01, 0.02, 0.05]
            found = False
            for d in deltas:
                if d > base:
                    continue
                for sign in [1, -1]:
                    a_try = a.copy()
                    a_try[i] += sign * d
                    a_try[j] -= sign * d
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1
        
        trial += 1
        if no_improve > 300:
            break
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Sine patterns
    for freq, name in [(1, 'sin1'), (2, 'sin2'), (3, 'sin3'), (5, 'sin5')]:
        v = np.sin(np.pi * freq * x / n) ** 2 + 0.01
        inits[name] = v / v.sum()
    
    # Cosine with phase
    v = np.cos(2 * np.pi * x / n + np.pi / 4) ** 2 + 0.01
    inits['cos_phase'] = v / v.sum()
    
    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Asymmetric bimodal (more mass on one side)
    bi2 = np.zeros(n)
    bi2[:n//4] = 1.5
    bi2[-n//4:] = 1.0
    inits['asym_bi'] = bi2 / bi2.sum()
    
    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Quick gradient scan across lengths
    Phase 2: Active-set LP deep optimization
    Phase 3: Multiplicative gradient + Nesterov polishing
    Phase 4: Coordinate descent fine-tuning
    Phase 5: Final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
            # Get parent score
            ps = float(evaluate_sequence(p.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = p.copy()
                print(f"Parent score: {ps:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    # =========================================================
    # Phase 1: Quick exploration (300s)
    # =========================================================
    print("\n=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()
    phase1_time = 300.0
    
    lengths_to_try = [512, 768, 1024, 1536]
    if parent is not None:
        lengths_to_try.insert(1, len(parent))  # Include parent length
    
    # Remove duplicates
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    candidates = []  # (score, n, a)
    
    n_per = max(20.0, (phase1_time - 10) / len(lengths_to_try))
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 1200:
            break
        if n < 16 or n > 4096:
            continue
        
        inits = generate_inits(n, parent, rng)
        
        # Prioritize
        priority = ['parent', 'parent_interp', 'uniform', 'sin1', 'asym_bi', 'random1']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]  # Limit to 3 per length
        
        t_per_init = max(10.0, n_per / len(init_names))
        
        for name in init_names:
            if time.monotonic() > deadline - 1100:
                break
            
            a0 = inits[name]
            
            # Quick Nesterov optimization
            t_alloc = min(t_per_init, deadline - time.monotonic() - 1000)
            if t_alloc < 5:
                break
            
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                            lr_init=0.015, momentum=0.92,
                                            top_k=30, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n}, {name}: {score:.6f}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: Active-set LP deep optimization (1200s)
    # =========================================================
    print("\n=== Phase 2: Active-Set LP Deep Optimization ===")
    t2 = time.monotonic()
    
    # Select top candidates (unique by length)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in candidates:
        if n not in seen_lengths:
            top_configs.append((n, a_opt))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break
    
    # Always include best
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 400.0
    if remaining < 60:
        remaining = 60
    
    t_per_lp = max(30.0, remaining * 0.7 / max(1, len(top_configs)))
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 300)
        if t_alloc < 20:
            break
        
        print(f"  AS-LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        
        a_opt, obj = lp_optimize_active(a0, time_limit=t_alloc, verbose=True, trust_frac=0.3)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  AS-LP {i} (n={n}): {score:.6f} ***NEW BEST***")
            else:
                print(f"  AS-LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Multiplicative + Nesterov polishing (300s)
    # =========================================================
    print("\n=== Phase 3: Polishing ===")
    t3 = time.monotonic()
    
    if best_sequence is not None:
        # Multiple polishing runs with different configs
        polish_configs = [
            ('multi', {'lr_init': 0.02, 'top_k': 30, 'temp_init': 0.002}),
            ('multi', {'lr_init': 0.01, 'top_k': 50, 'temp_init': 0.001}),
            ('nesterov', {'lr_init': 0.008, 'momentum': 0.93, 'top_k': 40, 'temp_init': 0.001}),
            ('nesterov', {'lr_init': 0.015, 'momentum': 0.9, 'top_k': 20, 'temp_init': 0.003}),
        ]
        
        remaining = deadline - time.monotonic() - 150.0
        for pi, (method, cfg) in enumerate(polish_configs):
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(60.0, remaining / (len(polish_configs) - pi) * 0.8)
            if t_alloc < 10:
                break
            
            if method == 'multi':
                a_opt, obj = multiplicative_optimize(best_sequence, time_limit=t_alloc, **cfg)
            else:
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc, **cfg)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {method} {pi}: {score:.6f} ***NEW BEST***")
            else:
                print(f"  Polish {method} {pi}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: AS-LP polish on single best (200s)
    # =========================================================
    print("\n=== Phase 4: Final AS-LP Polish ===")
    t4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            # Run AS-LP with tighter trust region for final polish
            a_opt, obj = lp_optimize_active(best_sequence, time_limit=remaining * 0.6,
                                             trust_frac=0.15)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final AS-LP: {score:.6f} ***NEW BEST***")
            
            # Short multiplicative polish
            remaining = deadline - time.monotonic() - 50.0
            if remaining > 10:
                a_opt, obj = multiplicative_optimize(best_sequence, time_limit=remaining,
                                                      lr_init=0.01, top_k=50, temp_init=0.001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final multi: {score:.6f} ***NEW BEST***")
    
    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Coordinate descent (remaining)
    # =========================================================
    print("\n=== Phase 5: Coordinate Descent ===")
    
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_cd = max(5.0, min(60.0, deadline - time.monotonic() - 10))
        a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  CoordDesc: {score:.6f} ***NEW BEST***")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np
