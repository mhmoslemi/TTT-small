def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate(w, f):
    """Compute cross-correlation R(j) = sum_k w(k) * f(k-j) for j=0..n-1."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    R = np.fft.irfft(W * np.conj(F), fft_size)
    return R[:n]

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def cross_correlate(w, f):

    n = len(f)

    R = np.fft.irfft(W * np.conj(F), fft_size)

            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps):
                best_seq = parent.copy()
                best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n = 384
        parent = np.ones(n) / n
        ps = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        best_score = ps
        print(f"Default init: n={n}, score={ps:.8f}")

    n = len(best_seq)

    # Phase 1: Dual-primal mirror descent
    rem = deadline - time.monotonic() - 950
    if rem > 30:
        print(f"\n=== Phase 1: Dual-Primal MD ({min(rem,800):.0f}s) ===")
        t_alloc = min(rem, 800)
        a_opt, _ = dual_primal_md(best_seq, time_limit=t_alloc, base_lr_primal=0.06, base_lr_dual=0.4)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 1 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 2: All-peaks mirror descent with different init
    rem = deadline - time.monotonic() - 600
    if rem > 20:
        print(f"\n=== Phase 2: All-Peaks MD ({min(rem,350):.0f}s) ===")
        t_alloc = min(rem, 350)
        a_opt, _ = md_all_peaks(best_seq, time_limit=t_alloc, base_lr=0.06, temp_init=0.001)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 2 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 3: Peak equalization
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        print(f"\n=== Phase 3: Peak Equalization ({min(rem,300):.0f}s) ===")
        t_alloc = min(rem, 300)
        a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, base_lr=0.015, num_peaks=30)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 3 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 4: Top-peaks mirror descent (different from phase 2)
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        print(f"\n=== Phase 4: Top-Peaks MD ({min(rem,200):.0f}s) ===")
        t_alloc = min(rem, 200)
        a_opt, _ = md_top_peaks(best_seq, time_limit=t_alloc, base_lr=0.03, top_k=10, temp_init=0.0005)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 4 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 5: Fine polish
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        print(f"\n=== Phase 5: Fine Polish ({min(rem,150):.0f}s) ===")
        t_alloc = min(rem, 150)
        a_opt, _ = fine_polish(best_seq, time_limit=t_alloc, init_scale=0.0008)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 5 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 6: Second round of dual-primal from best
    rem = deadline - time.monotonic() - 40
    if rem > 15:
        print(f"\n=== Phase 6: DP MD round 2 ({min(rem,100):.0f}s) ===")
        t_alloc = min(rem, 100)
        a_opt, _ = dual_primal_md(best_seq, time_limit=t_alloc, base_lr_primal=0.04, base_lr_dual=0.3)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 6 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Phase 7: Final fine polish
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        print(f"\n=== Phase 7: Final Polish ({min(rem,30):.0f}s) ===")
        t_alloc = min(rem, 30)
        a_opt, _ = fine_polish(best_seq, time_limit=t_alloc, init_scale=0.0003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        print(f"  Phase 7 result: {sc:.8f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
