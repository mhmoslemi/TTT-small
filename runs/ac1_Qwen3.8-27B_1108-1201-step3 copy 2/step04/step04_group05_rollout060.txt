import numpy as np
from scipy.optimize import linprog, minimize
import time

def fft_convolve(a, a2=None):
    """Linear convolution via FFT."""
    if a2 is None:
        a2 = a
    n = len(a)
    m = len(a2)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(a2, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def softmax(t):
    """Numerically stable softmax."""
    t_shifted = t - t.max()
    exp_t = np.exp(t_shifted)
    return exp_t / exp_t.sum()

    """Numerically stable log-sum-exp."""
    x_max = np.max(x, axis=axis, keepdims=True)
    result = x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))
    if axis is not None:
        result = np.squeeze(result, axis=axis)
    return result

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fft_convolve(a_norm, a_norm)
    return 2.0 * n * b.max()

    """Numerically stable softmax."""
    t_shifted = t - t.max()
    exp_t = np.exp(t_shifted)
    return exp_t / exp_t.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute objective value (score) of a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_n = a / s
    b = fft_convolve(a_n, a_n)
    return 2.0 * n * b.max()


def multiplicative_optimize(f0, time_limit, eta_init=0.3, eta_min=0.005,
                            n_peaks=40, cool_rate=0.9998, mom_init=0.5,
                            verbose=False, seed=42):
    """Optimize using multiplicative weight updates with momentum."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fft_convolve(a, a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    mom = mom_init
    temp = max(best_obj * 0.05, 1e-10)
    v = np.zeros(n)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fft_convolve(a, a)
        cur_obj = b.max()

        # Get top peaks
        m = len(b)
        k_peaks = min(n_peaks, m)
        if k_peaks < m:
            peaks = np.argpartition(b, -k_peaks)[-k_peaks:]
        else:
            peaks = np.arange(m)

        pv = b[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(k_peaks)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient
        grad = np.zeros(n)
        for idx in range(k_peaks):
            k = int(peaks[idx])
            wi = w[idx]
            if wi < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kj = k - j
                if k == 2 * j:
                    grad[j] += wi * a[kj]
                else:
                    grad[j] += 2.0 * wi * a[kj]

        # Momentum update
        v = mom * v - eta * grad
        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-30)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a_new = a_new / a_new_s

        b_new = fft_convolve(a_new, a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new

        eta = max(eta * cool_rate, eta_min)
        mom = min(mom * 1.0005, 0.95)
        temp *= 1.003

        if no_improve > 150:
            scale = 0.005 + 0.002 * (no_improve // 150)
            pert = best_a + rng.standard_normal(n) * scale
            a = project_simplex(pert)
            v = np.zeros(n)
            eta = max(eta * 2.0, eta_init * 0.5)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    Multiplicative iter {it}: best={best_obj:.10f}, eta={eta:.5f}, mom={mom:.3f}")

    return best_a, best_obj


def lse_obj_grad(t, sigma, reg=1e-10):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. t (softmax params)."""
    a = softmax_fn(t)
    n = len(a)
    b = fft_convolve(a, a)
    m = len(b)

    # Log-sum-exp of b/sigma
    b_scaled = b / sigma
    b_max = b_scaled.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b_scaled - b_max)))
    obj = sigma * log_sum_exp + 0.5 * reg * np.dot(t, t)

    # Weights: softmax(b/sigma)
    w = np.exp(b_scaled - b_max)
    w /= w.sum()

    # Gradient computation:
    # grad_t[j] = sum_k w[k] * a[j] * (a[k-j]*(2-delta_{k,2j}) - 2*b[k]) + reg * t[j]
    # = a[j] * (C[j] - 2*S) - a[j]^2 * w[2j] + reg * t[j]
    # where C[j] = sum_k w[k] * a[k-j], S = sum_k w[k] * b[k]

    S = np.dot(w, b)

    # Compute C[j] = sum_k w[k] * a[k-j] via cross-correlation
    # C[j] = sum_k w[k] * a[k-j] for 0<=k<m, 0<=a_idx<n
    # This is the cross-correlation of w (length m) and a (length n)
    # C[j] = sum_k w[k] * a[k-j]
    # Let's compute via FFT: pad both to size >= m+n-1
    fft_sz = 1 << (m + n - 2).bit_length()
    W = np.fft.rfft(w, fft_sz)
    A = np.fft.rfft(a, fft_sz)
    # Cross-correlation: ifft(conj(W) * A) gives sum_k w[k] * a[(j+k) mod N]
    # We want sum_k w[k] * a[k-j] = sum_k w[k-j] * a[k] ... let me think differently
    # C[j] = sum_k w[k] * a[k-j]
    # = sum_k w[k] * a[k-j]
    # If we define d[l] = w[l] and e[l] = a[l], then C[j] = sum_l d[l] * e[l-j]
    # This is the cross-correlation (d, e)[j] = sum_l d[l] * conj(e[l-j])
    # For real: sum_l d[l] * e[l-j]
    # = ifft(FFT(d) * conj(FFT(e)))[j] ... let me verify
    # ifft(FFT(d)*conj(FFT(e)))[j] = sum_l d[l*] * e[(j-l) mod N] ... hmm
    # Let me just use the direct definition via numpy correlate or manual FFT

    # Actually: circular cross-correlation r[j] = sum_k d[k] * e[(k-j) mod N]
    # = ifft(FFT(d) * conj(FFT(e)))[j]
    # We want C[j] = sum_k w[k] * a[k-j] which is exactly r[j] with d=w, e=a
    # (assuming we're in the circular regime, which is approximately true for our sizes)

    C_circ = np.fft.irfft(W * np.conj(A), fft_sz)[:m + n - 1]
    # C_circ[j] = sum_k w[k] * a[(k-j) mod fft_sz]
    # For j in [0, n-1], C_circ[j] = sum_k w[k] * a[k-j] (with a[k-j]=0 if out of range)
    # This is what we want for j = 0, 1, ..., n-1

    C = C_circ[:n]

    # Correction for the delta term: -a[j]^2 * w[2j]
    # w[2j] is only defined if 2j < m
    delta_correction = np.zeros(n)
    valid = 2 * np.arange(n) < m
    delta_correction[valid] = a[valid] ** 2 * w[2 * np.arange(n)[valid]]

    grad = a * (C - 2.0 * S) - delta_correction + reg * t

    return obj, grad


def lbfgs_optimize(f0, time_limit, sigma_init=None, sigma_min=None,
                   max_iter=300, verbose=False, seed=42):
    """L-BFGS optimization on softmax-parameterized log-sum-exp smoothed objective."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Initialize t from a
    t = np.log(np.maximum(a, 1e-30))

    b = fft_convolve(a, a)
    cur_max_b = b.max()

    if sigma_init is None:
        sigma_init = cur_max_b * 0.1
    if sigma_min is None:
        sigma_min = cur_max_b * 0.001

    best_obj = cur_max_b
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Annealing schedule: geometric from sigma_init to sigma_min
    n_anneal_steps = 8
    sigmas = np.geomspace(sigma_init, sigma_min, n_anneal_steps)

    for sigma_idx, sigma in enumerate(sigmas):
        t_sigma_end = min(t0 + time_limit * ((sigma_idx + 1) / n_anneal_steps),
                          t0 + time_limit)
        t_sigma_start = t0 + time_limit * (sigma_idx / n_anneal_steps) if sigma_idx > 0 else t0
        budget = max(1.0, t_sigma_end - max(t0, t_sigma_start) - 2.0)

        if time.monotonic() > t0 + time_limit - 5:
            break

        def func(t_in):
            obj, _ = lse_obj_grad(t_in, sigma)
            return obj

        def func_grad(t_in):
            obj, grad = lse_obj_grad(t_in, sigma)
            return obj, grad

        # Run L-BFGS-B
        res = minimize(func_grad, t.copy(), method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'gtol': 1e-8, 'ftol': 1e-12,
                                'maxls': 20})

        t_new = res.x
        a_new = softmax_fn(t_new)
        b_new = fft_convolve(a_new, a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            t = t_new
            if verbose:
                print(f"    L-BFGS sigma={sigma:.8f}: obj={best_obj:.10f}, iters={res.nit}")
        else:
            # Try with slight perturbation
            t_pert = t + rng.standard_normal(n) * 0.01
            a_pert = softmax_fn(t_pert)
            for _ in range(10):
                res2 = minimize(func_grad, t_pert.copy(), method='L-BFGS-B',
                                options={'maxiter': 50, 'maxfun': 100,
                                         'gtol': 1e-6, 'ftol': 1e-10})
                a_test = softmax_fn(res2.x)
                b_test = fft_convolve(a_test, a_test)
                if b_test.max() < best_obj:
                    best_obj = b_test.max()
                    best_a = a_test.copy()
                    t = res2.x
                    break
                t_pert = res2.x + rng.standard_normal(n) * 0.005

    return best_a, best_obj


def build_conv_matrix_row(f, k):
    """Get the row of the convolution matrix corresponding to lag k."""
    n = len(f)
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def fw_lp_optimize(f0, time_limit, verbose=False, seed=42):
    """Frank-Wolfe with linearized LP oracle (simplified)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fft_convolve(f, f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fft_convolve(f, f)
        current_obj = conv.max()

        # Find top peaks
        k_peaks = min(20, len(conv))
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # Build LP: min t s.t. t >= 2 * sum_j B[k,j] * g_j - conv[k] for peak k
        # where g = target - f, sum(g) = 0, g >= -f
        # Variables: [t, g_0, ..., g_{n-1}]
        c = np.zeros(n + 1)
        c[0] = 1.0

        A_ub = np.zeros((k_peaks, n + 1))
        A_ub[:, 0] = -1.0
        b_ub = np.zeros(k_peaks)

        for ii, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[ii, j + 1] = 2.0 * f[k - j]
            b_ub[ii] = -conv[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)

        bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            no_improve += 3
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        g = res.x[1:]
        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1.0 - alpha) * f + alpha * target
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            val = fft_convolve(f_new, f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_val < best_val + 0 and best_alpha > 0:
            # Fine search
            lo = max(0.01, best_alpha * 0.7)
            hi = min(1.0, best_alpha * 1.3)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                f_new = (1.0 - mid) * f + mid * target
                f_new = np.maximum(f_new, 0)
                fs = f_new.sum()
                if fs < 1e-15:
                    break
                f_new = f_new / fs
                val = fft_convolve(f_new, f_new).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0)
            f = f / f.sum()
            M = fft_convolve(f, f).max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0

    return best_f, best_obj


def gradient_descent_optimize(f0, time_limit, lr_init=0.005, lr_min=0.0001,
                               mom=0.9, n_peaks=50, verbose=False, seed=42):
    """Projected gradient descent with momentum and adaptive step size."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fft_convolve(a, a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    temp = best_obj * 0.05
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fft_convolve(a, a)
        cur_obj = b.max()

        # Weighted subgradient
        k_peaks = min(n_peaks, len(b))
        peaks = np.argpartition(b, -k_peaks)[-k_peaks:]
        pv = b[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        grad = np.zeros(n)
        for idx in range(k_peaks):
            k = int(peaks[idx])
            wi = w[idx]
            if wi < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                if k == 2 * j:
                    grad[j] += wi * a[k - j]
                else:
                    grad[j] += 2.0 * wi * a[k - j]

        # Momentum + projected gradient step
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        b_new = fft_convolve(a_new, a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = a_new

        lr = max(lr * 0.9995, lr_min)
        temp *= 1.002

        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = max(lr * 3, lr_init * 0.3)
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    GD iter {it}: best={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 0.8))

    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    for sigma_frac in [0.1, 0.2, 0.35]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())

    # Asymmetric bumps
    for offset_frac in [0.25, 0.35]:
        center_off = offset_frac * (n - 1)
        sigma = 0.15 * n
        g = np.exp(-0.5 * ((x - center_off) / sigma) ** 2)
        inits.append(g / g.sum())

    # Cosine variants
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.02
        inits.append(c / c.sum())

    if parent is not None and len(parent) != n:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining multiplicative updates,
    L-BFGS with annealing, Frank-Wolfe, and gradient descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            par_score = obj_val(parent)
            print(f"Parent loaded: n={len(parent)}, approx_score={par_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multiplicative updates (fast exploration) ===
    print("\n=== Phase 1: Multiplicative Updates ===")
    t_start = time.monotonic()
    remaining = deadline - t_start
    t_p1 = min(remaining * 0.25, 350)

    if t_p1 > 20:
        print(f"  Budget: {t_p1:.0f}s")
        a_opt, _ = multiplicative_optimize(parent, time_limit=t_p1,
                                           eta_init=0.4, eta_min=0.003,
                                           n_peaks=50, cool_rate=0.9997,
                                           mom_init=0.4, verbose=True, seed=seed)
        try_seq(a_opt, "multiplicative")

        # Also try gradient descent
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt, _ = gradient_descent_optimize(parent, time_limit=min(rem * 0.3, 100),
                                                  lr_init=0.003, lr_min=0.0001,
                                                  mom=0.92, n_peaks=50,
                                                  verbose=False, seed=seed+1)
            try_seq(a_opt, "gd")

    print(f"  Phase 1 best: {best_score:.10f}")

    # === Phase 2: L-BFGS with annealing (precise optimization) ===
    print("\n=== Phase 2: L-BFGS Annealing ===")
    remaining = deadline - time.monotonic()
    t_p2 = min(remaining * 0.35, 500)

    if t_p2 > 30 and best_seq is not None:
        print(f"  Budget: {t_p2:.0f}s, n={len(best_seq)}")
        n_best = len(best_seq)
        b_best = fft_convolve(best_seq / best_seq.sum())
        cur_max_b = b_best.max()

        # Run L-BFGS with different starting sigmas
        for sigma_mult in [0.15, 0.08, 0.04]:
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem * 0.35, 120)
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_alloc,
                                       sigma_init=cur_max_b * sigma_mult,
                                       sigma_min=cur_max_b * 0.001,
                                       max_iter=200, verbose=True, seed=seed+10)
            try_seq(a_opt, f"lbfgs_s{sigma_mult}")

        # L-BFGS from perturbed start
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            for pert_scale in [0.005, 0.01, 0.02]:
                rem2 = deadline - time.monotonic() - 40
                if rem2 < 20:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
                t_alloc = min(rem2 * 0.4, 80)
                a_opt, _ = lbfgs_optimize(f_pert, time_limit=t_alloc,
                                           sigma_init=cur_max_b * 0.1,
                                           sigma_min=cur_max_b * 0.0005,
                                           max_iter=150, verbose=False, seed=seed+20)
                try_seq(a_opt, f"lbfgs_pert{pert_scale}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: Frank-Wolfe with LP (larger structural changes) ===
    print("\n=== Phase 3: Frank-Wolfe LP ===")
    remaining = deadline - time.monotonic()
    t_p3 = min(remaining * 0.2, 250)

    if t_p3 > 30 and best_seq is not None:
        print(f"  Budget: {t_p3:.0f}s, n={len(best_seq)}")
        n_best = len(best_seq)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_p3 * 0.7,
                                   verbose=True, seed=seed+30)
        try_seq(a_opt, "fw_lp")

        # Perturbed FW
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=min(rem * 0.5, 100),
                                       verbose=False, seed=seed+31)
            try_seq(a_opt, "fw_pert")

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Fine multiplicative + L-BFGS polish ===
    print("\n=== Phase 4: Fine Polish ===")
    remaining = deadline - time.monotonic()
    t_p4 = min(remaining * 0.2, 200)

    if t_p4 > 20 and best_seq is not None:
        print(f"  Budget: {t_p4:.0f}s")
        n_best = len(best_seq)

        # Fine multiplicative
        t_mp = min(t_p4 * 0.4, 60)
        a_opt, _ = multiplicative_optimize(best_seq, time_limit=t_mp,
                                           eta_init=0.1, eta_min=0.001,
                                           n_peaks=60, cool_rate=0.9999,
                                           mom_init=0.7, verbose=False, seed=seed+40)
        try_seq(a_opt, "fine_mult")

        # Fine L-BFGS
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            b_best = fft_convolve(best_seq / best_seq.sum())
            cur_max_b = b_best.max()
            t_lb = min(rem * 0.6, 100)
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_lb,
                                       sigma_init=cur_max_b * 0.05,
                                       sigma_min=cur_max_b * 0.0003,
                                       max_iter=300, verbose=True, seed=seed+50)
            try_seq(a_opt, "fine_lbfgs")

    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Try different lengths ===
    print("\n=== Phase 5: Different Lengths ===")
    remaining = deadline - time.monotonic()
    t_p5 = min(remaining * 0.15, 150)

    if t_p5 > 20 and best_seq is not None:
        print(f"  Budget: {t_p5:.0f}s")
        n_best = len(best_seq)
        n_tries = [n_best - 16, n_best + 16, n_best + 32, 512, 448]
        n_tries = sorted(set(max(64, x) for x in n_tries if x != n_best))[:4]

        for n_try in n_tries:
            rem = deadline - time.monotonic() - 20
            if rem < 15 or n_try < 32:
                continue
            # Interpolate parent to new length
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps < 1e-10:
                continue
            px = px / ps

            # Quick multiplicative + L-BFGS
            t_each = min(rem * 0.3, 40)
            a_opt, _ = multiplicative_optimize(px, time_limit=t_each * 0.5,
                                               eta_init=0.2, n_peaks=30,
                                               seed=seed+60)
            try_seq(a_opt, f"len{n_try}_mult")

            rem2 = deadline - time.monotonic() - 15
            if rem2 > 10:
                b_try = fft_convolve(a_opt / a_opt.sum())
                a_opt2, _ = lbfgs_optimize(a_opt, time_limit=min(rem2 * 0.5, 25),
                                           sigma_init=b_try.max() * 0.05,
                                           sigma_min=b_try.max() * 0.001,
                                           max_iter=100, verbose=False, seed=seed+61)
                try_seq(a_opt2, f"len{n_try}_lbfgs")

    print(f"  Phase 5 best: {best_score:.10f}")

    # === Phase 6: Final intensive polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 20 and best_seq is not None:
        print(f"  Budget: {remaining:.0f}s")
        n_best = len(best_seq)

        # Multiple short L-BFGS runs from slightly different starts
        b_best = fft_convolve(best_seq / best_seq.sum())
        cur_max_b = b_best.max()

        for i in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break
            if i == 0:
                f_start = best_seq.copy()
            else:
                f_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002 * (i + 1))

            t_alloc = min(rem * 0.3, 30)
            a_opt, _ = lbfgs_optimize(f_start, time_limit=t_alloc,
                                       sigma_init=cur_max_b * 0.03,
                                       sigma_min=cur_max_b * 0.0002,
                                       max_iter=200, verbose=False, seed=seed+100+i)
            try_seq(a_opt, f"final_{i}")

        # One more multiplicative run
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = multiplicative_optimize(best_seq, time_limit=rem * 0.8,
                                               eta_init=0.05, eta_min=0.001,
                                               n_peaks=80, cool_rate=0.99995,
                                               mom_init=0.8, verbose=False, seed=seed+200)
            try_seq(a_opt, "final_mult")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f} (n=256)")

    return [float(x) for x in best_seq]
