Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adam optimization with log-sum-exp smoothed objective and adaptive temperature
2. LP-based Frank-Wolfe for large jumps
3. Spectral initialization (Gerchberg-Saxton style)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_max_grad(a, conv, T):
    """Compute gradient of (1/T)*log(sum(exp(T*conv_k))) w.r.t. a."""
    n = len(a)
    scaled = T * conv
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    # grad = 2 * cross_correlation(T*p, a)
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.fft(T * p, N)
    A = np.fft.fft(a, N)
    grad_a = 2.0 * np.real(np.fft.ifft(np.conj(W) * A, N)[:n])
    return grad_a


def obj_normalized(a):
    """Objective for a normalized (sum=1) sequence."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    T = T_init
    rng = np.random.default_rng(seed)

    best_f = f.copy()

def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None

def spectral_init(n, n_iters=20, seed=42):
    """Gerchberg-Saxton style spectral initialization."""
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()

    for _ in range(n_iters):
        # Compute power spectrum
        N = 1 << (2 * n - 1).bit_length()
        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2

        # Target: flat power spectrum (but keep DC component)
        target_power = np.ones(N)
        target_power[0] = power[0]  # Keep DC to maintain sum

        # Inverse FFT with target spectrum, original phases
        target_amplitude = np.sqrt(target_power)
        new_A = target_amplitude * (A / np.abs(A + 1e-15))
        a_new = np.real(np.fft.ifft(new_A, N)[:n])
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        # else keep old a

    return a


def cosine_init(n, m=5):
    """Initialize with a cosine pattern."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.7 * np.cos(2.0 * np.pi * m * x / n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def local_search(f0, time_limit, seed=123):
    """Random perturbation local search."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    scale = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of perturbations
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()

        scale *= 0.97
        if scale < 1e-5:
            scale = 0.003

        it += 1

    return best_f, best_obj


def coordinate_descent(f0, time_limit, seed=456):
    """Coordinate-wise descent: try adjusting pairs of coordinates."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Pick random pair of coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try to shift mass from i to j (or j to i)
        for frac in [0.1, 0.25, 0.5, 0.75]:
            f_try = best_f.copy()
            f_try[j] += frac * f_try[i]
            f_try[i] -= frac * f_try[i]
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            obj = fast_convolve(f_try).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_try.copy()
                break
        else:
            # Try shifting from j to i
            for frac in [0.1, 0.25, 0.5, 0.75]:
                f_try = best_f.copy()
                f_try[i] += frac * f_try[j]
                f_try[j] -= frac * f_try[j]
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-15:
                    continue
                f_try = f_try / s_try
                obj = fast_convolve(f_try).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f_try.copy()
                    break

    return best_f, best_obj
