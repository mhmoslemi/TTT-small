def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(v):
    v = v - v.max()
    e = np.exp(v)
    return e / e.sum()

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:

    return None

    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax(t)
    conv = fast_convolve(a)
    
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Objective (we only need it up to a constant for optimization)
    # F = conv_max + log(sum(exp(shifted))) / beta

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based Frank-Wolfe (primary local optimizer)
2. L-BFGS on smoothed log-sum-exp objective in softmax parameterization
3. Multiplicative Weight Updates with smooth gradient
4. Multi-length exploration with interpolation
5. Coordinate descent and subgradient polishing
Key innovation: L-BFGS in t-space with annealed beta provides a different
escape mechanism from local optima compared to LP-FW.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    conv = fast_convolve(f)

    best_obj = M

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)

def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space (for L-BFGS)."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * sum_m a[m] * w[m+i]  (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    # grad_t = a * (grad_a - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 30:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            res = minimize(
                _smooth_obj_grad_impl, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 300, 'gtol': 1e-9, 'maxls': 50,
                         'maxfun': 2000}
            )
            t = res.x
            a = softmax_fn(t)

            if a_sum > 1e-15:

def coordinate_descent(f0, time_limit):

def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    t2 = min(500, deadline - time.monotonic() - 600)
    if t2 > 40:
        a_opt, _ = lbfqg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # L-BFGS from perturbed start
        t2b = min(150, deadline - time.monotonic() - 500)
        if t2b > 20:
            noise = rng.standard_normal(len(base)) * 0.01
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b)
            try_seq(a_opt, "lbfqg_perturbed")

    # === Phase 3: MWU optimization (200s) ===
    print(f"\n=== Phase 3: MWU (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3 = min(200, deadline - time.monotonic() - 400)
    if t3 > 20:
        a_opt, _ = mwu_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "mwu_main")

    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 80)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Deep optimization on best (400s) ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 200)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5b: L-BFGS deep
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.6, 150)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 5c: Coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.7, 120)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

        # 5d: Subgradient polish
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.8, 80)
            a_opt, _ = sg_polish(best_seq, time_limit=t_alloc,
                                 lr=0.002, mom=0.95,
                                 top_k=max(20, n_best // 8),
                                 temp=0.00003)
            try_seq(a_opt, "deep_sg")

        # 5e: MWU on best
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem * 0.8, 60)
            a_opt, _ = mwu_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_mwu")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate between LP-FW and L-BFGS with small perturbations
        for attempt in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break

            if attempt % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.7, 60)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempt}")
            else:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.7, 60)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lbfqg_{attempt}")

        # One final CD pass
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

        # Fine SG
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 20),
                                 lr=0.001, mom=0.95,
                                 top_k=max(10, n_best // 4),
                                 temp=0.00001)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
