```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

corr[k] = sum_n x[n] * y[n-k] for k = -(Ly-1), ..., Lx-1

The length of the full cross-correlation is Lx + Ly - 1.

In our case, w has length 2n-1 and a has length n.
The full cross-correlation has length (2n-1) + n - 1 = 3n - 2.

corr[k] = sum_i w[i] * a[i-k] for k = -(n-1), ..., 2n-2

We want grad[j] = corr[j] for j = 0, ..., n-1.

So we need the cross-correlation at lags 0, 1, ..., n-1.

Using FFT:
- Pad w to length L ≥ 3n-2
- Pad a to length L
- cross = ifft(fft(w) * conj(fft(a)))
- The lags are at indices 0, 1, ..., L-1 (with negative lags wrapped around)
- Specifically, corr[k] = cross[k] for k ≥ 0, and corr[k] = cross[k+L] for k < 0

So grad = cross[0:n]

Let me verify: cross[k] = ifft(fft(w) * conj(fft(a)))[k] = sum_i w[i] * a[(i-k) mod L]

For the unmodded case (when no wraparound occurs):
cross[k] = sum_i w[i] * a[i-k] for i-k in [0, n-1]

This is exactly what we want for k = 0, ..., n-1 (assuming no wraparound, which is guaranteed if L ≥ 3n-2).

So:
```python
def compute_grad_fast(a, w):
    n = len(a)
    L = 1 << (3*n - 2 - 1).bit_length()  # next power of 2 ≥ 3n-2
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)
    return cross[:n]
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` for non-negative sequences. Key insights:

2. The gradient of `max(conv)` can be computed efficiently via FFT (cross-correlation of softmax weights with the sequence)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum a)^2 using:
- FFT-accelerated smooth (log-sum-exp) gradient descent with Nesterov momentum
- Adaptive beta schedule for progressive sharpening of the max approximation
- Multiple structured initializations (ramps, exponentials, parent-based)
- Deflation polish targeting the specific argmax convolution position
- Multi-scale exploration (several sequence lengths)
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w):
    """Compute gradient of smooth max: grad[j] = sum_i w[i]*a[i-j] via FFT cross-correlation."""
    n = len(a)
    if n == 1:
        return np.array([w[0] * a[0]])
    L = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)
    return cross[:n]


def smooth_grad_and_obj(a, beta):
    """Compute smooth objective (2n*LSE) and gradient for a on simplex."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted -= shifted.max()  # extra stability
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    logsum = (max_c + conv.max() - conv.max()) + (np.log(exp_s.sum()) + shifted.max() / beta) / beta
    # Actually let's redo: LSE = max_c + log(sum exp(beta*(conv - max_c)))/beta
    shifted2 = beta * (conv - max_c)
    exp2 = np.exp(shifted2 - shifted2.max())
    logsum = max_c + (shifted2.max() / beta + np.log(exp2.sum()) / beta)
    w2 = exp2 / exp2.sum()
    grad = compute_grad_fft(a, w2)
    obj = 2.0 * n * logsum
    return obj, grad, logsum


def exact_obj(a):
    """Compute exact objective 2n*max(conv)/sum^2 (a assumed on simplex => sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_nesterov(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                    lr_init=0.15, mu=0.85, check_interval=30):
    """Nesterov-accelerated gradient descent on smooth approximation."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_obj = exact_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    max_it = int(time_limit / 0.05)  # estimate max iterations

    while time.monotonic() - t0 < time_limit - 2.0:
        it += 1
        progress = min(1.0, it / max(1, max_it * 0.7))
        beta = beta_start * (beta_end / beta_start) ** progress

        _, grad, _ = smooth_grad_and_obj(a, beta)

        # Nesterov momentum
        v = mu * v + grad
        a_new = project_simplex(a - lr * v)

        # Backtracking: check if objective improved (use exact periodically)
        if it % check_interval == 1:
            obj_new = exact_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.95
                    if no_improve > 30:
                        # Restart with perturbation
                        rng = np.random.default_rng(it * 7 + 3)
                        a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                        v = np.zeros(n)
                        lr = lr_init * 0.5
                        no_improve = 0

        a = a_new

        # Decay learning rate
        lr = lr_init * (0.9995 ** it)
        lr = max(lr, 0.001)

        # Adapt momentum: increase as we converge
        mu = min(0.95, 0.85 + 0.1 * progress)

    # Final check
    obj_final = exact_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    return best_a, best_obj


def deflation_polish(a0, time_limit, n_inner=200):
    """Targeted deflation: repeatedly find argmax conv position and push mass away."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_obj = exact_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    step = 0.05

    for it in range(n_inner):
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Direction that increases conv[k_star]: c[j] = a[k_star - j]
        # We want to DECREASE conv[k_star], so push in direction -c
        c = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            c[j] = a[k_star - j]

        # Try multiple step sizes
        improved = False
        for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1]:
            c_norm = c / c.sum() if c.sum() > 0 else np.ones(n) / n
            a_try = project_simplex(a - step * frac * 10 * c_norm)
            obj_try = exact_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try small random perturbation
            rng = np.random.default_rng(it * 13 + 1)
            a_pert = project_simplex(a + rng.standard_normal(n) * 0.002)
            obj_pert = exact_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                a = a_pert
            else:
                step *= 0.9
                if step < 0.001:
                    break

    return best_a, best_obj


def exponential_mirror_descent(a0, time_limit, eta_init=0.3, eta_decay=0.999):
    """Mirror descent with exponential (multiplicative weights) update."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    best_obj = exact_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    eta = eta_init
    it = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        it += 1
        beta = 20.0 + 50.0 * it / max(1, int(time_limit / 0.02))
        _, grad, _ = smooth_grad_and_obj(a, beta)

        # Exponential update: a_new[j] prop to a[j] * exp(-eta * grad[j])
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()  # stability
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s

        if it % 20 == 0:
            obj_new = exact_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                eta *= 0.95
                a = a_new  # still update, just don't record as best

        eta *= eta_decay

    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse structured initializations on the simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Linear ramp increasing (mass at right)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 10.0]:
        r = (x + 1.0) ** p
        inits.append(r / r.sum())

    # Linear ramp decreasing (mass at left)
    for p in [1.0, 2.0, 5.0]:
        r = (n - x) ** p
        inits.append(r / r.sum())

    # Exponential (mass at right)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        e = np.exp(alpha * x / n)
        inits.append(e / e.sum())

    # Exponential (mass at left)
    for alpha in [1.0, 2.0]:
        e = np.exp(-alpha * x / n)
        inits.append(e / e.sum())

    # Two-scale: small base + spike region at right
    for frac in [0.1, 0.2, 0.3]:
        k = max(1, int(n * frac))
        a = np.ones(n)
        a[-k:] = n  # spike region
        inits.append(a / a.sum())

    # Cosine window (asymmetric)
    c = np.cos(np.pi * (x / (n - 1)) ** 2) ** 2 + 0.01
    inits.append(c / c.sum())
    c2 = np.cos(np.pi * (1 - x / (n - 1)) ** 2) ** 2 + 0.01
    inits.append(c2 / c2.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Parent-based
    if parent is not None:
        p = parent.copy()
        if len(p) != n:
            p = np.interp(x, np.linspace(0, 1, len(p)), p)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                p = p / ps
        else:
            p = p / p.sum()
        inits.append(p)
        # Blend parent with ramp
        ramp = (x + 1.0) ** 2
        ramp = ramp / ramp.sum()
        for t in [0.3, 0.5, 0.7]:
            blended = (1 - t) * p + t * ramp
            blended = np.maximum(blended, 0)
            inits.append(blended / blended.sum())

    return inits
```
