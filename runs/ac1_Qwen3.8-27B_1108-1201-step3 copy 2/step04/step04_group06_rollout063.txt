Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) Diverse initialization with Fourier-based constructions
(2) Exponentiated gradient (multiplicative weights) with adaptive parameters
(3) Efficient coordinate descent with O(n) convolution updates
(4) Cutting-plane QP refinement via CVXPY
(5) Final fine-grained coordinate descent polish

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the score directly (scale-invariant)."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)

def conv_update(a, conv, i, new_val):
    """Update conv after changing a[i] to new_val. Returns new conv."""
    n = len(a)
    old_val = a[i]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return conv
    new_conv = conv.copy()
    # For k where |k-i| < n (i.e., k-i in [-(n-1), n-1] and k in [0, 2n-2])
    # The affected k values are k = i+j for j in [0, n-1], i.e., k in [i, i+n-1]
    k_start = max(0, i)  # minimum k where k-i >= 0
    k_end = min(2*n-2, i + n - 1)  # maximum k where k-i < n
    for k in range(k_start, k_end + 1):
        j = k - i
        if k == 2*i:
            new_conv[k] += new_val**2 - old_val**2
        else:
            new_conv[k] += 2.0 * delta * a[j]
    return new_conv

                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.5f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay

        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 31 + 7)
            pert = 0.001 + 0.002 * min(1.0, no_improve / 5000.0)
            a = best_a + rng_local.standard_normal(n) * pert
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.1
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0

    return best_a.copy()


def coord_descent(a0, time_limit, mode="coarse", verbose=False):
    """
    Coordinate descent with O(n) conv updates.
    mode: 'coarse' for large multiplicative steps, 'fine' for small steps,
          'ultra' for very small steps (final polish).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    score = 2.0 * n * conv.max() / (s * s)
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    rng = np.random.default_rng(777)

    if mode == "coarse":
        factors = [0.0, 0.3, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.111, 1.25, 1.43, 1.67, 2.0, 3.0]
    elif mode == "fine":
        factors = [0.7, 0.8, 0.9, 0.95, 0.97, 1.0, 1.03, 1.05, 1.1, 1.25, 0.0]
    else:  # ultra
        factors = [0.97, 0.985, 0.99, 0.995, 1.0, 1.005, 1.01, 1.015, 1.03]

    no_improve_sweeps = 0
    max_no_improve = 80 if mode == "coarse" else 40 if mode == "fine" else 25

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False

        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-12:
                candidates = [0.001, 0.003, 0.01, 0.03]
            else:
                candidates = [old_val * f for f in factors]

            best_cand = old_val
            best_cand_score = score

            for cand in candidates:
                if cand < 0 or cand > 5:
                    continue
                delta = cand - old_val
                if abs(delta) < 1e-15:
                    continue
                new_s = s + delta
                if new_s < 1e-10:
                    continue
                dconv = compute_delta_conv(delta, i, n, a)
                new_max = (conv + dconv).max()
                new_score = 2.0 * n * new_max / (new_s * new_s)
                if new_score < best_cand_score - 1e-14:
                    best_cand_score = new_score
                    best_cand = cand

            if best_cand != old_val:
                delta = best_cand - old_val
                conv = conv + compute_delta_conv(delta, i, n, a)
                a[i] = best_cand
                s += delta
                score = best_cand_score
                sweep_improved = True

        # Periodically recompute conv from scratch to avoid drift
        if sweep % 50 == 0:
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max() / (s * s)

        if sweep_improved:
            no_improve_sweeps = 0
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                if verbose and sweep % 100 == 0:
                    print(f"    CD({mode}) sweep {sweep}: score={best_score:.10f}")
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > max_no_improve:
                break
            # Add small perturbation to escape plateau
            if no_improve_sweeps == max_no_improve // 2:
                idx = rng.integers(n)
                a[idx] = max(0, a[idx] + rng.standard_normal() * 0.001)
                s = a.sum()
                conv = fast_convolve(a)
                score = 2.0 * n * conv.max() / (s * s)

    return best_a.copy()


def paired_coord_descent(a0, time_limit, verbose=False):
    """Try changing pairs of coordinates jointly to escape single-coordinate plateaus."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    score = 2.0 * n * conv.max() / (s * s)
    best_score = score
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    it = 0
    factors = [0.7, 0.85, 1.0, 1.18, 1.4]

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick two random coordinates
        i, j = rng.choice(n, size=2, replace=False)

        for fi in factors:
            for fj in factors:
                delta_i = a[i] * fi - a[i]
                delta_j = a[j] * fj - a[j]
                new_s = s + delta_i + delta_j
                if new_s < 1e-10:
                    continue
                # Compute combined delta
                dconv = compute_delta_conv(delta_i, i, n, a)
                # Now update a temporarily for the j computation
                a[j] = a[j] + delta_j  # temporary
                dconv += compute_delta_conv(delta_j, j, n, a)
                a[j] = a[j] - delta_j  # restore

                new_max = (conv + dconv).max()
                new_score = 2.0 * n * new_max / (new_s * new_s)
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    a[i] += delta_i
                    a[j] += delta_j
                    s = new_s
                    conv = conv + dconv
                    best_a = a.copy()
                    break
        if time.monotonic() - t0 > time_limit:
            break

    return best_a.copy()


def generate_fourier_init(n, rng, n_harmonics=None):
    """Generate a non-negative sequence from Fourier coefficients with random phases."""
    if n_harmonics is None:
        n_harmonics = max(5, n // 4)
    freqs = np.arange(n_harmonics)
    mags = rng.dirichlet(np.ones(n_harmonics) * 2)
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    x = np.arange(n)
    t = 2 * np.pi * x / n
    sig = np.zeros(n)
    for f, m, p in zip(freqs, mags, phases):
        sig += m * np.cos(f * t + p)
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.ones(n) / n
    else:
        sig = sig / ss
    return sig
