"""

1. Smoothed log-sum-exp L-BFGS (continuation method) - different from LP-based FW

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Compute softmax(z) in a numerically stable way."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s


def smooth_obj_grad(z, T):
    """Compute smoothed objective F_T(softmax(z)) and gradient w.r.t. z."""
    n = len(z)
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = T * (conv - max_c)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_c = np.exp(scaled)
    log_sum = np.log(exp_c.sum())
    obj = max_c + log_sum / T

    w = exp_c / exp_c.sum()

    # dF/da[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    dF_da = np.zeros(n)
    for j in range(n):
        dF_da += 2.0 * a[j] * w[j:j + n]

    # dF/dz[i] = a[i] * (dF_da[i] - sum_j a[j]*dF_da[j])
    mean_grad = np.dot(a, dF_da)
    dF_dz = a * (dF_da - mean_grad)

    return obj, dF_dz


def smooth_minimize(f0, time_limit, T_start=1.0, T_end=5000.0, T_growth=1.4,
                    verbose=False):
    """Continuation method: minimize max(conv) using smoothed log-sum-exp with L-BFGS."""
    n = len(f0)
    # Initialize z from f0
    a_init = np.maximum(f0, 1e-15)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)  # z[n-1] will be the "free" direction

    conv0 = fast_convolve(a_init)
    best_obj = conv0.max() * n * 2  # actual score
    best_f = a_init.copy()

    t0 = time.monotonic()
    T = T_start

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Time allocation for this T level
        t_alloc = min(remaining * 0.6, 60.0)
        if T > T_end * 0.5:
            t_alloc = min(remaining * 0.4, 30.0)

        def func_grad(z_tmp):
            obj, grad = smooth_obj_grad(z_tmp, T)
            return obj, grad

        # Run L-BFGS-B
        res = minimize(func_grad, z, method='L-BFGS-B',
                       options={'maxiter': 200, 'gtol': 1e-8,
                                'maxfun': 500, 'maxls': 30})

        if res.success or res.fun < func_grad(z)[0]:
            z = res.x
            a = softmax_from_z(z)
            conv = fast_convolve(a)
            cur_obj = conv.max()
            if cur_obj < best_obj / (2.0 * n) - 1e-12:
                best_obj = cur_obj * 2.0 * n
                best_f = a.copy()
                if verbose:
                    print(f"    Smooth L-BFGS: T={T:.1f}, score={best_obj:.10f}")

        # Check if close to convergence at this T
        if T > 50:
            a_check = softmax_from_z(z)
            conv_check = fast_convolve(a_check)
            if abs(conv_check.max() - (best_f and fast_convolve(best_f).max())) < 1e-10:
                T *= T_growth
                continue

        T *= T_growth
        if T > T_end:
            T = T_end

    # Final: use the actual (unsmoothed) objective
    a_final = project_simplex(best_f)
    conv_final = fast_convolve(a_final)
    final_obj = conv_final.max() * 2.0 * n
    if final_obj < best_obj:
        best_obj = final_obj
        best_f = a_final

    return best_f, best_obj


def coord2_descent(a, time_limit, rng, n_sweeps_max=200):

    n = len(a)
    f = np.maximum(a, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # Random permutation of coordinate pairs
        pairs = []
        if n >= 2:
            # Generate random pairs, ensuring coverage
            idx = np.arange(n)
            for _ in range(n):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j:
                    pairs.append((i, j))
            # Add sequential pairs for structure
            for i in range(n - 1):
                pairs.append((i, i + 1))

        improved = False
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if i == j:
                continue

            c = f[i] + f[j]
            if c < 1e-15:
                continue

            # Compute base (convolution with a[i]=a[j]=0)
            f_zero = f.copy()
            f_zero[i] = 0.0
            f_zero[j] = 0.0
            base = fast_convolve(f_zero)

            # For each k, (a*a)[k] = base[k] + terms involving x and c-x
            # x = f[i], c-x = f[j]
            # Terms:
            #   l=i: x * a[k-i] (if 0<=k-i<n)
            #   l=j: (c-x) * a[k-j] (if 0<=k-j<n)
            # Special cases:
            #   k=2i: x*x adds to the l=i term
            #   k=2j: (c-x)^2 adds to the l=j term
            #   k=i+j: both l=i and l=j contribute x*(c-x) each

            # Build the quadratic coefficients
            # (a*a)[k] = A_k*x^2 + B_k*x + C_k where C_k = base[k]
            A_k = np.zeros(2 * n - 1)
            B_k = np.zeros(2 * n - 1)

            # l=i contribution: x * a[k-i] (for 0<=k-i<n)
            for k in range(2 * n - 1):
                j_idx = k - i
                if 0 <= j_idx < n:
                    if j_idx == i:
                        A_k[k] += 1.0  # x^2
                    elif j_idx == j:
                        A_k[k] += -1.0  # x*(c-x) = cx - x^2
                        B_k[k] += c
                    else:
                        B_k[k] += a_fixed = f[j_idx]

            # l=j contribution: (c-x) * a[k-j] (for 0<=k-j<n)
            for k in range(2 * n - 1):
                j_idx = k - j
                if 0 <= j_idx < n:
                    if j_idx == j:
                        A_k[k] += 1.0  # (c-x)^2 = c^2 - 2cx + x^2
                        B_k[k] += -2.0 * c
                    elif j_idx == i:
                        A_k[k] += -1.0  # (c-x)*x = cx - x^2
                        B_k[k] += c
                    else:
                        B_k[k] += -(f[j_idx])

            # Wait, this is getting complicated. Let me just evaluate directly.
            # For a given x, f[i]=x, f[j]=c-x, compute max of conv.
            # This is O(n) per evaluation.

            # Grid search + refinement
            x_lo = 0.0
            x_hi = c
            # Coarse grid
            x_grid = np.linspace(x_lo, x_hi, 30)
            best_x_val = f[i]
            best_val = np.inf
            for x_val in x_grid:
                f_test = f.copy()
                f_test[i] = x_val
                f_test[j] = c - x_val
                conv_test = fast_convolve(f_test)
                val = conv_test.max()
                if val < best_val:
                    best_val = val
                    best_x_val = x_val

            # Refine around best
            x_ref_lo = max(0.0, best_x_val - (x_hi - x_lo) / 30.0)
            x_ref_hi = min(c, best_x_val + (x_hi - x_lo) / 30.0)
            x_grid2 = np.linspace(x_ref_lo, x_ref_hi, 20)
            for x_val in x_grid2:
                f_test = f.copy()
                f_test[i] = x_val
                f_test[j] = c - x_val
                conv_test = fast_convolve(f_test)
                val = conv_test.max()
                if val < best_val:
                    best_val = val
                    best_x_val = x_val

            if best_val < f_conv_max(f) - 1e-12:
                f[i] = best_x_val
                f[j] = c - best_x_val
                improved = True

        if not improved:
            # Try with different random pairs
            if sweep > 5:
                break

    conv_final = fast_convolve(f)
    if conv_final.max() < best_obj:
        best_obj = conv_final.max()
        best_f = f.copy()

def peak_shaving(a, time_limit, rng, n_iter_max=50000):
    """Targeted peak shaving: identify peak lag, reduce contributing coordinates."""
    n = len(a)
    f = np.maximum(a, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    decay = 0.001
    rng_local = np.random.default_rng(rng.integers(1000000) if hasattr(rng, 'integers') else 99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > n_iter_max:
            break

        conv = fast_convolve(f)
        peak_val = conv.max()
        # Get all near-peak lags
        threshold = peak_val * 0.995
        peak_lags = np.where(conv >= threshold)[0]

        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(conv)])

        # Compute sensitivity: for each coordinate i, how much does it contribute to peaks?
        sens = np.zeros(n)
        for k in peak_lags:
            for j in range(n):
                ki = k - j
                if 0 <= ki < n and ki != j:
                    sens[j] += f[ki]
                elif ki == j:
                    sens[j] += f[j]  # quadratic contribution

        # Reduce high-sensitivity coordinates, increase low-sensitivity ones
        sens_mean = sens.mean()
        direction = sens - sens_mean
        # Add small random noise for exploration
        direction += rng_local.standard_normal(n) * 0.001

        # Project: f_new = f - decay * direction, then project to simplex
        f_new = f - decay * direction
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        if conv_new.max() < best_obj - 1e-14:
            best_obj = conv_new.max()
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            decay = min(decay * 1.01, 0.01)
        elif conv_new.max() < peak_val + 1e-6:
            f = f_new
            no_improve += 1
        else:
            no_improve += 1

        # Adaptive: reduce decay when stuck
        if no_improve > 50:
            decay *= 0.9
            if decay < 1e-6:
                # Large restart
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.005)
                decay = 0.001
                no_improve = 0

    return best_f, best_obj


def subgrad_optimize(f0, time_limit, lr=0.002, mom=0.92, top_k=30, temp=0.00005):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if no_improve > 400:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_f, best_obj


def local_search(f0, time_limit, rng, scale_init=0.003):
    """Random perturbation local search with adaptive scale."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    scale = scale_init

    while time.monotonic() - t0 < time_limit:
        batch = 200
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            f_pert = project_simplex(best_f + noise)
            obj = fast_convolve(f_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_pert.copy()

        scale *= 0.93
        if scale < 1e-7:
            scale = scale_init * 0.3
            f_jump = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            obj = fast_convolve(f_jump).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_jump.copy()

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    xc = 0.5 * (n - 1)

    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - xc) / sigma) ** 2)
        inits.append(g / g.sum())

    # Multi-bump
    for nb in [3, 5, 7]:
        bumps = np.zeros(n)
        positions = np.linspace(0.1 * n, 0.9 * n, nb)
        for p in positions:
            bumps += np.exp(-0.5 * ((x - p) / (0.05 * n)) ** 2)
        inits.append(bumps / bumps.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    # U-shape
    edge_w = 1.0 / (0.5 * np.abs(x - xc) / n + 0.05)
    inits.append(edge_w / edge_w.sum())

    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(rc / rc.sum())

    # Flat-top window
    ft = np.ones(n) * 0.9
        ft[:n//10] = 0.1
        ft[-n//10:] = 0.1
    inits.append(ft / ft.sum())

    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def lp_maxsum_step(f, M):
    """Solve max-sum LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Return g/sum(g)."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x / res.x.sum()
    return None


def linprog_lp_step(f, conv_f):
    """Solve linearized LP: min t s.t. t >= 2*(f*g)[k] + conv_f[k], sum(g)=0, g>=-f."""
    n = len(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    for j in range(n):
        A_ub[j:j + n, 1 + j] = 2.0 * f  # Wait, this isn't right
    # Actually (f*g)[k] = sum_j f[k-j]*g[j]... let me reconsider
    # (f*g)[k] = sum_j g[j]*f[k-j] where k-j in [0,n-1]
    # So A_ub[k, 1+j] = f[k-j] if 0 <= k-j < n
    A_ub[:, 1:] = 0.0
    for j in range(n):
        for k in range(max(j, 0), min(2*n-1, j+n)):
            A_ub[k, 1+j] = f[k - j]
    A_ub[:, 1:] *= 2.0
    b_ub = -conv_f

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def lp_fw_optimize(f0, time_limit, method='maxsum', verbose=False):
    """LP-based Frank-Wolfe for larger moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if method == 'maxsum':
            g = lp_maxsum_step(f, M * 1.05)
        else:
            g_dir = linprog_lp_step(f, conv)
            if g_dir is None:
                no_improve += 5
                if no_improve > 80:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    no_improve = 0
                continue
            g = f + g_dir
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-10:
                no_improve += 1
                continue
            g = g / gs

        if g is None:
            no_improve += 1
            if no_improve > 60:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-10:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj * 2 * n:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.003 + 0.0001 * no_improve))
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search:
    Phase 1: Diverse initialization
    Phase 2: Smoothed L-BFGS continuation (primary - different from LP)
    Phase 3: 2-coordinate descent polish
    Phase 4: Peak shaving + subgradient
    Phase 5: LP-based FW for larger moves
    Phase 6: Local search + different lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
    except Exception:
        pass

    # === Phase 1: Initialization (30s) ===
    print("=== Phase 1: Initialization ===")
    t1 = time.monotonic()

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent))
        n_p = len(parent)
    else:
        n_p = 384
        candidates.append(("uniform", np.ones(n_p) / n_p))

    # Add diverse inits at parent length
    other_inits = generate_inits(n_p, parent, rng)
    for i, ini in enumerate(other_inits[:8]):
        candidates.append((f"init_{i}", ini))

    # Different lengths
    for n_try in [256, 320, 448, 512, 640]:
        if n_try == n_p:
            continue
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_p), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                candidates.append((f"len{n_try}", px / ps))
        else:
            candidates.append((f"len{n_try}", np.ones(n_try) / n_try))

    for label, cand in candidates:
        if time.monotonic() + 5 > deadline:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq)})")

    # === Phase 2: Smoothed L-BFGS (primary optimizer, 700s) ===
    print("\n=== Phase 2: Smoothed L-BFGS Continuation ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc_p2 = min(remaining * 0.4, 700)

    if t_alloc_p2 > 30:
        print(f"  Smoothed L-BFGS on n={n_best}, budget={t_alloc_p2:.0f}s")
        # Run on best
        a_opt, _ = smooth_minimize(best_seq, time_limit=t_alloc_p2 * 0.5,
                                   T_start=1.0, T_end=5000.0, T_growth=1.5,
                                   verbose=True)
        try_seq(a_opt, "smooth_lbfgs")

        # Run on a perturbed version
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            a_opt, _ = smooth_minimize(f_pert, time_limit=min(rem * 0.5, 200),
                                       T_start=5.0, T_end=3000.0, T_growth=1.5)
            try_seq(a_opt, "smooth_pert")

    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: 2-Coordinate Descent (200s) ===
    print("\n=== Phase 3: 2-Coordinate Descent ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc_p3 = min(remaining * 0.15, 200)

    if t_alloc_p3 > 20:
        print(f"  2-coord descent, budget={t_alloc_p3:.0f}s")
        a_opt, _ = coord2_descent_v2(best_seq, time_limit=t_alloc_p3, rng=rng)
        try_seq(a_opt, "coord2")

        # Run again on result
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            a_opt, _ = coord2_descent_v2(best_seq, time_limit=min(rem * 0.5, 100), rng=rng)
            try_seq(a_opt, "coord2_2")

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Peak Shaving + Subgradient (300s) ===
    print("\n=== Phase 4: Peak Shaving + Subgradient ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc_p4 = min(remaining * 0.2, 300)

    if t_alloc_p4 > 20:
        # Peak shaving
        t_ps = min(t_alloc_p4 * 0.4, 150)
        a_opt, _ = peak_shaving(best_seq, time_limit=t_ps, rng=rng)
        try_seq(a_opt, "peak_shave")

        # Subgradient
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_sg = min(rem * 0.5, 150)
            a_opt, _ = subgrad_optimize(best_seq, time_limit=t_sg,
                                        lr=0.002, mom=0.93, top_k=n_best // 5, temp=0.00003)
            try_seq(a_opt, "subgrad")

    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: LP-based Frank-Wolfe (200s) ===
    print("\n=== Phase 5: LP-based Frank-Wolfe ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc_p5 = min(remaining * 0.15, 200)

    if t_alloc_p5 > 30:
        print(f"  LP-FW (maxsum), budget={t_alloc_p5:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc_p5 * 0.6,
                                  method='maxsum', verbose=True)
        try_seq(a_opt, "lpfw_maxsum")

        rem = deadline - time.monotonic() - 50
        if rem > 30:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.5, 100),
                                      method='linear')
            try_seq(a_opt, "lpfw_lin")

    print(f"  Phase 5 best: {best_score:.10f}")

    # === Phase 6: Local search + different lengths (remaining time) ===
    print("\n=== Phase 6: Local Search + Lengths ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    t_alloc_p6 = min(remaining * 0.6, 250)

    if t_alloc_p6 > 15:
        # Local search
        t_ls = min(t_alloc_p6 * 0.3, 80)
        a_opt, _ = local_search(best_seq, time_limit=t_ls, rng=rng, scale_init=0.002)
        try_seq(a_opt, "local")

        # Try nearby lengths
        rem = deadline - time.monotonic() - 30
        for n_try in [n_best - 16, n_best + 16, n_best - 32, n_best + 32]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                # Quick smooth optimization
                t_each = min(rem * 0.25, 30)
                a_opt, _ = smooth_minimize(px, time_limit=t_each,
                                           T_start=10.0, T_end=2000.0, T_growth=2.0)
                try_seq(a_opt, f"len{n_try}")
                rem = deadline - time.monotonic() - 20

        # Final local search
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = local_search(best_seq, time_limit=min(rem * 0.5, 30),
                                    rng=rng, scale_init=0.001)
            try_seq(a_opt, "final_local")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
