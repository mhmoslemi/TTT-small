"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved exponentiated gradient
method with momentum, adaptive step size, exact subgradients, periodic spectral
flattening, and multi-length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def exact_subgradient(a, k_star):
    """Compute exact subgradient of max(conv(a,a)) at position k_star."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for j in range(js, je + 1):
        sg[j] = 2.0 * a[k_star - j]
    return sg


def softmax_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient."""
    n = len(a)
    len_c = len(c)
    
    if top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je+1] += 2.0 * w[idx] * a[k-je:k-js+1][::-1]
    
    return sg


def spectral_flatten(a, alpha=0.7, epsilon=1e-6):
    """Apply spectral flattening to reduce peak."""
    n = len(a)
    fft_size = 1 << (2*n-1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = np.abs(A) ** 2
    # Compress dynamic range
    power_flat = power ** alpha
    # Scale to preserve total energy (Parseval)
    scale = np.sqrt(np.sum(power) / max(np.sum(power_flat), 1e-300))
    A_new = A / np.maximum(np.abs(A), 1e-300) * np.sqrt(power_flat * scale**2)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def local_search(a0, time_limit, max_rounds=500):

        it += 1
        improved = False
        
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5,

                    a = best_a.copy()
                    break

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_a.copy(), best_obj


def interpolate_sequence(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new
