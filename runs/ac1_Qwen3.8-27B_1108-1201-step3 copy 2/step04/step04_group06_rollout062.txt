= sum_j a[n-1-j] * p[i+n-1-j]

Let m = n-1-j, so j = n-1-m:
= sum_m a[m] * p[i+m] ✓

Great, so:
(H a)_i = conv(a[::-1], p)[i + n - 1]

```python
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def dual_eg(a0, time_limit, eta=1.0, inner_iters=30, inner_lr=None):
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize p uniformly
    p = np.ones(2*n - 1) / (2*n - 1)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    if inner_lr is None:
        inner_lr = 1.0 / (2 * np.dot(p, np.arange(2*n-1)))  # crude Lipschitz estimate
    
    while time.monotonic() - t0 < time_limit:
        # Inner: solve QP min a^T H(p) a s.t. a >= 0, sum(a) = 1
        a_qp = a.copy()
        for _ in range(inner_iters):
            grad = 2.0 * hankel_matvec(p, a_qp)
            a_qp = simplex_project(a_qp - inner_lr * grad)
        
        # Evaluate
        conv = fast_convolve(a_qp)
        obj = conv.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_qp.copy()
        
        # Outer: update p
        # Multiplicative weights: p_k *= exp(eta * conv_k)
        scaled = eta * conv
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        a = a_qp
    
    return best_a, best_obj
```

The problem min max_k conv(a,a)(k) s.t. sum(a) = 1, a >= 0 can be viewed as:
minimize lambda such that a^T H_k a <= lambda for all k, sum(a) = 1, a >= 0.

By the minimax theorem, this equals:
maximize min_{a: sum(a)=1, a>=0} sum_k p_k a^T H_k a = max_p min_a a^T H(p) a

The inner min is the Rayleigh quotient of H(p) restricted to the simplex. If the unconstrained minimizer (over all of R^n with sum constraint) is in the simplex, then:
min_a a^T H a s.t. sum(a) = 1 = 1^T H^{-1} 1 ... no wait.

Actually, min a^T H a s.t. 1^T a = 1 (ignoring non-negativity) has solution:
a* = H^{-1} 1 / (1^T H^{-1} 1)

And the minimum value is 1 / (1^T H^{-1} 1).

If a* >= 0, then the constrained minimum equals the unconstrained minimum.

So the dual becomes: max_p 1 / (1^T H(p)^{-1} 1) = min_p (1^T H(p)^{-1} 1)

This is a fractional programming problem in p. Not sure if there's a closed form.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Dual (game-theoretic) exponentiated gradient: maintains a distribution over
   lags, solves inner QP via projected gradient, updates lag distribution.
2. Improved primal EG with adaptive temperature and Nesterov-style averaging.
3. Parametric initializations (Gaussian, cosine, edge-heavy, Dirichlet).
4. Systematic multi-length exploration.
5. Local multiplicative refinement.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_matvec(p, a):
    """Compute H @ a where H[i,j] = p[i+j]. p has length 2n-1, a has length n."""
    n = len(a)
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size)
    return conv[n - 1:2 * n - 1]


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    if top_k is not None and 0 < top_k < len_c:
        indices = np.argpartition(c, -top_k)[-top_k:]
    else:
        indices = np.arange(len_c)
    peek = c[indices]
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    return sg


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.005, temp_decay=0.9997,
                eta_decay=0.99995, top_k_frac=0.25, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay

        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = max(eta0 * 0.1, 0.05)
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def dual_eg_optimize(a0, time_limit, eta_outer=0.5, inner_iters=40,
                     inner_lr=None, verbose=False):
    """
    Dual (game-theoretic) exponentiated gradient.
    Maintains distribution p over lags; inner loop solves QP min a^T H(p) a
    s.t. a in simplex; outer loop updates p via multiplicative weights.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Initialize p (distribution over lags 0..2n-2)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_obj = float('inf')
    best_a = a.copy()

    t0 = time.monotonic()
    if inner_lr is None:
        inner_lr = 0.5
    lr = inner_lr
    no_improve = 0
    it = 0

    avg_a = a.copy()
    count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: solve QP min a^T H(p) a s.t. a in simplex
        a_qp = a.copy()
        for s_iter in range(inner_iters):
            grad = 2.0 * hankel_matvec(p, a_qp)
            a_qp = simplex_project(a_qp - lr * grad)

        c = fast_convolve(a_qp)
        obj = c.max()

        count += 1
        avg_a = (count * avg_a + a_qp) / (count + 1)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_qp.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    DualEG iter {it}: obj={obj:.10f}, eta={eta_outer:.4f}")
        else:
            no_improve += 1

        # Outer: update p via multiplicative weights
        scaled = eta_outer * c
        scaled -= scaled.max()
        p_new = np.exp(scaled)
        ps = p_new.sum()
        if ps < 1e-300:
            p_new = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p_new /= ps
        p = p_new

        a = a_qp

        # Periodically check average
        if it % 200 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        # Adapt outer eta
        if no_improve > 500:
            eta_outer *= 1.2
            lr = max(lr * 0.9, 0.01)
            no_improve = 0
            # Perturb
            rng_local = np.random.default_rng(it * 17 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps2 = a.sum()
            if ps2 > 1e-10:
                a = a / ps2
            else:
                a = best_a.copy()
            avg_a = a.copy()
            count = 0

        # Decay lr slowly
        lr = max(lr * 0.9999, 0.005)

    return best_a.copy(), best_obj


def generate_parametric_inits(n, rng):
    """Generate diverse parametric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (various widths and centers)
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
        for center in [0.3, 0.5, 0.7]:
            g = np.exp(-0.5 * ((xn - center) / sigma) ** 2)
            inits.append(g / g.sum())

    # Cosine families
    for p_pow in [1, 2, 4, 8]:
        c1 = np.abs(np.cos(np.pi * xn)) ** p_pow + 0.01
        inits.append(c1 / c1.sum())

    # Edge-heavy
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        edge = 1.0 / (1.0 + alpha * np.minimum(xn, 1.0 - xn))
        inits.append(edge / edge.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(xn - 0.5)
    inits.append(np.maximum(tri, 0.01) / np.maximum(tri, 0.01).sum())

    # Sine
    s1 = np.sin(np.pi * xn) ** 2 + 0.01
    inits.append(s1 / s1.sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Smooth random
    for _ in range(3):
        r = rng.standard_normal(n)
        # Smooth via moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0)
        rs = r.sum()
        if rs > 1e-10:
            inits.append(r / rs)

    return inits


def local_polish(a0, time_limit, max_rounds=200):
    """Local search via multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    it = 0
    no_improve = 0

    factors = [0.5, 0.7, 0.85, 1.0 / 0.85, 1.0 / 0.7, 1.0 / 0.5,
               0.75, 1.0 / 0.75, 0.9, 1.0 / 0.9, 0.95, 1.0 / 0.95]

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        n_perturb = max(3, n // 12)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue

            for factor in factors:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a.copy(), best_obj


def try_length(a_best, n_new, rng):
    """Interpolate or transform best sequence to new length."""
    n_old = len(a_best)
    if n_new == n_old:
        return a_best.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_best)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new

        if a.sum() < 1e-10:

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(256) / 256
        print("No parent found, using uniform n=256")
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best_score={best_score:.8f}")

    # === Phase 1: Multi-length exploration with parametric inits (300s) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t1_deadline = time.monotonic() + 300

    candidate_lengths = sorted(set([
        128, 192, 256, 320, 384, 448, 512, 640, 768
    ]))

    # Filter to reasonable lengths
    candidate_lengths = [n for n in candidate_lengths if 64 <= n <= 1024]

    results_by_length = {}
    for n_try in candidate_lengths:
        if time.monotonic() > t1_deadline:
            break
        rem = t1_deadline - time.monotonic()
        if rem < 30:
            break
        time_per = max(5, min(25, rem / max(1, len(candidate_lengths) - candidate_lengths.index(n_try))))

        # Generate a few inits for this length
        inits = generate_parametric_inits(n_try, rng)[:6]
        # Also add interpolated parent
        if parent is not None:
            inits.append(try_length(parent, n_try, rng))

        best_for_len = None
        best_obj_for_len = float('inf')

        for init in inits:
            if time.monotonic() > t1_deadline:
                break
            t_alloc = max(3, time_per / max(1, len(inits)))
            a_opt, obj = eg_optimize(init, time_limit=t_alloc, eta0=1.0,
                                     temp0=0.01, temp_decay=0.9997,
                                     eta_decay=0.99995, top_k_frac=0.3)
            if obj < best_obj_for_len:
                best_obj_for_len = obj
                best_for_len = a_opt

        if best_for_len is not None:
            results_by_length[n_try] = best_for_len
            score = try_seq(best_for_len, f"p1_n{n_try}_obj{best_obj_for_len:.6f}")
            print(f"  n={n_try}: obj={best_obj_for_len:.8f}, score={score:.8f}")

    # Update best from phase 1
    if best_seq is not None:
        best_len = len(best_seq)
    else:
        best_len = n_parent

    # === Phase 2: Dual EG on best candidate (400s) ===
    print(f"\n=== Phase 2: Dual EG (n={best_len}) ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        t2_deadline = time.monotonic() + 400
        # Run dual EG with different settings
        dual_configs = [
            {"eta_outer": 0.3, "inner_iters": 30, "inner_lr": 0.5},
            {"eta_outer": 0.8, "inner_iters": 50, "inner_lr": 0.3},
            {"eta_outer": 0.5, "inner_iters": 60, "inner_lr": 0.4},
            {"eta_outer": 1.0, "inner_iters": 40, "inner_lr": 0.2},
        ]
        for ci, cfg in enumerate(dual_configs):
            if time.monotonic() > t2_deadline:
                break
            rem = t2_deadline - time.monotonic()
            t_alloc = min(100, max(20, rem / max(1, len(dual_configs) - ci)))
            print(f"  Dual EG config {ci}: eta={cfg['eta_outer']}, alloc={t_alloc:.0f}s")
            a_opt, obj = dual_eg_optimize(best_seq, time_limit=t_alloc, verbose=True, **cfg)
            try_seq(a_opt, f"dual_ci{ci}_obj{obj:.6f}")

    # === Phase 3: Primal EG with tuned parameters (400s) ===
    print(f"\n=== Phase 3: Primal EG Tuned ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        t3_deadline = time.monotonic() + 400
        eg_configs = [
            {"eta0": 0.8, "temp0": 0.003, "temp_decay": 0.9996, "eta_decay": 0.99995, "top_k_frac": 0.4},
            {"eta0": 2.0, "temp0": 0.001, "temp_decay": 0.9994, "eta_decay": 0.9999, "top_k_frac": 0.2},
            {"eta0": 1.5, "temp0": 0.01, "temp_decay": 0.9999, "eta_decay": 0.99999, "top_k_frac": 0.5},
            {"eta0": 0.5, "temp0": 0.005, "temp_decay": 0.9998, "eta_decay": 0.99995, "top_k_frac": 0.6},
            {"eta0": 3.0, "temp0": 0.002, "temp_decay": 0.9995, "eta_decay": 0.9999, "top_k_frac": 0.3},
        ]
        for ci, cfg in enumerate(eg_configs):
            if time.monotonic() > t3_deadline:
                break
            rem = t3_deadline - time.monotonic()
            t_alloc = min(80, max(15, rem / max(1, len(eg_configs) - ci)))
            print(f"  EG config {ci}: alloc={t_alloc:.0f}s")
            a_opt, obj = eg_optimize(best_seq, time_limit=t_alloc, verbose=False, **cfg)
            try_seq(a_opt, f"eg3_ci{ci}_obj{obj:.6f}")

    # === Phase 4: Dual EG on different lengths (250s) ===
    print(f"\n=== Phase 4: Dual EG Multi-length ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        t4_deadline = time.monotonic() + 250
        # Try neighboring lengths
        n_best = len(best_seq)
        nearby_lengths = [n_best // 2, int(n_best * 0.75), n_best,
                          int(n_best * 1.25), int(n_best * 1.5), n_best * 2]
        nearby_lengths = [n for n in nearby_lengths if 64 <= n <= 1024 and n != n_best]
        nearby_lengths = sorted(set(nearby_lengths))

        for n_alt in nearby_lengths:
            if time.monotonic() > t4_deadline:
                break
            rem = t4_deadline - time.monotonic()
            if rem < 30:
                break
            a_alt = try_length(best_seq, n_alt, rng)
            t_alloc = min(50, max(15, rem / max(1, len(nearby_lengths))))
            print(f"  Dual EG n={n_alt}, alloc={t_alloc:.0f}s")
            a_opt, obj = dual_eg_optimize(a_alt, time_limit=t_alloc,
                                          eta_outer=0.5, inner_iters=40, inner_lr=0.4,
                                          verbose=False)
            try_seq(a_opt, f"dual4_n{n_alt}_obj{obj:.6f}")

    # === Phase 5: Extended EG on best (200s) ===
    print(f"\n=== Phase 5: Extended EG on Best ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        t5_deadline = time.monotonic() + 200
        a_opt, obj = eg_optimize(best_seq, time_limit=min(120, t5_deadline - time.monotonic()),
                                 eta0=0.3, temp0=0.001, temp_decay=0.99995,
                                 eta_decay=0.99999, top_k_frac=0.5)
        try_seq(a_opt, "eg5_ext")

        # Dual EG continued
        rem = t5_deadline - time.monotonic()
        if rem > 30:
            a_opt, obj = dual_eg_optimize(best_seq, time_limit=min(70, rem),
                                          eta_outer=0.3, inner_iters=50, inner_lr=0.3,
                                          verbose=False)
            try_seq(a_opt, "dual5_ext")

    # === Phase 6: Local polish (100s) ===
    print(f"\n=== Phase 6: Local Polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = local_polish(best_seq, time_limit=min(rem, 80), max_rounds=150)
            try_seq(a_opt, "local_polish")

    # === Phase 7: Final refinement (remaining time) ===
    print(f"\n=== Phase 7: Final Refinement ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # Very fine EG
            t_fine = min(40, rem * 0.5)
            a_opt, _ = eg_optimize(best_seq, time_limit=t_fine,
                                   eta0=0.1, temp0=0.0005,
                                   temp_decay=0.99999, eta_decay=0.999999,
                                   top_k_frac=0.7)
            try_seq(a_opt, "final_eg")

            # One more local polish
            rem2 = deadline - time.monotonic() - 3
            if rem2 > 5:
                a_opt, _ = local_polish(best_seq, time_limit=min(rem2, 20), max_rounds=50)
                try_seq(a_opt, "final_local")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(max(0, x)) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in np.maximum(best_seq, 0.0)]
```
