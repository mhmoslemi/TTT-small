```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel "dual-primal balancing" 
iteration inspired by KKT conditions, combined with subgradient polishing and 
LP-based Frank-Wolfe steps. The balancing iteration directly targets the 
necessary condition that the cross-correlation of u with the dual weights 
should be constant on the support of u.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_fft(u, lam):
    """Compute h[i] = sum_k lam[k] * u[k-i] for i=0,...,2n-2 via FFT."""
    n = len(u)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    U = np.fft.rfft(u, fft_size)
    L = np.fft.rfft(lam, fft_size)
    # Cross-correlation: h[i] = sum_k lam[k] * u[k-i]
    # = ifft(FFT(lam) * conj(FFT(u)))
    h = np.fft.irfft(L * np.conj(U), fft_size)[:size]
    return h


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv) with softmax weighting."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def solve_lp(f, M, time_limit=10):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def balancing_optimize(u0, time_limit, verbose=False):
    """
    Dual-primal balancing iteration.
    At optimality: sum_k lambda[k] * u[k-i] = const for all i in support(u).
    We iteratively update u to satisfy this condition while updating lambda
    to concentrate on near-peak positions.
    """
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)
    
    conv = fast_convolve(u)
    best_obj = conv.max()
    best_u = u.copy()
    
    t0 = time.monotonic()
    temp = 0.01  # temperature for peak softmax
    eta = 0.3    # step size
    top_frac = 0.3  # fraction of peaks to use for lambda
    min_top = 8
    no_improve = 0
    it = 0
    eta_decay = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Update lambda (dual weights on peaks)
        n_peaks = max(min_top, int(n * top_frac))
        if n_peaks >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        lambda_vec = np.zeros(len(conv))
        lambda_vec[peaks] = w
        
        # Compute h[i] = sum_k lambda[k] * u[k-i]
        h = cross_corr_fft(u, lambda_vec)
        h_support = h[:n]  # h[i] for i in [0, n-1]
        
        # On support of u, h should be constant
        support = u > 1e-10
        if support.sum() < 2:
            break
        
        h_on_support = h_support[support]
        target = h_on_support.mean()
        
        if target < 1e-15:
            break
        
        # Multiplicative update: u[i] *= (target / h[i])^eta
        # Only on support; keep zeros as zeros
        ratio = np.ones(n)
        ratio[support] = (target / np.maximum(h_on_support, 1e-15)) ** eta
        u_new = u * ratio
        u_new = np.maximum(u_new, 0.0)
        s_new = u_new.sum()
        if s_new < 1e-15:
            break
        u_new = u_new / s_new
        
        # Evaluate
        conv = fast_convolve(u_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_u = u_new.copy()
            no_improve = 0
            u = u_new
        else:
            # Try a smaller step
            half_ratio = np.ones(n)
            half_ratio[support] = (target / np.maximum(h_on_support, 1e-15)) ** (eta * 0.5)
            u_half = u * half_ratio
            u_half = np.maximum(u_half, 0.0)
            u_half = u_half / u_half.sum()
            conv_half = fast_convolve(u_half)
            if conv_half.max() < best_obj - 1e-13:
                best_obj = conv_half.max()
                best_u = u_half.copy()
                u = u_half
                no_improve = 0
            else:
                no_improve += 1
                # Shrink step
                eta *= 0.9
                temp *= 0.95
        
        # Adaptive parameters
        eta = min(eta * eta_decay if no_improve == 0 else eta, 0.5)
        
        if verbose and it % 100 == 0:
            print(f"    Balancing it={it}, obj={best_obj:.8f}, eta={eta:.4f}, temp={temp:.6f}")
        
        if no_improve > 200:
            # Reset with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(u)
            best_obj = min(best_obj, conv.max())
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_u = u.copy()
            eta = 0.3
            temp = 0.01
            no_improve = 0
    
    return best_u, best_obj


def subgradient_refine(u0, time_limit, initial_lr=0.005, mom=0.93, top_k=64, temp=0.001):
    """Subgradient descent with momentum for polishing."""
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)
    
    conv = fast_convolve(u)
    best_obj = conv.max()
    best_u = u.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        sg = compute_subgradient(u, conv, top_k=max(top_k, n//8), temp=cur_temp)
        v = mom * v - cur_lr * sg
        u_new = project_simplex(u + v)
        conv = fast_convolve(u_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_u = u_new.copy()
            u = u_new
            no_improve = 0
        else:
            # Try without momentum
            u_no_mom = project_simplex(u - cur_lr * sg)
            conv_nm = fast_convolve(u_no_mom)
            if conv_nm.max() < best_obj - 1e-14:
                best_obj = conv_nm.max()
                best_u = u_no_mom.copy()
                u = u_no_mom
                conv = fast_convolve(u)
                no_improve = 0
                v = np.zeros(n)
            else:
                no_improve += 1
                u = u_new  # still update position
        
        cur_lr *= 0.9997
        cur_temp *= 1.002
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 11 + 5)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(u)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.5
            no_improve = 0
    
    return best_u, best_obj


def fw_lp_optimize(u0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)
    
    conv = fast_convolve(u)
    M = conv.max()
    best_obj = M
    best_u = u.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp(u, M * m_exp, time_limit=min(10, rem // 4))
        if g is None:
            m_exp *= 1.3
            if m_exp > 10:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - u)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ut = (1.0 - alpha) * u + alpha * g
            ut = np.maximum(ut, 0.0)
            ut = ut / ut.sum()
            val = fast_convolve(ut).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            # Fine search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                if hi - lo < 1e-5:
                    break
                mid = (lo + hi) / 2.0
                ut = (1.0 - mid) * u + mid * g
                ut = np.maximum(ut, 0.0)
                ut = ut / ut.sum()
                val = fast_convolve(ut).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            u = (1.0 - best_alpha) * u + best_alpha * g
            u = np.maximum(u, 0.0)
            u = u / u.sum()
            conv = fast_convolve(u)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_u = u.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 19 + 3)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(u)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_u, best_obj


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())
    
    # raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(rc / rc.sum())
    
    # Hanning-like
    hn = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(hn / hn.sum())
    
    # Exponential
    e1 = np.exp(-2 * x / n)
    inits.append(e1 / e1.sum())
    e2 = np.exp(-2 * (n - 1 - x) / n)
    inits.append(e2 / e2.sum())
    
    # Gaussian centered
    g = np.exp(-4 * ((x - n/2 + 0.5) / n) ** 2)
    inits.append(g / g.sum())
    
    # Two Gaussians
    g2 = np.exp(-8 * ((x - n*0.3) / n) ** 2) + np.exp(-8 * ((x - n*0.7) / n) ** 2)
    inits.append(g2 / g2.sum())
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    r2 = rng.dirichlet(np.ones(n) * 0.1)
    inits.append(r2)
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel dual-primal balancing 
iteration (targeting KKT conditions for the peak-minimization problem), 
combined with subgradient polishing and LP-based Frank-Wolfe steps.
The balancing iteration enforces the necessary optimality condition that 
cross-correlation of u with dual weights is constant on the support.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_fft(u, lam):
    """Compute h[i] = sum_k lam[k]*u[k-i] for i=0..2n-2 via FFT."""
    n = len(u)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    U = np.fft.rfft(u, fft_size)
    L = np.fft.rfft(lam, fft_size)
    h = np.fft.irfft(L * np.conj(U), fft_size)[:size]
    return h


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def solve_lp(f, M, time_limit=10):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def balancing_optimize(u0, time_limit):
    """
    Dual-primal balancing iteration targeting KKT condition:
    sum_k lambda[k] * u[k-i] = const for all i in support(u).
    Uses multiplicative updates with adaptive step size.
    """
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)

    conv = fast_convolve(u)
    best_obj = conv.max()
    best_u = u.copy()

    t0 = time.monotonic()
    temp = 0.005
    eta = 0.4
    top_frac = 0.25
    min_top = 10
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute dual weights (lambda) on top peaks
        n_peaks = max(min_top, int(n * top_frac))
        if n_peaks >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        lambda_vec = np.zeros(len(conv))
        lambda_vec[peaks] = w

        # Compute h[i] = sum_k lambda[k] * u[k-i]
        h = cross_corr_fft(u, lambda_vec)
        h_support = h[:n]

        support = u > 1e-12
        if support.sum() < 2:
            break

        h_on_sup = h_support[support]
        target = h_on_sup.mean()
        if target < 1e-15:
            break

        # Multiplicative update: u[i] *= (target/h[i])^eta on support
        ratio = np.ones(n)
        safe_h = np.maximum(h_on_sup, 1e-15)
        ratio[support] = (target / safe_h) ** eta
        u_new = u * ratio
        u_new = np.maximum(u_new, 0.0)
        s_new = u_new.sum()
        if s_new < 1e-15:
            break
        u_new = u_new / s_new

        conv = fast_convolve(u_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_u = u_new.copy()
            u = u_new
            no_improve = 0
            # Slightly increase eta if improving fast
            eta = min(eta * 1.005, 0.6)
        else:
            # Try half step
            ratio_half = np.ones(n)
            ratio_half[support] = (target / safe_h) ** (eta * 0.3)
            u_half = u * ratio_half
            u_half = np.maximum(u_half, 0.0)
            u_half = u_half / u_half.sum()
            conv_half = fast_convolve(u_half)
            if conv_half.max() < best_obj - 1e-14:
                best_obj = conv_half.max()
                best_u = u_half.copy()
                u = u_half
                no_improve = 0
            else:
                no_improve += 1
                eta *= 0.92
                temp *= 0.97

        eta = min(eta, 0.6)

def subgradient_refine(u0, time_limit, initial_lr=0.004, mom=0.93, top_k=64, temp=0.001):
    """Subgradient descent with momentum for polishing."""
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)

    conv = fast_convolve(u)
    best_obj = conv.max()
    best_u = u.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        sg = compute_subgradient(u, conv, top_k=max(top_k, n // 8), temp=cur_temp)
        v = mom * v - cur_lr * sg
        u_new = project_simplex(u + v)
        conv = fast_convolve(u_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_u = u_new.copy()
            u = u_new
            no_improve = 0
        else:
            # Try step without momentum
            u_nm = project_simplex(u - cur_lr * sg)
            conv_nm = fast_convolve(u_nm)
            if conv_nm.max() < best_obj - 1e-14:
                best_obj = conv_nm.max()
                best_u = u_nm.copy()
                u = u_nm
                conv = fast_convolve(u)
                v = np.zeros(n)
                no_improve = 0
            else:
                no_improve += 1
                u = u_new

        cur_lr *= 0.9996
        cur_temp *= 1.003

        if no_improve > 250:
            rng = np.random.default_rng(it * 17 + 3)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(u)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.6
            no_improve = 0

    return best_u, best_obj


def fw_lp_optimize(u0, time_limit):
    """Frank-Wolfe with LP oracle for global direction finding."""
    u = np.maximum(u0.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)

    conv = fast_convolve(u)
    M = conv.max()
    best_obj = M
    best_u = u.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 4:
            break

        g = solve_lp(u, M * m_exp, time_limit=min(8, rem // 5))
        if g is None:
            m_exp *= 1.4
            if m_exp > 12:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - u)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ut = (1.0 - alpha) * u + alpha * g
            ut = np.maximum(ut, 0.0)
            ut = ut / ut.sum()
            val = fast_convolve(ut).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                if hi - lo < 1e-5:
                    break
                mid = (lo + hi) / 2.0
                ut = (1.0 - mid) * u + mid * g
                ut = np.maximum(ut, 0.0)
                ut = ut / ut.sum()
                val = fast_convolve(ut).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        u = (1.0 - best_alpha) * u + best_alpha * g
        u = np.maximum(u, 0.0)
        u = u / u.sum()
        conv = fast_convolve(u)
        M = conv.max()
        m_exp = 1.0

        if M < best_obj - 1e-14:
            best_obj = M
            best_u = u.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 23 + 11)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(u)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_u, best_obj


def generate_inits(n, rng):

    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(rd / rd.sum())

    ri = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ri / ri.sum())

    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01

    e1 = np.exp(-2.5 * x / n)
    inits.append(e1 / e1.sum())

    e2 = np.exp(-2.5 * (n - 1 - x) / n)
    inits.append(e2 / e2.sum())

    gc = np.exp(-5.0 * ((x - n / 2 + 0.5) / n) ** 2)
    inits.append(gc / gc.sum())

    g2 = np.exp(-8.0 * ((x - 0.3 * n) / n) ** 2) + np.exp(-8.0 * ((x - 0.7 * n) / n) ** 2)
    inits.append(g2 / g2.sum())

    r1 = rng.dirichlet(np.ones(n) * 0.4)
    inits.append(r1)

    r2 = rng.dirichlet(np.ones(n) * 0.15)
    inits.append(r2)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: dual-primal balancing + subgradient + FW-LP with 
    adaptive multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(parent)}, sum={parent.sum():.4f}")
    except Exception:
        pass

    # Helper to try a sequence (inlined to avoid closure)
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # === Phase 1: Balancing on parent ===
    print("=== Phase 1: Balancing on Parent ===")
    if parent is not None:
        n_p = len(parent)
        u_parent = parent / parent.sum()
        t1 = min(250, deadline - time.monotonic() - 600)
        if t1 > 30:
            print(f"  Balancing (n={n_p}), t={t1:.0f}s")
            a_opt, obj = balancing_optimize(u_parent, time_limit=t1)
            _try(a_opt * parent.sum(), "bal_parent")

            rem = deadline - time.monotonic() - 500
            if rem > 25:
                a_opt2, _ = subgradient_refine(a_opt, time_limit=min(rem, 100),
                                               initial_lr=0.003, top_k=n_p // 6, temp=0.0005)
                _try(a_opt2 * parent.sum(), "bal_sg_parent")

    # === Phase 2: Multi-n exploration ===
    print("\n=== Phase 2: Multi-n Exploration ===")

    n_values = [256, 320, 384, 448, 512, 640]
    if parent is not None:
        if len(parent) in n_values:
            n_values.remove(len(parent))
        n_values.insert(0, len(parent))

    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        if parent is not None and n == len(parent):
            continue

        inits = generate_inits(n, rng)
        rem = deadline - time.monotonic() - 180
        n_remaining = len(n_values) - n_values.index(n) - 1
        t_per_n = max(30, min(200, rem / max(1, n_remaining + 1)))
        t_per_init = max(12, min(35, t_per_n / max(1, len(inits))))

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 8:
                break

            a_opt, obj = balancing_optimize(f_init, time_limit=t_alloc * 0.55)
            _try(a_opt * n, f"n{n}_bal{i}")

            rem2 = deadline - time.monotonic() - 80
            if rem2 > 12:
                a_opt2, _ = subgradient_refine(a_opt, time_limit=min(rem2, t_alloc * 0.4),
                                               initial_lr=0.005, top_k=max(32, n // 10), temp=0.001)
                _try(a_opt2 * n, f"n{n}_sg{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        u_best = best_seq / best_seq.sum()
        s_best = best_seq.sum()

        # FW-LP deep (multiple starts)
        for p in range(6):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem // (6 - p), 50)

            if p == 0:
                f_init = u_best.copy()
            elif p == 1:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.004)
            elif p == 2:
                f_init = project_simplex(0.92 * u_best + 0.08 * (u_best[::-1]))
            elif p == 3:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.008)
            elif p == 4:
                f_init = project_simplex(0.85 * u_best + 0.15 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.002)

            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            _try(a_opt * s_best, f"deep_fw_{p}")

        # Balancing deep
        for p in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem // (4 - p), 40)

            if p == 0:
                f_init = u_best.copy()
            elif p == 1:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.003)
            elif p == 2:
                f_init = project_simplex(0.95 * u_best + 0.05 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.006)

            a_opt, obj = balancing_optimize(f_init, time_limit=t_alloc)
            _try(a_opt * s_best, f"deep_bal_{p}")

        # SG deep
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(rem // (3 - p), 45)

            if p == 0:
                f_init = u_best.copy()
                cfg_lr, cfg_top, cfg_temp = 0.002, n_best // 5, 0.0003
            elif p == 1:
                f_init = project_simplex(u_best + rng.standard_normal(n_best) * 0.002)
                cfg_lr, cfg_top, cfg_temp = 0.004, n_best // 4, 0.001
            else:
                f_init = project_simplex(0.9 * u_best + 0.1 * (np.ones(n_best) / n_best))
                cfg_lr, cfg_top, cfg_temp = 0.001, n_best // 3, 0.0001

            a_opt, _ = subgradient_refine(f_init, time_limit=t_alloc,
                                          initial_lr=cfg_lr, top_k=cfg_top, temp=cfg_temp)
            _try(a_opt * s_best, f"deep_sg_{p}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        u_best = best_seq / best_seq.sum()
        s_best = best_seq.sum()

        # Fine balancing
        rem = deadline - time.monotonic() - 25
        if rem > 12:
            a_opt, _ = balancing_optimize(u_best, time_limit=min(rem, 35))
            _try(a_opt * s_best, "polish_bal")

        # Fine SG
        rem = deadline - time.monotonic() - 18
        if rem > 8:
            a_opt2, _ = subgradient_refine(a_opt, time_limit=min(rem, 25),
                                           initial_lr=0.001, top_k=n_best // 3, temp=0.00005)
            _try(a_opt2 * s_best, "polish_sg")

        # FW final
        rem = deadline - time.monotonic() - 12
        if rem > 8:
            a_opt3, _ = fw_lp_optimize(a_opt2, time_limit=min(rem, 20))
            _try(a_opt3 * s_best, "polish_fw")

        # Micro perturbations + quick polish
        rem = deadline - time.monotonic() - 5
        if rem > 4:
            for m in range(5):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                f_pert = project_simplex(a_opt3 + rng.standard_normal(n_best) * 0.0008)
                a_micro, _ = subgradient_refine(f_pert, time_limit=min(rem, 4),
                                               initial_lr=0.0004, top_k=n_best // 3, temp=0.00003)
                _try(a_micro * s_best, f"micro_{m}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
