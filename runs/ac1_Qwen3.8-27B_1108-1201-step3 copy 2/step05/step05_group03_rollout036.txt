    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_direction(a, M):
    """
    Solve the LP: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Returns the optimal b (unnormalized).
    """
    n = len(a)
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (a*b)_k <= M for k = 0, ..., 2n-2
    # (a*b)_k = sum_j C[k,j] * b_j
    # C[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A_ub[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n)

        b = lp_direction(a, M)
        b_sum = b.sum()

        best_t = 0
        best_sc = compute_score(a)
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:

            a_t = normalize(a_t)
            sc = compute_score(a_t)
            if sc < best_sc:

                best_t = t

            a = (1-best_t) * a + best_t * b
            a = normalize(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose and iteration % 10 == 0:
                    print(f"    LP iter {iteration}: score={best_score:.10f}, t={best_t}")
        else:
            # Try subgradient step as fallback
            k_star = np.argmax(conv)
            sg = np.zeros(n)
            for j in range(n):
                if 0 <= k_star - j < n:
                    sg[j] = 2.0 * a[k_star - j]
            lr = 0.01
            a_new = a - lr * sg
            a_new = normalize(a_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        # Periodic check
        if iteration % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 5:
                break
    
    return best_a, best_score

"""
LP-driven optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Primary method: iterative LP to find optimal mixing direction.
Secondary: subgradient averaging, coordinate descent polishing.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.fft import next_fast_len
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_mixing_lp(a, M):
    """Solve: max sum(b) s.t. b>=0, (a*b)_k <= M for all k."""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b)
    
    # Build constraint matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A_ub[k, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.success:
        return np.maximum(res.x, 0)
    return np.ones(n) / n

def subgradient_step(a, conv, lr, top_k=50):
    """Averaged subgradient step focusing on top peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    # Weight by how close to max
    temp = 0.01 * conv.max() + 1e-10
    weights = np.exp((conv[peaks] - conv.max()) / temp)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(k_actual):
        k = int(peaks[i])
        w = weights[i]
        if w < 1e-12:
            continue
        # sg_j = 2 * a[k-j] for valid j
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            sg[j_lo:j_hi+1] += 2.0 * w * a[k - np.arange(j_lo, j_hi+1)]
    
    return np.maximum(a * np.exp(-lr * sg), 1e-15)

def coord_descent(a, time_limit, n_passes_max=50):
    """Coordinate descent with bounded scalar optimization."""
    from scipy.optimize import minimize_scalar
    n = len(a)
    a = normalize(a.copy())
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    for pass_num in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = a[j]
            # Try to optimize coordinate j
            def obj(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                total = new_a.sum()
                if total < 1e-15:
                    return float('inf')
                new_a = new_a / total
                conv = fast_convolve(new_a)
                return conv.max()
            
            lo = max(0, old * 0.5)
            hi = min(0.5, old * 2 + 0.01)
            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-7, 'maxiter': 25})
                if res.fun < obj(old) - 1e-12:
                    new_a = a.copy()
                    new_a[j] = res.x
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except:
                pass
        if not improved and pass_num > 3:
            break
    
    return best_a, best_score

        if b_sum < 1e-10:

        b = b / b_sum
        
        # Line search
        best_t = 0
        best_sc_t = compute_score(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_t = normalize((1-t) * a + t * b)

"""
LP-driven optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Primary method: iterative linear programming to find optimal mixing directions.
Secondary: smoothed subgradient descent, coordinate descent polishing.
Key insight: Given current sequence a with peak M, solve LP to find b maximizing
sum(b) subject to (a*b)_k <= M for all k. The normalized b is a direction that
reduces the convolution peak when mixed with a.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import next_fast_len
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Project to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for the mixing LP."""
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_mixing_lp(a, M):
    """Solve: max sum(b) s.t. b>=0, (a*b)_k <= M for all k=0..2n-2."""
    n = len(a)
    c = -np.ones(n)
    A_ub = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b
    except Exception:
        pass
    return np.ones(n)


def subgradient_direction(a, conv, top_k=30):
    """Compute smoothed subgradient focusing on top peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    conv_max = conv.max()
    temp = 0.02 * conv_max + 1e-12
    weights = np.exp((conv[peaks] - conv_max) / temp)
    weights /= weights.sum()

    sg = np.zeros(n)
    for i in range(k_actual):
        w = weights[i]
        if w < 1e-12:
            continue
        k = int(peaks[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sg[idx] += 2.0 * w * a[k - idx]
    return sg


def lp_optimize(a, time_limit, verbose=False):
    """Main LP-based iterative optimization with subgradient fallback."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iter_count = 0
    stuck_count = 0
    last_lp_time = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Decide: LP step or subgradient step
        use_lp = (iter_count % 3 == 1) or (time.monotonic() - last_lp_time > 3.0)

        if use_lp:
            b = solve_mixing_lp(a, M)
            b_sum = b.sum()
            if b_sum < 1e-10:
                b = np.ones(n) / n
                b_sum = 1.0
            b = b / b_sum
            last_lp_time = time.monotonic()
        else:
            sg = subgradient_direction(a, conv, top_k=min(40, n // 15))
            lr = 0.03
            b = np.maximum(a * np.exp(-lr * sg), 1e-15)
            b = b / b.sum()

        # Line search for mixing ratio
        best_t = 0.0
        best_sc_t = compute_score(a)
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_t = normalize((1.0 - t) * a + t * b)
            sc = compute_score(a_t)
            if sc < best_sc_t:
                best_sc_t = sc
                best_t = t

        if best_t > 0.0 and best_sc_t < best_score - 1e-13:
            a = normalize((1.0 - best_t) * a + best_t * b)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            stuck_count = 0
        else:
            # Stuck: try aggressive subgradient
            sg = subgradient_direction(a, conv, top_k=min(60, n // 8))
            for lr_try in [0.01, 0.03, 0.05, 0.1, 0.2]:
                a_new = normalize(a * np.exp(-lr_try * sg))
                sc = compute_score(a_new)
                if sc < best_score:
                    a = a_new
                    best_score = sc
                    best_a = a.copy()
                    stuck_count = 0
                    break
            else:
                stuck_count += 1

        # Perturb if stuck
        if stuck_count > 30:
            noise = np.random.standard_normal(n) * 0.003
            a = normalize(a + noise)
            stuck_count = 0

        if verbose and iter_count % 50 == 0:
            print(f"      LP iter {iter_count}: score={best_score:.10f}, stuck={stuck_count}")

    return best_a, best_score


def coord_descent(a, time_limit):
    """Coordinate descent with bounded scalar optimization per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_score = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = a[j]
            if old < 1e-14:
                # Try small injection
                for v_try in [0.002, 0.005, 0.01]:
                    new_a = a.copy()
                    new_a[j] = v_try
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                continue

            # Bounded scalar optimization
            lo = max(0.0, old * 0.4)
            hi = min(0.5, old * 2.5 + 0.02)

            def obj_v(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                total = new_a.sum()
                if total < 1e-15:
                    return float('inf')
                new_a = new_a / total
                conv = fast_convolve(new_a)
                return conv.max()

            old_val = obj_v(old)
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-8, 'maxiter': 25})
                if res.fun < old_val - 1e-13:
                    new_a = a.copy()
                    new_a[j] = res.x
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            break

    return best_a, best_score


def pair_transfer(a, time_limit, n_trials=5000):
    """Random pair transfer local search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        if a[j] < 1e-13:
            continue

        best_delta = 0.0
        best_sc = best_score
        for f in [0.1, 0.3, 0.5, 0.7]:
            delta = a[j] * f
            new_a = a.copy()
            new_a[j] = a[j] - delta
            new_a[i] = a[i] + delta
            sc = compute_score(new_a)
            if sc < best_sc:
                best_sc = sc
                best_delta = delta

        if best_delta != 0.0:
            a[j] -= best_delta
            a[i] += best_delta
            a = normalize(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return normalize(np.interp(x_new, x_old, f))
