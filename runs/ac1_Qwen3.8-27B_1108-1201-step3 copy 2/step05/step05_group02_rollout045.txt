Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
adaptive M tracking, refined line search, multi-length exploration,
and structured local refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = f[k::-1]
    for k in range(n, 2 * n - 1):
        i_min = k - n + 1
        A[k, i_min:] = f[n - 1:k - n:-1]
    return A


def solve_lp_f(f, M):
    n = len(f)
    A_ub = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 30, 'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None


def compute_obj(f):
    """Compute max_conv for a normalized sequence (sum=1)."""
    return fast_convolve(f).max()


def evaluate_normalized(f):
    """Evaluate 2*n*max_conv for sequence f (assumed sum=1, f>=0)."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()


def line_search_manual(f, g, M_current):
    """Find best alpha in [0,1] minimizing max_conv of interpolation."""
    best_alpha = 0.0
    best_val = M_current
    
    # Coarse grid
    n_grid = 25
    alphas = np.linspace(0.0, 1.0, n_grid)
    for a in alphas:
        ft = (1.0 - a) * f + a * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = a
    
    if best_alpha == 0.0:
        return 0.0, M_current
    
    # Fine bisection around best_alpha
    if best_alpha <= alphas[1]:
        lo, hi = 0.0, alphas[2]
    elif best_alpha >= alphas[-2]:
        lo, hi = alphas[-3], 1.0
    else:
        idx = int(best_alpha * (n_grid - 1))
        lo = alphas[max(0, idx - 1)]
        hi = alphas[min(n_grid - 1, idx + 1)]

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M tracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    M_ub = M  # Running upper bound on optimal max_conv
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP with current M_ub
        g = solve_lp_f(f, M_ub)
        if g is None:
            # Try relaxing
            M_ub = M * 1.01
            continue
        
        g_sum = g.sum()
        if g_sum < 1.0 + 1e-10:
            # LP says we can't do better than sum=1, try relaxing M
            if M_ub < M * 0.999:
                M_ub = M_ub * 1.02
            else:
                no_improve += 1
            if no_improve > 50:
                break
            continue
        
        g = g / g_sum
        
        # Check if direction is useful
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            # Try slightly looser M
            M_ub = M_ub * 1.001
            continue
        
        # Line search
        best_alpha, best_val = line_search_manual(f, g, M)
        
        if best_alpha < 1e-10:
            no_improve += 1
            M_ub = M_ub * 1.001
            if no_improve > 50:
                break
            continue
        
        # Update
        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        
        new_M = fast_convolve(f_new).max()
        f = f_new
        M = new_M
        
        if M < M_ub:
            M_ub = M * 0.999  # Slightly tighten for next LP
        # Don't increase M_ub
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.002 * min(no_improve // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            M_ub = M
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent_pairs(f0, time_limit):
    """Pair coordinate descent: swap mass between pairs to reduce max_conv."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 100:
        # Get current convolution
        conv = fast_convolve(f)
        peak_k = conv.argmax()
        
        # Try a random set of pairs
        n_tries = min(200, n // 2)
        improved = False
        
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-10 or f[j] < 1e-10:
                continue
            
            # Try moving mass from i to j (and vice versa)
            # f[i] -> f[i] + d, f[j] -> f[j] - d (then renormalize)
            # Or use multiplicative: f[i] *= (1+d), f[j] *= (1-d)
            
            best_d = 0.0
            best_val = conv.max()
            
            for d in [-0.05, -0.02, -0.005, 0.005, 0.02, 0.05]:
                fi = f[i] * (1 + d)
                fj = f[j] * (1 - d)
                if fi < 0 or fj < 0:
                    continue
                # Renormalize to keep sum=1
                # Old sum contribution: f[i] + f[j]
                # New sum contribution: fi + fj
                # Scale factor for rest: (sum - fi - fj) / (sum - f[i] - f[j])
                old_contrib = f[i] + f[j]
                new_contrib = fi + fj
                if old_contrib > 1e-15 and abs(new_contrib - old_contrib) < old_contrib:
                    scale = (1.0 - new_contrib) / (1.0 - old_contrib)
                    f_trial = f.copy()
                    f_trial[i] = fi
                    f_trial[j] = fj
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val:
                        best_val = val
                        best_d = d
            
            if best_d != 0.0:
                fi = f[i] * (1 + best_d)
                fj = f[j] * (1 - best_d)
                old_contrib = f[i] + f[j]
                new_contrib = fi + fj
                scale = (1.0 - new_contrib) / (1.0 - old_contrib)
                f[i] = fi
                f[j] = fj
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                improved = True
                
                if best_val < best_obj - 1e-15:
                    best_obj = best_val
                    best_f = f.copy()
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
