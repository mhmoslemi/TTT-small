import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def multiplicative_descent(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.5
    temp = 0.01
    it = 0
    no_improve = 0

        it += 1
        c = fast_convolve(a)
        cmax = c.max()

            best_a = a.copy()
            no_improve = 0
        else:

        scaled -= scaled.max()

        if ws > 0:

        top_idx = np.argpartition(c, -K)[-K:]
        top_w = w[top_idx]
        tws = top_w.sum()
        if tws > 0:

def solve_lp_direction(f, M, n):
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_step(f, M, n):

    g = solve_lp_direction(f, M * 1.001, n)
    if g is None:
        g = solve_lp_direction(f, M * 1.01, n)
    if g is None:

def lp_fw_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

        it += 1
        g = lp_fw_step(f, M, n)
        if g is None:
            no_improve += 1
            if no_improve > 30:

            continue

            no_improve += 1
            if no_improve > 30:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative weight
descent with FFT-based subgradients, combined with coordinate descent,
LP-FW steps, and multi-length exploration. Key innovation: very fast
per-iteration cost (O(n log n)) enables thousands of optimization steps.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlation(w, a):
    """Compute (w * a_cross)[j] = sum_k w[k]*a[k-j] for j=0..len(a)-1."""
    n = len(a)
    m = len(w)
    needed = m + n - 1
    fft_size = 1 << (needed - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    return np.fft.ifft(W * np.conj(A)).real[:n]


def compute_subgradient(w, a):
    """Subgradient of sum_k w[k]*c[k] where c = a*a (auto-conv)."""
    return 2.0 * cross_correlation(w, a)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_descent(a0, time_limit, verbose=False):
    """
    Fast multiplicative weight descent for minimizing max(auto-conv(a)).
    Uses FFT-based subgradients for O(n log n) per iteration.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    c = fast_convolve(a)
    best_obj = c.max()
    t0 = time.monotonic()

    eta = 0.4
    temp_ratio = 0.015
    it = 0
    no_improve = 0
    momentum = 0.0
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        cmax = c.max()

        if cmax < best_obj - 1e-14:
            best_obj = cmax
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    MD iter {it} ({time.monotonic()-t0:.0f}s): obj={best_obj:.14f}")
        else:
            no_improve += 1

        # Compute soft weights: w[k] proportional to exp(c[k]/(temp*cmax))
        temp = temp_ratio * cmax
        scaled = (c - cmax) / temp
        scaled = np.clip(scaled, -80, 0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = compute_subgradient(w, a)

        # Normalize gradient so max is 1
        gmax = g.max()
        if gmax < 1e-15:
            break
        g = g / gmax

        # Momentum
        v = momentum * v + (1.0 - momentum) * g

        # Multiplicative update
        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        a = a / s

        # Annealing
        temp_ratio *= 1.001
        eta *= 0.99995

        # Periodic adjustments

            temp_ratio = max(temp_ratio * 0.5, 0.002)

            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = 0.3
            temp_ratio = 0.01
            momentum = 0.0
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiple trial values per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.4, 0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        c = fast_convolve(a)
        cur_max = c.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old = a[j]
            if old < 1e-12 or old > 0.9999:
                continue

            best_t = old
            best_val = cur_max

            for mult in multipliers:
                t = old * mult
                if t < 0 or t >= 0.9999:
                    continue
                denom = 1.0 - old
                if denom < 1e-12:
                    continue
                scale = (1.0 - t) / denom
                if scale < 0.001 or scale > 200:
                    continue
                a_trial = a * scale
                a_trial[j] = t
                val = fast_convolve(a_trial).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t

            if best_t != old:
                denom = 1.0 - old
                scale = (1.0 - best_t) / denom
                a = a * scale
                a[j] = best_t
                a = a / a.sum()
                cur_max = best_val
                if cur_max < best_obj - 1e-14:
                    best_obj = cur_max
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 2:
            break
        if verbose and passes % 3 == 0:
            print(f"    CD pass {passes} ({time.monotonic()-t0:.0f}s): obj={best_obj:.14f}")

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 45, 'presolve': True}
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 6.0:
                m_relax = 1.0
            no_improve += 1
            if no_improve > 40:
                break
            continue
        m_relax = 1.0

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it} ({time.monotonic()-t0:.0f}s): obj={best_obj:.14f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: fast multiplicative descent (main workhorse),
    coordinate descent, LP-FW, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n=768, score={best_score:.10f}")

    n_parent = len(parent)

    # === Phase 1: Fast multiplicative descent (long) ===
    print(f"\n=== Phase 1: MD round 1, n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.28)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = multiplicative_descent(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.10f}")

    # === Phase 2: Coordinate descent ===
    print(f"\n=== Phase 2: CD (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.18)
    if t2 > 15:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.10f}")

    # === Phase 3: MD round 2 (fine-tuning with smaller steps) ===
    print(f"\n=== Phase 3: MD round 2 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(200, rem * 0.18)
    if t3 > 15:
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = multiplicative_descent(best_seq, time_limit=t3, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.10f}")

    # === Phase 4: LP-FW for larger moves ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(150, rem * 0.15)
    if t4 > 30:
        print(f"  Budget: {t4:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.10f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.10f}) ===")
    n_best = len(best_seq)
    for delta in [-128, 128, -256, 256, -64, 64, -512, 512]:
        nl = n_best + delta
        if 64 < nl < 4096 and nl != n_best:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(80, rem * 0.12)
            if t_alloc < 15:
                continue
            print(f"  n={nl}, t={t_alloc:.0f}s")
            f_interp = interpolate_sequence(best_seq, nl)
            a_opt, _ = multiplicative_descent(f_interp, time_limit=t_alloc * 0.7)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.10f} (n={nl})")
                # Quick polish
                rem2 = deadline - time.monotonic()
                if rem2 > 20:
                    a_opt2, _ = coordinate_descent(a_opt, time_limit=min(40, rem2 * 0.3))
                    sc2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_opt2.copy()
                        print(f"  POLISHED: {sc2:.10f} (n={nl})")

    # === Phase 6: Final MD + CD ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_md = min(rem * 0.4, 120)
        a_opt, _ = multiplicative_descent(best_seq, time_limit=t_md, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  MD: {sc:.10f}")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.4, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  CD: {sc:.10f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse restarts (best={best_score:.10f}) ===")
    n_best = len(best_seq)
    inits = []

    noise1 = rng.standard_normal(n_best) * 0.003
    inits.append(("noise3k", project_simplex(best_seq + noise1)))

    noise2 = rng.standard_normal(n_best) * 0.008
    inits.append(("noise8k", project_simplex(best_seq + noise2)))

    f_uni = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
    inits.append(("mix_uni", project_simplex(f_uni)))

    f_rev = 0.5 * best_seq + 0.5 * best_seq[::-1]
    inits.append(("sym", project_simplex(f_rev)))

    for label, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(60, rem * 0.15)
        if t_alloc < 10:
            continue
        a_opt, _ = multiplicative_descent(f_init, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f}")

    # === Phase 8: Micro-perturbations ===
    print(f"\n=== Phase 8: Micro-search (best={best_score:.10f}) ===")
    n_best = len(best_seq)
    for trial in range(30):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0001 + trial * 0.00005
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.10f}")
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = multiplicative_descent(f_pert, time_limit=min(8, rem2 * 0.4))
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  POLISHED: {sc2:.10f}")

    # === Final: tiny MD for last bit ===
    rem = deadline - time.monotonic()
    if rem > 5:
        a_opt, _ = multiplicative_descent(best_seq, time_limit=min(rem * 0.8, 30))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL MD: {sc:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
