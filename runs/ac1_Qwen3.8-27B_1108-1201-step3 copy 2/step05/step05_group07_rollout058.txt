from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(f, b):
    n = len(f)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    B = rfft(b, fft_size)
    return irfft(F * B, fft_size)[:size]

def build_conv_matrix(f):
    """Build the matrix A such that (A @ b)[k] = (f*b)[k] for cross-convolution."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    return A

Uses successive linear programming (linearization of bilinear constraints)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A @ b)[k] = (f*b)[k] for cross-convolution."""
    n = len(f)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(size, n))
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    n = len(f)
    size = 2 * n - 1
    A = build_conv_matrix_sparse(f)
    
    # Maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M * ones(size)
    A_ub = A
    b_ub = M * np.ones(size)
    
    # Bounds: 0 <= b[i] <= 1 (upper bound to prevent numerical issues)
    bounds = [(0, 1)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x


def line_search(f, g, max_t=1.0, n_trials=10):
    """Find best t in [0, max_t] for f_new = (1-t)*f + t*g."""
    S = f.sum()
    best_t = 0.0
    best_score = 2 * len(f) * fast_convolve(f).max() / (S ** 2)
    
    ts = np.linspace(0, max_t, n_trials + 1)[1:]  # skip t=0
    for t in ts:
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        S_new = f_new.sum()
        if S_new < 0.01:
            continue
        score = 2 * len(f_new) * fast_convolve(f_new).max() / (S_new ** 2)
        if score < best_score:
            best_score = score
            best_t = t
    
    return best_t, best_score

    LP-based iterative optimization.
    f0: initial sequence (will be normalized to sum=1).

    S = f.sum()

        return f0

    n = len(f)

def gradient_polish(f0, time_limit=60.0, T=50):
    """Quick gradient-based polish using soft-max objective."""
    from scipy.optimize import minimize as scipy_minimize
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    S = f.sum()
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = 2 * n * fast_convolve(f).max()
    
    # Use log-softmax parameterization
    z = np.log(np.maximum(f, 1e-30))
    
    def obj_grad(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        w = exp_Tb / exp_Tb.sum()
        # Cross-correlation for gradient
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        S_dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S_dot)
        return obj, grad_z
    
    try:
        res = scipy_minimize(
            fun=lambda z: obj_grad(z)[0],
            x0=z,
            jac=lambda z: obj_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
        )
        z_opt = res.x
        z_shifted = z_opt - z_opt.max()
        exp_z = np.exp(z_shifted)
        a_opt = exp_z / exp_z.sum()
        score = 2 * n * fast_convolve(a_opt).max()
        if score < best_score:
            best_f = a_opt
            best_score = score
    except Exception as e:
        print(f"  Gradient polish error: {e}")
    
    return best_f

"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses successive linear programming (the LP direction-finding method) combined
with multi-length exploration, gradient polishing, and perturbation-based
restarts to escape local plateaus.

import numpy as np
from scipy.optimize import linprog, minimize as scipy_minimize
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A @ b)[k] = (f*b)[k] (cross-convolution)."""
    n = len(f)
    size = 2 * n - 1
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(size):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        nj = j_max - j_min + 1
        row_parts.append(np.full(nj, k, dtype=np.int32))
        col_parts.append(np.arange(j_min, j_max + 1, dtype=np.int32))
        js = np.arange(j_min, j_max + 1)
        data_parts.append(f[k - js])
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(size, n))


def solve_lp_max_sum(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0, b <= 1."""
    n = len(f)
    size = 2 * n - 1
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, 1.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def solve_lp_min_peak(f, M):
    """Solve LP: min t s.t. (f*b)[k] <= t*M for all k, sum(b)=1, b>=0."""
    n = len(f)
    size = 2 * n - 1
    # Variables: [b_0, ..., b_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: (f*b)[k] - t*M <= 0 for all k
    A_base = build_conv_matrix_sparse(f)
    # Add column for t: -M in each row
    t_col = csr_matrix(-M * np.ones((size, 1)))
    A_ub = A_base
    # We need to augment with the t column
    A_ub_data = np.hstack([A_base.toarray(), -M * np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub_data, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n], res.x[-1]


def line_search_conv(f, g, n_trials=15):
    """Find t in (0,1] minimizing max of auto-conv of (1-t)f + tg.
    Assumes f and g are normalized to sum=1.
    Returns (best_t, best_max_conv).
    """
    n = len(f)
    best_t = 0.0
    best_mc = fast_convolve(f).max()
    
    for i in range(1, n_trials + 1):
        t = i / n_trials
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        mc = fast_convolve(f_new).max()
        if mc < best_mc:
            best_mc = mc
            best_t = t
    return best_t, best_mc


def lp_iterate(f, time_limit, max_iter=200, verbose=True, use_min_peak=False):

    LP-based iterative optimization.
    f: initial sequence (normalized to sum=1).
    Returns (optimized_f, best_score).
    """
    t0 = time.monotonic()
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_mc = fast_convolve(f).max()
    best_score = 2.0 * n * best_mc  # with sum=1
    
    n_accepted = 0
    n_stuck = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        if use_min_peak:
            result = solve_lp_min_peak(f, M)
            if result is None:
                break
            g, t_val = result
        else:
            b = solve_lp_max_sum(f, M)
            if b is None:
                break
            bs = b.sum()
            if bs < 1e-15:
                break
            g = b / bs
            t_val = None
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Line search
        best_t, new_mc = line_search_conv(f, g, n_trials=15)
        
        if best_t > 0 and new_mc < best_mc - 1e-14:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            cur_mc = fast_convolve(f).max()
            if cur_mc < best_mc:
                best_mc = cur_mc
                best_f = f.copy()
                best_score = 2.0 * n * best_mc
            n_accepted += 1
            n_stuck = 0
            if verbose and n_accepted % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP step {n_accepted}: score={best_score:.6f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            n_stuck += 1
            if n_stuck >= 5:
                # Try perturbation
                rng_local = np.random.default_rng(it * 12345 + 678)
                pert = rng_local.standard_normal(n) * 0.005
                f_pert = f + pert
                f_pert = np.maximum(f_pert, 0)
                f_pert = f_pert / f_pert.sum()
                # Check if perturbation helps

                    f = f_pert
                    best_mc = pert_mc
                    best_f = f.copy()
                    best_score = 2.0 * n * best_mc
                    n_stuck = 0
                elif n_stuck >= 15:
                    if verbose:
                        print(f"    LP stuck after {n_accepted} accepted steps, score={best_score:.6f}")
                    break
    
    return best_f, best_score


def grad_polish(f0, time_limit=60.0, T=50.0):
    """Gradient-based polish using soft-max smoothed objective."""
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    
    # Log-space parameterization
    z = np.log(np.maximum(f, 1e-30))
    
    def obj_fn(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        return softmax_obj(z, T, n)[0]
    
    def jac_fn(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return softmax_obj(z, T, n)[1]
    
    try:
        res = scipy_minimize(
            fun=obj_fn,
            x0=z,
            jac=jac_fn,
            method='L-BFGS-B',
            options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
        )
        z_opt = res.x
        a_opt = softmax_from_z(z_opt)
        score = 2.0 * n * fast_convolve(a_opt).max()
        if score < best_score:
            best_f = a_opt
            best_score = score
    except Exception:
        pass
    
    return best_f, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def softmax_obj(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Cross-correlation: (w * a)[i] = sum_j w[j] * a[j-i]... 
    # Actually we need: grad_a_i = 2n * sum_k w_k * a_{k-i} (with bounds)
    # This is the cross-correlation of w and a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We want (w correlation a)[i] = sum_k w[k] * a[k-i]
    # This is the cross-correlation, which is ifft(conj(W) * A)
    grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
    S_dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S_dot)
    return obj, grad_z


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Precompute row/col indices for cross-convolution matrix."""
    size = 2 * n - 1

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    return rows, cols, size
