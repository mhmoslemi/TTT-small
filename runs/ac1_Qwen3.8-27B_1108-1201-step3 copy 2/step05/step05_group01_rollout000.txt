import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    v = v - v.max()
    e = np.exp(np.clip(v, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def nesterov_sg_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient with multi-temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Nesterov parameters
    momentum = 0.9
    lr = 0.005
    temp = 0.001
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute smoothed gradient
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(np.clip(shifted, -500, 500))
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad = 2.0 * R

        # Nesterov update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 5000 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
        else:
            no_improve += 1
            f = f_new  # still move, just didn't beat best

        # Adaptive schedule
        lr *= 0.9998
        temp *= 1.005
        momentum = min(momentum * 1.0001, 0.98)

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = 0.003
            temp = 0.003
            no_improve = 0

        lr = max(lr, 0.0005)

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Structured simulated annealing with fast convolution-change estimation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    T_start = 0.002
    T_end = 1e-6
    total_iters = 80000
    it = 0
    no_improve = 0

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and it < total_iters:
        it += 1
        # Temperature schedule (geometric)
        T = T_start * (T_end / T_start) ** (it / total_iters)

        # Choose move type
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Transfer mass between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = f[i] * 0.3 * (1 if rng.random() < 0.5 else -1)
            if f[i] + delta < 0 or f[j] - delta < 0:
                continue
            f_trial = f.copy()
            f_trial[i] += delta
            f_trial[j] -= delta
            f_trial = f_trial / f_trial.sum()
        elif move_type == 1:
            # Shift mass within a window
            win = rng.integers(3, min(20, n // 2) + 1)
            start = rng.integers(0, n - win)
            idx = start + rng.integers(0, win)
            jdx = start + rng.integers(0, win)
            if idx == jdx:
                continue
            delta = f[idx] * 0.2 * (1 if rng.random() < 0.5 else -1)
            if f[idx] + delta < 0 or f[jdx] - delta < 0:
                continue
            f_trial = f.copy()
            f_trial[idx] += delta
            f_trial[jdx] -= delta
            f_trial = f_trial / f_trial.sum()
        elif move_type == 2:
            # Mirror swap (swap a[i] and a[n-1-i] partially)
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            frac = 0.1 + 0.4 * rng.random()
            delta = (f[j] - f[i]) * frac
            if f[i] + delta < 0 or f[j] - delta < 0:
                continue
            f_trial = f.copy()
            f_trial[i] += delta
            f_trial[j] -= delta
            f_trial = f_trial / f_trial.sum()
        else:
            # Gaussian perturbation on a random subset
            k = rng.integers(3, min(15, n) + 1)
            idxs = rng.choice(n, size=k, replace=False)
            noise = rng.standard_normal(k) * 0.01
            f_trial = f.copy()
            f_trial[idxs] += noise
            f_trial = np.maximum(f_trial, 0)
            ss = f_trial.sum()
            if ss < 1e-15:
                continue
            f_trial /= ss

        conv_trial = fast_convolve(f_trial)
        obj_trial = conv_trial.max()

        # Metropolis acceptance
        delta_obj = obj_trial - fast_convolve(f).max()
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            f = f_trial
            cur_obj = obj_trial
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5000 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f}, T={T:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 20000:
            # Restart
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def aggressive_cd(f0, time_limit):
    """Aggressive coordinate descent with zero-out and large-step moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    multipliers = [0.1, 0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.7, 2.0, 3.0, 5.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-15:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def lbfqg_optimize_v2(f0, time_limit, verbose=False):
    """L-BFGS on smooth objective with more aggressive annealing and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    eps = 1e-15
    t = np.log(f + eps)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    def _smooth_obj(t_vec, beta):
        a = softmax_fn(t_vec)
        conv = fast_convolve(a)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(np.clip(shifted, -500, 500))
        w = exp_shifted / exp_shifted.sum()
        obj = conv_max + np.log(exp_shifted.sum()) / beta

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)
        return obj, grad_t

    betas = [3.0, 8.0, 20.0, 50.0, 120.0, 300.0, 600.0]
    t0_time = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(777)

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            res = minimize(
                _smooth_obj, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 400, 'gtol': 1e-10, 'maxls': 50, 'maxfun': 3000}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS v2 beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

        # After each beta, try a small perturbation restart
        if remaining > 30:
            noise = rng.standard_normal(n) * 0.005
            t_pert = t + noise
            try:
                res2 = minimize(
                    _smooth_obj, t_pert, args=(beta,),
                    jac=True, method='L-BFGS-B',
                    options={'maxiter': 200, 'gtol': 1e-10, 'maxfun': 1000}
                )
                t2 = res2.x
                a2 = softmax_fn(t2)
                a2 = np.maximum(a2, 0.0)
                a2_sum = a2.sum()
                if a2_sum > 1e-15:
                    a2 = a2 / a2_sum
                obj2 = fast_convolve(a2).max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_f = a2.copy()
                    t = t2
            except Exception:
                pass

    return best_f, best_obj


def deflation_restart(f0, n_restarts, time_per_restart, verbose=False):
    """Try structured perturbations (deflation) and re-optimize each."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    rng = np.random.default_rng(424242)

    for r in range(n_restarts):
        if time_per_restart <= 0:
            break

        # Choose perturbation type
        pert_type = r % 4
        if pert_type == 0:
            # Add sinusoidal perturbation
            freq = rng.uniform(1, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.01 + 0.02 * rng.random()
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            f_pert = project_simplex(f + pert)
        elif pert_type == 1:
            # Block shift
            block = rng.integers(n // 8, n // 3)
            shift = rng.integers(1, block)
            f_pert = np.roll(f, shift)
            f_pert = np.maximum(f_pert, 0)
            f_pert = f_pert / f_pert.sum()
        elif pert_type == 2:
            # Random subset amplification
            k = rng.integers(n // 10, n // 3)
            idxs = rng.choice(n, size=k, replace=False)
            f_pert = f.copy()
            f_pert[idxs] *= 1.5
            f_pert = np.maximum(f_pert, 0)
            f_pert = f_pert / f_pert.sum()
        else:
            # Gaussian with varying scale
            scale = 0.005 + 0.02 * rng.random()
            f_pert = project_simplex(f + rng.standard_normal(n) * scale)

        # Quick LP-FW on perturbation
        a_opt, obj_opt = lp_fw_optimize(f_pert, time_limit=time_per_restart, verbose=False)
        if obj_opt < best_obj - 1e-14:
            best_obj = obj_opt
            best_f = a_opt.copy()
            if verbose:
                print(f"    Deflation restart {r}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW, Nesterov SG, Simulated Annealing,
    aggressive CD, L-BFGS v2, and deflation restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Quick LP-FW (250s) ===
    print("\n=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic() - 1200
    t1 = min(250, max(30, rem))
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # Try reversed
    rem = deadline - time.monotonic() - 1100
    t1r = min(80, max(20, rem))
    if t1r > 20:
        a_rev = parent[::-1].copy()
        a_opt, _ = lp_fw_optimize(a_rev, time_limit=t1r)
        try_seq(a_opt, "lpfw_reversed")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Nesterov SG (400s) ===
    print(f"\n=== Phase 2: Nesterov SG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 800
    t2 = min(400, max(30, rem))
    if t2 > 30:
        a_opt, _ = nesterov_sg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "nsg_main")

    base = best_seq if best_seq is not None else base

    # === Phase 3: Simulated Annealing (350s) ===
    print(f"\n=== Phase 3: SA (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    t3 = min(350, max(30, rem))
    if t3 > 30:
        a_opt, _ = simulated_annealing(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "sa_main")

    base = best_seq if best_seq is not None else base

    # === Phase 4: L-BFGS v2 (300s) ===
    print(f"\n=== Phase 4: L-BFGS v2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    t4 = min(300, max(30, rem))
    if t4 > 30:
        a_opt, _ = lbfqg_optimize_v2(base, time_limit=t4, verbose=True)
        try_seq(a_opt, "lbfqg_v2")

    base = best_seq if best_seq is not None else base

    # === Phase 5: Aggressive CD + Nesterov SG cycle (200s) ===
    print(f"\n=== Phase 5: CD + NSG cycle (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    t5 = min(200, max(20, rem))
    if t5 > 20:
        t5a = t5 // 2
        t5b = t5 - t5a
        a_opt, _ = aggressive_cd(base, time_limit=t5a)
        try_seq(a_opt, "aggr_cd")
        base = best_seq if best_seq is not None else base
        if t5b > 15:
            a_opt, _ = nesterov_sg_optimize(base, time_limit=t5b, verbose=False)
            try_seq(a_opt, "nsg_after_cd")

    base = best_seq if best_seq is not None else base

    # === Phase 6: Deflation restarts (200s) ===
    print(f"\n=== Phase 6: Deflation restarts (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 100
    t6 = min(200, max(20, rem))
    if t6 > 20:
        n_rest = max(2, int(t6 / 50))
        t_per = t6 / n_rest
        a_opt, _ = deflation_restart(base, n_restarts=n_rest,
                                     time_per_restart=t_per, verbose=True)
        try_seq(a_opt, "deflation")

    base = best_seq if best_seq is not None else base

    # === Phase 7: Final polish with remaining time ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempts = 0
        while time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break
            attempts += 1

            if attempts % 3 == 1:
                # LP-FW with perturbation
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem, 50)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_lpfw_{attempts}")
            elif attempts % 3 == 2:
                # Nesterov SG
                t_alloc = min(rem, 40)
                a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_nsg_{attempts}")
            else:
                # CD
                t_alloc = min(rem, 30)
                a_opt, _ = aggressive_cd(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"final_cd_{attempts}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
