Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
1. FISTA with annealed log-sum-exp smoothing (fast, many iterations)
2. LP-dual-guided subgradient (informative direction from LP)
3. Nesterov-accelerated projected gradient on exact max
4. Coordinate descent for final polish
5. Multi-length exploration
Key improvement: replaces most expensive LP-FW iterations with fast
gradient steps guided by periodic LP dual information.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgrad(f, conv_f, weights=None, top_k=50, temp=0.0):
    """
    Compute (weighted) subgradient of max(conv) at f.
    If weights is None, use hard max (uniform over argmax).
    If temp > 0, use soft weighting: w_k ∝ exp((conv[k] - max) / temp)
    If weights is provided, use those weights.
    """
    n = len(f)
    if weights is not None:
        # Use provided weights (should be over all 2n-1 positions)
        # Zero out negligible weights
        w = np.abs(weights)
        w_sum = w.sum()
        if w_sum < 1e-15:
            # Fall back to hard max
            k_max = conv_f.argmax()
            w = np.zeros(len(conv_f))
            w[k_max] = 1.0
        else:
            w = w / w_sum
    elif temp > 0:
        mx = conv_f.max()
        w = np.exp((conv_f - mx) / temp)
        w_sum = w.sum()
        if w_sum < 1e-15:
            k_max = conv_f.argmax()
            w = np.zeros(len(conv_f))
            w[k_max] = 1.0
        else:
            w /= w_sum
    else:
        # Hard max with top-k
        k_act = min(top_k, len(conv_f))
        peaks = np.argpartition(conv_f, -k_act)[-k_act:]
        w = np.zeros(len(conv_f))
        pv = conv_f[peaks]
        # Uniform over top-k
        w[peaks] = 1.0 / len(peaks)
        return _apply_weights(f, w, n)
    
    return _apply_weights(f, w, n)


def _apply_weights(f, w, n):
    """Apply weights to f to get subgradient."""
    conv_len = len(w)
    # Only process non-negligible weights
    threshold = w.max() * 1e-10
    active = np.where(w > threshold)[0]
    if len(active) == 0:
        return np.zeros(n)
    
    sg = np.zeros(n)
    for k in active:
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        idx = np.arange(js, je + 1)
        sg[idx] += wk * f[int(k) - idx]
    return 2.0 * sg


def solve_lp_fw(f, M, n, time_limit=30):
    """Solve the FW linear program. Returns (g, dual_weights) or (None, None)."""
    c = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None:
        # Get dual variables (shadow prices)
        dual = None
        try:
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                dual = res.ineqlin.marginals
        except Exception:
            pass
        return res.x, dual
    return None, None
