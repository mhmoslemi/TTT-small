import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_with_grad(conv, beta):
    """Compute smooth approximation to max and its gradient weights."""
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    obj = conv_max + np.log(exp_s.sum()) / beta
    return obj, w

def grad_smooth_max(a, w):
    """Gradient of smooth_max(conv[a,a]) w.r.t. a.
    d/d a[i] of conv[k] = a[k-i] + a[i-k] (for valid indices)
    So grad[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    """
    n = len(a)
    # Use FFT for cross-correlation
    N = 1 << (3*n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # cross-correlation: sum_k w[k] * a[k-i] = (w * a)[i] in the "correlation" sense
    # Actually: sum_k w[k] * a[k-i] = (IFFT(FFT(w) * conj(FFT(a))))[i]
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using ramp-based initialization
with multi-strategy optimization:
1. Parameterized ramp/power-ramp starting points (E≈1.33-1.39 vs 1.505 parent)
2. Adam-based smooth gradient descent in softmax parameterization with annealed β
3. Peak-directed subgradient with adaptive step sizes
4. Simulated annealing with structural mutations
5. Coordinate descent polish
Key insight: linear ramp a[i]∝(i+1) gives E≈4/3 without optimization,
providing a much better starting point than uniform or parent.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v_shifted = v - v.max()
    e = np.exp(v_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_max_and_weights(conv, beta):
    """Compute log-sum-exp smooth approximation to max and softmax weights."""
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    obj = conv_max + np.log(exp_s.sum()) / beta
    return obj, w


def grad_smooth_max_fft(a, w):
    """Gradient of smooth_max(conv[a,a]) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    if N < 4:
        N = 4
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * R


def softmax_obj_grad(t, beta):
    """Objective and gradient in softmax parameter t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    obj, w = smooth_max_and_weights(conv, beta)
    grad_a = grad_smooth_max_fft(a, w)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def gen_ramp(n, power=1.0):
    """Generate a power ramp: a[i] ∝ (i+1)^power, normalized to sum=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    a = a / a.sum()
    return a


def gen_cosine_weighted(n, alpha=0.5):
    """Generate cosine-weighted sequence: a[i] ∝ 1 + alpha*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        a = 1.0 + alpha * np.cos(np.pi * i / (n - 1))
    else:
        a = np.ones(1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gen_two_bump(n, frac=0.3):
    """Two bumps: high mass at beginning and end, low in middle."""
    a = np.zeros(n)
    a[:int(frac * n)] = 1.0
    a[int((1 - frac) * n):] = 1.0
    a = a + 0.1
    return a / a.sum()


def gen_random_simplex(n, seed=42, Dirichlet_alpha=1.0):
    """Random point on simplex via Dirichlet distribution."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * Dirichlet_alpha)
    return a


def animate_beta(t_total, elapsed, beta_start, beta_end):
    """Exponentially animate beta from start to end over the time budget."""
    progress = min(elapsed / t_total, 1.0)
    return beta_start * (beta_end / beta_start) ** progress


def adam_smooth_optimize(a0, time_limit, beta_start=5.0, beta_end=2000.0,
                         lr_start=0.02, lr_end=0.001, verbose=False):
    """Adam optimizer in softmax t-space with annealed beta."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    eps = 1e-20
    t_param = np.log(a + eps)

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)

        beta = animate_beta(time_limit, elapsed, beta_start, beta_end)
        lr = lr_start * (lr_end / lr_start) ** progress

        obj, grad_t = softmax_obj_grad(t_param, beta)

        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t * grad_t
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        t_param = t_param - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = softmax_fn(t_param)
        actual_obj = fast_convolve(a_new).max()

        if actual_obj < best_obj - 1e-14:
            best_obj = actual_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 300:
            rng = np.random.default_rng(iter_count * 13 + 7)
            scale = min(0.005 + 0.003 * (no_improve // 300), 0.03)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            t_param = np.log(a + eps)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        if verbose and iter_count % 1000 == 0:
            print(f"    AdamSmooth iter {iter_count}: obj={best_obj:.10f}, "
                  f"beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_obj


def peak_directed_optimize(a0, time_limit, num_peaks=10, verbose=False):
    """Peak-directed subgradient: directly reduce the top convolution peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = 0.005
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        lr = 0.005 * (0.3 ** progress)

        conv = fast_convolve(a)
        conv_max = conv.max()

        k_actual = min(num_peaks, len(conv))
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        peak_values = conv[peak_indices]

        temp = conv_max * 0.01 + 1e-12
        scaled = peak_values / temp
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, ki in zip(weights, peak_indices):
            if wi < 1e-12:
                continue
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * wi * a[k - idx]

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        momentum = min(momentum * 1.0005, 0.95)

        if no_improve > 400:
            rng = np.random.default_rng(iter_count * 17 + 1)
            scale = min(0.003 + 0.002 * (no_improve // 400), 0.02)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            momentum = 0.85
            no_improve = 0

        if verbose and iter_count % 500 == 0:
            print(f"    PeakDir iter {iter_count}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def coordinate_descent_optimize(a0, time_limit, verbose=False):
    """Coordinate descent with local line search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_f = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = a.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_sum = f_trial.sum()
                if f_sum < 1e-10:
                    continue
                f_trial /= f_sum
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = a.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def simulated_annealing_optimize(a0, time_limit, T_start=0.01, T_end=0.0001,
                                  verbose=False):
    """Simulated annealing with peak-aware mutations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    cur_obj = best_obj
    cur_a = a.copy()

    t0 = time.monotonic()
    total_time = time_limit

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_time, 1.0)
        T = T_start * (T_end / T_start) ** progress

        conv = fast_convolve(cur_a)
        k_actual = min(5, len(conv))
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]

        if np.random.random() < 0.7:
            # Peak-directed mutation: reduce elements contributing to peaks
            peak_idx = peak_indices[np.random.randint(len(peak_indices))]
            k = int(peak_idx)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = cur_a[idx] * cur_a[k - idx]
            if contrib.sum() > 1e-15:
                # Pick a high-contribution index to reduce
                prob = contrib / contrib.sum()
               reduce_idx = np.random.choice(idx, p=prob)
                # Pick a low-value index to increase
                low_candidates = np.where(cur_a < np.median(cur_a))[0]
                if len(low_candidates) == 0:
                    low_candidates = np.arange(n)
                increase_idx = np.random.choice(low_candidates)
                delta = cur_a[reduce_idx] * 0.1 * T * 100
                cur_a[reduce_idx] -= delta
                cur_a[increase_idx] += delta
        else:
            # Random simplex perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            delta = rng.standard_normal(n) * T
            cur_a = project_simplex(cur_a + delta)

        cur_a = np.maximum(cur_a, 0.0)
        cur_s = cur_a.sum()
        if cur_s < 1e-10:
            cur_a = best_a.copy()
            cur_obj = best_obj
            continue
        cur_a = cur_a / cur_s

        new_obj = fast_convolve(cur_a).max()
        delta_obj = new_obj - cur_obj

        if delta_obj < 0 or np.random.random() < np.exp(-delta_obj / (T + 1e-15)):
            cur_a = cur_a.copy()
            cur_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = cur_a.copy()

    return best_a, best_obj


def lbfgs_smooth_optimize(a0, time_limit, beta_values=None, verbose=False):
    """L-BFGS on smooth objective in t-space with multiple betas."""
    if beta_values is None:
        beta_values = [10.0, 30.0, 100.0, 300.0, 1000.0]

    a = np.maximum(a0.astype(np.float64), 0.0)

    best_a = a.copy()

    t_param = np.log(a + eps)

    t0 = time.monotonic()

    for beta in beta_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        sub_time = remaining / (len(beta_values))

        try:
            res = minimize(
                softmax_obj_grad, t_param.copy(), args=(beta,),
                jac=True, method='L-BFGS-B',

            )
            t_param = res.x

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a
