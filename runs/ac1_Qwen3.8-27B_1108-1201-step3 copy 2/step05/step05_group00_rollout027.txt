def compute_grad(a, p):
    """Compute grad[m] = 2 * sum_l a[l] * p[l+m] for m=0..n-1."""
    n = len(a)

    P = np.fft.rfft(p, fft_len)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed optimization
with Adam and simplex projection, combined with LP-based Frank-Wolfe polishing.
Uses softmax parameterization to avoid projection issues and anneals the smoothing
parameter for increasing precision.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_lse(a, tau):
    """
    Compute gradient of tau*log(sum_k exp((a*a)_k/tau)) w.r.t. a.
    Returns (gradient, smoothed_max_value, max_conv_value).
    """
    n = len(a)
    conv = fast_convolve(a)

    # Softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    p /= p_sum

    # Smoothed max value
    smax = conv.max() + tau * np.log(p_sum * np.exp(scaled.max()))
    # More precisely: tau * (max(conv)/tau + log(sum(exp((conv - max(conv))/tau))))
    smax = conv.max() + tau * np.log(np.sum(np.exp((conv - conv.max()) / tau)))

    # Gradient: grad[m] = 2 * sum_l a[l] * p[l+m] for m=0..n-1
    # This is 2 * correlate(p, a, 'full')[m]
    fft_len = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_len)
    A = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(P * np.conj(A), fft_len)
    grad = 2.0 * corr[:n]

    return grad, smax, conv.max()

    Anneals the smoothing parameter tau over time.

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s

    conv0 = fast_convolve(f)
    tau_init = conv0.max() / 4.0
    tau_min = conv0.max() / 100000.0
    tau = tau_init

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = f.copy()
    best_obj = conv0.max()

    t0 = time.monotonic()
    no_improve = 0
    best_tau_for_restart = tau_init

    while time.monotonic() - t0 < time_limit:
        t += 1

        grad, smax, max_conv = compute_gradient_lse(f, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay_frac=0.15):
    """Create an asymmetric initial sequence (exponentially decaying)."""
    x = np.arange(n)
    center = n * 0.4  # Slightly off-center for asymmetry
    a = np.exp(-0.5 * ((x - center) / (n * decay_frac)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
