"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-domain (softmax) parameterization
with Adam optimization, multi-peak subgradient, spectral peaking, coordinate descent,
and adaptive multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return conv


def softmax_func(b):
    """Stable softmax."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    return exp_b / exp_b.sum()


def obj_from_conv(c, n):
    """Compute objective 2*n*max(c) assuming sum(a)=1."""
    return 2.0 * n * c.max()


def compute_grad_logspace(b, tau=10.0, top_k=40):
    """
    Compute gradient of 2n*max(conv(softmax(b), softmax(b))) w.r.t. b.
    Uses soft-max approximation for the max.
    Returns (grad, a, c, max_c).
    """
    n = len(b)
    a = softmax_func(b)
    c = fast_convolve(a)
    conv_len = len(c)

    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= w_sum

    C_w = np.dot(w, pv)

    # Compute g(m) = sum_l w_l * (a(m+l) + a(m-l))
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        wi = w[i]
        if k < n:
            # a(m+k) valid for m = 0..n-1-k
            g[:n - k] += wi * a[k:]
            # a(m-k) valid for m = k..n-1
            g[k:] += wi * a[:n - k]
        elif k >= n:
            # a(m+k): need m+k < n, but m>=0 and k>=n, so m+k>=n. No contribution.
            # a(m-k): need m-k >= 0, so m >= k >= n. But m < n. No contribution.
            pass

    grad = 2.0 * n * a * (g - 2.0 * C_w)
    return grad, a, c, c.max()


def compute_grad_logspace_fft(b, tau=10.0, top_k=40):
    """
    FFT-based gradient computation for better performance at larger n.
    """
    n = len(b)
    a = softmax_func(b)
    c = fast_convolve(a)
    conv_len = len(c)

    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(c, -k_actual)[-k_actual:]
    pv = c[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= w_sum

    C_w = np.dot(w, pv)

    # Build kernel K of length 2n-1
    K = np.zeros(2 * n - 1)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        if 0 <= k < 2 * n - 1:
            K[k] += w[i]

    # conv(K, a) gives sum_l K[l]*a[m-l] for valid m
    fft_size = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    K_fft = np.fft.rfft(K, fft_size)

    # term2: sum_l w_l * a(m-l) = conv(K, a)[m]
    term2 = np.fft.irfft(K_fft * A_fft, fft_size)[:n]

    # term1: sum_l w_l * a(m+l) = conv(K_flipped, a)[m]
    K_flipped = K[::-1]
    Kf_fft = np.fft.rfft(K_flipped, fft_size)
    term1 = np.fft.irfft(Kf_fft * A_fft, fft_size)[:n]

    g = term1 + term2
    grad = 2.0 * n * a * (g - 2.0 * C_w)
    return grad, a, c, c.max()


def adam_optimize_logspace(b_init, time_limit, verbose=False):
    """
    Adam optimization in log-domain (softmax parameterization).
    Minimize 2n*max(conv(a,a)) where a = softmax(b).
    """
    n = len(b_init)
    b = b_init.copy().astype(np.float64)
    a = softmax_func(b)
    c = fast_convolve(a)
    best_obj = c.max() * 2.0 * n
    best_b = b.copy()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    t = 0
    tau = 10.0
    top_k = 40
    no_improve = 0
    use_fft = n > 512

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Adaptive parameters
        if no_improve > 2000:
            tau = min(100.0, tau * 1.05)
        elif no_improve > 1000:
            tau = max(5.0, tau * 0.98)
            lr = max(0.005, lr * 0.995)
        elif no_improve > 500:
            top_k = min(80, top_k + 5)
        elif no_improve > 200:
            tau = min(50.0, tau * 1.02)
        else:
            lr = min(0.1, lr * 1.001)

        # Compute gradient
        if use_fft:
            grad, a, c, max_c = compute_grad_logspace_fft(b, tau=tau, top_k=top_k)
        else:
            grad, a, c, max_c = compute_grad_logspace(b, tau=tau, top_k=top_k)

        current_obj = 2.0 * n * max_c

        # Adam update
        beta1_t = 1.0 - beta1 ** t
        beta2_t = 1.0 - beta2 ** t
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / beta1_t
        v_hat = v / beta2_t
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_func(b)
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_b = b.copy()
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 4000:
            rng_r = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.1 + 0.05 * min(no_improve // 4000, 5)
            b = best_b + rng_r.standard_normal(n) * noise_scale
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 8.0
            top_k = 30

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {t}: obj={best_obj:.10f}, lr={lr:.5f}, tau={tau:.1f}, "
                  f\"no_imp={no_improve} ({elapsed:.0f}s)\")

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Randomize coordinate order each pass
        rng_p = np.random.default_rng(pass_count * 31)
        order = rng_p.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = 2.0 * n * fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a /= ss
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 2:
            break

    return best_a, best_obj


def spectral_peaking(a_init, time_limit):
    """
    Gerchberg-Saxton style spectral peaking:
    Iteratively reduce non-DC spectral components while maintaining non-negativity.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    damp = 0.95
    it = 0

    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        # Damp non-DC components
        if len(A) > 1:
            A[1:] *= damp
        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new /= ss

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Slowly increase damping
            damp = min(0.99, damp * 1.001)
        else:
            # Reduce damping
            damp = max(0.8, damp * 0.99)
            if it > 100:
                break

    return best_a, best_obj


def lbfqs_logspace(b_init, time_limit, verbose=False):
    """L-BFGS optimization in log-space."""
    n = len(b_init)

    def func_and_grad(b):
        b_shifted = b - b.max()
        exp_b = np.exp(b_shifted)
        a = exp_b / exp_b.sum()
        c = fast_convolve(a)
        obj = 2.0 * n * c.max()

        # Compute gradient
        conv_len = len(c)
        top_k = min(40, conv_len)

        pv = c[peaks]
        tau = 15.0
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        C_w = np.dot(w, pv)
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            wi = w[i]
            if k < n:
                g[:n - k] += wi * a[k:]
                g[k:] += wi * a[:n - k]

        grad = 2.0 * n * a * (g - 2.0 * C_w)
        return obj, grad

    result = scipy_minimize(
        lambda b: func_and_grad(b)[0],
        b_init,

        method='L-BFGS-B',

    )

    b_opt = result.x

    obj_opt = 2.0 * n * c_opt.max()

    if verbose:
        print(f"    L-BFGS: obj={obj_opt:.10f}, success={result.success}, iter={result.niter}")

    return a_opt, obj_opt


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def raised_cosine_init(n):
    \"\"\"Raised cosine initialization.\"\"\"
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    a = np.maximum(a, 0) + 1e-10
    return a / a.sum()


def interpolate_sequence(f, new_n):
    \"\"\"Interpolate a normalized sequence to a new length.\"\"\"
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    \"\"\"
    Multi-phase optimization: log-domain Adam, spectral peaking, L-BFGS,
    coordinate descent, and multi-length exploration.
    \"\"\"
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            a = a * (0.02 / max(a.sum(), 1e-30))
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Log-domain Adam on parent ===
    print(f"\\n=== Phase 1: Log-domain Adam (best={best_score:.6f}) ===")
    if parent is not None:
        n_par = len(parent)
        b_init = np.log(parent + 1e-30)
        t_adam = min(400, (deadline - time.monotonic()) * 0.3)
        print(f"  Adam n={n_par}, t={t_adam:.0f}s")
        a_opt, obj = adam_optimize_logspace(b_init, time_limit=t_adam, verbose=True)
        try_seq(a_opt, "adam_par")

        # L-BFGS refinement
        if best_seq is not None:
            b_best = np.log(best_seq + 1e-30)
            rem = deadline - time.monotonic() - 300
            if rem > 30:
                print(f"  L-BFGS n={n_par}, t={min(rem, 120):.0f}s")
                a_opt, obj = lbfqs_logspace(b_best, time_limit=min(rem, 120), verbose=True)
                try_seq(a_opt, "lbfqs_par")

    # === Phase 2: Spectral peaking ===
    print(f"\\n=== Phase 2: Spectral Peaking (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            a_opt, obj = spectral_peaking(best_seq, time_limit=min(rem, 100))
            try_seq(a_opt, "spec_peak")

            # Adam after spectral peaking
            if best_seq is not None:
                b_init = np.log(best_seq + 1e-30)
                rem2 = deadline - time.monotonic() - 200
                if rem2 > 20:
                    a_opt, obj = adam_optimize_logspace(b_init, time_limit=min(rem2, 80))
                    try_seq(a_opt, "adam_after_spec")

    # === Phase 3: Coordinate descent ===
    print(f"\\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 120))
            try_seq(a_opt, "cd")

    # === Phase 4: Try different initializations ===
    print(f"\\n=== Phase 4: Alternative Initializations (best={best_score:.6f}) ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [512, 640, 768, 896, 1024]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)

    for n_try in target_lengths[:4]:
        if n_try == (len(parent) if parent is not None else -1):
            continue  # Already tried parent length
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break

        # Best initialization for this length
        best_init = None
        best_init_obj = float('inf')
        methods = [
            ("flat", flat_spectrum_init(n_try, rng)),
            ("random", random_init(n_try, rng)),
            ("cosine", raised_cosine_init(n_try)),
        ]
        if parent is not None:
            methods.append(("interp", interpolate_sequence(parent, n_try)))

        for name, init in methods:
            init = np.maximum(init, 0)
            ss = init.sum()
            if ss < 1e-10:
                continue
            init = init / ss
            obj = 2.0 * n_try * fast_convolve(init).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init

        if best_init is not None:
            b_init = np.log(best_init + 1e-30)
            t_alloc = min(rem * 0.3, 100)
            print(f"  n={n_try} (best_init_obj={best_init_obj:.6f}), t={t_alloc:.0f}s")
            a_opt, obj = adam_optimize_logspace(b_init, time_limit=t_alloc)
            try_seq(a_opt, f"alt_n{n_try}")

            if best_seq is not None:
                rem2 = deadline - time.monotonic() - 100
                if rem2 > 15:
                    a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem2, 40))
                    try_seq(a_opt, f"cd_n{n_try}")

    # === Phase 5: Multi-restart Adam ===
    print(f"\\n=== Phase 5: Multi-Restart (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(6):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_alloc = min(rem / (6 - restart), 60)

            if restart == 0:
                f_init = best_seq.copy()
            elif restart == 1:
                noise = rng.standard_normal(n_best) * 0.02
                f_init = np.maximum(best_seq + noise, 0)
                f_init = f_init / f_init.sum()
            elif restart == 2:
                f_init = 0.7 * best_seq + 0.3 * rng.dirichlet(np.ones(n_best))
                f_init = f_init / f_init.sum()
            elif restart == 3:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif restart == 4:
                # Mirror average
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Random perturbation of spectrum
                A = np.fft.rfft(f_init, 1 << (2 * n_best - 1).bit_length())
                phase_noise = rng.standard_normal(len(A)) * 0.1
                A = A * np.exp(1j * phase_noise)
                f_init = np.fft.irfft(A, 1 << (2 * n_best - 1).bit_length())[:n_best]
                f_init = np.maximum(f_init, 0)
                ss = f_init.sum()
                if ss < 1e-10:
                    f_init = best_seq.copy()
                else:
                    f_init /= ss

            b_init = np.log(f_init + 1e-30)
            a_opt, obj = adam_optimize_logspace(b_init, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{restart}")

    # === Phase 6: Final coordinate descent + L-BFGS polish ===
    print(f"\\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            b_init = np.log(best_seq + 1e-30)
            a_opt, obj = lbfqs_logspace(b_init, time_limit=min(rem, 40), verbose=True)
            try_seq(a_opt, "final_lbfqs")

        # Final spectral peaking
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = spectral_peaking(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_spec")

    # Fallback
    if best_seq is None:
        n_fb = 512
        best_seq = flat_spectrum_init(n_fb, rng)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
