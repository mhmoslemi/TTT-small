import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        A_ub[start:end, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv, w


def smooth_optimize(f0, time_limit, T_start=0.5, T_end=0.01, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)
        lr = alpha * (0.3 ** (elapsed / (time_limit * 0.9)))
        
        grad, conv, w = smooth_grad(f, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

            m_state = np.zeros(n)

def coordinate_descent_refine(f0, time_limit, verbose=False):
    """Fine coordinate descent with golden section search on the exact objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    GOLDEN = 0.6180339887
    
    # Precompute which coordinates to try (focus on non-zero ones)
    active = np.where(f > 1e-8)[0]
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        # Try active coordinates in a shuffled order
        rng = np.random.default_rng(pass_count)
        order = active.copy()
        rng.shuffle(order)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            # Golden section search for optimal value at coordinate j
            lo = 0.0
            hi = min(0.999, old_val * 3.0)
            
            def eval_at(x):
                if x < 0 or x >= 1.0:
                    return 1e10
                denom = 1.0 - old_val
                if denom < 1e-10:
                    return 1e10
                scale = (1.0 - x) / denom
                f_trial = f.copy()
                f_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                return fast_convolve(f_trial).max()
            
            c = hi - GOLDEN * (hi - lo)
            d = lo + GOLDEN * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)
            
            for _ in range(12):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - GOLDEN * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + GOLDEN * (hi - lo)
                    fd = eval_at(d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_at(opt_val)
            
            if opt_obj < best_obj - 1e-14:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and pass_count > 5:
            break
    
    return best_f, best_obj


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.001
    return a / a.sum()


def make_super_gaussian(n, p=2.0):
    """exp(-|x|^p) window."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-np.abs(x) ** p)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_bell(n, sigma_rel=0.4):
    """Gaussian bell shape."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * sigma_rel**2))
    return a / a.sum()


def make_flat_with_edges(n, edge_decay=0.3):
    """Flat center with decaying edges."""
    a = np.ones(n)
    edge_len = int(n * edge_decay)
    for i in range(edge_len):
        frac = i / edge_len
        a[i] = frac
        a[n-1-i] = frac
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=1, temperature=0.0):
    """Compute subgradient of max(conv(f,f)) at coordinate j.
    Returns array s of length n where s[j] is the subgradient component."""
    n = len(f)
    s = np.zeros(n)
    js = np.arange(n)
    
    if top_k <= 1:
        k_star = np.argmax(conv)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        s[valid] = 2.0 * f[idx[valid]]
    else:
        # Soft subgradient over top-k peaks with exponential weighting
        top_indices = np.argsort(conv)[::-1][:top_k]
        if temperature > 0:
            raw_w = np.exp(conv[top_indices] / temperature)
        else:
            # Use top-1 with small weight on others
            raw_w = np.zeros(top_k)
            raw_w[0] = 1.0
            for i in range(1, top_k):
                raw_w[i] = 0.1
        raw_w = raw_w / raw_w.sum()
        for k_idx, w in zip(top_indices, raw_w):
            idx = k_idx - js
            valid = (idx >= 0) & (idx < n)
            s[valid] += 2.0 * w * f[idx[valid]]
    
    return s


def mirror_descent_optimize(f0, time_limit, top_k=1, use_averaging=True, verbose=False):
    """
    Mirror descent on simplex using exact subgradient of max(conv(f,f)).
    Uses multiplicative (KL mirror) updates with Friedman averaging.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    # Averaging accumulator
    f_avg = f.copy()
    avg_count = 1
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    c_lr = 1.5  # Learning rate constant
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Step size: c / sqrt(t)
        lr = c_lr / np.sqrt(it + 1)
        
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Check averaged solution
        if use_averaging and it > 100 and it % 100 == 0:
            f_avg_check = f_avg / avg_count
            f_avg_check = f_avg_check / f_avg_check.sum()
            avg_obj = fast_convolve(f_avg_check).max()
            if avg_obj < best_obj - 1e-15:
                best_obj = avg_obj
                best_f = f_avg_check.copy()
                no_improve = 0
        
        # Compute subgradient
        s_grad = compute_subgradient(f, conv, top_k=top_k)
        
        # Mirror descent update: f[j] *= exp(-lr * s_grad[j])
        # Use log-space for numerical stability
        log_f = np.log(f + 1e-30)
        log_f_shifted = log_f - log_f.max()
        log_f_shifted -= lr * s_grad
        log_f_shifted -= log_f_shifted.max()
        f_new = np.exp(log_f_shifted)
        f_new_sum = f_new.sum()
        
        if f_new_sum < 1e-10 or not np.isfinite(f_new_sum):
            # Numerical issue, skip this step
            lr *= 0.5
            continue
            
        f_new = f_new / f_new_sum
        f = f_new
        
        # Update averaging (start after initial transient)
        if it > 50:
            f_avg += f
            avg_count += 1
        
        # Recovery from stagnation
        if no_improve > 3000:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.002)
            f_avg = f.copy()
            avg_count = 1
            no_improve = 0
            it = max(it // 2, 1)  # Reset step counter for larger steps
        
        if verbose and it % 5000 == 0:
            print(f"      MD iter {it}: best={best_obj:.12f}, cur={cur_max:.12f}, lr={lr:.6f}")
    
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and fine line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 20.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 100:
                break
            m_exp = min(m_exp * 1.02, 10.0)
            continue
        
        # Multi-scale line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(18):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 10.0)
        
        if no_improve > 120:
            rng_p = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 100, 5)
            noise = rng_p.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T) via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, conv, w


def smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.012
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)
        T = max(T, T_end)
        lr = alpha * (0.25 ** (elapsed / (time_limit * 0.95)))
        
        grad, conv, w = smooth_grad(f, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 1500:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.004)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj


def cd_eval(f, n, j, x, old_val):
    """Evaluate max conv when f[j]=x, others scaled to maintain sum=1."""
    denom = 1.0 - old_val
    if denom < 1e-10:
        return 1e10
    scale = (1.0 - x) / denom
    f_trial = f.copy()
    f_trial[j] = x
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    f_trial[mask] *= scale
    return fast_convolve(f_trial).max()


def coordinate_descent_refine(f0, time_limit, verbose=False):
    """Fine coordinate descent with golden section search on exact objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    GOLDEN = 0.6180339887
    
    active = np.where(f > 1e-8)[0]
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        rng = np.random.default_rng(pass_count * 7 + 3)
        order = active.copy()
        rng.shuffle(order)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            lo = 0.0
            hi = min(0.999, old_val * 4.0 + 0.01)
            
            c = hi - GOLDEN * (hi - lo)
            d = lo + GOLDEN * (hi - lo)
            fc = cd_eval(f, n, j, c, old_val)
            fd = cd_eval(f, n, j, d, old_val)
            
            for _ in range(14):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - GOLDEN * (hi - lo)
                    fc = cd_eval(f, n, j, c, old_val)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + GOLDEN * (hi - lo)
                    fd = cd_eval(f, n, j, d, old_val)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = cd_eval(f, n, j, opt_val, old_val)
            
            if opt_obj < best_obj - 1e-15:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and pass_count > 8:
            break
    
    return best_f, best_obj


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric (rising-falling) geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.001
    return a / a.sum()


def make_super_gaussian(n, p=2.0):
    """Super-Gaussian window exp(-|x|^p)."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-np.abs(x) ** p)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_bell(n, sigma_rel=0.4):
    """Gaussian bell shape."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2.0 * sigma_rel**2))
    return a / a.sum()


def make_flat_edges(n, edge_frac=0.3):
    """Flat center with linearly decaying edges."""
    a = np.ones(n)
    edge_len = max(1, int(n * edge_frac))
    for i in range(edge_len):
        frac = (i + 1) / (edge_len + 1)
        a[i] = frac
        a[n - 1 - i] = frac
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_state, label):
    """Try a sequence and update best_state if improved.
    best_state is a dict with 'seq', 'score' keys."""
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < best_state['score']:
        old_score = best_state['score']
        best_state['score'] = score
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old_score:.8f}]")


def propose_candidate(seed=42, budget_s=1800, **kwargs):
