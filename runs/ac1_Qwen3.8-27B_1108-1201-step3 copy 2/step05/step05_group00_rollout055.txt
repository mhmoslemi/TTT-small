import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_normalized(a):
    """Compute 2*n*max_conv for a normalized sequence (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def mirror_descent_improved(a0, time_limit, top_k=50, temp_init=0.001, temp_growth=1.002, 
                            lr_init=0.01, lr_decay=0.9999, momentum=0.5, verbose=False):
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = objective_normalized(a)
    best_a = a.copy()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0
    
    # Momentum buffer
    sg_momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M
        
        # Select active peaks
        k_actual = min(top_k, len(conv))
        # Use top k peaks weighted by softmax
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Momentum
        sg_momentum = momentum * sg_momentum + (1 - momentum) * sg
        
        # Multiplicative update
        a_new = a * np.exp(-lr * sg_momentum)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        new_obj = objective_normalized(a_new)
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # If no improvement, try a smaller step
            if no_improve > 50:
                # Halve the learning rate
                lr *= 0.5
                no_improve = 0
        
        lr *= lr_decay
        temp *= temp_growth
        
        if verbose and it % 100 == 0:
            print(f"    MD+ iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, temp={temp:.6f}")
        
        if no_improve > 500:
            break
        if lr < 1e-10:
            break
    
    return best_a, best_obj

            # dc[k]/da[j] = 2*a[k-j] for k in valid range (k != 2j)
            # dc[2j]/da[j] = 2*a[j] (derivative of a[j]^2)
            
            # The "pressure" on a[j] is the gradient of the max:
            # If the max is at k*, the subgradient component for a[j] is 2*a[k*-j]
            # We want to decrease a[j] if this increases the max, and increase it otherwise.
            
            # Simple heuristic: try values around old_val
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.0, old_val * 1.2, old_val * 2.0, 0.0]
            best_candidate = old_val

            # This is expensive to do for each j. Instead, just evaluate the full conv for each candidate.
            
            # For n=768, one convolution is ~O(n log n) ≈ 7680 ops.

            # At 100M ops/s, that's ~0.35s per sweep. We can do ~200 sweeps in 60s.

                a = a / a.sum()
                if best_candidate_obj < best_obj - 1e-14:

                    best_a = a.copy()
                    improved = True

Wait, this is too slow. For each coordinate, we're doing 6 FFT convolutions. With n=768, that's 768*6 = 4608 FFTs per sweep, each of size ~2048. That's probably too slow.

Let me optimize: instead of doing a full FFT for each candidate, I can compute the change in the convolution analytically.

If I change a[j] from old_val to new_val (with renormalization), the change in c[k] is:
- Before renormalization: Δc[k] = (new_val - old_val) * a[k-j] + a[k] * (new_val - old_val) ... no, this isn't right because of the product structure.

Actually, c[k] = sum_{j'} a[j'] * a[k-j']. If I change a[j] from v to v', then:
c'[k] = c[k] - v*a[k-j] - a[k-j]*v + v'*a[k-j] + a[k-j]*v' (if j != k-j)
     = c[k] + (v'-v) * 2*a[k-j] (for k != 2j, where k-j is in [0, n-1])
     = c[k] - v^2 + v'^2 (for k = 2j)

But then we also renormalize: if old sum was S and new sum is S + (v'-v), then:
c''[k] = c'[k] * (S/(S+v'-v))^2

This is getting complicated. Let me just use a different approach.

**Better approach: Use the subgradient directly.**

At the current point a, the subgradient of f(a) = max_k c[k] is:
- If the max is unique at k*: g[j] = 2*a[k*-j] (for j in [k*-n+1, k*])
- If multiple peaks are active: g[j] = sum over active peaks k of w_k * 2*a[k-j]

The mirror descent step is: a_new[j] ∝ a[j] * exp(-eta * g[j])

This is O(n) per step (after computing the subgradient, which is O(n) if we know the active peaks). The expensive part is computing the convolution to find the active peaks, which is O(n log n) per step.

So each iteration is O(n log n). For n=768, that's about 7680 operations. We can do millions of iterations in 60 seconds.

The question is: how to choose eta (step size) well?

**Adaptive step size**: Use the fact that for convex optimization with mirror descent, a good step size is eta = c / sqrt(t) where t is the iteration count. But this is slow to converge.

Better: Use a **line search** or **backtracking line search**.

Or even better: Use the **FTRL (Follow The Regularized Leader)** algorithm, which has a closed-form update for the negative entropy regularizer.

FTRL with negative entropy:
a_new = argmin_{a >= 0, sum(a)=1} <g, a> + H(a) / eta + (1/eta) * D(a, a_old)

where H is the negative entropy and D is the KL divergence. This gives:
a_new[j] ∝ a_old[j] * exp(-eta * (g[j] + running_avg_g[j]))

This is essentially what we're already doing with the multiplicative weights.

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Multi-resolution accelerated multiplicative descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses coarse-to-fine upsample strategy, momentum-accelerated mirror descent with adaptive
step-size, peak-threshold subgradient selection, simulated annealing restarts, and
multi-length exploration to beat the previous best of 1.505008.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    return np.maximum(v - theta, 0.0)

        # Peak selection: all peaks above threshold, capped at top_k
        threshold = threshold_frac * M
        peak_mask = conv >= threshold
        n_above = peak_mask.sum()

        if n_above == 0:
            peaks = np.array([np.argmax(conv)])
        elif n_above > top_k:
            all_peak_idx = np.where(peak_mask)[0]
            all_peak_val = conv[all_peak_idx]
            top_sel = np.argsort(all_peak_val)[-top_k:]
            peaks = all_peak_idx[top_sel]
        else:
            peaks = np.where(peak_mask)[0]

        # Softmax weights over selected peaks
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Momentum update
        sg_mom = momentum * sg_mom + (1.0 - momentum) * sg

        # Multiplicative update
        exp_arg = -lr * sg_mom
        # Clip to avoid overflow
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a_new = a * np.exp(exp_arg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        new_obj = obj_val(a_new)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
            if no_improve > 150:
                # Simulated annealing restart
                rng_local = np.random.default_rng(it * 13 + 7)
                scale = 0.005 + 0.01 * np.sqrt(no_improve / 150.0)
                perturb = rng_local.standard_normal(n) * scale
                a = proj_simplex(a + perturb)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                sg_mom = np.zeros(n)
                no_improve = 0
                lr = min(lr * 2.0, lr_init * 0.8)

        temp *= temp_growth
        momentum = min(momentum * 1.0005, 0.85)

        if verbose and it % 500 == 0:
            print(f"    {tag} iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, temp={temp:.6f}, peaks={len(peaks)}")

        if no_improve > 600:
            break

    return best_a, best_obj


def upsample_seq(a, factor):
    """Upsample sequence by factor using cubic-like interpolation."""
    n = len(a)
    new_n = n * factor
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    # Use spline for smoother interpolation
    a_new = np.interp(new_x, old_x, a, kind='cubic')
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def make_gauss_init(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell_init(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def lr_sweep(a0, time_per_lr=2.0, lr_candidates=None):
    """Quick sweep to find a good learning rate."""
    if lr_candidates is None:
        lr_candidates = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    a_init = np.maximum(a0.astype(np.float64), 1e-15)
    a_init = a_init / a_init.sum()
    best_lr = lr_candidates[0]
    best_obj = obj_val(a_init)
    n = len(a_init)

    for lr_c in lr_candidates:
        a_test = a_init.copy()
        conv = fast_convolve(a_test)
        M = conv.max()
        t_start = time.monotonic()

        # Run a few iterations with this lr
        for _ in range(200):
            if time.monotonic() - t_start > time_per_lr:
                break
            peaks = np.array([np.argmax(conv)])
            k = int(peaks[0])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg = np.zeros(n)
            sg[idx] += 2.0 * a_test[k - idx]
            a_test = a_test * np.exp(-lr_c * np.clip(-sg, -50, 50))
            a_test = np.maximum(a_test, 1e-15)
            a_test = a_test / a_test.sum()
            conv = fast_convolve(a_test)
            M = conv.max()

        test_obj = 2.0 * n * M
        if test_obj < best_obj:
            best_obj = test_obj
            best_lr = lr_c

    return best_lr, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution multiplicative descent with:
    - Coarse-to-fine upsample + reoptimize
    - Adaptive learning rate sweep
    - Momentum-accelerated mirror descent
    - Simulated annealing restarts
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gauss_init(512, 0.3)
        print("No parent found, using Gaussian init n=512")

    # === Phase 1: Quick LR sweep on parent ===
    print("\n=== Phase 1: LR Sweep ===")
    rem = deadline - time.monotonic() - 600
    if rem > 5:
        best_lr, _ = lr_sweep(parent, time_per_lr=1.5)
        print(f"  Best LR from sweep: {best_lr}")
    else:
        best_lr = 0.03

    # === Phase 2: Main MD optimization at parent length ===
    print(f"\n=== Phase 2: MD at n={len(parent)} ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t2 = min(rem, 300)
        print(f"  Time allocated: {t2:.0f}s")
        a_opt, obj = md_core(parent, time_limit=t2, top_k=100,
                             temp_init=0.0005, temp_growth=1.0005,
                             lr_init=best_lr, momentum_init=0.4,
                             verbose=True, tag="P2")
        try_seq(a_opt, "md_parent")

    # === Phase 3: Upsample to 2x and optimize ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Upsample to n={2*len(best_seq)} ===")
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            a_up = upsample_seq(best_seq, 2)
            try_seq(a_up, "upsample2x")
            t3 = min(rem, 250)
            print(f"  MD n={len(a_up)}, t={t3:.0f}s")
            a_opt, obj = md_core(a_up, time_limit=t3, top_k=150,
                                 temp_init=0.0003, temp_growth=1.0003,
                                 lr_init=best_lr * 0.7, momentum_init=0.5,
                                 verbose=True, tag="P3")
            try_seq(a_opt, "md_2x")

    # === Phase 4: Upsample again ===
    if best_seq is not None and len(best_seq) <= 1536:
        print(f"\n=== Phase 4: Upsample to n={2*len(best_seq)} ===")
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            a_up = upsample_seq(best_seq, 2)
            try_seq(a_up, "upsample2x_v2")
            t4 = min(rem, 150)
            print(f"  MD n={len(a_up)}, t={t4:.0f}s")
            a_opt, obj = md_core(a_up, time_limit=t4, top_k=200,
                                 temp_init=0.0002, temp_growth=1.0002,
                                 lr_init=best_lr * 0.5, momentum_init=0.55,
                                 verbose=True, tag="P4")
            try_seq(a_opt, "md_4x")

    # === Phase 5: Fine-tuning with different parameters ===
    print("\n=== Phase 5: Fine-tuning ===")
    if best_seq is not None:
        variants = [
            dict(top_k=300, temp_init=0.00005, temp_growth=1.0002, lr_init=0.005, momentum_init=0.7, tag="F0"),
            dict(top_k=50, temp_init=0.005, temp_growth=1.002, lr_init=0.1, momentum_init=0.2, tag="F1"),
            dict(top_k=150, temp_init=0.001, temp_growth=1.001, lr_init=0.02, momentum_init=0.5, tag="F2"),
        ]
        for vi, params in enumerate(variants):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t5 = min(rem / (len(variants) - vi), 100)
            tag = params.pop('tag')
            print(f"  Variant {vi} ({tag}): t={t5:.0f}s, params={params}")

            if vi == 0:
                init = best_seq.copy()
            elif vi == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.008
                init = proj_simplex(best_seq + noise)
            else:
                init = 0.85 * best_seq + 0.15 * (np.ones(len(best_seq)) / len(best_seq))
                init = init / init.sum()

            a_opt, obj = md_core(init, time_limit=t5, verbose=(vi == 0), **{**params, 'tag': tag})
            try_seq(a_opt, f"fine_{vi}")

    # === Phase 6: Fresh starts at different lengths ===
    print("\n=== Phase 6: Fresh starts ===")
    fresh_lengths = [512, 1024, 1536]
    for ni, n_try in enumerate(fresh_lengths):
        rem = deadline - time.monotonic() - 40
        if rem < 20:
            break
        t6 = min(rem / (len(fresh_lengths) - ni), 120)

        # Try both Gaussian and Bell initializations
        a_g = make_gauss_init(n_try, 0.35)
        a_b = make_bell_init(n_try, 0.7)

        a_opt_g, obj_g = md_core(a_g, time_limit=t6 * 0.5, top_k=100,
                                  temp_init=0.002, temp_growth=1.001,
                                  lr_init=0.08, momentum_init=0.3, tag=f"FG{n_try}")
        try_seq(a_opt_g, f"gauss_n{n_try}")

        a_opt_b, obj_b = md_core(a_b, time_limit=t6 * 0.5, top_k=100,
                                  temp_init=0.002, temp_growth=1.001,
                                  lr_init=0.08, momentum_init=0.3, tag=f"FB{n_try}")
        try_seq(a_opt_b, f"bell_n{n_try}")

    # === Phase 7: Iterative restart + polish on best ===
    if best_seq is not None:
        print("\n=== Phase 7: Restart + Polish ===")
        for ri in range(4):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t7 = min(rem / (5 - ri), 60)

            if ri == 0:
                init = best_seq.copy()
            elif ri == 1:
                noise = rng.standard_normal(len(best_seq)) * 0.003
                init = proj_simplex(best_seq + noise)
            elif ri == 2:
                # Symmetrize
                init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                init = init / init.sum()
            else:
                # Blend with uniform
                n = len(best_seq)
                init = 0.9 * best_seq + 0.1 * (np.ones(n) / n)
                init = init / init.sum()

            a_opt, obj = md_core(init, time_limit=t7, top_k=200,
                                  temp_init=0.0001, temp_growth=1.0003,
                                  lr_init=0.01, momentum_init=0.6,
                                  verbose=False, tag=f"R{ri}")
            try_seq(a_opt, f"restart_{ri}")

    # === Phase 8: Final ultra-fine polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 40)
            print(f"\n=== Phase 8: Final Polish (t={t8:.0f}s) ===")
            a_opt, obj = md_core(best_seq, time_limit=t8, top_k=500,
                                  temp_init=0.00001, temp_growth=1.0001,
                                  lr_init=0.002, momentum_init=0.8,
                                  verbose=True, tag="FINAL")
            try_seq(a_opt, "final_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
