"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe optimization
with subgradient refinement and coordinate descent polishing.
Key innovation: Uses the LP from the inequality paper to find optimal search
directions, combined with adaptive subgradient steps and local search.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_weights(conv, tau, top_k=None):
    """Compute softmax weights for subgradient approximation."""
    if top_k is None or top_k >= len(conv):
        k = len(conv)
        vals = conv
        peaks = np.arange(k)
    else:
        k = top_k
        peaks = np.argpartition(conv, -k)[-k:]
        vals = conv[peaks]
    
    scaled = tau * (vals - vals.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    return w, peaks


def compute_subgradient(a, tau, top_k=200):
    """Compute smooth subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    w, peaks = compute_subgrad_weights(conv, tau, top_k)
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    
    return grad, conv


def solve_lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0.
    Returns (g, optimal_sum, dual_weights) or (None, None, None) on failure.
    """
    n = len(a)
    c = -np.ones(n)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            optimal_sum = -result.fun
            # Get dual variables (shadow prices)
            dual = result.ineqlin.marginals  # These are the y_k values
            return g, optimal_sum, dual
        else:
            return None, None, None
    except Exception:
        return None, None, None


def lp_frank_wolfe_step(a, M, t):
    """
    Perform one Frank-Wolfe step using the LP direction.
    Returns new point or None if LP failed.
    """
    g, s, _ = solve_lp_direction(a, M)
    if g is None or s <= 1.0 + 1e-10:
        return None
    # Normalize g to sum to 1
    g_norm = g / s
    # Move from a toward g_norm
    a_new = (1 - t) * a + t * g_norm
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss < 1e-15:
        return None
    a_new /= ss
    return a_new


def optimize_lp_subgrad_hybrid(a0, time_limit, verbose=True):
    """
    Hybrid optimizer: alternates between subgradient steps and LP-based
    Frank-Wolfe steps, with adaptive parameters.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    tau = 100.0
    lr = 0.05
    momentum = 0.8
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    lp_interval = max(5, n // 50)  # Solve LP every LP_INTERVAL iterations
    fw_step_size = 0.3
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        
        # Periodic LP-based Frank-Wolfe step
        if iter_count % lp_interval == 0:
            conv = fast_convolve(a)
            M = conv.max()
            a_new = lp_frank_wolfe_step(a, M, t=fw_step_size)
            if a_new is not None:
                obj_new = fast_convolve(a_new).max()
                if obj_new < best_obj:
                    a = a_new
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                    if verbose:
                        print(f"    [LP-FW] iter={iter_count}: obj={best_obj:.10f} ({elapsed:.0f}s)")
                else:
                    # Try smaller step
                    a_new2 = lp_frank_wolfe_step(a, M, t=fw_step_size * 0.3)
                    if a_new2 is not None:
                        obj_new2 = fast_convolve(a_new2).max()
                        if obj_new2 < best_obj:
                            a = a_new2
                            best_obj = obj_new2
                            best_a = a.copy()
                            no_improve = 0
            no_improve += 1
            continue
        
        # Subgradient step
        # Adaptive parameters
        if no_improve > 300:
            momentum = min(0.95, momentum + 0.003)
            lr = max(0.002, lr * 0.98)
        elif no_improve > 100:
            tau = min(300.0, tau * 1.02)
        elif no_improve > 20:
            tau = max(30.0, tau * 0.99)
        
        grad, conv = compute_subgradient(a, tau, top_k=min(300, 2*n-1))
        grad_centered = grad - grad.mean()
        
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj:
            # Try to extend
            a_ext = project_simplex(a + 2.0 * v, target_sum=1.0)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext
        
        a = a_new
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iter_count % 100 == 0:
            print(f"    [SG] iter={iter_count}: obj={best_obj:.10f}, lr={lr:.4f}, tau={tau:.0f}, no_imp={no_improve} ({elapsed:.0f}s)")
        
        # Restart if stuck
        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 17 + 3)
            noise = rng_r.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 80.0
            momentum = 0.8
    
    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit):
    """Fast coordinate descent: for each coordinate, optimize its value analytically."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Try scaling factor: change a[j] while keeping sum=1
            # If we change a[j] to new_val, scale all others by (1-new_val)/(1-old_val)
            best_j = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.75, 1.25, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest_sum
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j = new_val
            
            if best_j != old_val:
                rest_sum = 1.0 - old_val
                if rest_sum > 1e-10:
                    scale = (1.0 - best_j) / rest_sum
                    a[j] = best_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def local_pair_search(a0, time_limit, seed=42):
    """Local search: try small transfers between pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        improved_count = 0
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            for frac in [0.1, 0.2, 0.3]:
                transfer = frac * min(a[i], a[j])
                for direction in [1, -1]:
                    new_i = a[i] + direction * transfer
                    new_j = a[j] - direction * transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved_count += 1
        
        if improved_count == 0:
            break
    
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe optimization
combined with adaptive subgradient steps, coordinate descent, and local search.
Key innovation: Solves the LP from the inequality paper at each Frank-Wolfe step
to find optimal search directions, achieving faster convergence than pure subgradient.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, tau, top_k=300):
    """Compute smooth subgradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad, conv


def build_lp_matrix(a):
    """Build sparse constraint matrix for the LP: conv(a,g)_k <= M."""
    n = len(a)
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    return A


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)_k <= M for all k, g >= 0.
    Returns (g, optimal_sum) or (None, None).
    """
    n = len(a)
    c = -np.ones(n)
    num_constraints = 2 * n - 1
    A = build_lp_matrix(a)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            return result.x, -result.fun
    except Exception:
        pass
    return None, None


def lp_frank_wolfe_step(a, M, t):
    """
    One Frank-Wolfe step using LP direction.
    Returns (new_a, lp_sum) or (None, None).
    """
    g, s = solve_lp_direction(a, M)
    if g is None or s <= 1.0 + 1e-8:
        return None, None
    g_norm = g / s
    a_new = (1.0 - t) * a + t * g_norm
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return None, None
    a_new /= ss
    return a_new, s

def coordinate_descent(a0, time_limit):
    """Coordinate descent with 4 multiplier trials per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s_sum = a.sum()
    if s_sum < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s_sum
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.75, 1.33, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                rest = 1.0 - old_val
                if rest < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j = new_val
            if best_j != old_val:
                rest = 1.0 - old_val
                if rest > 1e-10:
                    scale = (1.0 - best_j) / rest
                    a[j] = best_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def local_pair_search(a0, time_limit, seed=42):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s_sum = a.sum()
    if s_sum < 1e-15:

    support = np.where(a > 1e-8)[0]

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:

            if i == j or a[i] < 1e-10 or a[j] < 1e-10:

            for frac in [0.1, 0.2, 0.35]:
                transfer = frac * min(a[i], a[j])

def flat_spectrum_init(n, rng):
    """Flat cyclic spectrum initialization."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s_sum = a.sum()
    if s_sum < 1e-10:
        return np.ones(n) / n
    return a / s_sum


def random_init(n, rng):

def rectangular_with_notch(n, rng):

def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = random_init(768, rng)
        p_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        best_score = p_score
        print(f"Random init: n={len(parent)}, score={p_score:.6f}")

    n_parent = len(parent)
    print(f"\nStarting optimization. Parent score: {best_score:.6f}, n={n_parent}")
    print(f"Deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Main LP-FW + Subgradient on parent length ===
    print(f"\n=== Phase 1: LP-FW Optimization n={n_parent} ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t1 = min(rem * 0.5, 500)
        print(f"  t={t1:.0f}s")
        a_opt, obj = optimize_lp_subgrad(parent, time_limit=t1, verbose=True)
        best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, f"lpfw_n{n_parent}")

    # === Phase 2: Subgradient refinement with perturbations ===
    print(f"\n=== Phase 2: Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = [
            ("direct", best_seq.copy()),
            ("noise1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003, target_sum=1.0)),
            ("mix_uniform", (0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)),
        ]
        for p_idx, (p_name, f_init) in enumerate(perturbations):
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem / (len(perturbations) - p_idx), 150)
            print(f"  [{p_name}] n={n_best} t={t_alloc:.0f}s")
            a_opt, obj = optimize_lp_subgrad(f_init, time_limit=t_alloc, verbose=(p_idx == 0))
            best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, f"refine_{p_name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        candidate_ns = [n_best]
        for n_try in [512, 640, 896, 1024, 1280, 1536]:
            if n_try != n_best:
                candidate_ns.append(n_try)
        # Sort by proximity to current best
        candidate_ns.sort(key=lambda x: abs(x - n_best))

        for n_try in candidate_ns[:4]:
            if n_try == n_best:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem * 0.25, 150)
            f_interp = interpolate_sequence(best_seq, n_try)
            # Also try a fresh initialization at this length
            f_alt = flat_spectrum_init(n_try, rng)
            obj_interp = fast_convolve(f_interp).max()
            obj_alt = fast_convolve(f_alt).max()
            if obj_alt < obj_interp:
                f_interp = f_alt
            print(f"  n={n_try} t={t_alloc:.0f}s (init obj={fast_convolve(f_interp).max():.6f})")
            a_opt, obj = optimize_lp_subgrad(f_interp, time_limit=t_alloc, verbose=False)
            best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, f"len_n{n_try}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.35, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, "cd1")
            # Second pass
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_opt2, obj2 = coordinate_descent(a_opt, time_limit=min(rem, 60))
                best_seq, best_score, _ = try_evaluate(a_opt2, best_seq, best_score, "cd2")

    # === Phase 5: Pair swap search ===
    print(f"\n=== Phase 5: Pair Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_ps = min(rem * 0.4, 80)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = local_pair_search(best_seq, time_limit=t_ps, seed=seed)
            best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, "pair")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_final = min(rem, 120)
            print(f"  Final t={t_final:.0f}s")
            a_opt, obj = optimize_lp_subgrad(best_seq, time_limit=t_final, verbose=True)
            best_seq, best_score, _ = try_evaluate(a_opt, best_seq, best_score, "final_lp")

            # Quick CD after final LP
            rem = deadline - time.monotonic() - 10
            if rem > 10 and best_score < 1.506:
                a_opt2, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
                best_seq, best_score, _ = try_evaluate(a_opt2, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
