def block_cd(f0, time_limit, block_size=2, verbose=False):
    """Block coordinate descent with direct evaluation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_blocks = min(200 * (n // 100 + 1), 2000)
        
        for _ in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Select a random block of coordinates
            if block_size == 1:
                j = int(rng.integers(n))
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                
                # Try several values
                best_x = old_val
                best_local = best_obj
                for frac in [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                    x_new = old_val * frac
                    if x_new < 0 or x_new >= 0.99:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - x_new) / denom
                    f_trial = f.copy()
                    f_trial[j] = x_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    val = fast_convolve(f_trial).max()
                    if val < best_local - 1e-14:
                        best_local = val
                        best_x = x_new
                
                if best_x != old_val:
                    denom = 1.0 - old_val
                    scale = (1.0 - best_x) / denom
                    f[j] = best_x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_local < best_obj - 1e-13:
                        best_obj = best_local
                        best_f = f.copy()
                        improved = True
            
            else:
                # Block of size 2
                idx = rng.integers(n, size=block_size)
                if idx[0] == idx[1]:
                    continue
                i, j = int(idx[0]), int(idx[1])
                old_i, old_j = f[i], f[j]
                total_ij = old_i + old_j
                if total_ij < 1e-12:
                    continue
                
                best_ratio = old_i / total_ij
                best_local = fast_convolve(f).max()
                
                for ratio in np.linspace(0.05, 0.95, 19):
                    if abs(ratio - best_ratio) < 0.03:
                        continue
                    f_trial = f.copy()
                    f_trial[i] = ratio * total_ij
                    f_trial[j] = (1.0 - ratio) * total_ij
                    val = fast_convolve(f_trial).max()
                    if val < best_local - 1e-14:
                        best_local = val
                        best_ratio = ratio
                
                if abs(best_ratio - old_i / total_ij) > 1e-6:
                    f[i] = best_ratio * total_ij
                    f[j] = (1.0 - best_ratio) * total_ij
                    if best_local < best_obj - 1e-13:
                        best_obj = best_local
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def lp_direction_fast(f, M, n):
    """Solve LP for Frank-Wolfe direction using structured constraints."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    # Build the constraint matrix: (f*g)[k] = sum_j f[k-j]*g[j] <= M
    for j in range(n):
        lo = max(0, j - (n - 1))
        hi = min(2 * n - 2, j + n - 1)
        A_ub[lo:hi+1, j] = f[lo-j : hi-j+1]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction_fast(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            m_exp = 1.0
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 100:
            rng_p = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()

def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    return a / a.sum()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Multi-stage optimizer combining mirror descent, FISTA with log-sum-exp smoothing,
block coordinate descent, and LP Frank-Wolfe for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: multiplicative weight updates for fast initial progress,
adaptive temperature scheduling, and aggressive final polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, T):
    """Compute gradient of T*log(sum(exp(conv(f,f)/T))) using FFT cross-correlation."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled_max = scaled.max()
    e = np.exp(scaled - scaled_max)
    e_sum = e.sum()
    w = e / e_sum

    # corr[j] = sum_{m=0}^{n-1} f[m] * w[m+j]
    # Use: corr[j] = conv(f[::-1], w)[j + n - 1]
    N = 1 << (n + (2 * n - 1) - 1).bit_length()
    fr = f[::-1]
    FR = np.fft.rfft(fr, N)
    WR = np.fft.rfft(w, N)
    conv_r = np.fft.irfft(FR * WR, N)
    corr = conv_r[n - 1: n - 1 + n]

    grad = 2.0 * corr
    lse_val = T * (scaled_max + np.log(e_sum))
    return grad, lse_val, conv, w


def fista_smooth(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """FISTA accelerated gradient on log-sum-exp smooth approximation with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    y = f.copy()
    t_k = 1.0
    f_prev = f.copy()

    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1e-6), 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        T = max(T, T_end)

        # Step size: conservative estimate of 1/L
        step = T / 4.0

        grad_y, lse_y, conv_y, w_y = compute_lse_gradient(y, T)

        # Gradient step + projection
        x_new = y - step * grad_y
        x_new = project_simplex(x_new)

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - f_prev)
        y = project_simplex(y)
        t_k = t_new
        f_prev = x_new.copy()
        f = x_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 300 == 0:

        else:
            no_improve += 1

        if no_improve > 1500:

            scale = 0.002 + 0.002 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng_r.standard_normal(n) * scale)
            y = f.copy()
            f_prev = f.copy()
            t_k = 1.0
            no_improve = 0

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) with subgradient from active set."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        eta = 0.5 / (np.sqrt(it + 1) + 1.0)

        conv = fast_convolve(f)
        max_val = conv.max()
        # Active set: lags within 5% of max
        active = conv >= max_val * 0.95
        w = active.astype(np.float64) / active.sum()

        # Compute subgradient: grad[j] = 2 * sum_k w[k] * f[k-j]
        N = 1 << (n + (2 * n - 1) - 1).bit_length()
        fr = f[::-1]
        FR = np.fft.rfft(fr, N)
        WR = np.fft.rfft(w, N)
        conv_r = np.fft.irfft(FR * WR, N)
        corr = conv_r[n - 1: n - 1 + n]
        grad = 2.0 * corr

        # Multiplicative update
        log_f = np.log(np.maximum(f, 1e-30))
        log_f_new = log_f - eta * grad
        # Subtract max for numerical stability
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum

        # Check if we improved
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    Mirror iter {it}: obj={best_obj:.10f}, eta={eta:.6f}")
        else:
            no_improve += 1
            # Try smaller step
            eta_half = eta * 0.5
            log_f_new2 = log_f - eta_half * grad
            log_f_new2 -= log_f_new2.max()
            f_check = np.exp(log_f_new2)
            f_check /= f_check.sum()
            obj_check = fast_convolve(f_check).max()
            if obj_check < obj_new:
                f = f_check

        if no_improve > 800:
            rng_r = np.random.default_rng(it * 23 + 11)
            f = project_simplex(best_f + rng_r.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def block_coordinate_descent(f0, time_limit, block_size=2, verbose=False):
    """Block coordinate descent with ratio search for pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_trials = min(300 * (n // 100 + 1), 3000)

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            if block_size == 1:
                j = int(rng.integers(n))
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue

                best_x = old_val
                best_local = best_obj
                # Try multiplicative factors
                for factor in [0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]:
                    x_new = old_val * factor
                    if x_new < 0 or x_new > 0.99:
                        continue
                    scale = (1.0 - x_new) / denom
                    f_trial = f.copy()
                    f_trial[j] = x_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    val = fast_convolve(f_trial).max()
                    if val < best_local - 1e-14:
                        best_local = val
                        best_x = x_new

                if best_x != old_val:
                    scale = (1.0 - best_x) / denom
                    f[j] = best_x
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / (f.sum() + 1e-30)
                    if best_local < best_obj - 1e-13:
                        best_obj = best_local
                        best_f = f.copy()
                        improved = True

            else:
                # Pair
                idx = rng.integers(n, size=2)
                i, j = int(idx[0]), int(idx[1])
                if i == j:
                    continue
                old_i, old_j = f[i], f[j]
                total_ij = old_i + old_j
                if total_ij < 1e-12:
                    continue

                cur_ratio = old_i / total_ij
                best_ratio = cur_ratio
                best_local = best_obj

                ratios = np.concatenate([
                    np.linspace(0.0, 0.5, 11),
                    np.linspace(0.5, 1.0, 11)
                ])
                for ratio in ratios:
                    if abs(ratio - cur_ratio) < 0.04:
                        continue
                    f_trial = f.copy()
                    f_trial[i] = ratio * total_ij
                    f_trial[j] = (1.0 - ratio) * total_ij
                    val = fast_convolve(f_trial).max()
                    if val < best_local - 1e-14:
                        best_local = val
                        best_ratio = ratio

                if abs(best_ratio - cur_ratio) > 1e-6:
                    f[i] = best_ratio * total_ij
                    f[j] = (1.0 - best_ratio) * total_ij
                    if best_local < best_obj - 1e-13:
                        best_obj = best_local
                        best_f = f.copy()
                        improved = True

        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes > 5:
                break

    return best_f, best_obj


def build_lp_matrix(n):
    """Build the (2n-1) x n constraint matrix for the LP direction."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = np.arange(n)  # placeholder, will be filled with f values
    return A


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            continue

def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def make_cosine(n, power=2.0):
    """Raised cosine window."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_symmetric_geo(n, r):
    """Symmetric rising-falling geometric."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    return a / a.sum()


def make_tapered(n, inner_frac=0.7, taper_power=1.5):
    """Sequence that's flat in the middle with tapered edges."""
    a = np.ones(n)
    edge = int(n * (1 - inner_frac) / 2)
    if edge > 0:
        for i in range(edge):
            frac = i / edge
            a[i] = frac ** taper_power
            a[n - 1 - i] = frac ** taper_power
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            _try(make_tapered(n_try, inner_frac=tf), f"taper_{tf}_n{n_try}")

    print(f"  After Phase 0: best={best_score:.6f} (n={len(best_seq)})")

    # === Phase 1: Mirror descent (fast, good for initial progress) ===
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.15, 200)
    if t1 > 20:
        print(f"  Mirror descent n={len(best_seq)}, t={t1:.0f}s")
        a_opt, _ = mirror_descent(best_seq, time_limit=t1, verbose=True)
        _try(a_opt, "mirror")

    # === Phase 2: Adam smooth optimization ===
    print(f"\n=== Phase 2: Adam Smooth (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350)
    if t2 > 30:
        print(f"  Adam n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = adam_smooth(best_seq, time_limit=t2,
                               T_start=0.4, T_end=0.008, verbose=True)
        _try(a_opt, "adam_main")

    # Second Adam with lower T
    rem = deadline - time.monotonic()
    t2b = min(rem * 0.12, 150)
    if t2b > 20:
        print(f"  Adam lowT n={len(best_seq)}, t={t2b:.0f}s")
        a_opt, _ = adam_smooth(best_seq, time_limit=t2b,
                              T_start=0.1, T_end=0.003)
        _try(a_opt, "adam_lowT")

    # === Phase 3: FISTA smooth ===
    print(f"\n=== Phase 3: FISTA Smooth (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200)
    if t3 > 20:
        print(f"  FISTA n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = fista_smooth(best_seq, time_limit=t3,
                                T_start=0.2, T_end=0.005, verbose=True)
        _try(a_opt, "fista")

    # === Phase 4: Block coordinate descent ===
    print(f"\n=== Phase 4: Block CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 250)
    if t4 > 20:
        print(f"  Block CD (pairs) n={len(best_seq)}, t={t4:.0f}s")
        a_opt, _ = block_coordinate_descent(best_seq, time_limit=t4, block_size=2)
        _try(a_opt, "cd_pairs")

    # Single coordinate
    rem = deadline - time.monotonic()
    t4b = min(rem * 0.10, 100)
    if t4b > 15:
        a_opt, _ = block_coordinate_descent(best_seq, time_limit=t4b, block_size=1)
        _try(a_opt, "cd_single")

    # === Phase 5: LP-FW ===
    print(f"\n=== Phase 5: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    if n_cur <= 1024:
        t5 = min(rem * 0.15, 200)
        if t5 > 20:
            print(f"  LP-FW n={n_cur}, t={t5:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t5, verbose=True)
            _try(a_opt, "lpfw")
    else:
        print(f"  Skipping LP-FW (n={n_cur} > 1024)")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidates = sorted(set([n_best - 64, n_best - 128, n_best + 64,
                            n_best + 128, n_best + 256, 896, 1024]))
    for target_n in candidates:
        if target_n < 100 or target_n > 2500:
            continue
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        t6 = min(rem * 0.15, 120)
        if t6 < 15:
            continue
        a_opt, _ = adam_smooth(f_interp, time_limit=t6,
                              T_start=0.2, T_end=0.01)
        _try(a_opt, f"adam_n{target_n}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.30, 150)
        a_opt, _ = adam_smooth(best_seq, time_limit=t7,
                              T_start=0.04, T_end=0.002, verbose=True)
        _try(a_opt, "adam_fine")

    rem = deadline - time.monotonic()
    if rem > 20:
        t7b = min(rem * 0.25, 80)
        a_opt, _ = block_coordinate_descent(best_seq, time_limit=t7b, block_size=2)
        _try(a_opt, "cd_fine")

    rem = deadline - time.monotonic()
    if rem > 15 and len(best_seq) <= 1024:
        t7c = min(rem * 0.30, 60)
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t7c)
        _try(a_opt, "lpfw_fine")

    # Perturbation restarts
    for trial in range(4):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        noise_scale = 0.001 + 0.001 * trial
        noise = rng.standard_normal(len(best_seq)) * noise_scale
        f_pert = project_simplex(best_seq + noise)
        t8 = min(rem * 0.25, 30)
        a_opt, _ = adam_smooth(f_pert, time_limit=t8,
                              T_start=0.03, T_end=0.002)
        _try(a_opt, f"perturb_{trial}")

    # FISTA fine
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.4, 40)
        a_opt, _ = fista_smooth(best_seq, time_limit=t9,
                                T_start=0.03, T_end=0.003, verbose=True)
        _try(a_opt, "fista_fine")

    # Blend with uniform
    rem = deadline - time.monotonic()
    if rem > 10:
        f_blend = 0.9 * best_seq + 0.1 * (np.ones(len(best_seq)) / len(best_seq))
        f_blend = project_simplex(f_blend)
        t10 = min(rem * 0.5, 20)
        a_opt, _ = adam_smooth(f_blend, time_limit=t10,
                              T_start=0.02, T_end=0.002)
        _try(a_opt, "blend")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
