def fast_convolve(a):

def project_simplex(v, target_sum=1.0):
    """Project onto simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(a, tau, top_k=None):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k is not None and top_k < len(conv):
        # Only consider top_k peaks
        idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_top = conv[idx]
        conv_shifted = conv_top - conv_top.max()
        w_top = np.exp(tau * conv_shifted)
        w_sum = w_top.sum()
        smooth_val = (tau * conv_top.max() + np.log(w_sum)) / tau
        W = np.zeros(len(conv))
        W[idx] = w_top / w_sum
    else:
        conv_shifted = conv - conv.max()
        w_vals = np.exp(tau * conv_shifted)
        w_sum = w_vals.sum()
        smooth_val = (tau * conv.max() + np.log(w_sum)) / tau
        W = w_vals / w_sum
    # Gradient: 2 * correlate(W, a, 'valid')
    grad = 2.0 * np.correlate(W, a, mode='valid')
    return smooth_val, grad, conv


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def triangular_fit_init(n, rng):
    """Triangular fit: optimize phases to make IFFT as close to triangular as possible."""
    target = np.maximum(1.0 - 2.0 * np.abs(np.arange(n) / (n - 1) - 0.5), 0)
    target = target / target.sum()
    # Random phases with target magnitude profile
    phases = rng.uniform(0, 2 * np.pi, n)
    mags = np.abs(np.fft.fft(target))
    mags[0] = 0
    for trial in range(20):
        A = mags * np.exp(1j * phases)
        A = (A + A[::-1].conj()) / 2  # enforce Hermitian symmetry
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        if fast_convolve(a).max() < fast_convolve(target).max():
            target = a
        phases = rng.uniform(0, 2 * np.pi, n)
        mags = np.abs(np.fft.fft(target))
        mags[0] = 0
    return target

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M):
    """Solve LP to find direction g that maximizes sum(g) s.t. g>=0, conv(f,g)<=M."""
    n = len(f)
    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # For efficiency, build as sparse-like (but use dense for n up to ~1000)
    # A has shape (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For coordinate j, it appears in conv[k] for k in [j, j+n-1] with coefficient f[k-j]
        A[j:j+n, j] = f[0:n]

    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    # Constraints: A @ g <= M (for all 2n-1 lags), g >= 0
    b = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': 10})

    if res.status == 0:
        g = res.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
        else:
            return f.copy()
    else:
        return f.copy()

def simulated_annealing(a0, time_limit, seed=0, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

    T_min = 1e-6
    cooling = 0.9995

        if ss < 1e-10:

        if verbose and total_steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA step {total_steps}: obj={obj:.10f}, best={best_obj:.10f}, T={T:.6f} ({elapsed:.0f}s)")

    return best_a, best_obj

            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Initialization (current best={best_score:.6f}) ===")

    # Try the parent at its native length and nearby lengths
    target_lengths = set()
    if parent is not None:
        target_lengths.add(parent_n)
        for delta in [-64, -32, 32, 64]:
            target_lengths.add(parent_n + delta)
    for n_try in [512, 640, 768, 896, 1024]:
        target_lengths.add(n_try)
    target_lengths = sorted([n for n in target_lengths if 256 <= n <= 2048])

    inits = {}
    for n in target_lengths[:6]:
        candidates = []

        # Flat spectrum
        for trial in range(3):
            init = flat_spectrum_init(n, rng)
            candidates.append(init)

        # Gaussian
        for center in [0.3, 0.5, 0.7]:
            for width in [0.1, 0.15, 0.25]:
                candidates.append(gaussian_init(n, center, width))

        # Random
        for _ in range(2):
            candidates.append(random_init(n, rng))

        # Triangular fit
        candidates.append(triangular_fit_init(n, rng))

        # Interpolated parent
        if parent is not None:
            candidates.append(interpolate_sequence(parent, n))

        # Uniform
        candidates.append(uniform_init(n))

        # Pick best
        best_init = None
        best_init_obj = float('inf')
        for init in candidates:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = fast_convolve(init).max()
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init.copy()
            except Exception:
                continue

        if best_init is not None:
            inits[n] = (best_init, best_init_obj)
            print(f"  n={n}: best init obj={best_init_obj:.6f}")

    # === Phase 2: Smooth optimization on top initializations ===
    print(f"\n=== Phase 2: Smooth Optimization (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: x[1][1])
    n_optimize = min(4, len(sorted_inits))

    for rank, (n_len, (init, init_obj)) in enumerate(sorted_inits[:n_optimize]):
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        time_fracs = [0.4, 0.3, 0.2, 0.1]
        time_alloc = min(rem * time_fracs[rank], 300)
        if time_alloc < 10:
            time_alloc = 10
        print(f"  Smooth opt n={n_len} (init_obj={init_obj:.6f}), t={time_alloc:.0f}s")
        a_opt, obj = optimize_smooth(init, time_limit=time_alloc, verbose=(rank == 0))
        try_seq(a_opt, f"smooth_n{n_len}_r{rank}")

    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Run LP on the best sequence
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_lp = min(rem * 0.6, 200)
            print(f"  LP n={n_best}, t={t_lp:.0f}s")
            a_opt, obj = lp_iterate(best_seq, time_limit=t_lp, verbose=True)
            try_seq(a_opt, f"lp_n{n_best}")

        # Also try LP on a different length if time allows
        rem = deadline - time.monotonic() - 300
        if rem > 30 and len(sorted_inits) > 1:
            n_alt = sorted_inits[1][0]
            if n_alt != n_best:
                f_alt = interpolate_sequence(best_seq, n_alt)
                t_lp2 = min(rem * 0.5, 100)
                print(f"  LP n={n_alt}, t={t_lp2:.0f}s")
                a_opt, obj = lp_iterate(f_alt, time_limit=t_lp2)
                try_seq(a_opt, f"lp_n{n_alt}_alt")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_cd = min(rem * 0.7, 250)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = fast_coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 5: Simulated annealing ===
    print(f"\n=== Phase 5: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sa = min(rem * 0.5, 120)
            print(f"  SA t={t_sa:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t_sa, seed=seed + 100, verbose=True)
            try_seq(a_opt, "sa")

    # === Phase 6: Refined optimization cycles ===
    print(f"\n=== Phase 6: Refined Cycles (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for cycle in range(5):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / (5 - cycle), 80)
            if t_alloc < 10:
                t_alloc = 10

            # Perturb and re-optimize
            if cycle == 0:
                f_init = best_seq.copy()
            elif cycle == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif cycle == 2:
                # Reverse and blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            elif cycle == 3:
                # Add uniform component
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Larger perturbation
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise, target_sum=1.0)

            # Quick smooth optimization
            a_opt, obj = optimize_smooth(f_init, time_limit=t_alloc * 0.6)
            try_seq(a_opt, f"refine_{cycle}")

            # Quick CD
            rem = deadline - time.monotonic() - 50
            if rem > 15:
                a_opt, obj = fast_coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt, f"refine_cd_{cycle}")

    # === Phase 7: Try extending to longer sequences ===
    print(f"\n=== Phase 7: Longer Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best + 64, n_best + 128, n_best * 2]:
            if target_n > 3000 or target_n <= n_best:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 80)
            a_opt, obj = optimize_smooth(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"long_n{target_n}")

            rem = deadline - time.monotonic() - 40
            if rem > 20:
                a_opt, obj = fast_coordinate_descent(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt, f"long_cd_n{target_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = fast_coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = optimize_smooth(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_smooth")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
