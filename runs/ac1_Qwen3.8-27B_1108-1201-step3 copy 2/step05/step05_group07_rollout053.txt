```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_lp_matrix(f):
    """Build the constraint matrix A where (f*g)_k = A[k,:] @ g."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    return A

def solve_lp(f, M, solver='CLARABEL'):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n)
    A = build_lp_matrix(f)
    constraints = [A @ g <= M, g >= 0]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def lp_iteration(f, t=0.5, n_iters=50, solver='CLARABEL'):
    """Iterate the LP-based refinement."""
    f = f.copy().astype(np.float64)
    S = f.sum()
    f = f / S  # normalize to sum=1 for simplicity
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    
    for it in range(n_iters):
        M = fast_convolve(f).max()
        g = solve_lp(f, M, solver=solver)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize g to sum=1
        f_new = (1 - t) * f + t * g
        peak = fast_convolve(f_new).max()
        if peak < best_peak:
            best_peak = peak
            best_f = f_new.copy()
        f = f_new
    
    # Rescale back
    best_f = best_f / best_f.sum() * np.sqrt(2 * len(best_f))
    return best_f, best_peak
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with L-BFGS on a soft-max smoothed objective, multi-start with
diverse initializations, and coordinate descent polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_cross_corr(w, a):
    """Compute c[i] = sum_k w[k] * a[k-i] for i=0,...,n-1."""
    n = len(a)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    return c

def soft_max_obj_grad(z, T):
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    c = compute_cross_corr(w, a)
    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def build_lp_matrix(f):
    n = len(f)
    rows = np.zeros(2*n-1, dtype=int)
    cols = np.zeros(2*n-1, dtype=int)
    # Actually, the number of non-zeros is n^2, not 2n-1
    # Let me use a different approach
    nnz = 0
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        nnz += je - js + 1
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    idx = 0
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M, solver='CLARABEL'):
    n = len(f)
    g_var = cp.Variable(n)
    A = build_lp_matrix(f)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)

def lp_iterate(f, time_budget, t_mix=0.1):
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_budget:
        M = fast_convolve(f).max()
        g = solve_lp(f, M)
        if g is None:
            # Try with ECOS
            g = solve_lp(f, M, solver='ECOS')
            if g is None:
                break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g = g / g_sum
        
        # Line search on t
        improved = False
        for t in [t_mix, 0.5*t_mix, 0.25*t_mix, 0.05, 0.01]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            fs = f_trial.sum()
            if fs < 1e-15:
                continue
            f_trial = f_trial / fs
            peak = fast_convolve(f_trial).max()
            if peak < best_peak - 1e-12:
                best_peak = peak
                best_f = f_trial.copy()
                f = f_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try smaller step or break
            if t_mix < 0.005:
                break
            t_mix *= 0.5
        
        iteration += 1
        if iteration % 50 == 0:
            elapsed = time.monotonic() - t0

def lbgfs_refine(a, time_budget, start_T=10, max_T=500):
    n = len(a)
    z = z_from_a(a)
    t0 = time.monotonic()
    
    def func(zv):
        if time.monotonic() - t0 > time_budget:
            return 1e15
        return soft_max_obj_grad(zv, start_T)[0]
    
    def jac(zv):
        if time.monotonic() - t0 > time_budget:
            return np.zeros_like(zv)
        return soft_max_obj_grad(zv, start_T)[1]
    
    try:
        res = minimize(
            func, z, jac=jac, method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 100000, 'disp': False}
        )
        z = res.x
    except Exception:
        pass
    
    a_result = softmax_from_z(z)
    return a_result

def coordinate_descent(a, time_budget):
    a = a.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    multipliers = np.array([0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0])
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a
