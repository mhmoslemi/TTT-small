def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, conv, top_k=30):
    """Compute subgradient of max(conv) using only top-K active indices.
    
    d conv[k] / d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    sorted_idx = np.argpartition(conv, -top_k)[-top_k:]
    for k in sorted_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            indices = np.arange(i_start, i_end + 1)
            a_indices = k_int - indices
            grad[indices] += 2.0 * a[a_indices]
    return grad, sorted_idx

def smooth_grad_fft(a, T):
    """Compute gradient and value of logsumexp(conv(a,a)/T) via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    mx = scaled.max()
    e = np.exp(scaled - mx)
    w = e / e.sum()
    lse_val = T * (mx + np.log(e.sum()))
    
    # grad_i = (2/T) * sum_j a[j] * w[i+j] for valid i+j
    # This is a cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w

def solve_lp_primal(f, M, n):
    """Solve LP: max sum(g) s.t. T@g <= M*1, g >= 0.
    T is (2n-1) x n with T[k,i] = f[k-i] if 0<=k-i<n else 0.
    """
    c = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # Column j: T[k,j] = f[k-j] if 0<=k-j<n
        # k-j in [0,n-1] means k in [j, j+n-1], intersected with [0, 2n-2]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_ub[k_start:k_end, j] = f[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_dual(f, M, n):
    """Solve dual LP: min M*sum(lambda) s.t. T^T @ lambda >= 1, lambda >= 0.
    (T^T)[i,k] = f[k-i] if 0<=k-i<n.
    """
    n_lambda = 2 * n - 1
    c = M * np.ones(n_lambda)
    # A_ub[i,k] = -(T^T)[i,k] = -f[k-i] if 0<=k-i<n
    # b_ub = -1
    A_ub = np.zeros((n, n_lambda))
    for i in range(n):
        # (T^T)[i,k] = f[k-i] if 0<=k-i<n, i.e., k in [i, i+n-1]
        k_start = i
        k_end = min(i + n, n_lambda)
        A_ub[i, k_start:k_end] = -f[:k_end - k_start]
    b_ub = -np.ones(n)
    bounds = [(0, None)] * n_lambda
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with exact 1D optimization."""
    n = len(a0)
    f = project_simplex(a0.astype(np.float64))
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            # Golden section search for optimal value of f[j]
            # keeping others proportional to maintain sum=1
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            lo = 0.0
            hi = 1.0
            
            def eval_at(x):
                scale = (1.0 - x) / denom
                ft = f.copy()
                ft[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                return fast_convolve(ft).max()
            
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)
            
            for _ in range(16):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_at(d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_at(opt_val)
            
            if opt_obj < best_obj - 1e-13:
                scale = (1.0 - opt_val) / denom
                f[j] = opt_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric (rising-falling) geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    elif len(a) > n:
        a = a[:n]
    return a / a.sum()


def make_bump(n, sigma_frac=0.3):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * (sigma_frac ** 2)))
    a += 0.001
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass
    
    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old_score:.8f}]")
    
    n_parent = len(parent) if parent is not None else 768
    
    # === Phase 0: Quick diverse initializations ===
    print("=== Phase 0: Initializations ===")
    for n_try in [n_parent, 1024]:
        for r in [0.97, 0.98, 0.99, 0.995]:
            try_seq(make_geometric(n_try, r), f"geo_r{r}_n{n_try}")
        for r in [0.97, 0.98, 0.99]:
            try_seq(make_symmetric_geo(n_try, r), f"symgeo_r{r}_n{n_try}")
        for pw in [1.5, 2.0, 3.0, 4.0, 6.0]:
            try_seq(make_cosine(n_try, power=pw), f"cos_p{pw}_n{n_try}")
        for sf in [0.2, 0.25, 0.3, 0.35, 0.4]:
            try_seq(make_bump(n_try, sigma_frac=sf), f"bump_s{sf}_n{n_try}")
    
    print(f"  After Phase 0: best={best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # Use the best as base for optimization
    base = best_seq.copy()
    n_base = len(base)
    
    # === Phase 1: Active-set subgradient (fast, many iterations) ===
    print(f"\n=== Phase 1: Active-Set Subgradient ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  n={n_base}, t={t_alloc:.0f}s")
        a_opt, obj = active_set_subgrad(base, time_limit=t_alloc, top_k=20, verbose=True)
        try_seq(a_opt, "activeSG_p1")
    
    # Also try with different top_k
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem, 150)
        a_opt, obj = active_set_subgrad(base, time_limit=t_alloc, top_k=50, verbose=False)
        try_seq(a_opt, "activeSG_topk50")
    
    # === Phase 2: LP-Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-Frank-Wolfe ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  n={n_base}, t={t_alloc:.0f}s")
        a_opt, obj = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_p2")
    
    # LP-FW with perturbation
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        noise = rng.standard_normal(n_base) * 0.004
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 100)
        a_opt, obj = lpfw_optimize(f_pert, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, "lpfw_pert")
    
    # === Phase 3: Smooth optimization ===
    print(f"\n=== Phase 3: Smooth Optimization ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  n={n_base}, t={t_alloc:.0f}s")
        a_opt, obj = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.2, T_end=0.005, verbose=True)
        try_seq(a_opt, "smooth_p3")
    
    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(rem, 80)
        print(f"  Single CD t={t_alloc:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "cd_single")
    
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_alloc = min(rem, 60)
        a_opt, obj = pairwise_cd(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "cd_pair")
    
    # === Phase 5: Try longer sequences ===
    print(f"\n=== Phase 5: Longer Sequences ===")
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 100
        if rem < 40 or target_n <= n_base:
            continue
        f_interp = interpolate_seq(best_seq, target_n)
        
        t1 = min(rem * 0.4, 60)
        a_opt, obj = active_set_subgrad(f_interp, time_limit=t1, top_k=20, verbose=False)
        try_seq(a_opt, f"activeSG_n{target_n}")
        
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t2 = min(rem * 0.5, 50)
            a_opt, obj = smooth_optimize(a_opt, time_limit=t2, T_start=0.1, T_end=0.01)
            try_seq(a_opt, f"smooth_n{target_n}")
    
    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Fine smooth optimization
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, obj = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.03, T_end=0.003, verbose=True)
            try_seq(a_opt, "smooth_fine")
        
        # LP-FW final
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, obj = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "lpfw_final")
        
        # Active-set final
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, obj = active_set_subgrad(best_seq, time_limit=t_alloc, top_k=15, verbose=False)
            try_seq(a_opt, "activeSG_final")
        
        # Small perturbations + quick optimize
        for pidx in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 15)
            a_opt, obj = active_set_subgrad(f_pert, time_limit=t_alloc, top_k=20, verbose=False)
            try_seq(a_opt, f"perturb_{pidx}")
        
        # Reversed sequence
        rem = deadline - time.monotonic() - 8
        if rem > 6:
            f_rev = best_seq[::-1].copy()
            a_opt, obj = active_set_subgrad(f_rev, time_limit=min(rem, 10), top_k=20, verbose=False)
            try_seq(a_opt, "reversed")
        
        # Blend with uniform
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            f_blend = 0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best)
            f_blend = project_simplex(f_blend)
            a_opt, obj = smooth_optimize(f_blend, time_limit=min(rem, 8), T_start=0.02, T_end=0.005)
            try_seq(a_opt, "blend")
    
    # Fallback
    if best_seq is None:
        best_seq = make_cosine(768, power=2.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
