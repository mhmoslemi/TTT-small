def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'dual feasibility tol': tol,
               'primal feasibility tol': tol, 'dual ratio tol': 1e-12,
               ' primal ratio tol': 1e-12}
    
    if prev_x is not None:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
    else:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with warm starts and adaptive expansion."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    prev_lp_sol = None

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    lp_time_budget = 5.0  # max seconds per LP solve

    while time.monotonic() - t0 < time_limit:
        it += 1
        lp_start = time.monotonic()
        
        # Use adaptive M expansion
        M_try = M * m_exp
        g = solve_lp_direction(f, M_try, n, prev_x=prev_lp_sol)
        
        lp_elapsed = time.monotonic() - lp_start
        if g is None:
            m_exp *= 1.2
            if m_exp > 20.0:
                m_exp = 1.0
                # Try with a much larger M
                g = solve_lp_direction(f, M * 10.0, n)
                if g is None:
                    break
        else:
            prev_lp_sol = g.copy()
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 10.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 31)
            scale = 0.002 + 0.004 * min(no_improve // 80, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            prev_lp_sol = None

    return best_f, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth approximation with aggressive beta annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [3.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            res = minimize(
                smooth_obj_and_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 50,
                         'maxfun': 5000, 'ftol': 1e-14}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            if verbose:
                print(f"    L-BFGS beta={beta} failed")
            break

    return best_f, best_obj


def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on the smooth objective with adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t = np.log(f + 1e-15)
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    lr = 0.01
    it = 0
    t0 = time.monotonic()
    no_improve = 0
    
    # Adaptive beta schedule
    base_beta = 5.0
    beta_growth = 1.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = base_beta * (beta_growth ** it)
        beta = min(beta, 2000.0)
        
        obj, grad = smooth_obj_and_grad(t, beta)
        
        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad**2
        m_hat = m_t / (1 - beta1**it)
        v_hat = v_t / (1 - beta2**it)
        t = t - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a = softmax_fn(t)
        obj_actual = fast_convolve(a).max()
        
        if obj_actual < best_obj - 1e-14:
            best_obj = obj_actual
            best_f = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            if lr < 0.0001:
                # Restart
                rng = np.random.default_rng(it * 104729 + 13)
                t = np.log(best_f + 1e-15) + rng.standard_normal(n) * 0.01
                m_t = np.zeros(n)
                v_t = np.zeros(n)
                it = 0
                lr = 0.005

        # Gradually decrease lr
        lr = max(lr * 0.9998, 0.00001)

    return best_f, best_obj


def peak_elimination(f0, time_limit, verbose=False):
    """
    Directly target the top-K convolution peaks.
    Uses a sequence of LPs that try to reduce the K most problematic lags.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    K = 5  # number of peaks to target

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -K)[-K:]
        peak_vals = conv[peaks]
        max_val = conv.max()
        
        # Find direction that reduces the top peaks
        # min sum |delta_a|^2 subject to:
        #   sum (conv)(f + delta)[k] <= max_val - eps for k in peaks
        #   sum delta = 0 (maintain normalization)
        #   f + delta >= 0
        
        # Linearize: conv(f+delta, f+delta)[k] ≈ conv(f,f)[k] + 2*conv(f, delta)[k]
        # So: 2*conv(f, delta)[k] <= max_val - eps - conv(f,f)[k] for k in peaks
        #     + conv(f,f)[k] <= max_val - eps
        #     => conv(f, delta)[k] <= (max_val - eps - conv(f,f)[k]) / 2
        
        eps = 1e-5 + 0.01 * max_val
        
        # LP: max -sum(delta)^2 approx -> min -sum(delta) subject to constraints
        # Actually, let's use a different approach:
        # Find the direction of steepest decrease of the max peak
        # This is the -subgradient direction projected onto the simplex
        
        # Subgradient approach:
        # The subgradient of max_k conv(f,f)[k] at f is:
        # 2 * sum_k w[k] * (f shifted by k) where w is a distribution on argmax set
        # For top-K peaks, use w proportional to peak values
        
        weights = peak_vals / peak_vals.sum()
        sg = np.zeros(n)
        for i in range(K):
            k = int(peaks[i])
            # conv(f, delta)[k] involves f shifted by k
            # d conv[k] / d f[j] = 2*(f[k-j] + f[j-k]) (with bounds)
            # For the subgradient of the max, we use:
            # sg[j] = 2 * sum_i w_i * (f[k_i - j] + f[j - k_i]) (with bounds)
            
            for j_idx in range(n):
                term1 = f[k - j_idx] if 0 <= k - j_idx < n else 0.0
                term2 = f[j_idx - k] if 0 <= j_idx - k < n else 0.0
                sg[j_idx] += 2.0 * weights[i] * (term1 + term2)
        
        # Actually, this is O(K*n) per iteration which is fine.
        # But let me use the FFT method for efficiency:
        # sg = 2 * sum_i w_i * cross_corr(f, delta_k_i)
        # where cross_corr is the "shifted" version
        
        # Use FFT:
        N = 1 << (3 * n - 2).bit_length()
        F = np.fft.rfft(f, N)
        W = np.zeros(N // 2 + 1)
        for i in range(K):
            k = int(peaks[i])
            W[k] += weights[i]
        W_full = np.fft.rfft(np.zeros(N))  # placeholder
        # Actually, let me just construct the gradient directly:
        grad = np.zeros(n)
        for i in range(K):
            k = int(peaks[i])
            # Contribution to grad[j] from peak at lag k:
            # 2 * w_i * (f[k-j] if 0<=k-j<n else 0 + f[j-k] if 0<=j-k<n else 0)
            if k < n:
                # f[k-j] valid when j <= k
                grad[:k+1] += 2.0 * weights[i] * f[k::-1][:k+1]  # f[k], f[k-1], ..., f[0]
                # f[j-k] valid when j >= k
                grad[k:] += 2.0 * weights[i] * f[:n-k]  # f[0], f[1], ..., f[n-1-k]
            else:
                # k >= n: f[k-j] valid when j <= k-(n-1) = k-n+1... 
                # Actually for linear conv, conv[k] = sum_j f[j]*f[k-j]
                # valid when 0<=j<n and 0<=k-j<n, i.e., k-n+1 <= j <= k
                # d conv[k]/d f[j] = f[k-j] + f[j-k] (with f[x]=0 outside [0,n-1])
                # For j in [0, n-1]:
                #   f[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
                #   f[j-k] valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
                # Since j in [0, n-1] and k in [n, 2n-2]:
                #   k-j in [k-n+1, k] and we need k-j < n, i.e., j > k-n
                #   j-k in [-k, n-1-k] and we need j-k >= 0, i.e., j >= k (impossible since j < n <= k)
                # So only f[k-j] contributes, for j in [k-n+1, n-1] (intersected with [0, n-1])
                lo = max(0, k - n + 1)
                for j in range(lo, n):
                    grad[j] += 2.0 * weights[i] * f[k - j]
        
        # Project gradient onto tangent space of simplex (remove mean)
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Take a step in the -gradient direction
        # Use backtracking line search
        alpha = 0.1
        best_alpha = 0.0
        best_val = max_val
        
        for _ in range(10):
            f_trial = f - alpha * grad_tangent
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5
        
        if best_alpha > 1e-10:
            f_new = f - best_alpha * grad_tangent
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    PE iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear (increasing)
    x = np.arange(n)
    a = x + 1
    inits.append(a / a.sum())
    
    # Linear (decreasing)
    a = n - x
    inits.append(a / a.sum())
    
    # Sine (various phases and frequencies)
    for phase in [0, 0.3, 0.5, 0.7, 1.0]:
        for freq in [1, 2]:
            a = np.maximum(np.sin(np.pi * x / (n - 1) * freq + phase), 0)
            if a.sum() > 0:
                inits.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.power(x + 1, p)
        inits.append(a / a.sum())
        a = np.power(n - x + 1, p)
        inits.append(a / a.sum())
    
    # Bimodal
    center = n // 2
    for spread in [n//4, n//3, n//2]:
        a = np.exp(-((x - center)**2) / (2 * spread**2))
        inits.append(a / a.sum())
    
    # Tukey window
    for r in [0.1, 0.2, 0.3, 0.5]:
        a = np.ones(n)
        nr = int(r * n)
        for i in range(nr):
            a[i] = 0.5 + 0.5 * np.cos(np.pi * (2 * i / (2 * nr) - 1))
            a[n - 1 - i] = a[i]
        inits.append(a / a.sum())
    
    # Random
    for _ in range(3):
        a = rng.random(n)
        inits.append(a / a.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse initialization and quick evaluation
    2. Fast Adam optimization on best init
    3. L-BFGS with beta annealing
    4. LP-Frank-Wolfe (primary optimizer)
    5. Peak elimination
    6. Multi-length exploration
    7. Final deep optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768
    if parent is None:
        parent = np.ones(n_parent) / n_parent

    # === Phase 1: Diverse initialization (60s) ===
    print("\n=== Phase 1: Diverse init ===")
    t1_end = time.monotonic() + 60
    
    # Try parent first
    try_seq(parent, "parent")
    
    # Try diverse inits at parent length
    inits = generate_diverse_inits(n_parent, rng)
    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end:
            break
        try_seq(init, f"init_{idx}")
    
    base = best_seq if best_seq is not None else parent

    # === Phase 2: Adam optimization (300s) ===
    print(f"\n=== Phase 2: Adam (best={best_score:.6f}) ===")
    t2_end = time.monotonic() + 300
    a_opt, _ = adam_optimize(base, time_limit=min(300, deadline - time.monotonic() - 600), verbose=True)
    try_seq(a_opt, "adam_main")
    
    # Adam from second best init
    if best_seq is not None:
        rem = t2_end - time.monotonic()
        if rem > 30:
            a_opt, _ = adam_optimize(base, time_limit=min(rem, 100))
            try_seq(a_opt, "adam_2nd")

    # === Phase 3: L-BFGS (400s) ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t3_end = time.monotonic() + 400
    a_opt, _ = lbfqg_optimize(base, time_limit=min(400, deadline - time.monotonic() - 500), verbose=True)
    try_seq(a_opt, "lbfqg_main")
    
    # L-BFGS from perturbed
    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 30:
            noise = rng.standard_normal(len(base)) * 0.02
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=min(rem, 150))
            try_seq(a_opt, "lbfqg_pert")

    # === Phase 4: LP-FW (500s) ===
    print(f"\n=== Phase 4: LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t4_end = time.monotonic() + 500
    a_opt, _ = lp_fw_optimize(base, time_limit=min(500, deadline - time.monotonic() - 400), verbose=True)
    try_seq(a_opt, "lpfw_main")
    
    # LP-FW on reversed
    if best_seq is not None:
        rem = t4_end - time.monotonic()
        if rem > 50:
            a_opt, _ = lp_fw_optimize(base[::-1].copy(), time_limit=min(rem, 150))
            try_seq(a_opt, "lpfw_rev")

    # === Phase 5: Peak Elimination (200s) ===
    print(f"\n=== Phase 5: Peak Elimination (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t5_end = time.monotonic() + 200
    a_opt, _ = peak_elimination(base, time_limit=min(200, deadline - time.monotonic() - 300), verbose=True)
    try_seq(a_opt, "peak_elim")

    # === Phase 6: Multi-length (200s) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [640, 896, 1024, 1280, 960]:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.3, 60)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")
        # Quick L-BFGS
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            a_opt, _ = lbfqg_optimize(f_interp, time_limit=min(rem, 20))
            try_seq(a_opt, f"lbfqg_len_{target_n}")

    # === Phase 7: Final deep optimization ===
    print(f"\n=== Phase 7: Final Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Deep LP-FW
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.5, 200), verbose=True)
            try_seq(a_opt, "deep_lpfw")
        
        # Deep L-BFGS
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=min(rem * 0.5, 100), verbose=True)
            try_seq(a_opt, "deep_lbfqg")
        
        # Deep Adam
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem * 0.6, 80))
            try_seq(a_opt, "deep_adam")
        
        # Peak elimination on best

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse analytical initializations with quick screening
2. Adam optimizer on log-sum-exp smoothed objective (fast pre-optimizer)
3. L-BFGS with aggressive beta annealing for local refinement
4. Restricted-constraint LP Frank-Wolfe (top-K active lags for speed)
5. Peak elimination via weighted subgradient
6. Multi-length interpolation and cross-length refinement
7. Final deep optimization with alternating methods
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_obj_and_grad(t, beta):
    """Compute log-sum-exp smoothed objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def solve_lp_restricted(f, M, n, K=100, active_lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K lags, g >= 0.
    Returns (solution, active_lags_used) for tracking."""
    conv = fast_convolve(f)
    if active_lags is None or len(active_lags) < K:
        active_lags = np.argpartition(conv, -K)[-K:]
    else:
        # Use provided active lags, add new top ones
        top_k_new = np.argpartition(conv, -K)[-K:]
        active_lags = np.unique(np.concatenate([active_lags, top_k_new]))
        if len(active_lags) > K:
            active_lags = active_lags[np.argsort(conv[active_lags])[-K:]]

    K_actual = len(active_lags)
    c = -np.ones(n)
    A_ub = np.zeros((K_actual, n))
    b_ub = np.full(K_actual, M)

    for row_idx, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        A_ub[row_idx, js] = f[k - js]

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 8.0})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0), active_lags
    return None, active_lags


def solve_lp_full(f, M, n):
    """Solve LP with all constraints (for final polishing)."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 15.0})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_restricted=True):
    """LP-based Frank-Wolfe with restricted constraints and warm tracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    active_lags = None
    K = min(100, 2 * n - 1)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if use_restricted:
            g, active_lags = solve_lp_restricted(f, M * 1.001, n, K, active_lags)
        else:
            g = solve_lp_full(f, M * 1.001, n)

        if g is None:
            # Try with larger M
            if use_restricted:
                g, active_lags = solve_lp_restricted(f, M * 1.1, n, K, active_lags)
            else:
                g = solve_lp_full(f, M * 1.1, n)
            if g is None:
                if no_improve > 50:
                    break
                no_improve += 1
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 31)
            scale = 0.002 + 0.004 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            active_lags = None

    return best_f, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()

    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 15:

        sub_time = remaining / (len(betas) - bi)

        try:

                smooth_obj_and_grad, t, args=(beta,),

                options={'maxiter': 800, 'gtol': 1e-12, 'maxls': 50,
                         'maxfun': 8000, 'ftol': 1e-15}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            if verbose:
                print(f"    L-BFGS beta={beta} failed")
            break

    return best_f, best_obj


def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on smooth objective with adaptive beta and LR."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t = np.log(f + 1e-15)
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    lr = 0.02
    it = 0
    t0 = time.monotonic()
    no_improve = 0
    base_beta = 3.0
    beta_growth = 1.0015
    lr_decay = 0.9997

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = min(base_beta * (beta_growth ** it), 3000.0)

        try:
            obj, grad = smooth_obj_and_grad(t, beta)
        except Exception:
            break

        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad * grad)
        m_hat = m_t / (1 - beta1 ** it)
        v_hat = v_t / (1 - beta2 ** it)
        t = t - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = softmax_fn(t)
        obj_actual = fast_convolve(a).max()

        if obj_actual < best_obj - 1e-14:
            best_obj = obj_actual
            best_f = a.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.85
            no_improve = 0
            if lr < 0.0005:
                rng = np.random.default_rng(it * 104729 + 13)
                t = np.log(best_f + 1e-15) + rng.standard_normal(n) * 0.005
                m_t = np.zeros(n)
                v_t = np.zeros(n)
                it = 0
                lr = 0.01
                base_beta = max(3.0, base_beta * 0.5)

        lr = max(lr * lr_decay, 0.0001)

    return best_f, best_obj


def peak_elimination(f0, time_limit, verbose=False):
    """Directly target top-K convolution peaks via weighted subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    K = min(10, 2 * n - 1)
    no_improve = 0
    it = 0
    alpha = 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        max_val = conv.max()
        peaks = np.argpartition(conv, -K)[-K:]
        peak_vals = np.maximum(conv[peaks] - max_val * 0.8, 0)
        if peak_vals.sum() < 1e-15:
            peak_vals = conv[peaks]
        weights = peak_vals / peak_vals.sum()

        # Compute weighted subgradient using FFT
        N = 1 << (3 * n - 2).bit_length()
        F = np.fft.rfft(f, N)
        w_vec = np.zeros(2 * n - 1)
        for i in range(K):
            w_vec[int(peaks[i])] = weights[i]
        W = np.fft.rfft(w_vec, N)
        R = np.fft.irfft(W * np.conj(F), N)[:n]
        grad = 2.0 * R

        # Project to tangent space
        grad_tangent = grad - grad.mean()

        # Backtracking line search
        best_alpha = 0.0
        best_val = max_val
        a_try = alpha
        for _ in range(15):
            f_trial = f - a_try * grad_tangent
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                a_try *= 0.5
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = a_try
            a_try *= 0.5

        if best_alpha > 1e-10:
            f_new = f - best_alpha * grad_tangent
            f_new = np.maximum(f_new, 0.0)
            f = f_new / f_new.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                alpha = min(alpha * 1.02, 2.0)
                if verbose and it % 200 == 0:
                    print(f"    PE iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
                alpha *= 0.95
        else:
            no_improve += 1
            alpha *= 0.9

        alpha = max(alpha, 0.01)

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
            alpha = 0.3

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Smooth with a small Gaussian
    if new_n > 4:
        from scipy.ndimage import gaussian_filter1d
        f_interp = gaussian_filter1d(f_interp, sigma=0.5)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences for exploration."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Asymmetric linear
    a = x + 1.0
    inits.append(a / a.sum())
    a = n - x
    inits.append(a / a.sum())

    # Sine with various phases
    for phase in [0.0, 0.5, 1.0, 1.5, 2.0, 2.5]:
        a = np.maximum(np.sin(np.pi * x / (n - 1) + phase), 0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())

    # Power laws
    for p in [0.3, 0.5, 1.0, 2.0, 4.0]:
        a = np.power(x + 1.0, p)
        inits.append(a / a.sum())
        a = np.power(n - x + 1.0, p)
        inits.append(a / a.sum())

    # Bimodal
    for frac in [0.2, 0.3, 0.4]:
        c1 = int(frac * n)
        c2 = n - 1 - c1
        a = np.zeros(n)
        a[c1] = 0.5
        a[c2] = 0.5
        # Smooth
        if n > 10:
            sigma = max(1.0, n * 0.05)
            a = np.exp(-((x - c1) ** 2) / (2 * sigma ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sigma ** 2))
        inits.append(a / a.sum())

    # Tukey windows
    for r in [0.1, 0.2, 0.3, 0.5, 0.7]:
        a = np.ones(n)
        nr = max(1, int(r * n / 2))
        for i in range(nr):
            val = 0.5 + 0.5 * np.cos(np.pi * (i / nr))
            a[i] = val
            if n - 1 - i != i:
                a[n - 1 - i] = val
        inits.append(a / a.sum())

    # Random
    for _ in range(2):
        a = rng.random(n)
        inits.append(a / a.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse initialization, Adam,
    L-BFGS, restricted LP-FW, peak elimination, and multi-length refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = [None]
    best_score = [float('inf')]

    def try_seq(a, label=""):
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score[0]:
            best_score[0] = score
            best_seq[0] = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768
    if parent is None:
        parent = np.ones(n_parent) / n_parent

    # === Phase 1: Diverse initialization screening (45s) ===
    print("\n=== Phase 1: Diverse Init ===")
    try_seq(parent, "parent")

    t1_end = time.monotonic() + 45
    inits = generate_diverse_inits(n_parent, rng)
    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end:
            break
        score = try_seq(init, f"init_{idx}")
        if score < 1.8:
            print(f"    Candidate {idx} has score {score:.4f}")

    base = best_seq[0] if best_seq[0] is not None else parent

    # === Phase 2: Adam pre-optimization (250s) ===
    print(f"\n=== Phase 2: Adam (best={best_score[0]:.6f}) ===")
    t2_limit = min(250, deadline - time.monotonic() - 1200)
    if t2_limit > 30:
        a_opt, _ = adam_optimize(base, time_limit=t2_limit, verbose=True)
        try_seq(a_opt, "adam_main")

        # Second Adam run from perturbed
        rem = min(80, deadline - time.monotonic() - 1100)
        if rem > 20 and best_seq[0] is not None:
            noise = rng.standard_normal(len(best_seq[0])) * 0.01
            f_pert = project_simplex(best_seq[0] + noise)
            a_opt, _ = adam_optimize(f_pert, time_limit=rem)
            try_seq(a_opt, "adam_pert")

    # === Phase 3: L-BFGS refinement (400s) ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score[0]:.6f}) ===")
    base = best_seq[0] if best_seq[0] is not None else parent
    t3_limit = min(400, deadline - time.monotonic() - 750)
    if t3_limit > 30:
        a_opt, _ = lbfqg_optimize(base, time_limit=t3_limit, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # L-BFGS from different start
        rem = min(100, deadline - time.monotonic() - 650)
        if rem > 20 and best_seq[0] is not None:
            noise = rng.standard_normal(len(best_seq[0])) * 0.015
            f_pert = project_simplex(best_seq[0] + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=rem)
            try_seq(a_opt, "lbfqg_pert")

    # === Phase 4: LP-FW main optimization (500s) ===
    print(f"\n=== Phase 4: LP-FW (best={best_score[0]:.6f}) ===")
    base = best_seq[0] if best_seq[0] is not None else parent
    t4_limit = min(500, deadline - time.monotonic() - 350)
    if t4_limit > 30:
        a_opt, _ = lp_fw_optimize(base, time_limit=t4_limit, verbose=True, use_restricted=True)
        try_seq(a_opt, "lpfw_main")

        # LP-FW on reversed
        rem = min(120, deadline - time.monotonic() - 250)
        if rem > 20 and best_seq[0] is not None:
            a_opt, _ = lp_fw_optimize(base[::-1].copy(), time_limit=rem, use_restricted=True)
            try_seq(a_opt, "lpfw_rev")

        # Full LP-FW for polishing (smaller budget)
        rem = min(80, deadline - time.monotonic() - 200)
        if rem > 20 and best_seq[0] is not None:
            a_opt, _ = lp_fw_optimize(best_seq[0], time_limit=rem, use_restricted=False)
            try_seq(a_opt, "lpfw_full")

    # === Phase 5: Peak elimination (150s) ===
    print(f"\n=== Phase 5: Peak Elim (best={best_score[0]:.6f}) ===")
    base = best_seq[0] if best_seq[0] is not None else parent
    t5_limit = min(150, deadline - time.monotonic() - 200)
    if t5_limit > 20:
        a_opt, _ = peak_elimination(base, time_limit=t5_limit, verbose=True)
        try_seq(a_opt, "peak_elim")

    # === Phase 6: Multi-length exploration (180s) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score[0]:.6f}) ===")
    base = best_seq[0] if best_seq[0] is not None else parent

    target_lengths = [640, 896, 960, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Quick Adam
        t_alloc = min(rem * 0.25, 35)
        a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"adam_n{target_n}")
        # LP-FW
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_alloc = min(rem * 0.4, 40)
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_restricted=True)
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 7: Final deep optimization ===
    print(f"\n=== Phase 7: Final Deep (best={best_score[0]:.6f}) ===")
    if best_seq[0] is not None:
        n_best = len(best_seq[0])

        # Deep LP-FW
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            a_opt, _ = lp_fw_optimize(best_seq[0], time_limit=min(rem * 0.5, 150),
                                      verbose=True, use_restricted=True)
            try_seq(a_opt, "deep_lpfw")

        # Deep L-BFGS
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            a_opt, _ = lbfqg_optimize(best_seq[0], time_limit=min(rem * 0.5, 80))
            try_seq(a_opt, "deep_lbfqg")

        # Deep Adam
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            a_opt, _ = adam_optimize(best_seq[0], time_limit=min(rem * 0.5, 60))
            try_seq(a_opt, "deep_adam")

        # Peak elimination
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_opt, _ = peak_elimination(best_seq[0], time_limit=min(rem * 0.6, 40))
            try_seq(a_opt, "deep_pe")

        # Coordinate descent polish
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq[0], time_limit=min(rem, 30))
            try_seq(a_opt, "deep_cd")

        # Final perturbations
        for attempt in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_start = project_simplex(best_seq[0] + noise)
            if attempt % 2 == 0:
                a_opt, _ = lp_fw_optimize(f_start, time_limit=min(rem * 0.6, 25),
                                          use_restricted=True)
            else:
                a_opt, _ = lbfqg_optimize(f_start, time_limit=min(rem * 0.6, 20))
            try_seq(a_opt, f"final_{attempt}")

    # Fallback
    if best_seq[0] is None:
        best_seq[0] = np.ones(384) / 384
        best_score[0] = float(evaluate_sequence(best_seq[0].tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq[0].tolist()))
    if np.isfinite(fs):
        best_score[0] = fs

    print(f"\nFINAL SCORE: {best_score[0]:.6f} (n={len(best_seq[0])})")
    return [float(x) for x in best_seq[0]]
