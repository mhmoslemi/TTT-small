def backtracking_line_search(f, g, M, max_iter=20):
    """Find a good step size alpha for f -> f + alpha*(g-f)."""

        elapsed = time.monotonic() - t0
        
        # Solve LP direction
        g = solve_lp_fast(f, M * m_relax, n, top_k_constraints=top_k_constraints,
                          time_limit=5, add_violated=True, max_rounds=3)
        if g is None:
            m_relax *= 1.2
            if m_relax > 10.0:
                m_relax = 1.0
                continue
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if g is close to f (convergence)
        diff = np.max(np.abs(g - f))
        if diff < 1e-13:
            no_improve += 5
            m_relax = min(m_relax * 1.05, 8.0)
            if no_improve > 60:
                break
            continue
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(max_backtrack):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= backtrack_factor
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            if val < M * (1 - 1e-6):
                # Good enough improvement, do a quick refinement
                alpha = alpha * 0.8
                break
            alpha *= backtrack_factor
            if alpha < 0.005:
                break
        
        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % verbose_every == 0:
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 8.0)
        
        # Perturbation to escape stagnation
        if no_improve > 100:
            scale = min(0.0005 + no_improve * 0.00001, 0.005)
            rng = np.random.default_rng(it * 104729 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr_init=0.002, mom_init=0.92,
                       top_k=80, temp_init=0.0003, verbose=True):
    """Subgradient method with momentum and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    mom = mom_init
    temp = temp_init
    no_improve = 0
    it = 0
    k = min(top_k, 2 * n - 1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 3 == 1:
            conv = fast_convolve(f)
        
        # Compute subgradient
        if k < len(conv):
            peaks = np.argpartition(conv, -k)[-k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            wi = w[i]
            if wi < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * wi * f[kk - idx]
        
        # Momentum update
        v = mom * v - lr * sg
        f = project_simplex(f + v)
        
        # Periodic evaluation
        if it % 10 == 0:
            conv = fast_convolve(f)
            co = conv.max()
            if co < best_obj - 1e-15:
                best_obj = co
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive parameters
            if no_improve > 200:
                lr *= 0.93
                temp *= 1.02
            if no_improve > 800:
                rng = np.random.default_rng(it * 31 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * lr * 5)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0002)
                no_improve = 0
                mom = mom_init
        
        lr *= 0.9998
        temp *= 1.0003
        mom = min(mom + 0.00005, 0.999)
        
        if verbose and it % 500 == 0:
            print(f"    SG iter {it} ({time.monotonic()-t0:.0f}s): best={best_obj:.12f}")
    
    return best_f, best_obj


def coordinate_refine(f0, time_limit, verbose=True):
    """Targeted coordinate descent focusing on high-impact coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Find top peaks
        k_peaks = min(20, len(conv))
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        
        # For each peak, find the coordinates that contribute most
        # and try adjusting them
        candidates = {}
        for kk in peaks:
            js = max(0, int(kk) - n + 1)
            je = min(n - 1, int(kk))
            for j in range(js, je + 1):
                contrib = 2.0 * f[int(kk) - j] * f[j]
                if j not in candidates:
                    candidates[j] = 0.0
                candidates[j] += contrib
        
        # Sort by contribution (most important first)
        sorted_coords = sorted(candidates.keys(), key=lambda x: -candidates[x])
        
        # Try adjusting top coordinates
        for j in sorted_coords[:50]:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-8:
                continue
            
            best_j_obj = M
            best_j_val = old_val
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for _ in range(5):
                perturb_idx = rng.integers(0, n, size=10)
                f_trial = f.copy()
                for j in perturb_idx:
                    f_trial[j] *= (1.0 + rng.standard_normal() * 0.02)
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-15:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
            
            if not improved:
                break
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
