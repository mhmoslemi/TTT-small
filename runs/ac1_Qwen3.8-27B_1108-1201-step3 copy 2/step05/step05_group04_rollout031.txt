```python
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    return np.maximum(v - theta, 0.0)

    Returns the optimal b.

    #   where b[k-i] is 0 if k-i is out of [0, n-1]

    cols = []
```

```python
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                            temp_init=0.008, temp_decay=0.9995, top_k=None):
    """Mirror descent with soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.7]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial /= f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_j_val) / denom
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding (from the
state-of-art LP formulation), combined with Adam subgradient polishing, mirror descent,
coordinate descent, random restarts, and multi-length exploration.

Key innovation: Uses the exact LP-optimal direction (solving max sum(g) s.t. (f*g)_inf
<= M) instead of approximate soft-max subgradients, which should drive faster convergence
to the local minimum.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_lp_matrix(n):
    """Build the sparse convolution constraint matrix structure for length n."""
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    fidx = np.tile(np.arange(n), n)
    return rows, cols, fidx


def solve_lp_direction(f, M, matrix_cache=None):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns (g, sum_g) or (None, 0).
    """
    n = len(f)
    if matrix_cache is None:
        rows, cols, fidx = _conv_lp_matrix(n)
    else:
        rows, cols, fidx = matrix_cache

    A = sparse.coo_matrix((f[fidx], (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if res.status == 0:
        return res.x, res.x.sum()
    return None, 0.0


def lp_iteration(f, time_limit, max_iters=10000, verbose=False, t_base=0.3):
    """
    LP-based iterative optimization.
    At each step: solve LP for optimal direction g, then mix f with g.
    """
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    matrix_cache = _conv_lp_matrix(n)

    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t_step = t_base

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g, sum_g = solve_lp_direction(f, best_M, matrix_cache)
        if g is None or sum_g < 1e-10:
            break

        g = g / sum_g

        # Try to find a good step size via a few candidates
        improved = False
        for tt in [t_step, 0.5 * t_step, 0.2 * t_step, 0.05 * t_step]:
            f_new = (1.0 - tt) * f + tt * g
            M_new = fast_convolve(f_new).max()
            if M_new < best_M * (1.0 - 1e-9):
                f = f_new
                best_M = M_new
                best_f = f.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: M={best_M:.12f}, t={t_step:.4f}")
        else:
            no_improve += 1
            t_step *= 0.7
            if t_step < 0.01:
                # Reset step and add perturbation
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(f + rng.standard_normal(n) * 0.008)
                best_M = fast_convolve(f).max()
                if best_M < best_f.sum() * 0:  # just checking
                    pass
                no_improve = 0
                t_step = t_base

            if no_improve > 200:
                # Try reverse mixing: f_new = (1-tt)*g + tt*f (move more toward g)
                for tt in [0.3, 0.5, 0.7]:
                    f_new = tt * g + (1.0 - tt) * f
                    M_new = fast_convolve(f_new).max()
                    if M_new < best_M * (1.0 - 1e-9):
                        f = f_new
                        best_M = M_new
                        best_f = f.copy()
                        no_improve = 0
                        t_step = t_base
                        break
                else:
                    break

    return best_f, best_M


def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam with soft-max subgradients and simplex projection."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.999,
                            temp_init=0.008, temp_decay=0.9995, top_k=None):
    """Mirror descent (multiplicative weights) with soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    temp = temp_init
    cur_lr = lr_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(len(conv))
            w_masked[idx] = w[idx]
            w_masked /= w_masked.sum()
            w = w_masked

        sg = np.correlate(w, f, 'valid')

        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(t * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            temp = max(temp_init * 0.6, 0.003)
            cur_lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial /= f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_j_val) / denom
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_gaussian_init(n, center=0.5, sigma=0.15):
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'cos':
        f = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP direction finding, Adam/mirror descent,
    coordinate descent, multi-start, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_symmetric_init(768, 'bump')
        print("No parent found, using bump init n=768")

    n_parent = len(parent)
    print(f"Initial best_score={best_score:.8f}, n={n_parent}")

    # === Phase 1: LP-based iteration on parent (primary method) ===
    print(f"\n=== Phase 1: LP iteration on parent (n={n_parent}) ===")
    t1_end = time.monotonic() + 320

    a_lp, m_lp = lp_iteration(parent, time_limit=min(180, t1_end - time.monotonic()),
                              verbose=True, t_base=0.3)
    try_seq(a_lp, "p1_lp")

    # Adam polish on LP result
    rem = t1_end - time.monotonic()
    if rem > 30 and best_seq is not None:
        a_adam, _ = adam_optimize(best_seq, time_limit=min(80, rem * 0.6),
                                  lr=0.004, temp_init=0.004,
                                  temp_decay=0.9996, top_k=max(20, len(best_seq) // 10))
        try_seq(a_adam, "p1_adam")

    # Second LP round with different step
    rem = t1_end - time.monotonic()
    if rem > 20 and best_seq is not None:
        a_lp2, _ = lp_iteration(best_seq, time_limit=min(60, rem * 0.7),
                                verbose=False, t_base=0.15)
        try_seq(a_lp2, "p1_lp2")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: Diverse initializations with LP ===
    print(f"\n=== Phase 2: Diverse inits + LP (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + 280

    inits = {
        'gauss15': make_gaussian_init(n_parent, 0.5, 0.15),
        'gauss25': make_gaussian_init(n_parent, 0.5, 0.25),
        'bump': make_symmetric_init(n_parent, 'bump'),
        'cos': make_symmetric_init(n_parent, 'cos'),
        'rand5': make_smooth_random_init(n_parent, rng, 5),
        'rand8': make_smooth_random_init(n_parent, rng, 8),
        'pert': project_simplex(parent + rng.standard_normal(n_parent) * 0.03),
    }

    for name, init in inits.items():
        rem = t2_end - time.monotonic()
        if rem < 20:
            break
        a_lp, _ = lp_iteration(init, time_limit=min(25, rem * 0.45),
                               verbose=False, t_base=0.3)
        try_seq(a_lp, f"p2_{name}")

        rem = t2_end - time.monotonic()
        if rem > 10:
            a_adam, _ = adam_optimize(a_lp, time_limit=min(15, rem * 0.45),
                                      lr=0.005, temp_init=0.006,
                                      top_k=max(15, n_parent // 15))
            try_seq(a_adam, f"p2_a_{name}")

    base = best_seq if best_seq is not None else base

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 320

    target_lengths = [512, 640, 896, 1024, 1280, 384]
    n_base = len(base)
    target_lengths = [t for t in target_lengths if t != n_base and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - n_base))
    target_lengths = target_lengths[:4]

    for target_n in target_lengths:
        rem = t3_end - time.monotonic()
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)

        a_lp, _ = lp_iteration(f_interp, time_limit=min(40, rem * 0.4),
                               verbose=False, t_base=0.3)
        try_seq(a_lp, f"p3_lp_{target_n}")

        rem = t3_end - time.monotonic()
        if rem > 15:
            a_adam, _ = adam_optimize(a_lp, time_limit=min(30, rem * 0.4),
                                      lr=0.004, temp_init=0.005,
                                      top_k=max(15, target_n // 12))
            try_seq(a_adam, f"p3_a_{target_n}")

    base = best_seq if best_seq is not None else base

    # === Phase 4: Deep LP + Adam on current best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 380

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple LP runs with different perturbations
        for ri in range(4):
            rem = t4_end - time.monotonic()
            if rem < 25:
                break
            noise = rng.standard_normal(n_best) * (0.005 + 0.005 * ri)
            f_pert = project_simplex(best_seq + noise)
            a_lp, _ = lp_iteration(f_pert, time_limit=min(50, rem * 0.25),
                                   verbose=False, t_base=0.2)
            try_seq(a_lp, f"p4_lp{ri}")

        # Fine Adam
        rem = t4_end - time.monotonic()
        if rem > 30:
            a_adam, _ = adam_optimize(best_seq, time_limit=min(80, rem * 0.4),
                                      lr=0.002, temp_init=0.003,
                                      temp_decay=0.9997, top_k=max(8, n_best // 25))
            try_seq(a_adam, "p4_adam_fine")

        # Mirror descent
        rem = t4_end - time.monotonic()
        if rem > 25:
            a_md, _ = mirror_descent_optimize(best_seq, time_limit=min(50, rem * 0.45),
                                              lr_init=0.01, temp_init=0.004,
                                              temp_decay=0.9996)
            try_seq(a_md, "p4_mirror")

        # LP with smaller step for fine-tuning
        rem = t4_end - time.monotonic()
        if rem > 30:
            a_lp3, _ = lp_iteration(best_seq, time_limit=min(60, rem * 0.5),
                                    verbose=False, t_base=0.1)
            try_seq(a_lp3, "p4_lp_fine")

    base = best_seq if best_seq is not None else base

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 130

    if best_seq is not None:
        rem = t5_end - time.monotonic()
        if rem > 10:
            a_cd, _ = coordinate_descent_polish(best_seq, time_limit=min(90, rem * 0.7))
            try_seq(a_cd, "p5_cd")

            # CD from perturbed start
            rem = t5_end - time.monotonic()
            if rem > 10 and best_seq is not None:
                n_best = len(best_seq)
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
                a_cd2, _ = coordinate_descent_polish(pert, time_limit=min(30, rem * 0.6))
                try_seq(a_cd2, "p5_cd2")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # One more LP round
        rem = deadline - time.monotonic() - 15
        if rem > 25:
            a_lp, _ = lp_iteration(best_seq, time_limit=min(35, rem * 0.4),
                                   verbose=False, t_base=0.08)
            try_seq(a_lp, "p6_lp")

        # Ultra-fine Adam
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_adam, _ = adam_optimize(best_seq, time_limit=min(25, rem * 0.5),
                                      lr=0.001, temp_init=0.002,
                                      temp_decay=0.9999, top_k=max(5, n_best // 40))
            try_seq(a_adam, "p6_fine")

        # Tiny perturbations + quick LP
        for pi in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_iteration(f_pert, time_limit=min(12, rem * 0.4),
                                    verbose=False, t_base=0.1)
            try_seq(a_opt, f"p6_p{pi}")

    # Fallback
    if best_seq is None:
        best_seq = make_symmetric_init(768, 'bump')
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
