Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. QP-based alternating minimization (new: more direct than LP-FW)
2. LP-FW with sparse constraint matrix for fast local optimization
3. Peak-targeted perturbation for escaping local optima

6. Coordination of all methods with adaptive time allocation

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def build_toeplitz_sparse(f):
    """Build sparse Toeplitz matrix T where T[k,j] = f[k-j] for valid indices."""
    n = len(f)

    cols = []
    data = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(f)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

            best_val = M
            for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                ft = f + alpha * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 0:
                lo = max(0.0, best_alpha - 0.03)
                hi = min(1.0, best_alpha + 0.03)
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    ft = f + mid * (g - f)
                    ft = np.maximum(ft, 0.0)
                    st = ft.sum()
                    if st < 1e-15:
                        break
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = mid
                        lo = mid
                    else:
                        hi = mid

            if best_alpha > 1e-10:
                f_new = f + best_alpha * (g - f)
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new / f_new.sum()
                conv = fast_convolve(f_new)
                M = conv.max()
                m_exp = 1.0
                f = f_new

                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f_new.copy()
                    no_improve = 0
                    if verbose and it % 100 == 0:
                        print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.1, 5.0)
        else:
            m_exp = min(m_exp * 1.15, 10.0)
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + 3)
                scale = 0.004 + 0.002 * min(no_improve // 100, 5)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0

    return best_f, best_obj

def qp_pg_optimize(f0, time_limit, verbose=False):
    """
    QP-based projected gradient alternating method.
    Alternates between fixing the Toeplitz matrix T (from current f)
    and doing projected gradient steps on the QP: min ||T a - c||^2 s.t. a in simplex.
    This is a different local optimizer than LP-FW.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    target = 1.0 / (2 * n - 1)
    c_vec = np.full(2 * n - 1, target)
    a = f.copy()
    f_fixed = f.copy()

    # Initial step size: use 1/L where L ~ ||T||^2
    # ||T||^2 = max eig of T^T T. Estimate via power iteration.
    v = np.random.default_rng(0).standard_normal(n)
    v /= np.linalg.norm(v)
    eig_est = 0.0
    for _ in range(30):
        Tv = np.convolve(f_fixed, v, mode='full')[:2 * n - 1]
        TtTv = fast_cross_corr(f_fixed, Tv)
        eig_est = np.dot(v, TtTv)
        v = TtTv / (np.linalg.norm(TtTv) + 1e-30)
    L = 2.0 * (eig_est + 1e-10)
    eta = 1.0 / L
    momentum = 0.0
    v_mom = np.zeros(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient of ||T a - c||^2 = 2 * T^T (T a - c)
        Ta = np.convolve(f_fixed, a, mode='full')[:2 * n - 1]
        residual = Ta - c_vec
        Tt_res = fast_cross_corr(f_fixed, residual)
        grad = 2.0 * Tt_res

        # Momentum
        v_mom = momentum * v_mom + grad
        step = v_mom

        # Backtracking line search
        alpha = eta
        obj_curr = np.dot(Ta, Ta) - 2 * target * np.dot(f_fixed, a) + target * target * (2 * n - 1)
        # Simpler: just compute the QP objective directly
        obj_curr = np.sum(residual ** 2)

        found = False
        for _ in range(20):
            a_trial = a - alpha * step
            a_trial = project_simplex(a_trial)
            Ta_trial = np.convolve(f_fixed, a_trial, mode='full')[:2 * n - 1]
            obj_trial = np.sum((Ta_trial - c_vec) ** 2)
            if obj_trial < obj_curr:
                found = True
                break
            alpha *= 0.5

        if found:
            a = a - alpha * step
            a = project_simplex(a)
        else:
            # No progress, update f_fixed and try again
            f_fixed = a.copy()
            # Reset momentum
            v_mom = np.zeros(n)
            momentum = min(momentum + 0.05, 0.7)
            continue

        # Update best
        obj_real = fast_convolve(a).max()
        if obj_real < best_obj:
            best_obj = obj_real
            best_f = a.copy()
            if verbose and it % 200 == 0:
                print(f"    QP-PG iter {it}: obj={best_obj:.10f}")

        # Periodically update f_fixed
        if it % 5 == 0:
            f_fixed = a.copy()

        # Increase momentum slowly
        momentum = min(momentum + 0.001, 0.7)

        # Convergence check
        if it > 200:
            if np.max(np.abs(a - best_f)) < 1e-12:
                break

    return best_f, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth log-sum-exp objective in softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break

        try:
            res = minimize(
                _smooth_obj_grad_impl, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'gtol': 1e-10, 'maxls': 50, 'maxfun': 1500}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def peak_targeted_perturb(f, n_peaks=5, strength=0.005):
    """Perturb f specifically targeting the peaks of the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    # Find top peaks
    peak_indices = np.argsort(conv)[::-1][:n_peaks]
    peak_values = conv[peak_indices]

    # For each peak at position k, the contributing pairs are (j, k-j)
    # We want to reduce mass at positions that heavily contribute to peaks
    reduce_score = np.zeros(n)
    for k in peak_indices:
        w = peak_values[np.where(peak_indices == k)[0][0]] if len(peak_indices) > 0 else 0
        weight = (conv[k] - conv.min()) / (conv.max() - conv.min() + 1e-30)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contributions = f[idx] * f[k - idx]
        # Distribute reduction proportionally to contribution
        total_contrib = contributions.sum() + 1e-30
        reduce_score[idx] += weight * contributions / total_contrib

    # Normalize
    reduce_score = reduce_score / (reduce_score.sum() + 1e-30)

    # Apply: reduce at high-contribution positions, add elsewhere
    perturbation = -strength * reduce_score + strength * (1.0 / n)
    return project_simplex(f + perturbation)


def random_perturb(f, scale, rng):
    """Random perturbation with varying scales."""
    n = len(f)
    noise = rng.standard_normal(n) * scale
    return project_simplex(f + noise)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=None, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(20, n // 8)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with smooth gradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 30.0
    eta = 0.05
    momentum = 0.7
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            eta *= 0.97
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                v = np.zeros(n)
                eta = 0.03
                no_improve = 0

        eta = max(eta, 0.002)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick LP-FW on parent + QP-PG alternative
    Phase 2: L-BFGS with annealing
    Phase 3: Multi-restart with peak-targeted perturbation
    Phase 4: Multi-length exploration
    Phase 5: Deep LP-FW + coordinate descent on best
    Phase 6: Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Quick optimization on parent (350s) ===
    print(f"\n=== Phase 1: Quick Opt (best={best_score:.6f}) ===")

    # 1a: LP-FW on parent (200s)
    t1a = min(200, max(30, deadline - time.monotonic() - 1600))
    if t1a > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1a, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # 1b: QP-PG on parent (150s) - NEW METHOD
    t1b = min(150, max(20, deadline - time.monotonic() - 1450))
    if t1b > 20:
        base = best_seq if best_seq is not None else parent
        a_opt, _ = qp_pg_optimize(base, time_limit=t1b, verbose=True)
        try_seq(a_opt, "qp_pg_parent")

    # === Phase 2: L-BFGS (350s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(350, max(30, deadline - time.monotonic() - 1300))
    if t2 > 30:
        a_opt, _ = lbfqg_optimize(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "lbfqg_main")

        # L-BFGS from perturbed start
        t2b = min(100, max(20, deadline - time.monotonic() - 1200))
        if t2b > 20:
            noise = rng.standard_normal(len(base)) * 0.008
            f_pert = project_simplex(base + noise)
            a_opt, _ = lbfqg_optimize(f_pert, time_limit=t2b)
            try_seq(a_opt, "lbfqg_pert")

    # === Phase 3: Multi-restart with peak-targeted perturbation (400s) ===
    print(f"\n=== Phase 3: Multi-Restart (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_best = len(base)

    restart_attempts = 8
    for ra in range(restart_attempts):
        rem = deadline - time.monotonic() - 900
        if rem < 40:
            break
        t_alloc = min(rem * 0.35, 50)
        if t_alloc < 15:
            break

        if ra % 3 == 0:
            # Peak-targeted perturbation
            f_start = peak_targeted_perturb(base, n_peaks=5 + ra, strength=0.003 + 0.002 * ra)
            label = f"peak_{ra}"
        elif ra % 3 == 1:
            # Random perturbation
            scale = 0.003 + 0.003 * (ra // 3)
            f_start = random_perturb(base, scale, rng)
            label = f"rand_{ra}"
        else:
            # Reversed + small noise
            f_start = base[::-1].copy()
            noise = rng.standard_normal(n_best) * 0.002
            f_start = project_simplex(f_start + noise)
            label = f"rev_{ra}"

        # Alternate between LP-FW and QP-PG
        if ra % 2 == 0:
            a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
        else:
            a_opt, _ = qp_pg_optimize(f_start, time_limit=t_alloc)
        try_seq(a_opt, label)

    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    # Try lengths around the current best
    n_cur = len(base)
    target_lengths = [n_cur - 1, n_cur + 1, n_cur + 2, n_cur - 2, n_cur + 4]
    # Also try some other lengths
    for extra in [640, 832, 896, 1024]:
        if extra not in target_lengths and abs(extra - n_cur) > 50:
            target_lengths.append(extra)

    target_lengths = [tl for tl in target_lengths if 128 <= tl <= 2048]

    for target_n in target_lengths[:5]:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.3, 50)
        if t_alloc < 15:
            continue
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Deep optimization on best (400s) ===
    print(f"\n=== Phase 5: Deep Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: Long LP-FW
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_alloc = min(rem * 0.5, 200)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5b: QP-PG deep
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = qp_pg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_qp")

        # 5c: L-BFGS deep
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 5d: MWU
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = mwu_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_mwu")

        # 5e: Coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate methods with tiny perturbations
        for attempt in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break

            if attempt % 4 == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"fin_lpfw_{attempt}")
            elif attempt % 4 == 1:
                f_start = peak_targeted_perturb(best_seq, n_peaks=3, strength=0.002)
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"fin_peak_{attempt}")
            elif attempt % 4 == 2:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"fin_lbfqg_{attempt}")
            else:
                f_start = best_seq.copy()
                t_alloc = min(rem * 0.6, 40)
                a_opt, _ = sg_polish(f_start, time_limit=t_alloc, lr=0.001,
                                     mom=0.95, top_k=max(10, n_best // 6), temp=0.00002)
                try_seq(a_opt, f"fin_sg_{attempt}")

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
        else:
            best_seq = best_seq  # keep as is

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
