(central binomial coefficients) with adaptive L-BFGS soft-max optimization,
projected subgradient descent, and coordinate descent polishing.
Key insight: the flat-head sequence (whose auto-convolution is perfectly flat 
for the first n lags) provides a near-optimal starting point, and the 
optimization focuses on filling in the tail of the convolution.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def flat_head(n):
    """Generate flat-head sequence using central binomial coefficient recurrence."""
    d = np.zeros(n)
    d[0] = 1.0
    for k in range(1, n):
        # d[k] = (1 - sum_{i=1}^{k-1} d[i]*d[k-i]) / 2
        if k > 1:
            s = np.dot(d[1:k], d[k-1:0:-1])
        else:
            s = 0.0
        d[k] = (1.0 - s) / 2.0
    # Normalize to sum 1
    total = d.sum()
    if total > 1e-15:
        d = d / total
    return d

def project_simplex(v):
