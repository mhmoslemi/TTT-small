```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
```

```python
def solve_lp_subproblem(f, M, n):
    """Solve: max sum(g) s.t. g>=0, sum(g)=1, (f*g)[k]<=M for all k"""
    # Build constraint matrix
    # A[k,j] = f[k-j] if 0<=k-j<n else 0, for k=0..2n-2, j=0..n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: 0 <= g_j <= 1
    bounds = [(0, 1)] * n
    
    # Inequality: A @ g <= M
    b = np.full(2*n-1, M)
    
    # Equality: sum(g) = 1
    A_eq = np.ones((1, n))
    b_eq = np.ones(1)
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x, 0)
        g = g / g.sum()
        return g
    else:
        return f.copy()
```

`= d(a[j]*a[k-j])/d(a[j]) * [if j is in the sum range] + d(a[k-j]*a[j])/d(a[j]) * [if k-j is in the sum range and k-j != j]`
`= a[k-j] * [if max(0,k-n+1) <= j <= min(n-1,k)] + a[k-j] * [if max(0,k-n+1) <= k-j <= min(n-1,k) and k-j != j]`

The first condition: `j` is a valid index AND `a[k-j]` is a valid term, which means `0 <= k-j < n`.
The second condition: `k-j` is valid AND `a[j]` is a valid term (which it always is since 0<=j<n), AND `k-j != j`.

So: `d(conv[k])/d(a[j]) = a[k-j] * [0<=k-j<n] + a[k-j] * [0<=k-j<n and k-j != j]`
`= a[k-j] * [0<=k-j<n] * (1 + (1 if k-j != j else 0))`
`= a[k-j] * [0<=k-j<n] * (2 if k != 2j else 1)`

Great, so the gradient of the smooth max approximation is:

`d(logsumexp(conv)/d(a[j]) = sum_k softmax_k(conv) * a[k-j] * [0<=k-j<n] * (2 if k!=2j else 1)`

This can be computed efficiently using the convolution structure!

Let me define: `h[k] = softmax_k(conv)` (the soft weights on the peaks).

Then: `grad[j] = sum_k h[k] * a[k-j] * [0<=k-j<n] * (2 if k!=2j else 1)`

The first part `sum_k h[k] * a[k-j] * [0<=k-j<n]` is exactly `(h * a)[j]` (a cross-correlation, which is a convolution with h reversed).

The correction for the `k=2j` terms can be handled separately.

So the gradient computation is O(n log n) using FFT!

This is much faster than the LP approach. Let me use this.

**Algorithm: Smoothed max with mirror descent**

1. Start with a good initial sequence
2. At each step:
   a. Compute conv = f*f
   b. Compute soft weights: w = softmax(conv / temperature)
   c. Compute gradient: grad[j] ≈ (w*f)[j] + correction (using FFT)
   d. Update: f = f * exp(-eta * grad), then normalize
   e. Decrease temperature (to make it sharper)
   f. Decrease eta (learning rate)

This is essentially a variant of the multiplicative weight updates / mirror descent.

The temperature controls the smoothness: at high temperature, it's like minimizing the mean of conv (which would be flat); at low temperature, it's like minimizing the max of conv.

This should converge faster than the LP approach.

Let me also add a **coordinate descent polish** at the end, where I do exact 1D optimization on a few key coordinates.

Now let me think about initialization. The geometric sequence a_i = r^i / sum(r^j) is a good start. But what's the optimal r for a given n?

From the analysis above, for a geometric sequence, conv[k] = (k+1)*r^k for k < n. The peak of (k+1)*r^k is at k ≈ 1/(-log(r)) - 1.

If we want the peak to be at k=n/2 (to spread the mass over the whole convolution), we need:
1/(-log(r)) - 1 ≈ n/2
=> -log(r) ≈ 2/(n+2)
=> r ≈ exp(-2/(n+2)) ≈ 1 - 2/(n+2)

For n=768: r ≈ exp(-2/770) ≈ 0.9974

But this is just for the geometric initialization. The actual optimum will be different.

Let me also consider a **symmetric** initialization: peak in the middle, decay both ways.

a_i = exp(-(i - n/2)^2 / (2*sigma^2))

The optimal sigma would be something like n/4 or n/6.

Now let me write the full algorithm:

"""
Minimizes 2*n*max(auto_conv)/sum^2 using smoothed-max mirror descent
with FFT-based gradient computation, geometric/symmetric initializations,
and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n)
    a /= a.sum()
    return a


def symmetric_gaussian(n, sigma_frac=0.3):
    """Gaussian centered in the middle."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    return normalize(a)


def compute_smooth_gradient(f, conv, temperature):
    """
    Compute gradient of logsumexp(conv/temperature) w.r.t. f.
    Uses FFT for efficiency.
    """
    n = len(f)
    
    # Softmax weights
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Main term: (w * f)[j] = sum_k w[k] * f[k-j] (cross-correlation)
    # This is the convolution of w_reversed with f, evaluated at appropriate indices
    # Actually, (w * f)[j] in our convention is sum_k w[k] * f[k-j]
    # which is the cross-correlation of w and f at lag j
    # = convolution of w with f_reversed at position (len-1-j + ...)
    
    # Let's compute via FFT:
    # cross_corr[j] = sum_k w[k] * f[k-j] (with bounds 0<=k-j<n)
    # This is (w * f_rev)[len_w + len_f - 1 - j] where * is linear convolution
    # Actually, let me just use the direct FFT approach:
    # sum_k w[k] * f[k-j] = ifft(fft(w) * fft(f)) evaluated appropriately
    
    # Simpler: pad both to size >= 2n-1, multiply DFTs, inverse DFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    
    # w * f (linear convolution) has length 2n-1
    # (w*f)[m] = sum_k w[k] * f[m-k]
    # We want sum_k w[k] * f[k-j] = (w * f_rev)[n-1-j + n-1] ... 
    # Hmm, let me think differently.
    
    # sum_k w[k] * f[k-j] where 0<=k-j<n
    # Let m = k-j, so k = m+j:
    # = sum_m w[m+j] * f[m] where 0<=m<n and 0<=m+j<2n-1
    # = (f * w_shifted)[j] where w_shifted[m] = w[m+j]
    
    # Alternatively, this is the cross-correlation:
    # corr[j] = sum_m f[m] * w[m+j]
    # = sum_m f[m] * w_reversed[W-1-m-j] where W = 2n-2 (since w has 2n-1 elements)
    
    # Let's use: corr of f and w at lag j = ifft(fft(f) * conj(fft(w))[shifted])
    
    # Actually the cleanest way:
    # corr[j] = sum_m f[m] * w[m+j]
    # This is the (j)-th element of the cross-correlation of f and w
    # Cross-correlation can be computed as: ifft(fft(f, N) * conj(fft(w, N)))
    # But we need to be careful about the indexing.
    
    # Let me just use: the linear cross-correlation of f (length n) and w (length 2n-1)
    # corr[j] = sum_{m=0}^{n-1} f[m] * w[j+m] (for j >= 0)
    #          = sum_{m=0}^{n-1} f[m] * w[|j|+m] ... no, just for j>=0: w[j+m]
    
    # Using FFT: pad f to length 2n-1 (or fft_size), zero-pad w to fft_size
    # Then ifft(FFT(f) * conj(FFT(w))) gives the circular cross-correlation
    # For linear: we need fft_size >= n + (2n-1) - 1 = 3n-2
    
    # Actually, let me just do it with a direct formula for clarity:
    # corr[j] = sum_m f[m] * w[m+j] for m such that 0<=m<n and 0<=m+j<2n-1
    # For j in range 0 to 2n-2-n = n-2... wait no.
    # j can range from 0 to 2n-2 (since w has indices 0 to 2n-2)
    # m ranges from 0 to n-1, and we need m+j <= 2n-2, so m <= 2n-2-j
    
    # For our purpose, j ranges from 0 to n-1 (since f has n elements)
    # Actually, we need grad[j] for j=0 to n-1.
    
    # Let me use a different approach: 
    # grad_main[j] = sum_k w[k] * f[k-j] * [0<=k-j<n]
    # = sum_k w[k] * f[k-j] * [0<=j<=k<n+j]  (since k-j>=0 means k>=j, and k-j<n means k<n+j)
    # For each j, k ranges from j to min(2n-2, n+j-1)
    
    # This is a "sliding dot product" which is a cross-correlation!
    # grad_main = cross_correlation(w, f) evaluated at lags 0 to n-1
    # where cross_correlation(w, f)[j] = sum_k w[k] * f[k-j]
    
    # Using FFT: 
    # N = fft_size >= len(w) + len(f) - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3*n - 2 - 1).bit_length()
    W_pad = np.fft.rfft(w, N)
    F_pad = np.fft.rfft(f, N)
    # Cross-correlation: corr[j] = sum_k w[k] * f[k-j]
    # In frequency: corr = ifft(FFT(w) * conj(FFT(f))) ... no
    # Actually: ifft(FFT(w) * FFT(f_reversed)) gives cross-correlation
    # Or: ifft(conj(FFT(w)) * FFT(f)) ... 
    
    # Let me use the standard formula:
    # cross_corr(x, y)[k] = sum_n x[n] * y[n+k] = (x * y_flipped)[len(y)-1+k]
    # In FFT: cross_corr = ifft(FFT(x) * conj(FFT(y)))
    # This gives: cross_corr[k] = sum_n x[n+k] * conj(y[n]) (for real: x[n+k]*y[n])
    
    # So: ifft(FFT(w) * conj(FFT(f))) [k] = sum_n w[n+k] * f[n]
    #     = sum_n f[n] * w[n+k]
    #     = our desired grad_main[k] for k >= 0!
    
    corr = np.fft.irfft(W_pad * np.conj(F_pad), N)
    grad_main = corr[:n].copy()
    
    # Correction for the k=2j terms: we need to add/subtract a[j] for the diagonal
    # grad[j] = sum_k w[k] * f[k-j] * [0<=k-j<n] * (2 if k != 2j else 1)
    # = 2 * grad_main[j] - w[2j] * f[j] * [2j < 2n-1]
    grad = 2.0 * grad_main
    for j in range(n):
        if 2*j < 2*n - 1:
            grad[j] -= w[2*j] * f[j]
    
    return grad


def smoothed_max_descent(f0, time_limit, n_iters=5000, verbose=False):
    """Mirror descent with smoothed max objective."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    
    if verbose:
        print(f"    Initial score: {best_score:.10f}")
    
    # Temperature schedule: start high, decrease geometrically
    temp_start = 0.01
    temp_end = 0.0001
    # Learning rate schedule
    lr_start = 0.1
    lr_end = 0.001
    
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Progress: 0 to 1
        progress = it / max(1, n_iters)
        
        # Temperature: linear interpolation in log space
        log_temp = np.log(temp_start) + progress * (np.log(temp_end) - np.log(temp_start))
        temperature = np.exp(log_temp)
        
        # Learning rate: decrease
        lr = lr_start * (lr_end / lr_start) ** progress
        
        conv = fast_convolve(f)
        grad = compute_smooth_gradient(f, conv, temperature)
        
        # Multiplicative update (mirror descent with entropy)
        f = f * np.exp(-lr * grad)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Check improvement
        if it % 50 == 0 or it < 5:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        
    # Final check
    score = compute_score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def lp_subproblem(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, sum(g)=1, (f*g)[k]<=M."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    # Build sparse constraint matrix
    # A[k,j] = f[k-j] if 0<=k-j<n else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For this column j, non-zero entries are at rows k where 0<=k-j<n
        # i.e., j <= k <= j+n-1, and k < 2n-1
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            idx = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    b_eq = np.array([1.0])
    A_eq = np.ones((1, n))
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            g /= s
        return g
    else:
        return f.copy()


def lp_alternating(f0, time_limit, max_iters=50, verbose=False):
    """LP-based alternating minimization."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.5
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if verbose:
            print(f"    LP iter {it}: M={M:.8f}, score_so_far={best_score:.8f}, t={t:.4f}")
        
        g = lp_subproblem(f, M, n)
        
        # Try update
        f_new = normalize((1-t) * f + t * g)
        score = compute_score(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
        else:
            # Halve step size
            t *= 0.7
            if t < 0.01:
                break
            f_new = normalize((1-t) * f + t * g)
            score = compute_score(f_new)
            if score < best_score:
                best_score = score
                best_f = f_new.copy()
            f = f_new
    
    return best_f, best_score


def coord_descent_polish(f, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization on top coordinates."""
    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Compute initial convolution to identify important positions
    conv = fast_convolve(f)
    peak_idx = conv.argmax()
    
    # Sort coordinates by their "importance" in the peak
    # Coordinates that contribute most to the peak convolution
    importance = np.zeros(n)
    for j in range(n):
        k = peak_idx
        if 0 <= k - j < n:
            importance[j] = f[j] * f[k-j]
    
    order = np.argsort(-importance)
    
    # Do coordinate descent on the most important coordinates first
    n_passes = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        
        # Cycle through coordinates (most important first in early passes)
        if n_passes <= 3:
            sweep_order = order
        else:
            sweep_order = np.random.permutation(n)
        
        for j in sweep_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            def obj(v):
                v = max(0.0, min(v, 0.5))
                new_f = f.copy()
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                new_f[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_f[mask] *= (1.0 - v) / other_sum
                conv = fast_convolve(new_f)
                return conv.max()
            
            try:
                from scipy.optimize import minimize_scalar
                res = minimize_scalar(obj, bounds=(0.0, min(0.5, 3*old_val + 0.01)),
                                      method='bounded', options={'xatol': 1e-10, 'maxiter': 40})
                if res.success and res.fun < obj(old_val) - 1e-15:
                    v_new = res.x
                    new_f = f.copy()
                    other_sum = 1.0 - old_val
                    new_f[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    new_f[mask] *= (1.0 - v_new) / other_sum
                    new_f = normalize(new_f)
                    sc = compute_score(new_f)
                    if sc < best_score - 1e-14:
                        f = new_f
                        best_score = sc
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                        peak_idx = conv.argmax()
            except:
                pass
        
        if not improved and n_passes > 3:
            break
    
    return best_f, best_score

"""
Minimizes 2*n*max(auto_conv)/sum^2 using a hybrid approach combining:
1. FFT-based smoothed-max mirror descent (fast gradient via cross-correlation)
2. LP-based alternating minimization (HiGHS solver)
3. Exact coordinate descent polish
4. Multi-scale initialization (geometric, symmetric Gaussian)
Key insight: d(conv[k])/d(a[j]) = a[k-j]*(2 if k!=2j else 1), enabling O(n log n) gradients.
"""

import numpy as np
from scipy.optimize import minimize_scalar, linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def symmetric_gaussian(n, sigma_frac=0.3):
    """Gaussian centered in the middle."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def compute_smooth_gradient(f, conv, temperature):
    """
    Compute gradient of logsumexp(conv/temperature) w.r.t. f using FFT.
    Returns ndarray of shape (n,).
    """
    n = len(f)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Need FFT size >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    W_pad = np.fft.rfft(w, N)
    F_pad = np.fft.rfft(f, N)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(f)))[j] = sum_m w[m+j] * f[m]
    corr = np.fft.irfft(W_pad * np.conj(F_pad), N)
    grad_main = corr[:n].copy()

    # Corrections for k=2j diagonal terms
    grad = 2.0 * grad_main
    max_k = 2 * n - 2
    for j in range(n):
        k_val = 2 * j
        if k_val <= max_k:
            grad[j] -= w[k_val] * f[j]

    return grad


def smoothed_max_descent(f0, time_limit, n_iters=5000, verbose=False):
    """Mirror descent with smoothed max objective and adaptive learning rate."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = compute_score(f)

    if verbose:
        print(f"    SM initial score: {best_score:.10f}")

    temp_start = 0.02
    temp_end = 0.001
    lr_start = 0.05
    lr_end = 0.001

    t0 = time.monotonic()
    last_improve_iter = 0
    stall_count = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, n_iters))
        log_temp = np.log(temp_start) + progress * (np.log(temp_end) - np.log(temp_start))
        temperature = np.exp(log_temp)
        lr = lr_start * (lr_end / lr_start) ** progress

        conv = fast_convolve(f)
        grad = compute_smooth_gradient(f, conv, temperature)

        # Adaptive LR: scale by inverse gradient magnitude
        grad_norm = np.sqrt(np.mean(grad ** 2))
        if grad_norm > 1e-12:
            lr_adaptive = lr / grad_norm * 0.1
        else:
            lr_adaptive = lr

        f = f * np.exp(-lr_adaptive * grad)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        if it % 100 == 0 or it < 10:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                last_improve_iter = it
                stall_count = 0
            else:
                stall_count += 1
            if verbose and it % 500 == 0:
                print(f"    SM iter {it}: score={best_score:.10f}, temp={temperature:.6f}")

        # Restart if stalled
        if stall_count > 20 and it > 200:
            f = normalize(best_f + np.random.standard_normal(n) * 0.005)
            stall_count = 0

    score = compute_score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def build_conv_matrix(f, n):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            idx = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[idx])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_subproblem(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, sum(g)=1, (f*g)[k]<=M for all k."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    b_eq = np.array([1.0])
    A_eq = np.ones((1, n))
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                g /= s
                return g
        return f.copy()
    except Exception:
        return f.copy()


def lp_alternating(f0, time_limit, max_iters=100, verbose=False):
    """LP-based alternating minimization with adaptive step size."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.5

    t0 = time.monotonic()

    for it in range(max_iters):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if verbose and it % 5 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best_score={best_score:.10f}, t={t:.4f}")

        g = lp_subproblem(f, M, n)

        # Try multiple step sizes
        best_t = 0.0
        best_new_score = best_score
        for t_try in [t, t * 0.7, t * 0.5, 0.9, 0.7, 0.4]:
            f_new = normalize((1.0 - t_try) * f + t_try * g)
            sc = compute_score(f_new)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t_try

        if best_t > 0:
            f = normalize((1.0 - best_t) * f + best_t * g)
            if best_new_score < best_score:
                best_score = best_new_score
                best_f = f.copy()
            t = min(1.0, best_t * 1.1)  # slight increase
        else:
            # No improvement, try different approach
            t *= 0.5
            if t < 0.02:
                break

    return best_f, best_score


def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization, priority-ordered."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    conv = fast_convolve(f)
    peak_idx = int(conv.argmax())

    # Importance: contribution to peak
    importance = np.zeros(n)
    for j in range(n):
        k_val = peak_idx
        if 0 <= k_val - j < n:
            importance[j] = f[j] * f[k_val - j]
        else:
            importance[j] = f[j] * 0.1

    order = np.argsort(-importance)
    rng = np.random.default_rng(42)
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False

        if n_passes <= 5:
            sweep_order = order
        else:
            sweep_order = rng.permutation(n)

        for j in sweep_order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]

            def obj(v):
                v = max(0.0, min(v, 0.4))
                new_f = f.copy()
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                new_f[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_f[mask] *= (1.0 - v) / other_sum
                conv_new = fast_convolve(new_f)
                return conv_new.max()

            try:
                lo = max(0.0, old_val - 0.05)
                hi = min(0.4, old_val + 0.05)
                if hi - lo < 1e-5:
                    lo = max(0.0, old_val * 0.5)
                    hi = min(0.4, old_val * 2.0 + 0.01)

                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 30})
                if res.success and res.fun < obj(old_val) - 1e-15:
                    v_new = float(res.x)
                    new_f = f.copy()
                    other_sum = 1.0 - old_val
                    new_f[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    new_f[mask] *= (1.0 - v_new) / other_sum
                    new_f = normalize(new_f)
                    sc = compute_score(new_f)
                    if sc < best_score - 1e-14:
                        f = new_f
                        best_score = sc
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                        peak_idx = int(conv.argmax())
                        if n_passes <= 5:
                            for jj in range(n):
                                if 0 <= peak_idx - jj < n:
                                    importance[jj] = f[jj] * f[peak_idx - jj]
                            order = np.argsort(-importance)
            except Exception:
                pass

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_score


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pair_transfer_search(f0, time_limit, n_trials=5000, verbose=False):
    """Randomized pair transfer local search with multiple transfer amounts."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = compute_score(f)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    no_improve = 0

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue

        a_i = f[i]
        a_j = f[j]
        if a_j < 1e-13:
            continue

        best_delta = 0.0
        best_sc = compute_score(f)

        fracs = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
        for fr in fracs:
            delta = a_j * fr
            if delta > a_j:
                continue
            new_f = f.copy()
            new_f[j] = a_j - delta
            new_f[i] = a_i + delta
            conv = fast_convolve(new_f)
            sc = 2.0 * n * conv.max()
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_delta = delta

        if best_delta != 0.0:
            f[j] -= best_delta
            f[i] += best_delta
            f = normalize(f)
            sc = compute_score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                f = best_f.copy()
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 500:
            break

    return best_f, best_score

            break
        t_alloc = min(t_per, rem)
        print(f"  SM descent [{label}] n={n_val} t={t_alloc:.0f}s")
        a_opt, sc = smoothed_max_descent(a_init, time_limit=t_alloc, n_iters=4000, verbose=False)
        try_seq(a_opt, label + "_sm")

    # === Phase 2: LP alternating minimization ===
    print(f"\n=== Phase 2: LP alternating (best={best_score:.8f}, n={len(best_a) if best_a is not None else '?'}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 200.0
        if rem > 30:
            t_alloc = min(rem * 0.6, 350.0)
            print(f"  LP alternating: t={t_alloc:.0f}s, max_iters=50")
            a_opt, sc = lp_alternating(best_a, time_limit=t_alloc, max_iters=50, verbose=True)
            try_seq(a_opt, "lp_alt")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: CD polish (best={best_score:.8f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 80.0
        if rem > 20:
            t_alloc = min(rem * 0.6, 250.0)
            print(f"  CD polish: t={t_alloc:.0f}s")
            a_opt, sc = coord_descent_polish(best_a, time_limit=t_alloc)
            try_seq(a_opt, "cd_polish")

    # === Phase 4: LP again on polished result ===
    print(f"\n=== Phase 4: LP on polished (best={best_score:.8f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 60.0
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            a_opt, sc = lp_alternating(best_a, time_limit=t_alloc, max_iters=30, verbose=False)
            try_seq(a_opt, "lp_polished")

    # === Phase 5: Pair transfer + final CD ===
    print(f"\n=== Phase 5: Pair transfer & final polish (best={best_score:.8f}) ===")

    if best_a is not None:
        rem = deadline - time.monotonic() - 30.0
        if rem > 20:
            t_ps = min(rem * 0.4, 120.0)
            a_opt, sc = pair_transfer_search(best_a, time_limit=t_ps, n_trials=3000)
            try_seq(a_opt, "pair_transfer")

        rem = deadline - time.monotonic() - 15.0
        if rem > 15:
            t_cd = min(rem, 100.0)
            a_opt, sc = coord_descent_polish(best_a, time_limit=t_cd)
            try_seq(a_opt, "final_cd")

    # === Phase 6: Try different dimensions on best ===
    print(f"\n=== Phase 6: Dimension exploration (best={best_score:.8f}) ===")

    if best_a is not None:
        best_n = len(best_a)
        for target_n in [best_n - 128, best_n + 128, best_n - 64, best_n + 64]:
            if target_n < 256 or target_n > 2048:
                continue
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 30.0
            if rem < 30:
                break
            # Interpolate
            old_x = np.linspace(0, 1, best_n)
            new_x = np.linspace(0, 1, target_n)
            a_interp = np.interp(new_x, old_x, best_a)
            a_interp = normalize(a_interp)
            t_alloc = min(rem * 0.3, 80.0)
            a_opt, sc = smoothed_max_descent(a_interp, time_limit=t_alloc, n_iters=2000)
            a_opt2, sc2 = coord_descent_polish(a_opt, time_limit=30.0)
            try_seq(a_opt2, f"dim_{target_n}")

    # === Final verification ===
    if best_a is None:
        best_a = geometric_sequence(768, 0.998)
        best_score = compute_score(best_a)

    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
        print(f"\n  evaluate_sequence confirms: {final_score:.8f}")

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
