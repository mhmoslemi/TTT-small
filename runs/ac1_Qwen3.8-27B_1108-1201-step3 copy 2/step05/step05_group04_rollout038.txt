def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

def lp_direction(a, M, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b and the optimal value."""
    n = len(a)
    
    # Build constraint matrix: A[m, j] = a[m-j] if 0 <= m-j < n
    # Number of rows: 2n-1, columns: n
    # Use sparse format
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        for k in range(n):
            m = j + k
            if m < 2 * n - 1:
                rows.append(m)
                cols.append(j)
                vals.append(a[k])  # a[m-j] = a[k]
    
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(b) = c^T b where c = ones(n)
    # => minimize -c^T b
    c = -np.ones(n)
    
    # Constraints: A @ b <= M * ones(2n-1), b >= 0
    b_ub = M * np.ones(2 * n - 1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x, -res.fun
    else:
        # Fallback: return uniform
        return np.ones(n) / n, 1.0

    return w

    else:

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-12:
                continue
            best_val = old
            best_obj_j = fast_convolve(f).max()
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                nv = old * mult
                if nv < 0 or nv > 0.99:
                    continue
                denom = 1.0 - old
                if denom < 1e-10:
                    continue
                scale = (1.0 - nv) / denom
                ft = f.copy()
                ft[j] = nv
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft /= ft.sum()
                ov = fast_convolve(ft).max()
                if ov < best_obj_j - 1e-15:
                    best_obj_j = ov
                    best_val = nv
            
            if best_val != old:
                denom = 1.0 - old
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                if fast_convolve(f).max() < best_obj - 1e-14:
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_bump_init(n):
    """Sin bump initialization."""
    x = np.linspace(0, 1, n)
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random(n, rng, n_comp=5):
    """Smooth random init as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_comp):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: LP direction finding, dual-guided subgradients,
    Adam/mirror descent, Polyak subgradient, length exploration, CD polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")
    
    n_parent = len(parent)
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    
    # === Phase 1: LP-based optimization on parent ===
    print(f"\n=== Phase 1: LP optimization (n={n_base}) ===")
    t1_end = time.monotonic() + 500
    rem = t1_end - time.monotonic()
    if rem > 20:
        print(f"  LP optimize, budget={rem:.0f}s")
        a_opt, obj = lp_optimize(base, time_limit=rem * 0.85, verbose=True)
        try_seq(a_opt, "lp_parent")
        if best_seq is not None:
            base = best_seq
            n_base = len(base)
    
    # === Phase 2: Diverse initializations + Adam ===
    print(f"\n=== Phase 2: Diverse inits + Adam (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + 300
    
    inits = {
        'parent': base.copy(),
        'gauss': make_gaussian_init(n_base, 0.5, 0.18),
        'bump': make_bump_init(n_base),
        'smooth': make_smooth_random(n_base, rng, 5),
        'pert': project_simplex(base + rng.standard_normal(n_base) * 0.01),
    }
    
    for name, init in inits.items():
        rem = t2_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 55)
        a_opt, _ = adam_optimize(init, t_alloc, lr=0.004, temp_init=0.006,
                                  temp_decay=0.9993, top_k=max(15, n_base // 12))
        try_seq(a_opt, f"adam_{name}")
    
    # Mirror descent on best
    rem = t2_end - time.monotonic()
    if rem > 20 and best_seq is not None:
        a_opt, _ = mirror_descent(best_seq, min(rem * 0.8, 60),
                                   lr_init=0.015, temp_init=0.006, temp_decay=0.9994)
        try_seq(a_opt, "mirror_best")
    
    # === Phase 3: Polyak subgradient ===
    print(f"\n=== Phase 3: Polyak subgradient (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 250
    if best_seq is not None:
        rem = t3_end - time.monotonic()
        if rem > 20:
            a_opt, _ = polyak_subgradient(best_seq, min(rem * 0.85, 200))
            try_seq(a_opt, "polyak")
    
    # === Phase 4: LP on current best ===
    print(f"\n=== Phase 4: LP on best (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 300
    if best_seq is not None:
        rem = t4_end - time.monotonic()
        if rem > 30:
            a_opt, _ = lp_optimize(best_seq, min(rem * 0.85, 250))
            try_seq(a_opt, "lp_best")
    
    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    t5_end = time.monotonic() + 200
    if best_seq is not None:
        target_ns = [512, 640, 896, 1024, 1280, 1536]
        target_ns = [t for t in target_ns if t != len(best_seq) and 256 <= t <= 2048]
        target_ns.sort(key=lambda x: abs(x - len(best_seq)))
        target_ns = target_ns[:4]
        
        for tn in target_ns:
            rem = t5_end - time.monotonic()
            if rem < 30:
                break
            fi = interpolate_sequence(best_seq, tn)
            # Quick LP
            t_lp = min(rem * 0.4, 60)
            a_opt, _ = lp_optimize(fi, t_lp)
            try_seq(a_opt, f"lp_n{tn}")
            # Quick Adam
            rem = t5_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                a_opt, _ = adam_optimize(best_seq, min(rem * 0.3, 30),
                                          lr=0.003, temp_init=0.005, temp_decay=0.9994)
                try_seq(a_opt, f"adam_n{tn}")
    
    # === Phase 6: Coordinate descent polishing ===
    print(f"\n=== Phase 6: CD polishing (best={best_score:.8f}) ===")
    t6_end = time.monotonic() + 150
    if best_seq is not None:
        rem = t6_end - time.monotonic()
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, min(rem * 0.8, 120))
            try_seq(a_opt, "cd")
    
    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            # Fine Adam
            a_opt, _ = adam_optimize(best_seq, min(rem * 0.4, 40),
                                      lr=0.001, temp_init=0.002, temp_decay=0.9998,
                                      top_k=max(10, n_best // 25))
            try_seq(a_opt, "fine_adam")
            # Fine LP
            rem = deadline - time.monotonic() - 10
            if rem > 20:
                a_opt, _ = lp_optimize(best_seq, min(rem * 0.7, 60))
                try_seq(a_opt, "fine_lp")
            # Tiny perturbations
            rem = deadline - time.monotonic() - 5
            for pi in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.002
                fp = project_simplex(best_seq + noise)
                a_opt, _ = adam_optimize(fp, min(rem * 0.5, 15),
                                          lr=0.002, temp_init=0.003, temp_decay=0.9997)
                try_seq(a_opt, f"pert{pi}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final check
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
