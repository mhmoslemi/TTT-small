import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

        f = f / s
    n = len(f)

    best_obj = M

    no_improve = 0

def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """Optimize using L-BFGS on smooth approximation with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    eps = 1e-15
    t = np.log(f + eps)
    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()
    total_time = time_limit

    for bi, beta in enumerate(betas):
        remaining = total_time - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)
        max_iter = max(50, int(sub_time * 3))
        try:
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-10, 'maxls': 50, 'maxfun': max_iter * 5}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")

    return best_f, best_obj

def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Now let me think about the coordinate descent more carefully. I want an efficient version.

For coordinate j, the objective as a function of x = a_j (with all other a_i fixed) is:
h(x) = 2n * max_k [conv_a[k] + c0[k] + c1[k]*x + c2[k]*x^2] / (S - a_j + x)^2

where S = sum(a), and the c's are as defined before.

The numerator is the max of linear (or quadratic) functions of x, so it's convex in x.
The denominator is (e + x)^2 where e = S - a_j > 0 (assuming a_j < S).

h(x) = g(x) / (e + x)^2 where g(x) = max_k [conv_a[k] + c0[k] + c1[k]*x + c2[k]*x^2]

For the minimum of h(x) over x >= 0:
- h'(x) = [g'(x)(e+x) - 2*g(x)] / (e+x)^3
- Setting h'(x) = 0: g'(x)(e+x) = 2*g(x)

Since g is piecewise linear/quadratic, g' is piecewise constant/linear.
In each piece (where a particular k is the argmax):
g(x) = conv_a[k] + c0[k] + c1[k]*x + c2[k]*x^2
g'(x) = c1[k] + 2*c2[k]*x

So: (c1[k] + 2*c2[k]*x)(e+x) = 2(conv_a[k] + c0[k] + c1[k]*x + c2[k]*x^2)
c1[k]*e + c1[k]*x + 2*c2[k]*e*x + 2*c2[k]*x^2 = 2*conv_a[k] + 2*c0[k] + 2*c1[k]*x + 2*c2[k]*x^2
c1[k]*e - c1[k]*x + 2*c2[k]*e*x = 2*conv_a[k] + 2*c0[k]
x*(2*c2[k]*e - c1[k]) = 2*(conv_a[k] + c0[k]) - c1[k]*e
x = [2*(conv_a[k] + c0[k]) - c1[k]*e] / [2*c2[k]*e - c1[k]]

For most k (c2[k]=0): x = [2*(conv_a[k] + c0[k]) - c1[k]*e] / [-c1[k]] = e - 2*(conv_a[k]+c0[k])/c1[k]

For k=2j (c2[k]=1, c1[k]=0): x = 2*(conv_a[2j]+c0[2j]) / (2*e) = (conv_a[2j]+c0[2j])/e

So for each k, there's at most one critical point. The optimal x is the minimum of h(x) over all these critical points (that are >= 0) and x=0.

The challenge is that we have 2n-1 potential k's, and we need to check which one is actually the argmax at the candidate x. This requires O(n) work per candidate, and we have O(n) candidates, giving O(n^2) per coordinate update.

For n=768, that's about 590K operations per coordinate, and 768 coordinates per sweep, so about 450M per sweep. Each "operation" involves a few flops, so maybe 1-2 seconds per sweep on a modern CPU. We could do about 100 sweeps in 1800 seconds. That's reasonable!

But actually, we can be smarter: we only need to consider the top ~20-50 k's by conv value, since those are the ones that can be the argmax. This reduces it to O(n * top_k) per coordinate.

            if time.monotonic() - t0 > time_limit:
                break

            e = 1.0 - old_val  # S - a_j, where S=1
            
            if e < 1e-10:

            # Compute c0[k], c1[k], c2[k] for the top-k indices
            conv_vals = conv.copy()

            # if k == 2*j and 2*j < 2*n-1: c1[k] = 0, c0[k] = -a[j]^2, c2[k] = 1
            # else: c0[k] = c1[k] = c2[k] = 0

            # Get top indices by current conv value
            if top_k < k_total:
                top_idx = np.argpartition(conv, -top_k)[-top_k:]
            else:
                top_idx = np.arange(k_total)
            
            candidates = [0.0, old_val]  # always include 0 and current value
            
            for k_idx in top_idx:
                k = int(k_idx)
                # Determine c0, c1, c2 for this k
                ki = k - j  # k - j, the index of a in the cross term
                if 0 <= ki < n:

                        c0 = -old_val * old_val
                        c1 = 0.0
                        c2 = 1.0

                        c1 = 2.0 * f[ki]
                        c2 = 0.0
                else:

                    c1 = 0.0
                    c2 = 0.0

                    if x_crit > 0 and x_crit < 10.0:  # reasonable range
                        candidates.append(x_crit)
            
            # Add some heuristic candidates
            candidates.append(old_val * 0.5)
            candidates.append(old_val * 1.5)

            best_x = old_val
            best_h = fast_convolve(f).max()  # current value
            
            for x in candidates:
                if x < 0:
                    continue
                # conv_new[k] = conv[k] + c0[k] + c1[k]*x + c2[k]*x^2
                # We need the max over all k, not just top_k

                max_top = conv_new_top.max()

                # (from the c0 and c1 terms), so if max_top is well above the next-highest
                # original conv value, we're safe.

                f[j] = best_x

                if best_x < 1.0 - 1e-10:

                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based Frank-Wolfe (primary local optimizer)
2. L-BFGS on smoothed log-sum-exp objective in softmax parameterization
3. Analytic coordinate descent (exact 1D subproblem)
4. Multi-length exploration with diverse parametric initializations
5. Iterative reweighting with aggressive line search
"""

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
