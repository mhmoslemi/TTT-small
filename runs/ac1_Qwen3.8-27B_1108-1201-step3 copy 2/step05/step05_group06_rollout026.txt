import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_of_max(a, k_star):
    """Compute subgradient of max_k (a*a)[k] at a where max is at k_star."""
    n = len(a)
    # grad[m] = 2 * a[k_star - m] if 0 <= k_star - m <= n-1, else 0
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    # For m in [lo, hi], k_star - m is in [0, n-1]
    # grad[m] = 2 * a[k_star - m]
    indices = np.arange(lo, hi + 1)
    grad[indices] = 2.0 * a[k_star - indices]
    return grad

    if s < 1e-15:

        spread = max_val - sorted_conv[n_top]
        rel_spread = spread / max_val if max_val > 0 else 1.0
        
        # Adaptive temperature
        base_T = T_start * (T_min / T_start) ** (progress ** 0.7)
        if rel_spread < 0.005:
            T = max(base_T * 2.0, T_min)
        elif rel_spread < 0.02:
            T = max(base_T * 1.3, T_min)
        else:
            T = max(base_T, T_min)
        
        # Smooth weights
        scaled = conv / T
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()
        
        # Gradient via FFT
        grad = smooth_grad_fft(f, w)
        
        # Adaptive learning rate
        lr = alpha * (1.0 + progress) ** (-0.5) * (1.0 - 0.5 * progress)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    SS iter {it}: obj={best_obj:.10f}, T={T:.5f}, spread={rel_spread:.5f}")
        else:
            no_improve += 1
        
        # Restart strategies
        if no_improve > 1500:
            strategy = it % 4
            if strategy == 0:
                noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise)
            elif strategy == 1:
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase) * 0.004
                f = project_simplex(best_f + pert)
            elif strategy == 2:
                # Block perturbation
                blk = max(1, n // 8)
                idx = int(rng.integers(0, n - blk))
                pert = np.zeros(n)
                pert[idx:idx + blk] = rng.uniform(0.003, 0.008)
                f = project_simplex(best_f + pert)
            else:
                # Edge boost
                pert = np.zeros(n)
                edge = max(1, n // 10)
                pert[:edge] = 0.005
                pert[-edge:] = 0.005
                f = project_simplex(best_f + pert)
            
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP Frank-Wolfe with adaptive active set and careful line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 0.96
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Use active set for speed, fallback to full
        g = active_set_lp(f, M * m_factor, n, active_frac=0.4)
        if g is None and m_factor < 0.99:
            g = active_set_lp(f, M * m_factor, n, active_frac=1.0)
        if g is None:
            m_factor = min(m_factor * 1.05, 1.5)
            if m_factor > 2.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 5e-11:
            no_improve += 1
            if no_improve > 100:
                break
            m_factor = min(m_factor * 1.02, 1.5)
            continue
        
        # Line search with more points
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0]
        for a_val in alphas:
            ft = f + a_val * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = a_val
        
        # Refine with golden section
        if best_alpha > 0.01:
            phi = 0.6180339887
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(12):
                c = hi - phi * (hi - lo)
                d = lo + phi * (hi - lo)
                ft_c = f + c * (g - f)
                ft_c = np.maximum(ft_c, 0.0)
                st_c = ft_c.sum()
                val_c = fast_convolve(ft_c / st_c).max() if st_c > 1e-15 else 1e10
                ft_d = f + d * (g - f)
                ft_d = np.maximum(ft_d, 0.0)
                st_d = ft_d.sum()
                val_d = fast_convolve(ft_d / st_d).max() if st_d > 1e-15 else 1e10
                if val_c < val_d:
                    hi = d
                else:
                    lo = c
            best_alpha = (lo + hi) / 2.0
            ft = f + best_alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st > 1e-15:
                best_val = fast_convolve(ft / st).max()
        
        if best_alpha > 1e-10 and best_val < M:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 0.96
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.04, 2.0)
        
        if no_improve > 80:
            strat = it % 3
            if strat == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            elif strat == 1:
                freq = rng.uniform(1, 4)
                pert = np.sin(2 * np.pi * freq * np.arange(n) / n) * 0.004
                f = project_simplex(best_f + pert)
            else:
                # Cross with geometric
                r = rng.uniform(0.97, 0.999)
                geo = r ** np.arange(n)
                geo = geo / geo.sum()
                mix = 0.9 * best_f + 0.1 * geo
                f = project_simplex(mix)
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 0.96
            no_improve = 0
    
    return best_f, best_obj


def cd_refine(f0, time_limit):
    """Coordinate descent refinement focusing on high-impact coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        max_val = conv.max()
        
        # Priority: coordinates involved in near-max convolution
        near_max = np.where(conv >= max_val * 0.98)[0]
        priority = set()
        for k in near_max:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            priority.update(range(lo, hi + 1))
        
        order = sorted(priority) + [j for j in range(n) if j not in priority]
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            best_x = old_val
            best_v = max_val
            
            # Try multiplicative perturbations
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 0.0]:
                x = old_val * mult
                if x > 0.9:
                    x = 0.9
                scale = (1.0 - x) / denom
                f_trial = f.copy()
                f_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                v = fast_convolve(f_trial).max()
                if v < best_v - 1e-15:
                    best_v = v
                    best_x = x
            
            if best_x != old_val:
                scale = (1.0 - best_x) / denom
                f[j] = best_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                max_val = best_v
                if best_v < best_obj - 1e-15:
                    best_obj = best_v
                    best_f = f.copy()
                improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def make_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    for r in [0.96, 0.97, 0.98, 0.99, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        inits.append(a / a.sum())
    for pw in [1.5, 2.0, 3.0, 5.0]:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** pw + 0.01
        inits.append(a / a.sum())
    # Symmetric
    for r in [0.97, 0.99]:
        half = (n + 1) // 2
        asc = r ** np.arange(half)
        desc = r ** np.arange(half - 1, 0, -1)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            pad = np.full(n - len(a), a[-1] * 0.5)
            a = np.concatenate([a, pad])
        else:
            a = a[:n]
        inits.append(a / a.sum())
    # Random smooth
    for _ in range(2):
        a = np.abs(rng.standard_normal(n)) ** 0.5
        k = np.array([1, 2, 3, 2, 1], dtype=float)
        a = np.convolve(a, k / k.sum(), mode='same')
        a = np.maximum(a, 0)
        if a.sum() > 0:
            inits.append(a / a.sum())
    # Edge-weighted
    x = np.linspace(0, 1, n)
    for c in [2.0, 4.0, 6.0]:
        a = (1 - np.abs(2 * x - 1)) ** c + 0.01
        inits.append(a / a.sum())
    return inits


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer: diverse inits -> spread-aware smooth -> 
    active-set LP-FW -> CD refine -> multi-length exploration -> final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def _score(a):
        try:
            return float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
    
    def _try(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            old = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) [was {old:.6f}]")
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            psc = _score(parent)
            print(f"Parent: n={len(parent)}, score={psc:.6f}")
            if np.isfinite(psc):
                best_seq = parent.copy()
                best_score = psc
    except Exception:
        pass
    
    # Phase 0: Try diverse inits at parent length and nearby
    print("=== Phase 0: Initializations ===")
    n_parent = len(parent) if parent is not None else 768
    
    trial_lengths = [n_parent]
    if n_parent != 512:
        trial_lengths.append(512)
    if n_parent != 768:
        trial_lengths.append(768)
    
    for n_try in trial_lengths:
        inits = make_inits(n_try, rng)
        for i, init in enumerate(inits):
            _try(init, f"init_{n_try}_{i}")
    
    # Also try parent-based perturbations
    if parent is not None:
        for _ in range(5):
            pert = parent + rng.standard_normal(n_parent) * 0.005
            _try(project_simplex(pert), "parent_pert")
    
    print(f"  Phase 0 done: best={best_score:.6f}")
    
    # Phase 1: Spread-aware smooth on best
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t1 = min(rem * 0.3, 400)
        a_opt, _ = spread_aware_smooth(best_seq, time_limit=t1, T_start=0.6, T_min=0.003, verbose=True)
        _try(a_opt, "smooth_p1")
    
    # Phase 2: LP-FW
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t2 = min(rem * 0.25, 300)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        _try(a_opt, "lpfw_p2")
    
    # Phase 3: CD
    print(f"\n=== Phase 3: CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t3 = min(rem * 0.15, 150)
        a_opt, _ = cd_refine(best_seq, time_limit=t3)
        _try(a_opt, "cd_p3")
    
    # Phase 4: Second round of smooth + LP-FW on current best
    print(f"\n=== Phase 4: Second Round (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.25, 250)
        a_opt, _ = spread_aware_smooth(best_seq, time_limit=t4, T_start=0.2, T_min=0.002, verbose=True)
        _try(a_opt, "smooth_p4")
    
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.3, 200)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=True)
        _try(a_opt, "lpfw_p4")
    
    # Phase 5: Try other lengths
    print(f"\n=== Phase 5: Other Lengths (best={best_score:.6f}) ===")
    n_best = len(best_seq) if best_seq is not None else 768
    other_lengths = [512, 640, 768, 896, 1024, 1280]
    other_lengths = [nl for nl in other_lengths if nl != n_best]
    
    for target_n in sorted(other_lengths, key=lambda x: abs(x - n_best)):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        _try(f_interp, f"interp_n{target_n}")

            _try(a_opt, f"smooth_n{target_n}")

            t_l = min(rem * 0.3, 80)
            a_opt, _ = lp_fw_optimize(best_seq if len(best_seq) == target_n else f_interp, time_limit=t_l)
            _try(a_opt, f"lpfw_n{target_n}")

        # Multiple short smooth runs with different T
        rem = deadline - time.monotonic()

            t_f = min(rem * 0.2, 80)
            a_opt, _ = spread_aware_smooth(best_seq, time_limit=t_f, T_start=0.05, T_min=0.001, verbose=False)
            _try(a_opt, "smooth_fine")
        
        rem = deadline - time.monotonic()

        rem = deadline - time.monotonic()
        if rem > 20:
            t_f = min(rem * 0.2, 40)
            a_opt, _ = cd_refine(best_seq, time_limit=t_f)
            _try(a_opt, "cd_fine")
        
        # Structured perturbations + quick optimize
        for pi in range(6):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            strat = pi % 3
            if strat == 0:
                pert = best_seq + rng.standard_normal(n_fin) * 0.002
            elif strat == 1:
                freq = rng.uniform(0.5, 6.0)
                pert = best_seq + np.sin(2 * np.pi * freq * np.arange(n_fin) / n_fin) * 0.003
            else:
                blk = max(1, n_fin // 6)
                idx = int(rng.integers(0, n_fin - blk))
                pert = best_seq.copy()
                pert[idx:idx + blk] *= 1.1
            f_p = project_simplex(pert)
            t_small = min(rem * 0.25, 30)
            a_opt, _ = spread_aware_smooth(f_p, time_limit=t_small, T_start=0.03, T_min=0.001)
            _try(a_opt, f"perturb_{pi}")
        
        # Reversed
        rem = deadline - time.monotonic()
        if rem > 15:
            f_rev = best_seq[::-1].copy()
            t_small = min(rem * 0.3, 25)
            a_opt, _ = spread_aware_smooth(f_rev, time_limit=t_small, T_start=0.02, T_min=0.001)
            _try(a_opt, "reversed")
        
        # Blend with uniform
        rem = deadline - time.monotonic()
        if rem > 12:
            f_blend = 0.93 * best_seq + 0.07 * (np.ones(n_fin) / n_fin)
            f_blend = project_simplex(f_blend)
            t_small = min(rem * 0.4, 18)
            a_opt, _ = spread_aware_smooth(f_blend, time_limit=t_small, T_start=0.02, T_min=0.001)
            _try(a_opt, "blend")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)
    
    # Final check
    final_score = _score(best_seq)
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
