def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)

    z_shifted = z - z.max()

def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def minimize_fun(z, T, n, t0, time_limit):
    """Objective function for scipy minimize (no closure)."""
    if time.monotonic() - t0 > time_limit:
        return 1e15
    obj, _ = soft_max_obj_grad(z, T, n)
    return obj


def minimize_jac(z, T, n, t0, time_limit):
    """Jacobian function for scipy minimize (no closure)."""
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z)
    _, g = soft_max_obj_grad(z, T, n)
    return g


def lbgfs_run(z0, T, time_limit, maxiter=5000):
    """Run L-BFGS optimization in log-space."""
    n = len(z0)
    t0 = time.monotonic()
    try:
        res = minimize(
            fun=minimize_fun, x0=z0, jac=minimize_jac,
            args=(T, n, t0, time_limit),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14,
                     'gtol': 1e-10, 'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x
    except Exception:
        return z0


def focused_peak_reduce(a0, time_limit, top_k):
    """Reduce top-k peaks by adjusting contributing elements."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()

        active = np.argsort(b)[-top_k:]
        improved = False

        for k_idx in active:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k_idx)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]

            for attempt in range(5):
                if time.monotonic() - t0 > time_limit:
                    break
                if attempt < len(idx):
                    max_local = np.argmax(contributions)
                else:
                    max_local = np.random.randint(len(idx))
                i = idx[max_local]
                j = k - i

                targets = [i, j] if i != j else [i]
                for target in targets:
                    if a[target] < 1e-14:
                        continue
                    for factor in [0.85, 0.7, 0.5, 0.3]:
                        new_val = a[target] * factor
                        diff = a[target] - new_val
                        if diff < 1e-15:
                            continue
                        mask = np.ones(n, dtype=bool)
                        mask[target] = False
                        others_sum = a[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[target] = new_val
                        a_trial[mask] += diff * a[mask] / others_sum
                        a_trial = a_trial / a_trial.sum()
                        new_max = fast_convolve(a_trial).max()
                        if new_max < M - 1e-13:
                            a = a_trial
                            M = new_max
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and iteration % 100 == 50:
            rng = np.random.default_rng(iteration * 7)
            pert = rng.standard_normal(n) * 0.002
            a_trial = project_simplex(a + pert)
            if fast_convolve(a_trial).max() < M:
                a = a_trial

    return best_a / best_a.sum()


def coordinate_descent_focused(a0, time_limit):
    """Coordinate descent focusing on elements near active peaks."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = np.array([0.4, 0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0])
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(a)
        M = b.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        active_lags = np.where(b > M * 0.75)[0]
        relevant = set()
        for k in active_lags:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            relevant.update(range(js, je + 1))

        for j in sorted(relevant):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            best_j_obj = M
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.01 or scale > 100:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                improved = True

        if not improved:
            if no_improve_count > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % 1000000)
                j = int(rng.integers(n))
                a[j] *= 1.0 + float(rng.standard_normal()) * 0.08
                a = project_simplex(a)
                no_improve_count = 0

    return best_a / best_a.sum()


def subgradient_mirror(a0, time_limit, lr, mom, top_k):
    """Mirror descent (exponentiated gradient) on the simplex."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.002

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a_new = a * np.exp(v)
        a_new = a_new / a_new.sum()
        b_new = fast_convolve(a_new)
        co = b_new.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1

        cur_lr *= 0.998
        cur_temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_a / best_a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: asymmetric starts, adaptive-temperature L-BFGS,
    focused peak reduction, coordinate descent, subgradient mirror descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Diverse asymmetric starts ===
    print(f"\n=== Phase 1: Asymmetric starts (time left: {deadline - time.monotonic():.0f}s) ===")

    starts = [parent.copy()]
    x = np.linspace(-1, 1, n_parent)

    for eps in [0.01, 0.03, 0.05, 0.08, 0.12]:
        tilted = parent * (1.0 + eps * x)
        tilted = np.maximum(tilted, 0.0)
        if tilted.sum() > 1e-10:
            starts.append(tilted / tilted.sum())

    for eps in [0.02, 0.05, 0.1]:
        pert = parent * (1.0 + eps * (x ** 2 - 1.0 / 3.0))
        pert = np.maximum(pert, 0.0)
        if pert.sum() > 1e-10:
            starts.append(pert / pert.sum())

    for frac in [0.03, 0.05, 0.08]:
        k = int(n_parent * frac)
        if k > 0 and k < n_parent:
            moved = parent[:k].copy()
            s1 = parent.copy()
            s1[:k] = 0
            s1[k:] += moved * (n_parent / (n_parent - k))
            if s1.sum() > 1e-10:
                starts.append(s1 / s1.sum())
            moved2 = parent[-k:].copy()
            s2 = parent.copy()
            s2[-k:] = 0
            s2[:-k] += moved2 * (n_parent / (n_parent - k))
            if s2.sum() > 1e-10:
                starts.append(s2 / s2.sum())

    for r in range(4):
        noise = rng.standard_normal(n_parent) * 0.08
        st = parent * (1.0 + noise)
        st = np.maximum(st, 0.0)
        if st.sum() > 1e-10:
            starts.append(st / st.sum())

    t1_total = min(300.0, (deadline - time.monotonic()) * 0.12)
    t1_per = max(5.0, t1_total / len(starts))

    for idx, start in enumerate(starts):
        rem = deadline - time.monotonic() - 400
        if rem < 10:
            break
        t_alloc = min(t1_per, rem * 0.15)
        if t_alloc < 5:
            break
        z0 = z_from_a(start)
        z_opt = lbgfs_run(z0, T=15, time_limit=t_alloc, maxiter=2000)
        a_opt = softmax_from_z(z_opt)
        try_seq(a_opt, f"s{idx}_T15")

    # === Phase 2: Deep L-BFGS with temperature annealing ===
    print(f"\n=== Phase 2: Deep L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        z_curr = z_from_a(best_seq)
        temps = [8, 20, 50, 120, 300]

        for T in temps:
            rem = deadline - time.monotonic() - 180
            if rem < 15:
                break
            t_alloc = min(rem * 0.22, 120.0)
            if t_alloc < 10:
                break
            print(f"  L-BFGS T={T}, n={n_best}, t={t_alloc:.0f}s")
            z_curr = lbgfs_run(z_curr, T=T, time_limit=t_alloc, maxiter=5000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"deep_T{T}")

        rem = deadline - time.monotonic() - 120
        if rem > 15:
            t_alloc = min(rem * 0.35, 80.0)
            z_curr = lbgfs_run(z_curr, T=600, time_limit=t_alloc, maxiter=5000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, "deep_T600")

    # === Phase 3: Focused peak reduction ===
    print(f"\n=== Phase 3: Peak reduction (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_pr = min(rem * 0.35, 80.0)
            a_pr = focused_peak_reduce(best_seq, time_limit=t_pr, top_k=3)
            try_seq(a_pr, "peak3")

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_pr = min(rem * 0.4, 50.0)
            a_pr = focused_peak_reduce(best_seq, time_limit=t_pr, top_k=5)
            try_seq(a_pr, "peak5")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_cd = min(rem * 0.45, 60.0)
            a_cd = coordinate_descent_focused(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 5: Subgradient mirror descent ===
    print(f"\n=== Phase 5: Mirror descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_sg = min(rem * 0.5, 50.0)
            top_k_sg = max(10, n_best // 20)
            a_sg = subgradient_mirror(best_seq, time_limit=t_sg,
                                      lr=0.004, mom=0.9, top_k=top_k_sg)
            try_seq(a_sg, "mirror")

    # === Phase 6: Alternating optimization ===
    print(f"\n=== Phase 6: Alternating (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        z_curr = z_from_a(best_seq)
        for it in range(6):
            rem = deadline - time.monotonic() - 25
            if rem < 25:
                break
            T_alt = [30, 60, 100, 150, 200, 300][it]
            t_lb = min(rem * 0.25, 35.0)
            z_curr = lbgfs_run(z_curr, T=T_alt, time_limit=t_lb, maxiter=3000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"alt{it}_T{T_alt}")

            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            t_pr = min(rem * 0.35, 20.0)
            a_pr = focused_peak_reduce(best_seq, time_limit=t_pr, top_k=3)
            try_seq(a_pr, f"alt{it}_pr")
            z_curr = z_from_a(best_seq)

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        deltas = [-32, -16, 16, 32, -64, 64]
        for delta in deltas:
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 4096:
                continue
            a_interp = interpolate_sequence(best_seq, new_n)
            t_alloc = min(rem * 0.25, 25.0)
            z0 = z_from_a(a_interp)
            z_opt = lbgfs_run(z0, T=25, time_limit=t_alloc, maxiter=3000)
            a_opt = softmax_from_z(z_opt)
            try_seq(a_opt, f"len_{new_n}")

    # === Phase 8: Final high-T polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            z0 = z_from_a(best_seq)
            t_alloc = min(rem * 0.6, 30.0)
            z_final = lbgfs_run(z0, T=800, time_limit=t_alloc, maxiter=3000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_T800")

        rem = deadline - time.monotonic() - 3
        if rem > 5:
            for attempt in range(3):
                rem = deadline - time.monotonic() - 2
                if rem < 4:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_seq + noise)
                z_pert = z_from_a(a_pert)
                t_alloc = min(rem * 0.7, 12.0)
                z_p = lbgfs_run(z_pert, T=200, time_limit=t_alloc, maxiter=2000)
                a_p = softmax_from_z(z_p)
                try_seq(a_p, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
