def compute_subgradient(a, b, w):
    """Efficient computation of weighted subgradient."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    return grad

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, n_iter=50):
    """Initialize a sequence using the spectral method with sinc-shaped target."""
    # Target: we want the linear convolution to be flat
    # Target power spectrum: DFT of a rectangle of length 2n-1
    N = 1 << (2*n - 2).bit_length()  # next power of 2 >= 2n-1
    # Compute target |A(f)|
    # c_target(k) = 1/(2n-1) for k=0,...,2n-2, 0 otherwise
    c_target = np.zeros(N)
    c_target[:2*n-1] = 1.0 / (2*n - 1)
    C_target = rfft(c_target, N)
    target_mag = np.abs(C_target)
    target_mag = np.maximum(target_mag, 1e-10)
    
    # Start with a random phase
    rng = np.random.default_rng(42)
    phase = rng.uniform(0, 2*np.pi, N//2 + 1)
    phase[0] = 0  # DC component is real
    
    A = target_mag * np.exp(1j * phase)
    a = np.real(irfft(A, N))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    
    # Iterate: FFT -> set magnitude -> IFFT -> clip -> normalize
    for _ in range(n_iter):
        A = rfft(a, N)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_new = target_mag * np.exp(1j * A_phase)
        a = np.real(irfft(A_new, N))[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
    
    return a

def nesterov_subgradient(a0, n_iter_max=50000, T_start=2.0, T_end=0.05, 
                          lr_start=0.01, lr_end=0.0001, momentum=0.9,
                          time_limit=60.0):
    """Nesterov-accelerated subgradient with adaptive temperature."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum velocity
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_obj = b.max()
    
    no_improve = 0
    max_no_improve = 500
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Progress fraction
        frac = min(1.0, it / max(1, n_iter_max))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Nesterov step
        v = momentum * v
        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute gradient at y
        b_y = fast_convolve(y)
        w = softmax_weights(b_y, T)
        grad = compute_subgradient(y, b_y, w)
        
        # Update
        v = v - lr * grad
        a = project_simplex(y - lr * grad)
        
        # Evaluate
        b_a = fast_convolve(a)
        obj = b_a.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj

def directed_anneal(a0, time_limit=60.0, T_start=0.1, T_end=0.001):
    """Directed simulated annealing: perturb elements contributing to peaks."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    b = fast_convolve(a)
    best_obj = b.max()
    
    T = T_start
    rng = np.random.default_rng(123)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule
        T = T_start * (T_end / T_start) ** (it / 100000.0)
        
        # Compute current peaks
        b = fast_convolve(a)
        curr_obj = b.max()
        
        # Find top peaks
        top_k = 5
        peak_indices = np.argsort(b)[-top_k:]
        
        # Choose a random peak
        k = peak_indices[rng.integers(top_k)]
        
        # Find elements contributing to this peak
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k - idx]
        
        # Choose the top contributor
        max_i = idx[np.argmax(contributions)]
        
        # Propose: reduce max_i and add to a random non-contributing element
        reduce_amount = a[max_i] * rng.uniform(0.01, 0.1)
        
        # Find candidate elements to add to (prefer low-contribution ones)
        candidate_mask = np.ones(n, dtype=bool)
        candidate_mask[max_i] = False
        # Prefer elements that don't contribute to top peaks
        for pk in peak_indices:
            pjs = max(0, pk - n + 1)
           pje = min(n - 1, pk)
            candidate_mask[pjs:pje+1] *= 0.5  # reduce priority
        
        candidates = np.where(candidate_mask)[0]
        if len(candidates) == 0:
            continue
        
        target = candidates[rng.integers(len(candidates))]
        
        # Propose new sequence
        a_new = a.copy()
        a_new[max_i] -= reduce_amount
        if a_new[max_i] < 0:
            a_new[max_i] = 0
            reduce_amount = a[max_i]
        a_new[target] += reduce_amount
        a_new = a_new / a_new.sum()
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        delta = obj_new - curr_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
    
    return best_a, best_obj
