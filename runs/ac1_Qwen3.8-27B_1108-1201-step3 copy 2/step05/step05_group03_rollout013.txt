        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Dual step: update p based on active constraints
        # Use "soft" active set: weight proportional to exp(conv/temp)
        conv_range = conv.max() - conv.min()
        if conv_range < 1e-12:
            temp = 1e-6
        else:
            temp = conv_range * 0.3
        p_new = np.exp((conv - conv.min()) / temp)
        p_new /= p_new.sum()
        
        # Also try concentrated on top peaks
        k_top = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -k_top)[-k_top:]
        p_concentrated = np.zeros(2 * n - 1)
        p_concentrated[top_peaks] = 1.0 / k_top
        
        # Mix soft and concentrated
        p = (1.0 - damping) * p + damping * (0.5 * p_new + 0.5 * p_concentrated)
        p = np.maximum(p, 1e-15)
        p /= p.sum()
        
        # Adapt damping: increase if oscillating
        if no_improve_count > 50:
            damping = min(0.8, damping + 0.05)
        
        if verbose and it % 200 == 0:
            print(f"    KKT iter {it}: score={best_score:.10f}, damping={damping:.2f}")
        
        if it > 20000:
            break
    
    return best_a, best_score


def nesterov_subgradient(a_init, time_limit=60, top_k=30, verbose=False):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a_init)
    a = normalize(a_init.copy())
    y = a.copy()  # lookahead
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = time.monotonic()
    step = 0
    base_lr = 0.08
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        
        # Subgradient at lookahead point
        conv_y = fast_convolve(y)
        k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv = conv_y[peaks]
        
        # Equal weights on top peaks for balanced descent
        w = np.ones(k_actual) / k_actual
        
        sg = np.zeros(n)
        for i in range(k_actual):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            partners = k - idx_range
            sg[idx_range] += 2.0 * w[i] * a[partners]
        
        # Adaptive step size
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            lr = base_lr
        else:
            lr = base_lr / (sg_norm * np.sqrt(1.0 + 0.1 * step))
        
        # Nesterov: step from y, then update y
        a_new = project_simplex(y - lr * sg)
        
        # Momentum update for y
        mu = (1.0 + np.sqrt(1.0 + 4.0 * step)) / 2.0
        if step == 1:
            y = a_new
        else:
            mu_prev = (1.0 + np.sqrt(1.0 + 4.0 * (step - 1))) / 2.0
            beta = (1.0 - mu_prev) / mu
            y = a_new + beta * (a_new - a)
            y = project_simplex(y)
        
        a = a_new
        
        # Track best
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = compute_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if verbose and step % 500 == 0:
            print(f"    NSG step {step}: score={best_score:.10f}")
        
        # Restart if stuck
        if step % 2000 == 0:
            noise = np.random.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            y = a.copy()
            base_lr = 0.05
    
    return best_a, best_score


def coord_descent(a_init, time_limit=120, verbose=False):
    """Precise coordinate descent with bounded scalar minimization."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            
            if old_val < 1e-14:
                lo, hi = 0.0, 0.1
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(0.999, old_val * 5.0)
                if hi - lo < 1e-5:
                    lo = max(0.0, old_val - 0.05)
                    hi = min(0.999, old_val + 0.05)
            
            def _obj(v):
                v = max(0.0, min(v, 1.0))
                new_a = a.copy()
                new_a[j] = v
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()
            
            try:
                res = minimize_scalar(_obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 50})
                if res.success and res.fun < _obj(old_val) - 1e-14:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-13:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")
        
        if not improved and pass_count > 3:
            break
        if pass_count > 300:
            break
    
    return best_a, best_score


def pair_swap_local(a_init, time_limit=60, n_trials=2000, verbose=False):
    """Random pair transfer local search."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(n_trials):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            if a[j] < 1e-12:
                continue
            
            fracs = [0.1, 0.3, 0.5, 0.7, 0.9]
            best_delta = 0.0
            best_sc = best_score
            
            for f in fracs:
                delta = a[j] * f
                if delta > a[j]:
                    continue
                new_a = a.copy()
                new_a[j] -= delta
                new_a[i] += delta
                conv = fast_convolve(new_a)
                sc = 2.0 * n * conv.max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = delta
            
            if best_delta > 0:
                a[j] -= best_delta
                a[i] += best_delta
                a = normalize(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score


def geometric_sequence(n, r):
    a = (1.0 - r) * r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def u_shape_sequence(n, power=0.5):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(-1, 1, n)
    a = np.abs(x) ** power + 0.1
    return normalize(a)


def exponential_decay(n, rate=0.01):
    a = np.exp(-rate * np.arange(n))
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def optimize_multi_stage(a, time_limit, verbose=False):
    """Multi-stage: KKT -> Nesterov -> CD -> PairSwap -> CD."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Stage 1: KKT fixed point
    rem = time_limit - (time.monotonic() - t0)
    t1 = min(rem * 0.35, 100)
    if rem > t1 + 10:
        a_kkt, sc_kkt = kkt_fixed_point(a, time_limit=t1, verbose=verbose)
        if sc_kkt < best_score:
            best_score = sc_kkt
            best_a = a_kkt.copy()
            a = a_kkt.copy()
        if verbose:
            print(f"      After KKT: {best_score:.10f}")
    
    # Stage 2: Nesterov subgradient
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.25, 80)
    if rem > t2 + 10:
        a_nsg, sc_nsg = nesterov_subgradient(a, time_limit=t2, verbose=verbose)
        if sc_nsg < best_score:
            best_score = sc_nsg
            best_a = a_nsg.copy()
            a = a_nsg.copy()
        if verbose:
            print(f"      After NSG: {best_score:.10f}")
    
    # Stage 3: Coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.3, 150)
    if rem > t3 + 5:
        a_cd, sc_cd = coord_descent(a, time_limit=t3, verbose=verbose)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
        if verbose:
            print(f"      After CD: {best_score:.10f}")
    
    # Stage 4: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    t4 = min(rem * 0.25, 60)
    if rem > t4 + 5:
        a_ps, sc_ps = pair_swap_local(a, time_limit=t4)
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps.copy()
            a = a_ps.copy()
        if verbose:
            print(f"      After PS: {best_score:.10f}")
    
    # Stage 5: Final CD
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_final, sc_final = coord_descent(a, time_limit=min(rem, 120))
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
        if verbose:
            print(f"      After final CD: {best_score:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: diverse initialization, KKT fixed-point, Nesterov
    subgradient, coordinate descent, dimension exploration, and local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)
    
    best_a = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # === Generate diverse starting points ===
    print("=== Generating starting points ===")
    starts = []
    
    # From parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            p_score = compute_score(parent)
            print(f"  Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    # Geometric
    for r in [0.998, 0.999, 0.9995, 0.9998]:
        for n in [512, 768, 1024]:
            starts.append((geometric_sequence(n, r), f"geom_r{r}_n{n}"))
    
    # Cosine
    for n in [512, 640, 768, 896, 1024]:
        starts.append((cosine_sequence(n), f"cos_n{n}"))
    
    # Gaussian
    for n in [512, 768, 1024]:
        for sf in [0.2, 0.3, 0.4]:
            starts.append((gaussian_sequence(n, sf), f"gauss_sf{sf}_n{n}"))
    
    # U-shaped
    for n in [512, 768, 1024]:
        for pw in [0.3, 0.5, 1.0]:
            starts.append((u_shape_sequence(n, pw), f"u_pw{pw}_n{n}"))
    
    # Exponential decay (asymmetric)
    for n in [512, 768, 1024]:
        for rate in [0.005, 0.01, 0.02]:
            starts.append((exponential_decay(n, rate), f"exp_r{rate}_n{n}"))
    
    # Random smooth
    for n in [640, 768, 896, 1024]:
        a = rng.random(n)
        k = 20
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize(a), f"rand_smooth_n{n}"))
    
    # Evaluate all starts
    print(f"  Evaluating {len(starts)} starting points...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])
    
    for i, (sc, a, label) in enumerate(start_scores[:5]):
        print(f"    Start #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)
    
    # === Phase 1: Optimize top diverse starts ===
    print("\n=== Phase 1: Optimize diverse starts ===")
    
    selected = []
    selected_types = set()
    for sc, a, label in start_scores:
        if len(selected) >= 5:
            break
        base = label.split('_n')[0] if '_n' in label else label
        if base not in selected_types:
            selected.append((a, label))
            selected_types.add(base)
    
    # Ensure parent is included
    has_parent = any(label == "parent" for _, label in selected)
    if not has_parent:
        for sc, a, label in start_scores:
            if label == "parent":
                selected.append((a, label))
                break
    
    n_sel = len(selected)
    rem_total = deadline - time.monotonic() - 400
    t_per = max(30, rem_total / max(1, n_sel))
    
    for a_init, label in selected:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}] (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, sc_opt = optimize_multi_stage(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")
    
    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        target_ns = set([384, 512, 640, 768, 896, 1024, 1280, 1536])
        target_ns.discard(best_n)
        
        for target_n in sorted(target_ns, key=lambda x: abs(x - best_n)):
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 150)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = optimize_multi_stage(a_interp, time_limit=t_alloc)
            if sc < best_score:
                try_seq(a_opt, f"dim_{target_n}")
    
    # === Phase 3: KKT with various dual initializations ===
    print(f"\n=== Phase 3: KKT with diverse duals (best={best_score:.8f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        conv_best = fast_convolve(best_a)
        
        dual_inits = []
        # Uniform
        dual_inits.append(np.ones(2 * n_best - 1) / (2 * n_best - 1))
        # Peaks
        top_k = min(50, 2 * n_best - 1)
        peaks = np.argpartition(conv_best, -top_k)[-top_k:]
        p_peak = np.zeros(2 * n_best - 1)
        p_peak[peaks] = 1.0 / top_k
        dual_inits.append(p_peak)
        # Soft max
        temp = max(conv_best.max() - conv_best.min(), 1e-6) * 0.5
        p_soft = np.exp(conv_best / temp)
        p_soft /= p_soft.sum()
        dual_inits.append(p_soft)
        # Edge-weighted
        p_edge = np.zeros(2 * n_best - 1)
        p_edge[:n_best // 4] = 0.3
        p_edge[-n_best // 4:] = 0.3
        p_edge[n_best // 2 - 5:n_best // 2 + 5] = 0.4
        p_edge /= p_edge.sum()
        dual_inits.append(p_edge)
        
        for di, p_init in enumerate(dual_inits):
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            # Warm-start with best_a, but use custom p
            # Temporarily modify kkt to accept p_init
            a_copy = best_a.copy()
            # Run kkt with more aggressive settings
            t_alloc = min(rem * 0.5, 80)
            
            # Manually run KKT with this p_init
            n = len(a_copy)
            a_kkt = normalize(a_copy.copy())
            p = p_init.copy()
            best_kkt_a = a_kkt.copy()
            best_kkt_sc = compute_score(a_kkt)
            t_kkt0 = time.monotonic()
            
            for it in range(5000):
                if time.monotonic() - t_kkt0 > t_alloc:
                    break
                a_cand = kkt_solve(p, n)
                a_cand = np.maximum(a_cand, 0.0)
                sc_sum = a_cand.sum()
                if sc_sum > 1e-15:
                    a_cand = a_cand / sc_sum
                a_kkt = 0.7 * a_kkt + 0.3 * a_cand
                a_kkt = normalize(a_kkt)
                
                conv_k = fast_convolve(a_kkt)
                sc_k = compute_score(a_kkt)
                if sc_k < best_kkt_sc:
                    best_kkt_sc = sc_k
                    best_kkt_a = a_kkt.copy()
                
                # Update p
                cr = conv_k.max() - conv_k.min()
                tp = max(cr * 0.3, 1e-8)
                p_new = np.exp((conv_k - conv_k.min()) / tp)
                p_new /= p_new.sum()
                p = 0.7 * p + 0.3 * p_new
                p = np.maximum(p, 1e-15)
                p /= p.sum()
            
            if best_kkt_sc < best_score:
                try_seq(best_kkt_a, f"kkt_dual{di}")
    
    # === Phase 4: Aggressive local search ===
    print(f"\n=== Phase 4: Aggressive local search (best={best_score:.8f}) ===")
    if best_a is not None:
        # Final precise CD
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.5, 200)
            print(f"  Final CD: t={t_cd:.0f}s")
            a_final, sc = coord_descent(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_final, "final_cd")
        
        # Pair swap
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_ps = min(rem * 0.6, 80)
            a_ps, sc = pair_swap_local(best_a, time_limit=t_ps)
            try_seq(a_ps, "final_ps")
        
        # Perturbation + re-optimize
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            n_best = len(best_a)
            for trial in range(4):
                rem = deadline - time.monotonic() - 15
                if rem < 15:
                    break
                noise = rng.standard_normal(n_best) * 0.005
                a_pert = project_simplex(best_a + noise)
                a_opt, sc = optimize_multi_stage(a_pert, time_limit=min(rem, 60),
                                                  verbose=False)
                try_seq(a_opt, f"perturb_{trial}")
    
    # === Final verification ===
    if best_a is None:
        best_a = cosine_sequence(768)
        best_score = compute_score(best_a)
    
    # Verify with actual evaluate_sequence
    final_score = float(evaluate_sequence([float(x) for x in best_a]))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
