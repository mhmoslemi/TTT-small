def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

    k_actual = min(n_active, 2 * n - 1)

    # Softmax weights
    pv = conv[peaks]
    temp = max_val * 1e-4
    scaled = (pv - max_val) / temp
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[0] = 1.0

        je = min(n - 1, k)

def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            multipliers = [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]
            best_j_val = old_val
            best_j_obj = cur_conv.max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_direction_v2(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0."""
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize t
    
    # (f*g)[k] - t <= 0 for all k
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = f
    A_ub[:, -1] = -1.0  # -t column
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    if res.success and res.x is not None:
        return res.x[:n], res.x[-1]  # g, t
    return None, None
