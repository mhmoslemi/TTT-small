```python
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal update directions, combined with
subgradient polishing and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(p):
    """Build (2n-1) x n matrix M where (M @ q)[k] = (p*q)[k]."""
    n = len(p)
    M = np.zeros((2*n-1, n))
    for j in range(n):
        start = j  # row where p[0] appears in column j
        length = n  # p[0], p[1], ..., p[n-1]
        M[start:start+length, j] = p
    return M

def solve_update_lp(p):
    """Solve LP to find optimal update direction q."""
    n = len(p)
    M = build_conv_matrix(p)
    
    # min t s.t. M @ q - t <= 0, sum(q) = 1, q >= 0, t >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_optimize(p0, time_limit, alpha_sched='decay', verbose=False):
    """LP-based iterative optimization."""
    p = p0.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    t0 = time.monotonic()
    iteration = 0
    
    conv = fast_convolve(p)
    M = conv.max()
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        q, t_opt = solve_update_lp(p)
        if q is None:
            break
        
        # Try damped updates
        best_alpha = 0
        best_trial_obj = best_obj
        
        for alpha in [0.3, 0.5, 0.7, 1.0, 0.2, 0.1]:
            p_trial = (1 - alpha) * p + alpha * q
            p_trial = np.maximum(p_trial, 0)
            ps = p_trial.sum()
            if ps < 1e-15:
                continue
            p_trial /= ps
            obj = fast_convolve(p_trial).max()
            if obj < best_trial_obj - 1e-12:
                best_trial_obj = obj
                best_alpha = alpha
        
        if best_alpha > 0:
            p = (1 - best_alpha) * p + best_alpha * q
            p = np.maximum(p, 0)
            p /= p.sum()
            obj = fast_convolve(p).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_p = p.copy()
                M = obj
                if verbose and iteration % 10 == 0:
                    print(f"    LP iter {iteration}: obj={best_obj:.10f}, alpha={best_alpha}")
        
        # Periodically update M (the constraint level)
        if iteration % 20 == 0:
            M = fast_convolve(p).max()
    
    return best_p, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative optimization,
subgradient polishing with adaptive annealing, and coordinate descent.
Key innovation: LP finds globally optimal update directions that minimize
cross-convolution peaks, enabling faster convergence than pure subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(p):
    """Build (2n-1) x n matrix M where (M @ q)[k] = (p*q)[k] (full conv)."""
    n = len(p)
    M = np.zeros((2 * n - 1, n))
    for j in range(n):
        M[j:j + n, j] = p
    return M


def solve_update_lp(p, verbose=False):
    """Solve LP: min t s.t. (p*q)[k] <= t, sum(q)=1, q>=0, t>=0. Returns (q, t_opt)."""
    n = len(p)
    M = build_conv_matrix(p)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([M, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_optimize(p0, time_limit, verbose=False, max_iter=100000):
    """LP-based iterative optimization with damped updates."""
    p = p0.astype(np.float64).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p0)) / len(p0)
    else:
        p = p / s
    n = len(p)

    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and iteration < max_iter:
        iteration += 1
        if time.monotonic() - t0 > time_limit:
            break

        q, t_opt = solve_update_lp(p)
        if q is None:
            break

        # Try multiple step sizes
        best_alpha = 0
        best_trial_obj = best_obj + 1e-10

        alphas = [1.0, 0.5, 0.75, 0.3, 0.2, 0.1]
        for alpha in alphas:
            p_trial = (1 - alpha) * p + alpha * q
            p_trial = np.maximum(p_trial, 0)
            ps = p_trial.sum()
            if ps < 1e-15:
                continue
            p_trial /= ps
            obj = fast_convolve(p_trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_alpha = alpha

        if best_alpha > 0:
            p = (1 - best_alpha) * p + best_alpha * q
            p = np.maximum(p, 0)
            p /= p.sum()
            obj = fast_convolve(p).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_p = p.copy()
                no_improve = 0
                if verbose and iteration % 50 == 0:
                    print(f"    LP iter {iteration}: obj={best_obj:.10f}, alpha={best_alpha:.3f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Reset on stagnation
        if no_improve > 100:
            if verbose:
                print(f"    LP stagnant at {best_obj:.10f} after iter {iteration}")
            break

    return best_p, best_obj


def compute_subgradient(p, w):
    """Compute subgradient: sg[j] = 2 * sum_k w[k] * p[k-j] (valid k-j)."""
    n = len(p)
    sg = np.zeros(n)
    for j in range(n):
        k_max = min(2 * n - 2, j + n - 1)
        if j <= k_max:
            sg[j] = 2.0 * np.dot(w[j:k_max + 1], p[:k_max + 1 - j])
    return sg


def get_softmax_weights(conv, temp, top_k=None):
    """Get soft-max weights over convolution values."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(w))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w


def subgradient_optimize(p0, time_limit, lr=0.01, temp_init=0.005,
                         temp_decay=0.9995, top_k=None, verbose=False):
    """Subgradient method with simplex projection and adaptive step size."""
    p = p0.astype(np.float64).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p0)) / len(p0)
    else:
        p = p / s
    n = len(p)

    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    temp = temp_init
    cur_lr = lr
    t0 = time.monotonic()
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(p)
        w = get_softmax_weights(conv, temp, top_k)
        sg = compute_subgradient(p, w)

        # Normalized step
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        step = cur_lr / sg_norm

        p_new = p - step * sg
        p = project_simplex(p_new)

        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.01)

        conv = fast_convolve(p)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_p = p.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    SG iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 500:
            break

    return best_p, best_obj


def coordinate_polish(p0, time_limit, verbose=False):
    """Coordinate descent: try scaling individual coordinates."""
    p = p0.astype(np.float64).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p0)) / len(p0)
    else:
        p = p / s
    n = len(p)

    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = p[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(p).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            multipliers = [0.5, 0.7, 0.85, 1.2, 1.4, 1.7]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.98:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                p_trial = p.copy()
                p_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                p_trial[mask] *= scale
                p_trial = p_trial / p_trial.sum()

                val = fast_convolve(p_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    p[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    p[mask] *= scale
                    p = p / p.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_p = p.copy()
                    improved = True

        if not improved and passes > 1:
            break

    return best_p, best_obj

        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.3)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    x = np.linspace(0, 1, n)
    if shape == 'bump':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'u_shape':
        f = (x - 0.5) ** 2 + 0.1
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based iterations, subgradient methods,
    and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")

    n_parent = len(parent)
    print(f"\nStarting optimization (n={n_parent}, score={best_score:.8f})")

    # === Phase 1: Quick exploration with different initializations ===
    print(f"\n=== Phase 1: Exploration (budget=150s) ===")
    t_p1_end = time.monotonic() + 150

    initializations = {
        'parent': parent.copy(),
        'uniform': np.ones(n_parent) / n_parent,
        'bump': make_symmetric_init(n_parent, 'bump'),
        'u_shape': make_symmetric_init(n_parent, 'u_shape'),
        'smooth_rand': make_smooth_random_init(n_parent, rng, 5),
    }

    # Perturbed parent
    initializations['parent_pert'] = project_simplex(parent + rng.standard_normal(n_parent) * 0.02)

    for name, init in initializations.items():
        rem = t_p1_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.35, 30)
        if t_alloc < 5:
            break

        # Quick LP optimization
        a_opt, obj = lp_optimize(init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"p1_lp_{name}")

    # === Phase 2: Main LP-based optimization ===
    print(f"\n=== Phase 2: LP optimization (best={best_score:.8f}) ===")
    t_p2_end = time.monotonic() + 1000
    base = best_seq.copy() if best_seq is not None else parent
    n_base = len(base)

    # Run LP with the best sequence found so far
    rem = t_p2_end - time.monotonic()
    if rem > 30:
        t_alloc = rem * 0.6
        print(f"  LP on main (n={n_base}), t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "p2_lp_main")

    # Try LP from perturbed versions
    if best_seq is not None:
        for pert_i in range(3):
            rem = t_p2_end - time.monotonic()
            if rem < 20:
                break
            noise_scale = 0.01 * (pert_i + 1)
            perturbed = project_simplex(best_seq + rng.standard_normal(n_base) * noise_scale)
            t_alloc = min(rem * 0.3, 80)
            print(f"  LP perturbed {pert_i} (scale={noise_scale}), t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(perturbed, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"p2_lp_pert{pert_i}")

    # === Phase 3: Subgradient optimization ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    t_p3_end = time.monotonic() + 350
    if best_seq is not None:
        n_cur = len(best_seq)

        # Multiple subgradient runs with different temperatures
        sg_configs = [
            {'lr': 0.008, 'temp_init': 0.004, 'temp_decay': 0.9994, 'top_k': max(10, n_cur // 15)},
            {'lr': 0.005, 'temp_init': 0.008, 'temp_decay': 0.9996, 'top_k': max(20, n_cur // 10)},
            {'lr': 0.012, 'temp_init': 0.003, 'temp_decay': 0.9992, 'top_k': max(5, n_cur // 25)},
        ]

        for ci, cfg in enumerate(sg_configs):
            rem = t_p3_end - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 100)
            print(f"  SG config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, verbose=False, **cfg)
            try_seq(a_opt, f"p3_sg_{ci}")

        # Subgradient from mixed starting point
        rem = t_p3_end - time.monotonic()
        if rem > 20:
            mixed = project_simplex(0.7 * best_seq + 0.3 * make_symmetric_init(n_cur, 'bump'))
            t_alloc = min(rem * 0.5, 80)
            a_opt, _ = subgradient_optimize(mixed, time_limit=t_alloc,
                                            lr=0.006, temp_init=0.005,
                                            temp_decay=0.9995, top_k=max(10, n_cur // 12))
            try_seq(a_opt, "p3_sg_mixed")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
    t_p4_end = time.monotonic() + 200
    if best_seq is not None:
        n_cur = len(best_seq)
        target_lengths = [512, 640, 768, 896, 1024, 1152]
        target_lengths = [t for t in target_lengths if t != n_cur and 256 <= t <= 1280]
        target_lengths.sort(key=lambda x: abs(x - n_cur))
        target_lengths = target_lengths[:3]

        for target_n in target_lengths:
            rem = t_p4_end - time.monotonic()
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.5, 60)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"p4_n{target_n}_lp")

            rem = t_p4_end - time.monotonic()
            if rem > 15:
                a_opt, _ = subgradient_optimize(f_interp, time_limit=min(rem * 0.4, 30),
                                                lr=0.005, temp_init=0.005,
                                                temp_decay=0.9995)
                try_seq(a_opt, f"p4_n{target_n}_sg")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    t_p5_end = time.monotonic() + 150
    if best_seq is not None:
        rem = t_p5_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.7, 100)
            print(f"  CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p5_cd")

            # CD from slightly perturbed
            rem = t_p5_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                n_cur = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_cur) * 0.005)
                a_opt, _ = coordinate_polish(perturbed, time_limit=min(rem * 0.6, 40))
                try_seq(a_opt, "p5_cd_pert")

    # === Phase 6: Final fine LP + subgradient combo ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        rem = deadline - time.monotonic() - 10

        if rem > 30:
            t_alloc = min(rem * 0.5, 60)
            print(f"  Final LP, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "p6_lp")

        rem = deadline - time.monotonic() - 8
        if rem > 15 and best_seq is not None:
            t_alloc = min(rem * 0.5, 40)
            print(f"  Final SG, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc,
                                            lr=0.003, temp_init=0.002,
                                            temp_decay=0.9998, top_k=max(5, n_cur // 30))
            try_seq(a_opt, "p6_sg")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 8
        for pi in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 5 or best_seq is None:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_optimize(f_pert, time_limit=min(rem * 0.5, 15), verbose=False)
            try_seq(a_opt, f"p6_pert{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
