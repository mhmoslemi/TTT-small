def auto_convolve(a):
    """Compute linear auto-convolution of a with itself."""
    return np.convolve(a, a, mode='full')


def compute_subgradient(a, top_k=20, temperature=0.01):
    """
    Compute softmax-weighted subgradient of max_k (a*a)_k.
    Returns the subgradient vector of length len(a).
    """
    n = len(a)
    conv = auto_convolve(a)
    
    # Select top peaks
    k_sel = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_sel)[-k_sel:]
    
    # Softmax weights
    pv = conv[peak_indices]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute subgradient: sum_i w_i * 2 * a_{k_i - j}
    sg = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if w[i] < 1e-12:
            continue
        k = int(k)
        # (a*a)_k = sum_j a_j * a_{k-j}
        # d/d a_j = 2 * a_{k-j} if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        partner = k - idx  # these are the k-j values
        sg[idx] += 2.0 * w[i] * a[partner]
    
    return sg

from scipy.optimize import linprog
import time


def auto_convolve(a):
    """Compute linear auto-convolution of a with itself."""
    return np.convolve(a, a, mode='full')


def compute_soft_subgradient(a, top_k=30, temperature=0.01):
    """Compute softmax-weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    conv = auto_convolve(a)
    k_sel = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_sel)[-k_sel:]
    pv = conv[peak_indices]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peak_indices))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if w[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        partner = k - idx
        sg[idx] += 2.0 * w[i] * a[partner]
    return sg, conv


def mwd_momentum(a0, time_limit, lr_init=0.04, top_k=30, temp_init=0.01,
                 momentum=0.7, verbose=False):
    """
    Mirror descent (multiplicative weight) with heavy-ball momentum.
    Minimizes max_k (a*a)_k subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = auto_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg, conv = compute_soft_subgradient(a, top_k=top_k, temperature=temp)

        # Heavy ball momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative update with momentum-smoothed gradient
        a_new = a * np.exp(-lr * v)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        conv = auto_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9997
        temp *= 1.002

        if verbose and it % 200 == 0:
            print(f"    MWD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")

        if no_improve > 300:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr_init * 0.3, 0.005)
            no_improve = 0
            temp = temp_init

        if lr < 1e-6:
            break

    return best_a, best_obj


def restricted_lp_fw(f0, time_limit, m_constraints=40, verbose=False):
    """
    LP-Frank-Wolfe with only the top-m most binding constraints.
    Much faster than full LP-FW for large n.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = auto_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find active constraints (top peaks)
        conv = auto_convolve(f)
        M = conv.max()
        k_sel = min(m_constraints, len(conv))
        peak_indices = np.argpartition(conv, -k_sel)[-k_sel:]

        # Build restricted LP
        c = -np.ones(n)
        A_ub = np.zeros((k_sel, n))
        for i, k in enumerate(peak_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[i, j] = f[k - j]
        b_ub = np.full(k_sel, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success:
            m_exp *= 1.1
            if m_exp > 5.0:
                break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            break

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = auto_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = auto_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"    RLPFW iter {it}: obj={best_obj:.10f}")
        else:
            m_exp = min(m_exp * 1.08, 5.0)

    return best_f, best_obj


def make_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Gaussian
    x = np.arange(n) - n / 2.0
    for sigma in [0.2, 0.3, 0.4]:
        a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    # Bell (flat top)
    for width in [0.4, 0.6, 0.8]:
        half = n * width / 2.0
        t = np.clip(np.abs(x) / half, 0, 1.0)
        a = 1.0 - t ** 4
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    # Inverted-V (triangle)
    a = 1.0 - np.abs(x) / (n / 2.0)
    a = np.maximum(a, 0)
    a = a / a.sum()
    inits.append(a)
    # Random smooth
    a = np.abs(rng.standard_normal(n)) + 0.1
    # Smooth with moving average
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    a = a / a.sum()
    inits.append(a)
    return inits


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
