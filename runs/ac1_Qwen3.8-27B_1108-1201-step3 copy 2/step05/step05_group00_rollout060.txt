"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with full
FFT-based gradients, adaptive beta scheduling, multi-length exploration,
and LP-Frank-Wolfe polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent_optimize(a0, time_limit, beta_init=50.0, beta_max=500.0, 
                            lr_init=0.1, verbose=False):

    over the simplex, using log-sum-exp smoothing.

    a = a / a.sum()
    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()

        scaled -= scaled.max()

        if p_sum < 1e-300:

            p[np.argmax(conv)] = 1.0
        else:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, p):

    Compute grad_m = 2 * sum_k p[k] * a[k-m] for m in [0, n-1].
    Uses FFT for O(n log n) computation.

    ft_size = 1 << (N_conv - 1).bit_length()

    AR = np.fft.rfft(a_rev, ft_size)
    c = np.fft.irfft(P * AR, ft_size)[:N_conv]

def make_linear_ramp(n, direction=1):

    return a


def make_gaussian(n, center_frac=0.5, sigma_frac=0.25):

    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center_frac) / sigma_frac) ** 2)

def make_quadratic(n, direction=1):

    x = np.arange(n, dtype=np.float64) / (n - 1)
    if direction == 1:
        a = x ** 2
    else:
        a = (1 - x) ** 2
    a = a + 0.01  # Small floor to avoid too many zeros
    a /= a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Quick screening of initializations ===
    print("=== Phase 1: Initialization Screening ===")
    t_phase1 = 60
    init_results = []

    n_values = [384, 512, 768, 1024]
    if parent is not None:
        n_values.append(len(parent))

    for n in n_values:
        if time.monotonic() > deadline - t_phase1:
            break
        inits = [
            (make_linear_ramp(n, 1), f"lin_up_{n}"),
            (make_linear_ramp(n, -1), f"lin_dn_{n}"),
            (make_exponential(n, 3.0, 1), f"exp_up_{n}"),
            (make_exponential(n, 3.0, -1), f"exp_dn_{n}"),
            (make_quadratic(n, 1), f"quad_up_{n}"),
            (make_quadratic(n, -1), f"quad_dn_{n}"),
            (make_gaussian(n, 0.5, 0.25), f"gauss_{n}"),
            (make_sinusoidal_modulated(n, 1.5, 0.4), f"sin_{n}"),
        ]
        # Random initialization
        a_rand = np.abs(rng.standard_normal(n))
        a_rand /= a_rand.sum()
        inits.append((a_rand, f"rand_{n}"))

        for a_init, label in inits:
            sc = try_seq(a_init, label)
            if np.isfinite(sc):
                init_results.append((sc, a_init, n, label))

    # Sort by score
    init_results.sort(key=lambda x: x[0])
    print(f"  Best init: {init_results[0][3]} score={init_results[0][0]:.8f}")

    # === Phase 2: Mirror descent on top initializations ===
    print("\n=== Phase 2: Mirror Descent ===")
    t_alloc_total = max(600, (deadline - time.monotonic()) * 0.5)
    n_top = min(6, len(init_results))
    t_per = t_alloc_total / n_top

    for idx in range(n_top):
        if time.monotonic() > deadline - 300:
            break
        sc, a_init, n, label = init_results[idx]
        rem = min(t_per, deadline - time.monotonic() - 200)
        if rem < 30:
            break
        print(f"  MD on {label} (n={n}, score={sc:.6f}): {rem:.0f}s")
        a_opt, obj = mirror_descent(a_init, time_limit=rem, beta_init=30,
                                    beta_max=1500, lr_init=0.5, verbose=True, tag=label)
        try_seq(a_opt, f"md_{label}")

    # === Phase 3: Mirror descent on parent / best with different params ===
    print("\n=== Phase 3: Parameter Sweep ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for beta_i, lr_i in [(50, 0.3), (100, 0.2), (200, 0.1), (500, 0.05)]:
            if time.monotonic() > deadline - 200:
                break
            rem = min(60, deadline - time.monotonic() - 100)
            if rem < 20:
                break
            print(f"  MD sweep beta={beta_i}, lr={lr_i}: {rem:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=rem,
                                        beta_init=beta_i, beta_max=beta_i * 20,
                                        lr_init=lr_i, verbose=False, tag="sweep")
            try_seq(a_opt, f"sweep_b{beta_i}")

    # === Phase 4: LP-FW on best ===
    print("\n=== Phase 4: LP-FW Polishing ===")
    if best_seq is not None:
        rem = min(120, deadline - time.monotonic() - 100)
        if rem > 30:
            print(f"  LP-FW: {rem:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=rem, verbose=True, tag="main")
            try_seq(a_opt, "lpfw")

    # === Phase 5: Try interpolating to longer lengths ===
    print("\n=== Phase 5: Longer Sequences ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 1536]:
            if time.monotonic() > deadline - 150:
                break
            if target_n <= len(best_seq):
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_{target_n}")

            rem = min(90, deadline - time.monotonic() - 80)
            if rem < 30:
                break
            print(f"  MD n={target_n}: {rem:.0f}s")
            a_opt, obj = mirror_descent(f_interp, time_limit=rem,
                                        beta_init=50, beta_max=1000,
                                        lr_init=0.4, verbose=False, tag=f"n{target_n}")
            try_seq(a_opt, f"md_interp_{target_n}")

    # === Phase 6: Random restarts from best ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(4):
            if time.monotonic() > deadline - 100:
                break
            rem = min(60, deadline - time.monotonic() - 60)
            if rem < 20:
                break

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.005
                a_init = project_simplex(best_seq + noise)
            elif r == 1:
                # Blend with linear ramp
                a_init = 0.7 * best_seq + 0.3 * make_linear_ramp(n_best, 1)
                a_init = a_init / a_init.sum()
            elif r == 2:
                # Blend with reverse
                a_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                a_init = a_init / a_init.sum()
            else:
                # Scaled noise
                noise = rng.standard_normal(n_best) * 0.01
                a_init = np.maximum(best_seq + noise, 0)
                a_init = a_init / a_init.sum()

            print(f"  Restart {r}: {rem:.0f}s")
            a_opt, obj = mirror_descent(a_init, time_limit=rem,
                                        beta_init=100, beta_max=1000,
                                        lr_init=0.3, verbose=False, tag=f"r{r}")
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Final LP-FW squeeze ===
    print("\n=== Phase 7: Final LP-FW ===")
    if best_seq is not None:
        rem = min(60, deadline - time.monotonic() - 20)
        if rem > 20:
            print(f"  Final LP-FW: {rem:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=rem, verbose=False, tag="final")
            try_seq(a_opt, "final_lpfw")

    # === Phase 8: Final MD polish ===
    if best_seq is not None:
        rem = min(40, deadline - time.monotonic() - 10)
        if rem > 15:
            print(f"  Final MD: {rem:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=rem,
                                        beta_init=500, beta_max=2000,
                                        lr_init=0.1, verbose=False, tag="final_md")
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
