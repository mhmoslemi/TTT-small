import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)[k] = (f*g)[k]"""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    return A

def build_conv_matrix_sparse(f):
    """Build the convolution matrix more efficiently."""
    n = len(f)
    size = 2 * n - 1
    # For each k, the valid j range is max(0, k-n+1) to min(k, n-1)
    A = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            i_vals = k - np.arange(j_start, j_end + 1)
            A[k, j_start:j_end+1] = f[i_vals]
    return A

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

        rem = deadline - time.monotonic() - 100.0
        if rem > 20:
            t4 = max(10.0, min(rem * 0.7, 200.0))
            print(f"  LP after CD: t={t4:.0f}s")
            a_opt, sc_opt = lp_iteration(best_a, time_limit=t4, max_iters=300, verbose=True)
            try_seq(a_opt, "lp_after_cd")

    print(f"\n=== Phase 5: Final CD polish (score={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 50.0
        if rem > 10:
            t5 = max(5.0, min(rem * 0.8, 200.0))
            print(f"  Final CD: t={t5:.0f}s")
            a_cd, sc_cd = coord_descent_fast(best_a, time_limit=t5)
            try_seq(a_cd, "final_cd")

    # Fallback
    if best_a is None:
        best_a = geometric_init(768, 0.9995)
        best_score = compute_score(best_a)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_a]
    final_score = evaluate_sequence(final_list)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
