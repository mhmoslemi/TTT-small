"""
LP-based iterative refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a linear programming approach to find the best direction to move at each step,
combined with L-BFGS polishing and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_lp_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[k, js:je+1] = a[k - np.arange(js, je+1)]
    return A_ub


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k."""
    n = len(a)
    c = -np.ones(n)
    A_ub = build_lp_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.success:
        g = -res.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            g /= s

"""
LP-based iterative refinement combined with L-BFGS polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses linear programming to find optimal directions
at each step, multi-length exploration, and adaptive step-size line search.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[k, js:je + 1] = a[k - np.arange(js, je + 1)]
    return A_ub


def solve_lp_direction(a, M):
    n = len(a)
    c = -np.ones(n)
    A_ub = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 1.0)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = -res.x
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                g /= s
            return g
    except Exception:
        pass
    return None


def lp_refine(a0, time_limit=60.0, verbose=False):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_convolve(a0).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    stall_count = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = best_obj
        g = solve_lp_direction(a, M)
        if g is None:
            break

        if np.max(np.abs(g - a)) < 1e-9:
            if verbose:
                print(f"    LP converged at iter {it}")
            break

        best_t = 0.0
        best_val = best_obj
        step_candidates = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        for t in step_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial /= ss
            val = fast_convolve(a_trial).max()
            if val < best_val - 1e-13:
                best_val = val
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    LP stalled at iter {it}")
                break
            if stall_count <= 3:
                rng = np.random.default_rng(42 + it)
                a = project_simplex(a + rng.standard_normal(n) * 0.002)
                obj_check = fast_convolve(a).max()
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a /= a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
            if verbose and it % 50 == 0:
                print(f"    LP iter {it}: obj = {obj:.10f}, t = {best_t:.4f}")
        else:
            a = best_a.copy()
            stall_count += 1
            if stall_count > 8:
                break

    return best_a, best_obj


def lbgfs_polish(a0, time_limit=30.0, T=50):
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': 75000, 'disp': False})
        return softmax_from_z(res.x)
    except Exception:
        return a0.copy()


def multiplicative_weights(a0, time_limit=30.0, eta=0.1, top_k=10):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), fast_convolve(a0).max()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        max_pv = pv.max()
        if max_pv < 1e-15:
            break
        w = pv / max_pv
        w = w ** 2
        ws = w.sum()
        if ws < 1e-15:
            break
        w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= eta * w[i] * a[k - idx]

        a = a * np.exp(log_update)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            break
        a /= ss

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            eta *= 0.8
            if eta < 1e-5:
                break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_inp = f_interp / ss
    return f_interp
