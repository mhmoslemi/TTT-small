"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L-BFGS with soft-max smoothing at multiple temperatures and lengths
2. Nesterov accelerated projected subgradient with top-k objective
3. MCM (Minimum Constant Modulus) style multiplicative weight updates
4. Greedy peak-peeling local search
5. Final polishing with exact max objective

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, b, top_k=1):
    """Compute subgradient of max(b) or topk_avg(b) w.r.t. a."""
    n = len(a)
    k = min(top_k, len(b))
    peaks = np.argpartition(b, -k)[-k:]
    g = np.zeros(n)
    for idx in peaks:
        ki = int(idx)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        idx_arr = np.arange(js, je + 1)
        g[idx_arr] += a[ki - idx_arr]
    g /= k
    return g, peaks


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]
    # Actually compute cross-correlation properly
    grad_a = np.zeros(n)
    for i in range(n):
        ki = int(np.argmax(w * np.arange(len(w))))  # placeholder, not used
    # Proper cross-correlation: grad_a[i] = sum_k w[k] * a[k-i]
    # Use FFT-based approach
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a *= 2.0 * n
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit."""
    t0 = time.monotonic()
    best_z = z0.copy()
    best_fun = soft_max_obj_grad(z0, T, n)[0]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return best_fun + 1e10
        return soft_max_obj_grad(z, T, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return soft_max_obj_grad(z, T, n)[1]

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        if res.fun < best_fun:
            best_z = res.x
            best_fun = res.fun
    except Exception:
        pass
    return best_z, best_fun


def nesterov_subgradient(a0, time_limit, top_k=1, lr_init=0.01, decay=0.9995):
    """Nesterov accelerated projected subgradient method."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)
    y = a.copy()
    lr = lr_init
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        b_y = fast_convolve(y)
        g, _ = compute_subgradient(y, b_y, top_k=top_k)
        # Projected gradient step
        a_new = project_simplex(y - lr * g)
        # Nesterov update
        t_curr = 1.0 / (np.sqrt(momentum) * 0.5 + 1)
        y = (1 - t_curr) * a_new + t_curr * (a_new - (a_new - a) / t_curr)
        y = project_simplex(y)
        a = a_new
        lr *= decay
        b = fast_convolve(a)
        cur_obj = b.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
    return best_a, best_obj


def mcm_iterations(a0, time_limit, alpha=0.1, n_iter_max=50000):
    """MCM-style multiplicative weight updates to reduce peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        # Find peaks
        threshold = M * 0.9
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            break
        # Compute blame for each element
        blame = np.zeros(n)
        for ki in peaks:
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx_arr = np.arange(js, je + 1)
            blame[idx_arr] += a[ki - idx_arr]
        blame /= len(peaks)
        # Multiplicative update
        update = 1.0 - alpha * blame
        update = np.maximum(update, 0.01)
        a = a * update
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        # Reduce alpha over time
        alpha *= 0.9999
        if alpha < 0.001:
            break
    return best_a, best_obj


def greedy_peak_peel(a0, time_limit, n_trials=10000):
    """Greedy peak peeling: reduce the element contributing most to the max peak."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        k_peak = int(np.argmax(b))
        # Find the pair (i, k-i) contributing most to b[k_peak]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx_arr = np.arange(js, je + 1)
        contributions = a[idx_arr] * a[k_peak - idx_arr]
        max_local = np.argmax(contributions)
        i_star = idx_arr[max_local]
        j_star = k_peak - i_star

        # Try reducing a[i_star] or a[j_star] and redistributing
        target = i_star if a[i_star] >= a[j_star] else j_star
        if a[target] < 1e-10:
            # Try the other
            target = i_star if target == j_star else j_star
            if a[target] < 1e-10:
                break

        # Try several reduction factors
        best_factor = None
        best_new_obj = best_obj
        for factor in [0.7, 0.8, 0.85, 0.9, 0.95]:
            new_val = a[target] * factor
            diff = a[target] - new_val
            if diff < 1e-12:
                continue
            a_trial = a.copy()
            a_trial[target] = new_val
            # Redistribute to others proportionally
            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others_sum = a_trial[mask].sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] += diff * a_trial[mask] / others_sum
            a_trial /= a_trial.sum()
            obj = fast_convolve(a_trial).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_factor = factor

        if best_factor is not None:
            new_val = a[target] * best_factor
            diff = a[target] - new_val
            a[target] = new_val
            mask = np.ones(n, dtype=bool)
            mask[target] = False
            others_sum = a[mask].sum()
            if others_sum > 1e-15:
                a[mask] += diff * a[mask] / others_sum
            a /= a.sum()
            if best_new_obj < best_obj:
                best_obj = best_new_obj
                best_a = a.copy()
        else:
            # Try adding to a different position
            # Find position with minimum "peak contribution"
            min_contrib = None
            min_val = M + 1
            for j in range(0, n, max(1, n // 50)):
                if a[j] > 0.5:
                    continue
                # Estimate effect of increasing a[j]
                effect = 0.0
                for ki in range(max(0, j - n + 1), min(2 * n - 1, j + n)):
                    other_idx = ki - j
                    if 0 <= other_idx < n:
                        effect = max(effect, a[other_idx])
                if effect < min_val:
                    min_val = effect
                    min_contrib = j

            if min_contrib is not None and min_contrib != target:
                # Move small amount from target to min_contrib
                epsilon = a[target] * 0.05
                a_trial = a.copy()
                a_trial[target] -= epsilon
                a_trial[min_contrib] += epsilon
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_obj:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
            else:
                break
    return best_a, best_obj


def local_search_swaps(a0, time_limit, n_trials=5000):
    """Local search: try swapping/adjusting pairs of elements."""
    a = a0.copy().astype(np.float64)
    s = a.sum()

    a = a / s

        j = rng.integers(0, n)
        if i == j:

        if a[i] < 1e-10 and a[j] < 1e-10:
            continue
        # Try redistributing between i and j
        total = a[i] + a[j]
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:

        count += 1
        b = fast_convolve(a)
        g, _ = compute_subgradient(a, b, top_k=top_k)

        lr *= 0.999

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def upsample_with_zeros(f, factor):
    """Upsample by inserting zeros between elements."""
    old_n = len(f)
    new_n = old_n * factor
    result = np.zeros(new_n)
    result[::factor] = f
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

            print(f"  Greedy peel n={len(best_seq)}, t={t_gp:.0f}s")
            a_gp, _ = greedy_peak_peel(best_seq, t_gp)
            try_seq(a_gp, "greedy")

            rem = deadline - time.monotonic() - 60
            if rem > 10:
                t_ls = min(rem * 0.4, 50.0)
                a_ls, _ = local_search_swaps(best_seq, t_ls)
                try_seq(a_ls, "local_search")

    # === Phase 7: Combined refinement on best ===
    print(f"\n=== Phase 7: Combined refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Apply MCM then Nesterov in sequence
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_a = min(rem * 0.3, 50.0)
            a1, _ = mcm_iterations(best_seq, t_a, alpha=0.03)
            if np.issubdtype(type(a1), np.floating) or isinstance(a1, np.ndarray):
                try_seq(a1, "combined_1")
            if best_seq is not None:
                rem = deadline - time.monotonic() - 40
                if rem > 15:
                    t_b = min(rem * 0.4, 40.0)
                    a2, _ = nesterov_subgradient(best_seq, t_b, top_k=5, lr_init=0.01)
                    try_seq(a2, "combined_2")

        # Final L-BFGS polish with very high T
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            z0 = z_from_a(best_seq)
            t_final = min(rem * 0.5, 25.0)
            z_final, _ = lbgfs_optimize(z0, 500, n_best, t_final, maxiter=2000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_lbgfs")

        # Perturbation + re-optimize
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            for attempt in range(5):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_seq + noise)
                t_p = min(rem * 0.4, 10.0)
                if t_p < 3:
                    continue
                a_pol, _ = topk_subgradient(a_pert, t_p, top_k=3, lr_init=0.005)
                try_seq(a_pol, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
