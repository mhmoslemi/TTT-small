Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth L-BFGS with adaptive temperature
2. Exact subgradient method with heavy-ball momentum
3. LP-based game iteration for polishing
4. Coordinate descent for final refinement

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_of_max_conv(a, k):
    """Compute subgradient of b_k = sum_i a_i * a_{k-i} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(max(0, k - n + 1), min(n, k + 1)):
        j = k - i
        grad[i] += a[j]
        if j != i:
            grad[j] += a[i]
    return grad

def subgradient_topk(a, top_k=10):
    """Subgradient of max(conv(a,a)) using top-k peaks."""
    n = len(a)
    b = fast_convolve(a)
    # Find top-k peaks
    peaks = np.argpartition(b, -top_k)[-top_k:]
    # Weight by proximity to max
    max_b = b.max()
    min_peak = b[min(peaks)]
    if max_b - min_peak < 1e-12:
        weights = np.ones(top_k) / top_k
    else:
        # Exponential weighting
        t = b[peaks]
        t_shift = t - t.max()
        w = np.exp(t_shift * 50)  # Sharp weighting
        w /= w.sum()
    
    total_grad = np.zeros(n)
    for i, peak in enumerate(peaks):
        total_grad += w[i] * subgradient_of_max_conv(a, int(peak))
    
    # Multiply by 2n for the objective 2n*max(conv)
    return 2.0 * n * total_grad, b, peaks, w

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: log-softmax L-BFGS, exact subgradient with momentum,
LP-based game iteration, and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cross_corr_valid(w, a):
    """
    Compute valid cross-correlation: result[j] = sum_k w[k] * a[k-j]
    where w has length 2n-1 and a has length n.
    result has length n.
    """
    n = len(a)
    conv_size = 2 * n - 1
    # Pad a to conv_size
    a_pad = np.zeros(conv_size)
    a_pad[:n] = a
    # FFT size for linear cross-correlation
    fft_size = 1 << (2 * conv_size - 1).bit_length()
    W = rfft(w, fft_size)
    Ap = rfft(a_pad, fft_size)
    # Cross-correlation: IFFT(conj(W) * Ap)
    result_full = irfft(np.conj(W) * Ap, fft_size)
    # The valid part: result[j] for j in [0, n-1]
    return result_full[:n]


def _soft_max_obj_grad_params(z, T, n):
    """
    Compute smooth max objective and gradient w.r.t. z (log-domain params).
    a = softmax(z), objective = (2n/T) * log(sum exp(T * b_k)) where b = conv(a,a).
    """
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute b = conv(a, a)
    b = _fast_convolve(a)
    
    # Soft max
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient w.r.t. a: grad_a = 2n * cross_corr(w, a)
    grad_a = 2.0 * n * _cross_corr_valid(w, a)
    
    # Chain rule through softmax: grad_z = a * (grad_a - a * dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def _subgradient_exact(a, top_k=20, sharpness=100.0):
    """
    Compute the exact subgradient of 2n*max(conv(a,a)) using top-k peaks.
    Returns (gradient, current_max, peak_indices, weights).
    """
    n = len(a)
    b = _fast_convolve(a)
    max_b = b.max()
    
    # Find top-k peaks
    if top_k >= len(b):
        peaks = np.arange(len(b))
    else:
        peaks = np.argpartition(b, -top_k)[-top_k:]
    
    # Exponential weighting based on proximity to max
    peak_vals = b[peaks]
    t = peak_vals - peak_vals.max()
    exp_t = np.exp(sharpness * t)
    w = exp_t / exp_t.sum()
    
    # Compute subgradient: 2n * cross_corr(w, a)
    # Where w is placed at the peak positions in a (2n-1)-length vector
    w_full = np.zeros(2 * n - 1)
    w_full[peaks] = w
    
    grad = 2.0 * n * _cross_corr_valid(w_full, a)
    
    return grad, max_b, peaks, w


def _phase_lbgfs(z0, n, temps, time_budget, maxiter_per_T=2000):
    """Run L-BFGS with temperature annealing in log-space."""
    z_curr = z0.copy()
    per_T_time = time_budget / len(temps)
    
    for T in temps:
        t0 = time.monotonic()
        remaining = max(1.0, per_T_time - (time.monotonic() - t0))
        
        def func(z):
            obj, _ = _soft_max_obj_grad_params(z, T, n)
            return obj
        
        def jac(z):
            _, g = _soft_max_obj_grad_params(z, T, n)
            return g
        
        try:
            res = minimize(
                fun=func,
                x0=z_curr,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': maxiter_per_T * 20, 'disp': False}
            )
            z_curr = res.x
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")
        
        elapsed = time.monotonic() - t0
        a_curr = _softmax_from_z(z_curr)
        score = _quick_score(a_curr)

        z_curr = _z_from_a(a_curr)
        
        if time.monotonic() - t0 > per_T_time * 1.5:
            break
    
    return z_curr


def _softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def _z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def _quick_score(a):
    """Quick score computation (assumes a is non-negative, sum > 0)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = _fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)


def _phase_subgradient(a0, time_budget=300.0, top_k=30, sharpness=50.0):
    """
    Exact subgradient method with heavy-ball momentum.
    Optimizes 2n*max(conv(a,a)) on the simplex.
    """
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * _fast_convolve(a).max()
    lb = 2.0 * n / (2.0 * n - 1)  # Lower bound
    
    v = np.zeros(n)  # Momentum
    lr = 0.01
    momentum = 0.9
    t0 = time.monotonic()
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_budget:
        step += 1
        grad, max_b, peaks, w = _subgradient_exact(a, top_k, sharpness)
        obj = 2.0 * n * max_b
        
        # Update velocity
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        
        # Evaluate
        new_obj = 2.0 * n * _fast_convolve(a_new).max()
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # Still move (might find a path to improvement)
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 50:
            lr *= 0.9
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(step)
                a = _project_simplex(best_a + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                lr = 0.01
                no_improve = 0
                sharpness *= 2.0  # Focus more on dominant peaks
        
        # Periodically increase sharpness (focus on top peaks)
        if step % 500 == 0:
            sharpness *= 1.1
    
    return best_a, best_obj


def _phase_lp_game(a0, time_budget=200.0):
    """
    LP-based alternating game.
    Alternate between:
    a. Fix f, solve LP for best response b
    b. Update f towards b
    """
    n = len(a0)
    f = a0.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = 2.0 * n * _fast_convolve(f).max()
    t0 = time.monotonic()
    
    iteration = 0
    while time.monotonic() - t0 < time_budget:
        iteration += 1
        conv_f = _fast_convolve(f)
        M = conv_f.max()
        cur_obj = 2.0 * n * M
        
        # Solve LP: max sum(b) s.t. b >= 0, (f * b)_k <= M for all k
        # The constraint (f * b)_k = sum_i f[i] * b[k-i] <= M
        # This is linear in b.
        
        # Build constraint matrix A_ub: A_ub[k, j] = f[k-j] (if 0<=k-j<n, else 0)
        # But this matrix is 2n-1 by n, which is 1535 x 768 for n=768.
        # That's fine for linprog.
        
        n_vars = n
        n_cons = 2 * n - 1
        
        # c = -ones (we want to maximize sum, so minimize -sum)
        c = -np.ones(n_vars)
        
        # Build A_ub: for each constraint k, A_ub[k, j] = f[k-j] if 0<=k-j<n
        A_ub = np.zeros((n_cons, n_vars))
        for k in range(n_cons):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A_ub[k, j] = f[k - j]
        
        b_ub = np.full(n_cons, M)
        
        bounds = [(0, None)] * n_vars
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-10:
                    b_norm = b / b_sum
                    
                    # Update f
                    t = 0.3  # Mixing parameter
                    f_new = (1 - t) * f + t * b_norm
                    f_new = _project_simplex(f_new)
                    
                    new_obj = 2.0 * n * _fast_convolve(f_new).max()
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_f = f_new.copy()
                    f = f_new
            else:
                print(f"  LP failed at iter {iteration}: {res.message}")
                break
        except Exception as e:
            print(f"  LP error at iter {iteration}: {e}")
            break
        
        if iteration % 20 == 0:
            print(f"  LP game iter {iteration}: obj={cur_obj:.6f}, best={best_obj:.6f}")
    
    return best_f, best_obj


def _phase_coord_descent(a0, time_budget=100.0):
    """Coordinate descent: adjust one element at a time."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * _fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = 2.0 * n * _fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj
