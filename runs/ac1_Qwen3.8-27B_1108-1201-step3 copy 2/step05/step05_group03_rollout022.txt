from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def fast_coord_descent(a, time_limit, verbose=True):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = normalize(a.copy()).astype(np.float64)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()  # score with sum=1
    
    t0 = time.monotonic()
    pass_count = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        # Priority-based ordering: start with coordinates that affect the hottest conv positions
        conv_max = conv.max()
        hot_threshold = conv_max * 0.95
        hot_positions = np.where(conv >= hot_threshold)[0]
        
        # Coordinates that affect hot positions
        affected = set()
        for k in hot_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected.update(range(js, je + 1))
        
        if affected:

            rest = np.array([j for j in range(n) if j not in affected])
            order = np.concatenate([priority, np.random.permutation(rest)])
        else:
            order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            v_old = a[j]
            
            # Golden section search bounds
            if v_old < 1e-12:
                lo, hi = 0.0, 0.1
            else:
                lo = max(0.0, v_old * 0.5)
                hi = min(1.0, v_old * 2.0)
            
            def obj(v):
                # Update conv incrementally
                delta = v - v_old
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if k_end <= k_start:
                    return conv[k_start:k_end].max() if len(conv) > 0 else 0
                idx = np.arange(k_start, k_end) - j
                updates = 2.0 * delta * a[idx]
                if 2 * j >= k_start and 2 * j < k_end:

                    updates[pos] = v * v - v_old * v_old

                a[j] = v
                result = conv.max()
                # Revert
                conv[k_start:k_end] -= updates
                a[j] = v_old
                return result
            
            try:

                if res.success and res.fun < conv.max() - 1e-12:
                    # Apply the update for real
                    v_new = res.x
                    delta = v_new - v_old
                    k_start = j
                    k_end = min(j + n, 2 * n - 1)
                    idx = np.arange(k_start, k_end) - j
                    updates = 2.0 * delta * a[idx]
                    if 2 * j >= k_start and 2 * j < k_end:
                        pos = 2 * j - k_start
                        updates[pos] = v_new * v_new - v_old * v_old
                    conv[k_start:k_end] += updates
                    a[j] = v_new
                    
                    new_score = 2.0 * n * conv.max()
                    if new_score < best_score - 1e-12:

        else:

    else:

        best_score = 2.0 * n * final_conv.max()

  - As the "second factor": a[k-j] * a[j] (term where i=k-j)
  
  If k-j ≠ j: total contribution = 2 * a[j] * a[k-j] = 2 * v_old * a[k-j]
  After change: 2 * v_new * a[k-j]
  Delta = 2 * (v_new - v_old) * a[k-j] ✓

  If k-j = j (i.e., k = 2j): contribution = a[j]^2 = v_old^2
  After change: v_new^2
  Delta = v_new^2 - v_old^2 ✓

So the formula is correct, and a[k-j] in the delta formula refers to the value BEFORE the change (which is the old value unless k-j = j, in which case we use the special formula).

"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using fast coordinate descent with
incremental O(n) convolution updates, priority-based coordinate ordering,
diverse structured initializations (power-law, cosine, Gaussian, arcsine,
bimodal), multi-scale refinement, and larger sequence lengths.
Key innovation: O(n) per coordinate evaluation (vs O(n log n) for FFT),
enabling many more optimization passes within the time budget.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fast_coord_descent(a, time_limit, verbose=False, hot_ratio=0.9, max_passes=500):
    """
    Fast coordinate descent with incremental O(n) convolution updates.
    Uses priority-based ordering (hot spots first) and golden-section 1D search.
    """
    n = len(a)
    a = normalize(a.copy()).astype(np.float64)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    t0 = time.monotonic()
    pass_count = 0
    stagnation = 0
    recompute_interval = max(1, n // 512)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        # Periodically recompute conv from scratch to avoid drift
        if pass_count % recompute_interval == 0:
            conv = fast_convolve(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()

        # Priority-based ordering
        conv_max = conv.max()
        hot_threshold = conv_max * hot_ratio
        hot_positions = np.where(conv >= hot_threshold)[0]

        if len(hot_positions) > 0:
            affected = set()
            for k in hot_positions:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                step = max(1, (je - js + 1) // 50)
                for jj in range(js, je + 1, step):
                    affected.add(jj)
            priority = np.array(sorted(affected), dtype=int)
            rest_mask = np.ones(n, dtype=bool)
            rest_mask[priority] = False
            rest = np.where(rest_mask)[0]
            np.random.shuffle(rest)
            order = np.concatenate([priority, rest])
        else:
            order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            v_old = a[j]

            # Determine search bounds
            if v_old < 1e-12:
                lo, hi = 0.0, 0.05
            else:
                lo = max(0.0, v_old * 0.4)
                hi = min(0.8, v_old * 2.5)
                if hi - lo < 1e-8:
                    lo = max(0.0, v_old - 0.02)
                    hi = min(0.8, v_old + 0.02)

            # Precompute index range for this coordinate
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            idx_arr = np.arange(k_start, k_end) - j
            special_pos = 2 * j - k_start if (2 * j >= k_start and 2 * j < k_end) else -1

            def obj_func(v):
                delta = v - v_old
                updates = 2.0 * delta * a[idx_arr]
                if special_pos >= 0:
                    updates[special_pos] = v * v - v_old * v_old
                conv[k_start:k_end] += updates
                a[j] = v
                result = conv.max()
                conv[k_start:k_end] -= updates
                a[j] = v_old
                return result

            try:
                res = minimize_scalar(obj_func, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 22})
                if res.success and res.fun < conv.max() - 1e-12:
                    v_new = res.x
                    delta = v_new - v_old
                    updates = 2.0 * delta * a[idx_arr]
                    if special_pos >= 0:
                        updates[special_pos] = v_new * v_new - v_old * v_old
                    conv[k_start:k_end] += updates
                    a[j] = v_new

                    new_score = 2.0 * n * conv.max()
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved:
            stagnation += 1
        else:
            stagnation = 0

        if stagnation >= 4:
            break
        if pass_count > max_passes:
            break

    # Final verification with FFT
    final_conv = fast_convolve(best_a)
    final_score = 2.0 * n * final_conv.max()
    if final_score < best_score:
        best_score = final_score
    else:
        best_score = final_score
        best_a = a.copy()
        best_final = 2.0 * n * fast_convolve(best_a).max()
        if best_final < best_score:
            best_score = best_final

    return best_a, best_score


def block_coord_descent(a, time_limit, block_size=8, verbose=False):
    """
    Block coordinate descent: optimize multiple coordinates simultaneously
    using scipy minimize with L-BFGS-B on the block.
    """
    n = len(a)
    a = normalize(a.copy()).astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Pick a random block
        start = rng.integers(0, n - block_size)
        block_idx = np.arange(start, start + block_size)

        def block_obj(x):
            new_a = a.copy()
            new_a[block_idx] = x
            return fast_convolve(new_a).max()

        x0 = a[block_idx].copy()
        try:
            res = minimize_scalar(lambda t: block_obj(
                np.maximum(x0 + t * (rng.standard_normal(block_size) - 0.5) * 0.1, 0)),
                bounds=(-1.0, 1.0), method='bounded', options={'xatol': 1e-8, 'maxiter': 30})
            if res.success:
                new_a = a.copy()
                new_a[block_idx] = np.maximum(
                    x0 + res.x * (rng.standard_normal(block_size) - 0.5) * 0.1, 0)
                new_a = normalize(new_a)
                sc = 2.0 * n * fast_convolve(new_a).max()
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = new_a.copy()
                    a = new_a.copy()
        except Exception:
            pass

    return best_a, best_score


def geometric_seq(n, r):
    a = r ** np.arange(n)
    return normalize(a)


def centered_geometric_seq(n, r):
    center = (n - 1) / 2.0
    a = r ** np.abs(np.arange(n) - center)
    return normalize(a)


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_seq(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2.0 * sigma ** 2))
    return normalize(a)


def triangular_seq(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def power_law_seq(n, p):
    a = (np.arange(n) + 1.0) ** (-p)
    return normalize(a)


def arcsine_seq(n):
    j = np.arange(n) + 0.5
    a = 1.0 / np.sqrt(j * (n - j))
    return normalize(a)


def bimodal_seq(n, frac=0.25):
    mu1 = frac * n
    mu2 = (1.0 - frac) * n
    sigma = 0.1 * n
    a = np.exp(-(np.arange(n) - mu1) ** 2 / (2 * sigma ** 2)) + \
        np.exp(-(np.arange(n) - mu2) ** 2 / (2 * sigma ** 2))
    return normalize(a)


def u_shaped_seq(n, sharpness=3.0):
    x = np.arange(n) / n - 0.5
    a = (1.0 + sharpness * (2.0 * x) ** 2)
    return normalize(a)


def smooth_random_seq(n, rng, kernel_size=21):
    a = rng.random(n)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
