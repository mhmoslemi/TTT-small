Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with softmax parameterization, continuation on smoothing parameter,
Adam optimizer, and intensive coordinate descent polishing.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_smoothed_conv(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    # For numerical stability
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(exp_shifted.sum())
    smooth_max = max_c + tau * log_sum
    w = exp_shifted / exp_shifted.sum()
    # Gradient of smooth_max w.r.t. a
    grad = 2.0 * np.correlate(a, w, 'full')[:n]
    return smooth_max, grad, conv, w

def softmax(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def objective_value(a):
    """Compute the actual objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def gradient_descent_softmax(a_init, time_limit, tau_start=5.0, tau_end=0.1, 
                              verbose=False):
    """
    Gradient descent on the smoothed objective using softmax parameterization.
    a = softmax(x), optimize over x.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    # Initialize x from a (inverse softmax)
    x = np.log(a + 1e-30)
    
    # Adam parameters
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    
    # Continuation schedule
    n_stages = 8
    tau_schedule = np.geomspace(tau_start, tau_end, n_stages)
    
    for stage, tau in enumerate(tau_schedule):
        stage_time = (time_limit * (stage+1)/n_stages - time.monotonic() + t0)
        if time.monotonic() - t0 > time_limit:
            break
        
        # Reset Adam for each stage
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.5 / np.sqrt(n)  # Initial learning rate
        
        while time.monotonic() - t0 < time_limit * (stage+1) / n_stages:
            t += 1
            
            # Compute gradient
            smooth_max, grad_a, conv, w = compute_smoothed_conv(softmax(x), tau)
            
            # Convert to gradient w.r.t. x
            # grad_x = a * (grad_a - a @ grad_a)
            a_curr = softmax(x)
            grad_x = a_curr * (grad_a - a_curr @ grad_a)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * grad_x**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check improvement
            a_new = softmax(x)
            obj = objective_value(a_new)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_new.copy()
            
            # Adaptive lr
            if t % 100 == 0 and verbose:
                print(f"      Stage {stage}, tau={tau:.3f}, obj={best_obj:.8f}, t={t}")
        
        if verbose:
            print(f"    Stage {stage} done (tau={tau:.3f}): best_obj={best_obj:.8f}")
    
    return best_a, best_obj

    log_sum = max_shifted + np.log(exp_shifted.sum())
    smooth_max = max_c + tau * log_sum
    w = exp_shifted / exp_shifted.sum()
    # Gradient: d smooth_max / da[i] = 2 * sum_k w[k] * a[k-i]
    grad = 2.0 * np.correlate(a, w, 'full')[:n]
    return smooth_max, grad, conv, w


def smooth_obj_and_grad(x, tau, n):
    """Objective and gradient for scipy minimize. x is the raw (unnormalized) vector."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a_norm = a / s
    smooth_max, grad_a, conv, w = compute_smoothed_grad(a_norm, tau)
    obj = 2.0 * n * smooth_max
    # d obj / da_norm = 2*n * grad_a
    # da_norm/dx is more complex; use chain rule with normalization
    # Simpler: work with normalized a directly
    grad = 2.0 * n * grad_a
    return obj, grad


def lbfgs_smooth(a_init, time_limit, tau_start=3.0, tau_end=0.05, verbose=False):
    """
    L-BFGS-B optimization with continuation on smoothing parameter.
    Works on the normalized simplex (sum=1, a>=0).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()

    # Continuation schedule
    n_stages = 10
    tau_schedule = np.geomspace(tau_start, tau_end, n_stages)

    for stage, tau in enumerate(tau_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        def fun_and_grad(x):
            a_local = np.maximum(x, 0.0)
            s_local = a_local.sum()
            if s_local < 1e-10:
                return 1e10, np.zeros(n)
            a_norm = a_local / s_local
            smooth_max, grad_a, conv, w = compute_smoothed_grad(a_norm, tau)
            obj = 2.0 * n * smooth_max
            # Gradient w.r.t. x (unnormalized):
            # a_norm = x / s, so da_norm[i]/dx[j] = (delta_ij * s - x[i]) / s^2
            # d obj / dx[j] = sum_i (d obj / da_norm[i]) * (da_norm[i]/dx[j])
            # = sum_i (2*n*grad_a[i]) * (delta_ij - x[i]/s) / s
            # = 2*n * (grad_a[j] - sum_i grad_a[i]*x[i]/s) / s
            # = 2*n * (grad_a[j] - a_norm @ grad_a) / s
            grad_x = 2.0 * n * (grad_a - a_norm @ grad_a) / s_local
            return obj, grad_x

        # Run L-BFGS-B
        try:
            res = minimize(fun_and_grad, a.copy(), method='L-BFGS-B',
                           jac=True, bounds=[(0, None)] * n,
                           options={'maxiter': 500, 'ftol': 1e-12,
                                    'gtol': 1e-8, 'maxfun': 2000,
                                    'maxls': 50})
            a_new = np.maximum(res.x, 0.0)
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new / sn
                obj_new = objective_value(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at stage {stage}: {e}")

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Continuation stage {stage} (tau={tau:.4f}): "
                  f"best_obj={best_obj:.10f} ({elapsed:.1f}s)")

    return best_a, best_obj


def coordinate_descent_gs(a0, time_limit, verbose=False):
    """
    Intensive coordinate descent with golden section search.
    For each coordinate, find optimal value via 1D search while rescaling others.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            a_rest_sum = a.sum() - old_val  # = 1 - old_val

            if a_rest_sum < 1e-12:
                continue

            # Golden section search for optimal a[i]
            # The objective as function of t=a[i] (others scaled by (1-t)/(1-old_val))
            # is the max of quadratics, hence convex in t.
            lo = 0.0
            hi = min(1.0 - 1e-10, max(2.0 * old_val + 0.005, 0.01))

            if old_val > 0.95:
                hi = 0.999
                lo = 0.5

            # Evaluate at golden section points
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618

            def eval_at(t):
                if t < 0 or t > 1 - 1e-10:
                    return 1e10
                if abs(t - old_val) < 1e-15:
                    return best_obj
                a_trial = a.copy()
                a_trial[i] = t
                scale = (1.0 - t) / a_rest_sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] *= scale
                return objective_value(a_trial)

            c = lo + (1.0 - gr) * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = lo + (1.0 - gr) * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_at(d)

            optimal_t = (lo + hi) / 2.0
            if abs(optimal_t - old_val) < 1e-14:
                continue

            # Apply the best value found
            a_trial = a.copy()
            a_trial[i] = optimal_t
            scale = (1.0 - optimal_t) / a_rest_sum
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            a_trial[mask] *= scale
            a_trial = np.maximum(a_trial, 0)
            sn = a_trial.sum()
            if sn < 1e-10:
                continue
            a_trial /= sn

            obj_trial = objective_value(a_trial)
            if obj_trial < best_obj - 1e-13:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True

        if verbose and pass_count % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f} ({elapsed:.1f}s)")

        if not improved and pass_count > 1:
            break
        if not improved:
            # Try random restart within coordinate descent
            rng = np.random.default_rng(hash((pass_count, n)) % 2**31)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise, 1.0)

    return best_a, best_obj


def pair_exchange_search(a0, time_limit, seed=0):
    """Local search by exchanging mass between pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = objective_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    support = np.where(a > 1e-6)[0]
    if len(support) < 2:
        support = np.arange(n)

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        idx = rng.integers(0, len(support), size=2)
        i, j = int(support[idx[0]]), int(support[idx[1]])
        if i == j:
            continue
        if a[i] < 1e-10 or a[j] < 1e-10:
            continue

        # Try transferring some mass from j to i (and vice versa)
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            for direction in [1, -1]:
                transfer = direction * frac * min(a[i], a[j])
                new_i = a[i] + transfer
                new_j = a[j] - transfer
                if new_i < -1e-12 or new_j < -1e-12:
                    continue
                new_i = max(0.0, new_i)
                new_j = max(0.0, new_j)

                a_trial = a.copy()
                a_trial[i] = new_i
                a_trial[j] = new_j
                # Rescale to sum=1
                sn = a_trial.sum()
                if sn < 1e-10:
                    continue
                a_trial /= sn

                obj = objective_value(a_trial)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat cyclic spectrum, tapered at edges."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    # Apply a mild taper to reduce boundary effects
    taper = np.ones(n)
    edge = max(2, n // 20)
    for i in range(edge):
        taper[i] = 0.5 * (1 - np.cos(np.pi * i / edge))
        taper[n - 1 - i] = taper[i]
    a *= taper
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    a = rng.exponential(1.0, n)
    return a / a.sum()


def gaussian_init(n):
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()


def tapered_uniform_init(n, rng):
    """Uniform with random taper."""
    a = np.ones(n)
    edge = rng.integers(5, n // 4)
    for i in range(edge):
        f = 0.3 * (1 - np.cos(np.pi * i / edge))
        a[i] = f
        a[n - 1 - i] = f
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_step(a, tau, lr, momentum, v, top_k=80):
    """Single subgradient step with momentum."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        # grad contribution: 2 * w[i] * a[k - idx] for valid idx
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]

    grad_centered = grad - grad.mean()
    v = momentum * v - lr * grad_centered
    a_new = project_simplex(a + v, 1.0)
    return a_new, v, conv.max()


def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient optimization with adaptive parameters and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = objective_value(a)
    v = np.zeros(n)
    tau = 50.0
    lr = 0.03
    momentum = 0.85
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        if no_improve > 500:
            momentum = min(0.95, momentum + 0.003)
            lr = max(0.002, lr * 0.97)
        elif no_improve > 200:
            tau = min(400.0, tau * 1.02)
            momentum = min(0.93, momentum + 0.002)
        else:
            tau = min(300.0, tau * 1.003)

        a, v, conv_max = subgradient_step(a, tau, lr, momentum, v)
        obj = objective_value(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 600:
            rng_r = np.random.default_rng(iter_count * 13 + 7)
            noise_scale = 0.003 + 0.003 * min(no_improve // 600, 5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, 1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 50.0
            momentum = 0.85

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS-B with continuation,
    subgradient methods, intensive coordinate descent, and pair search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a = a / a.sum()
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = objective_value(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Generate diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    target_lengths = []
    if parent is not None:
        pn = len(parent)
        target_lengths.append(pn)
        for delta in [-64, -32, 0, 32, 64, 96]:
            tl = pn + delta
            if 256 <= tl <= 2048 and tl not in target_lengths:
                target_lengths.append(tl)

    inits = {}
    for n in target_lengths[:6]:
        candidates = []
        if parent is not None:
            candidates.append(("parent_interp", interpolate_sequence(parent, n)))
        candidates.append(("flat_spec", flat_spectrum_init(n, rng)))
        candidates.append(("tapered_uniform", tapered_uniform_init(n, rng)))
        candidates.append(("gaussian", gaussian_init(n)))
        candidates.append(("random", random_init(n, rng)))

        best_init = None
        best_init_obj = float('inf')
        for name, init in candidates:
            try:
                init = np.maximum(init, 0)
                ss = init.sum()
                if ss < 1e-10:
                    continue
                init = init / ss
                obj = objective_value(init)
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init
            except Exception:
                pass
        if best_init is not None:
            inits[n] = best_init
            print(f"  n={n}: best init obj={best_init_obj:.8f}")

    # === Phase 2: L-BFGS-B with continuation ===
    print(f"\n=== Phase 2: L-BFGS-B Continuation (best={best_score:.6f}) ===")
    sorted_inits = sorted(inits.items(), key=lambda x: objective_value(x[1]))
    n_optimize = min(3, len(sorted_inits))

    total_lbfgs_time = min(400, max(60, (deadline - time.monotonic()) * 0.3))
    for rank, (n_len, init) in enumerate(sorted_inits[:n_optimize]):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        fracs = [0.5, 0.3, 0.2]
        t_alloc = min(total_lbfgs_time * fracs[rank], rem)
        print(f"  L-BFGS n={n_len}, t={t_alloc:.0f}s")
        try:
            a_opt, obj = lbfgs_smooth(init, time_limit=t_alloc, verbose=(rank == 0))
            try_seq(a_opt, f"lbfgs_n{n_len}_r{rank}")
        except Exception as e:
            print(f"    Error: {e}")

    # === Phase 3: Subgradient optimization ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        for rank, (n_len, init) in enumerate(sorted_inits[:2]):
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 100)
            print(f"  SG n={n_len}, t={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(init, time_limit=t_alloc)
            try_seq(a_opt, f"sg_n{n_len}_r{rank}")

            # Also try from current best
            if rank == 0 and best_seq is not None:
                t_alloc2 = min(rem * 0.2, 50)
                if t_alloc2 > 10:
                    a_opt2, obj2 = subgradient_optimize(best_seq, time_limit=t_alloc2)
                    try_seq(a_opt2, "sg_best")

    # === Phase 4: Intensive coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.5, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_gs(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_main")

    # === Phase 5: Pair exchange search ===
    print(f"\n=== Phase 5: Pair Exchange (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_ps = min(rem * 0.3, 80)
            print(f"  PS t={t_ps:.0f}s")
            for ps_seed in [0, 1, 2]:
                a_opt, obj = pair_exchange_search(best_seq, time_limit=t_ps // 3, seed=ps_seed + seed)
                try_seq(a_opt, f"ps_{ps_seed}")

    # === Phase 6: More L-BFGS from best with perturbations ===
    print(f"\n=== Phase 6: Refined L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem / (5 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, 1.0)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init, 1.0)
            else:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, 1.0)

            a_opt, obj = lbfgs_smooth(f_init, time_limit=t_alloc,
                                       tau_start=2.0, tau_end=0.03)
            try_seq(a_opt, f"lbfgs_refine_{p}")

            # Quick CD polish
            rem = deadline - time.monotonic() - 40
            if rem > 15:
                a_opt2, obj2 = coordinate_descent_gs(best_seq, time_limit=min(rem, 40))
                try_seq(a_opt2, f"cd_refine_{p}")

    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best - 32, n_best + 32, n_best + 64]:
            if target_n < 256 or target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 60)
            print(f"  Length n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_smooth(f_interp, time_limit=t_alloc,
                                       tau_start=2.0, tau_end=0.05)
            try_seq(a_opt, f"len_{target_n}")

            rem = deadline - time.monotonic() - 25
            if rem > 10:
                a_opt2, obj2 = coordinate_descent_gs(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt2, f"cd_len_{target_n}")

    # === Phase 8: Final intensive coordinate descent ===
    print(f"\n=== Phase 8: Final CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = coordinate_descent_gs(best_seq, time_limit=rem, verbose=True)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
