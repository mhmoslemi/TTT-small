from scipy.optimize import linprog
import numpy as np

def lp_direction(f, M):
    n = len(f)
    # Build constraint matrix: A[k,j] = f[k-j] if valid
    # conv(f,g)_k = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if 0<=k-j<n
    
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (vector of M's)
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
Nesterov-accelerated subgradient, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, M, top_k=50):
    """Compute subgradient of max(conv(a,a)) at a, using top-k positions."""
    n = len(a)
    conv = fast_convolve(a)
    # Find top-k positions
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    # Weight by how close to max
    vals = conv[peaks]
    max_val = vals.max()
    # Softmax weighting
    tau = 0.01 * max_val
    w = np.exp((vals - max_val) / tau)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue

        if sg > 1e-10:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, top_k=50):
    """Compute subgradient of max(conv(a,a)) using top-k positioning."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    vals = conv[peaks]
    max_val = vals.max()
    if max_val <= 0:
        return np.zeros(n), conv
    tau = 0.05 * max_val
    w = np.exp((vals - max_val) / tau)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]

    c = -np.ones(n)
    bounds = [(0, None)] * n

                        options={'time_limit': time_limit, 'presolve': True})

def coordinate_descent(a0, time_limit):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()
            
            for mult in [0.6, 0.8, 1.25, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
Nesterov-accelerated subgradient fallback, coordinate descent, and
multi-restart strategy. Key innovation: LP finds the optimal feasible
direction at each step, enabling much faster convergence than pure
subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, top_k=50):
    """Compute subgradient of max(conv(a,a)) using top-k positioning."""
    n = len(a)
    conv = fast_convolve(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    vals = conv[peaks]
    max_val = vals.max()
    if max_val <= 0:
        return np.zeros(n)
    tau = 0.05 * max_val
    w = np.exp((vals - max_val) / tau)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(vals)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, time_limit=5.0, use_active_set=True):
    """
    Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0.
    Uses active set reduction for speed. Returns g normalized to sum 1, or None.
    """
    n = len(f)
    t_start = time.monotonic()

    if use_active_set:
        conv_ff = fast_convolve(f)
        K = min(40, 2 * n - 1)
        top_k = np.argpartition(conv_ff, -K)[-K:]
        active_indices = set(int(x) for x in top_k)

        for round_i in range(8):
            if time.monotonic() - t_start > time_limit * 0.8:
                break
            idx_list = sorted(active_indices)
            A = build_conv_matrix(f)[idx_list, :]
            b = np.full(len(idx_list), M)
            c = -np.ones(n)
            bounds = [(0, None)] * n
            try:
                result = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs',
                                 options={'time_limit': time_limit * 0.3})
            except Exception:
                break
            if not result.success or result.x is None:
                break
            g = np.maximum(result.x, 0)
            sg = g.sum()
            if sg < 1e-10:
                return None
            # Check all constraints
            A_full = build_conv_matrix(f)
            conv_fg = A_full @ g
            violated = np.where(conv_fg > M * (1 + 1e-6) + 1e-12)[0]
            if len(violated) == 0:
                return g / sg
            for v in violated:
                active_indices.add(int(v))

    # Fallback: full LP
    A_full = build_conv_matrix(f)
    b_full = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_full, b_ub=b_full, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit})
    except Exception:
        return None
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        sg = g.sum()
        if sg > 1e-10:
            return g / sg
    return None


def subgrad_direction(f, top_k=50):
    """Compute subgradient-based direction normalized to sum 1."""
    n = len(f)
    grad = compute_subgrad(f, top_k=top_k)
    direction = -grad
    g = project_simplex(direction + 0.001 * np.ones(n), target_sum=1.0)
    return g


def max_conv_scalar(f, g, t):
    """Compute max(conv((1-t)f + tg, (1-t)f + tg))."""
    new_f = (1.0 - t) * f + t * g
    return fast_convolve(new_f).max()


def simple_line_search(f, g, M):
    """Try several step sizes and return the best (t, obj)."""
    candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
    best_t = 0.0
    best_obj = M
    for t in candidates:
        obj = max_conv_scalar(f, g, t)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    # Refine
    if 0 < best_t < 1:
        for dt in [0.03, -0.03, 0.01, -0.01]:
            t_new = best_t + dt
            if 0 <= t_new <= 1:
                obj = max_conv_scalar(f, g, t_new)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t_new
    return best_t, best_obj

def coordinate_descent(a0, time_limit):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()

            for mult in [0.5, 0.7, 0.85, 1.15, 1.35, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def pair_swap(a0, time_limit, seed=0, max_iter=300):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        support = np.arange(n)

    for _ in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        idx = rng.integers(0, len(support), size=2)
        i, j = int(support[idx[0]]), int(support[idx[1]])
        if i == j:
            continue
        if a[i] < 1e-12 or a[j] < 1e-12:
            continue
        for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
            for direction in [1, -1]:
                transfer = direction * frac * min(a[i], a[j])
                new_i = a[i] + transfer
                new_j = a[j] - transfer
                if new_i < 0 or new_j < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_i
                a_trial[j] = new_j
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_obj


def nesterov_subgrad(f0, time_limit, verbose=False):

    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    M = fast_convolve(a).max()
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = 0.05
    tau_smooth = 50.0
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        # Adaptive parameters
        if no_improve > 100:
            beta = min(0.98, beta + 0.003)
            lr = max(0.005, lr * 0.98)
        elif no_improve > 40:
            tau_smooth = min(200.0, tau_smooth * 1.02)

        grad = compute_subgrad(y, top_k=80)
        grad_centered = grad - grad.mean()

        v = beta * v - lr * grad_centered
        a_new = project_simplex(y + v, target_sum=1.0)

        # Nesterov update for y
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        alpha = (1.0 + beta) / t_curr
        y = a_new + alpha * (a_new - a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = a_new.copy()

        a = a_new
        obj = fast_convolve(a).max()

        if obj < best_M - 1e-12:
            best_M = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG iter {iter_count}: M={best_M:.10f}, lr={lr:.4f} ({elapsed:.0f}s)")

        if no_improve > 200:
            rng = np.random.default_rng(iter_count * 31 + 7)
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(best_a + noise, target_sum=1.0)
            y = a.copy()
            v = np.zeros(n)
            no_improve = 0
            lr = 0.04

    return best_a, best_M


def evaluate_and_update(state, a, label=""):
    """Evaluate sequence and update state dict if better."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-based direction finding,
    Nesterov subgradient, coordinate descent, and pair swap search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state['best_seq'] = parent.copy()
            state['best_score'] = p_score
    except Exception:
        parent = None

    if parent is None:
        n_init = 640
        parent = np.ones(n_init) / n_init
        p_score = float(evaluate_sequence(parent.tolist()))
        state['best_seq'] = parent.copy()
        state['best_score'] = p_score
        print(f"Default init: n={n_init}, score={p_score:.6f}")

    n_work = len(state['best_seq'])
    print(f"\nWorking length: n={n_work}, initial score={state['best_score']:.6f}")

    # === Phase 1: LP-based optimization (main, ~55% time) ===
    print(f"\n=== Phase 1: LP Optimization ({state['best_score']:.6f}) ===")
    t1 = min(850, (deadline - time.monotonic()) * 0.55)
    if t1 > 30:
        a_opt, obj = optimize_lp_subgrad(parent, time_limit=t1, verbose=True)
        evaluate_and_update(state, a_opt, "lp_main")

    # === Phase 2: Nesterov subgradient refinement (~15% time) ===
    print(f"\n=== Phase 2: Nesterov SG ({state['best_score']:.6f}) ===")
    if state['best_seq'] is not None:
        t2_total = min(250, (deadline - time.monotonic()) * 0.15)
        if t2_total > 30:
            n_best = len(state['best_seq'])
            for p_idx in range(3):
                rem = deadline - time.monotonic() - 120
                if rem < 20:
                    break
                t_alloc = min(rem / (3 - p_idx), 90)
                if p_idx == 0:
                    f_init = state['best_seq'].copy()
                elif p_idx == 1:
                    noise = rng.standard_normal(n_best) * 0.003
                    f_init = project_simplex(state['best_seq'] + noise, target_sum=1.0)
                else:
                    noise = rng.standard_normal(n_best) * 0.007
                    f_init = project_simplex(state['best_seq'] + noise, target_sum=1.0)
                a_opt, obj = nesterov_subgrad(f_init, time_limit=t_alloc, verbose=False)
                evaluate_and_update(state, a_opt, f"nsg_{p_idx}")

    # === Phase 3: LP refinement on best (~10% time) ===
    print(f"\n=== Phase 3: LP Refine ({state['best_score']:.6f}) ===")
    if state['best_seq'] is not None:
        t3 = min(180, (deadline - time.monotonic()) * 0.10)
        if t3 > 20:
            a_opt, obj = optimize_lp_subgrad(state['best_seq'], time_limit=t3, verbose=False)
            evaluate_and_update(state, a_opt, "lp_refine")

    # === Phase 4: Coordinate descent (~10% time) ===
    print(f"\n=== Phase 4: Coordinate Descent ({state['best_score']:.6f}) ===")
    if state['best_seq'] is not None:
        t4 = min(120, (deadline - time.monotonic()) * 0.10)
        if t4 > 15:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=t4)
            evaluate_and_update(state, a_opt, "cd")

    # === Phase 5: Pair swap (~8% time) ===
    print(f"\n=== Phase 5: Pair Swap ({state['best_score']:.6f}) ===")
    if state['best_seq'] is not None:
        t5 = min(100, (deadline - time.monotonic()) * 0.08)
        if t5 > 15:
            a_opt, obj = pair_swap(state['best_seq'], time_limit=t5, seed=seed, max_iter=400)
            evaluate_and_update(state, a_opt, "pair_swap")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish ({state['best_score']:.6f}) ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic() - 25
        if rem > 40:
            # Another LP pass
            t6a = min(rem * 0.5, 80)
            a_opt, obj = optimize_lp_subgrad(state['best_seq'], time_limit=t6a, verbose=False)
            evaluate_and_update(state, a_opt, "final_lp")

        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 40))
            evaluate_and_update(state, a_opt, "final_cd")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = pair_swap(state['best_seq'], time_limit=min(rem, 20),
                                   seed=seed + 1, max_iter=100)
            evaluate_and_update(state, a_opt, "final_ps")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = min(state['best_score'], fs)
    else:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
