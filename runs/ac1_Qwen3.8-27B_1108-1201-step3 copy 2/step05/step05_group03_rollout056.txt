```python
Minimizes 2*n*max(auto_conv)/(sum)^2 using smooth gradient descent
with log-sum-exp approximation, coordinate descent polishing,
and multi-restart strategy.

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, tau):
    """Compute gradient of log-sum-exp(b/tau) w.r.t. a, using FFT-based cross-correlation."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / tau
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    # grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
    cross = np.correlate(weights, a, mode='full')
    grad = 2.0 * cross[:n]
    return grad
```

```python
"""
Minimizes 2*n*max(auto_conv)/(sum)^2 using:
1. Smooth gradient descent (log-sum-exp) with adaptive temperature and Nesterov momentum
2. Precise coordinate descent with exact 1D bounded search
3. Pair-move local search
4. Multi-restart with diverse structured initializations
5. Dimension exploration and interpolation
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, tau):
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / tau
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    cross = np.correlate(weights, a, mode='full')
    grad = 2.0 * cross[:n]
    return grad


def smooth_gradient_descent(a, time_limit, tau0=None, lr0=0.05, momentum=0.9,
                            tau_decay=0.999, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    b = fast_convolve(a)
    if tau0 is None:
        tau0 = 0.05 * b.max()
    tau = tau0
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    step = 0
    f_prev = compute_score(a)

    while time.monotonic() - t0 < time_limit:
        step += 1
        grad = smooth_gradient(a, tau)
        v = momentum * v - lr0 * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        tau *= tau_decay
        if tau < 1e-10 * fast_convolve(a).max():
            tau = 1e-10 * fast_convolve(a).max()
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step {step}: score={best_score:.10f}, tau={tau:.6e}, t={elapsed:.1f}s")
    return best_a, best_score


def coordinate_descent(a, time_limit, max_passes=200, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit and no_improve_count < 3:
        pass_count += 1
        improved = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                candidates = [0.0001, 0.001, 0.005, 0.01]
            else:
                candidates = None

            def obj_1d(v):
                vv = max(0.0, v)
                new_a = a.copy()
                new_a[j] = vv
                conv = fast_convolve(new_a)
                s = new_a.sum()
                if s < 1e-15:
                    return float('inf')
                return 2.0 * n * conv.max() / (s * s)

            if candidates is not None:
                best_c = old_val
                best_f = obj_1d(old_val)
                for c in candidates:
                    f = obj_1d(c)
                    if f < best_f:
                        best_f = f
                        best_c = c
                if best_c != old_val and obj_1d(best_c) < best_score:
                    a[j] = best_c
                    a = normalize(a)
                    sc = compute_score(a)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            else:
                lo = max(0.0, old_val * 0.2)
                hi = min(1.0, old_val * 5.0)
                if hi - lo < 1e-8:
                    lo = max(0.0, old_val - 0.02)
                    hi = min(1.0, old_val + 0.02)
                try:
                    res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-10, 'maxiter': 40})
                    if res.success:
                        new_val = res.x
                        if new_val != old_val and res.fun < obj_1d(old_val) - 1e-14:
                            a[j] = new_val
                            a = normalize(a)
                            sc = compute_score(a)
                            if sc < best_score - 1e-14:
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                except Exception:
                    pass

        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0
        if verbose and pass_count % 10 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

    return best_a, best_score


def pair_local_search(a, time_limit, n_tries=500, verbose=False):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(777)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break
            count += 1
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            a_i, a_j = a[i], a[j]
            if a_i < 1e-14 and a_j < 1e-14:
                continue
            fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]
            best_f = best_score
            best_delta = 0.0
            for f in fracs:
                delta = a_j * f
                if delta < 1e-15:
                    continue
                new_a = a.copy()
                new_a[j] = a_j - delta
                new_a[i] = a_i + delta
                if new_a[j] < -1e-12:
                    continue
                conv = fast_convolve(new_a)
                s = new_a.sum()
                if s < 1e-15:
                    continue
                sc = 2.0 * n * conv.max() / (s * s)
                if sc < best_f:
                    best_f = sc
                    best_delta = delta
            if best_delta != 0.0:
                a[j] -= best_delta
                a[i] += best_delta
                a = normalize(a)
                if best_f < best_score:
                    best_score = best_f
                    best_a = a.copy()

        if verbose and count % 5000 == 0:
            print(f"    PS count {count}: score={best_score:.10f}")
        if count % 2000 == 0:
            current_sc = compute_score(a)
            if current_sc < best_score:
                best_score = current_sc
                best_a = a.copy()

    return best_a, best_score


def geometric_seq(n, r):
    a = r ** np.arange(n, dtype=float)
    return normalize(a)


def cosine_seq(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return normalize(a)


def gaussian_seq(n, sf=0.3):
    x = np.arange(n) - n / 2.0
    sig = sf * n
    a = np.exp(-x ** 2 / (2 * sig ** 2))
    return normalize(a)


def powerlaw_seq(n, beta, alpha=50.0):
    x = np.arange(n, dtype=float)
    a = (1.0 + x / alpha) ** (-beta)
    return normalize(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def two_bump_seq(n, center_frac=0.35, width_frac=0.15, ratio=0.4):
    x = np.arange(n, dtype=float)
    c1 = center_frac * n
    c2 = (1.0 - center_frac) * n
    w = width_frac * n
    a = np.exp(-(x - c1) ** 2 / (2 * w ** 2)) + ratio * np.exp(-(x - c2) ** 2 / (2 * w ** 2))
    return normalize(a)
```
