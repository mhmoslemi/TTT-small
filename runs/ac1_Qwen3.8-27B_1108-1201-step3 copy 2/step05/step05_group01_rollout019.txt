    n = len(f)

    F = np.fft.rfft(f, fft_size)

    """Smooth log-sum-exp objective in softmax parameter space."""

    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    # Gradient
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t

def lbfqg_optimize(f0, time_limit, verbose=False):

        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_f = f.copy()

    t = np.log(f + eps)
    
    betas = [10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]
    t0_time = time.monotonic()

        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break

        max_iter = max(50, int(200 * sub_time / max(remaining/len(betas), 1)))
        
        try:
            res = minimize(
                smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': min(max_iter, 500), 'gtol': 1e-10, 'maxls': 100,
                         'maxfun': min(5000, max_iter * 20)}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def support_refinement(f, threshold, time_limit):
    """Optimize only on the support of f."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)
    
    support = np.where(f > threshold)[0]
    if len(support) < 2:
        return f, fast_convolve(f).max()
    
    # Optimize on support
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # LP on support
        n_s = len(support)
        f_s = f[support]
        M = fast_convolve(f).max()
        
        # Build constraint matrix (only for support variables)
        A_ub = np.zeros((2*n-1, n_s))
        for i, j in enumerate(support):
            A_ub[j:j+n, i] = f_s[i]
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n_s)
        bounds = [(0, None)] * n_s
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g_s = res.x
        g_sum = g_s.sum()
        if g_sum < 1e-15:
            break
        g_s = g_s / g_sum
        
        # Map back to full space
        g = np.zeros(n)
        g[support] = g_s
        g = g / g.sum()
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            break
        
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        M = fast_convolve(f_new).max()
        f = f_new

            best_obj = M
            best_f = f.copy()
            # Update support
            support = np.where(f > threshold * 0.5)[0]
            if len(support) < 2:
                break
        
    return best_f, best_obj

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses cutting-plane LP-based Frank-Wolfe, annealed L-BFGS in softmax space,
multi-length exploration with interpolation, and support-based refinement.
Key innovation: cutting-plane LP solver allows many more FW iterations,
and aggressive multi-length exploration finds better n/2 trade-offs.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution (f*g) via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_cutting_plane(f, M, n, max_cp=80, w_init=60, w_max=200):
    """Solve LP max sum(g) s.t. (f*g)[k] <= M, g >= 0 using cutting-plane."""
    conv_ff = fast_convolve(f)
    w = min(w_init, 2 * n - 1)
    K = np.sort(np.argpartition(conv_ff, -w)[-w:])

    for _ in range(max_cp):
        A_ub = np.zeros((len(K), n))
        for i in range(len(K)):
            k = int(K[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[i, js:je + 1] = f[js:je + 1]

        b_ub = np.full(len(K), M)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        g = res.x

        fg = cross_convolve(f, g)
        violation = fg.max() - M
        if violation < 1e-8 * M + 1e-12:
            return g

        new_k = int(np.argmax(fg))
        if new_k not in K:
            K = np.append(K, new_k)
            K = np.sort(K)
            if len(K) > w_max:
                fg_K = fg[K]
                remove_idx = int(np.argmin(fg_K))
                K = np.delete(K, remove_idx)
                K = np.sort(K)

    if res.success:
        return res.x
    return None

def smooth_obj_grad(t, beta):
    """Smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-18
    t = np.log(f + eps)

    betas = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0, 3000.0]
    t0_time = time.monotonic()
    total = len(betas)

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 15:
            break
        sub_time = remaining / (total - bi)

        max_iter = max(100, int(300 * sub_time / max(remaining / total, 1)))
        max_iter = min(max_iter, 800)

        try:
            res = minimize(
                smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-12, 'maxls': 200,
                         'maxfun': min(8000, max_iter * 30)}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def interpolate_sinc(f, new_n):
    """Interpolate using sinc-based (bandlimited) interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    # Use FFT-based interpolation (bandlimited)
    scale = new_n / old_n
    # Zero-pad in frequency domain
    fft_old = np.fft.rfft(f, 2 * old_n)
    fft_new = np.fft.rfft(np.zeros(new_n))
    n_copy = min(new_n // 2 + 1, old_n)
    fft_new[:n_copy] = fft_old[:n_copy]
    f_interp = np.fft.irfft(fft_new, new_n)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def support_based_fw(f0, threshold, time_limit, verbose=False):
    """Frank-Wolfe restricted to the support of f0."""
    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, fast_convolve(f).max()
    f = f / s
    n = len(f)

    support = np.where(f > threshold)[0]
    if len(support) < 3:
        support = np.where(f > 0)[0]
    if len(support) < 3:
        return f, fast_convolve(f).max()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        M = fast_convolve(f).max()
        n_s = len(support)

        # Build constraints only for support variables
        A_ub = np.zeros((2 * n - 1, n_s))
        for i, j in enumerate(support):
            A_ub[j:j + n, i] = f[support[i]]
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n_s)
        bounds = [(0, None)] * n_s

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g_s = res.x
        gs = g_s.sum()
        if gs < 1e-15:
            break
        g_s = g_s / gs

        g = np.zeros(n)
        g[support] = g_s
        g = g / g.sum()

        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            break

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        M = fast_convolve(f_new).max()
        f = f_new

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose:
                print(f"    Support-FW: obj={best_obj:.10f}, |S|={len(support)}")
        else:
            no_improve += 1

        # Update support occasionally
        if no_improve % 20 == 0:
            new_support = np.where(f > threshold * 0.5)[0]
            if len(new_support) >= 3:
                support = new_support

        if no_improve > 100:
            break

    return best_f, best_obj


def mwu_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 50.0
    eta = 0.04
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            eta *= 0.98
            if no_improve > 400:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                eta = 0.03
                no_improve = 0

        eta = max(eta, 0.003)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining cutting-plane LP-FW, L-BFGS annealing,
    multi-length exploration, support-based refinement, and MWU.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    t_start = time.monotonic()

    # === Phase 1: Quick LP-FW on parent (150s) ===
    print("\n=== Phase 1: LP-FW cutting-plane on parent ===")
    t1 = min(150, max(30, int((deadline - time.monotonic()) * 0.12)))
    a_opt, obj1 = lp_fw_optimize(parent, time_limit=t1, verbose=True, use_cp=True, w_init=50, w_max=150)
    _try(a_opt, "lpfw_cp_parent")

    # === Phase 2: L-BFGS on best so far (250s) ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    t2 = min(250, max(40, int((deadline - time.monotonic()) * 0.18)))
    a_opt, obj2 = lbfqg_optimize(base, time_limit=t2, verbose=True)
    _try(a_opt, "lbfqg_main")

    # === Phase 3: Multi-length exploration (500s) ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_parent = len(base)

    # Try longer lengths
    longer_lengths = []
    for factor in [4.0 / 3.0, 2.0, 2.5, 3.0]:
        nl = int(n_parent * factor)
        if nl > n_parent and nl <= 4096 and nl not in longer_lengths:
            longer_lengths.append(nl)
    longer_lengths = sorted(longer_lengths)

    for target_n in longer_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 80:
            break

        f_interp = interpolate_sequence(base, target_n)
        also_sinc = interpolate_sinc(base, target_n)

        # LP-FW on linear interpolation
        t_alloc = min(200, int(rem * 0.45))
        print(f"  Multi-length n={target_n} (linear), t={t_alloc}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_cp=True, w_init=50, w_max=150)
        _try(a_opt, f"ml_lin_{target_n}")

        # L-BFGS on best at this length
        rem2 = deadline - time.monotonic() - 250
        if rem2 > 60 and best_seq is not None and len(best_seq) == target_n:
            t_alloc2 = min(120, int(rem2 * 0.4))
            print(f"  Multi-length n={target_n} (L-BFGS), t={t_alloc2}s")
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc2)
            _try(a_opt, f"ml_lbfqg_{target_n}")

        # Try sinc interpolation if time allows
        rem3 = deadline - time.monotonic() - 200
        if rem3 > 60:
            t_alloc3 = min(100, int(rem3 * 0.35))
            print(f"  Multi-length n={target_n} (sinc), t={t_alloc3}s")
            a_opt, _ = lp_fw_optimize(also_sinc, time_limit=t_alloc3, use_cp=True, w_init=50, w_max=150)
            _try(a_opt, f"ml_sinc_{target_n}")

    # === Phase 4: Support-based refinement (150s) ===
    print(f"\n=== Phase 4: Support refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        for thresh in [0.02, 0.01, 0.005]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(50, int(rem * 0.6))
            a_opt, _ = support_based_fw(best_seq, thresh, time_limit=t_alloc, verbose=True)
            _try(a_opt, f"support_{thresh}")

    # === Phase 5: Deep optimization on global best (300s) ===
    print(f"\n=== Phase 5: Deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        # 5a: LP-FW with wider working set
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            t_alloc = min(150, int(rem * 0.55))
            print(f"  Deep LP-FW, t={t_alloc}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True, use_cp=True, w_init=100, w_max=300)
            _try(a_opt, "deep_lpfw")

        # 5b: L-BFGS
        rem = deadline - time.monotonic() - 100
        if rem > 50:
            t_alloc = min(100, int(rem * 0.6))
            print(f"  Deep L-BFGS, t={t_alloc}s")
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "deep_lbfqg")

        # 5c: MWU
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(60, int(rem * 0.7))
            print(f"  Deep MWU, t={t_alloc}s")
            a_opt, _ = mwu_optimize(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "deep_mwu")

    # === Phase 6: Final polishing (remaining) ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempt = 0
        while time.monotonic() < deadline - 15:
            attempt += 1
            rem = deadline - time.monotonic() - 10
            if rem < 20:
                break

            if attempt % 4 == 1:
                noise = rng.standard_normal(n_best) * 0.001
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(40, int(rem * 0.7))
                a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc, use_cp=True, w_init=80, w_max=250)
                _try(a_opt, f"polish_lpfw_{attempt}")
            elif attempt % 4 == 2:
                t_alloc = min(40, int(rem * 0.7))
                a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc)
                _try(a_opt, f"polish_lbfqg_{attempt}")
            elif attempt % 4 == 3:
                t_alloc = min(30, int(rem * 0.6))
                a_opt, _ = support_based_fw(best_seq, 0.005, time_limit=t_alloc)
                _try(a_opt, f"polish_sup_{attempt}")
            else:
                # Try slightly different length
                new_n = n_best + (64 if n_best < 2048 else -64)
                if new_n < 128:
                    continue
                f_interp = interpolate_sequence(best_seq, new_n)
                t_alloc = min(30, int(rem * 0.6))
                a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_cp=True, w_init=50, w_max=150)
                _try(a_opt, f"polish_len{new_n}_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
