    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_direction_weighted(f, M, n, weights):
    """Solve LP: max sum(w*g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -weights
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """Improved LP-based Frank-Wolfe with adaptive M and better line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 20.0:
                # Try with weighted objective
                conv_now = fast_convolve(f)
                w = np.exp(np.maximum(0, (conv_now.max() - conv_now) * 50.0))
                w = w / w.sum()
                g = solve_lp_direction_weighted(f, M * m_exp, n, w)
                m_exp = 1.0
            if g is None:
                m_exp = 1.0
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 10.0)
            continue

        # Line search: backtracking + golden section
        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(20):
                mid = 0.618 * (hi - lo) + lo
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    hi = mid
                else:
                    lo = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 10.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * sum_m a[m] * w[m+i] (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def nesterov_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient descent on smooth surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)
    t_prev = t.copy()

    betas = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)

        # Nesterov parameters
        mu = 0.99
        lr = 0.5 / (beta * np.sqrt(n))
        y = t.copy()

        it = 0
        obj_prev = smooth_obj_grad(t, beta)[0]
        no_improve = 0

        while time.monotonic() - t0_time < t0_time + sub_time:
            it += 1
            obj, grad = smooth_obj_grad(y, beta)

            t_new = y - lr * grad
            y = mu * t_new + (1 - mu) * t

            a = softmax_fn(y)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj_actual = fast_convolve(a).max()

            if obj_actual < best_obj - 1e-14:
                best_obj = obj_actual
                best_f = a.copy()
                no_improve = 0
                t = y.copy()
                if verbose and it % 50 == 0:
                    print(f"    Nesterov beta={beta} iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
                if no_improve > 100:
                    break
                t = y.copy()

            if objective_decreased := (obj < obj_prev):
                lr = min(lr * 1.02, 2.0 / (beta * np.sqrt(n)))
            else:
                lr *= 0.95
            obj_prev = obj

        t = y.copy()
        a = softmax_fn(t)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a / a_sum
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = a.copy()

    return best_f, best_obj


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth approximation with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [10.0, 30.0, 60.0, 120.0, 250.0, 500.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            res = minimize(
                smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 50,
                         'maxfun': 5000, 'ftol': 1e-12}
            )
            t = res.x
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={beta} failed: {e}")
            break

    return best_f, best_obj


def peak_surgery(f0, time_limit, verbose=False):
    """Direct peak-targeting: reduce elements contributing to top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = 0.02
    top_k = 10
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Identify top peaks and their responsibility
        k_actual = min(top_k + it // 100, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks_sorted = peaks[np.argsort(conv[peaks])[::-1]]

        # Compute responsibility: for each element, how much does it contribute to peaks?
        resp = np.zeros(n)
        for k in peaks_sorted:
            k = int(k)
            contrib = conv[k]
            if contrib < 1e-15:
                continue
            # (a*a)[k] = sum_j a[j]*a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                partner = int(k - j)
                prod = f[j] * f[partner]
                if prod > 1e-20:
                    # Distribute responsibility: reduce the larger one more
                    frac = f[partner] / (f[j] + f[partner] + 1e-20)
                    frac2 = 1.0 - frac
                    resp[j] += contrib * frac * 0.5
                    resp[partner] += contrib * frac2 * 0.5

        # Normalize responsibility
        resp_max = resp.max()
        if resp_max > 1e-20:
            resp = resp / resp_max

        # Multiplicative update: reduce high-responsibility elements
        log_f = np.log(f + 1e-30) - eta * resp
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    PeakSurgery iter {it}: obj={best_obj:.10f}, eta={eta:.5f}")
        else:
            no_improve += 1
            eta *= 0.98

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            eta = 0.015
            no_improve = 0

        eta = max(eta, 0.001)

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Multiplicative Weight Update with smooth gradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta_ls = 50.0
    eta = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta_ls * (conv - conv_max)
        shifted -= shifted.max()
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        N = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(f, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        v = momentum * v + (1.0 - momentum) * grad_a

        log_f = np.log(f + 1e-30) - eta * v
        f_new = softmax_fn(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            eta *= 0.97
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                v = np.zeros(n)
                eta = 0.02
                no_improve = 0

        eta = max(eta, 0.002)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_perturb(f, rng, mode=0):
    """Generate structured perturbations of f."""
    n = len(f)
    if mode == 0:
        # Gaussian noise
        return project_simplex(f + rng.standard_normal(n) * 0.01)
    elif mode == 1:
        # Reversal blend
        return project_simplex(0.5 * f + 0.5 * f[::-1] + rng.standard_normal(n) * 0.005)
    elif mode == 2:
        # Circular shift
        shift = rng.integers(1, n // 2)
        shifted = np.roll(f, shift)
        return project_simplex(0.5 * f + 0.5 * shifted + rng.standard_normal(n) * 0.003)
    elif mode == 3:
        # Fourier phase perturbation
        F = np.fft.rfft(f)
        mag = np.abs(F)
        phase = np.angle(F) + rng.standard_normal(len(F)) * 0.1
        F_new = mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n)
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return project_simplex(f_new)
    elif mode == 4:
        # Piecewise constant perturbation
        block = max(1, n // 32)
        f_pert = f.copy()
        for start in range(0, n, block):
            end = min(start + block, n)
            f_pert[start:end] *= (1.0 + rng.standard_normal() * 0.05)
        f_pert = np.maximum(f_pert, 0)
        f_pert = f_pert / f_pert.sum()
        return project_simplex(f_pert)
    else:
        return project_simplex(f + rng.standard_normal(n) * 0.01)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW, Nesterov, L-BFGS, peak surgery,
    MWU, coordinate descent, and structured perturbation restarts.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 1200
    t1 = min(350, max(60, rem))
    if t1 > 40:
        a_opt, _ = lpfw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # LP-FW from reversed

        a_rev = parent[::-1].copy()
        a_opt, _ = lpfw_optimize(a_rev, time_limit=min(100, rem))
        try_seq(a_opt, "lpfw_reversed")

    # === Phase 2: Nesterov accelerated gradient ===
    print(f"\n=== Phase 2: Nesterov (best={best_score:.8f}) ===")

    rem = deadline - time.monotonic() - 1000

        a_opt, _ = nesterov_optimize(base, time_limit=t2, verbose=True)

    # === Phase 3: L-BFGS smooth optimization ===

    rem = deadline - time.monotonic() - 900

    if t3 > 40:
        a_opt, _ = lbfqg_optimize(base, time_limit=t3, verbose=True)
        try_seq(a_opt, "lbfqg_main")

    # L-BFGS from perturbed start
    rem = deadline - time.monotonic() - 800
    if rem > 40:
        f_pert = structured_perturb(base, rng, mode=3)
        a_opt, _ = lbfqg_optimize(f_pert, time_limit=min(120, rem))
        try_seq(a_opt, "lbfqg_fourier_pert")

    # === Phase 4: Peak surgery ===
    print(f"\n=== Phase 4: Peak Surgery (best={best_score:.8f}) ===")

        base = best_seq if best_seq is not None else parent
        a_opt, _ = fw_optimize(base, time_limit=t5)
        try_seq(a_opt, "mwu_main")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [640, 832, 896, 1024, 1152, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 80)
        if t_alloc < 25:
            t_alloc = 25
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc)
        try_seq(a_opt, f"len_{target_n}_lpfw")

        rem2 = deadline - time.monotonic() - 250
        if rem2 > 30:
            a_opt2, _ = lbfqg_optimize(f_interp, time_limit=min(rem2 * 0.5, 50))
            try_seq(a_opt2, f"len_{target_n}_lbfqg")

    # === Phase 7: Deep optimization on best ===
    print(f"\n=== Phase 7: Deep Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 7a: LP-FW deep
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem * 0.4, 180)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 7b: Nesterov deep
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 120)
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_nesterov")

        # 7c: L-BFGS deep
        rem = deadline - time.monotonic() - 100
        if rem > 25:
            t_alloc = min(rem * 0.5, 100)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # 7d: Peak surgery deep
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.6, 80)
            a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_surgery")

        # 7e: Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "deep_cd")

    # === Phase 8: Structured perturbation exploration ===
    print(f"\n=== Phase 8: Structured Perturbation (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        modes = [0, 1, 2, 3, 4]
        for attempt in range(10):
            rem = deadline - time.monotonic() - 20
            if rem < 20:
                break
            mode = modes[attempt % len(modes)]
            f_start = structured_perturb(best_seq, rng, mode=mode)
            t_alloc = min(rem * 0.5, 50)
            if t_alloc < 15:
                t_alloc = 15

            # Alternate between methods
            if attempt % 2 == 0:
                a_opt, _ = lpfw_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"pert_lpfw_{mode}")
            else:
                a_opt, _ = lbfqg_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"pert_lbfqg_{mode}")

    # === Phase 9: Final polishing ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Final CD pass
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_cd")

        # Final small LP-FW
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
