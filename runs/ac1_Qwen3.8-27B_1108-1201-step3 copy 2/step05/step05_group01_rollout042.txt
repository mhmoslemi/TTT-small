from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def solve_lp_primal(f, M, n):
    """Solve primal LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def solve_lp_dual(f, M, n):
    """Solve dual LP: min M*sum(y) s.t. A^T y >= 1, y >= 0.
    Then recover primal from dual via complementary slackness (or just use strong duality).
    Actually, for FW we need the primal solution. Let's solve the primal directly.
    But we can use the dual to get a better bound.
    """
    # Dual: min M * sum(y) s.t. for j=0..n-1: sum_{i=0}^{n-1} f[i]*y[i+j] >= 1, y >= 0
    c_dual = M * np.ones(2 * n - 1)
    A_ub_dual = np.zeros((n, 2 * n - 1))
    # A_ub_dual[j, i] = -f[i-j] for valid i (i.e., j <= i <= j+n-1)
    # We need: sum_{i=j}^{j+n-1} f[i-j] * y[i] >= 1
    # => -sum_{i=j}^{j+n-1} f[i-j] * y[i] <= -1
    for j in range(n):
        for i in range(j, j + n):
            A_ub_dual[j, i] = -f[i - j]
    b_ub_dual = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds, method='highs')
    return res

```python
def riemannian_adam(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, beta_smooth=50):
    n = len(f0)
    f = np.maximum(f0, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    t = np.log(f + 1e-30)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    step = 0
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_convolve(softmax_fn(t))
        conv_max = conv.max()
        shifted = beta_smooth * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient: grad_a = 2 * (w * f) (cross-correlation)
        N = 1 << (3 * n - 2).bit_length()
        F = np.fft.rfft(softmax_fn(t), N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(F), N)[:n]
        grad_a = 2.0 * R
        
        a = softmax_fn(t)
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        t = t - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = softmax_fn(t)
        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if step % 500 == 0:
                print(f"  RAdam step {step}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 200:
                # Anneal and restart
                lr *= 0.9
                no_improve = 0
                if lr < 1e-5:
                    break
        
        # Adaptively increase beta_smooth
        if step % 100 == 0:
            beta_smooth *= 1.05
    
    return best_f, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based Frank-Wolfe with adaptive M (primary)
2. Riemannian Adam on log-softmax parameterization (new adaptive optimizer)
3. Spectral POCS / frequency-domain equalization (new geometric approach)
4. Multi-length exploration with interpolation
5. Coordinate descent and subgradient polishing
Key innovations over previous: Riemannian Adam for adaptive per-coordinate
learning rates, and spectral POCS which operates in a fundamentally different
geometry (frequency domain) to escape local optima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n, time_budget=2.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M, line search, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 20.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_exp = min(m_exp * 1.1, 10.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        # Standard FW step size
        alpha_fw = 2.0 / (it + 2)
        alphas = sorted(set([0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5,
                             0.7, 1.0, alpha_fw, 2 * alpha_fw]))

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 10.0)

        if no_improve > 120:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 120, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def riemannian_adam(f0, time_limit, lr=0.05, beta_smooth_init=20.0):
    """Riemannian Adam optimizer on log-softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    t = np.log(f + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta_smooth = beta_smooth_init

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    # Precompute FFT size
    N = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        step += 1
        if step % 1000 == 0:
            beta_smooth = min(beta_smooth * 1.1, 500.0)

        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()
        shifted = beta_smooth * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient computation via FFT
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_t
        v = 0.999 * v + 0.001 * grad_t ** 2
        m_hat = m / (1 - 0.9 ** step)
        v_hat = v / (1 - 0.999 ** step)

        t = t - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = softmax_fn(t)
        obj = fast_convolve(f_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if step % 5000 == 0:
                print(f"    RAdam step {step}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 500:
                lr = max(lr * 0.9, 0.001)
                no_improve = 0
                if lr < 0.001:
                    break

    return best_f, best_obj


def spectral_pocs(f0, time_limit, relax=0.3):
    """Spectral POCS: iterate between frequency-domain flattening and time-domain projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    N = 1 << (2 * n - 1).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Zero-pad and FFT
        fp = np.zeros(N, dtype=np.complex128)
        fp[:n] = f
        F = np.fft.fft(fp)

        # Compute power spectrum
        power = np.abs(F) ** 2
        avg_power = power.sum() / N

        # Flatten: move magnitudes toward average
        F_new = F.copy()
        for k in range(1, N // 2 + 1):
            mag = np.abs(F[k])
            if mag > 1e-30:
                new_mag = (1 - relax) * mag + relax * np.sqrt(avg_power)
                phase = np.angle(F[k])
                F_new[k] = new_mag * np.exp(1j * phase)
            # Handle conjugate symmetry
            if k < N // 2:
                F_new[N - k] = np.conj(F_new[k])

        # IDFT
        fp_new = np.fft.ifft(F_new).real

        # Truncate and project to non-negative
        f_new = np.maximum(fp_new[:n], 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            # Try with less aggressive flattening
            relax = max(relax * 0.5, 0.01)
            continue
        f_new = f_new / s_new

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            if it % 5000 == 0:
                print(f"    POCS iter {it}: obj={best_obj:.10f}, relax={relax:.3f}")
        elif it % 5000 == 0 and relax > 0.05:
            relax = max(relax * 0.8, 0.05)

        f = f_new

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.75, 2.0]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    if scale >= 0:
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_f


def diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    j = np.arange(n)

    inits.append(rc / rc.sum())

    inits.append(hann / hann.sum())

    inits.append(tri / tri.sum())

    sigma = n / 6

    g2 = np.exp(-0.5 * ((j - n * 0.7) / sigma) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    exp_t = np.exp(-j / (n / 4))
    inits.append(exp_t / exp_t.sum())

    # Random with smoothing
    r = rng.random(n)
    # Simple box filter smoothing
    kernel = np.ones(5) / 5
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based Frank-Wolfe with adaptive M and time-limited solves (primary)
2. Riemannian Adam on log-softmax parameterization (adaptive per-coordinate LR)
3. Spectral POCS / frequency-domain equalization (different geometry)
4. Multi-length exploration with interpolation
5. Coordinate descent and subgradient polishing
Key innovations: Riemannian Adam for adaptive learning, spectral POCS for
frequency-domain escape from local optima, diverse structured initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n, time_limit_lp=1.5):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_lp})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    """LP-based Frank-Wolfe with adaptive M, line search, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    it = 0

        if g_sum < 1e-15:

            no_improve += 1
            if no_improve > 80:

            m_exp = min(m_exp * 1.1, 10.0)
            continue

        best_val = M

                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 10.0)

        if no_improve > 120:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 120, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def riemannian_adam(f0, time_limit, lr=0.05, beta_smooth_init=20.0):
    """Riemannian Adam optimizer on log-softmax parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    t = np.log(f + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta_smooth = beta_smooth_init

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    N = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        step += 1
        if step % 800 == 0:
            beta_smooth = min(beta_smooth * 1.15, 800.0)

        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()
        shifted = beta_smooth * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Gradient via FFT cross-correlation
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R

        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_t
        v = 0.999 * v + 0.001 * grad_t ** 2
        m_hat = m / (1.0 - 0.9 ** step)
        v_hat = v / (1.0 - 0.999 ** step)

        t = t - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = softmax_fn(t)
        obj = fast_convolve(f_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if step % 5000 == 0:
                print(f"    RAdam step {step}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 400:
                lr = max(lr * 0.85, 0.0005)
                no_improve = 0
                if lr < 0.001:
                    break

    return best_f, best_obj


def spectral_pocs(f0, time_limit, relax=0.3):
    """Spectral POCS: iterate between frequency flattening and time-domain projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    N = 1 << (2 * n - 1).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        fp = np.zeros(N, dtype=np.complex128)
        fp[:n] = f
        F = np.fft.fft(fp)

        power = np.abs(F) ** 2
        avg_power = power.sum() / N
        target_mag = np.sqrt(avg_power)

        F_new = F.copy()
        for k in range(1, N // 2 + 1):
            mag = np.abs(F[k])
            if mag > 1e-30:
                new_mag = (1.0 - relax) * mag + relax * target_mag
                phase = np.angle(F[k])
                F_new[k] = new_mag * np.exp(1j * phase)
            if k < N // 2:
                F_new[N - k] = np.conj(F_new[k])

        fp_new = np.fft.ifft(F_new).real
        f_new = np.maximum(fp_new[:n], 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            relax = max(relax * 0.5, 0.02)
            continue
        f_new = f_new / s_new

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            if it % 5000 == 0:
                print(f"    POCS iter {it}: obj={best_obj:.10f}, relax={relax:.3f}")
        elif it % 3000 == 0 and relax > 0.05:
            relax = max(relax * 0.85, 0.05)

        f = f_new

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.75, 2.0]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    if scale >= 0:
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    j = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * (j + 0.5) / n))
    inits.append(rc / rc.sum())

    # Hann window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * j / n))
    inits.append(hann / hann.sum())

    # Triangle (Fejer-like)
    tri = np.minimum(j + 1.0, n - j)
    inits.append(tri / tri.sum())

    # Two Gaussians
    sigma = n / 6.0
    g1 = np.exp(-0.5 * ((j - n * 0.3) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((j - n * 0.7) / sigma) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Exponential taper
    exp_t = np.exp(-j / (n / 4.0))
    inits.append(exp_t / exp_t.sum())

    # Random smoothed
    r = rng.random(n)
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW, Riemannian Adam, Spectral POCS,
    coordinate descent, and subgradient methods with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    remaining = deadline - time.monotonic() - 750
    t1 = min(350, remaining)
    if t1 > 30:
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        a_opt = np.maximum(a_opt, 0.0)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  NEW BEST [lpfw_parent]: {score:.6f}")

    # LP-FW on reversed
    remaining = deadline - time.monotonic() - 700
    t1r = min(80, remaining)
    if t1r > 15:
        a_rev = parent[::-1].copy()
        a_rev = a_rev / a_rev.sum()
        a_opt, obj = lp_fw_optimize(a_rev, time_limit=t1r)
        a_opt = np.maximum(a_opt, 0.0)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  NEW BEST [lpfw_reversed]: {score:.6f}")

    # === Phase 2: Riemannian Adam ===
    print(f"\n=== Phase 2: Riemannian Adam (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    remaining = deadline - time.monotonic() - 600
    t2 = min(300, remaining)
    if t2 > 30:
        a_opt, obj = riemannian_adam(base, time_limit=t2, lr=0.05, beta_smooth_init=20.0)
        a_opt = np.maximum(a_opt, 0.0)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  NEW BEST [radam_main]: {score:.6f}")

        # Perturbed RAdam
        remaining = deadline - time.monotonic() - 500
        t2b = min(100, remaining)
        if t2b > 20:
            noise = rng.standard_normal(len(base)) * 0.02
            f_pert = project_simplex(base + noise)
            a_opt, obj = riemannian_adam(f_pert, time_limit=t2b, lr=0.03, beta_smooth_init=10.0)
            a_opt = np.maximum(a_opt, 0.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  NEW BEST [radam_perturbed]: {score:.6f}")

    # === Phase 3: Spectral POCS ===
    print(f"\n=== Phase 3: Spectral POCS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    remaining = deadline - time.monotonic() - 420

    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths[:5]:
        remaining = deadline - time.monotonic() - 150
        if remaining < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(remaining * 0.3, 70)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)

            best_score = score
            best_seq = a_opt.copy()
            print(f"  NEW BEST [len_{target_n}]: {score:.6f}")

        # RAdam on interpolated

        if remaining > 30:

            a_opt = np.maximum(a_opt, 0.0)
            score = float(evaluate_sequence(a_opt.tolist()))

                best_score = score
                best_seq = a_opt.copy()
                print(f"  NEW BEST [radam_len_{target_n}]: {score:.6f}")

    # === Phase 6: Polishing ===
    print(f"\n=== Phase 6: Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # SG polish
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:

            a_opt, obj = sg_polish(best_seq, time_limit=t_alloc,

                                   top_k=max(20, n_best // 8), temp=0.00005)

                best_seq = a_opt.copy()
                print(f"  NEW BEST [sg_polish]: {score:.6f}")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 50

            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  NEW BEST [cd]: {score:.6f}")

        # Fine RAdam
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:

    # Fallback

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
