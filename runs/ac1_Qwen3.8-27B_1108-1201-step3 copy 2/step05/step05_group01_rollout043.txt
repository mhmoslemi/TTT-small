def project_simplex(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    n=v.size
    if n==0: return v
    if not np.all(np.isfinite(v)): return np.ones(n)/n
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    ind=np.arange(1,n+1)
    cond = u*ind > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n)/n
    rho=int(np.nonzero(cond)[0][-1])
    theta=(cssv[rho]-1.0)/(rho+1.0)
    out=np.maximum(v-theta,0.0)
    s=float(np.sum(out))
    if s < 1e-300: return np.ones(n)/n
    return out/s

"""
Off-line search for non-negative step-function heights minimizing
    2*n*max(np.convolve(a,a)) / (sum(a)**2).

The script works on normalized sequences sum(a)=1 and minimizes
M = max(conv(a,a)).  It combines:
  - a structured multi-start initialization,
  - smooth subgradient / projected-polish in Fourier-accelerated form,
  - a one-dimensional exact coordinate search over multiplicative
    simplex-preserving moves,
  - L-BFGS on an annealed log-sum-exp smooth maximum approximation,
  - final zero-cleaning, symmetrization, and evaluation-based checks.

The function propose_candidate() is the entry point.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.asarray(np.convolve(a, a), dtype=np.float64)
    size = 2 * n - 1
    N = _next_power_of_two(size)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    return out[:size]


def local_m(p):
    n = len(p)
    if n == 0:
        return float("inf")
    c = fast_convolve(p)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if m < 0.0:
        m = 0.0
    return m


def local_score(p):
    n = len(p)
    if n == 0:
        return float("inf")
    return 2.0 * float(n) * local_m(p)


def eval_score(p):
    """Use the official evaluator if available, otherwise use the local formula."""
    arr = np.asarray(p, dtype=np.float64).ravel()
    if arr.size == 0:
        return float("inf")
    try:
        lst = arr.tolist()
        sc = evaluate_sequence(lst)
        if np.isfinite(sc):
            return float(sc)
    except Exception:
        pass
    return local_score(arr)


def softmax_fn(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        return np.ones(n, dtype=np.float64) / n
    e = np.exp(v - vmax)
    s = float(np.sum(e))
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.ones(n, dtype=np.float64) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u * ind > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return out / s


def cross_corr(p, w):
    """
    R[i] = sum_j p[j] * w[i+j] for i = 0..len(p)-1.
    Used to compute gradients of weighted convolution coefficients.
    """
    n = len(p)
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_power_of_two(3 * n - 2)
    A = np.fft.rfft(p, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)
    return R[:n]


def make_weights(c, top_k=24, temp_ratio=0.05):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = len(c)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = float(np.max(c))
    if m < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)

    if top_k is None or int(top_k) <= 0:
        s = 20.0 * (c - m)
        e = np.exp(s)
        ss = float(np.sum(e))
        if ss < 1e-300:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return e / ss

    k = int(top_k)
    if k >= n:
        idx = np.arange(n)
    else:
        idx = np.argpartition(c, -k)[-k:]

    vals = c[idx]
    vmax = float(np.max(vals))
    temp = max(m * float(temp_ratio), 1e-300)
    s = (vals - vmax) / temp
    e = np.exp(s)
    ss = float(np.sum(e))
    if ss < 1e-300:
        wtop = np.full(len(vals), 1.0 / len(vals), dtype=np.float64)
    else:
        wtop = e / ss

    w = np.zeros(n, dtype=np.float64)
    w[idx] = wtop
    return w


def smooth_obj_grad_t(t, beta):
    a = softmax_fn(t)
    c = fast_convolve(a)
    m = float(np.max(c))

    s = float(beta) * (c - m)
    e = np.exp(s)
    essum = float(np.sum(e))
    if essum < 1e-300:
        w = np.zeros(len(c), dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        obj = m
    else:
        w = e / essum
        obj = m + float(np.log(essum)) / float(beta)

    grad_a = 2.0 * cross_corr(a, w)
    dot = float(np.dot(a, grad_a))
    grad_t = a * (grad_a - dot)
    if not np.all(np.isfinite(grad_t)):
        return obj, np.zeros_like(t, dtype=np.float64)
    return obj, grad_t


def gradient_polish(p0, time_limit, top_k=24, temp_ratio=0.08,
                    min_temp=0.002, lr=0.02, eps=1e-18, max_no=800):
    p = normalize(p0)
    c = fast_convolve(p)
    best_m = float(np.max(c))
    best_p = p.copy()
    if time_limit <= 0:
        return best_p, best_m

    t0 = time.monotonic()
    limit = t0 + float(time_limit)
    cur_lr = float(lr)
    tr = float(temp_ratio)
    no = 0

    while time.monotonic() < limit:
        M = float(np.max(c))
        if M < best_m - max(1e-15, 1e-12 * best_m):
            best_m = M
            best_p = p.copy()

        w = make_weights(c, top_k, tr)
        grad = 2.0 * cross_corr(p, w)
        if not np.all(np.isfinite(grad)):
            break
        gnorm = float(np.linalg.norm(grad, ord=np.inf))
        if gnorm < 1e-18:
            break

        base = cur_lr / max(1.0, 10.0 * gnorm)
        if not np.isfinite(base) or base <= 0.0:
            break

        logp0 = np.log(p + eps)
        trials = (base, base * 0.5, base * 2.0)
        cand_q = None
        cand_m = M

        for trial in trials:
            q1 = project_simplex(p - trial * grad)
            m1 = float(np.max(fast_convolve(q1)))
            if m1 < cand_m:
                cand_m = m1
                cand_q = q1

            q2 = normalize(softmax_fn(logp0 - trial * grad))
            m2 = float(np.max(fast_convolve(q2)))
            if m2 < cand_m:
                cand_m = m2
                cand_q = q2

        tol = max(1e-15, 1e-12 * M)
        if cand_q is not None and cand_m < M - tol:
            p = cand_q
            c = fast_convolve(p)
            cur_lr = min(1.0, cur_lr * 1.05)
            tr = max(float(min_temp), tr * 0.97)
            no = 0
            M = float(np.max(c))
            if M < best_m - tol:
                best_m = M
                best_p = p.copy()
        else:
            no += 1
            cur_lr *= 0.7
            tr = min(0.2, tr * 1.15)
            if no > max_no:
                break

    return best_p, best_m


def coord_eval_scalar(A_valid, B_valid, A_out, i, x):
    s = 1.0 - x
    ss = s * s
    val = A_valid * ss + B_valid * (x * s)
    val[i] = val[i] + x * x
    m = float(np.max(val))
    ov = A_out * ss
    if ov > m:
        m = ov
    return m


def coord_search_best(A_valid, B_valid, A_out, i, old_x, cur_m,
                      num_samples=24, refine=35):
    num_samples = int(max(4, num_samples))
    xs = np.linspace(0.0, 1.0, num_samples, dtype=np.float64)
    extras = np.array([
        old_x,
        0.5 * old_x,
        1.5 * old_x,
        old_x + 0.05,
        old_x - 0.05,
    ], dtype=np.float64)
    xs = np.concatenate((xs, extras))
    xs = np.clip(xs, 0.0, 1.0)
    xs = np.unique(xs)

    S = xs.reshape(-1, 1)
    one_S = 1.0 - S
    s2 = S * S
    val = A_valid[None, :] * s2 + B_valid[None, :] * (S * one_S)
    val[:, i] += S[:, 0] * S[:, 0]
    vals = np.max(val, axis=1)
    outv = A_out * (S[:, 0] * S[:, 0])
    np.maximum(vals, outv, out=vals)

    idx = int(np.argmin(vals))
    best_x = float(xs[idx])
    best_val = float(vals[idx])

    if len(xs) > 1:
        if idx == 0:
            lo = 0.0
            hi = float(xs[1])
        elif idx == len(xs) - 1:
            lo = float(xs[-2])
            hi = 1.0
        else:
            lo = float(xs[idx - 1])
            hi = float(xs[idx + 1])
    else:
        lo, hi = 0.0, 1.0

    if hi <= lo:
        lo = max(0.0, best_x - 0.01)
        hi = min(1.0, best_x + 0.01)

    if hi - lo > 1e-9:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = coord_eval_scalar(A_valid, B_valid, A_out, i, c1)
        f2 = coord_eval_scalar(A_valid, B_valid, A_out, i, c2)
        for _ in range(int(refine)):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = coord_eval_scalar(A_valid, B_valid, A_out, i, c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = coord_eval_scalar(A_valid, B_valid, A_out, i, c2)
        xm = 0.5 * (lo + hi)
        fx = coord_eval_scalar(A_valid, B_valid, A_out, i, xm)
        if fx < best_val:
            best_x = xm
            best_val = fx

    return best_x, best_val


def coordinate_descent(p0, time_limit, samples=24, refine=35, tol=1e-14):
    p = normalize(p0)
    if time_limit <= 0:
        c = fast_convolve(p)
        return p, float(np.max(c))

    n = len(p)
    c = fast_convolve(p)
    cur = float(np.max(c))
    best_m = cur
    best_p = p.copy()

    t0 = time.monotonic()
    limit = t0 + float(time_limit)
    passes = 0

    while time.monotonic() < limit:
        passes += 1
        improved = False

        for i in range(n):
            if time.monotonic() >= limit:
                break

            pi = float(p[i])
            denom = 1.0 - pi
            if denom < 1e-12:
                continue

            contrib = 2.0 * pi * p
            contrib[i] = pi * pi
            c_no_i = c[i:i + n] - contrib
            np.maximum(c_no_i, 0.0, out=c_no_i)

            invD = 1.0 / denom
            invD2 = invD * invD
            A_valid = c_no_i * invD2
            B_valid = p * (2.0 * invD)
            B_valid[i] = 0.0

            out_max = 0.0
            if i > 0:
                m_out = float(np.max(c[:i]))
                if m_out > out_max:
                    out_max = m_out
            if i + n < len(c):
                m_out = float(np.max(c[i + n:]))
                if m_out > out_max:
                    out_max = m_out
            A_out = out_max * invD2

            best_x, best_val = coord_search_best(
                A_valid, B_valid, A_out, i, pi, cur,
                num_samples=int(samples), refine=int(refine)
            )

            if best_val < cur - float(tol):
                x = best_x
                scale = (1.0 - x) * invD
                ss = scale * scale

                p_backup = p.copy()
                c_backup = c.copy()

                if i > 0:
                    c[:i] *= ss
                if i + n < len(c):
                    c[i + n:] *= ss

                one = 1.0 - x
                new_valid = A_valid * (one * one) + B_valid * (x * one)
                new_valid[i] += x * x
                c[i:i + n] = new_valid

                p *= scale
                p[i] = x
                psum = float(np.sum(p))
                if psum > 0:
                    p /= psum

                new_m = float(np.max(fast_convolve(p)))
                if new_m < cur - float(tol):
                    cur = new_m
                    improved = True
                    if cur < best_m - float(tol):
                        best_m = cur
                        best_p = p.copy()
                else:
                    p[:] = p_backup
                    c[:] = c_backup

        c = fast_convolve(p)
        recomp = float(np.max(c))
        if recomp < best_m - 1e-14:
            best_m = recomp
            best_p = p.copy()
            cur = best_m
        elif recomp > cur + 1e-9:
            p = best_p.copy()
            c = fast_convolve(p)
            cur = best_m
        else:
            cur = recomp

        if not improved:
            break
        if passes > 5000:
            break

    return best_p, best_m


def lbfqg_optimize(p0, time_limit, betas=None, maxiter=None):
    p = normalize(p0)
    best_p = p.copy()
    best_m = local_m(p)
    if time_limit <= 0:
        return best_p, best_m

    n = len(p)
    if betas is None:
        if n <= 1024:
            betas = (20, 40, 80, 160, 320, 640)
        else:
            betas = (20, 40, 80, 160)

    if maxiter is None:
        if time_limit < 50:
            maxiter = 100
        elif time_limit < 200:
            maxiter = 200
        else:
            maxiter = 350

    pmax = float(np.max(p))
    eps = max(1e-12, 1e-7 * pmax)
    t = np.log(p + eps)

    t0 = time.monotonic()
    limit = t0 + float(time_limit)

    for beta in betas:
        if time.monotonic() >= limit:
            break
        try:
            res = minimize(
                smooth_obj_grad_t,
                t,
                args=(beta,),
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter),
                    "gtol": 1e-9,
                    "maxls": 30,
                }
            )
            t_new = res.x
            if not np.all(np.isfinite(t_new)):
                break
            t = t_new
        except Exception:
            break

        a = normalize(softmax_fn(t))
        m = local_m(a)
        if m < best_m - max(1e-15, 1e-12 * best_m):
            best_m = m
            best_p = a.copy()

    return best_p, best_m


def interpolate_sequence(f, new_n):
    f = normalize(f)
    old_n = len(f)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        g = np.full(new_n, float(f[0]), dtype=np.float64)
        return normalize(g)
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    g = np.interp(new_x, old_x, f)
    g = np.maximum(g, 0.0)
    return normalize(g)


def make_parametric_starts(n, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = []

    out.append(normalize(np.ones(n, dtype=np.float64)))

    base = np.sin(np.pi * x)
    for alpha in (0.5, 1.0, 2.0, 4.0):
        out.append(normalize(base ** alpha))

    for sig in (0.12, 0.18, 0.25, 0.32, 0.40):
        p = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        out.append(normalize(p))

    for amp in (0.3, 0.6, 0.85):
        p = 1.0 + amp * np.cos(2.0 * np.pi * x)
        p = np.maximum(p, 0.0)
        if float(np.sum(p)) > 0.0:
            out.append(normalize(p))

    for a in (0.5, 1.0, 2.0, 3.0):
        p = x ** a + (1.0 - x) ** a
        out.append(normalize(p))

    for _ in range(4):
        noise = rng.standard_normal(n).astype(np.float64)
        klen = min(16, n)
        kern = np.ones(klen, dtype=np.float64)
        kern /= float(np.sum(kern))
        noise = np.convolve(noise, kern, mode="same")
        p = np.exp(0.5 * noise)
        out.append(normalize(p))

    return out


def clean_small(p, frac=1e-10):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0)
    s = float(np.sum(p))
    if s < 1e-300:
        return p
    q = p.copy()
    q[q < frac * s] = 0.0
    qs = float(np.sum(q))
    if qs < 0.999 * s:
        return p
    return q / qs


def optimize_candidate(p0, time_budget, seed=0):
    p = normalize(p0)
    best_p = p.copy()
    best_m = local_m(p)
    n = len(p)
    rng = np.random.default_rng(seed)

    t_start = time.monotonic()
    limit = t_start + max(1.0, float(time_budget))

    rem = limit - time.monotonic()
    if rem > 30:
        t = min(0.22 * float(time_budget), rem - 5.0)
        if t > 10:
            q, m = gradient_polish(p, t, top_k=24, temp_ratio=0.08, lr=0.02)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()
            p = q.copy() if m < best_m * 1.03 else best_p.copy()

    rem = limit - time.monotonic()
    if rem > 40 and n <= 2048:
        t = min(0.10 * float(time_budget), rem - 5.0)
        if t > 10:
            pert = project_simplex(best_p + rng.standard_normal(n) * 0.01)
            q, m = gradient_polish(pert, t, top_k=8, temp_ratio=0.03, lr=0.02)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()
            p = q.copy() if m < best_m * 1.02 else best_p.copy()

    rem = limit - time.monotonic()
    if rem > 50:
        samples = 32 if n <= 1024 else 16
        t = min(0.26 * float(time_budget), rem - 5.0)
        if t > 15:
            q, m = coordinate_descent(best_p, t, samples=samples, refine=35)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()
            p = q.copy() if m < best_m * 1.02 else best_p.copy()

    rem = limit - time.monotonic()
    if rem > 80:
        t = min(0.18 * float(time_budget), rem - 10.0)
        if t > 20:
            start_l = project_simplex(best_p + rng.standard_normal(n) * 0.003)
            q, m = lbfqg_optimize(start_l, t)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()
            p = q.copy() if m < best_m * 1.02 else best_p.copy()

    rem = limit - time.monotonic()
    if rem > 30:
        t = min(0.10 * float(time_budget), rem - 5.0)
        if t > 10:
            q, m = gradient_polish(p, t, top_k=6, temp_ratio=0.015, lr=0.015)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()
            p = q.copy() if m < best_m * 1.02 else best_p.copy()

    rem = limit - time.monotonic()
    if rem > 40:
        samples = 40 if n <= 1024 else 16
        t = min(0.12 * float(time_budget), rem - 5.0)
        if t > 15:
            q, m = coordinate_descent(best_p, t, samples=samples, refine=40)
            if m < best_m - max(1e-15, 1e-12 * best_m):
                best_m = m
                best_p = q.copy()

    return best_p, best_m


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 90.0)
    rng = np.random.default_rng(seed)

    candidates = []
    parent = None

    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            parent = normalize(np.clip(arr, 0.0, 1000.0))
            candidates.append(parent)
            sym = 0.5 * (parent + parent[::-1])
            candidates.append(normalize(sym))
    except Exception:
        parent = None

    n_base = len(parent) if parent is not None else 768
    if n_base < 32:
        n_base = 768

    starts = make_parametric_starts(n_base, rng)
    if parent is not None:
        for s in starts[:8]:
            candidates.append(s)
    else:
        for s in starts:
            candidates.append(s)

    scored = []
    for p in candidates:
        p = normalize(p)
        if p.size == 0:
            continue
        sc = local_score(p)
        if np.isfinite(sc):
            scored.append((sc, len(scored), p))

    if not scored:
        fallback = np.ones(768, dtype=np.float64)
        scored.append((2.0, 0, fallback))

    scored.sort()
    best_p = scored[0][2].copy()
    best_score = eval_score(best_p)

    selected = []
    for _, _, p in scored[:5]:
        if len(selected) < 3:
            selected.append(p)
    if parent is not None and len(selected) < 4:
        selected.append(parent)

    allocs = (600, 150, 80, 40)
    for idx, p in enumerate(selected):
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        alloc = allocs[idx] if idx < len(allocs) else 40
        alloc = min(float(alloc), float(rem) * 0.9)
        q, _ = optimize_candidate(p, alloc, seed=idx + 10)
        sc = eval_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    rem = deadline - time.monotonic()
    if rem > 150:
        n_best = len(best_p)
        targets = []
        for N in (n_best, n_best * 2, int(n_best * 1.5), 1024, 1536, 2048):
            N = int(N)
            if N not in targets and 32 <= N <= 4096:
                targets.append(N)
        for N in targets:
            rem = deadline - time.monotonic()
            if rem < 90:
                break
            if N == n_best:
                continue
            q0 = interpolate_sequence(best_p, N)
            alloc = min(90.0, float(rem) * 0.45)
            q, _ = optimize_candidate(q0, alloc, seed=1000 + N)
            sc = eval_score(q)
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = q.copy()

    rem = deadline - time.monotonic()
    if rem > 200:
        alloc = min(float(rem) * 0.55, 500.0)
        q, _ = optimize_candidate(best_p, alloc, seed=555)
        sc = eval_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    rem = deadline - time.monotonic()
    if rem > 60:
        t = min(180.0, float(rem) * 0.6)
        samples = 40 if len(best_p) <= 1024 else 16
        q, _ = coordinate_descent(best_p, t, samples=samples, refine=45)
        sc = eval_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    rem = deadline - time.monotonic()
    if rem > 40:
        t = min(90.0, float(rem) * 0.7)
        q, _ = gradient_polish(best_p, t, top_k=5, temp_ratio=0.01, lr=0.012)
        sc = eval_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    rem = deadline - time.monotonic()
    if rem > 30:
        t = min(60.0, float(rem) * 0.7)
        q, _ = lbfqg_optimize(best_p, t)
        sc = eval_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    best_p = normalize(best_p)
    final_candidates = [best_p]
    final_candidates.append(clean_small(best_p, 1e-10))
    sym = normalize(0.5 * (best_p + best_p[::-1]))
    final_candidates.append(sym)

    for q in final_candidates:
        if q.size == 0:
            continue
        sc = eval_score(q)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()

    best_p = normalize(best_p)
    fs = eval_score(best_p)
    if not np.isfinite(fs) or best_p.size == 0:
        best_p = np.array([1.0], dtype=np.float64)

    return best_p.astype(np.float64).tolist()
