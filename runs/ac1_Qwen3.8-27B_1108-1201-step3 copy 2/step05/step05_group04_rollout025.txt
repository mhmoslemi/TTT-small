from scipy.sparse import csr_matrix

def solve_lp_direction(f, M):
    """
    Solve: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0
    Returns the optimal b vector.
    """
    n = len(f)
    # Constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=(0, None), method='highs')
    
    if res.success:
        return res.x
    return None
