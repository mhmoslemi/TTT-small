def compute_subgradient(f, w):
    """Compute the weighted subgradient of the max-conv objective."""
    n = len(f)
    # sg[j] = 2 * sum_m w[j+m] * f[m]
    # = 2 * conv(w, flip(f))[j + n - 1]
    flip_f = f[::-1]
    # Use FFT for convolution
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(flip_f, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)
    sg = 2.0 * conv_result[n-1 : 2*n-2]
    return sg

from scipy.optimize import linprog
import time


def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(f, w):
    """Compute weighted subgradient using FFT.
    sg[j] = 2 * sum_m w[j+m] * f[m]
    """
    n = len(f)
    flip_f = f[::-1]
    size = len(w) + n - 1

    F = np.fft.rfft(flip_f, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)

            m_exp *= 1.1
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                M = fast_convolve(f).max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 37 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and (it % 50 == 0):
                    print(f"    {label} LP-FW {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_refine(f0, time_limit, verbose=False):
    """Coordinate descent: try adjusting each element to reduce max peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_sweep_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)

        # Focus on elements that contribute to the top peaks
        top_k = min(20, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        # Get candidate indices from peaks
        candidate_j = set()
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                candidate_j.add(j)

        # Also add a random sample
        rng = np.random.default_rng(hash(tuple(f[::max(1, n//100)])))
        for j in rng.integers(0, n, size=min(50, n)):
            candidate_j.add(int(j))

        for j in sorted(candidate_j):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            # Try a range of adjustments
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                trial = f.copy()
                trial[j] = new_val
                # Renormalize others to keep sum = 1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = max(0, (1.0 - best_j_val) / denom)
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_sweep_improve += 1
            if no_sweep_improve >= 3:
                break
        else:
            no_sweep_improve = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: mirror descent with FFT subgradients,
    LP-Frank-Wolfe, coordinate refinement, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default_gauss")

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    init_candidates = [parent]

    pn = len(parent)
    for n_try in [512, 768, 1024]:
        if n_try == pn:
            continue
        init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_gaussian(n_try, 0.25))
        init_candidates.append(make_bell(n_try, 0.45))
        # Asymmetric: exponential decay
        a_asym = np.exp(-np.arange(n_try) / (n_try * 0.4))
        a_asym /= a_asym.sum()
        init_candidates.append(a_asym)
        # Half-Gaussian (concentrated on one side)
        x = np.arange(n_try) / n_try - 0.2
        a_half = np.exp(-3.0 * x ** 2)
        a_half = np.maximum(a_half, 0)
        a_half /= a_half.sum()
        init_candidates.append(a_half)

    # Evaluate all inits
    best_init = None
    best_init_obj = float('inf')
    for a in init_candidates:
        s = a.sum()
        if s < 1e-10:
            continue
        a_norm = a / s
        obj = fast_convolve(a_norm).max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm
        sc = try_seq(a_norm, f"init_n{len(a_norm)}")

    print(f"  Best init obj: {best_init_obj:.8f} (n={len(best_init)})")

    # === Phase 2: Main mirror descent on best init ===
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 350)
        print(f"  MD: t={t_alloc:.0f}s, n={len(best_init)}")
        a_opt, obj, iters = mirror_descent(best_init, time_limit=t_alloc,
                                           lr_init=0.015, T_init=0.003,
                                           T_decay=0.9997, verbose=True, label="main")
        try_seq(a_opt, "md_main")
        print(f"  MD done: {iters} iters, obj={obj:.8f}")

    # === Phase 3: LP-Frank-Wolfe ===
    print(f"\n=== Phase 3: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 40:
            t_alloc = min(rem, 250)
            print(f"  LP-FW: t={t_alloc:.0f}s, n={len(best_seq)}")
            a_opt, obj = lp_fw_step(best_seq, time_limit=t_alloc, max_iters=300,
                                    verbose=True, label="main")
            try_seq(a_opt, "lpfw_main")

    # === Phase 4: Second round MD (refined) ===
    print(f"\n=== Phase 4: Refined Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  MD refined: t={t_alloc:.0f}s, n={len(best_seq)}")
            # Use smaller temperature for more focus
            a_opt, obj, iters = mirror_descent(best_seq, time_limit=t_alloc,
                                               lr_init=0.01, T_init=0.001,
                                               T_decay=0.9993, verbose=True, label="refine")
            try_seq(a_opt, "md_refine")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    target_lengths = [1024, 896, 640, 512]
    if len(base) in target_lengths:
        target_lengths.remove(len(base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_n{target_n}")

        t_alloc = min(rem * 0.4, 120)
        print(f"  MD on n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj, iters = mirror_descent(f_interp, time_limit=t_alloc,
                                           lr_init=0.012, T_init=0.002,
                                           T_decay=0.9995, verbose=False, label=f"n{target_n}")
        try_seq(a_opt, f"md_n{target_n}")

        # Quick LP-FW on best of this length
        rem2 = deadline - time.monotonic() - 250
        if rem2 > 30:
            t_alloc2 = min(rem2 * 0.3, 60)
            a_opt2, obj2 = lp_fw_step(a_opt, time_limit=t_alloc2, max_iters=100,
                                      verbose=False, label=f"n{target_n}")
            try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 6: Coordinate refinement ===
    print(f"\n=== Phase 6: Coordinate Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  CD: t={t_alloc:.0f}s, n={len(best_seq)}")
            a_opt, obj = coordinate_refine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 7: Deep MD with multiple restarts ===
    print(f"\n=== Phase 7: Deep MD with Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem / (4 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
                lr, T = 0.008, 0.0008
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
                lr, T = 0.01, 0.0015
            elif p == 2:
                # Blend with uniform
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
                lr, T = 0.012, 0.002
            else:
                # Smaller perturbation
                noise = rng.standard_normal(n_best) * 0.002
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()
                lr, T = 0.009, 0.001

            print(f"  Deep MD variant {p}: t={t_alloc:.0f}s, lr={lr}, T={T}")
            a_opt, obj, iters = mirror_descent(f_init, time_limit=t_alloc,
                                               lr_init=lr, T_init=T,
                                               T_decay=0.9994, verbose=False, label=f"deep{p}")
            try_seq(a_opt, f"deep_{p}")

    # === Phase 8: Final LP-FW squeeze ===
    print(f"\n=== Phase 8: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_step(best_seq, time_limit=t_alloc, max_iters=200,
                                    verbose=False, label="final")
            try_seq(a_opt, "final_lpfw")

    # === Phase 9: Final CD polish ===
    print(f"\n=== Phase 9: Final CD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt, obj = coordinate_refine(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
