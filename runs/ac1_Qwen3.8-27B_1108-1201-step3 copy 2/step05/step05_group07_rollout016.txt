def frank_wolfe_step(a, T, n, step_size=None):
    """
    One Frank-Wolfe step using soft-max gradient.
    Returns new a.
    """
    b = fast_convolve(a)
    # Compute soft-max weights
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # Compute gradient
    grad_a = compute_soft_max_grad_a(a, n, T)
    
    # Frank-Wolfe: find vertex with minimum gradient
    j_star = np.argmin(grad_a)
    
    if step_size is None:
        # Standard FW step size
        # We'll use a fixed small step
        step_size = 0.05
    
    e = np.zeros(n)
    e[j_star] = 1.0
    a_new = (1 - step_size) * a + step_size * e
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new, grad_a, j_star


def fw_optimize(a0, T, n, time_limit, initial_step=0.1, decay=0.9995):

    step = initial_step
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        a_new, grad_a, j_star = frank_wolfe_step(a, T, n, step)
        # Check if this improved the exact max
        b_new = fast_convolve(a_new)
        max_new = b_new.max()
        
        if max_new < best_max - 1e-15:
            best_max = max_new

            a = a_new

                a = project_simplex(a + noise)
