"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (exponentiated gradient)
with adaptive top-K peak tracking, FFT-accelerated subgradients, coordinate descent
local refinement, and multi-phase time-budgeted search. Uses multiplicative updates
naturally suited to the simplex constraint, with progressive peak focus and periodic
local polishing to escape the slow convergence of pure subgradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """
    Cross-correlation: result[i] = sum_k w[k] * a[k-i] for i in [0, len(a)-1].
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    m = len(w)
    size = n + m - 1
    if size <= 0:
        return np.zeros(n)
    fft_size = 1 << (size - 1).bit_length()
    if fft_size < 4:
        fft_size = 4
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(W) * A, fft_size)[:n]
    return result


def compute_subgradient(a, w):
    """
    Compute subgradient of weighted sum of convolution peaks.
    g[i] = 2 * sum_k w[k] * a[k-i] (with a[x]=0 for x out of [0,n-1])
    """
    n = len(a)
    # g[i] = 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)[i]
    g = 2.0 * cross_corr_fft(w, a)
    return g


def mirror_descent_step(a, g, eta):
    """Multiplicative weight update: a_new[i] proportional to a[i] * exp(-eta * g[i])."""
    # Normalize g to avoid overflow
    g_max = g.max()
    if g_max < 1e-18:
        return a.copy()
    g_norm = g / g_max
    log_update = -eta * g_norm
    # Clamp to avoid numerical issues
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    a_new = a_new / s
    return a_new


def mirror_descent(a0, n_steps, eta0, eta_decay, beta0, beta_growth, time_limit, top_k_frac=0.1, verbose=False):
    """
    Mirror descent with progressive peak focus.
    
    Parameters:
    - a0: initial sequence (will be normalized to sum=1)
    - n_steps: maximum number of steps
    - eta0: initial step size
    - eta_decay: multiplicative decay per step for step size
    - beta0: initial softmax sharpness for peak weighting
    - beta_growth: multiplicative growth factor for beta per step
    - time_limit: seconds to run
    - top_k_frac: fraction of peaks to consider in weighting
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta0
    beta = beta0
    t0 = time.monotonic()
    
    # Progressively focus from broader peaks to the actual max
    # beta starts moderate and grows -> softmax weights become more peaked on the max
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        c_max = c.max()
        
        if c_max < best_obj - 1e-15:
            best_obj = c_max
            best_a = a.copy()
        
        # Compute softmax weights over convolution values
        # Focus on top peaks by using high beta
        c_shifted = c - c_max
        beta_c = beta * c_shifted
        beta_c = np.clip(beta_c, -100, 0)
        w = np.exp(beta_c)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(conv_len)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
        
        # Compute subgradient
        g = compute_subgradient(a, w)
        
        # Adaptive step size: scale by inverse of gradient magnitude
        g_max = g.max()
        if g_max < 1e-15:
            break
        eta_adapt = eta * (1.0 / (g_max + 0.01))
        eta_adapt = min(eta_adapt, 2.0)  # Cap step size
        
        a = mirror_descent_step(a, g, eta_adapt)
        
        # Update schedule
        eta *= eta_decay
        beta *= beta_growth
        
        if verbose and step % 10000 == 0:
            print(f"    MD step {step}: obj={c_max:.8f}, best={best_obj:.8f}, eta={eta:.4e}, beta={beta:.2f}")
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60.0, n_trials_per_pos=5):
    """
    Coordinate descent: for each position, try to reduce the max convolution
    by adjusting that position's value (redistributing to others proportionally).
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = [0.5, 0.7, 0.8, 1.2, 1.5, 1.8]
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                if new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a_sum = a.sum()
                    if a_sum > 1e-15:
                        a = a / a_sum
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_a, best_obj


def pairwise_swap_descent(a0, time_limit=60.0):
    """
    Try swapping mass between pairs of positions to reduce the max convolution.
    More effective than single-coordinate descent for flat landscapes.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    fractions = [0.1, 0.2, 0.3, 0.5, 0.7]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try random pairs
        n_attempts = max(50, n // 2)
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            for f in fractions:
                if f >= 1.0:
                    continue
                # Move fraction f of the difference from larger to smaller
                ai, aj = a[i], a[j]
                if ai > aj:
                    delta = (ai - aj) * f
                    a_trial = a.copy()
                    a_trial[i] = ai - delta
                    a_trial[j] = aj + delta
                else:
                    delta = (aj - ai) * f
                    a_trial = a.copy()
                    a_trial[j] = aj - delta
                    a_trial[i] = ai + delta
                
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            # Try more aggressive: move mass from high-value position to low-value
            for _ in range(n // 4):
                if time.monotonic() - t0 > time_limit:
                    break
                # Pick from top-value positions
                top_pos = np.argsort(a)[-min(20, n):]
                bot_pos = np.argsort(a)[:min(20, n)]
                i = top_pos[rng.integers(0, len(top_pos))]
                j = bot_pos[rng.integers(0, len(bot_pos))]
                if i == j or a[i] < 1e-12:
                    continue
                for f in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    delta = a[i] * f
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_bump_init(n, width_frac=0.6, seed=42):
    """Generate a bump-like initial condition (Gaussian centered)."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    center = 0.5
    width = width_frac / 2.0
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def generate_flat_init(n, seed=42):
    """Generate a slightly perturbed uniform initial condition."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) + rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def generate_two_bump_init(n, seed=42):
    """Two-bump initial condition."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.3) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.15) ** 2)
    a = a / a.sum()
    return a
