def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: optimize each coordinate while keeping sum=1.
    For coordinate i, the optimal value (given others fixed) minimizes
    the peak of the convolution.
    """

    s = a.sum()

    t0 = time.monotonic()
    epoch = 0
    
    while time.monotonic() - t0 < time_limit:
        epoch += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            # Try several values for a[i]
            old = a[i]
            best_i_val = old
            best_i_obj = best_obj
            
            # The contribution of a[i] to conv[k] is:
            # conv[k] += a[i] * a[k-i] (for each valid k)
            # So changing a[i] by delta changes conv[k] by delta * a[k-i]
            
            # Try reducing a[i] (redistribute to others proportionally)
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3]
            for mult in multipliers:
                new_val = old * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                trial = a.copy()
                trial[i] = new_val
                # Redistribute to maintain sum=1
                diff = old - new_val
                if abs(diff) > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a[mask].sum()
                    if others_sum > 1e-15:
                        trial[mask] += diff * a[mask] / others_sum
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                
                obj = fast_convolve(trial).max()
                if obj < best_i_obj - 1e-15:
                    best_i_obj = obj
                    best_i_val = new_val
            
            if best_i_val != old:
                a[i] = best_i_val
                diff = old - best_i_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum > 1e-15:
                    a[mask] += diff * a[mask] / others_sum
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_i_obj < best_obj - 1e-15:
                    best_obj = best_i_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:

                break
    
    return best_a, 2.0 * n * best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
