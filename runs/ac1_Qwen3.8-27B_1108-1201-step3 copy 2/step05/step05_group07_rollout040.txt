import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.0005, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):

    Projected subgradient method with Nesterov momentum and adaptive
    temperature-weighted peak selection. Optimizes on the simplex (sum=1).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        batch_end = min(it_count + 50, it_count + 50)
        while it_count < batch_end and time.monotonic() - t0 < time_limit:
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()
            
            # Compute weights over active peaks using temperature
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
            
            # Only use top peaks to reduce computation
            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx]
                w_sel /= w_sel.sum()
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full
            
            # Compute subgradient: g[i] = sum_k w[k] * a[k-i] * I[0<=k-i<n]
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def fast_convolve_grad(a):
    """Returns (convolution, and a helper for cross-correlation)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    return conv


def cross_corr_fft(w, a):
    """Cross-correlation via FFT: result[i] = sum_k w[k+i] * a[i] (valid range)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    best_obj = soft_max_obj_grad(z0, T, n)[0]
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    
    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit, n_trials_per_elem=8):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey (tapered cosine) window."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_skewed_init(n, alpha=2.0):
    """Beta-distribution shaped (skewed) initialization."""
    from scipy.stats import beta as beta_dist
    x = np.linspace(0, 1, n)
    a = beta_dist.pdf(x, alpha, alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_linear_init(n):
    """Linearly increasing initialization."""
    a = np.linspace(0.1, 1.0, n)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):

        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")
    
    n_parent = len(parent)

    print(f"\n=== Phase 1: Subgradient from parent (n={n_parent}) ===")
