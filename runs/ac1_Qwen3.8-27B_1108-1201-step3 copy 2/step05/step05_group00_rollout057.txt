def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')

    conv = fast_convolve(a_norm)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, n, num_peaks=50, temperature=0.001):
    """Compute a weighted subgradient of max_k (f*f)_k."""
    # Find peaks
    if num_peaks >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        # Use top peaks

        pv = conv[peak_idx]
        # Softmax weights
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        peaks = peak_idx
        weights = w
    
    # Compute subgradient
    sg = np.zeros(n)

        k = int(peaks[i])

        f = np.maximum(f, 0)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, n, num_peaks=50, temperature=0.001):
    """Compute a smoothed subgradient of max_k (f*f)_k."""
    if num_peaks >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peak_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
        pv = conv[peak_idx]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        peaks = peak_idx
        weights = w

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-10:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]

    return sg

        grad_b = f * (sg_a - mean_sg)

        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_b
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_b * grad_b)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)

        b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.exp(b_new - np.max(b_new))
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            b = b_new
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1

        temperature *= 1.001

        if no_improve > 250:
            rng = np.random.default_rng(it * 11 + 5)
            f_pert = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f_pert = f_pert / f_pert.sum()
            b = np.log(f_pert)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            f = f_pert
            no_improve = 0
            lr = min(lr * 1.5, 0.08)
            temperature = 0.0005

    return best_f, best_obj


def targeted_peak_reduction(f0, time_limit, verbose=False):
    """
    Target the largest convolution peak by reducing high-product pairs
    and redistributing to low-impact positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[peak_k - idx]

        order = np.argsort(products)[::-1]
        improved = False

        for num_reduce in [1, 2, 3, 5, 8]:
            if num_reduce >= len(idx):
                continue
            top_idx = idx[order[:num_reduce]]
            bot_idx = idx[order[-num_reduce:]]

            for frac in [0.03, 0.05, 0.08, 0.12, 0.18, 0.25]:
                trial = f.copy()
                total_red = 0.0
                for j in top_idx:
                    red = f[j] * frac
                    trial[j] -= red
                    total_red += red
                for j in bot_idx:
                    trial[j] += total_red / len(bot_idx)

                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    no_improve = 0
                    if verbose:
                        print(f"    TPR: k={peak_k}, frac={frac}, obj={best_obj:.10f}")
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 0)
                f = f / f.sum()
                no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False, m_expand_max=5.0):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': max(1, time_limit - (time.monotonic() - t0))})
        except Exception:
            break

        if not res.success or res.x is None:
            m_exp *= 1.1
            if m_exp > m_expand_max:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.08, m_expand_max)
            if no_improve > 80:
                break
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 100:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: mirror descent with momentum, Adam in log-space,
    targeted peak reduction, LP-FW, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        n = 512
        x = np.linspace(0, 1, n)
        parent = np.exp(-3 * (x - 0.5) ** 2)
        parent = parent / parent.sum()
        try_seq(parent, "default_init")

    n_parent = len(parent)
    print(f"  Starting with n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Mirror descent with momentum on parent ===
    print("\n=== Phase 1: Mirror Descent (Momentum) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 250)
        print(f"  MD-momentum: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_momentum(parent, time_limit=t_alloc, verbose=True,
                                            base_lr=0.008, num_peaks=50,
                                            temperature_init=0.0004, momentum=0.85)
        try_seq(a_opt, "md_mom_parent")

    # === Phase 2: Adam in log-space ===
    print("\n=== Phase 2: Adam Log-Space ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem, 150)
        print(f"  Adam: t={t_alloc:.0f}s")
        a_opt, obj = adam_log_optimize(base, time_limit=t_alloc, verbose=True, base_lr=0.03)
        try_seq(a_opt, "adam_parent")

    # === Phase 3: Targeted peak reduction ===
    print("\n=== Phase 3: Targeted Peak Reduction ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  TPR: t={t_alloc:.0f}s")
            a_opt, obj = targeted_peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "tpr")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length Exploration ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [n_parent]
    for t in [512, 640, 768, 896, 1024, 1280, 1536]:
        if t not in target_lengths and t >= 256:
            target_lengths.append(t)
    target_lengths = sorted(set(target_lengths))

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        if target_n == n_parent and best_seq is not None and len(best_seq) == n_parent:
            continue  # Already optimized at this length

        t_alloc = min(rem * 0.35, 80)
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_n{target_n}")
        if sc < best_score:
            base = f_interp

        print(f"  MD n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_momentum(f_interp, time_limit=t_alloc, verbose=False,
                                            base_lr=0.006, num_peaks=40,
                                            temperature_init=0.0005, momentum=0.85)
        sc = try_seq(a_opt, f"md_n{target_n}")
        if sc < best_score:
            base = a_opt

    # === Phase 5: LP-FW on best ===
    print("\n=== Phase 5: LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 6: Second round MD with different params ===
    print("\n=== Phase 6: MD Round 2 (different hyperparams) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  MD2 high-temp: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_momentum(best_seq, time_limit=t_alloc, verbose=False,
                                                base_lr=0.005, num_peaks=100,
                                                temperature_init=0.002, momentum=0.75,
                                                reheat_every=200)
            try_seq(a_opt, "md2_hightemp")

    # === Phase 7: Adam round 2 ===
    print("\n=== Phase 7: Adam Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  Adam2: t={t_alloc:.0f}s")
            a_opt, obj = adam_log_optimize(best_seq, time_limit=t_alloc, verbose=False, base_lr=0.01)
            try_seq(a_opt, "adam2")

    # === Phase 8: More TPR ===
    print("\n=== Phase 8: TPR Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  TPR2: t={t_alloc:.0f}s")
            a_opt, obj = targeted_peak_reduction(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "tpr2")

    # === Phase 9: Final diversified polish ===
    print("\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem / (6 - p), 40)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init = f_init / f_init.sum()
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif p == 3:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                noise = rng.standard_normal(n_best) * 0.001
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init = f_init / f_init.sum()

            if p % 2 == 0:
                a_opt, obj = mirror_descent_momentum(f_init, time_limit=t_alloc,
                                                    verbose=False, base_lr=0.003,
                                                    num_peaks=30, temperature_init=0.0002,
                                                    momentum=0.9)
            else:
                a_opt, obj = adam_log_optimize(f_init, time_limit=t_alloc,
                                              verbose=False, base_lr=0.01)
            try_seq(a_opt, f"polish_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
