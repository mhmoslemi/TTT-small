```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_reduced_lp(f, M, active_lags, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (f*g)[k] = sum_j f[j] * g[k-j] = sum_j f[k-j] * g[j] for valid j
        # So A_ub[i, j] = f[k-j] for j in [js, je]
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    b_ub = np.full(len(active_lags), M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent (softmax tempering of peak autocorrelation)
2. LP-based Frank-Wolfe with reduced active constraints
3. Fine polishing with tiny LP steps
4. Multi-length exploration and diverse restarts
"""

import numpy as np
from scipy.optimize import linprog
from numpy.lib.stride_tricks import sliding_window_view
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_subgradient(a, t):
    """Compute subgradient of softmax-smoothed max autocorrelation.
    
    Returns (objective_value, subgradient).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax probabilities
    scaled = conv / t
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    obj = t * (scaled.max() + np.log(exp_scaled.sum()))
    
    # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # Using sliding window view for efficiency
    windows = sliding_window_view(p, n)  # shape (n, n), windows[j, m] = p[j+m]
    grad = 2.0 * (windows @ a)
    
    return obj, grad


def smooth_grad_descent(f0, time_limit, t_init=0.005, t_min=0.0001, t_decay=0.999,
                        lr_init=0.002, momentum=0.9, verbose=False):
    """Smoothed gradient descent with momentum and decreasing temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0))
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)
    t = t_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        obj, grad = smooth_subgradient(f, t)
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate actual (unsmoothed) objective
        actual_obj = fast_convolve(f_new).max()
        f = f_new
        
        if actual_obj < best_obj - 1e-15:
            best_obj = actual_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      GD iter {it}: obj={best_obj:.12f}, t={t:.6f}")
        else:
            no_improve += 1
        
        # Anneal
        t *= t_decay
        lr *= 0.9998
        
        if no_improve > 200:
            # Restart with fresh momentum
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.0001)
            no_improve = 0
            
            # Small perturbation
            if it % 400 == 0:
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
        
        if t < t_min:
            t = t_min
    
    return best_f, best_obj
```

`[1'  0] [λ] = [1]`

We can solve this directly! And if some a_j come out negative, we set them to zero and re-solve (active set method).

This is essentially a **quadratic program with a Hankel matrix**! And QPs with structured matrices can be solved very efficiently.

```python
def hankel_qp_solve(w, n, max_active_set_iters=50):
    """Solve: min a^T H a s.t. sum(a)=1, a>=0 where H is Hankel with H[j,j']=w[j+j']."""
    # H[j, j'] = w[j+j'] where w has length 2n-1, zero-padded if needed
    # Build the constraint: 2*H@a + lambda*1 + mu = 0, mu >= 0, mu_j * a_j = 0
    # Start with all indices active (a_j > 0 assumed for all j)
    
    active = set(range(n))
    for _ in range(max_active_set_iters):
        # Build the linear system for the active indices
        idx = sorted(active)
        m = len(idx)
        if m == 0:
            return None
        
        # Build 2*H restricted to active indices
        # H[idx_i, idx_j] = w[idx_i + idx_j]
        H_sub = np.zeros((m, m))
        for i in range(m):
            for j in range(m):
                k = idx[i] + idx[j]
                if k < len(w):
                    H_sub[i, j] = w[k]
        
        # Solve: [2*H_sub  1] [a_sub]   [0]
        #        [1'       0] [lambda] = [1]
        A = np.zeros((m+1, m+1))
        A[:m, :m] = 2.0 * H_sub
        A[:m, m] = 1.0
        A[m, :m] = 1.0
        rhs = np.zeros(m+1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(A, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
        
        a_sub = sol[:m]
        lam = sol[m]
        
        # Check non-negativity
        if np.all(a_sub >= -1e-10):
            # Feasible! Check KKT for inactive indices
            a_full = np.zeros(n)
            a_full[idx] = np.maximum(a_sub, 0)
            a_full /= a_full.sum()  # Ensure sum = 1 (should already be)
            
            # Check: for inactive j, the gradient should be >= lambda
            # grad_j = 2 * sum_k w[k] * a[k-j] (valid indices)
            # We need grad_j >= lambda for all inactive j
            violated = False
            for j in range(n):
                if j not in active:
                    grad_j = 2.0 * sum(w[k] * a_full[k-j] for k in range(max(0, k-n+1), min(n-1, k)+1) if 0 <= k-j < n)
                    # Simplified: grad_j = 2 * sum_{k: 0<=k-j<n} w[k]*a[k-j]
                    # = 2 * sum_{m: 0<=m<n} w[j+m] * a[m]
                    grad_j = 2.0 * np.dot(w[j:j+n], a_full)
                    if grad_j < lam - 1e-8:
                        violated = True
                        active.add(j)
                        break
            
            if not violated:
                return a_full
        else:
            # Infeasible: remove the most negative component
            min_idx = np.argmin(a_sub)
            active.remove(idx[min_idx])
    
    # If we didn't converge, return the best we have
    if len(active) > 0:
        idx = sorted(active)
        m = len(idx)
        H_sub = np.zeros((m, m))
        for i in range(m):
            for j in range(m):
                k = idx[i] + idx[j]
                if k < len(w):
                    H_sub[i, j] = w[k]
        A = np.zeros((m+1, m+1))
        A[:m, :m] = 2.0 * H_sub
        A[:m, m] = 1.0
        A[m, :m] = 1.0
        rhs = np.zeros(m+1)
        rhs[m] = 1.0
        sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
        a_full = np.zeros(n)
        a_full[idx] = np.maximum(sol[:m], 0)
        ss = a_full.sum()
        if ss > 0:
            a_full /= ss
        return a_full
    return None
```

```python
def mirror_descent_dual(f0, time_limit, T_inner=50, eta_init=0.1, eta_decay=0.999, t_smoothing=0.001):
    """Dual mirror descent for min max autocorrelation.
    
    Outer: w = softmax(w + eta * autocorrelation(a*))
    Inner: a* = argmin sum_k w_k * g_k(a) (approximately, via GD in b-space)
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Initialize dual weights (uniform over all lags)
    w = np.ones(2*n - 1) / (2*n - 1)
    eta = eta_init
    
    # Softmax parameterization: a = softmax(b)
    b = np.log(np.maximum(f, 1e-30))
    
    outer_it = 0
    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        
        # Inner: minimize sum_k w_k * g_k(a) over a in simplex
        # Using gradient descent in b-space
        for _ in range(T_inner):
            a = softmax(b)
            conv = fast_convolve(a)
            
            # Gradient of sum_k w_k * g_k(a) with respect to a
            # = 2 * sum_k w_k * a_flipped_k
            # where a_flipped_k[j] = a[k-j] (valid indices)
            # = 2 * sum_k w_k * (sliding window of a)
            # More precisely: grad_a[j] = 2 * sum_k w_k * a[k-j] (valid)
            # = 2 * sum_{m: 0<=m<n} w[j+m] * a[m]
            
            # Compute using dot product for each j
            # grad_a[j] = 2 * dot(w[j:j+n], a)
            # This is a matrix-vector product
            W = sliding_window_view(w, n)  # shape (n, n), W[j, m] = w[j+m]
            grad_a = 2.0 * (W @ a)
            
            # Gradient with respect to b: da/db = a * (I - 1*a^T)
            # db_grad[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
            dot = np.dot(a, grad_a)
            grad_b = a * (grad_a - dot)
            
            # Update b
            b -= 0.01 * grad_b  # Fixed learning rate for inner GD
        
        # Evaluate the solution
        a_opt = softmax(b)
        conv = fast_convolve(a_opt)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = a_opt.copy()
        
        # Outer update: w = softmax(w + eta * conv)
        # Use a temperature for the outer softmax to avoid nums
        w_new = w + eta * (conv - conv.mean())
        # Normalize w to be a probability distribution
        w_new = np.exp(w_new / max(w_new.max() - w_new.min(), 1e-10))
        w = w_new / w_new.sum()
        # Truncate: keep only top-k lags
        if outer_it > 10:
            k = min(100, len(w))
            top_k = np.argsort(w)[-k:]
            w_masked = np.zeros_like(w)
            w_masked[top_k] = w[top_k]
            w = w_masked / w_masked.sum()
        
        eta *= eta_decay
    
    return best_f, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced combination of:
1. LP-based Frank-Wolfe with reduced active constraints (top-k lags)
2. Alternating dual-primal (Hankel QP) method for diverse basins
3. Coordinate descent and subgradient polishing
Key improvements over prior: much faster LPs (reduced constraints),
alternating method for escape from local minima, better line search,
adaptive M relaxation, and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
from numpy.lib.stride_tricks import sliding_window_view
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_safe(x, t=1.0):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted / t)
    return ex / ex.sum()


def solve_lp_reduced(f, M, active_lags, n, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    
    Only uses the specified active_lags as constraints (reduced LP).
    """
    k = len(active_lags)
    if k == 0:
        return np.ones(n) / n
    
    c = -np.ones(n)
    A_ub = np.zeros((k, n))
    for i, lag in enumerate(active_lags):
        js = max(0, lag - n + 1)
        je = min(n - 1, lag)
        for j in range(js, je + 1):
            A_ub[i, j] = f[lag - j]
    b_ub = np.full(k, M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_reduced(f0, time_limit, top_k=40, verbose=False):
    """LP-based Frank-Wolfe with reduced active constraints.
    
    Only includes top_k autocorrelation lags in the LP at each step.
    This makes each LP much faster, allowing more iterations.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0))
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t_start = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    lp_time = 20  # seconds per LP

    while time.monotonic() - t_start < time_limit:
        it += 1
        elapsed = time.monotonic() - t_start
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        # Get top-k active lags
        k_actual = min(top_k, 2 * n - 1)
        active_idx = np.argpartition(conv, -k_actual)[-k_actual:]
        active_idx = active_idx[np.argsort(conv[active_idx])[::-1]]

        # Solve reduced LP
        g = solve_lp_reduced(f, M * m_relax, active_idx, n, time_limit=lp_time)
        if g is None:
            m_relax = min(m_relax * 1.2, 10.0)
            if m_relax >= 10.0:
                # LP is infeasible even with large relaxation; perturb and reset
                rng = np.random.default_rng(it * 104729 + 7)
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 60:
                break
            m_relax = min(m_relax * 1.01, 3.0)
            continue

        # Line search: coarse then fine bisection
        best_alpha = 0.0
        best_val = M

        # Coarse grid
        alphas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.8, 1.0
            elif best_alpha <= 0.02:
                lo, hi = 0.0, 0.08
            else:
                lo = max(0.0, best_alpha - 0.12)
                hi = min(1.0, best_alpha + 0.12)

            for _ in range(30):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.14f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 5.0)

        # Diversify when stuck
        if no_improve > 50:
            scale = 0.002 + 0.003 * min((no_improve - 50) // 50, 5)
            rng = np.random.default_rng(it * 7919 + 13 + no_improve)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def alternating_dual_primal(f0, time_limit, top_k=50, t_outer=0.005, verbose=False):
    """Alternating method:
    - Primal: solve QP min a'H(a) a s.t. simplex (Hakel Hessian from w)
    - Dual: update w = softmax(autocorr(a)) with temperature
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0))
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t_start = time.monotonic()

    # Initialize dual weights: top-k lags of initial autocorrelation
    conv = fast_convolve(f)
    k_actual = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(conv, -k_actual)[-k_actual:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k_actual

    outer_it = 0
    no_improve = 0
    t_current = t_outer

    while time.monotonic() - t_start < time_limit:
        outer_it += 1
        elapsed = time.monotonic() - t_start
        if time_limit - elapsed < 0.5:
            break

        # Primal step: solve QP min a^T H(w) a s.t. sum(a)=1, a>=0
        # H[i,j] = w[i+j] (Hankel matrix)
        a_new = solve_hankel_qp(w, n)
        if a_new is not None:
            f = a_new
        else:
            # Fallback: use current f
            pass

        # Evaluate
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and outer_it % 20 == 0:
                print(f"    ADP iter {outer_it} ({elapsed:.0f}s): obj={best_obj:.14f}")
        else:
            no_improve += 1

        # Dual step: update w
        # Use softmax of autocorrelation with cooling temperature
        t_eff = max(t_current, best_obj * 0.01)
        scaled = conv / t_eff
        scaled -= scaled.max()
        w_new = np.exp(scaled)
        w_new /= w_new.sum()

        # Keep only top-k to maintain sparsity
        k_cur = min(top_k, len(w_new))
        top_idx = np.argpartition(w_new, -k_cur)[-k_cur:]
        w_masked = np.zeros_like(w_new)
        w_masked[top_idx] = w_new[top_idx]
        w = w_masked / w_masked.sum()

        # Cool down
        t_current *= 0.995

        # Diversify
        if no_improve > 30:
            rng = np.random.default_rng(outer_it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
            t_current = max(t_outer * 0.5, t_current)

    return best_f, best_obj


def solve_hankel_qp(w, n, max_iters=30):
    """Solve: min a^T H a s.t. sum(a)=1, a>=0 where H[i,j]=w[i+j].
    
    Uses active set method with regularization for numerical stability.
    """
    active = set(range(n))
    w_padded = np.zeros(2 * n - 1)
    w_padded[:len(w)] = w

    for _ in range(max_iters):
        idx = sorted(active)
        m = len(idx)
        if m == 0:
            return None
        if m == 1:
            return np.eye(1) / 1  # Just [1.0]

        idx_arr = np.array(idx, dtype=np.int64)
        # Build H_sub using broadcasting
        H_sub = w_padded[idx_arr[:, None] + idx_arr[None, :]]

        # Add small regularization for numerical stability
        H_sub += 1e-8 * np.eye(m)

        # Solve the KKT system:
        # [2*H_sub  1] [a]   [0]
        # [1'       0] [lam] = [1]
        A = np.zeros((m + 1, m + 1))
        A[:m, :m] = 2.0 * H_sub
        A[:m, m] = 1.0
        A[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0

        try:
            sol = np.linalg.solve(A, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A, rhs, rcond=None)[0]

        a_sub = sol[:m]
        lam = sol[m]

        if np.min(a_sub) >= -1e-9:
            # All active variables non-negative: check KKT for inactive
            a_full = np.zeros(n)
            a_full[idx] = np.maximum(a_sub, 0.0)
            ss = a_full.sum()
            if ss < 1e-15:
                return None
            a_full /= ss

            # Check gradient condition for inactive indices
            # grad_j = 2 * sum_m w[j+m] * a[m]
            # Need grad_j >= lam for all inactive j
            violated = False
            for j in range(n):
                if j not in active:
                    # grad_j = 2 * dot(w_padded[j:j+n], a_full)
                    grad_j = 2.0 * np.dot(w_padded[j:j + n], a_full)
                    if grad_j < lam - 1e-7:
                        active.add(j)
                        violated = True
                        break
            if not violated:
                return a_full
        else:
            # Remove most negative variable
            min_pos = np.argmin(a_sub)
            active.remove(idx[min_pos])

    # Fallback: return best effort
    idx = sorted(active)
    m = len(idx)
    if m == 0:
        return None
    idx_arr = np.array(idx, dtype=np.int64)
    H_sub = w_padded[idx_arr[:, None] + idx_arr[None, :]]
    H_sub += 1e-8 * np.eye(m)
    A = np.zeros((m + 1, m + 1))
    A[:m, :m] = 2.0 * H_sub
    A[:m, m] = 1.0
    A[m, :m] = 1.0
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
    a_full = np.zeros(n)
    a_full[idx] = np.maximum(sol[:m], 0.0)
    ss = a_full.sum()
    if ss > 1e-15:
        a_full /= ss
    return a_full


def coordinate_descent_fast(f0, time_limit, verbose=False):
    """Fast coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0))
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t_start = time.monotonic()

    while time.monotonic() - t_start < time_limit:
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(0, n, max(1, n // 500)):
            if time.monotonic() - t_start > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = cur_max

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def squeezed_lp_fw(f0, time_limit, top_k=15, verbose=False):
    """Very tight LP-FW for final polishing with minimal constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0))

        f = f / s

            break

            g = solve_lp_reduced(f, M * 1.005, active_idx, n, time_limit=15)
        if g is None:
            g = solve_lp_reduced(f, M * 1.01, active_idx, n, time_limit=15)
        if g is None:
            # Try coordinate descent as fallback
            rem = time_limit - (time.monotonic() - t_start)
            if rem > 2:
                f_cd, obj_cd = coordinate_descent_fast(f, min(rem * 0.3, 10))
                if obj_cd < best_obj - 1e-15:
                    best_obj = obj_cd
                    best_f = f_cd.copy()
                    f = f_cd
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            rem = time_limit - (time.monotonic() - t_start)
            if rem > 2:
                f_cd, obj_cd = coordinate_descent_fast(f, min(rem * 0.3, 8))
                if obj_cd < best_obj - 1e-15:
                    best_obj = obj_cd
                    best_f = f_cd.copy()
                    f = f_cd
            else:
                break
            continue

        # Small step with line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-14:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-16:
                best_obj = new_obj
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    Squeeze iter {it}: obj={best_obj:.14f}")

    return best_f, best_obj

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic() - 120
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.3)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_reduced(f_interp, time_limit=t_alloc, top_k=40)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.8f}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD polish (best={best_score:.8f}) ===")
    n_cur = len(best_seq)
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_cd = min(rem * 0.5, 120)
        print(f"  Budget: {t_cd:.0f}s")
        a_opt, _ = coordinate_descent_fast(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")

    # === Phase 6: Squeeze (very tight LP-FW) ===
    print(f"\n=== Phase 6: Squeeze (best={best_score:.8f}) ===")
    n_cur = len(best_seq)
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_sq = min(rem * 0.6, 100)
        print(f"  Budget: {t_sq:.0f}s")
        a_opt, _ = squeezed_lp_fw(best_seq, time_limit=t_sq, top_k=12, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.8f}")

    # === Phase 7: Micro-perturbations ===
    print(f"\n=== Phase 7: Micro-perturbations (best={best_score:.8f}) ===")
    n_cur = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scales = [0.0002, 0.0004, 0.0006, 0.001, 0.0015, 0.002,
                  0.003, 0.005, 0.008, 0.01, 0.015, 0.02]
        scale = scales[min(trial, len(scales) - 1)]
        noise = rng.standard_normal(n_cur) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.8f}")
            # Quick squeeze on the new best
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = squeezed_lp_fw(f_pert, time_limit=min(rem2 * 0.3, 15), top_k=10)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sq]: {sc2:.8f}")

    # === Final squeeze ===
    rem = deadline - time.monotonic()
    if rem > 5:
        a_opt, _ = squeezed_lp_fw(best_seq, time_limit=min(rem * 0.8, 30), top_k=10)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL SQUEEZE: {sc:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
