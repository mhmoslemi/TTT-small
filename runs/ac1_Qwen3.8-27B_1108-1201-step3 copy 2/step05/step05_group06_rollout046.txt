"""
Optimized step function search using LP-Frank-Wolfe, mirror descent,
and Nesterov-accelerated subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def conv_subgradient(f, k_star, n):
    """Compute subgradient of conv(f,f) at the maximizing index k_star."""
    g = np.zeros(n)
    for m in range(n):
        km = k_star - m
        if 0 <= km < n:
            if km == m:
                g[m] = 2.0 * f[m]
            else:
                g[m] = f[km]
    return g


def mirror_descent_step(f, g, eta):
    """Exponential family (mirror) descent step on the simplex."""
    log_ratio = -eta * g
    # Numerical stability
    log_ratio -= log_ratio.max()
    weights = f * np.exp(log_ratio)
    s = weights.sum()
    if s < 1e-300:
        return f.copy()
    return weights / s


def nesterov_subgrad(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    y = f.copy()  # Nesterov's y
    t_step = 1.0
    t0 = time.monotonic()
    
    # Subgradient step size: eta_t = c / sqrt(t)
    # For simplicity, use a fixed eta and adjust
    eta_base = 0.1
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_star = int(np.argmax(conv_y))
        g = conv_subgradient(y, k_star, n)
        
        # Project gradient onto tangent space of simplex
        g_proj = g - g.mean()
        
        # Step size
        eta = eta_base / np.sqrt(it)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t_step**2)) / 2
        beta = (t_step - 1) / t_new
        y_new = f + beta * (f - y)
        y_new = y_new - eta * g_proj
        f_new = project_simplex(y_new)
        
        # Check improvement
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        y = y_new if no_improve == 0 else f  # Reset momentum if no improvement
        t_step = t_new
        
        if no_improve > 5000:
            break
    
    return best_f, best_obj

def smooth_optimize(f0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Log-sum-exp smooth optimization with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    T = T_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = max(T_start * (T_end / T_start) ** (elapsed / time_limit), T_end)
        lr = alpha * (0.5 ** (elapsed / (time_limit * 0.7)))
        
        # Compute smooth gradient
        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()
        
        # grad_i = (2/T) * sum_j f[j] * w[i+j] (cross-correlation)
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T) * corr
        
        # Project gradient onto tangent space
        grad = grad - grad.mean()
        
        # Adam update
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Smooth iter {it}: obj={best_obj:.12f}, T={T:.6f}")
        else:
            no_improve += 1
        
        if no_improve > 2000:
            seed_val = int(elapsed * 1000) % 1000000
            rng_p = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj

def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    elif len(a) > n:
        a = a[:n]
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def make_skewed(n, skew=0.3):
    """Asymmetric cosine: peak shifted from center."""
    center = 0.5 + skew
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / (0.3 * (1 - abs(skew))))**2) + 0.001
    return a / a.sum()


def make_half_gauss(n, offset=0.3):
    """Half Gaussian: peak at offset fraction of the length."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - offset) / 0.25)**2) + 0.001
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
