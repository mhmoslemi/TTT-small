import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_weighted(w, a):
    """Compute sum_m w[m+i]*a[m] for i=0,...,n-1 using np.correlate."""
    n = len(a)
    # w has length 2n-1, a has length n
    # result[i] = sum_{m=0}^{n-1} w[m+i] * a[m]
    return np.correlate(w, a, mode='valid')


def soft_max_obj_grad_direct(a, T):
    """
    Compute soft-max smoothed objective and gradient w.r.t. a (unconstrained).
    Assumes sum(a) = 1.
    Obj = 2n * soft_max(conv(a,a), T)
    Grad[i] = 4n * cross_corr(w, a)[i] where w = soft_max_weights(conv(a,a), T)
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad = 4.0 * n * cross_corr_weighted(w, a)
    return obj, grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad_z(z, T):
    """
    Compute soft-max objective and gradient w.r.t. z where a = softmax(z).
    """
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # grad_a = 4n * cross_corr(w, a)
    grad_a = 4.0 * n * cross_corr_weighted(w, a)
    # Chain rule: grad_z_j = a_j * (grad_a_j - a_j * sum_i grad_a_i * a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def projected_subgradient(a0, time_limit, lr_init=0.002, decay=0.9995,

    cur_temp = temperature
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        ws = w_peaks.sum()

        else:
            w_peaks /= ws

        w_full = np.zeros(2 * n - 1)

                w_full[int(peaks[i])] += w_peaks[i]
        
        # Gradient: 4n * cross_corr(w_full, a)
        sg = 4.0 * n * cross_corr_weighted(w_full, a)

        a_new = project_simplex(a + v)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        co = b_new.max()
        
        if co < best_obj - 1e-15:

                v = np.zeros(n)

def coordinate_descent(a0, time_limit, n_multipliers=5):
    """Coordinate descent cycling through multiplicative updates."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 1.7, 2.0])
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        # Randomize the order each pass
        order = np.random.default_rng(passes).permutation(n)
        
        for idx in range(n):
            j = order[idx]
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def peak_shave_refine(a0, time_limit, n_peaks=10):
    """
    Peak shaving: identify the top peaks and try to reduce them
    by adjusting contributing elements.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)

        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        k_actual = min(n_peaks, len(b_inner))
        peak_indices = np.argpartition(b_inner, -k_actual)[-k_actual:]
        peak_indices = sorted([int(k) for k in peak_indices if b[k] > best_obj * 0.9])
        
        if not peak_indices:
            break
        
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:

            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:

            contributions = a[idx] * a[k - idx]
            
            # Try reducing the largest contributor

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def raised_cosine(n):
    """Create a raised-cosine shaped sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gaussian_bump(n, sigma_frac=0.3):
    """Create a Gaussian-shaped sequence."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * sigma_frac**2))
    return a / a.sum()


def triangle_seq(n):
    """Create a triangular (tent) sequence."""
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            try_seq(a_curr, f"deep_T{T}")
    
    # === Phase 3: Try larger lengths ===
    print(f"\n=== Phase 3: Larger length exploration (best={best_score:.8f}) ===")
    
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1536, 2048, 3072]:
            rem = deadline - time.monotonic() - 400
            if rem < 60:
                break
            
            f_interp = interpolate_sequence(base, target_n)
            z0 = z_from_a(f_interp)
            
            t_alloc = min(rem * 0.35, 120.0)
            print(f"  n={target_n}, L-BFGS T=100, t={t_alloc:.0f}s")
            z_curr, _ = lbgfs_optimize_z(z0, T=100, time_limit=t_alloc, maxiter=3000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"n{target_n}_T100")
            
            if best_score < 1.504:
                rem = deadline - time.monotonic() - 300
                if rem > 30:
                    t_alloc2 = min(rem * 0.3, 80.0)
                    print(f"  n={target_n}, L-BFGS T=500, t={t_alloc2:.0f}s")
                    z_curr, _ = lbgfs_optimize_z(z_curr, T=500, time_limit=t_alloc2, maxiter=3000)
                    a_curr = softmax_from_z(z_curr)
                    try_seq(a_curr, f"n{target_n}_T500")
    
    # === Phase 4: Projected subgradient deep refinement ===
    print(f"\n=== Phase 4: Deep subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            n_best = len(best_seq)
            t_sg = min(rem * 0.5, 120.0)
            a_sg, _ = projected_subgradient(best_seq, time_limit=t_sg,
                                            lr_init=0.001, decay=0.9999,
                                            top_k=max(20, n_best // 10),
                                            temperature=0.005, seed=123)
            try_seq(a_sg, "deep_sg")
            
            # Second round with different params
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_sg2 = min(rem * 0.5, 80.0)
                a_sg2, _ = projected_subgradient(best_seq, time_limit=t_sg2,
                                                 lr_init=0.0005, decay=0.99995,
                                                 top_k=max(5, n_best // 50),
                                                 temperature=0.001, seed=456)
                try_seq(a_sg2, "deep_sg_fine")
    
    # === Phase 5: L-BFGS at very high T (near-exact max) ===
    print(f"\n=== Phase 5: High-T L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        z0 = z_from_a(best_seq)
        for T in [2000, 5000]:
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            z_opt, _ = lbgfs_optimize_z(z0, T, time_limit=t_alloc, maxiter=5000)
            a_opt = softmax_from_z(z_opt)
            try_seq(a_opt, f"highT_{T}")
            if isinstance(best_seq, np.ndarray):
                z0 = z_from_a(best_seq)
    
    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.6, 90.0)
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "coord_desc")
    
    # === Phase 7: Peak shaving ===
    print(f"\n=== Phase 7: Peak shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_ps = min(rem * 0.6, 60.0)
            a_ps, _ = peak_shave_refine(best_seq, time_limit=t_ps, n_peaks=15)
            try_seq(a_ps, "peak_shave")
    
    # === Phase 8: Final micro-optimization ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Perturb + re-optimize
        for attempt in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            noise_scale = 0.001 * (attempt + 1)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_seq + noise)
            z_pert = z_from_a(a_pert)
            t_alloc = min(rem * 0.3, 20.0)
            z_p, _ = lbgfs_optimize_z(z_pert, 1000, time_limit=t_alloc, maxiter=2000)
            a_p = softmax_from_z(z_p)
            try_seq(a_p, f"perturb_{attempt}")
            
            # Also try short coordinate descent on the perturbed result
            rem = deadline - time.monotonic() - 25
            if rem > 8 and isinstance(best_seq, np.ndarray):
                a_cd, _ = coordinate_descent(best_seq, time_limit=min(rem * 0.5, 15.0))
                try_seq(a_cd, f"cd_fin_{attempt}")
    
    # Final L-BFGS with max T
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            n_best = len(best_seq)
            z0 = z_from_a(best_seq)
            z_final, _ = lbgfs_optimize_z(z0, 10000, time_limit=min(rem * 0.7, 20.0), maxiter=3000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_maxT")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
