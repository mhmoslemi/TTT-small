1. Initialize a (e.g., from the previous best).
2. For t = 1, 2, ...:
   a. Compute c = convolve(a, a).

def fast_convolve(a):

    if s < 1e-15:

    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, active_set):
    """
    Compute the subgradient of max_k c[k] with respect to a.
    
    For lag k:
    - If k <= n-1: g_k[j] = 2*a[k-j] for j in [0, k], else 0
    - If k >= n: g_k[j] = 2*a[k-j] for j in [k-n+1, n-1], else 0
    
    The subgradient of max is the average of gradients of active terms.
    """
    n = len(a)
    g = np.zeros(n)
    k_active = len(active_set)
    if k_active == 0:
        return g
    
    for k in active_set:
        if k <= n - 1:
            # g_k[j] = 2*a[k-j] for j in [0, k]
            js = np.arange(0, k + 1)
            g[js] += 2.0 * a[k - js]
        else:
            # g_k[j] = 2*a[k-j] for j in [k-n+1, n-1]
            js = np.arange(k - n + 1, n)
            g[js] += 2.0 * a[k - js]
    
    g /= k_active
    return g


def subgradient_method(a, time_limit, verbose=False):
    """
    Subgradient descent with Nesterov averaging for minimizing max auto-conv.
    Assumes a is normalized to sum=1.
    """
    n = len(a)
    a = normalize(a.copy())
    conv = fast_convolve(a)
    best_score = compute_score(a)
    best_a = a.copy()
    
    # Running average (Nesterov-Yudin)
    avg_a = np.zeros(n)
    count = 0
    
    # Step size parameters
    c0 = 1.0  # Initial step size scale
    t0 = time.monotonic()
    step = 0
    patience = 0
    patience_limit = 5000
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        count += 1
        
        # Compute auto-conv
        conv = fast_convolve(a)
        max_c = conv.max()
        
        # Find active set (lags within epsilon of max)
        # Use adaptive epsilon
        eps = max(1e-8, 0.01 * max_c)
        active_set = np.where(conv >= max_c - eps)[0]
        
        # Compute subgradient
        g = compute_subgradient(a, conv, active_set)
        
        # Step size: c / sqrt(step)
        alpha = c0 / np.sqrt(step)
        
        # Update
        a_new = a - alpha * g
        a_new = project_simplex(a_new)
        
        # Update average
        avg_a += a_new
        avg_candidate = avg_a / count
        
        # Evaluate
        sc = compute_score(a_new)
        sc_avg = compute_score(avg_candidate)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            patience = 0
        elif sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_candidate.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        # Restart if stuck
        if patience > patience_limit:
            patience = 0
            a = project_simplex(best_a + np.random.standard_normal(n) * 0.01)
            avg_a = a.copy()
            count = 1
            c0 = c0 * 0.5  # Reduce step size on restart
            if verbose:
                print(f"      SG restart at step {step}: score={best_score:.10f}, c0={c0:.4f}")
        
        if verbose and step % 2000 == 0:
            print(f"      SG step {step}: score={best_score:.10f}, alpha={alpha:.6f}, |A|={len(active_set)}")
    
    return best_a, best_score


def coord_descent_precise(a, time_limit, verbose=False):
    """Precise coordinate descent with golden-section search per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        if pass_count > 1:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            
            def obj_v(v):
                if v < -1e-10 or v > 1.0:
                    return float('inf')
                v = max(0.0, v)
                new_a = a.copy()
                new_a[j] = v
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                other_sum = 1.0 - old_val
                if other_sum < 1e-12:
                    return float('inf')
                scale = (1.0 - v) / other_sum
                new_a[mask] *= scale
                conv = fast_convolve(new_a)
                return conv.max()

            lo = max(0.0, old_val * 0.5)
            hi = min(0.999, old_val * 2.0 + 0.01)
            
            try:
                res = minimize_scalar(obj_v, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 50})
                if res.success and res.fun < obj_v(old_val) - 1e-15:
                    v_new = res.x
                    new_a = a.copy()
                    new_a[j] = v_new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = 1.0 - old_val
                    if other_sum > 1e-12:
                        scale = (1.0 - v_new) / other_sum
                        new_a[mask] *= scale
                    new_a = normalize(new_a)
                    sc = compute_score(new_a)
                    if sc < best_score - 1e-15:
                        a = new_a
                        best_score = sc
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: score={best_score:.10f}")

        if not improved and pass_count > 3:
            break
        if pass_count > 300:
            break

    return best_a, best_score


def block_descent(a, block_size, time_limit):
    """Block coordinate descent: optimize a block of consecutive variables."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Pick a random block
        start = rng.integers(0, n - block_size + 1)
        end = start + block_size
        
        # Try perturbations of this block
        best_block_score = best_score
        best_block = a[start:end].copy()
        
        for trial in range(20):
            # Random perturbation of the block
            noise = rng.standard_normal(block_size) * 0.01
            new_block = np.maximum(best_block + noise, 0)
            new_a = a.copy()
            new_a[start:end] = new_block
            new_a = normalize(new_a)
            sc = compute_score(new_a)
            if sc < best_block_score:
                best_block_score = sc
                best_block = new_block.copy()
        
        if best_block_score < best_score:
            a[start:end] = best_block
            a = normalize(a)
            best_score = compute_score(a)
            best_a = a.copy()
    
    return best_a, best_score


def geometric_sequence(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def cosine_sequence(n):
    a = np.sin(np.linspace(0, np.pi, n)) ** 2
    return normalize(a)


def gaussian_sequence(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return normalize(a)


def triangular_sequence(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)
