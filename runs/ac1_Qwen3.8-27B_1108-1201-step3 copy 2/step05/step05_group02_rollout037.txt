Let me focus on practical improvements to the algorithm:

1. **Faster subgradient/mirror descent** using FFT
2. **Adaptive K** (number of peaks to consider)
3. **Better eta schedule**
4. **Multiple starting points**
5. **Try different n values**

Let me also consider using **ADMM** or **proximal methods**:
- Split the problem into: minimize max_k (f*f)[k] + indicator(f >= 0, sum(f) = 1)
- The proximal operator of the simplex constraint is easy (projection)
- The subgradient of max_k (f*f)[k] is easy to compute

But I think mirror descent will be the most effective.

Let me also think about a **two-scale approach**:
1. Coarse: Use a shorter sequence (n/4 or n/8) to find a good shape
2. Fine: Upsample and refine

This could help escape local minima.

"""
Hybrid optimization: Nesterov-accelerated mirror descent + LP-FW refinement
+ coordinate descent polish + multi-length search.
Key innovation: FFT-based mirror descent with Nesterov momentum for fast
many-iteration optimization, followed by expensive but precise LP-FW steps.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Linear convolution of two arrays via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent(f0, time_limit, eta0=0.05, eta_decay=0.9995, K=30, 
                   nesterov=True, verbose=False):
    """Nesterov-accelerated mirror descent for minimizing max autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    cur_obj = best_obj
    
    t0 = time.monotonic()
    eta = eta0
    f_prev = f.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        if nesterov and it > 1:
            momentum = (it - 1) / (it + 1)
            y = f + momentum * (f - f_prev)
            y = np.maximum(y, 1e-30)
            y_sum = y.sum()
            y = y / y_sum
        else:
            y = f
            f_prev = f.copy()
        
        # Compute convolution at y
        conv_y = fast_convolve(y)
        cur_val = conv_y.max()
        
        # Find top-K peaks with softmax weighting
        K_actual = min(K, len(conv_y))
        peak_idx = np.argpartition(conv_y, -K_actual)[-K_actual:]
        peak_vals = conv_y[peak_idx]
        
        # Softmax weights (temperature relative to current max)
        temp = 0.01 * cur_val
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum
        
        # Build peak kernel
        w_kernel = np.zeros(len(conv_y))
        w_kernel[peak_idx] = w
        
        # Compute subgradient via cross-correlation
        # sg[z] = 2 * sum_k w_kernel[k] * y[k-z]
        # = 2 * (w_kernel * y_rev)[z + n - 1]
        y_rev = y[::-1]
        cross = fast_convolve2(w_kernel, y_rev)
        sg = 2.0 * cross[n-1 : 2*n-1]
        
        # Mirror descent update (multiplicative)
        log_update = -eta * sg
        log_update -= log_update.max()  # numerical stability
        f_new = y * np.exp(log_update)
        f_new_sum = f_new.sum()
        f_new = f_new / f_new_sum
        
        f = f_new
        eta *= eta_decay
        
        # Track best
        f_conv = fast_convolve(f)
        f_max = f_conv.max()
        if f_max < best_obj - 1e-15:
            best_obj = f_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            f_prev = f.copy()
            eta = eta0 * 0.5
            no_improve = 0
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it} ({elapsed:.0f}s): best={best_obj:.10f}")
    
    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.1
            if m_relax > 10.0:
                m_relax = 1.0
                continue
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.03, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new
            
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 104729 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
            best_j_val = old_val
            best_j_obj = cur_conv.max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

We have f of length n. The linear convolution f*f has length 2n-1.
(f*f)[k] = sum_{i=max(0,k-n+1)}^{min(k, n-1)} f[i] * f[k-i]

The derivative of (f*f)[k] with respect to f[z] (for 0 <= z < n):
= sum_{i} [delta(i,z) * f[k-i] + f[i] * delta(k-i, z)]
= f[k-z] (if 0 <= k-z < n) + f[k-z] (if 0 <= k-z < n, since i = k-z must be in [0, n-1])

Wait, that's the same condition. So the derivative is 2*f[k-z] when 0 <= k-z < n, else 0.

But wait, when z = k/2 (i.e., k is even and z = k/2), the term f[z]*f[k-z] = f[z]^2 appears only once in the sum, and its derivative is 2*f[z]. My formula gives 2*f[k-z] = 2*f[z], which is correct.

So sg[z] = 2 * sum_{k in peaks} w_k * f[k-z] (when 0 <= k-z < n)

Now, to compute this via convolution:
Define w_kernel of length 2n-1 where w_kernel[k] = w_k for k in peaks, 0 otherwise.
Define y_rev = y[::-1] which has length n. y_rev[j] = y[n-1-j].

Linear convolution: (w_kernel * y_rev)[m] = sum_{j=0}^{n-1} w_kernel[m-j] * y_rev[j]
= sum_{j=0}^{n-1} w_kernel[m-j] * y[n-1-j]

Let m = z + n - 1:
(w_kernel * y_rev)[z + n - 1] = sum_{j=0}^{n-1} w_kernel[z + n - 1 - j] * y[n-1-j]

Let k = z + n - 1 - j, so j = z + n - 1 - k:
= sum_{k} w_kernel[k] * y[n-1-(z+n-1-k)]
= sum_{k} w_kernel[k] * y[k-z]

And the valid range of j is [0, n-1], which means k = z+n-1-j ranges from z+n-1-(n-1) = z to z+n-1-0 = z+n-1.
So k ranges from z to z+n-1, and we also need 0 <= k < 2n-1 (which is automatically satisfied for z in [0, n-1]).

Also we need 0 <= k-z < n, i.e., 0 <= k-z <= n-1, i.e., z <= k <= z+n-1.
This is exactly the range we have! So:

sg[z] = 2 * (w_kernel * y_rev)[z + n - 1] for z = 0, ..., n-1

Great, so my formula is correct.

"""
Hybrid optimization: Nesterov-accelerated mirror descent (FFT-based) as the
main workhorse, combined with LP-FW refinement, coordinate descent polish,
multi-length exploration, and pattern-injection restarts.
Key innovation: fast subgradient computation via cross-correlation in FFT domain
enables 100k+ optimization iterations within the time budget, far exceeding
the ~200-500 iterations achievable with LP-based methods alone.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    """Linear convolution of two arrays via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(y, peak_idx, weights, n):
    """
    Compute subgradient of max_k (y*y)[k] using cross-correlation.
    sg[z] = 2 * sum_k w_k * y[k-z] for valid k-z in [0, n-1].
    Computed via: sg[z] = 2 * (w_kernel * y_rev)[z + n - 1]
    """
    conv_len = 2 * n - 1
    w_kernel = np.zeros(conv_len)
    w_kernel[peak_idx] = weights
    y_rev = y[::-1]
    cross = fast_convolve2(w_kernel, y_rev)
    sg = 2.0 * cross[n - 1:2 * n - 1]
    return sg


def mirror_descent(f0, time_limit, eta0=1.0, eta_decay=0.9999, K=40,
                   nesterov=True, adaptive_eta=True, verbose=False):
    """
    Nesterov-accelerated mirror descent for minimizing max autocorrelation.
    Uses multiplicative (entropy) mirror map, naturally preserving non-negativity.
    """
    f = np.maximum(f0.astype(np.float64), 1e-30)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()

    t0 = time.monotonic()
    eta = eta0
    f_prev = f.copy()
    it = 0
    no_improve = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        if nesterov and it > 1:
            momentum = (it - 1) / (it + 1)
            y = f + momentum * (f - f_prev)
            y = np.maximum(y, 1e-30)
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = f.copy()
            else:
                y = y / y_sum
        else:
            y = f
            f_prev = f.copy()

        # Compute convolution and find peaks
        conv_y = fast_convolve(y)
        cur_val = conv_y.max()

        # Find top-K peaks
        K_actual = min(K, len(conv_y))
        peak_idx = np.argpartition(conv_y, -K_actual)[-K_actual:]
        peak_vals = conv_y[peak_idx]

        # Softmax weights with adaptive temperature
        temp = 0.02 * cur_val + 1e-30
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(K_actual)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient
        sg = compute_subgradient(y, peak_idx, w, n)

        # Adaptive eta: normalize step to be O(1) relative to sg scale
        if adaptive_eta:

            if sg_scale < 1e-15:
                sg_scale = 1.0

            # Cap to avoid too-large updates

            effective_eta = min(effective_eta, 50.0)
        else:
            effective_eta = eta

        log_update = -effective_eta * sg
        max_log = log_update.max()
        if max_log > 500:
            log_update -= max_log
        f_new = y * np.exp(log_update)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15 or not np.isfinite(f_new_sum):
            f_new = y.copy()
            f_new_sum = 1.0
        f_new = f_new / f_new_sum

        f = f_new
        f_prev = f_prev if not nesterov or it == 1 else f.copy()

        # Track best

            f_conv = fast_convolve(f)
            f_max = f_conv.max()
            if f_max < best_obj - 1e-15:
                best_obj = f_max
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 50
        else:
            no_improve += 1

        # Restart mechanism

            rng = np.random.default_rng(it * 31 + restart_count)
            scale = 0.001 + 0.002 * min(restart_count, 5)

            f_prev = f.copy()
            eta = eta0 * (0.5 ** min(restart_count, 3))

            print(f"    MD iter {it} ({elapsed:.0f}s): best={f_conv_check.max():.12f}")

    # Final check
    final_conv = fast_convolve(best_f)
    return best_f, final_conv.max()


def solve_lp_dual(f, M, n):
    """
    Solve the dual LP for direction finding.
    Dual: min M * sum(y) s.t. F^T y >= 1, y >= 0
    where (F^T y)[j] = sum_k y[k] * f[k-j] (valid k)
    We extract the primal solution from the dual.
    For simplicity, we solve the primal directly but with a tighter formulation.
    """
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 45, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_dual(f, M * m_relax, n)
        if g is None:
            m_relax *= 1.15
            if m_relax > 10.0:
                m_relax = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_relax = min(m_relax * 1.03, 8.0)
            continue

        # Line search: coarse
        best_alpha = 0.0
        best_val = M
        for alpha in [0.03, 0.07, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            if best_alpha >= 1.0:
                lo, hi = 0.7, 1.0
            elif best_alpha <= 0.03:
                lo, hi = 0.0, 0.1
            else:
                lo = max(0.0, best_alpha - 0.12)
                hi = min(1.0, best_alpha + 0.12)
            for _ in range(22):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M = conv_new.max()
            m_relax = 1.0
            f = f_new
            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.06, 8.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
            best_j_val = old_val
            best_j_obj = cur_max

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def pattern_init(n, pattern_type='raised_cosine', rng=None):
    """Generate structured initial patterns."""
    if rng is None:
        rng = np.random.default_rng(0)

    if pattern_type == 'uniform':
        f = np.ones(n)
    elif pattern_type == 'raised_cosine':
        x = np.linspace(0, 1, n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif pattern_type == 'gaussian':
        x = np.linspace(-3, 3, n)
        f = np.exp(-0.5 * x ** 2)
    elif pattern_type == 'sinc_envelope':
        x = np.linspace(-1, 1, n)
        f = np.sinc(0.8 * x) ** 2
        f = np.maximum(f, 0)
    elif pattern_type == 'tapered':
        x = np.linspace(0, 1, n)
        f = x ** 0.5 * (1 - x) ** 0.5
    elif pattern_type == 'dual_peak':
        x = np.linspace(0, 1, n)
        f = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif pattern_type == 'bump':
        x = np.linspace(0, 1, n)
        f = np.exp(-5 * (x - 0.5) ** 2)
    else:
        f = np.ones(n)

    f = np.maximum(f, 1e-10)
    f = f + rng.standard_normal(n) * 0.01
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    return f

            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: LP-FW refinement ===
    print(f"\n=== Phase 2: LP-FW Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(350, rem * 0.25)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: Second MD with different params ===
    print(f"\n=== Phase 3: MD Round 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(300, rem * 0.22)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s, K=60, eta0=0.5")
        a_opt, _ = mirror_descent(best_seq, time_limit=t3,
                                   eta0=0.5, eta_decay=0.99992,
                                   K=60, nesterov=True,
                                   adaptive_eta=True, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 4: Multi-length search ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [64, -64, 128, -128, 192, -192, 256]:
        nl = n_best + delta
        if 128 <= nl <= 2500 and nl != n_best:
            candidate_lengths.append(nl)

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 120
        if rem < 25:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_md = min(60, rem * 0.25)
        if t_md < 10:
            continue
        print(f"  n={target_n}, MD t={t_md:.0f}s")
        a_opt, _ = mirror_descent(f_interp, time_limit=t_md,
                                   eta0=0.8, eta_decay=0.9999, K=35)
        # Short LP-FW
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 15:
            t_lp = min(30, rem2 * 0.4)
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_lp)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")

    # === Phase 5: Pattern-injection diverse restarts ===
    print(f"\n=== Phase 5: Pattern restarts (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    patterns = ['raised_cosine', 'tapered', 'bump', 'sinc_envelope']
    for pat in patterns:
        rem = deadline - time.monotonic() - 100
        if rem < 20:
            break
        f_pat = pattern_init(n_best, pat, rng)
        # Blend with best
        blend = 0.7 * best_seq + 0.3 * f_pat
        blend = project_simplex(blend)
        t_alloc = min(100, rem * 0.2)
        if t_alloc < 10:
            continue
        print(f"  pattern={pat}: t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(blend, time_limit=t_alloc,
                                   eta0=0.8, eta_decay=0.9999, K=35)
        rem2 = deadline - time.monotonic() - 80
        if rem2 > 10:
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=min(rem2 * 0.4, 25))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{pat}]: {sc:.6f}")

    # === Phase 6: Noisy restart + MD ===
    print(f"\n=== Phase 6: Noisy restarts (best={best_score:.6f}) ===")
    for noise_scale in [0.005, 0.01, 0.02]:
        rem = deadline - time.monotonic() - 60
        if rem < 15:
            break
        f_noisy = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
        t_alloc = min(80, rem * 0.25)
        if t_alloc < 10:
            continue
        a_opt, _ = mirror_descent(f_noisy, time_limit=t_alloc,
                                   eta0=0.5, eta_decay=0.9999, K=30)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [noise={noise_scale}]: {sc:.6f}")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 40
    if rem > 30:
        t_cd = min(rem * 0.5, 120)
        print(f"  Budget: {t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 8: Final LP-FW squeeze ===
    print(f"\n=== Phase 8: Final LP-FW (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_f = min(rem, 60)
        print(f"  Budget: {t_f:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_f)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 9: Final micro MD polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_opt, _ = mirror_descent(best_seq, time_limit=min(rem * 0.8, 20),
                                   eta0=0.1, eta_decay=0.99999, K=80)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL MD: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
