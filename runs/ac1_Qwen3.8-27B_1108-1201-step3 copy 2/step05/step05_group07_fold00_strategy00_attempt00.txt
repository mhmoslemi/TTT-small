    a = softmax_from_z(z)
    n = len(a)
    b = fast_Convolve(a)  # full convolution of length 2n-1
    max_b = b.max() if len(b) > 0 else 0
    obj = (2.0 * n / T) * log_sum_exp(T * b)
    # Gradient calculation
    w = np.zeros_like(a)
    max_index = np.argmax(b)
    w[max_index] = b[max_index] - b.max() + np.log(len(b))  # relatively incorrect weighting
    # But we do it properly
    weights = b - b.min() if len(b) > 0 else np.ones(len(b))
    weights = weights / weights.sum()
    grad = cross_corr(weights, a)
    grad = grad / grad.sum() if np.any(grad) else grad
    return obj, grad, max_b

def simplex_proj(v):
    """Simplex projection: Euclidean onto simplex."""
    n = len(v)
    if n == 1:
        return [max(v[0], 0.0)]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n)/n
    idx = np.where(cond)[0][-1] - 1
    theta = (cssv[idx+1] - 1.0) / (idx+1.0)
    return np.maximum(v - theta, 0.0) / np.maximum(v - theta, 0.0).sum()

def optimize_z(z0, T, time_limit, simplex_proj_func=simplex_proj, n_steps=1000):

        if np.isfinite(res.fun):
