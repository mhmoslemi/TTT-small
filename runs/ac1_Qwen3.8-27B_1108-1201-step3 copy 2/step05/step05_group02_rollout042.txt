```python
def peak_redistribute(a, k_star, alpha):
    """Redistribute mass to reduce the peak at lag k_star."""
    n = len(a)
    # Contribution of each position j to the peak
    # Peak = sum_j a[j] * a[j + k_star]
    # The "responsibility" of position j is a[j] * a[j + k_star]
    
    k = k_star
    # For lag k, the pairs are (j, j-k) for j >= k (or (j-k, j))
    # Actually, (a*a)_k = sum_j a[j] * a[j+k] where j ranges over valid indices
    
    if k == 0:
        # Peak is sum a_j^2, responsibility is a_j^2
        resp = a ** 2
    else:
        resp = np.zeros(n)
        for j in range(n - k):
            resp[j] += a[j] * a[j + k]
            resp[j + k] += a[j] * a[j + k]
    
    # Reduce high-responsibility positions, increase low-responsibility
    # Use a softmax-like redistribution
    max_resp = resp.max()
    if max_resp < 1e-15:
        return a
    
    # Scale factor: reduce resp proportionally
    reduction = alpha * (resp / max_resp)
    # New values
    a_new = a * (1 - reduction)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s
    return a_new
```

```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

Let me now think about what specific improvements to make:

1. **Adaptive constraint selection**: Instead of using all 2n-1 constraints in the LP, use only the top K (e.g., K = n). This can speed up the LP significantly.

2. **Warm-start the LP**: Use the previous solution as a warm start. (HIHOS doesn't directly support warm-starting, but we can try.)

3. **Symmetric optimization**: Enforce `a_i = a_{n-1-i}` to halve the problem.

4. **Structured initializations**: 
   - Hann window: `a_i = sin^2(pi * i / (n-1))` normalized
   - Raised cosine
   - Bimodal: two peaks at 1/4 and 3/4
   - Random with decay

5. **Peak redistribution heuristic**:
   - Find the peak lag k*.
   - Compute the "responsibility" of each position.
   - Shift mass from high-responsibility to low-responsibility positions.
   - This is a cheap local improvement that can be done many times.

6. **Interleaved approach**: Alternate between LP-FW steps and peak redistribution steps.

Actually, one more thought: the objective function is `2*n*max_conv/sum^2`. If I normalize `sum(a) = 1`, it becomes `2*n*max_conv`. So I want to minimize `n * max_conv`. For fixed n, this is just minimizing `max_conv`.

But the "optimal" n depends on the tradeoff. For larger n, we have more degrees of freedom to spread out the convolution, but the factor `n` in the objective works against us.

Actually wait, with `sum(a) = 1`, `max_conv >= 1/(2n-1)` (since the convolution has 2n-1 entries summing to 1). So the objective is at least `2n/(2n-1) ≈ 1` for large n. But this lower bound is not achievable.

For a uniform sequence, `max_conv = n/n^2 = 1/n`, giving objective `2n * 1/n = 2`.

The previous best is 1.505, which means `max_conv ≈ 1.505/(2*768) ≈ 0.000981`. Compare with uniform: `1/768 ≈ 0.001302`. So the optimal is about 75% of the uniform max.

For larger n, say n=2048, if we could achieve the same ratio, we'd get `max_conv ≈ 0.000981 * (768/2048) ≈ 0.000368`, and objective `2*2048*0.000368 ≈ 1.505`. So the objective would be about the same!

This suggests that the objective might be roughly invariant to n (for large n), and the key is finding the right "shape" of the sequence. The length mainly affects the granularity.

This is an important insight: we should focus on finding the right shape, and the length is less important (as long as it's large enough for good resolution).

Given this, my strategy should be:
1. Find a good "shape" at moderate resolution (n=256 or 512).
2. Upscale to the target resolution (n=768 or higher).
3. Refine at the target resolution.

For step 1, I can use faster methods since n is smaller.
For step 3, I need to be careful about computational cost.

```python
"""
Multi-phase optimizer for min 2*n*max(auto_conv)/(sum a)^2.
Key ideas: adaptive-constraint LP-Frank-Wolfe, peak redistribution,
multi-resolution search, window-function initializations, coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, k_vals):
    """Build LP constraint matrix for selected lags."""
    n = len(f)
    A = np.zeros((len(k_vals), n))
    for idx, k in enumerate(k_vals):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            cols = np.arange(i_start, i_end + 1)
            A[idx, cols] = f[k - cols]
    return A

def solve_lp_direction(f, M, n, n_constraints=None, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    if n_constraints is not None and n_constraints < 2 * n - 1:
        conv_f = fast_convolve(f)
        k_vals = np.argsort(conv_f)[::-1][:n_constraints]
    else:
        k_vals = np.arange(2 * n - 1)
    
    A_ub = build_lp_matrix(f, k_vals)
    b_ub = np.full(len(k_vals), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None and res.x.sum() > 1e-15:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, n_constraints=None, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if n_constraints is None:
        n_constraints = min(2*n-1, max(n, 256))
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        g = solve_lp_direction(f, M * m_relax, n, n_constraints, 
                               time_limit=min(30, (time_limit - elapsed) * 0.3))
        if g is None:
            m_relax = min(m_relax * 1.2, 5.0)
            continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                # Perturb
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                m_relax = 1.0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if 0 < best_alpha < 1.0:
            lo = max(0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(20):
                mid = (lo + hi) / 2
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        elif best_alpha >= 1.0:
            best_val = fast_convolve(np.maximum(g, 0.0)).max()
            # g is already normalized
            ft = g.copy()
            best_val = fast_convolve(ft).max()
        
        if best_alpha > 1e-13:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_relax = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 5.0)
    
    return best_f, best_obj

def peak_redistribute(a, time_limit, alpha=0.1, verbose=False):
    """Reduce auto-convolution peaks by redistributing mass."""
    f = np.maximum(a.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_star = conv.argmax()
        
        # Compute responsibility
        k = k_star
        resp = np.zeros(n)
        if k == 0:
            resp = f.copy()  # d(sum a_j^2)/da_j = 2*a_j, proportional to a_j
        else:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                # contribution of j is f[j] * f[j+k] (if valid) or f[j-k] * f[j]
                partner = j + k if j + k < n else j - k
                if 0 <= partner < n:
                    contrib = f[j] * f[partner]
                    resp[j] = max(resp[j], contrib)
        
        if resp.max() < 1e-15:
            break
        
        # Redistribute: reduce high-resp, increase low-resp
        # Use a soft approach
        resp_norm = resp / (resp.sum() + 1e-30)
        shift = alpha * (resp_norm - 1.0 / n)
        f_new = f * (1.0 - 2.0 * f * shift)  # multiplicative
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f
        else:
            f_new /= ss
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller alpha
            alpha = max(alpha * 0.9, 0.001)
            if alpha < 0.005:
                break
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-15:
                continue
            
            best_j_obj = best_obj  # Use current best as threshold
            best_j_val = old_val
            
            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj

def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.005
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def hann_window(n):
    t = np.arange(n) / max(n - 1, 1)
    w = (1 - np.cos(2 * np.pi * t)) / 2
    w[w < 0] = 0
    ss = w.sum()
    if ss > 0:
        w /= ss
    return w

def blackman_harris(n):
    t = np.arange(n) / max(n - 1, 1)
    w = (0.35875 - 0.48829 * np.cos(2*np.pi*t) + 0.14128 * np.cos(4*np.pi*t) 
         - 0.01168 * np.cos(6*np.pi*t))
    w[w < 0] = 0
    ss = w.sum()
    if ss > 0:
        w /= ss
    return w
```

This is essentially: minimize the variance of |F(ω)|^2 over the non-DC frequencies, subject to the time-domain constraints.

One approach: **iteratively reduce the peak of |F(ω)|^2**.
1. Find the frequency ω* where |F(ω*)|^2 is maximum (excluding DC).
2. Add a small negative cosine at that frequency to f: f ← f - ε * cos(2πω*t).
3. Project onto the simplex.
4. Repeat.

This is like a "frequency-domain coordinate descent." Let me implement this:


But there's a subtlety: reducing the peak in the frequency domain might increase the peak in the time domain (auto-convolution). The relationship is:
- Auto-convolution at lag k = IDFT(|F|^2) at lag k.
- So making |F|^2 flat in frequency means making the auto-convolution flat in time (which is what we want!).

So the frequency domain and time domain are connected by the Fourier transform. Making |F|^2 flat IS making the auto-convolution flat.

"""
Multi-phase optimizer for min 2*n*max(auto_convolve(a))/(sum a)^2.
Combines: adaptive-constraint LP-Frank-Wolfe, Gerchberg-Saxton spectrum flattening,
peak-weighted subgradient, coordinate descent, and window-function initializations.
Key new idea: Gerchberg-Saxton style alternating between time-domain (simplex projection)
and frequency-domain (power spectrum flattening) to drive toward flat auto-convolution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, k_vals):
    """Build LP constraint matrix: A[k, i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((len(k_vals), n))
    for idx, k in enumerate(k_vals):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            cols = np.arange(i_start, i_end + 1)
            A[idx, cols] = f[k - cols]
    return A


def solve_lp_direction(f, M, n, n_constraints=None, time_limit=25):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for selected k, g >= 0."""
    total_constraints = 2 * n - 1
    if n_constraints is not None and n_constraints < total_constraints:
        conv_f = fast_convolve(f)
        k_vals = np.argsort(conv_f)[::-1][:n_constraints]
    else:
        k_vals = np.arange(total_constraints)

    A_ub = build_lp_matrix(f, k_vals)
    b_ub = np.full(len(k_vals), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None and res.x.sum() > 1e-15:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, n_constraints=None, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(n, 256))

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        lp_tl = min(20, remaining * 0.25)
        g = solve_lp_direction(f, M * m_relax, n, n_constraints, time_limit=lp_tl)
        if g is None:
            m_relax = min(m_relax * 1.15, 6.0)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                m_relax = 1.0
            continue

        # Line search: coarse
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Line search: bisection refinement
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(20):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        elif best_alpha == 1.0:
            ft = g.copy()
            best_val = fast_convolve(ft).max()

        if best_alpha > 1e-13:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_relax = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.03, 5.0)

    return best_f, best_obj


def gs_flat_spectrum(f0, time_limit, verbose=False):
    """
    Gerchberg-Saxton style: alternate between flattening |F|^2 in frequency domain
    and projecting onto the simplex in time domain.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = 0.4  # blending parameter

    while time.monotonic() - t0 < time_limit:
        it += 1

        F = np.fft.rfft(f, fft_size)
        power = np.abs(F) ** 2
        phase = np.angle(F)

        p_nc = power[1:]
        if p_nc.size == 0:
            break

        # Target: compress dynamic range of power spectrum
        # Use power-law compression: p_new = p^alpha (alpha < 1 flattens)
        # Also blend towards the mean
        mean_p = p_nc.mean()
        if mean_p < 1e-20:
            break

        # Normalize for stable power-law
        p_norm = p_nc / mean_p
        p_compressed = p_norm ** alpha
        p_compressed = p_compressed / p_compressed.mean() * mean_p

        # Blend with original
        p_new = (1 - 0.3) * p_nc + 0.3 * p_compressed

        # Build new F with modified magnitude, same phase
        F_new = F.copy()
        F_new[1:] = np.sqrt(p_new) * np.exp(1j * phase[1:])

        # Inverse transform and project
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adapt alpha: increase compression when stuck
        if no_improve > 30:
            alpha = min(alpha * 0.85, 0.05)
            no_improve = 0
        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            alpha = 0.4
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, alpha={alpha:.3f}")

    return best_f, best_obj


def gs_peak_trim(f0, time_limit, verbose=False):
    """
    Variant: specifically target the peak frequency in |F|^2.
    Reduces the max power spectral component iteratively.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    t_indices = np.arange(n)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    eps = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1

        F = np.fft.rfft(f, fft_size)
        power = np.abs(F) ** 2

        # Find top 5 non-DC peaks
        p_nc = power[1:]
        if p_nc.size < 5:
            top_idxs = np.argsort(p_nc)[::-1][:p_nc.size] + 1
        else:
            top_idxs = np.argpartition(p_nc, -5)[-5:] + 1

        improved = False
        for pi in top_idxs:
            freq = pi / fft_size
            cos_pat = np.cos(2 * np.pi * freq * t_indices)
            sin_pat = np.sin(2 * np.pi * freq * t_indices)

            for pattern in [cos_pat, sin_pat]:
                for sign in [-1, 1]:
                    f_trial = f + sign * eps * pattern
                    f_trial = project_simplex(f_trial)
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            eps *= 0.8
            if eps < 1e-5:
                break

        if it % 500 == 0:
            # Gradually increase step
            eps = min(eps * 1.5, 0.02)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS-PeakTrim iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, eps={eps:.6f}")

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def hann_window(n):
    t = np.arange(n) / max(n - 1, 1)
    w = (1 - np.cos(2 * np.pi * t)) / 2
    w[w < 0] = 0
    ss = w.sum()
    if ss > 0:
        w /= ss
    return w


def blackman_harris(n):
    t = np.arange(n) / max(n - 1, 1)
    w = (0.35875 - 0.48829 * np.cos(2 * np.pi * t) + 0.14128 * np.cos(4 * np.pi * t)
         - 0.01168 * np.cos(6 * np.pi * t))
    w[w < 0] = 0
    ss = w.sum()
    if ss > 0:

    return w


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-FW, Gerchberg-Saxton spectrum flattening,
    subgradient polishing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = hann_window(512)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent (hann512): n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: GS spectrum flattening on parent ===
    print(f"\n=== Phase 1: GS Spectrum (n={n_parent}) ===")
    t_rem = deadline - time.monotonic()
    t1 = min(200, t_rem * 0.15)
    if t1 > 10:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = gs_flat_spectrum(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}, best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    t_rem = deadline - time.monotonic()
    t2 = min(350, t_rem * 0.22)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2,
                                   n_constraints=min(n_cur, 512), verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: GS Peak Trim ===
    print(f"\n=== Phase 3: GS PeakTrim (best={best_score:.6f}) ===")
    t_rem = deadline - time.monotonic()
    t3 = min(150, t_rem * 0.1)
    if t3 > 10:
        print(f"  Budget: {t3:.0f}s")
        a_opt, _ = gs_peak_trim(best_seq, time_limit=t3, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128, 256]:
        nl = n_best + delta
        if 128 <= nl <= 2048 and nl != n_best:
            candidate_lengths.append(nl)
    # Also try fixed good lengths
    for nl in [512, 1024]:
        if nl not in candidate_lengths and nl != n_best:
            candidate_lengths.append(nl)
    candidate_lengths = list(set(candidate_lengths))[:6]

    for target_n in candidate_lengths:
        t_rem = deadline - time.monotonic()
        if t_rem - 300 < 15:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(100, (t_rem - 300) * 0.2)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        # Use GS + LP-FW combo
        a_gs, _ = gs_flat_spectrum(f_interp, time_limit=t_alloc * 0.4)
        sc_gs = float(evaluate_sequence(a_gs.tolist()))
        a_opt, _ = lp_fw_optimize(a_gs, time_limit=t_alloc * 0.6,
                                   n_constraints=min(target_n, 256))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc_gs) and sc_gs < best_score:
            best_score = sc_gs
            best_seq = a_gs.copy()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
        if np.isfinite(sc) and sc < 1e10:
            print(f"    n={target_n} score={sc:.6f}")

    # === Phase 5: Window function initializations at best length ===
    print(f"\n=== Phase 5: Window Inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    t = np.linspace(0, 1, n_best)
    bimodal = 0.5 * np.exp(-15 * (t - 0.25) ** 2) + 0.5 * np.exp(-15 * (t - 0.75) ** 2)
    bimodal /= bimodal.sum()

    init_list = [
        ("hann", hann_window(n_best)),
        ("blackman", blackman_harris(n_best)),
        ("bimodal", bimodal),
        ("parent_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)),
        ("parent_large_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)),
    ]

    for label, f_init in init_list:
        t_rem = deadline - time.monotonic()
        if t_rem - 200 < 15:
            break
        t_alloc = min(80, (t_rem - 200) * 0.15)
        if t_alloc < 10:
            continue
        print(f"  {label}: t={t_alloc:.0f}s")
        a_gs, _ = gs_flat_spectrum(f_init, time_limit=t_alloc * 0.5)
        a_opt, _ = lp_fw_optimize(a_gs, time_limit=t_alloc * 0.5,
                                   n_constraints=min(n_best, 256))
        sc = float(evaluate_sequence(a_opt.tolist()))
        sc_gs = float(evaluate_sequence(a_gs.tolist()))
        if np.isfinite(sc_gs) and sc_gs < best_score:
            best_score = sc_gs
            best_seq = a_gs.copy()
            print(f"  NEW BEST [{label}_gs]: {best_score:.6f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {best_score:.6f}")

    # === Phase 6: SG Polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        t_sg = min(t_rem * 0.2, 80)
        a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                             lr=0.0015, mom=0.94,
                             top_k=max(20, n_best // 5),
                             temp=0.00003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 7: Coordinate Descent ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        t_cd = min(t_rem * 0.25, 150)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
