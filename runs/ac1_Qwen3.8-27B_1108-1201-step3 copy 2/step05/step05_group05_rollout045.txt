Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed gradient
with L-BFGS-B in softmax parameterization, temperature annealing,
spectral equalization refinement, and fine coordinate descent.

import numpy as np
import time
from scipy.optimize import minimize as sci_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def compute_lse_obj_grad(x, beta, n, fft_size):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. log-params x."""
    a = softmax_fn(x)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled_max = scaled.max()
    exp_scaled = np.exp(scaled - scaled_max)
    log_sum_exp = scaled_max + np.log(exp_scaled.sum())
    obj = (2.0 * n / beta) * log_sum_exp

    p = exp_scaled / exp_scaled.sum()

    # Cross-correlation: (p * a)[j] = sum_k p[k] * a[k-j]
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)[:n]

    # grad f w.r.t. a: 4n * (p * a)
    grad_f_a = 4.0 * n * corr

    # Chain rule through softmax: grad_x = a * (grad_f_a - dot(grad_f_a, a))
    dot_val = np.dot(grad_f_a, a)
    grad_x = a * (grad_f_a - dot_val)

    return obj, grad_x


def optimize_lbfgs(a_init, n, betas, max_iter_per_beta=3000, tol=1e-12):
    """Temperature annealing with L-BFGS-B in softmax parameterization."""
    a = a_init.copy()

            o, _ = compute_lse_obj_grad(x, beta, n, fft_size)
            return o

def spectral_equalize(a, alpha=0.3, n_iter=20):
    """Iterative spectral equalization: flatten the magnitude spectrum."""
    n = len(a)
    a = a.copy()
    fft_size = 1 << (2 * n - 2).bit_length()

    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15

        # Raise magnitude toward flat: new_mag = mag^(-alpha) normalized
        # This reduces dynamic range of the spectrum
        new_mag = mag ** (-alpha)
        new_mag /= new_mag.max()

        # New spectrum = new_mag * exp(i*phase)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_try = np.fft.irfft(A_new, fft_size)[:n]
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a = a_try / s

    return a

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(best_seq)}, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: L-BFGS-B with temperature annealing on parent length ===
    print(f"\n=== Phase 1: L-BFGS-B Annealing (n={n_parent}) ===")
    print(f"  Current best: {best_score:.6f}")

    a_work = best_seq.copy()
    a_work = project_simplex(a_work, 1.0)

    # Temperature schedule: start smooth, end sharp
    betas = [15, 30, 60, 120, 250, 500, 1000]

    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 400  # reserve time for later phases
    if remaining > 60:
        # Adjust betas based on time
        n_betas = min(len(betas), max(3, int(remaining / 40)))
        betas_use = betas[:n_betas]
        max_iter = max(500, int(remaining / n_betas / 0.05))
        print(f"  Using {n_betas} temperatures: {betas_use}, max_iter each: {max_iter}")

        a_opt = optimize_lbfgs(a_work, n_parent, betas_use, max_iter_per_beta=max_iter, tol=1e-10)
        try_seq(a_opt, f"lbfgs_n{n_parent}")

    elapsed_p1 = time.monotonic() - t0
    print(f"  Phase 1 total time: {elapsed_p1:.1f}s")

    # === Phase 2: Spectral equalization refinement ===
    print(f"\n=== Phase 2: Spectral Equalization (best={best_score:.6f}) ===")
    if best_seq is not None:
        for alpha in [0.1, 0.2, 0.3, 0.5]:
            rem = deadline - time.monotonic() - 200
            if rem < 10:
                break
            a_eq = spectral_equalize(best_seq, alpha=alpha, n_iter=15)
            try_seq(a_eq, f"spec_eq_a{alpha}")

    # === Phase 3: L-BFGS-B refinement at high temperature ===
    print(f"\n=== Phase 3: High-temperature L-BFGS-B (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            for beta in [2000, 5000]:
                rem = deadline - time.monotonic() - 150
                if rem < 10:
                    break
                n_curr = len(best_seq)
                a_opt = optimize_lbfgs(best_seq, n_curr, [beta], max_iter_per_beta=2000, tol=1e-12)
                try_seq(a_opt, f"lbfgs_hi_beta{beta}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_fine")

    # === Phase 5: Pair-wise coordinate descent ===
    print(f"\n=== Phase 5: Pair Wise CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_pw = min(rem * 0.5, 80)
            print(f"  PW time: {t_pw:.0f}s")
            a_opt, _ = coordinate_descent_pairs(best_seq, time_limit=t_pw, seed=seed)
            try_seq(a_opt, "pair_cd")

    # === Phase 6: L-BFGS-B restart from best with perturbation ===
    print(f"\n=== Phase 6: Perturbed L-BFGS-B (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_curr = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            if p == 0:
                noise = rng.standard_normal(n_curr) * 0.003
                a_init = project_simplex(best_seq + noise, 1.0)
            elif p == 1:
                a_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                a_init = project_simplex(a_init, 1.0)
            else:
                a_init = 0.85 * best_seq + 0.15 * (np.ones(n_curr) / n_curr)
                a_init = project_simplex(a_init, 1.0)

            betas_refine = [100, 300, 800]
            a_opt = optimize_lbfgs(a_init, n_curr, betas_refine, max_iter_per_beta=1500, tol=1e-11)
            try_seq(a_opt, f"perturb_lbfgs_{p}")

    # === Phase 7: Try slightly different lengths ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_curr = len(best_seq)
        for mult in [0.75, 1.25, 1.5]:
            target_n = int(n_curr * mult)
            if target_n < 256 or target_n > 2048 or target_n == n_curr:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break

            a_interp = interpolate_sequence(best_seq, target_n)
            # Quick L-BFGS-B
            a_opt = optimize_lbfgs(a_interp, target_n, [50, 200, 500], max_iter_per_beta=1000, tol=1e-10)
            try_seq(a_opt, f"len_{target_n}")

            rem = deadline - time.monotonic() - 25
            if rem > 15:
                a_opt2, _ = coordinate_descent_fine(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt2, f"len_{target_n}_cd")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_final = min(rem, 50)
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_final)
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 10
        if rem > 5:
            n_curr = len(best_seq)
            a_opt = optimize_lbfgs(best_seq, n_curr, [3000], max_iter_per_beta=2000, tol=1e-13)
            try_seq(a_opt, "final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
