def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_subgradient(conv, a, n, top_k=20, temperature=0.01):
    """Compute softmax-weighted subgradient of max(conv)."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    return 2.0 * sg, peaks, w

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

This is a 1D optimization problem that can be solved by binary search on the max constraint.

Specifically, `(a*a)_k = a_j^2 * [j+j=k] + a_j * (sum_{l: l+j=k, l≠j} a_l) + (contribution not involving a_j)`.

We want `max_k (Q_k + a_j * (P + a_j * [2j=k])) <= M`.

- For k ≠ 2j: `a_j <= (M - Q_k) / P_k` where `P_k = sum_{l: l+j=k, l≠j} a_l`.

def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: optimize each coordinate exactly via 1D search
    while maintaining sum=1 and non-negativity.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        # Random order to avoid bias
        perm = np.random.default_rng(it).permutation(n)

        for j in perm:
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            old_val = a[j]

            # 1D optimization: set a[j] = t, scale others to maintain sum=1
            # Others become: a[l] = (1-t) * a[l] / (1 - old_val) for l != j

            # P_k = sum of a[l] for l != j such that l + j = k (times the scaling)
            # R_k = 1 if 2j = k else 0 (times scaling)

            # a'[l] = t if l=j, else (1-t)*a[l]/denom
            # (a'*a')[k] = sum_{l+m=k} a'[l] a'[m]
            # = t^2 * [2j=k] + 2t * sum_{m: m+j=k, m!=j} (1-t)*a[m]/denom
            #   + (1-t)^2/denom^2 * sum_{l+m=k, l!=j, m!=j} a[l]*a[m]

            # Let's just do golden section search on t in [0, min(1, old_val*3)]
            t_lo = 0.0
            t_hi = min(1.0 - 1e-10, old_val * 3.0 + 0.01)
            if t_hi <= t_lo:
                t_hi = min(1.0 - 1e-10, 0.5)

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = t_lo + (1.0 - gr) * (t_hi - t_lo)
            d = t_lo + gr * (t_hi - t_lo)

            a_c = a.copy()
            a_d = a.copy()
            a_c[j] = c
            a_d[j] = d
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_c[mask] = (1.0 - c) * a_c[mask] / denom
            a_d[mask] = (1.0 - d) * a_d[mask] / denom
            a_c = np.maximum(a_c, 0)
            a_d = np.maximum(a_d, 0)
            fc = fast_convolve(a_c / a_c.sum()).max() * 2.0 * n
            fd = fast_convolve(a_d / a_d.sum()).max() * 2.0 * n

            for _ in range(20):
                if fc < fd:
                    t_hi = d
                    d = c
                    fd = fc
                    c = t_lo + (1.0 - gr) * (t_hi - t_lo)
                    a_c[j] = c
                    a_c[mask] = (1.0 - c) * a.original[mask] / denom if hasattr(a, 'original') else (1.0 - c) * a[mask] / denom
                    # Recompute properly
                    a_c = a.copy()
                    a_c[j] = c
                    a_c[mask] = (1.0 - c) * a_c[mask] / denom
                    a_c = np.maximum(a_c, 0)
                    sc = a_c.sum()
                    if sc < 1e-15:
                        fc = float('inf')
                    else:
                        fc = 2.0 * n * fast_convolve(a_c / sc).max()
                else:
                    t_lo = c
                    c = d
                    fc = fd
                    d = t_lo + gr * (t_hi - t_lo)
                    a_d = a.copy()
                    a_d[j] = d
                    a_d[mask] = (1.0 - d) * a_d[mask] / denom
                    a_d = np.maximum(a_d, 0)
                    sd = a_d.sum()
                    if sd < 1e-15:
                        fd = float('inf')
                    else:
                        fd = 2.0 * n * fast_convolve(a_d / sd).max()

            t_best = (c + d) / 2.0
            a_trial = a.copy()
            a_trial[j] = t_best
            a_trial[mask] = (1.0 - t_best) * a_trial[mask] / denom
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()

            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True

        if not improved:
            # Try a small random perturbation
            rng = np.random.default_rng(it * 7 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            else:
                a = best_a.copy()
                break

    return best_a, best_obj


def symmetry_polish(a0, time_limit):
    """Exploit reversal symmetry: blend with reversed version and re-optimize."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_obj(a0)
    a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()

    # Try different blend ratios
    for alpha in [0.5, 0.3, 0.7, 0.2, 0.8]:
        if time.monotonic() - t0 > time_limit:
            break
        a_blend = alpha * a + (1 - alpha) * a[::-1]
        a_blend = np.maximum(a_blend, 1e-15)
        a_blend = a_blend / a_blend.sum()
        obj = compute_obj(a_blend)
        if obj < best_obj:
            best_obj = obj
            best_a = a_blend.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse analytical initializations
    2. Fast mirror descent on promising candidates
    3. Symmetry exploitation
    4. Coordinate descent polish
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    n_base = 512
    inits = {}
    inits['gauss_025'] = make_gaussian(n_base, 0.25)
    inits['gauss_030'] = make_gaussian(n_base, 0.30)
    inits['gauss_035'] = make_gaussian(n_base, 0.35)
    inits['gauss_040'] = make_gaussian(n_base, 0.40)
    inits['semicircle'] = make_semicircle(n_base)
    inits['beta_05_05'] = make_beta(n_base, 0.5, 0.5)
    inits['beta_07_07'] = make_beta(n_base, 0.7, 0.7)
    inits['beta_1_1'] = make_beta(n_base, 1.0, 1.0)
    inits['beta_2_2'] = make_beta(n_base, 2.0, 2.0)
    inits['cosine'] = make_cosine_bell(n_base)
    inits['flat_top'] = make_flat_top(n_base, 0.7)
    inits['flat_top_05'] = make_flat_top(n_base, 0.5)
    inits['uniform'] = np.ones(n_base) / n_base
    inits['mixture_1'] = make_mixture(n_base, [n_base*0.3, n_base*0.7],
                                       [n_base*0.15, n_base*0.15], [0.7, 0.3])
    inits['mixture_2'] = make_mixture(n_base, [n_base*0.2, n_base*0.5, n_base*0.8],
                                       [n_base*0.12, n_base*0.15, n_base*0.12], [0.3, 0.4, 0.3])

    if parent is not None:
        inits['parent'] = interpolate_sequence(parent, n_base)

    # Score all initializations
    init_scores = {}
    for name, a in inits.items():
        obj = compute_obj(a)
        init_scores[name] = (obj, a)
        sc = try_seq(a, f"init_{name}")

    # Sort by score
    sorted_inits = sorted(init_scores.items(), key=lambda x: x[1][0])
    print(f"  Best init: {sorted_inits[0][0]} with obj={sorted_inits[0][1][0]:.8f}")
    for name, (obj, _) in sorted_inits[:5]:
        print(f"    {name}: {obj:.8f}")

    # === Phase 2: Mirror descent on top initializations ===
    print("\n=== Phase 2: Mirror Descent ===")
    # Allocate time to top 4 initializations
    top_names = [name for name, _ in sorted_inits[:4]]
    time_per_init = 200  # seconds each

    for name in top_names:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(time_per_init, rem - 300)  # reserve time for later phases
        if t_alloc < 30:
            break
        a_init = inits[name]
        print(f"  MD on {name}: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(a_init, time_limit=t_alloc,
                                             lr_init=0.05, top_k=60,
                                             temp_init=0.02, label=name)
        try_seq(a_opt, f"md_{name}")

    # === Phase 3: Continue MD on current best ===
    print(f"\n=== Phase 3: Extended MD on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem, 300)
            print(f"  Extended MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc,
                                                 lr_init=0.02, top_k=100,
                                                 temp_init=0.01, label="ext")
            try_seq(a_opt, "md_ext")

    # === Phase 4: Symmetry polish ===
    print(f"\n=== Phase 4: Symmetry Polish ===")
    if best_seq is not None:
        a_sym, obj_sym = symmetry_polish(best_seq, time_limit=30)
        try_seq(a_sym, "symmetry")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    if best_seq is not None:
        for target_n in [384, 640, 768]:
            rem = deadline - time.monotonic() - 60
            if rem < 50:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            t_alloc = min(80, rem - 80)
            if t_alloc < 30:
                break
            print(f"  MD n={target_n}: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                                 lr_init=0.04, top_k=50,
                                                 temp_init=0.02,
                                                 label=f"n{target_n}")
            try_seq(a_opt, f"md_n{target_n}")

    # === Phase 7: Final fine-grained MD ===
    print(f"\n=== Phase 7: Final Fine MD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  Fine MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc,
                                                 lr_init=0.005, top_k=150,
                                                 temp_init=0.005, label="fine")
            try_seq(a_opt, "fine_md")

    # === Phase 8: Final CD polish ===
    print(f"\n=== Phase 8: Final CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Final CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(512, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
        # Re-normalize to ensure consistency
        s = best_seq.sum()
        if s > 1e-10:
            best_seq = best_seq / s

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
