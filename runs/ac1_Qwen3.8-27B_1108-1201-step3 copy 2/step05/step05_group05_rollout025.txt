import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - target_sum) / rho) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the convolution matrix C where C @ g = conv(f, g)."""
    n = len(f)
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0

    all_cols = []

        all_cols.extend([j] * n)

def logsumexp_smooth(f, temp):
    """Compute smooth approximation of max(conv(f,f)) using log-sum-exp."""
    conv = fast_convolve(f)
    scaled = conv / temp
    max_scaled = scaled.max()
    if max_scaled > 700:
        return conv.max()  # numerical stability
    return temp * (max_scaled + np.log(np.sum(np.exp(scaled - max_scaled))))

def logsumexp_grad(f, temp, top_k=None):
    """Compute gradient of log-sum-exp smoothing of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    k = len(conv) if top_k is None else min(top_k, len(conv))
    if top_k is not None and top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    else:
        peaks = np.arange(k)
        pv = conv
    
    scaled = pv / temp
    max_s = scaled.max()
    w = np.exp(scaled - max_s)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k_idx = int(peaks[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * f[k_idx - idx]
    
    return grad, conv.max()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - target_sum) / rho) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1)
    return np.maximum(v - theta, 0.0)

def logsumexp_gradient(f, temp, top_k=200):

    n = len(f)

    if m > 500:

        max_idx = peaks[np.argmax(pv)]

        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * f[max_idx - idx]
        return grad, conv.max()

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Temperature schedule: list of (fraction_start, fraction_end, temperature)
    temps = [
        (0.0, 0.30, temp_start),
        (0.30, 0.55, temp_start * 0.4),
        (0.55, 0.75, temp_start * 0.15),
        (0.75, 0.90, temp_start * 0.05),
        (0.90, 1.00, temp_start * 0.01),
    ]

    for frac_s, frac_e, temp in temps:
        phase_start = t0 + frac_s * time_limit
        phase_end = t0 + frac_e * time_limit
        now = time.monotonic()
        if now >= phase_end:
            continue
        if now < phase_start:
            time.sleep(min(phase_start - now, 0.5))

        def obj_func(x):
            x = np.maximum(x, 0.0)
            ss = x.sum()
            if ss < 1e-15:
                return 1e20
            x = x / ss
            return logsumexp_value(x, temp)

        def grad_func(x):
            x = np.maximum(x, 0.0)
            ss = x.sum()
            if ss < 1e-15:
                return np.zeros(len(x))
            x = x / ss
            grad, _ = logsumexp_gradient(x, temp)
            return grad

        try:
            result = minimize(
                obj_func, a.copy(), jac=grad_func,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 1000}
            )
            if result.x is not None:
                a_new = np.maximum(result.x, 0.0)
                ss = a_new.sum()
                if ss > 1e-10:
                    a_new = a_new / ss
                    obj = fast_convolve(a_new).max()
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_a = a_new.copy()
                        a = a_new.copy()
                        if verbose:
                            print(f"    L-BFGS temp={temp:.2e}: obj={obj:.10f}")
        except Exception:
            pass

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                remainder = 1.0 - old_val
                if remainder < 1e-12:
                    continue
                scale = (1.0 - new_val) / remainder
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                remainder = 1.0 - old_val
                if remainder > 1e-12:
                    scale = (1.0 - best_j_val) / remainder
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def subgradient_optimize(f0, time_limit):
    """Subgradient optimization with softmax smoothing and momentum."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    tau = 50.0
    lr = 0.03
    momentum = 0.8
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k = min(150, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k_idx = int(peaks[i])
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k_idx - idx]

        grad_centered = grad - grad.mean()
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        if no_improve > 300:
            lr = max(0.002, lr * 0.95)
            momentum = min(0.95, momentum + 0.003)
            tau = min(400.0, tau * 1.02)
        elif no_improve > 100:
            tau = max(20.0, tau * 0.99)
        else:
            tau = min(300.0, tau * 1.005)

        if no_improve > 500:
            rng_r = np.random.default_rng(int(time.monotonic() * 1000))
            noise = rng_r.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.02
            tau = 40.0
            momentum = 0.8

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining LP-based iteration, L-BFGS with log-sum-exp
    smoothing, subgradient methods, coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = flat_spectrum_init(n_init, rng)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Flat spectrum init: n={n_init}, score={p_score:.6f}")
        best_seq = parent.copy()
        best_score = p_score

    def record(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # === Phase 1: LP Optimization ===
    print(f"\n=== Phase 1: LP Optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 120
    t_lp = min(rem * 0.35, 600)
    print(f"  Time budget: {t_lp:.0f}s")
    a_opt, obj = lp_optimize(parent, time_limit=t_lp, verbose=True)
    record(a_opt, "lp1")

    # === Phase 2: Subgradient from multiple starts ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_sg = min(rem * 0.2, 150)
            print(f"  Time: {t_sg:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg)
            record(a_opt, "sg")

    # === Phase 3: L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_lbs = min(rem * 0.2, 150)
            print(f"  Time: {t_lbs:.0f}s")
            a_opt, obj = lbfgs_optimize(best_seq, time_limit=t_lbs, temp_start=0.0005, verbose=True)
            record(a_opt, "lbfgs")

    # === Phase 4: LP Refinement ===
    print(f"\n=== Phase 4: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_lp2 = min(rem * 0.25, 200)
            print(f"  Time: {t_lp2:.0f}s")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_lp2, verbose=False)
            record(a_opt, "lp_refine")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_cd = min(rem * 0.25, 100)
            print(f"  Time: {t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            record(a_opt, "cd")

    # === Phase 6: Perturbation + LP ===
    print(f"\n=== Phase 6: Perturbation + LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = [
            ("noise_small", best_seq + rng.standard_normal(n_best) * 0.003),
            ("noise_med", best_seq + rng.standard_normal(n_best) * 0.008),
            ("reverse_mix", 0.7 * best_seq + 0.3 * best_seq[::-1]),
            ("flat_mix", 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)),
            ("mask", best_seq * (rng.random(n_best) > 0.05)),
            ("flat_spec", flat_spectrum_init(n_best, rng)),
        ]
        for label, f_pert in perturbations:
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break
            f_init = project_simplex(f_pert, target_sum=1.0)
            t_alloc = min(rem / 2.0, 50)
            if t_alloc < 10:
                t_alloc = rem * 0.6
            a_opt, obj = lp_optimize(f_init, time_limit=t_alloc, verbose=False)
            record(a_opt, f"pert_{label}")

            rem = deadline - time.monotonic() - 10
            if rem > 10 and best_score < 1.504:
                a_opt2, obj2 = coordinate_descent(a_opt, time_limit=min(rem, 30))
                record(a_opt2, f"cd_pert_{label}")

    # === Phase 7: Different lengths ===
    print(f"\n=== Phase 7: Different Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for target_n in [n_best - 64, n_best + 64, n_best + 128, n_best - 128]:
            if target_n < 256 or target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 15
            if rem < 25:
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 60)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(f_interp, time_limit=t_alloc, verbose=False)
            record(a_opt, f"len_{target_n}")

            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_opt2, obj2 = coordinate_descent(a_opt, time_limit=min(rem, 25))
                record(a_opt2, f"cd_len_{target_n}")

    # === Phase 8: Final intense LP + CD ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_final_lp = min(rem * 0.5, 40)
            a_opt, obj = lp_optimize(best_seq, time_limit=t_final_lp, verbose=False)
            record(a_opt, "final_lp")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 30))
            record(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
