```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a combination of:
1. LP-based direction finding (projection onto convolution ball)
2. Log-softmax L-BFGS with temperature annealing
3. Coordinate descent and subgradient polishing
The LP approach provides principled descent directions by solving:
  max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, time_limit=10.0):
    """
    Solve the LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # Constraint matrix: A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # Shape: (2n-1) x n
    # For memory efficiency with large n, we can use the structure
    
    if n > 3000:
        # Too large for dense LP, fall back to subgradient
        return subgradient_direction(f, M, n_steps=50)
    
    # Build constraint matrix
    # A[k, i] = f[k-i] for valid (k-i)
    k_range = np.arange(2 * n - 1)
    i_range = np.arange(n)
    # A[k, i] = f[k-i] if 0 <= k-i < n
    # This is a Toeplitz matrix structure
    # For memory: (2n-1) * n elements. For n=768: 1535*768 = ~1.18M, fine.
    # For n=2048: 4095*2048 = ~8.4M, still fine.
    # For n=4096: 8191*4096 = ~33.5M, getting large.
    # For n=8192: 16383*8192 = ~134M, too large for dense.
    
    if n <= 4096:
        diff = k_range[:, None] - i_range[None, :]
        A = np.where((diff >= 0) & (diff < n), f[diff], 0.0)
    else:
        return subgradient_direction(f, M, n_steps=50)
    
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum(g)
    
    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception as e:
        print(f"    LP failed: {e}")
    
    return None

def subgradient_direction(f, M, n_steps=50):
    """
    Approximate the LP direction using subgradient projection.
    """
    n = len(f)
    t0 = time.monotonic()
    g = np.ones(n) / n
    lr = 1.0 / n
    
    for step in range(n_steps):
        if time.monotonic() - t0 > 5.0:
            break
        fg = fast_cross_convolve(f, g)
        # Find peaks
        peaks = fg[fg > M]
        if len(peaks) == 0:
            # Already feasible, increase g
            g *= 1.01
            break
        # Subgradient of max(fg - M, 0) w.r.t. g
        # = sum over peaks of (indicator of peak) * f reversed
        sg = np.zeros(n)
        for k in np.argwhere(fg > M).flatten():
            # d(fg[k])/dg[i] = f[k-i] for valid i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += f[k - np.arange(js, je+1)]
        
        g -= lr * sg
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            g /= s
    
    return g

def compute_objective_normalized(a):
    """Compute R = 2*n*max(conv)/sum^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_based_optimize(f0, time_limit=120.0, max_iter=1000):
    """
    Iterative LP-based optimization:
    1. Compute M = ||f*f||_inf
    2. Find g = argmax sum(g) s.t. g>=0, ||f*g||_inf <= M
    3. Normalize g to have same sum as f
    4. Update f = (1-t)*f + t*g
    """
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum=1
    n = len(f)
    
    best_f = f.copy()
    best_obj = f * f  # conv
    best_obj_val = fast_convolve(f).max()
    
    step_size = 0.3
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        # Solve LP for direction
        g = lp_direction(f, M, time_limit=min(15.0, (time_limit - (time.monotonic() - t0)) * 0.3))
        
        if g is None:
            # Fallback to subgradient
            g = subgradient_direction(f, M, n_steps=30)
        
        if g is None or g.sum() < 1e-15:
            break
        
        # Normalize g to have same sum as f
        g = g / g.sum()
        
        # Update
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Log-softmax L-BFGS with adaptive temperature annealing for smooth optimization
2. LP-based projection (max sum(g) s.t. g>=0, f*g<=M) for principled descent directions
3. Coordinate descent and subgradient polishing for fine-tuning
The LP approach provides complementary descent directions that help escape local minima
that the smooth L-BFGS approach may get stuck in. Multi-length exploration and diverse
restarts improve the chance of finding the global minimum.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_obj_and_grad(z, T, n):
    """
    Compute the soft-max smoothed objective and its gradient w.r.t. z.
    Objective: (2n/T) * log(sum_k exp(T * b_k)) where b = conv(a,a), a = softmax(z).
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Weights for each lag
    w = exp_Tb / exp_Tb.sum()

    # h[i] = sum_k w[k] * a[k-i] for valid (k-i)
    # h[i] = (w * a_flip)[n-1+i] where a_flip[j] = a[n-1-j]
    a_flip = a[::-1]
    conv_w_af = fast_cross(w, a_flip)
    h = conv_w_af[n - 1:n - 1 + n]

    # Gradient: dF/dz_j = 4n * a_j * (h[j] - S) where S = sum h[i]*a[i]
    grad_a = 4.0 * n * h
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def lbgfs_run(z0, T, n, time_limit, maxiter=5000):
    """Run L-BFGS optimization in log-space with soft-max objective."""
    t0 = time.monotonic()
    elapsed_check = [0.0]

    def func(z):
        elapsed_check[0] = time.monotonic()
        if elapsed_check[0] - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_and_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def build_conv_matrix(f):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0. Shape: (2n-1) x n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_direction(f, M, time_limit=10.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None on failure.
    """
    n = len(f)
    if n > 3000:
        return None

    t0 = time.monotonic()
    rem = time_limit - (t0 - t0)
    if rem < 1.0:
        return None

    try:
        A = build_conv_matrix(f)
    except Exception:
        return None

    try:
        b = np.full(2 * n - 1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': max(1.0, rem - 0.5)})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception as e:
        print(f"    LP error: {e}")

    return None


def lp_optimize(f0, time_limit=120.0, max_iter=500):
    """
    LP-based iterative optimization:
    1. M = max(f*f)
    2. g = argmax sum(g) s.t. g>=0, f*g <= M
    3. Normalize g, update f = (1-t)*f + t*g
    """
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    step_size = 0.3
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv_f = fast_convolve(f)
        M = conv_f.max()

        lp_time = min(8.0, (time_limit - (time.monotonic() - t0)) * 0.4)
        if lp_time < 1.5:
            break

        g = lp_direction(f, M, time_limit=lp_time)

        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Try a few step sizes
        best_step = step_size
        best_trial_obj = best_obj
        for t in [step_size * 0.5, step_size, step_size * 1.5]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            obj_new = fast_convolve(f_new).max()
            if obj_new < best_trial_obj:
                best_trial_obj = obj_new
                best_step = t

        f_new = (1 - best_step) * f + best_step * g
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            break
        f = f_new / sn

        obj_now = fast_convolve(f).max()
        if obj_now < best_obj - 1e-12:
            best_obj = obj_now
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            step_size *= 0.6
            no_improve = 0
            if step_size < 0.02:
                break

    return best_f, best_obj


def coordinate_descent(a0, time_limit=30.0):
    """Coordinate descent cycling through multipliers on the simplex."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def subgradient_polish(a0, time_limit=60.0, lr_init=0.004, mom=0.92, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = 0.001

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        co = fast_convolve(a).max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.008

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("Using default uniform parent of length 768")

    # === Phase 1: L-BFGS at parent length with temperature annealing ===
    print("\n=== Phase 1: L-BFGS log-softmax at parent length ===")
    n_parent = len(parent)
    t1_budget = min(500.0, (deadline - time.monotonic()) * 0.35)

    if t1_budget > 30 and n_parent <= 8192:
        z0 = z_from_a(parent)
        temps = [10, 30, 80, 200, 500]
        z_curr = z0
        per_temp = t1_budget / len(temps)

        for idx, T in enumerate(temps):
            rem = deadline - time.monotonic() - (len(temps) - idx - 1) * per_temp * 0.3
            if rem < 10:
                break
            t_alloc = min(per_temp * 1.2, 120.0)
            print(f"  L-BFGS n={n_parent}, T={T}, t={t_alloc:.0f}s")
            z_curr, _ = lbgfs_run(z_curr, T, n_parent, t_alloc, maxiter=3000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"lbfgs_T{T}")

    # === Phase 2: LP-based optimization ===
    print(f"\n=== Phase 2: LP-based optimization (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    if n_base <= 3000:
        t2_budget = min(300.0, (deadline - time.monotonic()) * 0.2)
        if t2_budget > 20:
            print(f"  LP optimize n={n_base}, t={t2_budget:.0f}s")
            a_lp, lp_obj = lp_optimize(base, time_limit=t2_budget, max_iter=300)
            try_seq(a_lp, "lp_opt")

            # Continue with another round if improved and time allows
            if best_seq is not None and best_score < 1.51:
                rem = deadline - time.monotonic() - 100
                if rem > 30:
                    t2b = min(rem * 0.5, 120.0)
                    print(f"  LP round 2 n={n_base}, t={t2b:.0f}s")
                    a_lp2, _ = lp_optimize(best_seq, time_limit=t2b, max_iter=200)
                    try_seq(a_lp2, "lp_opt2")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [1024, 1536, 2048, 3072]
    total_p3 = min(500.0, (deadline - time.monotonic()) * 0.3)
    per_len = total_p3 / len(target_lengths)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break

        f_interp = interpolate_sequence(base, target_n)
        z0 = z_from_a(f_interp)

        # Quick L-BFGS with moderate T
        t_alloc = min(per_len * 0.5, 100.0)
        if t_alloc < 10:
            continue
        print(f"  L-BFGS n={target_n}, T=20, t={t_alloc:.0f}s")
        z_curr, _ = lbgfs_run(z0, 20, target_n, t_alloc, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"lbfgs_n{target_n}_T20")

        # Refine with higher T
        rem = deadline - time.monotonic() - 100
        if rem < 15:
            continue
        t_alloc2 = min(per_len * 0.35, 60.0)
        if t_alloc2 < 8:
            continue
        print(f"  Refine n={target_n}, T=100, t={t_alloc2:.0f}s")
        z_curr, _ = lbgfs_run(z_curr, 100, target_n, t_alloc2, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"lbfgs_n{target_n}_T100")

        # Try LP on the best at this length if n allows
        if target_n <= 2048 and best_seq is not None:
            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_lp = min(rem * 0.4, 60.0)
                print(f"  LP n={target_n}, t={t_lp:.0f}s")
                a_lp, _ = lp_optimize(best_seq, time_limit=t_lp, max_iter=100)
                try_seq(a_lp, f"lp_n{target_n}")

    # === Phase 4: Deep refinement on best ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # High-T L-BFGS (near-exact max)
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            z0 = z_from_a(best_seq)
            for T in [200, 500, 1000]:
                rem = deadline - time.monotonic() - 60
                if rem < 8:
                    break
                t_alloc = min(rem * 0.3, 40.0)
                if t_alloc < 5:
                    break
                print(f"  Deep L-BFGS n={n_best}, T={T}, t={t_alloc:.0f}s")
                z_curr, _ = lbgfs_run(z0, T, n_best, t_alloc, maxiter=2000)
                a_curr = softmax_from_z(z_curr)
                try_seq(a_curr, f"deep_T{T}")
                z0 = z_curr

        # LP on the best
        if n_best <= 2500:
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                t_lp = min(rem * 0.5, 60.0)
                print(f"  Final LP n={n_best}, t={t_lp:.0f}s")
                a_lp, _ = lp_optimize(best_seq, time_limit=t_lp, max_iter=150)
                try_seq(a_lp, "final_lp")

    # === Phase 5: Fine-tuning ===
    print(f"\n=== Phase 5: Fine-tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_cd = min(rem * 0.5, 45.0)
            print(f"  CD n={n_best}, t={t_cd:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

        # Subgradient polish
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_sg = min(rem * 0.6, 30.0)
            print(f"  Subgradient n={n_best}, t={t_sg:.0f}s")
            a_sg, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                         lr_init=0.003, mom=0.93,
                                         top_k=max(20, n_best // 15))
            try_seq(a_sg, "sg")

    # === Phase 6: Random restarts + quick optimize ===
    print(f"\n=== Phase 6: Random restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            attempts = min(5, int(rem // 12))
            for attempt in range(attempts):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                noise_scale = 0.003 * (attempt + 1)
                noise = rng.standard_normal(n_best) * noise_scale
                a_pert = project_simplex(best_seq + noise)
                z_pert = z_from_a(a_pert)
                t_alloc = min(rem * 0.5, 15.0)
                if t_alloc < 4:
                    break
                z_p, _ = lbgfs_run(z_pert, 300, n_best, t_alloc, maxiter=1000)
                a_p = softmax_from_z(z_p)
                try_seq(a_p, f"restart_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Shouldn't happen, but fallback
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
