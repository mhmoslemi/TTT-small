```python
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = np.sum(a)
    if s < 1e-15:
        return np.inf, a
    a_norm = a / s
    n = len(a_norm)
    conv = fast_convolve(a_norm)
    max_b = conv.max()
    return 2.0 * n * max_b, a_norm

def project_nonneg_sum(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

    Iterative spectral shaping: alternate between flattening the power spectrum
    in frequency domain and projecting to non-negativity in time domain.
    """

        a = np.ones(n) / n
    else:
        a = a / s
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax parameterization with L-BFGS-B and temperature annealing
2. Block coordinate descent
3. Multi-restart strategy
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def softmax_grad_assemble(n, p, a, N_fft):
    """Compute gradient of soft_tmax(conv(a,a)) w.r.t. a using FFT."""
    p_pad = np.zeros(N_fft)
    p_pad[:2*n-1] = p
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    P = np.fft.fft(p_pad)
    A = np.fft.fft(a_pad)
    r = np.fft.ifft(P * np.conj(A)).real[:n]
    return 2.0 * r

    Optimize using softmax parameterization + L-BFGS-B with temperature annealing.
    """

    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    
    N_fft = 1 << (3 * n - 2).bit_length()
    
    state = {'t': t_start, 'counts': [0, 0]}  # [func evals, grad evals]
    
    def obj_and_grad(theta):
        state['counts'][0] += 1
        if state['counts'][1] > 0:
            state['counts'][0] -= 1  # L-BFGS calls both, count once
        state['counts'][1] += 1
        
        t = state['t']
        # theta to a
        th = theta - theta.max()
        a = np.exp(th)
        s = a.sum()
        if s < 1e-300:
            return 1e10, np.zeros(n)
        a = a / s
        
        c = fast_convolve(a)
        c_max = c.max()
        # Soft max
        c_shifted = c / t - c_max / t
        p = np.exp(c_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            p /= p_sum
        soft_max = c_max + t * np.log(p_sum)
        obj = 2.0 * n * soft_max
        
        # Gradient
        grad_a = 2.0 * n * softmax_grad_assemble(n, p, a, N_fft)
        # Convert to theta gradient
        G = np.dot(grad_a, a)
        grad_theta = a * (grad_a - G)
        
        return obj, grad_theta

    theta0 = np.log(np.maximum(a, 1e-15))
    
    best_a = a.copy()
    best_true_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    n_anneal_steps = 20
    anneal_interval = time_limit / n_anneal_steps
    
    for step in range(n_anneal_steps):
        state['t'] = t_start * (t_end / t_start) ** (step / (n_anneal_steps - 1))
        t_alloc = min(anneal_interval * 1.5, time_limit - (time.monotonic() - t0) - 10)
        if t_alloc < 5:

        try:

                          jac=True, options={'maxiter': 200, 'maxfun': 500,
                                            'iprint': -1, 'gtol': 1e-8})
            theta0 = res.x
        except Exception as e:
            print(f"  L-BFGS error at step {step}: {e}")
            break

        a_new = np.exp(th)
        a_new = a_new / a_new.sum()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized L-BFGS-B
with temperature annealing, efficient FFT-based gradients, coordinate descent,
and structured pair-move local search. Key innovation: exact gradient of the
smoothed max via cross-correlation in frequency domain, combined with a
multi-resolution temperature schedule and diverse restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def soft_max_and_grad(a, t, N_fft):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_value, gradient_array).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    c_shifted = (c - c_max) / t
    exp_vals = np.exp(np.maximum(c_shifted, -50))
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
        soft_max = c_max
    else:
        p = exp_vals / p_sum
        soft_max = c_max + t * np.log(p_sum)

    # Gradient: grad[j] = 2 * sum_m p[m+j] * a[m] for j=0..n-1
    p_pad = np.zeros(N_fft)
    p_pad[:2 * n - 1] = p
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    P = np.fft.fft(p_pad)
    A = np.fft.fft(a_pad)
    r = np.fft.ifft(P * np.conj(A)).real[:n]
    grad = 2.0 * r
    return soft_max, grad


def lbfgsb_optimize(a0, time_limit, t_list, max_iter_per_step=150, verbose=False):
    """
    L-BFGS-B optimization with softmax parameterization and temperature annealing.
    a0: initial non-negative sequence (will be normalized).
    t_list: list of temperatures (descending).
    Returns (best_a, best_true_max).
    """
    n = len(a0)
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()

    N_fft = 1 << (3 * n - 2).bit_length()

    state = {'t': t_list[0]}

    def obj_and_grad(theta):
        t = state['t']
        th = theta - theta.max()
        a_local = np.exp(th)
        s = a_local.sum()
        if s < 1e-300:
            return 1e10, np.zeros(n)
        a_local = a_local / s

        soft_max, grad_a = soft_max_and_grad(a_local, t, N_fft)
        obj = 2.0 * n * soft_max

        # Convert gradient to theta space
        G = np.dot(grad_a, a_local)
        grad_theta = a_local * (grad_a - G)

        return obj, grad_theta

    theta0 = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    best_true = fast_convolve(a).max()
    t0 = time.monotonic()

    for si, t_val in enumerate(t_list):
        rem = time_limit - (time.monotonic() - t0) - 5
        if rem < 3:
            break
        state['t'] = t_val

        try:
            res = minimize(obj_and_grad, theta0, method='L-BFGS-B',
                           jac=True, options={'maxiter': max_iter_per_step,
                                              'maxfun': max_iter_per_step * 3,
                                              'iprint': -1, 'gtol': 1e-10})
            theta0 = res.x
        except Exception:
            break

        th = theta0 - theta0.max()
        a_new = np.exp(th)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        true_max = fast_convolve(a_new).max()
        if true_max < best_true:
            best_true = true_max
            best_a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LBFGS step {si}: t={t_val:.2e}, true_max={best_true:.10f} ({elapsed:.1f}s)")

    return best_a, best_true


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent with rescaling to maintain sum=1."""
    if multipliers is None:
        multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-12:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[np.arange(n) != j] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-12:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-12:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    a[np.arange(n) != j] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and pass_count > 1:
            break

    return best_a, best_obj


def pair_move_search(a0, time_limit, seed=0):
    """Structured pair-move local search."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    fracs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]

    while time.monotonic() - t0 < time_limit:
        # Try random pairs
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue
            for frac in fracs:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

        # Also try structured pairs (adjacent, symmetric)
        for i in range(0, n, 4):
            if time.monotonic() - t0 > time_limit:
                break
            j = n - 1 - i
            if j <= i or a[i] < 1e-12 or a[j] < 1e-12:
                continue
            for frac in [0.05, 0.1, 0.2]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_obj


def block_descent(a0, time_limit, block_size=4):
    """Block coordinate descent: optimize groups of block_size coordinates."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random blocks
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            start = rng.integers(0, max(1, n - block_size))
            block = np.arange(start, start + block_size)
            block_mask = np.zeros(n, dtype=bool)
            block_mask[block] = True

            block_vals = a[block].copy()
            others_sum = 1.0 - block_vals.sum()
            if others_sum < 1e-12:
                continue

            best_block = block_vals.copy()
            best_obj_trial = best_obj

            # Try perturbations of the block
            for _ in range(20):
                perturbation = rng.standard_normal(block_size) * 0.1 * block_vals
                new_block = np.maximum(block_vals + perturbation, 0)
                nb_sum = new_block.sum()
                if nb_sum < 1e-12:
                    continue
                new_block = new_block / nb_sum * block_vals.sum()
                if new_block.sum() > 1.0:
                    continue
                a_trial = a.copy()
                a_trial[block] = new_block
                # Rescale others
                new_others_sum = 1.0 - new_block.sum()
                if new_others_sum < 1e-12:
                    continue
                scale = new_others_sum / others_sum
                a_trial[~block_mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-12:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_obj_trial - 1e-15:
                    best_obj_trial = val
                    best_block = new_block.copy()

            if best_obj_trial < best_obj - 1e-15:
                a[block] = best_block
                new_others_sum = 1.0 - best_block.sum()
                if new_others_sum > 1e-12:
                    scale = new_others_sum / others_sum
                    a[~block_mask] *= scale
                a = np.maximum(a, 0)
                a /= a.sum()
                best_obj = best_obj_trial
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. L-BFGS-B with softmax parameterization and temperature annealing
    2. Coordinate descent
    3. Block descent
    4. Pair-move search
    5. Multi-length exploration
    6. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback parent: score={best_score:.6f}")

    # === Phase 1: L-BFGS-B with temperature annealing ===
    print(f"\n=== Phase 1: L-BFGS-B Optimization (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Temperature schedule (in units of the convolution values)
    # Typical c value ~ 1/(2n), peak ~ 1.5/(2n)
    typical_c = 1.0 / (2 * n_best)
    t_list = [typical_c * 5, typical_c * 2, typical_c * 1, typical_c * 0.5,
              typical_c * 0.25, typical_c * 0.1, typical_c * 0.05]

    rem = deadline - time.monotonic() - 600
    t_lbfgs = min(rem * 0.3, 120)
    print(f"  L-BFGS-B: n={n_best}, t_list={t_list}, time={t_lbfgs:.0f}s")
    a_opt, obj = lbfgsb_optimize(best_seq, time_limit=t_lbfgs, t_list=t_list,
                                 max_iter_per_step=100, verbose=True)
    try_seq(a_opt, "lbfgs_main")

    # Second L-BFGS run with different initialization
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(best_seq + noise, target_sum=1.0)
            t_list2 = [typical_c * 2, typical_c * 1, typical_c * 0.5,
                       typical_c * 0.25, typical_c * 0.1, typical_c * 0.05]
            t_alloc = min(rem * 0.2, 60)
            a_opt, obj = lbfgsb_optimize(a_pert, time_limit=t_alloc, t_list=t_list2,
                                         max_iter_per_step=80)
            try_seq(a_opt, "lbfgs_perturb")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 15:
            t_cd = min(rem * 0.3, 150)
            print(f"  CD time={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 3: Block Descent ===
    print(f"\n=== Phase 3: Block Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_bd = min(rem * 0.2, 100)
            print(f"  BD time={t_bd:.0f}s")
            a_opt, obj = block_descent(best_seq, time_limit=t_bd, block_size=4)
            try_seq(a_opt, "block4")
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                a_opt, obj = block_descent(best_seq, time_limit=min(rem, 80), block_size=8)
                try_seq(a_opt, "block8")

    # === Phase 4: Pair Move Search ===
    print(f"\n=== Phase 4: Pair Move Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_ps = min(rem * 0.25, 100)
            print(f"  PS time={t_ps:.0f}s")
            a_opt, obj = pair_move_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair")

    # === Phase 5: Another L-BFGS round on the improved sequence ===
    print(f"\n=== Phase 5: L-BFGS Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        typical_c2 = 1.0 / (2 * n_cur)
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_list3 = [typical_c2 * 1, typical_c2 * 0.5, typical_c2 * 0.25,
                       typical_c2 * 0.1, typical_c2 * 0.05, typical_c2 * 0.02]
            t_alloc = min(rem * 0.3, 80)
            a_opt, obj = lbfgsb_optimize(best_seq, time_limit=t_alloc, t_list=t_list3,
                                         max_iter_per_step=120)
            try_seq(a_opt, "lbfgs_refine")

            # CD after refinement
            rem = deadline - time.monotonic() - 60
            if rem > 10:
                a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 40))
                try_seq(a_opt, "cd_after_refine")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        lengths_to_try = []
        for mult in [0.75, 0.5, 1.25, 1.5]:
            target_n = int(n_best * mult)
            target_n = max(256, min(2048, target_n))
            if target_n != n_best and target_n not in lengths_to_try:
                lengths_to_try.append(target_n)

        for target_n in lengths_to_try[:3]:
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            typical_c3 = 1.0 / (2 * target_n)
            t_alloc = min(rem * 0.3, 60)
            t_list4 = [typical_c3 * 2, typical_c3 * 1, typical_c3 * 0.5,
                       typical_c3 * 0.2, typical_c3 * 0.1]
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lbfgsb_optimize(f_interp, time_limit=t_alloc, t_list=t_list4,
                                         max_iter_per_step=80)
            try_seq(a_opt, f"len_{target_n}")

            rem = deadline - time.monotonic() - 40
            if rem > 10:
                a_opt, obj = coordinate_descent(a_opt, time_limit=min(rem, 30))
                try_seq(a_opt, f"cd_len_{target_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_final = len(best_seq)

        # Multiple short CD rounds with perturbation
        for p in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            if p == 0:
                f_init = best_seq.copy()
            else:
                noise = rng.standard_normal(n_final) * (0.001 + 0.002 * p)
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            a_opt, obj = coordinate_descent(f_init, time_limit=min(rem, 20))
            try_seq(a_opt, f"polish_cd_{p}")

        # Final L-BFGS with very low temperature
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            typical_c4 = 1.0 / (2 * n_final)
            t_list5 = [typical_c4 * 0.1, typical_c4 * 0.05, typical_c4 * 0.02]
            a_opt, obj = lbfgsb_optimize(best_seq, time_limit=min(rem, 20),
                                         t_list=t_list5, max_iter_per_step=100)
            try_seq(a_opt, "final_lbfgs")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
