import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def spectral_equalize(a, n_iter=500, alpha_start=0.9, alpha_end=0.3, verbose=False):
    """Iteratively flatten the power spectrum via IFTA."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-15:
        return np.ones(n) / n
    a = a / S
    
    fft_size = 1 << (2 * n - 1).bit_length()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iter):
        # Linear interpolation of alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(n_iter - 1, 1))
        
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress non-DC magnitudes
        mag_new = mag.copy()
        if len(mag) > 1:
            # Scale so that the product of compression preserves some energy
            non_dc_mag = mag[1:]
            if non_dc_mag.max() > 1e-30:
                mag_new[1:] = non_dc_mag ** alpha
                # Renormalize to preserve total non-DC energy
                old_energy = (non_dc_mag ** 2).sum()
                new_energy = (mag_new[1:] ** 2).sum()
                if new_energy > 1e-30:
                    mag_new[1:] *= np.sqrt(old_energy / new_energy)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Clip and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"    SE iter {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")
    
    return best_a, best_obj


def peak_reduction(a, n_iter=2000, verbose=False):
    """Targeted peak reduction: directly minimize the max convolution value."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > 300:  # 5 min timeout
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find the top few peaks
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Compute weighted gradient (subgradient of max)
        pv = conv[peaks]
        # Use softmax weighting
        scaled = 10.0 * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]
        
        # Center gradient
        grad_centered = grad - grad.mean()
        
        # Adaptive step size
        step = 0.02 / (1.0 + no_improve * 0.01)
        
        a_new = project_simplex(a - step * grad_centered, target_sum=1.0)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Restart with small perturbation

                a = project_simplex(best_a + rng.standard_normal(n) * 0.003, target_sum=1.0)

        if verbose and it % 200 == 0:

def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj

def cosine_init(n, freq, rng):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
