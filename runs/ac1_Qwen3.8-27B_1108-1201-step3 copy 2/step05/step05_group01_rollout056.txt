Here's a concrete algorithm I'll implement:

**Algorithm: Iterative Convolution Shaping (ICS)**

1. Initialize a from the parent with some perturbation.
2. Repeat:
   a. Compute c = conv(a, a) (zero-padded to length N for FFT).
   b. Compute the "excess" e = c - target, where target is the flat level 1/(2n-1).
   c. In the frequency domain: E = FFT(e), and we want E to be small.
   d. Since c = IFFT(A^2), we have A^2 = FFT(c). We want to modify A^2 to reduce the excess.
   e. Update: A^2_new = A^2 - lambda * FFT(excess_clipped)
      where excess_clipped is the positive part of (c - threshold) for some threshold.
   f. A_new = sqrt(A^2_new) (with appropriate phase handling)
   g. a_new = IFFT(A_new)
   h. Project a_new to be non-negative (and renormalize).
3. Output the best a found.

The tricky part is step (f): taking the square root of a complex number while maintaining conjugate symmetry.

For a real sequence a, we have A[N-k] = conj(A[k]). So A^2[N-k] = conj(A^2[k]).
When we compute A^2_new, we need to maintain this property.
Then A_new[k] = sqrt(A^2_new[k]). For conjugate symmetry: A_new[N-k] should = conj(A_new[k]).
If A^2_new[N-k] = conj(A^2_new[k]), then sqrt(A^2_new[N-k]) = sqrt(conj(A^2_new[k])) = conj(sqrt(A^2_new[k])) (taking the principal square root). So conjugate symmetry is automatically maintained!

Great, so the algorithm is:
1. A = FFT_N(a) (padded to size N)
2. C = A^2 (element-wise)
3. c = IFFT_N(C) (real, since C has conjugate symmetry)
4. Compute the excess: e = max(c[:2n-1] - threshold, 0)
5. Extend e to length N (pad with zeros)
6. E = FFT_N(e)
7. C_new = C - lambda * E
8. Ensure C_new has conjugate symmetry (it should, by construction)
9. A_new = sqrt(C_new) (element-wise complex square root)
10. a_new = real(IFFT_N(A_new))
11. Project: a_new = max(a_new, 0), renormalize
12. Check if a_new has significant mass in the padding region; if so, zero it out and renormalize.
13. Update a = a_new, go to step 2.

The parameter lambda controls the step size. The threshold controls which peaks we're trying to reduce.

This is essentially a "projection" method in the frequency domain, similar to the Gerchberg-Saxton algorithm used in X-ray crystallography and optics.

    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)

    N = 1 << (2 * n - 1 + 1).bit_length()  # Power of 2 >= 2n
    # Actually we need N >= 2n-1 for the linear convolution
    N = 1 << (2 * n).bit_length()  # Slightly larger for safety
    
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:

        C = A * A
        c = np.fft.ifft(C).real[:2*n-1]
        
        current_max = c.max()
        if current_max < best_obj:
            best_obj = current_max
            best_a = a.copy()
        
        # Compute excess
        threshold = current_max * 0.95  # Reduce peaks above 95% of max
        excess = np.maximum(c - threshold, 0)
        excess_full = np.zeros(N)
        excess_full[:2*n-1] = excess
        
        E = np.fft.fft(excess_full)
        C_new = C - lambda_lr * E

        # Take square root
        A_new = np.sqrt(C_new)  # Complex square root
        
        # Compute new a
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]  # Keep only the first n elements

        s_new = a_new.sum()

        lambda_lr = max(lambda_lr * 0.999, 0.001)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Iterative Convolution Shaping (ICS)
in the frequency domain, combined with asymmetric LP-FW and peak-directed methods.
Key innovation: exploits the non-symmetry of auto-convolution (unlike autocorrelation)
by directly shaping FFT(a)^2 in frequency domain via iterative projected updates.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.08, 5.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 80, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def ics_optimize(a0, time_limit, verbose=False):
    """
    Iterative Convolution Shaping: works in frequency domain to flatten
    the auto-convolution by modifying FFT(a)^2 directly.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    # Choose FFT size (power of 2, >= 2n-1)
    N = 1 << (2 * n).bit_length()

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.fft(a, N)
        C = A * A
        c_full = np.fft.ifft(C).real
        c = c_full[:2 * n - 1]

        current_max = c.max()
        current_min_flat = 1.0 / (2 * n - 1)

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    ICS iter {it}: max_conv={best_obj:.10f}, obj={2*n*best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            lr = max(lr * 0.5, 0.05)
            continue

        # Compute excess above threshold
        threshold = current_max - 0.02 * (current_max - current_min_flat)
        excess = np.maximum(c - threshold, 0.0)

        # Build full-length excess vector
        excess_full = np.zeros(N)
        excess_full[:2 * n - 1] = excess

        # Frequency domain update
        E = np.fft.fft(excess_full)
        C_new = C - lr * E

        # Complex square root with sign fixing
        A_new = np.sqrt(C_new + 0j)

        # Fix signs to be close to old A
        dots = np.real(A_new * np.conj(A))
        neg_mask = dots < 0
        A_new[neg_mask] *= -1

        # New time domain
        a_new = np.fft.ifft(A_new).real
        a_trunc = a_new[:n]

        # Project: non-negative, normalize
        a_trunc = np.maximum(a_trunc, 0.0)
        s_new = a_trunc.sum()
        if s_new < 1e-10:
            break
        a_trunc = a_trunc / s_new

        a = a_trunc
        lr = max(lr * 0.9995, 0.05)

    return best_a, best_obj


def peak_directed_descent(a0, time_limit, verbose=False):
    """
    Directly reduce the current peak of the auto-convolution using
    the subgradient of max(conv(a,a)) with respect to a.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.01
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Weighted peaks (softmax over top values)
        conv_max = conv.max()
        flat_level = 1.0 / (2 * n - 1)
        temp = max(0.01 * (conv_max - flat_level), 1e-6)
        shifted = (conv - conv_max) / temp
        w = np.exp(shifted)
        w = w / w.sum()

        # Subgradient: for each weighted peak position k, grad contribution is 2*a[k-j]
        sg = np.zeros(n)
        # Use FFT-based cross-correlation for efficiency
        N_fft = 1 << (3 * n - 2).bit_length()
        A = np.fft.rfft(a, N_fft)
        W = np.fft.rfft(w, N_fft)
        R = np.fft.irfft(W * np.conj(A), N_fft)[:n]
        sg = 2.0 * R

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and best_obj < 0.76:
                print(f"    PDD: max_conv={best_obj:.10f}, obj={2*n*best_obj:.10f}")
        else:
            no_improve += 1
            lr *= 0.98
            if no_improve > 300:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.008)
                v = np.zeros(n)
                lr = 0.008
                no_improve = 0

        lr = max(lr, 0.0005)

    return best_a, best_obj


def spectral_phase_optimize(a0, time_limit, verbose=False):
    """
    Optimize by directly manipulating the phase of FFT(a) to flatten
    FFT(a)^2 in a controlled way, using error reduction in frequency domain.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    N = 1 << (2 * n).bit_length()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = 0.3
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.fft(a, N)
        C = A * A
        c = np.fft.ifft(C).real[:2 * n - 1]

        current_max = c.max()
        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
            if verbose and best_obj < 0.76:
                print(f"    SPO: max_conv={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(no_improve * 7 + 1)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            no_improve = 0
            alpha = max(alpha * 0.7, 0.05)
            continue

        # Target: make c flat at level 1/(2n-1)
        target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        C_target_flat = np.fft.fft(np.concatenate([target, np.zeros(N - 2 * n + 1)]))

        # Blend: move C toward flat target
        C_blended = (1 - alpha) * C + alpha * C_target_flat

        # Ensure C_blended is a valid "squared spectrum" (non-negative real part at k=0)
        # Take square root
        A_new = np.sqrt(C_blended + 0j)

        # Fix signs relative to old A
        dots = np.real(A_new * np.conj(A))
        neg_mask = dots < 0
        A_new[neg_mask] *= -1

        # Enforce conjugate symmetry explicitly
        for k in range(1, N // 2 + 1):
            A_new[N - k] = np.conj(A_new[k])
        if N % 2 == 0:
            A_new[N // 2] = np.real(A_new[N // 2])

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

    return best_a, best_obj


def asymmetric_perturb(base, rng, mode="linear"):
    """Create an asymmetric perturbation of the base sequence."""
    n = len(base)
    if mode == "linear":
        w = np.linspace(0.5, 1.5, n)
    elif mode == "quad":
        w = 0.5 + np.linspace(0, 1, n) ** 2
    elif mode == "exp":
        w = np.exp(np.linspace(-0.5, 0.5, n))
    elif mode == "bump":
        center = rng.uniform(0.3, 0.7) * n
        w = 1.0 + 0.3 * np.exp(-((np.arange(n) - center) / (n * 0.2)) ** 2)
    elif mode == "two_bump":
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w = 1.0 + 0.4 * (np.exp(-((np.arange(n) - c1) / (n * 0.15)) ** 2) +
                          np.exp(-((np.arange(n) - c2) / (n * 0.15)) ** 2) * 0.7)
    else:
        w = np.ones(n)

    a = base * w
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            try_seq(a_opt, "combined_deep")

    # === Phase 7: Final polishing (remaining) ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick ICS refinement
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_opt, _ = ics_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "final_ics")

        # Quick PDD
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = peak_directed_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_pdd")

        # LP-FW short
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
