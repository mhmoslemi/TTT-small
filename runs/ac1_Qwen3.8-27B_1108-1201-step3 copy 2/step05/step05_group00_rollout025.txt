def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        inits.append(a / a.sum())
    
    # Triangular (peak at different positions)
    for peak_frac in [0.3, 0.5, 0.7]:
        peak = int(n * peak_frac)
        a = 1 - np.abs(np.arange(n) - peak) / n
        a = np.maximum(a, 0)
        inits.append(a / a.sum())
    
    # Sinusoidal modulation
    for freq in [1, 2, 3]:
        a = 1 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0)
        inits.append(a / a.sum())
    
    # Cosine window (Hann)
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(a / a.sum())
    
    # Raised cosine
    a = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))
    inits.append(a / a.sum())
    
    # Bump function (flat top)
    for width in [0.3, 0.5, 0.7]:
        center = n / 2
        half = n * width / 2
        t = np.abs(np.arange(n) - center) / half
        a = np.where(t < 1, (1 - t**4), 0)
        inits.append(a / a.sum())
    
    # Random
    for _ in range(5):
        a = rng.random(n)
        inits.append(a / a.sum())
    
    # Power law
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = np.power(np.arange(n) + 1, -p)
        inits.append(a / a.sum())
    
    return inits

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False, verbose_freq=20):
    """LP-based Frank-Wolfe with adaptive M, momentum, and diversity."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    momentum = 0.0
    prev_direction = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Solve LP with current M (or expanded M if stuck)
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                rng = np.random.default_rng(it * 13 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005 * min(m_exp, 3))
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                momentum = 0.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                noise_scale = 0.003 + 0.002 * (no_improve - 80) / 20
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                momentum = 0.0
            continue
        else:
            if np.max(np.abs(g - f)) > 1e-6:
                no_improve = 0

        # Compute direction with momentum
        direction = g - f
        if prev_direction is not None and momentum > 0:
            direction = (1 - momentum) * direction + momentum * prev_direction
        prev_direction = direction

        # Line search
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine search
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * direction
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * direction
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            momentum = min(momentum * 0.95 + 0.05, 0.3)  # Slowly build momentum

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % verbose_freq == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
            momentum *= 0.9

        if no_improve > 150:
            break

    return best_f, best_obj

def subgradient_descent(f0, time_limit, verbose=False, verbose_freq=50):
    """Subgradient descent with top-k peaks and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    velocity = np.zeros(n)  # Momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        k_top = min(15, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        # Weight by value (softer than max)
        w = pv ** 2
        wsum = w.sum()
        if wsum > 0:
            w = w / wsum
        else:
            w = np.ones(len(peaks)) / len(peaks)

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Update with momentum
        velocity = 0.9 * velocity - lr * sg
        f_new = f + velocity
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss
        
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % verbose_freq == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f} (t={time.monotonic()-t0:.1f}s)")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 0 and no_improve % 50 == 0:
            lr *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.01, 0)
            f = f / f.sum()
            lr = 0.01
            velocity = np.zeros(n)
            no_improve = 0

        if lr < 1e-8:
            lr = 0.005
            no_improve = 0
            if no_improve > 500:
                break

    return best_f, best_obj

def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with structured moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    current_obj = best_obj

    t0 = time.monotonic()
    T = 0.001  # Initial temperature
    T_min = 1e-8
    cooling = 0.999
    rng = np.random.default_rng(12345)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        T *= cooling

        # Generate a structured perturbation
        move_type = rng.integers(0, 4)
        
        trial = f.copy()
        
        if move_type == 0:
            # Pair swap: exchange mass between two random positions
            i, j = rng.integers(0, n, size=2)
            delta = trial[i] - trial[j]
            trial[i] = trial[j]
            trial[j] = trial[i]
        elif move_type == 1:
            # Block swap: exchange two contiguous blocks
            i = rng.integers(0, n - 10)
            j = rng.integers(i + 10, n - 5)
            length = rng.integers(3, 10)
            j = min(j, n - length - 1)
            block_i = trial[i:i+length].copy()
            block_j = trial[j:j+length].copy()
            trial[i:i+length] = block_j
            trial[j:j+length] = block_i
        elif move_type == 2:
            # Gaussian noise
            noise = rng.standard_normal(n) * 0.01 * T * 100
            trial = f + noise
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
        else:
            # Local shuffle: pick a window and randomize
            center = rng.integers(10, n - 10)
            window = min(20, n // 4)
            lo = max(0, center - window)
            hi = min(n - 1, center + window)
            trial[lo:hi+1] = rng.dirichlet(np.ones(hi - lo + 1)) * f[lo:hi+1].sum()
        
        trial = np.maximum(trial, 0)
        ss = trial.sum()
        if ss < 1e-15:
            continue
        trial = trial / ss

        trial_conv = fast_convolve(trial)
        trial_obj = trial_conv.max()

        delta = trial_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            f = trial
            current_obj = trial_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 1000 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f} (t={time.monotonic()-t0:.1f}s)")
        else:
            no_improve += 1

        if no_improve > 5000:
            break

    return best_f, best_obj

def frequency_flat_init(n, rng):
    """
    Try to initialize a sequence whose FFT magnitude is approximately flat.
    This is done by iteratively projecting onto the non-negative constraint
    while trying to flatten the spectrum.
    """
    a = rng.random(n)
    a = a / a.sum()
    
    for _ in range(200):
        A = np.fft.fft(a)
        phases = np.angle(A)
        # Target: flat magnitude
        target_mag = np.ones(n)
        # Inverse FFT to get "target" time-domain signal
        a_target = np.fft.ifft(target_mag * phases).real
        a_target = np.maximum(a_target, 0)
        ss = a_target.sum()
        if ss < 1e-15:
            break
        a_target = a_target / ss
        # Blend
        a = 0.5 * a + 0.5 * a_target
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.arange(n) - n / 2
    for sf in [0.12, 0.18, 0.22, 0.28, 0.35]:
        a = np.exp(-0.5 * (x / (n * sf)) ** 2)
        a = np.maximum(a, 0)
        inits.append(a / a.sum())
    
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-rate * np.arange(n) / n)
        inits.append(a / a.sum())
        a = np.exp(-rate * (n - 1 - np.arange(n)) / n)
        inits.append(a / a.sum())
    
    for pf in [0.25, 0.35, 0.5, 0.65, 0.75]:
        peak = int(n * pf)
        a = 1 - np.abs(np.arange(n) - peak) / n
        a = np.maximum(a, 0)
        inits.append(a / a.sum())
    
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-10:
                inits.append(a / ss)
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = np.power(np.arange(n) + 1, -p)
        inits.append(a / a.sum())
        a = np.power(n - np.arange(n), -p)
        inits.append(a / a.sum())
    
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(a / a.sum())
    
    for width in [0.25, 0.4, 0.6, 0.8]:
        center = n / 2
        half = n * width / 2
        t = np.abs(np.arange(n) - center) / max(half, 1)
        a = np.where(t < 1, (1 - t**6), 0)
        ss = a.sum()
        if ss > 1e-10:
            inits.append(a / ss)
    
    for _ in range(8):
        a = rng.random(n)
        inits.append(a / a.sum())
    
    return inits


def frequency_flat_init(n, rng, iterations=300):
    """Initialize by iteratively flattening the frequency spectrum."""
    a = rng.random(n)
    a = a / a.sum()
    
    for _ in range(iterations):
        A = np.fft.fft(a)
        mag = np.abs(A)
        phases = np.angle(A)
        # Target: flat magnitude (all ones)
        target_mag = np.full(n, 1.0)
        a_target = np.fft.ifft(target_mag * phases).real
        a_target = np.maximum(a_target, 0)
        ss = a_target.sum()
        if ss < 1e-15:
            # Try with modified target
            target_mag = 1.0 + 0.1 * np.cos(np.arange(n) * 0.1)
            a_target = np.fft.ifft(target_mag * phases).real
            a_target = np.maximum(a_target, 0)
            ss = a_target.sum()
            if ss < 1e-15:
                break
        a_target = a_target / ss
        a = 0.3 * a + 0.7 * a_target
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / ss
    
    return a


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, verbose_freq=20):
    """LP-based Frank-Wolfe with adaptive M, momentum, and diversity."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    momentum = 0.0
    prev_direction = None
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.12
            lp_fail_count += 1
            if m_exp > 6.0 or lp_fail_count > 30:
                rng_local = np.random.default_rng(it * 17 + 3)
                noise_scale = 0.004 * min(m_exp, 3)
                f = project_simplex(best_f + rng_local.standard_normal(n) * noise_scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                lp_fail_count = 0
                momentum = 0.0
            continue

        lp_fail_count = 0
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 15:
                m_exp = min(m_exp * 1.06, 5.0)
            if no_improve > 60:
                rng_local = np.random.default_rng(it * 31 + 7)
                noise_scale = 0.003 + 0.003 * min((no_improve - 60) / 30, 2.0)
                f = project_simplex(best_f + rng_local.standard_normal(n) * noise_scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                momentum = 0.0
                prev_direction = None
            continue
        else:
            if np.max(np.abs(g - f)) > 1e-6:
                no_improve = max(0, no_improve - 1)

        direction = g - f
        if prev_direction is not None and momentum > 0.01:
            direction = (1 - momentum) * direction + momentum * prev_direction
        prev_direction = direction.copy()

        best_alpha = 0.0
        best_val = M

        alphas = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * direction
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * direction
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = max(1.0, m_exp * 0.95)
            momentum = min(momentum * 0.97 + 0.03, 0.25)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % verbose_freq == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f} (t={elapsed:.1f}s)")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)
            momentum *= 0.9

        if no_improve > 200:
            break

    return best_f, best_obj


def subgradient_descent(f0, time_limit, verbose=False, verbose_freq=50):
    """Subgradient descent with top-k peaks, momentum, and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0
    it = 0
    velocity = np.zeros(n)
    beta = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_top = min(20, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        w = pv ** 3
        wsum = w.sum()
        if wsum > 1e-15:
            w = w / wsum
        else:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        velocity = beta * velocity - lr * sg
        f_new = f + velocity
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / ss
        
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % verbose_freq == 0:
                print(f"    SGD iter {it}: obj={best_obj:.10f} (t={time.monotonic()-t0:.1f}s)")
        else:
            no_improve += 1

        if no_improve > 0 and no_improve % 40 == 0:
            lr *= 0.85
        
        if no_improve > 250:
            rng_local = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng_local.standard_normal(n) * 0.008, 0)
            f = f / f.sum()
            lr = 0.008 * (0.5 + 0.5 * rng_local.random())
            velocity = np.zeros(n)
            no_improve = 0
            beta = 0.8 + 0.1 * rng_local.random()

        if lr < 1e-9:
            break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with structured moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    current_obj = best_obj

    t0 = time.monotonic()
    T = 0.0005
    T_min = 1e-9
    cooling = 0.9995
    rng = np.random.default_rng(99999)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        T *= cooling

        move_type = rng.integers(0, 5)
        trial = f.copy()

        if move_type == 0:
            i, j = rng.integers(0, n, size=2)
            trial[i], trial[j] = trial[j], trial[i]
        elif move_type == 1:
            i = rng.integers(0, max(1, n - 15))
            blen = rng.integers(3, min(15, n // 3))
            j = rng.integers(i + blen + 2, max(i + blen + 3, n - blen))
            if j + blen <= n:
                bi = trial[i:i+blen].copy()
                bj = trial[j:j+blen].copy()
                trial[i:i+blen] = bj
                trial[j:j+blen] = bi
        elif move_type == 2:
            noise = rng.standard_normal(n) * 0.005
            trial = f + noise
        elif move_type == 3:
            center = rng.integers(5, n - 5)
            w = min(15, n // 5)
            lo = max(0, center - w)
            hi = min(n - 1, center + w)
            mass = f[lo:hi+1].sum()
            if mass > 1e-10:
                trial[lo:hi+1] = rng.dirichlet(np.ones(hi - lo + 1)) * mass
        else:
            i = rng.integers(0, n - 1)
            delta = (trial[i] - trial[i+1]) * rng.uniform(0.5, 1.5)
            trial[i] -= delta
            trial[i+1] += delta

        trial = np.maximum(trial, 0)
        ss = trial.sum()
        if ss < 1e-15:
            continue
        trial = trial / ss

        trial_obj = fast_convolve(trial).max()
        delta = trial_obj - current_obj
        
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            f = trial
            current_obj = trial_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 2000 == 0:
                    print(f"    SA iter {it}: obj={best_obj:.10f} (t={time.monotonic()-t0:.1f}s)")
        else:
            no_improve += 1

        if no_improve > 8000:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = max(n * sigma_frac, 1.0)
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def evaluate_score(a):
    """Compute the objective score for a sequence (normalized to sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    conv = fast_convolve(a_norm)
    return float(2 * n * conv.max())
