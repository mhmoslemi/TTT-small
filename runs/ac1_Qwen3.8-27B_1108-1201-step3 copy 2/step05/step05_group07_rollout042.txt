```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L-BFGS with soft-max in log-space for global optimization
2. Targeted peak equalization using SLSQP on active peaks

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

# Global variables for optimization parameters (to avoid closures)
_global_T = 10.0
_global_n = 0
_global_deadline = 0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]

def soft_max_obj(z):
    """Soft-max objective for L-BFGS (uses global T and n)."""
    global _global_T, _global_n, _global_deadline
    if time.monotonic() > _global_deadline:
        return 1e15
    T = _global_T
    n = _global_n
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp

def soft_max_grad(z):
    """Soft-max gradient for L-BFGS (uses global T and n)."""
    global _global_T, _global_n, _global_deadline
    if time.monotonic() > _global_deadline:
        return np.zeros_like(z)
    T = _global_T
    n = _global_n
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return grad_z
```

```python
    # Chain rule for softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return grad_z
```

This is cleaner. Now I can use L-BFGS-B (unconstrained) on z.

But there's a subtlety: the "top k" set can change during optimization, making the objective non-smooth. In practice, this is usually fine because the peaks don't change ordering very quickly.

Let me also think about the coordinate descent approach. The idea is:
- For each element a[j], try changing it to a[j] * multiplier
- Redistribute the difference to other elements proportionally
- Accept if the max convolution decreases

This is O(n * num_multipliers * conv_cost) per full pass. For n=768 and conv_cost ≈ O(n log n), each pass is about 768 * 6 * (768 * 10) ≈ 3.5M operations. Should be fast enough.

Let me now write the complete solution:

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. L-BFGS with soft-max in log-space (global exploration)
2. Targeted peak equalization in log-space (local refinement)
3. Coordinate descent (fine-tuning)
4. Multi-length exploration and random restarts

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

_g_T = 10.0
_g_n = 512
_g_deadline = 0.0
_g_top_k = 20
_g_peq_T = 50.0

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]

def _obj_func(z):
    if time.monotonic() > _g_deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * _g_n / _g_T) * (max_Tb + np.log(exp_Tb.sum()))

def _grad_func(z):
    if time.monotonic() > _g_deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * _g_n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    return a * (grad_a - a * S)

def _peq_obj(z):
    if time.monotonic() > _g_deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    k = _g_top_k
    sorted_idx = np.argsort(b)[::-1][:k]
    Tb = _g_peq_T * b[sorted_idx]
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (1.0 / _g_peq_T) * (max_Tb + np.log(exp_Tb.sum()))

def _peq_grad(z):
    if time.monotonic() > _g_deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    n = len(a)
    k = _g_top_k
    sorted_idx = np.argsort(b)[::-1][:k]
    Tb = _g_peq_T * b[sorted_idx]
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for i in range(len(sorted_idx)):
        if w[i] < 1e-15:
            continue
        k_val = int(sorted_idx[i])
        js = max(0, k_val - n + 1)
        je = min(n - 1, k_val)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[i] * a[k_val - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - a * S)

def run_lbgfs_obj(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS for the full soft-max objective."""
    global _g_T, _g_n, _g_deadline
    _g_T = T
    _g_n = n
    _g_deadline = time.monotonic() + time_limit

                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': maxiter * 25})
        return res.x, res.fun
    except Exception:
        return z0, 1e10

def run_peq(z0, top_k, peq_T, time_limit, maxiter=2000):

def coordinate_descent(a0, time_limit=30.0):
    """Coordinate descent with proportion redistribution."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 1.7, 2.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                # Try small positive values
                for new_val in [0.001, 0.01, 0.05]:
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-14:
                        continue
                    a_trial[mask] = a_trial[mask] / others_sum * (1.0 - new_val)
                    val = fast_convolve(a_trial).max()
                    if val < best_max - 1e-14:
                        a = a_trial
                        best_max = val
                        best_a = a.copy()
                        improved = True
                continue
            
            best_j_max = best_max
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                others_sum = 1.0 - new_val
                if others_sum < 1e-10:
                    continue
                scale = others_sum / (1.0 - old_val) if (1.0 - old_val) > 1e-10 else 1.0
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] = a_trial[mask] * scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_max - 1e-14:
                    best_j_max = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                others_sum = 1.0 - best_j_val
                scale = others_sum / (1.0 - old_val)
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = a[mask] * scale
                a = a / a.sum()
                if best_j_max < best_max - 1e-14:
                    best_max = best_j_max
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_max

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.998
        cur_temp *= 1.005
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-initialization with various window functions
2. L-BFGS with soft-max in log-space (global optimization)
3. Targeted peak equalization in log-space (precise refinement)
4. Symmetrization exploits (reversal invariance)
5. Coordinate descent for fine-tuning
6. Multi-length exploration and random restarts
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

# Global state for optimization callbacks
_g_T = 10.0
_g_n = 512
_g_deadline = 0.0
_g_top_k = 20
_g_peq_T = 50.0
_g_best_seq = None
_g_best_score = float('inf')


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr(w, a):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def symmetrize(a):
    """Symmetrize: (a + reverse(a)) / 2, then normalize. Safe due to reversal invariance."""
    s = a + a[::-1]
    s = s / s.sum()
    return np.maximum(s, 0)


def try_seq(a, label):
    """Evaluate a sequence and update global best."""
    global _g_best_seq, _g_best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < _g_best_score:
        _g_best_score = score
        _g_best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def _obj_func(z):
    if time.monotonic() > _g_deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * _g_n / _g_T) * (max_Tb + np.log(exp_Tb.sum()))


def _grad_func(z):
    if time.monotonic() > _g_deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * _g_n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    return a * (grad_a - a * S)


def _peq_obj(z):
    if time.monotonic() > _g_deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    k = _g_top_k
    sorted_idx = np.argsort(b)[::-1][:k]
    Tb = _g_peq_T * b[sorted_idx]
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (1.0 / _g_peq_T) * (max_Tb + np.log(exp_Tb.sum()))


def _peq_grad(z):
    if time.monotonic() > _g_deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    n = len(a)
    k = _g_top_k
    sorted_idx = np.argsort(b)[::-1][:k]
    Tb = _g_peq_T * b[sorted_idx]
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for i in range(len(sorted_idx)):
        if w[i] < 1e-15:
            continue
        k_val = int(sorted_idx[i])
        js = max(0, k_val - n + 1)
        je = min(n - 1, k_val)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[i] * a[k_val - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - a * S)


def run_lbgfs_obj(z0, T, n, time_limit, maxiter=3000):
    global _g_T, _g_n, _g_deadline
    _g_T = T
    _g_n = n
    _g_deadline = time.monotonic() + time_limit
    try:
        res = minimize(_obj_func, z0, jac=_grad_func, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': maxiter * 25})
        return res.x, res.fun
    except Exception:
        return z0, 1e10


def run_peq(z0, top_k, peq_T, time_limit, maxiter=2000):
    global _g_T, _g_n, _g_deadline, _g_top_k, _g_peq_T
    _g_top_k = top_k
    _g_peq_T = peq_T
    _g_n = len(z0)
    _g_deadline = time.monotonic() + time_limit
    try:
        res = minimize(_peq_obj, z0, jac=_peq_grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': maxiter * 25})
        return res.x, res.fun
    except Exception:
        return z0, 1e10


def coordinate_descent(a0, time_limit=30.0):
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()
    multipliers = np.array([0.4, 0.6, 0.75, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                for new_val in [0.001, 0.005, 0.02]:
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-14:
                        continue
                    a_trial[mask] = a_trial[mask] / others_sum * (1.0 - new_val)
                    val = fast_convolve(a_trial).max()
                    if val < best_max - 1e-14:
                        a = a_trial
                        best_max = val
                        best_a = a.copy()
                        improved = True
                continue

            best_j_max = best_max
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                others_target = 1.0 - new_val
                others_current = 1.0 - old_val
                if others_current < 1e-12:
                    continue
                scale = others_target / others_current
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] = a_trial[mask] * scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_max - 1e-14:
                    best_j_max = val
                    best_j_val = new_val

            if best_j_val != old_val:
                others_target = 1.0 - best_j_val
                others_current = 1.0 - old_val
                if others_current > 1e-12:
                    scale = others_target / others_current
                else:
                    scale = 1.0
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] = a[mask] * scale
                a = a / a.sum()
                if best_j_max < best_max - 1e-14:
                    best_max = best_j_max
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_max


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_polish(a0, time_limit=60.0, lr=0.005, mom=0.9, top_k=30):
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001
    rng_local = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = b.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.998
        cur_temp *= 1.005

        if no_improve > 400:
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_a, best_obj


def peak_shave(a0, n_peaks=8, n_steps=200, time_limit=30.0):
    """Iteratively reduce the highest peaks by adjusting contributing elements."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -1
        b_inner[-1] = -1
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = [k for k in peak_indices if b[k] > best_obj * 0.92]
        if not peak_indices:
            break

        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            max_local = np.argmax(contributions)
            i = idx[max_local]
            j = k - i

            for target in [i, j]:
                old_val = a[target]
                if old_val < 1e-10:
                    continue
                for factor in [0.85, 0.9, 0.95]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[target] = new_val
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-12:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    global _g_best_seq, _g_best_score, _g_T, _g_n, _g_deadline, _g_top_k, _g_peq_T

    _g_best_seq = None
    _g_best_score = float('inf')

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # Generate diverse starting points
    starts = [parent.copy()]

    # Symmetrized parent
    starts.append(symmetrize(parent))

    # Various window functions at parent length
    x = np.linspace(0, 1, n_parent)
    for name, fn in [
        ("hann", 0.5 * (1 - np.cos(2 * np.pi * x))),
        ("raised_cos", 0.5 * (1 + np.cos(2 * np.pi * x))),
        ("parabolic", np.maximum(1 - (2 * x - 1) ** 2, 0)),
        ("quartic", np.maximum(1 - (2 * x - 1) ** 4, 0)),
        ("super_gauss", np.exp(-4 * (2 * x - 1) ** 2)),
    ]:
        w = fn.copy()
        ws = w.sum()
        if ws > 1e-10:
            starts.append(w / ws)

    # Sparse edge version (zero out 10% at each edge)
    edge_frac = 0.1
    a_sparse = parent.copy()
    n_pad = int(n_parent * edge_frac)
    a_sparse[:n_pad] = 0
    a_sparse[-n_pad:] = 0
    a_sparse = symmetrize(a_sparse)
    starts.append(a_sparse)

    # Evaluate all starts
    start_scores = []
    for i, st in enumerate(starts):
        st = np.maximum(st, 0)
        ss = st.sum()
        if ss < 1e-10:
            continue
        st = st / ss
        score = try_seq(st, f"start_{i}")
        if np.isfinite(score):
            start_scores.append((score, st))

    start_scores.sort(key=lambda t: t[0])

    # === Phase 1: L-BFGS on best start(s) at parent length ===
    print(f"\n=== Phase 1: L-BFGS at n={n_parent} ===")
    best_start = start_scores[0][1] if start_scores else parent
    z_curr = z_from_a(best_start)

    temps = [5, 10, 20, 50, 100, 200]
    for T in temps:
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_alloc = min(rem / max(1, len(temps) - temps.index(T)), 100.0)
        if t_alloc < 10:
            break
        print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
        z_curr, _ = run_lbgfs_obj(z_curr, T, n_parent, t_alloc, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        a_curr = symmetrize(a_curr)
        try_seq(a_curr, f"lbfgs_T{T}")
        # Also try non-symmetrized
        try_seq(softmax_from_z(z_curr), f"lbfgs_T{T}_ns")

    # Also try the second-best start
    if len(start_scores) > 1:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            z2 = z_from_a(start_scores[1][1])
            t_alloc = min(rem * 0.3, 50.0)
            z2, _ = run_lbgfs_obj(z2, 20, n_parent, t_alloc, maxiter=1500)
            a2 = symmetrize(softmax_from_z(z2))
            try_seq(a2, "second_start_lbfgs")

    # === Phase 2: Peak equalization (the key refinement) ===
    print(f"\n=== Phase 2: Peak equalization (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        n_best = len(_g_best_seq)
        z_peq = z_from_a(_g_best_seq)

        # Sweep over top_k and T values
        peq_configs = [
            (10, 100), (20, 100), (30, 100), (50, 100), (100, 100),
            (10, 200), (20, 200), (50, 200), (100, 200),
            (20, 500), (50, 500), (50, 1000),
        ]

        for top_k, T_peq in peq_configs:
            rem = deadline - time.monotonic() - 250
            if rem < 10:
                break
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 8:
                break
            print(f"  PEQ k={top_k}, T={T_peq}, t={t_alloc:.0f}s")
            z_new, _ = run_peq(z_peq, top_k, float(T_peq), t_alloc, maxiter=1500)
            a_new = softmax_from_z(z_new)
            try_seq(a_new, f"peq_k{top_k}_T{T_peq}")
            a_new_s = symmetrize(a_new)
            try_seq(a_new_s, f"peq_k{top_k}_T{T_peq}_sym")
            if _g_best_seq is not None:
                z_peq = z_from_a(_g_best_seq)

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_ps = min(rem * 0.4, 50.0)
            a_ps, _ = peak_shave(_g_best_seq, n_peaks=8, n_steps=300, time_limit=t_ps)
            try_seq(a_ps, "peak_shave")
            a_ps_s = symmetrize(a_ps)
            try_seq(a_ps_s, "peak_shave_sym")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={_g_best_score:.6f}) ===")
    base = _g_best_seq if _g_best_seq is not None else parent
    for target_n in [n_parent * 2, n_parent * 2 + 128, n_parent * 3]:
        if target_n < 512 or target_n > 8192:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        f_interp = symmetrize(f_interp)
        z_new = z_from_a(f_interp)

        t_alloc = min(rem * 0.25, 70.0)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, L-BFGS T=10, t={t_alloc:.0f}s")
        z_new, _ = run_lbgfs_obj(z_new, 10, target_n, t_alloc, maxiter=1500)
        a_new = symmetrize(softmax_from_z(z_new))
        try_seq(a_new, f"len_{target_n}")

        # Peak equalization on new length
        rem = deadline - time.monotonic() - 150
        if rem > 20 and _g_best_seq is not None and len(_g_best_seq) == target_n:
            t_alloc2 = min(rem * 0.3, 50.0)
            if t_alloc2 > 10:
                z_eq, _ = run_peq(z_new, 30, 200.0, t_alloc2, maxiter=1500)
                a_eq = symmetrize(softmax_from_z(z_eq))
                try_seq(a_eq, f"len_{target_n}_peq")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem * 0.6, 45.0)
            a_cd, _ = coordinate_descent(_g_best_seq, time_limit=t_cd)
            try_seq(a_cd, "CD")
            a_cd_s = symmetrize(a_cd)
            try_seq(a_cd_s, "CD_sym")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            n_best = len(_g_best_seq)
            t_sg = min(rem * 0.5, 35.0)
            a_sg, _ = subgradient_polish(_g_best_seq, time_limit=t_sg, lr=0.003,
                                          mom=0.92, top_k=max(20, n_best // 8))
            try_seq(a_sg, "SG")

    # === Phase 7: Random perturbations + re-optimize ===
    print(f"\n=== Phase 7: Perturbations (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        n_best = len(_g_best_seq)
        for attempt in range(8):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            noise_scale = 0.0005 * (attempt + 1)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(_g_best_seq + noise)
            a_pert = symmetrize(a_pert)
            z_pert = z_from_a(a_pert)

            # Peak equalization on perturbed
            t_alloc = min(rem * 0.35, 20.0)
            if t_alloc < 5:
                break
            z_p, _ = run_peq(z_pert, 30, 500.0, t_alloc, maxiter=800)
            a_p = symmetrize(softmax_from_z(z_p))
            try_seq(a_p, f"pert_{attempt}")

            # Full L-BFGS on perturbed
            rem = deadline - time.monotonic() - 15
            if rem > 8:
                t_alloc2 = min(rem * 0.35, 12.0)
                if t_alloc2 > 5:
                    z_p2, _ = run_lbgfs_obj(z_pert, 100, n_best, t_alloc2, maxiter=500)
                    a_p2 = symmetrize(softmax_from_z(z_p2))
                    try_seq(a_p2, f"pert_{attempt}_lbfgs")

    # === Phase 8: One final intensive peak equalization ===
    print(f"\n=== Phase 8: Final PEQ (best={_g_best_score:.6f}) ===")
    if _g_best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            n_best = len(_g_best_seq)
            z_final = z_from_a(_g_best_seq)
            # Very high T to essentially minimize the exact max
            t_final = min(rem * 0.6, 30.0)
            z_final, _ = run_peq(z_final, 20, 2000.0, t_final, maxiter=1000)
            a_final = symmetrize(softmax_from_z(z_final))
            try_seq(a_final, "final_peq_T2000")

            # One more with different k
            rem = deadline - time.monotonic() - 5
            if rem > 8:
                z_f2 = z_from_a(_g_best_seq)
                t_f2 = min(rem * 0.7, 15.0)
                z_f2, _ = run_peq(z_f2, 5, 5000.0, t_f2, maxiter=500)
                a_f2 = symmetrize(softmax_from_z(z_f2))
                try_seq(a_f2, "final_peq_k5_T5000")

    # Fallback
    if _g_best_seq is None:
        _g_best_seq = parent.copy()
        _g_best_score = float(evaluate_sequence(_g_best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(_g_best_seq.tolist()))
    if np.isfinite(fs):
        _g_best_score = fs

    print(f"\nFINAL SCORE: {_g_best_score:.6f} (n={len(_g_best_seq)})")
    return [float(x) for x in _g_best_seq]
```
