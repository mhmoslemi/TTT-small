def exact_subgrad_optimize(a0, time_limit, lr=0.05, mom=0.9,
                            temp_init=0.01, temp_decay=0.99):
    """
    Exact subgradient descent with Nesterov momentum and adaptive step size.
    Works on the simplex directly (a >= 0, sum a = 1).
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    G = np.zeros(n)  # ADAGrad accumulator
    cur_mom = mom
    cur_temp = temp_init
    cur_lr = lr
    step = 0
    t0 = time.monotonic()
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        max_c = c.max()
        
        # Soft active set
        scaled = (c - max_c) / cur_temp
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        grad = compute_subgrad(w, a)
        
        # ADAGrad update with Nesterov momentum
        G += grad * grad
        lr_t = cur_lr / (np.sqrt(G) + 1e-6)
        v = cur_mom * v - lr_t * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        c_new = fast_convolve(a_new)
        max_new = c_new.max()
        
        if max_new < best_obj - 1e-12:
            best_obj = max_new
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1
        
        a = a_new
        step += 1
        cur_temp *= temp_decay
        cur_mom = min(0.999, cur_mom * 0.9999)
        cur_lr *= 0.9995
        
        # Restart if stale
        if stale > 200:
            # Perturb best and restart
            rng = np.random.default_rng(step)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            G = np.zeros(n)
            cur_lr = lr * 0.5
            stale = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent: for each position, try scaling its value and
    redistributing to others proportionally.
    """
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0])
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    improved_any = True
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = np.maximum(a[mask] * scale, 0)
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_any = True
    
    return best_a, best_obj


def make_cosine_window(n, alpha=1.0):
    """Create a raised-cosine (Hann-like) window normalized to sum=1."""
    i = np.arange(n)
    w = 0.5 * (1 + np.cos(np.pi * (2*i - (n-1)) / (n-1)))
    # Or use a generalized cosine
    w = np.cos(np.pi * (2*i / (n-1) - 1) / 2.0)**(2*alpha)
    w = np.maximum(w, 0)
    w = w / w.sum()
    return w


def make_gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window normalized to sum=1."""
    i = np.arange(n)
    center = (n-1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((i - center) / sigma)**2)
    w = w / w.sum()
    return w


def make_two_bump_window(n, frac=0.3, width=0.1):
    """Create a two-bump window with mass at two positions."""
    i = np.arange(n)
    c1 = frac * n

    sigma = width * n
    w = np.exp(-0.5 * ((i - c1) / sigma)**2) + np.exp(-0.5 * ((i - c2) / sigma)**2)
    w = w / w.sum()
    return w

    """Interpolate sequence to new length preserving sum."""
    old_n = len(a)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_pair(x, y):
    """Linear convolution of x and y via FFT."""
    size = len(x) + len(y) - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:size]


def compute_subgrad(w, a):
    """
    Compute subgradient contribution: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices).
    w has length 2n-1, a has length n. Returns array of length n.
    """
    n = len(a)
    b = a[::-1]
    conv = fast_convolve_pair(w, b)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Convert log-space z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-30))


def soft_obj_and_grad(z, T):
    """
    Soft-max smoothed objective and gradient w.r.t. z.
    a = softmax(z), obj = (2n/T) * log(sum exp(T*b_k)) where b = conv(a,a).
    Returns (obj, grad_z).
    """
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)

    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2n * sum_k w[k] * 2*a[k-j] = 4n * (cross-corr)_j
    # compute_subgrad returns 2*(cross-corr), so grad_a = 2n * compute_subgrad(w, a)
    grad_a = 2.0 * n * compute_subgrad(w, a)

    # Chain rule to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)

    return obj, grad_z


def _smooth_obj(z, T):
    """Objective function for L-BFGS (with deadline check)."""
    if time.monotonic() > _GLOBAL_DEADLINE:
        return 1e15
    obj, _ = soft_obj_and_grad(z, T)
    return obj


def _smooth_jac(z, T):
    """Jacobian for L-BFGS (with deadline check)."""
    if time.monotonic() > _GLOBAL_DEADLINE:
        return np.zeros(len(z))
    _, g = soft_obj_and_grad(z, T)
    return g


def lbgfs_optimize_smooth(z0, T, time_limit, maxiter=3000):
    """Run L-BFGS on the smooth soft-max objective with time limit."""
    global _GLOBAL_DEADLINE
    t0 = time.monotonic()
    _GLOBAL_DEADLINE = t0 + time_limit
    n = len(z0)

    try:
        res = minimize(
            _smooth_obj, z0, args=(T,),
            jac=_smooth_jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, _smooth_obj(res.x, T)
    except Exception:
        return z0, _smooth_obj(z0, T)


def exact_subgrad_optimize(a0, time_limit, lr_init=0.05, mom_init=0.88,
                           temp_init=0.008, temp_decay=0.992,
                           top_k=50, restart_interval=150):
    """
    Exact subgradient descent with Nesterov momentum, ADAGrad,
    and adaptive soft active set. Works on the simplex.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    G = np.zeros(n)
    cur_mom = mom_init
    cur_temp = temp_init
    cur_lr = lr_init
    step = 0
    stale = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        max_c = c.max()

        # Adaptive top-k active set
        k = min(top_k, 2 * n - 1)
        # Use soft weights over top-k peaks
        peak_vals = np.sort(c)[::-1][:k]
        # Set temperature so that peaks within a factor of the top are active
        cur_temp_eff = max(cur_temp, 1e-6)
        scaled = (c - max_c) / cur_temp_eff
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= ws

        # Subgradient
        grad = compute_subgrad(w, a)

        # ADAGrad + Nesterov momentum
        G += grad * grad
        lr_t = cur_lr / (np.sqrt(G) + 1e-8)
        v = cur_mom * v - lr_t * grad
        a_new = project_simplex(a + v)

        # Evaluate
        c_new = fast_convolve(a_new)
        max_new = c_new.max()

        if max_new < best_obj - 1e-12:
            best_obj = max_new
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1

        a = a_new
        step += 1
        cur_temp *= temp_decay
        cur_mom = min(0.999, cur_mom * 0.9998)
        cur_lr *= 0.9997

        # Periodic restart with perturbation
        if stale > restart_interval:
            rng_restart = np.random.default_rng(step * 7919 + 13)
            perturb_scale = 0.003 + 0.005 * (stale / restart_interval)
            a = project_simplex(best_a + rng_restart.standard_normal(n) * perturb_scale)
            v = np.zeros(n)
            G = np.zeros(n)
            cur_lr = lr_init * max(0.3, 1.0 - stale / (4.0 * restart_interval))
            stale = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent: scale one coordinate, redistribute to others."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)

    if multipliers is None:
        multipliers = np.array([0.4, 0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.6])

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Precompute which indices contribute to top peaks for prioritization
    c = fast_convolve(a)
    top_peaks = np.argsort(c)[::-1][:20]
    # Count how often each index appears in contributing pairs
    priority = np.zeros(n)
    for k in top_peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        priority[js:je + 1] += 1
    # Sort indices by priority (descending)
    order = np.argsort(priority)[::-1]

    pass_count = 0
    improved_any = True
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        pass_count += 1
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] = a_trial[mask] * scale
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] = np.maximum(a[mask] * scale, 0)
                    a_s = a.sum()
                    if a_s > 1e-15:
                        a = a / a_s

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_any = True

    return best_a, best_obj


def make_cosine_window(n, alpha=1.0):
    """Generalized raised-cosine window."""
    i = np.arange(n, dtype=np.float64)
    if n <= 1:
        return np.ones(1)
    x = 2.0 * i / (n - 1) - 1.0
    w = np.cos(np.pi * x / 2.0) ** (2.0 * alpha)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def make_gaussian_window(n, sigma_frac=0.25):
    """Gaussian window centered."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = max(1.0, sigma_frac * n)
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = w.sum()
    return w / s


def make_two_bump(n, frac, width):
    """Two-bump Gaussian mixture."""
    i = np.arange(n, dtype=np.float64)
    c1 = frac * n
    c2 = (1.0 - frac) * n
    sigma = max(1.0, width * n)
    w = np.exp(-0.5 * ((i - c1) / sigma) ** 2) + np.exp(-0.5 * ((i - c2) / sigma) ** 2)
    s = w.sum()
    return w / s


def make_smooth_random(n, seed):
    """Smooth random positive sequence."""
    rng = np.random.default_rng(seed)
    r = rng.standard_normal(n)
    kernel_len = max(3, n // 20)
    kernel = np.exp(-0.5 * (np.arange(kernel_len * 2 + 1) / (kernel_len * 0.7)) ** 2)
    r = np.convolve(r, kernel, 'same')
    r = np.maximum(r, 0)
    s = r.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return r / s


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp


def evaluate_wrapper(a, best_state):
    """Evaluate sequence and update best state in-place."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a_norm.copy()
        print(f"  NEW BEST: {score:.6f} (n={len(a_norm)})")
    return score
