    return best_a, best_score


def coord_descent_polish(a, time_limit=60, n_passes=100):
    """Coordinate descent with multi-scale perturbation for polishing."""
    n = len(a)
    a_curr = normalize(a.copy())
    best_a = a_curr.copy()
    best_score = compute_score(a_curr)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = a_curr[j]
            if old < 1e-14:
                continue

            tried = 0
            for delta_frac in [0.05, 0.1, 0.2, 0.3, -0.05, -0.1, -0.2, -0.3, 0.5, -0.5]:
                tried += 1
                new_val = max(0.0, old * (1.0 + delta_frac))
                new_a = a_curr.copy()
                new_a[j] = new_val
                new_a = normalize(new_a)
                sc = compute_score(new_a)
                if sc < best_score - 1e-12:
                    a_curr = new_a
                    best_score = sc
                    best_a = a_curr.copy()
                    improved = True
                    break

        if not improved and p > 3:
            break

    return best_a, best_score


def pair_swap_local(a, time_limit=40, n_trials=500):
    """Random pair transfer local search."""
    n = len(a)
    a_curr = normalize(a.copy())
    best_a = a_curr.copy()
    best_score = compute_score(a_curr)
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        i, j = rng.integers(0, n, 2)
        if i == j:
            continue

        a_j = a_curr[j]
        a_i = a_curr[i]
        if a_j < 1e-13:
            continue

        best_delta = 0.0
        best_sc = best_score

        for f in [0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = a_j * f
            new_a = a_curr.copy()
            new_a[j] = a_j - delta
            new_a[i] = a_i + delta
            sc = compute_score(new_a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_delta = delta

        if best_delta != 0.0:
            a_curr[j] -= best_delta
            a_curr[i] += best_delta
            a_curr = normalize(a_curr)
            sc = compute_score(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    return best_a, best_score


def optimize_single(a_init, time_limit=120):
    """Full optimization pipeline for a single initial sequence."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Stage 1: Game iterations (LP-based)
    t1 = min(time_limit * 0.5, 80)
    if time_limit - (time.monotonic() - t0) > t1 + 10:
        a_g, sc_g = game_iterations(a, time_limit=t1, t_init=0.08, t_decay=0.999)
        if sc_g < best_score:
            best_score = sc_g
            best_a = a_g.copy()
            a = a_g.copy()

    # Stage 2: Subgradient
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(rem * 0.3, 40)
    if rem > t2 + 5:
        a_s, sc_s = subgradient_descent(a, time_limit=t2, lr0=0.03, top_k=max(10, n // 30))
        if sc_s < best_score:
            best_score = sc_s
            best_a = a_s.copy()
            a = a_s.copy()

    # Stage 3: Coordinate descent polish
    rem = time_limit - (time.monotonic() - t0)
    t3 = min(rem * 0.4, 30)
    if rem > t3 + 3:
        a_c, sc_c = coord_descent_polish(a, time_limit=t3)
        if sc_c < best_score:
            best_score = sc_c
            best_a = a_c.copy()
            a = a_c.copy()

    # Stage 4: Pair swap
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_p, sc_p = pair_swap_local(a, time_limit=min(rem, 20), n_trials=200)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_p.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization starting from non-symmetric ramp initializations.
    Uses game-theoretic LP iterations, subgradient, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Generate diverse non-symmetric initializations ===
    print("=== Generating non-symmetric initializations ===")
    candidates = []

    # Linear ramps (key insight: these achieve ~4/3 score)
    for n in [512, 640, 768, 896, 1024]:
        candidates.append((linear_ramp(n), f"ramp1_n{n}"))
        candidates.append((linear_ramp(n, offset=0.5), f"ramp1.5_n{n}"))
        candidates.append((linear_ramp(n, offset=-0.5), f"ramp0.5_n{n}"))

    # Power ramps
    for p in [0.5, 0.7, 1.3, 1.5, 2.0]:
        candidates.append((power_ramp(768, p), f"pow{p}_n768"))
        candidates.append((power_ramp(1024, p), f"pow{p}_n1024"))

    # Log ramp
    candidates.append((log_ramp(768), "log_n768"))
    candidates.append((log_ramp(1024), "log_n1024"))

    # From parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            candidates.append((parent, "parent"))
    except Exception:
        pass

    # Evaluate all candidates quickly
    print(f"  Evaluating {len(candidates)} candidates...")
    cand_scores = []
    for a, label in candidates:
        sc = compute_score(a)
        cand_scores.append((sc, a, label))
    cand_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(cand_scores[:8]):
        print(f"    #{i+1} [{label}]: {sc:.8f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse candidates ===
    print(f"\n=== Phase 1: Optimize top candidates (best={best_score:.8f}) ===")

    # Select diverse candidates (different base types)
    selected = []
    seen_types = set()
    for sc, a, label in cand_scores:
        if len(selected) >= 8:
            break
        base = label.split('_n')[0]
        if base not in seen_types:
            selected.append((a, label))
            seen_types.add(base)

    # Ensure parent is included
    if not any(l == "parent" for _, l in selected):
        for sc, a, label in cand_scores:
            if label == "parent" and sc < best_score + 0.1:
                selected.append((a, label))
                break

    t_per = min((deadline - time.monotonic() - 400) / max(1, len(selected)), 150)
    print(f"  {len(selected)} candidates, ~{t_per:.0f}s each")

    for a_init, label in selected:
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        t_alloc = min(t_per, rem)
        print(f"  Optimizing [{label}]: t={t_alloc:.0f}s")
        a_opt, sc_opt = optimize_single(a_init, time_limit=t_alloc)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Long game iterations on best ===
    print(f"\n=== Phase 2: Extended game iterations (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.7, 300)
            print(f"  Game iterations on best: t={t_alloc:.0f}s")
            a_g, sc_g = game_iterations(best_a, time_limit=t_alloc,
                                        t_init=0.05, t_decay=0.9995, max_iters=500)
            try_seq(a_g, "game_ext")
            if sc_g < best_score:
                best_a = a_g.copy()

            # Polish
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                t_alloc = min(rem * 0.5, 60)
                a_c, sc_c = coord_descent_polish(best_a, time_limit=t_alloc)
                try_seq(a_c, "cd_ext")
                if sc_c < best_score:
                    best_a = a_c.copy()

    # === Phase 3: Try different n from best shape ===
    print(f"\n=== Phase 3: Dimension exploration (best={best_score:.8f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        for target_n in [512, 640, 768, 896, 1024, 1280, 1536]:
            if target_n == best_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            old_x = np.linspace(0, 1, best_n)
            new_x = np.linspace(0, 1, target_n)
            a_interp = normalize(np.interp(new_x, old_x, best_a))
            t_alloc = min(rem * 0.4, 60)
            a_opt, sc = optimize_single(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim{target_n}")

    # === Phase 4: Final aggressive polishing ===
    print(f"\n=== Phase 4: Final polishing (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            a_c, sc = coord_descent_polish(best_a, time_limit=t_alloc, n_passes=200)
            try_seq(a_c, "final_cd")

            rem = deadline - time.monotonic() - 30
            if rem > 15:
                a_p, sc = pair_swap_local(best_a, time_limit=min(rem, 40), n_trials=1000)
                try_seq(a_p, "final_ps")

            rem = deadline - time.monotonic() - 20
            if rem > 15:
                # Tiny perturbations + re-optimize
                n_best = len(best_a)
                for trial in range(3):
                    rem = deadline - time.monotonic() - 15
                    if rem < 10:
                        break
                    noise = rng.standard_normal(n_best) * 0.003
                    a_pert = normalize(best_a + noise)
                    a_opt, sc = optimize_single(a_pert, time_limit=min(rem, 30))
                    try_seq(a_opt, f"perturb{trial}")

    # === Final verification ===
    if best_a is None:
        best_a = linear_ramp(768)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_seq = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_seq))
    if np.isfinite(final_score):
        if final_score != best_score:
            print(f"  Verification: computed={best_score:.8f}, actual={final_score:.8f}")
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return final_seq
