            m_exp = 1.0
            best_alpha_history.append(best_alpha)

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, m_expand_limit)

        # Smart restart: increasing perturbation when stuck
        if no_improve > 80:
            rng = np.random.default_rng(it * 331 + 7 + (it % 7))
            scale = 0.001 + 0.001 * min(no_improve / 50.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if no_improve > 250:
            break

    return best_f, best_obj


def subgradient_fw(f0, time_limit, verbose=False):
    """
    Frank-Wolfe using subgradient direction (no LP needed).
    Faster per iteration, useful as a complement to LP-FW.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    temperature = 30.0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        c = fast_convolve(f)
        M = c.max()
        if M < 1e-30:
            break

        # Compute weighted subgradient using softmax over top peaks
        k_top = min(30, len(c))
        peaks = np.argpartition(c, -k_top)[-k_top:]
        pv = c[peaks]
        scaled = temperature * (pv - M)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient of max_k (a*a)_k
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = 2.0 * w[i] * f[k - idx]
            grad[idx] += contrib
            if k % 2 == 0:
                j_mid = k // 2
                if 0 <= j_mid < n:
                    grad[j_mid] -= w[i] * f[j_mid]

        # FW direction: minimize grad^T * v over simplex => v = e_argmin(grad)
        # But this gives a vertex, which is too aggressive. Use the projection instead.
        # direction = project_simplex(-grad) - f
        # Actually for FW, the LMO for simplex is: v = e_j where j = argmin(grad_j)
        # Let's use a blend: move towards the vertex AND use gradient info
        j_min = np.argmin(grad)
        vertex = np.zeros(n)
        vertex[j_min] = 1.0

        # Use subgradient projection as direction (more stable than vertex)
        c_min = np.min(grad)
        proj_dir = project_simplex(f - 0.1 * (grad - c_min))
        direction = 0.5 * (vertex - f) + 0.5 * (proj_dir - f)

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = f + best_alpha * direction
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            new_obj = fast_convolve(f).max()
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        temperature *= 1.005
        if temperature > 200:
            temperature = 200

        if no_improve > 150:
            rng = np.random.default_rng(it * 7919 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
            temperature = max(temperature * 0.5, 30.0)

        if no_improve > 400:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.28):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.55):
    """Smooth bell with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = (1.0 - t) ** 3 * (1.0 + 2.0 * t)  # Smoothstep-like
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: focused LP-FW with convex line search,
    subgradient FW complement, and adaptive multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(512)

    # === Phase 1: Quick exploration of lengths and shapes (60s budget) ===
    print("=== Phase 1: Quick Exploration ===")
    phase1_budget = 60.0
    t_phase1_start = time.monotonic()

    init_candidates = []
    # Interpolate parent to various lengths
    for n_try in [512, 640, 768]:
        if len(parent) != n_try:
            a_interp = interpolate_sequence(parent, n_try)
        else:
            a_interp = parent.copy()
        init_candidates.append((a_interp, f"parent_n{n_try}"))

    # Fresh initializations
    for n_try in [512, 640]:
        for name, func in [("gauss", make_gaussian), ("bell", make_bell)]:
            a_init = func(n_try)
            init_candidates.append((a_init, f"{name}_n{n_try}"))

    # Evaluate all
    best_init = None
    best_init_score = float('inf')
    for a_init, label in init_candidates:
        if time.monotonic() - t_phase1_start > phase1_budget:
            break
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        # Also get actual evaluate_sequence score
        sc = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(sc) and sc < best_init_score:
            best_init_score = sc
            best_init = a_norm.copy()
        if obj < best_init_score:
            pass  # Use evaluate_sequence score for comparison
        print(f"    {label}: score={sc:.6f}, obj_unscaled={obj:.6f}")

    if best_init is None:
        best_init = parent
        best_init_score = best_score

    # Also try the parent directly
    sc_parent = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(sc_parent) and sc_parent < best_init_score:
        best_init = parent.copy()
        best_init_score = sc_parent

    print(f"  Best init for Phase 2: n={len(best_init)}, score={best_init_score:.6f}")

    # === Phase 2: Focused LP-FW on best initialization (majority of time) ===
    print(f"\n=== Phase 2: Focused LP-FW (n={len(best_init)}) ===")
    rem = deadline - time.monotonic()
    t_lpfw = min(rem - 120, max(rem * 0.55, 300))
    print(f"  LP-FW budget: {t_lpfw:.0f}s")

    a_opt, obj = lp_fw_optimize(best_init, time_limit=t_lpfw, verbose=True)
    try_seq(a_opt, "lpfw_main")

    # === Phase 3: Subgradient FW on current best ===
    print(f"\n=== Phase 3: Subgradient FW ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        t_sg = min(rem - 60, 150)
        print(f"  Subgradient FW budget: {t_sg:.0f}s")
        a_opt, obj = subgradient_fw(best_seq, time_limit=t_sg, verbose=True)
        try_seq(a_opt, "sg_fw")

    # === Phase 4: LP-FW with perturbation variants ===
    print(f"\n=== Phase 4: LP-FW Perturbation Variants ===")
    if best_seq is not None:
        n_best = len(best_seq)
        variants = []

        # Small perturbation
        noise = rng.standard_normal(n_best) * 0.002
        variants.append(project_simplex(best_seq + noise))

        # Symmetrize
        sym = 0.5 * best_seq + 0.5 * best_seq[::-1]
        variants.append(sym / sym.sum())

        # Blend with Gaussian
        gauss = make_gaussian(n_best, 0.28)
        blends = [0.9, 0.8, 0.7]
        for b in blends:
            v = b * best_seq + (1 - b) * gauss
            variants.append(v / v.sum())

        # Interpolate to nearby lengths
        for n_alt in [n_best + 64, n_best - 64]:
            if n_alt >= 256:
                variants.append(interpolate_sequence(best_seq, n_alt))

        for vi, v in enumerate(variants):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / (len(variants) - vi), 80)
            if t_alloc < 15:
                break
            print(f"  Variant {vi} (n={len(v)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(v, time_limit=t_alloc, verbose=(vi == 0))
            try_seq(a_opt, f"variant_{vi}")

    # === Phase 5: Final polish LP-FW ===
    print(f"\n=== Phase 5: Final LP-FW Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_polish = min(rem - 10, 120)
            print(f"  Polish budget: {t_polish:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_polish, verbose=True)
            try_seq(a_opt, "polish")

    # === Phase 6: Try longer sequences via interpolation + LP-FW ===
    print(f"\n=== Phase 6: Longer Sequence Exploration ===")
    if best_seq is not None:
        for target_n in [896, 1024]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            t_alloc = min(rem - 30, 80)
            if t_alloc > 20:
                print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 7: Ultra-final squeeze ===
    print(f"\n=== Phase 7: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_squeeze = min(rem - 5, 50)
            print(f"  Squeeze budget: {t_squeeze:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_squeeze)
            try_seq(a_opt, "squeeze")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
