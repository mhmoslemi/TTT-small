def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """FFT-based linear cross-convolution."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    ind = np.nonzero(cond)[0][-1]
    theta = (cssv[ind] - 1.0) / (ind + 1.0)
    return np.maximum(v - theta, 0.0)

    best_f = f.copy()

    v = np.zeros(n)  # velocity
    t0 = time.monotonic()
    no_improve = 0
    cur_temp = temp_init
    cur_lr = lr
    it = 0

    while time.monotonic() - t0 < time_limit:
        if n_iter_max is not None and it >= n_iter_max:
            break
        it += 1

        # Find active peaks with softmax weighting
        k_actual = min(top_k, 2 * n - 1)
        # Use top-k via argpartition
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Softmax weights over peaks
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient: sg_i = 2 * sum_k w_k * f_{k-i} (valid range)
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            # (f*f)[k] = sum_j f_j * f_{k-j}, d/d f_i = 2 * f_{k-i} (if valid)
            # f_{k-i} is valid when 0 <= k-i < n
            # So i ranges from max(0, k-n+1) to min(n-1, k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                sg[i_start:i_end + 1] += 2.0 * w[idx] * f[k - (i_start:i_end + 1)][::-1]

        # Nesterov momentum update
        v = momentum * v - cur_lr * sg
        f_new = f + v
        f_new = project_simplex(f_new)

        # Evaluate
        conv = fast_convolve(f_new)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            cur_lr *= 0.95
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(it * 104729 + 7)
                scale = 0.002 + 0.003 * min((no_improve - 500) // 500, 5)
                f_new = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                cur_lr = max(lr * 0.3, 0.001)
                no_improve = 0
                conv = fast_convolve(f_new)
                cur_obj = conv.max()

        f = f_new
        # Slowly increase temperature (explore more peaks)
        cur_temp *= temp_mult

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {it} ({elapsed:.1f}s): obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, max_iters=200, verbose=False):
    """LP-based Frank-Wolfe for polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_relax, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_relax)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 30, 'presolve': True})
        except Exception:
            res = None

        if res is None or not res.success or res.x is None:
            m_relax *= 1.2
            if m_relax > 10.0:
                m_relax = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.05, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            if best_alpha >= 1.0:
                lo, hi = 0.8, 1.0
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-8:
                continue

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = cur_max

            # Try scaling f[j] by different factors
            for mult in [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_skewed_init(n, skew=0.7, rng=None):
    """Create a skewed initialization with more mass on one side."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)
    # Beta-like distribution skewed to one side
    alpha = 1.0 + skew
    beta = 1.0
    # Simple approximation: t^alpha normalized
    f = t ** alpha
    f = f / f.sum()
    return f


def make_bimodal_init(n, peak_frac=0.3):
    """Bimodal initialization with peaks near both ends."""
    t = np.linspace(0, 1, n)
    center1 = 0.1
    center2 = 0.9
    width = peak_frac
    f = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2)
    f = f / f.sum()
    return f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining fast subgradient, LP-FW,
    coordinate descent, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    def try_sequence(seq):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        sc = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.clip(seq, 0, 1000).astype(np.float64)
            print(f"  NEW BEST: {sc:.6f} (n={len(best_seq)})")

    # === Phase 1: Subgradient on parent ===
    print(f"\n=== Phase 1: Subgradient on parent n={n_parent} ===")
    t_rem = deadline - time.monotonic()
    t1 = min(t_rem * 0.3, 400)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        f_opt, obj = subgradient_optimize(parent, time_limit=t1,
                                          lr=0.008, momentum=0.92,
                                          top_k_frac=0.04, temp_init=0.0005,
                                          temp_mult=1.003, verbose=True)
        try_sequence(f_opt)

    # === Phase 2: LP-FW polish on best ===
    print(f"\n=== Phase 2: LP-FW polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic()
    t2 = min(t_rem * 0.15, 120)
    if t2 > 10:
        print(f"  Budget: {t2:.0f}s")
        f_opt, obj = lp_fw_optimize(best_seq, time_limit=t2, max_iters=50, verbose=True)
        try_sequence(f_opt)

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for nl in [128, 192, 256, 320, 384, 448, 512, 576, 640, 704, 768,
               832, 896, 960, 1024, 1152, 1280, 1408, 1536, 1792, 2048]:
        if nl != n_best and 64 <= nl <= 2500:
            candidate_lengths.append(nl)

    # Sort by distance to current best
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    for target_n in candidate_lengths[:10]:
        t_rem = deadline - time.monotonic() - 60
        if t_rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(t_rem * 0.12, 80)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        f_opt, obj = subgradient_optimize(f_interp, time_limit=t_alloc * 0.7,
                                          lr=0.006, momentum=0.91,
                                          top_k_frac=0.04, temp_init=0.0004,
                                          temp_mult=1.003)
        try_sequence(f_opt)

        # Quick LP-FW polish
        t_rem2 = deadline - time.monotonic() - 60
        if t_rem2 > 15:
            f_opt2, _ = lp_fw_optimize(f_opt, time_limit=min(t_rem2 * 0.3, 30), max_iters=15)
            try_sequence(f_opt2)

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse inits (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    inits = []
    # Skewed variants
    for skew in [0.5, 1.0, 2.0, 3.0]:
        inits.append((f"skew{skew}", make_skewed_init(n_best, skew=skew)))
    # Bimodal
    inits.append(("bimodal", make_bimodal_init(n_best, peak_frac=0.25)))
    # Noise around best
    noise = rng.standard_normal(n_best) * 0.01
    inits.append(("noise1", project_simplex(best_seq + noise)))
    # Reversed best
    inits.append(("reversed", project_simplex(best_seq[::-1] + 1e-10)))
    # Mix with uniform
    inits.append(("mix_uni", project_simplex(0.7 * best_seq + 0.3 * np.ones(n_best) / n_best)))

    for label, f_init in inits[:6]:
        t_rem = deadline - time.monotonic() - 50
        if t_rem < 20:
            break
        t_alloc = min(t_rem * 0.1, 60)
        if t_alloc < 10:
            continue
        print(f"  {label}: t={t_alloc:.0f}s")
        f_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc,
                                          lr=0.005, momentum=0.90,
                                          top_k_frac=0.05, temp_init=0.001,
                                          temp_mult=1.004)
        try_sequence(f_opt)

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic() - 40
    if t_rem > 20:
        t_cd = min(t_rem * 0.3, 100)
        print(f"  t={t_cd:.0f}s")
        f_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
        try_sequence(f_opt)

    # === Phase 6: Final subgradient + LP-FW ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    t_rem = deadline - time.monotonic() - 20
    if t_rem > 15:
        # Subgradient with very small lr
        t_sg = min(t_rem * 0.5, 60)
        f_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg,
                                          lr=0.002, momentum=0.95,
                                          top_k_frac=0.02, temp_init=0.0001,
                                          temp_mult=1.001)
        try_sequence(f_opt)

    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        t_lp = min(t_rem * 0.7, 60)
        print(f"  LP-FW final: t={t_lp:.0f}s")
        f_opt, obj = lp_fw_optimize(best_seq, time_limit=t_lp, max_iters=20, verbose=True)
        try_sequence(f_opt)

    # === Phase 7: Micro-perturbations ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        t_rem = deadline - time.monotonic()
        if t_rem < 3:
            break
        scale = 0.0002 + trial * 0.0001
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        try_sequence(f_pert)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Final verification
    fs = float(evaluate_sequence([float(max(0, min(1000, x))) for x in best_seq]))
    if not np.isfinite(fs):
        best_seq = np.ones(384) / 384
        fs = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {fs:.6f} (n={len(best_seq)})")
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
