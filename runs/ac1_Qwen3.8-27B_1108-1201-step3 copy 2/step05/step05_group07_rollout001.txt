```python
def project_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero.cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 4.0 * n * corr

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbfgs_with_time_limit(z0, T, n, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with time limit."""
    t0 = time.monotonic()
    flag = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            flag[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad_z(z, T)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            flag[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad_z(z, T)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def nesterov_projected_gradient(a0, time_limit, lr_init=0.002, top_k=50):
    """
    Nesterov-accelerated projected gradient on the simplex using exact max.
    Objective: 2n * max_k conv(a,a)[k], constraint: a >= 0, sum a = 1.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.9
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Softmax weights on active peaks (temperature adapts)
        temp = max((pv.max() - pv.min()) * 5.0, 0.001)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient: 2n * sum_k w_k * 2*a_{k-j} for active k
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[idx] * a[k - idx_arr]
        grad *= 4.0 * n

        # Nesterov step
        a_y = a + momentum * v
        a_new = project_simplex(a_y - lr * grad)
        v = a_new - a
        a = a_new

        # Evaluate
        b = fast_convolve(a)
        cur_val = b.max()
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.998
        momentum = min(0.999, momentum * 1.001)

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

    return best_a


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex for final polish."""
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5])

    improved_global = True
    while time.monotonic() - t0 < time_limit and improved_global:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                # Try turning on
                for mult in [0.001, 0.005, 0.01]:
                    a_trial = a.copy()
                    a_trial[j] = mult
                    denom = 1.0 - mult
                    if denom < 1e-10:
                        continue
                    a_trial[:j] *= a[:j].sum() / (a[:j].sum() + a[j:] .sum() - a[j]) if (a[:j].sum() + a[j:].sum() - a[j]) > 0 else 1
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= (1.0 - mult) / (a[mask].sum() + 1e-300)
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        a = a_trial
                        best_obj = val
                        best_a = a.copy()
                        improved_global = True
                        old_val = a[j]
                        break
                continue

            best_j_obj = best_obj
            best_j_val = old_val
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= (1.0 - new_val) / denom
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - best_j_val) / denom
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initial_sequence(n, seed=42, structure="random"):
    """Create initial sequence with various structures."""
    rng = np.random.default_rng(seed)
    if structure == "random":
        a = rng.random(n) + 0.01
    elif structure == "ramp":
        a = np.linspace(0.1, 2.0, n)
    elif structure == "bimodal":
        a = np.zeros(n)
        center = n // 2
        width = n // 4
        a[center - width:center] = rng.random(width) + 0.5
        a[center:center + width] = rng.random(width) + 0.5
        a += 0.01
    elif structure == "periodic":
        freq = rng.integers(3, max(4, n // 50))
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    elif structure == "quadratic":
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x**2
    else:
        a = rng.random(n) + 0.01
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: log-softmax L-BFGS with temperature annealing,
    multi-length exploration, Nesterov projected gradient, and CD polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Quick L-BFGS at parent length, low T ===
    print("\n=== Phase 1: Quick L-BFGS (low T) ===")
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(40.0, rem * 0.05)

    if t1 > 10:
        z0 = z_from_a(parent)
        z_curr, _ = lbfgs_with_time_limit(z0, T=5, n=n_parent, time_limit=t1, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"quick_T5_n{n_parent}")

    # === Phase 2: Temperature annealing at parent length ===
    print("\n=== Phase 2: Temperature annealing ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    z_base = z_from_a(base)

    temps = [10, 25, 50, 100, 200, 400]
    for T in temps:
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(rem * 0.2, 80.0)
        if t_alloc < 10:
            break
        print(f"  T={T}, n={n_base}, t={t_alloc:.0f}s")
        z_base, _ = lbfgs_with_time_limit(z_base, T=T, n=n_base, time_limit=t_alloc, maxiter=3000)
        a_curr = softmax_from_z(z_base)
        try_seq(a_curr, f"anneal_T{T}_n{n_base}")

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length exploration ===")
    target_lengths = [1024, 1536, 2048, 3072]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break

        f_interp = interpolate_sequence(base, target_n)
        # Add small asymmetry to break palindromic symmetry
        asym = np.linspace(-0.02, 0.02, target_n)
        f_interp = np.maximum(f_interp + asym, 0)
        f_interp = f_interp / f_interp.sum()
        z0 = z_from_a(f_interp)

        t_alloc = min(rem * 0.25, 90.0)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, T=15, t={t_alloc:.0f}s")
        z_curr, _ = lbfgs_with_time_limit(z0, T=15, n=target_n, time_limit=t_alloc, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"len_{target_n}_T15")

        # Refine with higher T
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            continue
        t_alloc2 = min(rem * 0.2, 60.0)
        if t_alloc2 < 10:
            continue
        print(f"  n={target_n}, T=60, t={t_alloc2:.0f}s")
        z_curr, _ = lbfgs_with_time_limit(z_curr, T=60, n=target_n, time_limit=t_alloc2, maxiter=2000)
        a_curr = softmax_from_z(z_curr)
        try_seq(a_curr, f"len_{target_n}_T60")

    # === Phase 4: Deep annealing on best sequence ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        z_best = z_from_a(best_seq)
        deep_temps = [50, 100, 200, 400, 800]
        for T in deep_temps:
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 60.0)
            print(f"  Deep T={T}, n={n_best}, t={t_alloc:.0f}s")
            z_best, _ = lbfgs_with_time_limit(z_best, T=T, n=n_best, time_limit=t_alloc, maxiter=3000)
            a_curr = softmax_from_z(z_best)
            try_seq(a_curr, f"deep_T{T}")

    # === Phase 5: Nesterov projected gradient ===
    print(f"\n=== Phase 5: Nesterov PG (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_nest = min(rem * 0.5, 80.0)
            n_best = len(best_seq)
            top_k = max(30, n_best // 8)
            a_nest = nesterov_projected_gradient(best_seq, time_limit=t_nest, lr_init=0.001, top_k=top_k)
            try_seq(a_nest, "nesterov")

            # Second pass if improved
            rem = deadline - time.monotonic() - 40
            if rem > 15 and best_seq is not None:
                t_nest2 = min(rem * 0.5, 30.0)
                a_nest2 = nesterov_projected_gradient(best_seq, time_limit=t_nest2, lr_init=0.0005, top_k=top_k)
                try_seq(a_nest2, "nesterov2")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.6, 60.0)
            a_cd = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 7: Additional starts with different structures ===
    print(f"\n=== Phase 7: Alternative starts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        structures = ["ramp", "bimodal", "periodic"]
        for struct in structures:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            a_init = make_initial_sequence(n_best, seed=seed + hash(struct) % 1000, structure=struct)
            z_init = z_from_a(a_init)
            t_alloc = min(rem * 0.3, 50.0)
            if t_alloc < 10:
                continue
            z_opt, _ = lbfgs_with_time_limit(z_init, T=20, n=n_best, time_limit=t_alloc, maxiter=2000)
            a_opt = softmax_from_z(z_opt)
            try_seq(a_opt, f"alt_{struct}")

    # === Phase 8: Final high-T polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            z0 = z_from_a(best_seq)
            t_alloc = min(rem * 0.5, 30.0)
            z_final, _ = lbfgs_with_time_limit(z0, T=1000, n=n_best, time_limit=t_alloc, maxiter=2000)
            a_final = softmax_from_z(z_final)
            try_seq(a_final, "final_T1000")

            # Perturbation + re-optimize
            rem = deadline - time.monotonic() - 5
            if rem > 8:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 3
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_best) * 0.005
                    a_pert = project_simplex(best_seq + noise)
                    z_pert = z_from_a(a_pert)
                    t_alloc = min(rem * 0.4, 12.0)
                    z_p, _ = lbfgs_with_time_limit(z_pert, T=500, n=n_best, time_limit=t_alloc, maxiter=1000)
                    a_p = softmax_from_z(z_p)
                    try_seq(a_p, f"perturb_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
