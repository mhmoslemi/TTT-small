```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    """Solve the LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0."""
    n = len(f)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with optimal line search, diverse initializations, and length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_convolve_matrix(f):
    """Build matrix A such that (A @ g)[k] = (f * g)[k] for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end+1, j] = f[:length]
    return A

def solve_lp_direction(f, M, verbose=False):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns normalized g (sum to 1) or None if failed.
    """
    n = len(f)
    A = build_convolve_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    
    g = g / s
    return g

def golden_section_line_search(f, g, ff, fg, gg, n_iter=50):
    """
    Find alpha in [0,1] that minimizes max_k ((1-alpha)f + alpha g * (1-alpha)f + alpha g)_k.
    Uses precomputed convolutions ff=f*f, fg=f*g, gg=g*g.
    """
    a = 0.0
    b = 1.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    
    def objective(alpha):
        h = (1 - alpha) ** 2 * ff + 2 * alpha * (1 - alpha) * fg + alpha ** 2 * gg
        return h.max()
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = objective(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = objective(d)
    
    alpha = (a + b) / 2
    return alpha, objective(alpha)

def lp_iteration(f, M_multiplier=1.0, verbose=False):
    """
    One LP-based iteration: find direction via LP, do line search, return new f.
    """
    n = len(f)
    ff = fast_convolve(f)
    M = ff.max() * M_multiplier
    
    g = solve_lp_direction(f, M)
    if g is None:
        return f, M, 0.0
    
    fg = cross_convolve(f, g)
    gg = fast_convolve(g)
    
    alpha, obj = golden_section_line_search(f, g, ff, fg, gg)
    
    if alpha <= 0.001:
        return f, M, alpha
    
    f_new = (1 - alpha) * f + alpha * g
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f, M, alpha
    f_new = f_new / s
    
    return f_new, ff.max(), alpha

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

    return best_f, best_obj

    """
    LP-based optimization with optimal line search, diverse initializations,
    and length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform init n=512")
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Diverse initializations (n={n_parent}) ===")
    t_phase1_end = time.monotonic() + 200
    
    initializations = {
        'parent': parent.copy(),
        'uniform': np.ones(n_parent) / n_parent,
        'gauss': make_gaussian_init(n_parent, 0.5, 0.2),
        'bump': make_symmetric_init(n_parent, 'bump'),
        'multi3': make_multi_peak_init(n_parent, 3, rng),
        'multi5': make_multi_peak_init(n_parent, 5, rng),
        'smooth_rand': make_smooth_random_init(n_parent, rng, 5),
    }
    
    # Perturbed parent
    initializations['parent_pert'] = project_simplex(parent + rng.standard_normal(n_parent) * 0.02)
    
    phase1_best = None
    phase1_best_obj = float('inf')
    
    for name, init in initializations.items():
        rem = t_phase1_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 30)
        
        print(f"  LP from '{name}', t={t_alloc:.0f}s...")
        a_opt, obj = lp_optimize(init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"p1_{name}")
        if obj < phase1_best_obj:
            phase1_best_obj = obj
            phase1_best = a_opt.copy()
    
    if phase1_best is not None:
        try_seq(phase1_best, "p1_best")
    
    base = best_seq if best_seq is not None else parent
    
    # === Phase 2: Extended LP optimization ===
    print(f"\n=== Phase 2: Extended LP optimization (best={best_score:.8f}) ===")
    t_phase2_end = time.monotonic() + 800
    n_base = len(base)
    
    # Run LP with different M multipliers
    M_mults = [1.0, 1.01, 1.05, 1.1, 0.99]
    
    for mi, mm in enumerate(M_mults):
        rem = t_phase2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 150)
        
        print(f"  LP M_mult={mm}, t={t_alloc:.0f}s...")
        # Start from best or slightly perturbed
        if mi == 0:
            start = base.copy()
        else:
            start = project_simplex(base + rng.standard_normal(n_base) * 0.01)
            start = np.maximum(start, 0)
            start /= start.sum()
        
        a_opt, _ = lp_optimize(start, time_limit=t_alloc, M_multiplier=mm, verbose=False)
        try_seq(a_opt, f"p2_mm{mm}")
