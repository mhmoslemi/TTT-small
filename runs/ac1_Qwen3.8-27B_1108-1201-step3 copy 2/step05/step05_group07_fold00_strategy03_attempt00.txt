    a = project_simplex(a)
    # But standard implementation:
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.minimum(np.where(np.cumsum(u) > s)[0] - 1, n - 1)
    if rho == 0:
        return np.ones(n) / n * s
    theta = (cssv[rho] - s) / rho
    return np.maximum(z0 - theta * np.ones(n), 0.0)

def compute_convolve_full(x):
    """Compute full convolution of a."""
    x_np = np.array(x)
    if len(x_np) == 0:
        return np.array([])
    result = np.convolve(x_np, x_np, mode='full')
    return result

    """Objective function callable for optimization."""
    a = project_simplex(z)

    # Proper dummy difference method (finitedifference)
    # Compute the ratio
    denom = np.sum(a)**2
    num = 2.0 * n * max_b
    return num / denom if denom > 1e-15 else num
