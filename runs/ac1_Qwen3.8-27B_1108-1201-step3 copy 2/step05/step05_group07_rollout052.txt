    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """Cross-correlation: c[i] = sum_k w[k] * a[k-i] (valid part, length n)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to non-negative vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert non-negative vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def _soft_max_core(z, T, n):
    """Core computation: returns (objective, gradient) for soft-max smoothed objective."""
    a = softmax_from_z(z)
    b = fft_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def _scipy_obj(x):
    """Objective for scipy (uses module-level state)."""
    return _soft_max_core(x, _MLS_T, _MLS_N)[0]


def _scipy_jac(x):
    """Jacobian for scipy (uses module-level state)."""
    return _soft_max_core(x, _MLS_T, _MLS_N)[1]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_run(z0, T, n, time_limit, maxiter=5000):
    """Run L-BFGS-B on soft-max objective with time limit."""
    global _MLS_T, _MLS_N, _MLS_DEADLINE
    _MLS_T = T
    _MLS_N = n
    _MLS_DEADLINE = time.monotonic() + time_limit
    
    try:
        res = minimize(
            _scipy_obj,
            z0,
            jac=_scipy_jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        return res.x, res.fun
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return z0, _soft_max_core(z0, T, n)[0]


def anneal_lbgfs(a0, n, time_budget, T_start=0.5, T_end=2000.0, n_steps=8):
    """Temperature annealing with L-BFGS. Returns optimized a."""
    z = z_from_a(a0)
    T = T_start
    elapsed = 0.0
    t0 = time.monotonic()
    
    for i in range(n_steps):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 5.0:
            break
        t_alloc = min(rem * 0.85 / max(1, n_steps - i), 120.0)
        if t_alloc < 3.0:
            break
        _obj_val, _ = _soft_max_core(z, T, n)
        print(f"    T={T:.1f}, obj={_obj_val:.6f}, t={t_alloc:.0f}s")
        z, _ = lbgfs_run(z, T, n, t_alloc, maxiter=3000)
        T = T * (T_end / T_start) ** (1.0 / max(1, n_steps - 1))
    return softmax_from_z(z)


def nesterov_subgradient(a0, time_limit, lr_init=0.002, top_k=50):
    """Nesterov-accelerated projected subgradient on the true max objective."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_val = fft_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    k = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        b = fft_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        max_b = b.max()
        # Scaled subgradient: weight peaks by their excess
        excess = b[peaks] - 0.9 * max_b
        excess = np.maximum(excess, 0)
        w = excess / max(excess.sum(), 1e-15)
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            lag = int(peaks[i])
            js = max(0, lag - n + 1)
            je = min(n - 1, lag)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[lag - idx]
        
        # Nesterov momentum
        y = project_simplex(a + v)
        # Recompute subgradient at y
        b_y = fft_convolve(y)
        peaks_y = np.argpartition(b_y, -k_actual)[-k_actual:]
        max_y = b_y.max()
        excess_y = b_y[peaks_y] - 0.9 * max_y
        excess_y = np.maximum(excess_y, 0)
        w_y = excess_y / max(excess_y.sum(), 1e-15)
        
        sg_y = np.zeros(n)
        for i in range(len(peaks_y)):
            if w_y[i] < 1e-12:
                continue
            lag = int(peaks_y[i])
            js = max(0, lag - n + 1)
            je = min(n - 1, lag)
            idx = np.arange(js, je + 1)
            sg_y[idx] += 2.0 * w_y[i] * y[lag - idx]
        
        lr_k = lr_init / (1.0 + 0.01 * k)
        a_new = project_simplex(y - lr_k * sg_y)
        v = (a - a_new) / (1.0 + 0.01 * k)
        a = a_new
        
        val = fft_convolve(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    return best_a, best_val


def coord_descent(a0, time_limit=60.0, n_mults=5):
    """Coordinate descent: adjust one element at a time, redistribute to others."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fft_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.linspace(0.3, 2.0, n_mults)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fft_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def sine_init(n):
    """Sine-weighted initialization (zero at edges)."""
    j = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (j + 0.5) / n) ** 2
    return a / a.sum()


def raised_cosine_init(n):
    """Raised cosine initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (j - (n - 1) / 2.0) / (n - 1)) ** 2
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def active_set_refine(a0, time_limit=60.0, n_active=20):
    """Iteratively refine by identifying active peaks and doing local optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fft_convolve(a).max()
    t0 = time.monotonic()
    
    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        b = fft_convolve(a)
        # Find active set (top peaks)
        k_actual = min(n_active, len(b))
        active = np.argpartition(b, -k_actual)[-k_actual:]
        active = sorted(set(int(x) for x in active))
        max_b = b.max()
        
        # Try to reduce max by solving a small QP via active constraints
        # Use scipy minimize on the max of active constraints
        # Parameterize: minimize t s.t. conv(a,a)_k <= t for k in active, a>=0, sum=1
        # Instead, use a simple approach: move mass away from peaks
        
        improved = False
        for lag in active[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            if b[int(lag)] < max_b * 0.9:
                continue
            # Find contributing pairs
            js = max(0, int(lag) - n + 1)
            je = min(n - 1, int(lag))
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[int(lag) - idx]
            if contributions.max() < 1e-15:
                continue
            # Reduce the largest contributor
            mi = np.argmax(contributions)
            i = int(idx[mi])
            j2 = int(lag) - i
            old_i = a[i]
            old_j = a[j2]
            # Try reducing the larger of the two
            if old_i >= old_j:
                target = i
            else:
                target = j2
            old_val = a[target]
            if old_val < 1e-12:
                continue
            # Reduce by factor and redistribute
            for factor in [0.8, 0.6, 0.4]:
                new_val = old_val * factor
                diff = old_val - new_val
                a_trial = a.copy()
                a_trial[target] = new_val
                mask = np.ones(n, dtype=bool)
                mask[target] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fft_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    a = a_trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try random perturbation to escape
            rng = np.random.default_rng(hash((n, iteration)) % 100000)
            a_pert = project_simplex(a + rng.standard_normal(n) * 0.01)
            val = fft_convolve(a_pert).max()
            if val < best_obj:
                a = a_pert
                best_obj = val
                best_a = a.copy()
            else:
                break
    
    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses soft-max L-BFGS with temperature annealing, Nesterov subgradient,
    coordinate descent, and active-set refinement across multiple lengths.
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = sine_init(768)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: score={best_score:.6f}")
    
    # === Phase 1: Optimize at parent length with L-BFGS annealing ===
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(300.0, rem * 0.18)
    if t1 > 30:
        print(f"\n=== Phase 1: L-BFGS anneal n={n_parent}, budget={t1:.0f}s ===")
        a_opt = anneal_lbgfs(parent, n_parent, t1, T_start=0.5, T_end=1000, n_steps=8)
        try_seq(a_opt, f"anneal_n{n_parent}")
    
    # === Phase 2: Try multiple lengths ===
    lengths_to_try = []
    base = best_seq if best_seq is not None else parent
    
    # Always try larger lengths
    for L in [1024, 2048, 4096, 8192]:
        if L != n_parent and L > 256:
            lengths_to_try.append(L)
    
    # Also try the parent length if not already done
    if n_parent not in lengths_to_try and n_parent >= 512:
        lengths_to_try.insert(0, n_parent)
    
    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            print(f"\n  Skipping n={target_n} (insufficient time)")
            break
        
        # Try multiple initializations
        inits = []
        # Interpolated parent
        if best_seq is not None:
            inits.append(interpolate_to_length(best_seq, target_n))
        # Sine init
        inits.append(sine_init(target_n))
        # Raised cosine init
        inits.append(raised_cosine_init(target_n))
        # Random (smoothed)
        r = rng.standard_normal(target_n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0)
        if r.sum() > 1e-15:
            inits.append(r / r.sum())
        
        n_inits = len(inits)
        t_per_init = min((rem * 0.7) / n_inits, 80.0)
        
        for ii, a_init in enumerate(inits):
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(t_per_init, rem * 0.8)
            if t_alloc < 10:
                break
            
            label = ["interpolated", "sine", "raised_cos", "random_smooth"][ii] if ii < 4 else f"init{ii}"
            print(f"\n  n={target_n}, init={label}, t={t_alloc:.0f}s")
            
            if t_alloc > 15:
                a_opt = anneal_lbgfs(a_init, target_n, t_alloc, T_start=1.0, T_end=500, n_steps=5)
                try_seq(a_opt, f"n{target_n}_{label}")
            
            # Quick subgradient polish
            rem = deadline - time.monotonic() - 200
            if rem > 15 and best_seq is not None:
                a_best_for_len = interpolate_to_length(best_seq, target_n) if len(best_seq) != target_n else best_seq
                t_sg = min(rem * 0.3, 30.0)
                if t_sg > 5:
                    a_sg, _ = nesterov_subgradient(a_best_for_len, t_sg, lr_init=0.003, top_k=min(50, target_n // 10))
                    try_seq(a_sg, f"n{target_n}_sg_{label}")
    
    # === Phase 3: Deep refinement on best ===
    rem = deadline - time.monotonic() - 100
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Deep refinement (best={best_score:.6f}, n={n_best}) ===")
        
        # L-BFGS with very high T
        t_lbgfs = min(rem * 0.4, 120.0)
        if t_lbgfs > 15:
            z0 = z_from_a(best_seq)
            z_opt, _ = lbgfs_run(z0, 500, n_best, t_lbgfs, maxiter=5000)
            a_opt = softmax_from_z(z_opt)
            try_seq(a_opt, "deep_T500")
            
            # Even higher T
            rem = deadline - time.monotonic() - 100
            if rem > 20:
                t2 = min(rem * 0.3, 60.0)
                z_opt2, _ = lbgfs_run(z_opt, 1500, n_best, t2, maxiter=5000)
                a_opt2 = softmax_from_z(z_opt2)
                try_seq(a_opt2, "deep_T1500")
        
        # Nesterov subgradient
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_sg = min(rem * 0.5, 60.0)
            a_sg, _ = nesterov_subgradient(best_seq, t_sg, lr_init=0.002, top_k=min(80, n_best // 8))
            try_seq(a_sg, "nesterov_deep")
        
        # Coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem * 0.5, 45.0)
            a_cd, _ = coord_descent(best_seq, t_cd, n_mults=7)
            try_seq(a_cd, "coord_deep")
        
        # Active set refinement
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_as = min(rem * 0.6, 40.0)
            a_as, _ = active_set_refine(best_seq, t_as, n_active=30)
            try_seq(a_as, "active_set")
    
    # === Phase 4: Try a significantly different length ===
    rem = deadline - time.monotonic() - 50
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        # Try 2x the best length
        new_n = n_best * 2
        if new_n <= 16384 and new_n != n_best:
            print(f"\n=== Phase 4: Upscale to n={new_n} ===")
            a_interp = interpolate_to_length(best_seq, new_n)
            t_alloc = min(rem * 0.6, 100.0)
            if t_alloc > 20:
                a_opt = anneal_lbgfs(a_interp, new_n, t_alloc, T_start=5.0, T_end=800, n_steps=5)
                try_seq(a_opt, f"upscale_n{new_n}")
        
        # Try a smaller length (sometimes helpful)
        if n_best > 1024:
            small_n = n_best // 2
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 30:
                a_small = interpolate_to_length(best_seq, small_n)
                t_alloc2 = min(rem2 * 0.5, 50.0)
                if t_alloc2 > 15:
                    a_opt = anneal_lbgfs(a_small, small_n, t_alloc2, T_start=5.0, T_end=500, n_steps=4)
                    try_seq(a_opt, f"downscale_n{small_n}")
    
    # === Phase 5: Final micro-optimization ===
    rem = deadline - time.monotonic() - 10
    if rem > 10 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Final micro-optimization (best={best_score:.6f}) ===")
        
        # Very aggressive subgradient with small steps
        t_final = min(rem * 0.6, 30.0)
        if t_final > 5:
            a_f, _ = nesterov_subgradient(best_seq, t_final, lr_init=0.0005, top_k=min(40, n_best // 10))
            try_seq(a_f, "final_sg")
        
        # Tiny L-BFGS at very high T
        rem2 = deadline - time.monotonic() - 5
        if rem2 > 8:
            z0 = z_from_a(best_seq)
            z_f, _ = lbgfs_run(z0, 3000, n_best, min(rem2 * 0.7, 20.0), maxiter=3000)
            a_f = softmax_from_z(z_f)
            try_seq(a_f, "final_T3000")
        
        # One more coord descent pass
        rem3 = deadline - time.monotonic() - 3
        if rem3 > 8:
            a_cd, _ = coord_descent(best_seq, min(rem3 * 0.8, 15.0), n_mults=9)
            try_seq(a_cd, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
