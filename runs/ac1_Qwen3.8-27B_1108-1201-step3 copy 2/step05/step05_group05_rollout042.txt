       A = np.fft.rfft(a, fft_size)
       return np.fft.irfft(A * A, fft_size)[:size]

   def solve_lp_for_a(b, n):
       """Given b, find a >= 0, sum(a)=1, minimizing max_k (a*b)[k]."""
       # Variables: [a_0, ..., a_{n-1}, t]
       c = np.zeros(n + 1)
       c[-1] = 1.0  # minimize t
       
       # Constraints: (a*b)[k] - t <= 0 for k = 0, ..., 2n-2
       A_ub = np.zeros((2*n - 1, n + 1))
       for k in range(2*n - 1):
           lo = max(0, k - n + 1)
           hi = min(n - 1, k)
           for j in range(lo, hi + 1):
               A_ub[k, j] = b[k - j]
           A_ub[k, n] = -1.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP optimization
with diverse initializations, multi-length exploration, and coordinate descent polish.
Core idea: alternate between LP subproblems (fix one argument, optimize the other)
to find sequences with minimal peak auto-convolution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(b, n):
    """Build the LP constraint matrix for min t s.t. (a*b)[k] <= t, sum(a)=1, a>=0."""
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A_ub[k, j] = b[k-j] for j in [lo, hi]
        # k-j ranges from k-lo down to k-hi
        A_ub[k, lo:hi + 1] = b[k - hi:k - lo + 1][::-1]
        A_ub[k, n] = -1.0
    return A_ub


def solve_lp_conv(b, n):
    """
    Solve: min t s.t. (a*b)[k] <= t for all k, a >= 0, sum(a) = 1.
    Returns optimal a and t.
    """
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = build_lp_matrix(b, n)
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        a_opt = np.maximum(result.x[:n], 0.0)
        t_opt = result.x[n]
        return a_opt, t_opt
    return None, None


def alternating_lp(a_init, max_iter=300, tol=1e-12, verbose=False):
    """
    Alternating LP: repeatedly solve min max(a*b) to find fixed point.
    Returns (best_a, best_peak).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(max_iter):
        b_new, t_val = solve_lp_conv(a, n)
        if b_new is None:
            break
        b_new = np.maximum(b_new, 0.0)
        sb = b_new.sum()
        if sb < 1e-15:
            break
        b_new = b_new / sb

        diff = np.max(np.abs(b_new - a))
        peak = fast_convolve(b_new).max()
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = b_new.copy()

        if diff < tol:
            a = b_new
            break
        a = b_new

        if verbose and it % 50 == 0:
            print(f"    ALP iter {it}: peak={best_peak:.10f}, diff={diff:.2e}")

    return best_a, best_peak

def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def step_init(n, ratio, frac):
    """Step function: first `frac*n` entries are `ratio`, rest are 1."""
    k = int(n * frac)
    a = np.zeros(n)
    a[:k] = ratio
    a[k:] = 1.0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random exponential initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def ramp_init(n, direction=1):
    """Linear ramp initialization."""
    a = np.arange(1, n + 1, dtype=np.float64)
    if direction == -1:
        a = a[::-1]
    return a / a.sum()


def cosine_init(n):
    """Cosine bell initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    return a / a.sum()


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    return a / a.sum()


def antisymmetric_init(n, rng):
    """High on one side, low on the other."""
    a = np.zeros(n)
    half = n // 2
    a[:half] = rng.uniform(1.5, 3.0, half)
    a[half:] = rng.uniform(0.3, 0.8, n - half)
    return a / a.sum()


def dual_step_init(n, ratio, frac, rng):
    """Two-step: high, medium, or low regions."""
    k1 = int(n * frac)
    a = np.zeros(n)
    a[:k1] = ratio
    a[k1:] = 1.0
    # Add small perturbation
    a += rng.uniform(0, 0.05, n)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with multiplier search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

    return best_a, best_obj


def subgradient_refine(a0, time_limit, verbose=False):
    """Subgradient method with momentum for local refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    tau = 50.0
    lr = 0.03
    momentum = 0.85
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        if no_improve > 300:
            momentum = min(0.95, momentum + 0.003)
            lr = max(0.002, lr * 0.98)
        elif no_improve > 100:
            tau = min(300.0, tau * 1.01)
        else:
            tau = min(200.0, tau * 1.005)

        # Compute smooth subgradient
        conv = fast_convolve(a)
        k_actual = min(100, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        grad_centered = grad - grad.mean()
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, target_sum=1.0)

        obj_new = fast_convolve(a_new).max()
        a = a_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 17 + 3)
            noise = 0.003 + 0.003 * min(no_improve // 500, 5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise, target_sum=1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.02
            tau = 40.0
            momentum = 0.83

    return best_a, best_obj
