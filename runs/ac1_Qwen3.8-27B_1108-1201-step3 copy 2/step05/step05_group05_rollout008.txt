"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Gerchberg-Saxton iterative
spectral shaping (primary), subgradient refinement, coordinate descent,
and multi-length/multi-init exploration. The GS approach treats the problem
as a PAR (peak-to-average ratio) minimization of the auto-convolution,
alternating between time-domain non-negativity projection and frequency-domain
spectral flattening.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gs_optimize(a0, time_limit, alpha=0.3, power_mode=False, power=0.7,
                relaxation=1.0, verbose=False):
    """GS-style iterative spectral shaping to minimize max auto-convolution."""
    a = project_nonneg_sum1(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute auto-convolution via FFT
        A = np.fft.fft(a, fft_size)
        A_sq = A * A
        b = np.fft.ifft(A_sq).real[:size]
        cur_max = b.max()

        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Create target b (flatten peaks)
        mean_b = b.sum() / size
        if power_mode:
            b_target = np.power(np.maximum(b, 1e-30), power)
        else:
            # Adaptive alpha: increase when stagnant
            eff_alpha = alpha + 0.1 * min(stagnant / 200.0, 2.0)
            eff_alpha = min(eff_alpha, 0.95)
            b_target = (1.0 - eff_alpha) * b + eff_alpha * mean_b
            b_target = np.maximum(b_target, 0.0)

        # Renormalize to preserve total sum
        s_tgt = b_target.sum()
        if s_tgt > 1e-30:
            b_target *= b.sum() / s_tgt

        # Compute new A from b_target
        B = np.fft.fft(b_target, fft_size)
        # Complex square root (principal branch)
        A_new = np.sqrt(B.astype(np.complex128))
        a_candidate = np.fft.ifft(A_new).real[:n]

        # Relaxation and projection
        a_new = project_nonneg_sum1((1.0 - relaxation) * a + relaxation * a_candidate)

        # If stuck, add perturbation
        if stagnant > 300:
            rng_p = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.005 + 0.005 * min(stagnant // 300, 3)
            a_new = project_nonneg_sum1(a_new + rng_p.standard_normal(n) * noise_scale)
            stagnant = 0

        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS iter {it}: best={best_obj:.8f}, cur={cur_max:.8f}, "

    return best_a, best_obj


def subgrad_optimize(a0, time_limit, verbose=False):
    """Subgradient optimization with momentum and adaptive parameters."""
    a = project_nonneg_sum1(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    tau = 60.0
    lr = 0.035
    momentum = 0.82
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Smoothed subgradient of max(b)
        conv = fast_convolve(a)
        k_actual = min(200, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = tau * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        grad -= grad.mean()

        v = momentum * v - lr * grad
        a_new = project_nonneg_sum1(a + v)

        # Adaptive line search
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj:
            # Try to extend step
            a_ext = project_nonneg_sum1(a + 1.5 * v)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext

        a = a_new

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 500:
            momentum = min(0.95, momentum + 0.003)
            lr = max(0.004, lr * 0.96)
        elif no_improve > 200:
            tau = min(400.0, tau * 1.01)
        elif no_improve > 50:
            tau = max(30.0, tau * 0.995)
            lr = min(0.06, lr * 1.003)
        else:
            tau = min(200.0, tau * 1.003)

        # Restart if very stuck
        if no_improve > 700:
            rng_r = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.003 + 0.003 * min(no_improve // 700, 4)
            a = project_nonneg_sum1(best_a + rng_r.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.03
            tau = 50.0
            momentum = 0.82

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}: best={best_obj:.8f}, lr={lr:.4f}, "
                  f"tau={tau:.1f} ({elapsed:.1f}s)")

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent with multiplier-based local search."""
    a = project_nonneg_sum1(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            old_remaining = 1.0 - old_val
            if old_remaining < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = project_nonneg_sum1(a_trial)
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / old_remaining
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = project_nonneg_sum1(a)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def gaussian_init(n):
    """Gaussian-shaped initialization."""
    x = np.linspace(-3, 3, n)
    a = np.exp(-x ** 2 / 2)
    return a / a.sum()


def triangular_init(n):
    """Triangular distribution initialization."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
