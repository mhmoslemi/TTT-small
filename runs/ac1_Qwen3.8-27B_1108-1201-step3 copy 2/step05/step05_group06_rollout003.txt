def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def srsa_optimize(f0, time_limit, gamma0=0.1, gamma_decay=0.999, alpha0=0.05, alpha_decay=0.9995):

    n = len(f0)
    # Work in log-space with offset to ensure positivity
    # a[i] = exp(log_weights[i]) / sum(exp(log_weights))
    # Initialize from f0

    f = f / f.sum()
    log_w = np.log(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    rng = np.random.default_rng(42)
    gamma = gamma0
    alpha = alpha0
    t0 = time.monotonic()
    it = 0

        it += 1
        # Generate random perturbation
        c = gamma
        delta = rng.choice([-1.0, 1.0], size=n)

        log_w_plus = log_w + c * delta
        f_plus = np.exp(log_w_plus - log_w_plus.max())
        f_plus = f_plus / f_plus.sum()
        obj_plus = fast_convolve(f_plus).max()
        
        log_w_minus = log_w - c * delta
        f_minus = np.exp(log_w_minus - log_w_minus.max())
        f_minus = f_minus / f_minus.sum()
        obj_minus = fast_convolve(f_minus).max()

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Compute smooth gradient of logsumexp(conv(f,f)/T) via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def peak_subgrad(f, epsilon=0.0):
    """Compute uniform subgradient over active peaks."""
    n = len(f)
    conv = fast_convolve(f)
    max_conv = conv.max()
    active = conv >= (max_conv - epsilon)
    active_indices = np.where(active)[0]
    if len(active_indices) == 0:
        return np.zeros(n), conv, max_conv
    # grad[j] = sum_{k in active} a[k-j] / len(active)
    # Use FFT for efficiency
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    I_fft = np.fft.rfft(indicator, N)
    corr = np.fft.irfft(A_fft * np.conj(I_fft), N)[:n]
    grad = 2.0 * corr / len(active_indices)
    return grad, conv, max_conv


def srsa_optimize(f0, time_limit, rng_seed=42, verbose=False):
    """SPSA in log-space with softmax normalization."""
    n = len(f0)
    f = np.maximum(f0, 1e-10)
    f = f / f.sum()
    log_w = np.log(f)
    log_w -= log_w.max()  # normalize for stability

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    rng = np.random.default_rng(rng_seed)
    c0 = 0.1
    a0 = 0.05
    c_decay = 0.9997
    a_decay = 0.9999

    t0 = time.monotonic()
    it = 0
    best_it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = (time.monotonic() - t0) / time_limit
        c = c0 / (1.0 + it * 0.001)
        a = a0 / (1.0 + it * 0.0005)

        delta = rng.choice([-1.0, 1.0], size=n)

        log_w_plus = log_w + c * delta
        shifted = log_w_plus - log_w_plus.max()
        f_plus = np.exp(shifted)
        f_plus = f_plus / f_plus.sum()
        obj_plus = fast_convolve(f_plus).max()

        log_w_minus = log_w - c * delta
        shifted = log_w_minus - log_w_minus.max()
        f_minus = np.exp(shifted)
        f_minus = f_minus / f_minus.sum()
        obj_minus = fast_convolve(f_minus).max()

        g = (obj_plus - obj_minus) / (2.0 * c * delta)

        log_w = log_w - a * g
        # Renormalize log_w for stability
        log_w -= log_w.max()

        # Periodically compute true objective
        if it % 10 == 0:
            f_cur = np.exp(log_w)
            f_cur = f_cur / f_cur.sum()
            obj = fast_convolve(f_cur).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_cur.copy()
                best_it = it
                if verbose and it % 1000 == 0:
                    print(f"    SPSA iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def smooth_adam(f0, time_limit, T_start=0.3, T_end=0.003, verbose=False, seed_offset=0):
    """Adam optimization with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345 + seed_offset)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        lr = alpha * (1.0 - 0.5 * progress)

        grad, lse_val, conv, w = smooth_grad(f, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)

        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1

        if no_improve > 2000:
            scale = 0.002 + 0.004 * min(no_improve // 2000, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_f, best_obj


def peak_subgrad_opt(f0, time_limit, epsilon=0.0, verbose=False):
    """Optimize using uniform subgradient over active peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    eta = 0.02
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = (time.monotonic() - t0) / time_limit
        eta_cur = eta * (1.0 - 0.6 * elapsed)

        grad, conv, max_conv = peak_subgrad(f, epsilon=epsilon * (1.0 - elapsed))

        f = f - eta_cur * grad
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    PeakSub iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 1500:
            scale = 0.003 + 0.003 * min(no_improve // 1500, 3)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            no_improve = 0

    return best_f, best_obj


def lp_direction_fast(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_direction_fast(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.1, 3.0)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.05, 2.5)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = fast_convolve(f).max()
            m_relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.08, 3.5)

        if no_improve > 100:
            rng_p = np.random.default_rng(it * 13 + 3)
            noise = rng_p.standard_normal(n) * 0.004
            f = project_simplex(best_f + noise)
            M = fast_convolve(f).max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent_peaks(f0, time_limit, verbose=False):
    """Coordinate descent focusing on peak-reducing moves."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        max_conv = conv.max()
        # Find active peaks
        active = np.where(conv >= max_conv * 0.999)[0]

        # For each active peak, try reducing contributing coordinates
        n_trials = min(100 * (n // 100 + 1), 500)

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            k = int(rng.choice(active))
            # Coordinates j that contribute to conv[k]: a[j]*a[k-j]
            # Try shifting mass from high-contributing coordinates
            j_candidates = []
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                other = k - j
                if 0 <= other < n:
                    prod = f[j] * f[other]
                    j_candidates.append((j, prod))

            if not j_candidates:
                continue

            j_candidates.sort(key=lambda x: -x[1])
            j = j_candidates[0][0]

            # Try reducing f[j] and redistributing
            old_val = f[j]
            if old_val < 1e-8:
                continue

            best_new_val = old_val
            best_local_obj = max_conv

            for frac in [0.8, 0.7, 0.6, 0.5, 0.4, 0.3]:
                new_val = old_val * frac
                denom = 1.0 - old_val
                if denom < 1e-10:
                    break
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                val = fast_convolve(f_trial).max()
                if val < best_local_obj - 1e-14:
                    best_local_obj = val
                    best_new_val = new_val

            if best_new_val < old_val - 1e-10:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_new_val) / denom
                    f[j] = best_new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 5:
            # Try increasing low coordinates
            pass2_improved = False
            for _ in range(n_trials // 2):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(n))
                old_val = f[j]
                best_new_val = old_val
                best_local_obj = fast_convolve(f).max()

                for frac in [1.2, 1.5, 2.0, 3.0]:
                    new_val = old_val * frac
                    if new_val > 0.5:
                        break
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        break
                    scale = (1.0 - new_val) / denom
                    if scale < 0:
                        break
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    val = fast_convolve(f_trial).max()
                    if val < best_local_obj - 1e-14:
                        best_local_obj = val
                        best_new_val = new_val

                if best_new_val > old_val + 1e-10:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_new_val) / denom
                        f[j] = best_new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    if best_local_obj < best_obj - 1e-13:
                        best_obj = best_local_obj
                        best_f = f.copy()
                        pass2_improved = True
                        improved = True

            if not improved and passes > 8:
                break

    return best_f, best_obj


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.005
    return a / a.sum()


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * a[-1]])
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

    LP-Frank-Wolfe, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old_score:.8f}]")

    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    n_parent = len(parent) if parent is not None else 768

    for n_try in set([n_parent, 512, 768]):
        for r in [0.98, 0.99, 0.995]:

            _try(g, f"geo_r{r}_n{n_try}")
        for r in [0.98, 0.99]:

            _try(g, f"symgeo_r{r}_n{n_try}")

    # === Phase 2: Smooth Adam optimization ===
    print(f"\n=== Phase 2: Smooth Adam (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 550

        _try(f_interp, f"interp_n{target_n}")

        t1 = min(rem * 0.4, 60)
        if t1 > 10:
            a_opt, _ = smooth_adam(f_interp, time_limit=t1, T_start=0.3, T_end=0.01, seed_offset=target_n)
            _try(a_opt, f"adam_n{target_n}")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t2 = min(rem * 0.4, 40)
            a_opt, _ = smooth_adam(f_interp, time_limit=t2, T_start=0.1, T_end=0.005, seed_offset=target_n + 1)
            _try(a_opt, f"adam2_n{target_n}")

    # Try shorter lengths too
    for target_n in [640, 800]:
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        t1 = min(rem * 0.5, 30)
        if t1 > 10:
            a_opt, _ = smooth_adam(f_interp, time_limit=t1, T_start=0.2, T_end=0.005, seed_offset=target_n + 50)
            _try(a_opt, f"adam_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine smooth optimization
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem, 60)
            print(f"  Fine smooth t={t_alloc:.0f}s, T: 0.02->0.001")
            a_opt, _ = smooth_adam(best_seq, time_limit=t_alloc, T_start=0.02, T_end=0.001, verbose=True, seed_offset=200)
            _try(a_opt, "smooth_fine")

        # Fine peak subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, _ = peak_subgrad_opt(best_seq, time_limit=t_alloc, epsilon=0.0, verbose=False)
            _try(a_opt, "peaksub_fine")

        # Fine coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 20)
            a_opt, _ = coord_descent_peaks(best_seq, time_limit=t_alloc)
            _try(a_opt, "coord_fine")

        # Multiple small perturbations with quick optimization
        for i in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 10)
            a_opt, _ = smooth_adam(f_pert, time_limit=t_alloc, T_start=0.01, T_end=0.001, seed_offset=300 + i)
            _try(a_opt, f"perturb_{i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
