        grad[indices] = 2.0 * f[m - indices]
    return grad

        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)
        
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1
        
        # Restart with perturbation
        if no_improve > 2000:
            scale = 0.003 + 0.002 * min((no_improve - 2000) // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with exact evaluation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        # Random order
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12 and rng.random() > 0.1:
                continue
            
            # Binary/golden search for optimal value of f[j]
            # Keep sum = 1 by scaling others
            lo = 0.0
            hi = 0.9999
            best_local_val = old_val
            best_local_obj = fast_convolve(f).max()
            
            # Try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                         old_val, 0.0, hi * 0.1, hi * 0.5]
            candidates = [max(0, min(0.9999, c)) for c in candidates]
            
            for cv in candidates:
                if abs(cv - old_val) < 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[j] = cv
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - cv) / denom
                obj = fast_convolve(f_trial).max()
                if obj < best_local_obj - 1e-14:
                    best_local_obj = obj
                    best_local_val = cv
            
            if best_local_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    f[j] = best_local_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_local_val) / denom
                    f = f / f.sum()
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try perturbation
            scale = 0.002
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
    
    return best_f, best_obj


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    if half > 1:
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
    else:
        a = asc
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * 0.001])
    a = a[:n]
    a = np.maximum(a, 0)
    return a / a.sum()


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.003
    return a / a.sum()


def make_feker(n):
    """Fejér kernel-like: (1 - k/n)^2 for k=0..n-1"""
    k = np.arange(n, dtype=np.float64)
    a = (1.0 - k / n) ** 2
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_bell(n, sigma_frac=0.3):
    """Gaussian bell."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Let me try a different approach: **direct optimization using the subgradient at the argmax**.

The idea:
1. Compute conv = f*f
2. Find m* = argmax(conv)
3. Compute subgradient: grad[j] = 2*f[m*-j] (with zero padding)
4. Project grad onto the tangent space of the simplex: grad_tan = grad - (grad @ f) * ones
5. Take a step: f_new = f - lr * grad_tan
6. Project to simplex
7. Repeat

The key advantage: this uses the exact subgradient at the current max, not a smoothed version. It should converge faster to the local optimum.

But the issue is: the subgradient can change abruptly when the argmax changes, causing oscillations. The log-sum-exp smoothing avoids this by using a weighted average.

"""

- Hybrid exact/smoothed subgradient with Nesterov momentum

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_lag(f, m):
    """Compute subgradient of (f*f)[m] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, m - (n - 1))
    hi = min(n - 1, m)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * f[m - indices]
    return grad


def smooth_gradient_fft(f, T):
    """Compute gradient of log-sum-exp(conv(f,f)/T) via FFT cross-correlation."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    total = e.sum()
    w = e / total
    
    N = 1 << (3 * n - 2).bit_length()
    W_fft = np.fft.rfft(w, N)
    F_fft = np.fft.rfft(f, N)
    corr = np.fft.irfft(W_fft * np.conj(F_fft), N)
    
    grad = 2.0 * corr[:n]
    lse_val = T * (scaled.max() + np.log(total))
    return grad, lse_val, conv, w


def smooth_optimize_v2(f0, time_limit, T_start=0.5, T_end=0.005, alpha_init=0.01, verbose=False):

    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        # Exponential temperature decay
        T = max(T_end, T_start * np.exp(-3.0 * elapsed / time_limit))
        # Learning rate decay
        lr = alpha_init * (0.2 ** (elapsed / time_limit))
        
        grad, lse_val, conv, w = smooth_gradient_fft(f, T)
        
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)

def lag_targeted_optimize(f0, time_limit, K=5, verbose=False):
    """
    Optimize by targeting top K lags where convolution is near max.
    Uses linearized LP directions to find descent directions.
    """

    if s < 1e-15:

        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_expansion = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top K lags
        top_indices = np.argsort(conv)[-K:]
        
        # LP: minimize t s.t. for each m in top_K:
        #   grad_m @ g - t <= grad_m @ f - conv[m]  (linearization)
        #   sum(g) = 1, g >= 0, t >= 0

            A_ub[idx, n] = -1.0
            b_ub[idx] = grad @ f - conv[m]
        
        A_eq = np.zeros((1, n + 1))

        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:

            if res.success:

                            if st > 1e-15:

                                val = fast_convolve(ft).max()
                                if val < best_val:

                                    lo = mid
                                else:
                                    hi = mid
                        
                        f_new = (1.0 - best_alpha) * f + best_alpha * g
                        f_new = np.maximum(f_new, 0.0)

                        f = f_new

def exact_subgrad_optimize(f0, time_limit, verbose=False):

    velocity = np.zeros(n)
    mu = 0.9  # momentum
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0
        lr = 0.008 * (0.3 ** (elapsed / time_limit))
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Use top 3 lags for stability
        top_k = min(3, len(conv))
        top_indices = np.argsort(conv)[-top_k:]

        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-12:

        # Restart

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

def coordinate_descent(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

        order = rng.permutation(n)
        improved = False

            if time.monotonic() - t0 > time_limit:
                break

            if old_val < 1e-12 and rng.random() > 0.05:
                continue
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_local_val = old_val
            best_local_obj = fast_convolve(f).max()
            
            # Try multiplicative and additive perturbations
            candidates = [
                old_val * 0.3, old_val * 0.6, old_val * 0.8,
                old_val * 1.2, old_val * 1.5, old_val * 2.0,
                0.0, 0.01, 0.05,
            ]
            # Add context-dependent candidates
            mean_val = 1.0 / n
            candidates.extend([mean_val * 0.5, mean_val * 2.0])
            
            for cv in candidates:
                cv = max(0.0, min(0.9999, cv))
                if abs(cv - old_val) < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = cv
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - cv) / denom
                if (f_trial < 0).any():
                    continue
                obj = fast_convolve(f_trial).max()
                if obj < best_local_obj - 1e-14:
                    best_local_obj = obj
                    best_local_val = cv
            
            if best_local_val != old_val:
                f[j] = best_local_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= (1.0 - best_local_val) / denom
                f = f / f.sum()
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Perturbation to escape
            scale = 0.001 + 0.001 * (time.monotonic() - t0) / time_limit
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
    
    return best_f, best_obj


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    """Symmetric rising-falling geometric sequence."""
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    if half > 1:
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
    else:
        a = asc.copy()
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a)) * 0.01])
    a = a[:n].copy()
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_cosine(n, power=2.0):
    """Cosine window raised to a power."""
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.002
    return a / a.sum()


def make_feker(n):
    """Fejér kernel-like: (1 - k/n)^2."""
    k = np.arange(n, dtype=np.float64)
    a = (1.0 - k / n) ** 2
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_bell(n, sigma_frac=0.3):
    """Gaussian bell centered in the sequence."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def make_two_bump(n, pos1=0.3, pos2=0.7):
    """Two-bump distribution."""
    x = np.arange(n, dtype=np.float64)
    sigma = 0.1 * n
    a = np.exp(-0.5 * ((x - pos1 * n) / sigma) ** 2) + np.exp(-0.5 * ((x - pos2 * n) / sigma) ** 2)
    return a / a.sum()


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer combining:
    - Diverse structured initializations
    - Log-sum-exp smooth optimization (Adam + Nesterov)
    - Lag-targeted LP directions
    - Exact subgradient with momentum
    - Coordinate descent
    - Multi-length exploration
    - Final polishing with very low temperature
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass
    
    n_parent = len(parent) if parent is not None else 768
    
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}) [was {old_score:.6f}]")
    
    # === Phase 0: Diverse initializations ===
    print("=== Phase 0: Diverse Initializations ===")
    t_phase = time.monotonic()
    
    n_list = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1152]))
    
    for n_try in n_list:
        if time.monotonic() > deadline - 600:
            break
        for r in [0.96, 0.97, 0.98, 0.985, 0.99, 0.995]:
            _try(make_geometric(n_try, r), f"geo_{r}_{n_try}")
        for r in [0.95, 0.97, 0.985, 0.99]:
            _try(make_symmetric_geo(n_try, r), f"sym{r}_{n_try}")
        for pw in [1.5, 2.0, 2.5, 3.0, 4.0, 6.0]:
            _try(make_cosine(n_try, power=pw), f"cos{pw}_{n_try}")
        _try(make_feker(n_try), f"feker_{n_try}")
        for sf in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
            _try(make_bell(n_try, sigma_frac=sf), f"bell{sf}_{n_try}")
        _try(make_two_bump(n_try), f"2bump_{n_try}")
    
    print(f"  Phase 0 ({time.monotonic()-t_phase:.1f}s): best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # === Phase 1: Smooth optimization (Adam) on best ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem, 350)
        print(f"  Adam smooth n={len(best_seq)}, T:[0.4->0.003], t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_v2(best_seq, time_limit=t_alloc,
                                       T_start=0.4, T_end=0.003, alpha_init=0.012,
                                       verbose=True)
        _try(a_opt, "adam_smooth1")
    
    # Second run with different schedule
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  Adam smooth (slow) n={len(best_seq)}, T:[1.0->0.01], t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_v2(best_seq, time_limit=t_alloc,
                                       T_start=1.0, T_end=0.01, alpha_init=0.008,
                                       verbose=True)
        _try(a_opt, "adam_smooth2")
    
    # Nesterov
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem, 150)
        print(f"  Nesterov n={len(best_seq)}, T:[0.3->0.002], t={t_alloc:.0f}s")
        a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc,
                                      T_start=0.3, T_end=0.002, verbose=True)
        _try(a_opt, "nesterov")
    
    # === Phase 2: Lag-targeted LP ===
    print(f"\n=== Phase 2: Lag-Targeted LP (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t_alloc = min(rem, 180)
        print(f"  Lag-targeted K=5, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lag_targeted_optimize(best_seq, time_limit=t_alloc, K=5, verbose=True)
        _try(a_opt, "lag_K5")
    
    rem = deadline - time.monotonic() - 180
    if rem > 15:
        t_alloc = min(rem, 100)
        print(f"  Lag-targeted K=10, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lag_targeted_optimize(best_seq, time_limit=t_alloc, K=10, verbose=True)
        _try(a_opt, "lag_K10")
    
    # === Phase 3: Exact subgradient ===
    print(f"\n=== Phase 3: Exact Subgradient (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(rem, 120)
        print(f"  Exact subgrad n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = exact_subgrad_optimize(best_seq, time_limit=t_alloc, verbose=True)
        _try(a_opt, "exact_subgrad")
    
    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_cd = min(rem, 80)
        print(f"  CD n={len(best_seq)}, t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        _try(a_opt, "cd")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    
    n_best = len(best_seq) if best_seq is not None else 768
    target_lengths = []
    for delta in [-128, -64, -32, 32, 64, 128, 256]:
        tl = n_best + delta
        if 200 <= tl <= 2048:
            target_lengths.append(tl)
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 60
        if rem < 40:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        
        t1 = min(rem * 0.35, 50)
        a_opt, _ = smooth_optimize_v2(f_interp, time_limit=t1,
                                       T_start=0.3, T_end=0.005, alpha_init=0.01)
        _try(a_opt, f"smooth_n{target_n}")
    
    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very low temperature smooth
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 50)
            print(f"  Fine smooth T:[0.015->0.0005], t={t_alloc:.0f}s")
            a_opt, _ = smooth_optimize_v2(best_seq, time_limit=t_alloc,
                                           T_start=0.015, T_end=0.0005, alpha_init=0.005,
                                           verbose=True)
            _try(a_opt, "smooth_fine")
        
        # Final CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 25))
            _try(a_opt, "cd_final")
        
        # Small perturbations with very low T
        for pi in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise_scale = 0.0008 * (pi + 1)
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_optimize_v2(f_pert, time_limit=min(rem, 12),
                                           T_start=0.008, T_end=0.0005, alpha_init=0.003)
            _try(a_opt, f"perturb_{pi}")
        
        # Reversed
        rem = deadline - time.monotonic() - 8
        if rem > 6:
            f_rev = best_seq[::-1].copy()
            a_opt, _ = smooth_optimize_v2(f_rev, time_limit=min(rem, 8),
                                           T_start=0.008, T_end=0.0005, alpha_init=0.003)
            _try(a_opt, "reversed")
        
        # Blend with uniform
        rem = deadline - time.monotonic() - 6
        if rem > 5:
            for blend_frac in [0.05, 0.1, 0.2]:
                f_blend = (1 - blend_frac) * best_seq + blend_frac * (np.ones(n_best) / n_best)
                f_blend = project_simplex(f_blend)
                a_opt, _ = smooth_optimize_v2(f_blend, time_limit=min(rem, 5),
                                               T_start=0.005, T_end=0.0005, alpha_init=0.003)
                _try(a_opt, f"blend_{blend_frac}")
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
