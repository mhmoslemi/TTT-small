"""
Multi-stage optimizer using SCA-LP (Successive Convex Approximation with LP)
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
1. Direct SCA-LP: solve min t s.t. linearized quadratic constraints (not FW direction)
2. Adaptive active set: focus on near-maximal convolution positions
3. Multi-scale exploration with diverse initializations
4. Coordinate descent polishing
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def sca_lp_step(a, c, active_positions, t_bound=None):
    """
    Solve one SCA iteration:
    min t
    s.t. 2 * sum_i a[k-i] * x[i] - c[k] <= t for k in active_positions
         sum(x) = 1
         x >= 0
         t <= t_bound (optional upper bound)
    
    Returns (x, t) or (None, None) if infeasible.
    """
    n = len(a)
    # Variables: x[0..n-1], t
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: for each k in active_positions:
    #   2 * sum_i a[k-i] * x[i] - t <= c[k]
    #   => 2 * sum_i a[k-i] * x[i] - t <= c[k]
    #   => [2*a[k-0], 2*a[k-1], ..., 2*a[k-(n-1)], -1] . [x, t] <= c[k]
    
    A_ub = []
    b_ub = []
    
    for k in active_positions:
        row = np.zeros(n + 1)
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                row[i] = 2.0 * a[idx]
        row[-1] = -1.0  # coefficient of t
        A_ub.append(row)
        b_ub.append(c[k])
    
    if t_bound is not None:
        # t <= t_bound
        row = np.zeros(n + 1)
        row[-1] = 1.0
        A_ub.append(row)
        b_ub.append(t_bound)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x >= 0, t is free (or t >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    
    return None, None

            a = project_simplex(best_a + pert)
            c = fast_convolve(a)
            M = c.max()
            no_improve = 0
            active_frac = max(0.1, active_frac * 0.8)
