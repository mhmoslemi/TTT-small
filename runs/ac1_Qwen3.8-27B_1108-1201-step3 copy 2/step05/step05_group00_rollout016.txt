"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-method approach:
- Enhanced LP-Frank-Wolfe with reduced constraint sets and multi-direction exploration
- Subgradient method with backtracking and iterate averaging
- Diverse initializations (asymmetric, two-bump, cosine-modulated)
- Multi-length exploration with interpolation
- Structured perturbations to escape local minima

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n, top_k=None):
    """Build LP constraint matrix. If top_k is set, only include constraints for top_k peaks."""
    conv_f = fast_convolve(f)
    if top_k is not None and top_k < len(conv_f):
        peak_indices = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(len(conv_f))
    
    num_cons = len(peak_indices)
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            # A_ub[idx, js:je+1] = f[k - je : k - js + 1] reversed
            # (f*g)[k] = sum_{j=js}^{je} f[k-j] * g[j]
            A_ub[idx, js:je+1] = f[k-je:k-js+1][::-1]
            b_ub[idx] = 1.0  # We'll scale by M later
    
    return A_ub, b_ub, peak_indices, conv_f


def solve_lp_direction(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected k, g >= 0."""
    num_cons = min(top_k, 2*n-1) if top_k else (2*n-1)
    A_ub = np.zeros((num_cons, n))
    b_ub = np.full(num_cons, M)
    
    if top_k:
        conv_f = fast_convolve(f)
        peak_indices = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(2*n-1)
    
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A_ub[idx, js:je+1] = f[k-je:k-js+1][::-1]
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            current_peak = conv.max()
            f = f_new

            if current_peak < best_obj - 1e-14:
                best_obj = current_peak
                best_f = f.copy()
                no_improve = 0
                # Tighten M when improving
                m_expand = max(1.01, m_expand * 0.95)
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, M_exp={m_expand:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_expand = min(m_expand * 1.03, 3.0)

        if no_improve > 100:
            f = _structured_restart(f, n, it)
            f = f / f.sum()
            conv = fast_convolve(f)
            current_peak = conv.max()
            m_expand = 1.01
            no_improve = 0

    return best_f, best_obj


def _structured_restart(f, n, seed):
    """Create a perturbed version of f for restart."""
    rng = np.random.default_rng(seed * 7919 + 3)
    # Mix of small random perturbation and structured (low-freq) perturbation
    noise_scale = 0.003
    random_noise = rng.standard_normal(n) * noise_scale
    
    # Low-frequency structured perturbation
    freqs = rng.integers(1, 5, size=3)
    phases = rng.uniform(0, 2*np.pi, size=3)
    struct_noise = np.zeros(n)
    for freq, phase in zip(freqs, phases):
        struct_noise += np.sin(2*np.pi*freq*np.arange(n)/n + phase)
    struct_noise = struct_noise * noise_scale * 0.5
    
    f_new = f + random_noise + struct_noise
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def subgradient_optimize(f0, time_limit, verbose=False, num_avg=50):
    """
    Subgradient method with softmax weighting over peaks,
    backtracking line search, and iterate averaging.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Maintain average of recent iterates
    avg_f = f.copy()
    avg_count = 0
    step_size = 0.1
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temperature = 0.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2:
            break

        conv = fast_convolve(f)
        current_peak = conv.max()

        # Select top peaks with softmax weighting
        k_top = min(30, len(conv))
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peak_idx]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Backtracking line search
        best_ft = f.copy()
        best_val = current_peak
        alpha = step_size
        for _ in range(8):
            ft = f - alpha * sg
            ft = np.maximum(ft, 0.0)
            ts = ft.sum()
            if ts < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / ts
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
            alpha *= 0.5

        f = best_ft
        step_size = min(step_size * 1.1, 0.5)

        # Update average
        avg_f = (avg_count * avg_f + f) / (avg_count + 1)
        avg_count += 1
        if avg_count > num_avg:
            avg_f = avg_f  # Keep running average (it's already the average)

        # Check improvement
        avg_conv = fast_convolve(avg_f)
        avg_peak = avg_conv.max()
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 30 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, avg={avg_peak:.10f}")
        elif avg_peak < best_obj - 1e-14:
            best_obj = avg_peak
            best_f = avg_f.copy()
            no_improve = 0
            if verbose and it % 30 == 0:
                print(f"    SG avg iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temperature *= 1.003

        if no_improve > 200:
            # Restart from best with perturbation
            f = _structured_restart(best_f, n, it)
            f = f / f.sum()
            avg_f = f.copy()
            avg_count = 0
            step_size = 0.1
            no_improve = 0

    return best_f, best_obj


def make_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    """Gaussian initialization with optional off-center."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump(n, sep_frac=0.3, sigma_frac=0.1):
    """Two Gaussian bumps."""
    a = np.zeros(n)
    c1 = n * (0.5 - sep_frac / 2)
    c2 = n * (0.5 + sep_frac / 2)
    sigma = n * sigma_frac
    a += np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2)
    a += np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_modulated(n, sigma_frac=0.25, mod_freq=3):
    """Gaussian modulated by cosine."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    mod = 0.7 + 0.3 * np.cos(2 * np.pi * mod_freq * np.arange(n) / n)
    a = gauss * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gaussian(n, center_frac=0.35, sigma_frac=0.22):
    """Off-center Gaussian."""
    return make_gaussian(n, center_frac=center_frac, sigma_frac=sigma_frac)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
