import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fienup_optimize(f0, time_limit, alpha=0.5, relax=0.5, verbose=False):
    """
    Fienup/PSO-type iteration for minimizing max(conv(a,a)).
    Alternates between:
    - Time domain: project to simplex (a >= 0, sum = 1)
    - Convolution domain: flatten the convolution
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_alpha = alpha
    cur_relax = relax
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Forward: compute convolution via FFT
        fft_size = 1 << (2*n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        C = A * A  # This is FFT of the zero-padded convolution
        c = np.fft.irfft(C, fft_size)[:2*n-1]
        
        # Modify: flatten c
        # c_flat = c^alpha (power compression)
        c_flat = np.maximum(c, 0) ** cur_alpha
        # Also mix with uniform
        c_target = np.full(2*n-1, 1.0/(2*n-1))
        c_flat = (1 - 0.3) * c_flat + 0.3 * c_target
        
        # Backward: recover a from c_flat
        C_flat = np.fft.rfft(c_flat, fft_size)
        # Take square root (element-wise)
        # C_flat should be approximately non-negative real if c is a valid convolution
        # But after modification, it might not be. Take complex sqrt.
        B_new = np.sqrt(C_flat + 0j)
        a_new = np.fft.irfft(B_new, fft_size)[:n]
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        # Relaxed update
        a = (1 - cur_relax) * a + cur_relax * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    Fienup iter {it}: obj={best_obj:.10f}, alpha={cur_alpha:.3f}")
        
        # Anneal: gradually reduce compression
        cur_alpha = max(0.1, cur_alpha * 0.999)
        cur_relax = min(1.0, cur_relax * 1.001)
        
        if it % 500 == 0 and obj > best_obj * 1.01:
            # Stuck, restart with perturbation
            rng = np.random.default_rng(it)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            cur_alpha = alpha
    
    return best_a, best_obj

"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def fienup_optimize(f0, time_limit, verbose=False):
    """
    Fienup/PSO-type iteration: alternates between simplex projection (time domain)
    and magnitude flattening (frequency domain) to minimize max(conv(a,a)).
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    fft_size = 1 << (2 * n - 1).bit_length()

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Frequency domain modification: compress dynamic range of |A|
        # Target: |A[k]| proportional to Dirichlet kernel of length 2n-1
        target_mag = np.ones_like(mag)
        for k in range(len(mag)):
            freq = 2 * np.pi * k / fft_size
            if abs(np.sin(freq / 2)) > 1e-15:
                target_mag[k] = abs(np.sin((2 * n - 1) * freq / 2) / np.sin(freq / 2))
            else:
                target_mag[k] = 2 * n - 1
        target_mag /= target_mag[0] if target_mag[0] > 1e-15 else 1.0

        # Blend: oscillate between pure compression and target-matching
        frac = (it % 200) / 200.0
        if frac < 0.5:
            # Power compression
            alpha = 0.5 + 0.5 * frac
            mag_new = mag ** alpha
            mag_new = mag_new / mag_new.max() * mag.max()
        else:
            # Match target magnitude with some mixing
            blend = 0.2 + 0.6 * (frac - 0.5)
            mag_new = (1 - blend) * mag + blend * target_mag * (mag.max() / target_mag.max())

        # Reconstruct with phase
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        # Relaxed update
        relax = 0.4 + 0.3 * (it % 100) / 100.0
        a = (1 - relax) * a + relax * a_new
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a / a_sum

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Fienup iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.02)
            no_improve = 0

    return best_a, best_obj


def fienup_v2(f0, time_limit, verbose=False):

        A_recovered = np.sqrt(mag_C) * np.exp(1j * phase_C)

        a_new = np.fft.irfft(A_recovered, fft_size)[:n]

        # Project
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_obj_grad_impl(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t

def sg_optimize(f0, time_limit, verbose=False):
    """Subgradient optimization with top-k peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = 0.003
    cur_temp = 0.00005
    top_k = max(20, n // 8)

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = 0.94 * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and co < 1.52:
                print(f"    SG: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(0.001, cur_lr * 0.5)
            no_improve = 0

    return best_f, best_obj


def anneal_optimize(f0, time_limit, verbose=False):
    """
    Simulated annealing: randomly perturb coordinates and accept based on
    energy (max of convolution). Uses adaptive temperature.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    temp = 0.01
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Random perturbation: pick a few coordinates and modify them
        n_perturb = max(3, n // 50)
        idx = rng.choice(n, size=n_perturb, replace=False)
        deltas = rng.standard_normal(n_perturb) * temp
        a_trial = a.copy()
        a_trial[idx] += deltas
        a_trial = np.maximum(a_trial, 0.0)
        a_sum = a_trial.sum()
        if a_sum < 1e-15:
            continue
        a_trial = a_trial / a_sum

        trial_obj = fast_convolve(a_trial).max()
        delta = trial_obj - cur_obj

        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-15)):
            a = a_trial
            cur_obj = trial_obj
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and best_obj < 1.52:
                    print(f"    Anneal: obj={best_obj:.10f}, T={temp:.6f}")
        else:
            no_improve += 1

        # Cool down
        temp *= 0.9995
        if no_improve > 500:
            temp = 0.01
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            cur_obj = fast_convolve(a).max()
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_cosine_window(n):
    """Cosine-tapered window normalized to sum 1."""
    j = np.arange(n)
    a = np.cos(np.pi * j / (2 * n)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian_window(n, sigma_frac=0.3):
    """Gaussian window normalized to sum 1."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_chebyshev_window(n):
    """Chebyshev (raised cosine) window."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.ones(1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    # === Phase 2: Fienup v2 + LP-FW on best (400s) ===
    print(f"\n=== Phase 2: Fienup2 + LP-FW (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(150, rem * 0.4)
        a_opt, _ = fienup_v2(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "fienup2_main")

    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(150, rem * 0.5)
        a_opt, _ = lp_fw_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lpfw_main")

    # === Phase 3: Multi-length exploration (300s) ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [512, 896, 1024, 1280, 1536]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 350
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        # Also try cosine window at this length
        f_cos = make_cosine_window(target_n)

        # Fienup on interpolated
        t_alloc = min(rem * 0.3, 60)
        if t_alloc > 10:
            a_opt, _ = fienup_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"fienup_len{target_n}")

        # Fienup on cosine
        t_alloc = min(rem * 0.2, 40)
        if t_alloc > 10:
            a_opt, _ = fienup_optimize(f_cos, time_limit=t_alloc)
            try_seq(a_opt, f"fienup_cos_len{target_n}")

    # === Phase 4: L-BFGS refinement (250s) ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(200, rem * 0.7)
        a_opt, _ = lbfqg_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lbfqg_main")

    # L-BFGS from perturbed
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        noise = rng.standard_normal(len(base)) * 0.01
        f_pert = project_simplex(base + noise)
        t_alloc = min(100, rem * 0.6)
        a_opt, _ = lbfqg_optimize(f_pert, time_limit=t_alloc)
        try_seq(a_opt, "lbfqg_pert")

    # === Phase 5: SG + Anneal (200s) ===
    print(f"\n=== Phase 5: SG + Anneal (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(100, rem * 0.5)
        a_opt, _ = sg_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "sg_main")

    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(80, rem * 0.6)
        a_opt, _ = anneal_optimize(base, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "anneal_main")

    # === Phase 6: Deep optimization on best (remaining) ===
    print(f"\n=== Phase 6: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Deep Fienup
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(100, rem * 0.5)
            a_opt, _ = fienup_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fienup")

        # Deep Fienup v2
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(80, rem * 0.5)
            a_opt, _ = fienup_v2(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fienup2")

        # Deep LP-FW
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(60, rem * 0.6)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # Deep L-BFGS
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(40, rem * 0.7)
            a_opt, _ = lbfqg_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lbfqg")

        # Final SG
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_sg")

        # Final anneal
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = anneal_optimize(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "final_anneal")

    # Fallback
    if best_seq is None:
        best_seq = make_cosine_window(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
