import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgradient(p, top_k=5):
    """Compute exact subgradient of max(p*p) weighted at top-k lag positions."""
    conv = fast_convolve(p)
    n = len(p)
    m = len(conv)
    
    # Get top-k positions with weights based on proximity to max
    max_val = conv.max()
    indices = np.argsort(conv)[-top_k:]
    # Weight: closer to max gets higher weight
    weights = np.zeros(m)
    for k in indices:
        w = (conv[k] - max_val + 1e-30)
        w = max(0.0, (conv[k] / max(max_val, 1e-15)) ** 8)
        weights[k] = w
    ws = weights.sum()
    if ws < 1e-30:
        weights[indices[0]] = 1.0
        ws = 1.0
    weights /= ws
    
    # Compute subgradient: for each k, d(p*p)[k]/dp[j] = p[k-j]*I(0<=k-j<n) + p[j]*I(k-j!=j)
    sg = np.zeros(n)
    for k in indices:
        wk = weights[k]
        if wk < 1e-15:
            continue
        # Vectorized: p[k-j] for j=0..n-1 where 0<=k-j<n
        # k-j valid when j in [k-(n-1), k] intersect [0, n-1]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # p[k-j] for j in [lo, hi] is p[k-hi : k-lo+1] reversed
        rev = p[lo:hi + 1][::-1]
        sg[lo:hi + 1] += wk * rev
        # p[j] for j where k-j != j and 0<=k-j<n, i.e., j != k/2
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n and 0 <= k - mid < n:
                # j=mid has k-j=j, so no extra p[j] term (already counted as 2*p[mid] in rev)
                pass
        # Add p[j] for all j in [lo,hi] where k-j != j
        # k-j != j means j != k/2
        for j in range(lo, hi + 1):
            if k - j != j:
                sg[j] += wk * p[j]
    
    return sg, conv, max_val

def golden_section_line_search(p, j, f_current, n):
    """1D golden-section search along coordinate j (scaling, with sum constraint)."""
    # We fix all other coordinates and scale them to maintain sum=1
    # p[j] = t, others scaled by (1-t)/(1-p[j])
    old_val = p[j]
    if old_val < 1e-12 or old_val > 0.999:
        return old_val, f_current
    
    def obj_1d(t):
        if t < 0 or t > 0.999:
            return 1e10
        denom = 1.0 - old_val
        if denom < 1e-12:
            return 1e10
        scale = (1.0 - t) / denom
        p_trial = p.copy()
        p_trial[j] = t
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        p_trial[mask] *= scale
        p_trial = np.maximum(p_trial, 0)
        s = p_trial.sum()
        if s < 1e-15:
            return 1e10
        p_trial /= s
        return fast_convolve(p_trial).max()
    
    # Search in [max(0, old_val*0.1), min(0.5, old_val*3)]
    lo = max(0.0, old_val * 0.1)
    hi = min(0.99, old_val * 3.0)
    if hi <= lo:
        hi = lo + 0.01
    if hi > 0.99:
        hi = 0.99
    
    result = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                             options={'xatol': 1e-8, 'maxiter': 50})
    
    if result.success and result.fun < f_current - 1e-14:
        t_opt = result.x
        denom = 1.0 - old_val
        if denom > 1e-12:
            scale = (1.0 - t_opt) / denom
            p_new = p.copy()
            p_new[j] = t_opt
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            p_new[mask] *= scale
            p_new = np.maximum(p_new, 0)
            s = p_new.sum()
            if s > 1e-15:
                p_new /= s
            return p_new, result.fun
    return p, f_current

        active_coords = set()
        for k in active_lags:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                active_coords.add(j)
        
        active_list = list(active_coords)
        # Sort by contribution (larger values first)
        active_list.sort(key=lambda j: -p[j])
        
        improved = False
        for j in active_list:
            if time.monotonic() - t0 > time_limit:
                break
            if total_evals > 50000:
                break
            
            p, obj = golden_section_line_search(p, j, fast_convolve(p).max(), n)
            total_evals += 50  # approximate
            conv_now = fast_convolve(p).max()
            if conv_now < best_obj - 1e-14:
                best_obj = conv_now
                best_p = p.copy()
                improved = True
        
        if verbose and (passes + 1) % 10 == 0:
            print(f"      CD pass {passes+1}: obj={best_obj:.10f}, active_coords={len(active_list)}")
        
        if not improved:
            # Try a small random perturbation to escape
            rng = np.random.default_rng(passes * 17 + 3)
            p = project_simplex(best_p + rng.standard_normal(n) * 0.002)
    
    return best_p, best_obj


def try_scipy_cobyła(p0, time_limit, verbose=False):
    """Scipy COBYLA optimization on the full sequence."""
    p = p0.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    def objective(x):
        xi = np.maximum(x, 0)
        s = xi.sum()
        if s < 1e-15:
            return 1e10
        xi = xi / s
        return fast_convolve(xi).max()
    
    t0 = time.monotonic()
    
    try:
        # COBYLA with time limit via maxiter
        # Each eval takes ~O(n log n), so for n=768, ~1ms per eval
        # Allow at most time_limit/0.002 = ~5000 evals max, but COBYLA maxiter is per function eval
        max_iter = min(500, int(time_limit / 0.005))
        result = scipy_minimize(objective, p, method='COBYLA',
                                options={'maxiter': max_iter, 'rhobeg': 0.005, 'catol': 1e-10})
        if np.isfinite(result.fun) and result.fun < best_obj:
            best_obj = result.fun
            best_p = np.maximum(result.x, 0)
            s = best_p.sum()
            if s > 1e-15:
                best_p /= s
    except Exception as e:
        if verbose:
            print(f"      COBYLA error: {e}")
    
    if verbose:
        print(f"      scipy COBYLA: obj={best_obj:.10f}")
    
    return best_p, best_obj


def make_gaussian_init(n, center=0.5, sigma=0.15):
    """Gaussian-like initialization."""
    x = np.linspace(0, 1, n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_symmetric_init(n, shape='bump'):
    """Symmetric initialization."""
    if shape == 'bump':
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    elif shape == 'plateau':
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.5) / 0.3) ** 4)
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_smooth_random_init(n, rng, n_components=5):
    """Smooth random initialization as sum of Gaussians."""
    f = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(n_components):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.05, 0.25)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining exact subgradient descent,
    active coordinate descent, scipy local optimization, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform init n=768")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick diverse exploration (200s) ===
    print(f"\n=== Phase 1: Diverse exploration (n={n_parent}) ===")
    t1_end = time.monotonic() + 200
    
    initializations = {
        'parent': parent.copy(),
        'gauss_012': make_gaussian_init(n_parent, 0.5, 0.12),
        'gauss_020': make_gaussian_init(n_parent, 0.5, 0.20),
        'gauss_030': make_gaussian_init(n_parent, 0.5, 0.30),
        'bump': make_symmetric_init(n_parent, 'bump'),
        'plateau': make_symmetric_init(n_parent, 'plateau'),
        'smooth_rand': make_smooth_random_init(n_parent, rng, 5),
        'parent_pert1': project_simplex(parent + rng.standard_normal(n_parent) * 0.01),
        'parent_pert2': project_simplex(parent + rng.standard_normal(n_parent) * 0.03),
    }
    
    for name, init in initializations.items():
        rem = t1_end - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.4, 20)
        print(f"  Phase 1: '{name}', t={t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(init, time_limit=t_alloc,
                                          momentum=0.7, initial_step=0.8/np.sqrt(n_parent))
        try_seq(a_opt, f"p1_{name}")
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # === Phase 2: Focused optimization on best (500s) ===
    print(f"\n=== Phase 2: Focused optimization (best={best_score:.8f}, n={len(best_seq)}) ===")
    t2_end = time.monotonic() + 500
    n_best = len(best_seq)
    
    # Multiple subgradient runs with different hyperparameters
    configs = [
        {'momentum': 0.7, 'step': None, 'tk': None},
        {'momentum': 0.8, 'step': None, 'tk': 5},
        {'momentum': 0.6, 'step': None, 'tk': 10},
        {'momentum': 0.85, 'step': 0.5/np.sqrt(n_best), 'tk': 3},
        {'momentum': 0.75, 'step': None, 'tk': 8},
    ]
    
    for ci, cfg in enumerate(configs):
        rem = t2_end - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.5, 100)
        
        # Start from best or perturbed best
        if ci % 2 == 0:
            start = best_seq.copy()
        else:
            start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
        
        print(f"  Phase 2: SGD config {ci}, t={t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(start, time_limit=t_alloc,
                                          momentum=cfg['momentum'],
                                          initial_step=cfg['step'],
                                          top_k_override=cfg['tk'])
        try_seq(a_opt, f"p2_sgd_{ci}")
    
    # Coordinate descent on current best
    rem = t2_end - time.monotonic()
    if rem > 50:
        t_cd = min(rem * 0.7, 150)
        print(f"  Phase 2: Active CD, t={t_cd:.0f}s")
        a_opt, obj = coordinate_descent_active(best_seq, time_limit=t_cd)
        try_seq(a_opt, "p2_cd")
    
    # === Phase 3: Length exploration (350s) ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + 350
    
    target_lengths = [512, 640, 768, 896, 1024, 1152, 1280]
    target_lengths = [t for t in target_lengths if t != n_best and 256 <= t <= 1536]
    target_lengths.sort(key=lambda x: abs(x - n_best))
    target_lengths = target_lengths[:4]
    
    for target_n in target_lengths:
        rem = t3_end - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.45, 80)
        
        print(f"  Phase 3: n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = subgradient_descent(f_interp, time_limit=t_alloc,
                                          momentum=0.75, initial_step=0.8/np.sqrt(target_n))
        try_seq(a_opt, f"p3_n{target_n}")
        
        # Quick CD
        rem = t3_end - time.monotonic()
        if rem > 20 and best_seq is not None:
            if len(best_seq) == target_n:
                a_opt, obj = coordinate_descent_active(best_seq, time_limit=min(rem*0.4, 40))
                try_seq(a_opt, f"p3_cd_n{target_n}")
    
    # === Phase 4: Deep local optimization (400s) ===
    print(f"\n=== Phase 4: Deep local optimization (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + 400
    n_best = len(best_seq)
    
    # Multiple rounds of SGD -> CD -> SGD
    for round_i in range(3):
        rem = t4_end - time.monotonic()
        if rem < 30:
            break
        
        # SGD with fine parameters
        t_sgd = min(rem * 0.4, 100)
        print(f"  Phase 4 round {round_i}: SGD t={t_sgd:.0f}s")
        start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
        a_opt, _ = subgradient_descent(start, time_limit=t_sgd,
                                        momentum=0.8, initial_step=0.3/np.sqrt(n_best))
        try_seq(a_opt, f"p4r{round_i}_sgd")
        
        # CD
        rem = t4_end - time.monotonic()
        if rem > 20:
            t_cd = min(rem * 0.4, 80)
            print(f"  Phase 4 round {round_i}: CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_active(best_seq, time_limit=t_cd)
            try_seq(a_opt, f"p4r{round_i}_cd")
    
    # === Phase 5: scipy COBYLA (if time permits) ===
    print(f"\n=== Phase 5: scipy local opt (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 60 and len(best_seq) <= 1024:
        t_scipy = min(rem * 0.5, 120)
        print(f"  Phase 5: scipy COBYLA, t={t_scipy:.0f}s")
        a_opt, _ = try_scipy_cobyla_patched(best_seq, time_limit=t_scipy, verbose=True)
        try_seq(a_opt, "p5_scipy")
    
    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Very fine SGD
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_alloc = min(rem * 0.5, 60)
            print(f"  Final SGD, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_descent(best_seq, time_limit=t_alloc,
                                            momentum=0.85, initial_step=0.1/np.sqrt(n_best),
                                            top_k_override=3)
            try_seq(a_opt, "final_sgd")
        
        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for pi in range(4):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = subgradient_descent(f_pert, time_limit=min(rem * 0.4, 15),
                                            momentum=0.8, initial_step=0.05/np.sqrt(n_best),
                                            top_k_override=3)
            try_seq(a_opt, f"final_pert_{pi}")
        
        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent_active(best_seq, time_limit=min(rem * 0.8, 30))
            try_seq(a_opt, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
