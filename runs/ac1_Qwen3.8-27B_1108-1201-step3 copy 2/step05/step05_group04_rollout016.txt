"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
parametric initialization, subgradient with top-k active sets,
incremental coordinate descent polishing, and length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_all_R(a):
    """Compute all cross-correlation values R(m) = sum_j a[j]*a[j+m] for m=0..n-1.
    Returns array of length n."""
    n = len(a)
    # R(m) = conv(a, reversed(a))[n-1+m] for m >= 0
    # Actually, let's use the fact that conv(a,a)[n-1+m] = R(m) for m >= 0
    # and conv(a,a)[n-1-m] = R(m) for m >= 0 (by symmetry)
    # So we just need the upper half of the convolution.
    conv = fast_convolve(a)
    return conv[n-1:n-1+n]  # R(0), R(1), ..., R(n-1)


def compute_R_grad(a, m):
    """Compute gradient of R(m) = sum_j a[j]*a[j+m] with respect to a.
    Returns array of length n."""
    n = len(a)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 2.0 * a
    else:
        # a[i] appears in a[i]*a[i+m] (if i+m < n) and a[i-m]*a[i] (if i-m >= 0)
        grad[:n-m] += a[m:]      # a[i+m] for i = 0..n-1-m
        grad[m:] += a[:n-m]      # a[i-m] for i = m..n-1
    return grad


def compute_subgradient(a, top_k):
    """Compute subgradient of max_m R(m) using top-k active set."""
    n = len(a)
    conv = fast_convolve(a)
    R = conv[n-1:n-1+n]
    
    # Find top-k positions
    if top_k is None or top_k >= n:
        k = n
    else:
        k = min(top_k, n)
    
    # Get top-k indices
    if k == n:
        top_indices = np.arange(n)
        weights = np.ones(n) / n
    else:
        top_indices = np.argpartition(R, -k)[-k:]
        # Get weights proportional to R values (or uniform)
        top_R = R[top_indices]
        if top_R.max() > 0:
            weights = top_R / top_R.sum()
        else:
            weights = np.ones(k) / k
    
    # Average gradients
    sg = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        sg += w * compute_R_grad(a, idx)
    
    return sg, conv.max(), R


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
