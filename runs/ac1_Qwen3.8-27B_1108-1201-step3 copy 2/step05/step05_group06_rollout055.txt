import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, active_set):
    """
    Compute subgradient of max_k conv(a,a)[k] using the given active set.

def smooth_logsumexp_opt(a0, time_limit, T_start=0.3, T_end=0.005, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / max(time_limit, 1e-6))
        T = max(T, T_end)
        lr = alpha * (0.5 ** (elapsed / (time_limit * 0.7)))
        
        # Compute smooth objective and gradient
        b = fast_convolve(a)
        scaled = b / T
        scaled_max = scaled.max()
        e = np.exp(scaled - scaled_max)
        w = e / e.sum()
        
        # Gradient: g[j] = (2/T) * sum_k w[k] * a[k-j]
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(a, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T) * corr
        
        # Adam update
        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        
        cur_obj = fast_convolve(a).max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            rng_r = np.random.default_rng(int(elapsed * 1000) % 1000000)
            # Structured restart
            freq = int(rng_r.integers(1, max(2, n // 6)))
            amp = 0.005 + 0.005 * min(no_improve // 200, 3)
            phase = float(rng_r.uniform(0, 2 * np.pi))
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a = project_simplex(best_a + pert)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    ORDER = 0.6180339887
    
    # Optimize "heavy" coordinates first (those with larger values)
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Process coordinates in order of magnitude (descending)
        order = np.argsort(a)[::-1]
        
        for idx, j in enumerate(order):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Golden section search for a[j]
            # We fix sum=1, so if a[j] changes, others scale proportionally
            lo = 0.0
            hi = 0.999
            
            def eval_at(x):
                """Evaluate max conv when a[j]=x, others proportional."""
                denom = 1.0 - old_val
                if denom < 1e-10:
                    return 1e10
                if x >= 1.0 or x < 0:
                    return 1e10
                scale = (1.0 - x) / denom
                a_trial = a.copy()
                a_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                return fast_convolve(a_trial).max()
            
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)
            
            for _ in range(12):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_at(d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_at(opt_val)
            
            if opt_obj < best_obj - 1e-13:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    a[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def parametric_search(time_limit, n_values, verbose=False):
    """
    Search over a parametric family of sequences.
    a[j] = w(j) * (1 + sum_m c_m * cos(2*pi*m*j/n + phi_m)) / norm
    """
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()
    
    for n in n_values:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Try different base shapes
        base_shapes = []
        
        # Uniform
        base_shapes.append(np.ones(n))
        
        # Cosine windows
        for power in [1.5, 2.0, 3.0, 4.0, 6.0]:
            x = np.linspace(0, np.pi, n)
            base_shapes.append(np.sin(x) ** power + 0.01)
        
        # Geometric (asymmetric)
        for r in [0.95, 0.97, 0.99, 0.995]:
            base_shapes.append(r ** np.arange(n, dtype=np.float64))
        
        # Symmetric geometric (triangle-like)
        half = (n + 1) // 2
        for r in [0.93, 0.96, 0.98, 0.99]:
            asc = r ** np.arange(half, dtype=np.float64)
            desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
            shape = np.concatenate([asc, desc])
            if len(shape) < n:
                shape = np.concatenate([shape, np.ones(n - len(shape))])
            shape = shape[:n]
            base_shapes.append(shape)
        
        # Trapezoid
        for w_frac in [0.3, 0.5, 0.7]:
            w_start = int(n * (1 - w_frac) / 2)
            w_end = int(n * (1 + w_frac) / 2)
            shape = np.zeros(n)
            shape[:w_start] = np.linspace(0, 1, w_start + 1)[:w_start] if w_start > 0 else 0
            shape[w_start:w_end] = 1.0
            shape[w_end:] = np.linspace(1, 0, n - w_end) if n - w_end > 0 else 0
            shape = np.maximum(shape, 0.001)
            base_shapes.append(shape)
        
        # Exponential decay from center
        center = n // 2
        for sigma in [n/4, n/3, n/2, n/3, 2*n/5]:
            x = np.arange(n) - center
            shape = np.exp(-x**2 / (2*sigma**2))
            base_shapes.append(shape)
        
        # Modulated shapes
        for mod_freq in [1, 2, 3]:
            base = np.sin(np.linspace(0, np.pi, n)) + 0.1
            for mod_depth in [0.1, 0.2, 0.3, 0.4]:
                mod = 1 + mod_depth * np.cos(2 * np.pi * mod_freq * np.arange(n) / n)
                base_shapes.append(np.maximum(base * mod, 0.001))
        
        for shape in base_shapes:
            if time.monotonic() - t0 > time_limit:
                break
            shape = np.maximum(shape, 0)
            ss = shape.sum()
            if ss < 1e-15:
                continue
            shape = shape / ss
            b = fast_convolve(shape)
            score = 2 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = shape.copy()
    
    return best_a, best_score

def lp_frank_wolfe(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    M = b.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_boost = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M*m_boost for all k, g >= 0
        # (a*g)[k] = sum_j a[j] * g[k-j]
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_boost)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                m_boost = min(m_boost * 1.1, 5.0)
                continue
            g = res.x
        except Exception:
            m_boost = min(m_boost * 1.1, 5.0)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check if g is close to a
        if np.max(np.abs(g - a)) < 1e-8:
            no_improve += 1
            if no_improve > 50:
                break
            m_boost = min(m_boost * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_boost = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_boost = min(m_boost * 1.08, 4.0)
        
        if no_improve > 80:
            rng_r = np.random.default_rng(it * 99 + 7)
            scale = 0.004 + 0.002 * min(no_improve // 50, 5)
            noise = rng_r.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            M = fast_convolve(a).max()
            m_boost = 1.0
            no_improve = 0
        
        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, boost={m_boost:.2f}")
    
    return best_a, best_obj

def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: parametric family exploration, subgradient mirror descent (multiplicative
weights), log-sum-exp smooth optimization with Adam, LP Frank-Wolfe, and
coordinate descent. Uses structured low-frequency perturbations for restarts
and adaptive time allocation across phases.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, active_set):
    """
    Compute subgradient of max_k conv(a,a)[k] using the given active set.
    g[j] = (2/|S|) * sum_{k in S} a[k-j] (with a[m]=0 for m outside [0,n-1])
    """
    n = len(a)
    S_size = len(active_set)
    if S_size == 0:
        return np.zeros(n)
    indicator = np.zeros(2 * n - 1)
    indicator[active_set] = 1.0 / S_size
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    I_fft = np.fft.rfft(indicator, N)
    corr = np.fft.irfft(A_fft * np.conj(I_fft), N)[:n]
    return 2.0 * corr


def subgradient_mirror_descent(a0, time_limit, verbose=False):
    """
    Mirror descent (multiplicative weights) with subgradient of max-convolution.
    Uses adaptive step size and structured restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.8
    no_improve = 0
    it = 0
    decay_rate = 0.9995

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        # Active set with slight tolerance
        threshold = max_b * (1.0 - 1e-5) - 1e-12
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        g = compute_subgradient(a, active)
        # Multiplicative update
        exponent = -eta * g
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        # Backtracking line search
        best_alpha = 1.0
        best_val = max_b
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * a_new
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            eta *= 0.7
        # Adaptive step size
        eta = max(eta * decay_rate, 0.005)
        # Structured restart if stuck
        if no_improve > 150:
            rng = np.random.default_rng(it * 7919 + 13)
            freq = int(rng.integers(1, max(2, n // 6)))
            phase = float(rng.uniform(0, 2.0 * np.pi))
            amp = 0.008 + 0.004 * min(no_improve // 100, 4)
            pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
            a = project_simplex(best_a + pert)
            no_improve = 0
            eta = 0.5
        if verbose and it % 80 == 0:
            print(f"    SubGrad iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, |S|={len(active)}")
    return best_a, best_obj


def smooth_logsumexp_opt(a0, time_limit, T_start=0.3, T_end=0.004, verbose=False):
    """Optimize using log-sum-exp smooth approximation with Adam and structured restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    alpha_lr = 0.012
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / max(time_limit, 1e-9))
        T = max(T, T_end)
        lr = alpha_lr * (0.4 ** (elapsed / (time_limit * 0.75)))
        b = fast_convolve(a)
        scaled = b / T
        scaled_max = scaled.max()
        e = np.exp(scaled - scaled_max)
        w = e / e.sum()
        # Gradient via FFT cross-correlation
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(a, N)
        W_fft = np.fft.rfft(w, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T) * corr
        # Adam update
        t_adam += 1
        m_state = beta1 * m_state + (1.0 - beta1) * grad
        v_state = beta2 * v_state + (1.0 - beta2) * grad * grad
        m_hat = m_state / (1.0 - beta1 ** t_adam)
        v_hat = v_state / (1.0 - beta2 ** t_adam)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
        cur_obj = fast_convolve(a).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Structured restart
        if no_improve > 400:
            rng_r = np.random.default_rng(int(elapsed * 137) % 1000000)
            freq = int(rng_r.integers(1, max(2, n // 5)))
            amp = 0.004 + 0.003 * min(no_improve // 200, 5)
            phase = float(rng_r.uniform(0, 2.0 * np.pi))
            pert = amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            a = project_simplex(best_a + pert)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0
        if verbose and it % 200 == 0:
            print(f"    Smooth iter {it}: obj={best_obj:.10f}, T={T:.5f}")
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with golden section search on heavy coordinates first."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    ORDER = 0.6180339887
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        # Process coordinates in descending order of value
        order = np.argsort(a)[::-1]
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            # Golden section search
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            lo = 0.0
            hi = 0.999
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)

            def eval_c(cx):
                scale = (1.0 - cx) / denom
                a_t = a.copy()
                a_t[j] = cx
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_t[mask] *= scale
                return fast_convolve(a_t).max()

            fc = eval_c(c)
            fd = eval_c(d)
            for _ in range(12):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_c(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_c(d)
            opt_val = (lo + hi) / 2.0
            opt_obj = eval_c(opt_val)
            if opt_obj < best_obj - 1e-13:
                scale = (1.0 - opt_val) / denom
                a[j] = opt_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                best_obj = opt_obj
                best_a = a.copy()
                improved = True
        if not improved and passes > 2:
            break
    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation for max-convolution minimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    best_obj = M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_boost = 1.0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c_vec = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * m_boost)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                m_boost = min(m_boost * 1.1, 5.0)
                continue
            g = res.x
        except Exception:
            m_boost = min(m_boost * 1.1, 5.0)
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - a)) < 1e-8:
            no_improve += 1
            if no_improve > 60:
                break
            m_boost = min(m_boost * 1.05, 3.0)
            continue
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1.0 - alpha) * a + alpha * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_boost = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_boost = min(m_boost * 1.08, 4.0)
        if no_improve > 70:
            rng_r = np.random.default_rng(it * 61 + 3)
            freq = int(rng_r.integers(1, max(2, n // 8)))
            phase = float(rng_r.uniform(0, 2.0 * np.pi))
            amp = 0.005 + 0.003 * min(no_improve // 40, 4)

            a = project_simplex(best_a + pert)

def parametric_search(time_limit, n_values, verbose=False):
    """Search over diverse parametric families to find good initializations."""
    best_a = None
    best_score = float('inf')
    t0 = time.monotonic()
    for n in n_values:
        if time.monotonic() - t0 > time_limit:
            break
        base_shapes = []
        # Uniform
        base_shapes.append(np.ones(n))
        # Cosine windows with various powers
        for power in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0, 8.0]:
            x = np.linspace(0, np.pi, n)
            base_shapes.append(np.sin(x) ** power + 0.005)
        # Geometric (asymmetric)
        for r in [0.92, 0.95, 0.97, 0.99, 0.995, 0.998]:
            base_shapes.append(r ** np.arange(n, dtype=np.float64))
        # Symmetric geometric
        half = (n + 1) // 2
        for r in [0.9, 0.93, 0.96, 0.98, 0.99]:
            asc = r ** np.arange(half, dtype=np.float64)
            desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
            shape = np.concatenate([asc, desc])
            if len(shape) < n:
                shape = np.concatenate([shape, np.ones(n - len(shape))])
            base_shapes.append(shape[:n])
        # Gaussian centered
        center = n / 2.0
        for sigma_frac in [0.2, 0.25, 0.3, 0.4, 0.5]:
            sigma = sigma_frac * n
            x = np.arange(n) - center
            base_shapes.append(np.exp(-x * x / (2.0 * sigma * sigma)))
        # Trapezoid
        for w_frac in [0.3, 0.5, 0.6, 0.8]:
            w_start = int(n * (1.0 - w_frac) / 2.0)
            w_end = int(n * (1.0 + w_frac) / 2.0)
            shape = np.zeros(n)
            if w_start > 0:
                shape[:w_start] = np.linspace(0.05, 1.0, w_start)
            shape[w_start:w_end] = 1.0
            if n - w_end > 0:
                shape[w_end:] = np.linspace(1.0, 0.05, n - w_end)
            base_shapes.append(np.maximum(shape, 0.001))
        # Modulated shapes
        for mod_freq in [1, 2, 3, 4]:
            base = np.sin(np.linspace(0, np.pi, n)) + 0.15
            for mod_depth in [0.1, 0.2, 0.35, 0.5]:
                mod = 1.0 + mod_depth * np.cos(2.0 * np.pi * mod_freq * np.arange(n) / n)
                base_shapes.append(np.maximum(base * mod, 0.001))
        # Skewed cosine
        for skew in [0.5, 1.0, 1.5, 2.0]:
            x = np.linspace(0, 1, n)
            shape = (1.0 + np.cos(2.0 * np.pi * x * skew)) ** 2 + 0.01
            base_shapes.append(shape)
        for shape in base_shapes:
            if time.monotonic() - t0 > time_limit:
                break
            shape = np.maximum(shape, 0.0)
            ss = shape.sum()
            if ss < 1e-15:
                continue
            shape = shape / ss
            b = fast_convolve(shape)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = shape.copy()
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
