def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.003, decay=0.9998):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric (exponential decay) initial sequence."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n, A_ub_cache=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if A_ub_cache is None:
        A_ub_cache = build_lp_matrix(f, n)
    else:
        # Update the constraint matrix (f changed)
        A_ub_cache = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub_cache, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_line_search(f, g, f_max, n, max_iter=12):
    """Golden section line search for alpha in [0,1] minimizing conv max."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    def eval_alpha(a):
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return f_max
        ft = ft / s
        return fast_convolve(ft).max()

    # Initial coarse grid
    best_alpha = 0.0
    best_val = f_max
    for a in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
        v = eval_alpha(a)
        if v < best_val:
            best_val = v
            best_alpha = a

    # Golden section refinement around best
    span = 0.15
    lo = max(0.0, best_alpha - span)
    hi = min(1.0, best_alpha + span)
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    c_val = eval_alpha(c_pt)
    d_val = eval_alpha(d_pt)

    for _ in range(max_iter):
        if c_val < d_val:
            hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = eval_alpha(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = eval_alpha(d_pt)
        if abs(hi - lo) < 1e-6:
            break
        mid_alpha = (lo + hi) / 2.0
        mid_val = eval_alpha(mid_alpha)
        if mid_val < best_val:
            best_val = mid_val
            best_alpha = mid_alpha

    return best_alpha, best_val

def multiplicative_descent(f0, time_limit, verbose=False, lr_init=0.004, decay=0.9997):
    """Mirror descent with multiplicative weight updates."""
    f = np.maximum(f0.astype(np.float64), 1e-15)

        conv = fast_convolve(f)
        M = conv.max()

        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co

            no_improve = 0

            f = f / f.sum()
            lr = max(lr_init * 0.6, 1e-6)
            no_improve = 0

        if lr < 1e-8:

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay initial sequence."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_peak")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10 and best_seq is not None:
                t2_actual = min(t2, rem2 - 10)
                if t2_actual > 5:
                    print(f"  Final MD: t={t2_actual:.0f}s")
                    a_opt, obj = multiplicative_descent(best_seq, time_limit=t2_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_md")

            rem3 = deadline - time.monotonic() - 5
            if rem3 > 5 and best_seq is not None:
                t3_actual = min(t3, rem3)
                if t3_actual > 5:
                    print(f"  Final LP-FW: t={t3_actual:.0f}s")
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
