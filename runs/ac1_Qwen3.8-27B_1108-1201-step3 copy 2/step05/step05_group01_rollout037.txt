def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_smooth_obj_grad(f, beta):
    """
    Compute smooth log-sum-exp objective and gradient.
    f is on the simplex (sum=1, non-negative).
    Returns (obj_value, gradient) where obj = 2n * LSE_beta(conv(f,f)).
    """
    n = len(f)
    conv = fast_convolve(f)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    w = exp_shifted / s
    lse = conv_max + np.log(s) / beta
    obj = 2.0 * n * lse

    # Gradient: grad[i] = 2 * sum_j w[i+j] * f[j] for i=0..n-1
    # Compute via FFT cross-correlation
    N_fft = 1 << (3 * n - 2).bit_length()
    W = np.fft.fft(w, N_fft)
    F = np.fft.fft(f, N_fft)
    R = np.fft.ifft(np.conj(W) * F)
    # R[(N-i) % N] = sum_k w[k] * f[k-i] for valid k-i
    # For i=0: R[0], for i=1: R[N-1], ..., for i=n-1: R[N-n+1]
    idx = np.zeros(n, dtype=int)
    if n > 1:
        idx[1:] = N_fft - np.arange(1, n)
    grad = 2.0 * R[idx].real
    grad *= 2.0 * n  # Scale by 2n for the full objective

    return obj, grad


def nesterov_pgd(f0, time_limit, beta_start=20.0, beta_end=2000.0,
                 momentum_start=0.85, momentum_end=0.99,
                 verbose=False, rng=None):
    """
    Nesterov accelerated projected gradient descent with adaptive log-sum-exp smoothing.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    total_time = time_limit
    v = np.zeros(n)
    no_improve = 0
    max_no_improve = 80

    # Precompute FFT size
    N_fft = 1 << (3 * n - 2).bit_length()

    it = 0
    while time.monotonic() - t0 < total_time:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / total_time)

        # Anneal beta and momentum
        beta = beta_start * (beta_end / beta_start) ** progress
        mu = momentum_start + (momentum_end - momentum_start) * progress

        # Compute gradient
        conv = fast_convolve(f)
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        s_exp = exp_shifted.sum()
        w = exp_shifted / s_exp
        lse = conv_max + np.log(s_exp) / beta

        # Gradient via FFT
        W = np.fft.fft(w, N_fft)
        F = np.fft.fft(f, N_fft)
        R = np.fft.ifft(np.conj(W) * F)
        idx = np.zeros(n, dtype=int)
        if n > 1:
            idx[1:] = N_fft - np.arange(1, n)
        grad = 2.0 * R[idx].real

        # Nesterov momentum update
        v = mu * v + (1.0 - mu) * grad

        # Adaptive step size
        grad_norm = np.linalg.norm(v)
        if grad_norm < 1e-15:
            break
        alpha = 1.0 / (grad_norm + 1e-10)
        # Scale down for safety
        alpha *= 0.8

        # Projected gradient step
        f_new = f - alpha * v
        f_new = project_simplex(f_new)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    NPGD it={it}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 5:
                alpha *= 0.5
                f_new = f - alpha * v
                f_new = project_simplex(f_new)
                conv_new = fast_convolve(f_new)
                if conv_new.max() < best_obj - 1e-14:
                    best_obj = conv_new.max()
                    best_f = f_new.copy()
                    no_improve = 0
                    f = f_new

        # Restart if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.005 + 0.01 * no_improve / max_no_improve
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0
            max_no_improve = max(50, int(max_no_improve * 1.3))

    return best_f, best_obj


def spectral_flat_restart(f, rng, strength=0.5):
    """
    Perturb f via spectral flattening: flatten the DFT magnitude,
    project back to non-negative simplex.
    """
    n = len(f)
    F = np.fft.fft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    # Flatten magnitudes: interpolate between current and uniform
    target_mag = np.full_like(mag, mag.mean())
    new_mag = (1.0 - strength) * mag + strength * target_mag
    # Ensure DC component is preserved (real, positive)
    new_F = new_mag * np.exp(1j * phase)
    new_F[0] = np.abs(new_F[0])  # Keep DC real and positive
    # Ensure Hermitian symmetry for real output
    new_F[-(n//2):] = np.conj(new_F[1:(n//2)+1][::-1]) if n % 2 == 0 else np.conj(new_F[1:(n-1)//2+1][::-1])
    f_new = np.real(np.fft.ifft(new_F))
    f_new = np.maximum(f_new, 0.0)
    f_sum = f_new.sum()
    if f_sum < 1e-15:
        return f.copy()
    f_new /= f_sum
    return f_new


def block_coordinate_descent(f0, time_limit, block_size=32, verbose=False):
    """
    Block coordinate descent: optimize blocks of coordinates at a time.
    Within each block, use a few iterations of projected gradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    n_blocks = (n + block_size - 1) // block_size

    while time.monotonic() - t0 < time_limit:
        # Shuffle block order
        order = rng.permutation(n_blocks)
        improved_any = False

        for bi in order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            size = end - start

            # Current block values
            block = f[start:end].copy()
            rest_mask = np.ones(n, dtype=bool)
            rest_mask[start:end] = False
            rest = f[rest_mask]
            rest_sum = rest.sum()

            # Optimize block: minimize max conv with block variables
            # Use a few iterations of gradient descent on the block
            beta = 100.0
            for inner_it in range(20):
                # Assemble full f
                f_full = np.empty(n)
                f_full[start:end] = block
                f_full[rest_mask] = rest
                f_sum = f_full.sum()
                if f_sum < 1e-15:
                    break
                f_norm = f_full / f_sum

                conv = fast_convolve(f_norm)
                conv_max = conv.max()
                shifted = beta * (conv - conv_max)
                exp_s = np.exp(shifted)
                w = exp_s / exp_s.sum()

                # Gradient w.r.t. block coordinates
                N_fft = 1 << (3 * n - 2).bit_length()
                W = np.fft.fft(w, N_fft)
                F = np.fft.fft(f_norm, N_fft)
                R = np.fft.ifft(np.conj(W) * F)
                idx = np.zeros(n, dtype=int)
                if n > 1:
                    idx[1:] = N_fft - np.arange(1, n)
                grad_full = 2.0 * R[idx].real

                # Only use gradient for block coordinates
                grad_block = grad_full[start:end]
                block_norm = np.linalg.norm(grad_block)
                if block_norm < 1e-12:
                    break

                lr = 0.5 / (block_norm + 1e-10)
                block_new = block - lr * grad_block
                block_new = np.maximum(block_new, 0.0)
                b_sum = block_new.sum()
                if b_sum < 1e-15:
                    break
                # Normalize: block should sum to (1 - rest_sum * scale)
                # Actually, let's just project the whole thing
                f_full[start:end] = block_new
                f_full = project_simplex(f_full)
                f_trial = f_full.copy()

                conv_t = fast_convolve(f_trial)
                obj_t = conv_t.max()
                if obj_t < best_obj - 1e-14:
                    best_obj = obj_t
                    best_f = f_trial.copy()
                    f = f_trial
                    block = f[start:end].copy()
                    rest_mask = np.ones(n, dtype=bool)
                    rest_mask[start:end] = False
                    rest = f[rest_mask]
                    improved_any = True
                    break  # Move to next block
                else:
                    block = block_new  # Keep the new block for next inner iteration

        if not improved_any:
            break

    return best_f, best_obj


def coarse_to_fine(f_coarse, n_fine, time_limit, verbose=False):

def coordinate_polish(f0, time_limit, verbose=False):
    """
    Coordinate-wise polishing: try small multiplicative changes to each coordinate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    multipliers = [0.5, 0.7, 0.85, 1.0/0.85, 1.3, 1.5]

    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        # Random order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_mult = 1.0
            best_obj_j = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-10 or new_val > 0.99:
                    continue
                # Adjust other coordinates proportionally
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest_sum
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0.0)
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-15:
                    continue
                f_trial /= f_trial_sum

                obj = fast_convolve(f_trial).max()
                if obj < best_obj_j - 1e-15:
                    best_obj_j = obj
                    best_mult = mult

            if best_mult != 1.0:
                new_val = old_val * best_mult
                rest_sum = 1.0 - old_val
                if rest_sum > 1e-10:
                    scale = (1.0 - new_val) / rest_sum
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f_sum = f.sum()
                    if f_sum > 1e-15:
                        f /= f_sum
                    if best_obj_j < best_obj - 1e-14:
                        best_obj = best_obj_j
                        best_f = f.copy()
                        improved = True

        if not improved:
            # Try random restart within coordinate polish
            if sweeps > 3:
                break
            f = project_simplex(f + rng.standard_normal(n) * 0.001)

    return best_f, best_obj

    candidates = {}

    # Parent
    candidates['parent'] = parent.copy()

    # Reversed
    candidates['reversed'] = parent[::-1].copy()

    # Uniform
    candidates['uniform'] = np.ones(n_parent) / n_parent

    # Gaussian bump
    x = np.linspace(-1, 1, n_parent)
    g = np.exp(-3 * x ** 2)
    candidates['gaussian'] = g / g.sum()

    # U-shape (high at ends)
    u = np.exp(2 * x ** 2)
    candidates['ushape'] = u / u.sum()

    # Cosine window
    c = 0.5 * (1 + np.cos(np.pi * x))
    c = np.maximum(c, 0.01)
    candidates['cosine'] = c / c.sum()

    # Spectral flat version of parent
    candidates['spectral'] = spectral_flat_restart(parent, rng, strength=0.7)

    # Random
    r = rng.random(n_parent)
    r = np.maximum(r, 0.01)
    candidates['random'] = r / r.sum()

    # Quick PGD on each candidate (15s each, but limit total to 150s)
    quick_time = min(20, (150 / len(candidates)))
    quick_results = {}
    for name, f in candidates.items():
        if time.monotonic() > time.monotonic() + 150 - (time.monotonic() - (time.monotonic() - 150)):
            break
        a_opt, obj = nesterov_pgd(f, time_limit=quick_time,
                                   beta_start=30.0, beta_end=500.0,
                                   rng=rng)
        s = float(evaluate_sequence(a_opt.tolist()))
        quick_results[name] = (a_opt, s)
        print(f"  Quick [{name}]: score={s:.6f}, n={len(a_opt)}")
        try_seq(a_opt, f"quick_{name}")

    # Pick best quick result as main starting point
    best_quick_name = min(quick_results, key=lambda k: quick_results[k][1])
    main_start = quick_results[best_quick_name][0].copy()
    print(f"  Best quick start: {best_quick_name} (score={quick_results[best_quick_name][1]:.6f})")

    # === Phase 2: Main Nesterov PGD optimization (450s) ===
    print(f"\n=== Phase 2: Main PGD (best={best_score:.6f}) ===")
    t2 = min(450, deadline - time.monotonic() - 800)
    if t2 > 30:
        a_opt, obj = nesterov_pgd(main_start, time_limit=t2,
                                   beta_start=15.0, beta_end=5000.0,
                                   momentum_start=0.85, momentum_end=0.99,
                                   verbose=True, rng=rng)
        try_seq(a_opt, "main_pgd")

    # === Phase 3: Alternating PGD + Spectral restarts (300s) ===
    print(f"\n=== Phase 3: PGD + Spectral (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else main_start
    t3 = min(300, deadline - time.monotonic() - 500)
    if t3 > 30:
        f_cur = base.copy()
        t3_start = time.monotonic()
        no_global_improve = 0
        restart_count = 0

        while time.monotonic() - t3_start < t3:
            rem = t3 - (time.monotonic() - t3_start)
            if rem < 20:
                break
            seg_time = min(rem * 0.6, 60)

            a_opt, obj = nesterov_pgd(f_cur, time_limit=seg_time,
                                       beta_start=50.0, beta_end=5000.0,
                                       momentum_start=0.9, momentum_end=0.995,
                                       rng=rng)
            s = float(evaluate_sequence(a_opt.tolist()))
            if s < best_score - 1e-8:
                best_score = s
                best_seq = a_opt.copy()
                no_global_improve = 0
            else:
                no_global_improve += 1

            # Spectral restart
            restart_count += 1
            strength = min(0.3 + 0.1 * restart_count, 0.8)
            f_cur = spectral_flat_restart(a_opt, rng, strength=strength)
            # Mix with best
            mix = 0.5 + 0.3 * rng.random()
            f_cur = mix * f_cur + (1 - mix) * best_seq
            f_cur = project_simplex(f_cur)

            if verbose:
                print(f"  Phase 3 iter {restart_count}: best={best_score:.6f}")

    # === Phase 4: Multi-length exploration (250s) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else main_start

    target_lengths = [1024, 1536, 2048, 3072]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        # Interpolate
        old_n = len(base)
        x_old = np.linspace(0, 1, old_n, endpoint=False)
        x_new = np.linspace(0, 1, target_n, endpoint=False)
        f_interp = np.interp(x_new, x_old, base)
        f_interp = np.maximum(f_interp, 0.0)
        f_interp_sum = f_interp.sum()
        if f_interp_sum > 1e-15:
            f_interp /= f_interp_sum
        else:
            f_interp = np.ones(target_n) / target_n

        t_alloc = min(rem * 0.35, 60)
        print(f"  Trying n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = nesterov_pgd(f_interp, time_limit=t_alloc,
                                   beta_start=30.0, beta_end=3000.0,
                                   rng=rng)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Deep optimization on best (400s) ===
    print(f"\n=== Phase 5: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: Long Nesterov PGD
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.5, 180)
            a_opt, obj = nesterov_pgd(best_seq, time_limit=t_alloc,
                                       beta_start=100.0, beta_end=20000.0,
                                       momentum_start=0.92, momentum_end=0.998,
                                       verbose=True, rng=rng)
            try_seq(a_opt, "deep_pgd1")

        # 5b: Block CD
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            a_opt, obj = block_coordinate_descent(best_seq, time_limit=t_alloc,
                                                   block_size=max(16, n_best // 32))
            try_seq(a_opt, "deep_bcdd")

        # 5c: Another PGD round with different params
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            # Perturb slightly
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, obj = nesterov_pgd(f_pert, time_limit=t_alloc,
                                       beta_start=50.0, beta_end=10000.0,
                                       momentum_start=0.88, momentum_end=0.995,
                                       rng=rng)
            try_seq(a_opt, "deep_pgd2")

        # 5d: Spectral restart + PGD
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.6, 80)
            f_spec = spectral_flat_restart(best_seq, rng, strength=0.4)
            f_spec = 0.7 * f_spec + 0.3 * best_seq
            f_spec = project_simplex(f_spec)
            a_opt, obj = nesterov_pgd(f_spec, time_limit=t_alloc,
                                       beta_start=100.0, beta_end=20000.0,
                                       momentum_start=0.9, momentum_end=0.997,
                                       rng=rng)
            try_seq(a_opt, "deep_spectral")

    # === Phase 6: Final polishing (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.4, 60)
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_cd")

        # Final fine PGD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.6, 50)
            a_opt, obj = nesterov_pgd(best_seq, time_limit=t_alloc,
                                       beta_start=500.0, beta_end=50000.0,
                                       momentum_start=0.95, momentum_end=0.999,
                                       rng=rng)
            try_seq(a_opt, "polish_pgd")

        # One more coordinate polish
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 20)
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "polish_cd2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
