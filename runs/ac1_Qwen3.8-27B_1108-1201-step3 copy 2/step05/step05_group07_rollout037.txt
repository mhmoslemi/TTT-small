"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) log-softmax L-BFGS with temperature annealing, (2) momentum-based
subgradient descent with exact max, (3) SLP with LP subproblems, (4) local
peak refinement, and (5) multi-length exploration.

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_shift(a, k):
    """Return vector g where g[i] = 2*a[k-i] (with zero extension)."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2 * a[k-i] if 0 <= k-i < n
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def soft_max_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    # Gradient: grad_a = 2n * sum_k w_k * shift(a, k)
    # shift(a, k)[i] = a[k-i]
    # So grad_a[i] = 2n * sum_k w_k * a[k-i]
    # This is the cross-correlation of w with a, scaled by 2n
    # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] (valid range)
    # Let's compute this via FFT
    # Actually, sum_k w_k * a_{k-i} is the (i-th) coefficient of the 
    # "anti-convolution" or cross-correlation
    # Let's use FFT: if W = FFT(w), then FFT(sum_k w_k a_{k-i}) = conj(W) * A
    # Wait, that's the cross-correlation.
    # cross_corr(w, a)[i] = sum_k w[k] * conj(a[k-i]) = for real: sum_k w[k] * a[k-i]
    # Using FFT: cross_corr = IFFT(conj(FFT(w)) * FFT(a))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * n * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

def lbgfs_optimize(z0, T, n, time_limit, maxiter=3000):
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)

def subgradient_momentum(a0, time_limit, lr=0.01, momentum=0.9, top_k=50, 
                         warmup=100, lr_decay=0.9999):
    """Subgradient method with momentum for the exact max objective."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    cur_lr = lr
    count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        cur_max = b.max()
        
        # Top-k peaks for smoothed subgradient
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        # Smooth weights
        peak_vals = b[peaks]
        max_pv = peak_vals.max()
        # Using exponential weights
        w = np.exp(10 * (peak_vals - max_pv))
        w /= w.sum()
        
        # Compute smoothed subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            sg += w[i] * cross_corr_shift(a, k)
        
        # Momentum update
        v = momentum * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        # Check improvement
        b_new = fast_convolve(a_new)
        cur_max_new = b_new.max()
        
        if cur_max_new < best_obj - 1e-15:
            best_obj = cur_max_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        cur_lr *= lr_decay
        count += 1
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
        
        if count % 10000 == 0:
            print(f"    SG iter {count}: best_obj={best_obj:.8f}, cur={cur_max_new:.8f}, lr={cur_lr:.6f}")
    
    return best_a, best_obj

def slp_optimize(a0, time_limit, max_iter=200, top_k=20):
    """Successive Linear Programming: at each step, solve LP to minimize 
    linearized max autocorrelation."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        cur_max = b.max()
        
        # Select top-k peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        peaks = sorted(peaks)
        
        # Build LP: min t
        # s.t. t - 2 * sum_i a_{k-i} * a_i >= -b_k  for k in peaks
        #      sum a_i = 1
        #      a_i >= 0
        # Variables: a_0, ..., a_{n-1}, t
        # Objective: minimize t
        # In the convention for linprog: c^T x, A_ub x <= b_ub, A_eq x = b_eq
        
        # c = [0, 0, ..., 0, 1] (minimize t, which is the last variable)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Inequality constraints: for each peak k:
        # t - 2 * sum_i a_{k-i} * a_i >= -b_k
        # => -t + 2 * sum_i a_{k-i} * a_i <= b_k
        # A_ub row: [2*a_{k-0}, 2*a_{k-1}, ..., 2*a_{k-(n-1)}, -1] <= b_k
        # where a_{k-i} = 0 if k-i < 0 or k-i >= n
        
        m = len(peaks)
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        
        for idx, k in enumerate(peaks):
            # 2 * a_{k-i} for i = 0, ..., n-1
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[i] = 2.0 * a[j]
            A_ub[idx, :n] = row
            A_ub[idx, n] = -1.0  # coefficient of t

def local_peak_refine(a0, time_limit, n_peaks=5):
    """Try to reduce specific peaks by adjusting individual elements."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    improved_iter = 0
    while time.monotonic() - t0 < time_limit and improved_iter < 100:
        b = fast_convolve(a)
        # Find top peaks (excluding boundary)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        
        any_improved = False
        for k in sorted(peak_indices, key=lambda x: -b[x]):
            if time.monotonic() - t0 > time_limit:
                break
            if b[k] <= best_obj * 1.001:
                continue
            
            # Find contributing pairs
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k - idx]
            
            # Try reducing the element with largest contribution
            for trial_count in range(min(5, len(idx))):
                if trial_count >= len(idx):
                    break
                # Get top contributors
                top_contrib_idx = idx[np.argsort(contribs)[::-1][:5]]
                
                for i in top_contrib_idx:
                    old_val = a[i]
                    if old_val < 1e-12:
                        continue
                    # Try reducing a[i] by various factors
                    for factor in [0.7, 0.8, 0.9]:
                        new_val = old_val * factor
                        diff = old_val - new_val
                        if diff < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[i] = new_val
                        # Redistribute to others proportional to their values
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = a_trial / a_trial.sum()
                        
                        trial_max = fast_convolve(a_trial).max()
                        if trial_max < best_obj - 1e-14:
                            best_obj = trial_max
                            best_a = a_trial.copy()
                            a = a_trial
                            any_improved = True
                            # Recompute contributions
                            b = fast_convolve(a)
                            if b[k] <= best_obj * 1.001:
                                break
                            contribs = a[idx] * a[k - idx]
                            break
                    if b[k] <= best_obj * 1.001:
                        break
            
            if any_improved:
                break
        
        if any_improved:
            improved_iter += 1
        else:
            break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent with line search on each coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0])
    
    prev_obj = best_obj
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try a few random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a_pert = project_simplex(a + rng.standard_normal(n) * 0.01)

                a = a_pert
                best_obj = pert_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) log-softmax L-BFGS with temperature annealing, (2) exact
subgradient with momentum and adaptive step size, (3) SLP with LP subproblems,
(4) systematic peak refinement via coordinate moves, (5) multi-length exploration,
and (6) simulated-annealing stochastic restarts to escape local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """Cross-correlation cc[i] = sum_k w[k] * a[k-i] for k=0..len(w)-1, valid i=0..n-1."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_grad(z, T):
    """Soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = log_sum_exp / T
    w = exp_Tb / exp_Tb.sum()
    # grad_a[i] = 2 * sum_k w_k * a[k-i] = 2 * cross_corr(w, a)[i]
    cc = cross_corr_fft(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_optimize(z0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with time limit."""
    t0 = time.monotonic()
    n = len(z0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def subgradient_momentum(a0, time_limit, lr_init=0.02, momentum=0.95,
                         top_k=30, smooth_T=5.0):
    """
    Subgradient method with momentum for exact max of autocorrelation.
    Uses smoothed subgradient (softmax over top-k peaks) for stability.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    b_init = fast_convolve(a)
    best_obj = b_init.max()
    v = np.zeros(n)
    t0 = time.monotonic()
    cur_lr = lr_init
    count = 0
    no_improve = 0
    sa_temp = 0.001  # simulated annealing temperature

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        cur_max = b.max()

        # Select top-k peaks with softmax weights
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        peak_vals = b[peaks]
        scaled = peak_vals * smooth_T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws

        # Compute smoothed subgradient: sum_i w_i * 2*a[k_i - i]
        # Using FFT: sg = 2 * cross_corr(w_extended, a)
        w_ext = np.zeros(len(b))
        for idx in range(len(peaks)):
            w_ext[int(peaks[idx])] = w[idx]
        sg = 2.0 * cross_corr_fft(w_ext, a)

        # Momentum update
        v = momentum * v - cur_lr * sg
        a_new = project_simplex(a + v)

        # Evaluate
        b_new = fast_convolve(a_new)
        cur_max_new = b_new.max()

        # Simulated annealing: accept worse with some probability
        if cur_max_new < best_obj - 1e-15:
            best_obj = cur_max_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        elif cur_max < cur_max_new:
            # Accept if within SA temperature
            delta = cur_max_new - cur_max
            if np.random.random() < np.exp(-delta / max(sa_temp, 1e-10)):
                a = a_new
            else:
                # Revert step
                v = 0.5 * v
                a = project_simplex(a + v)
            no_improve += 1
        else:
            a = a_new
            no_improve += 1

        cur_lr *= 0.99995
        sa_temp *= 0.999
        count += 1

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            scale = 0.003 + 0.003 * np.exp(-count / 50000.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.3, 0.002)
            no_improve = 0

        if count % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {count}/{int(elapsed * 2000)}: best={best_obj:.8f}, lr={cur_lr:.6f}")

    return best_a, best_obj


def slp_lp_optimize(a0, time_limit, max_iter=100, top_k=15):
    """
    SLP: at each step, solve LP to minimize linearized max autocorrelation.
    Uses top-k active peaks for the LP constraints.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)

        # Select top-k peaks
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        peaks_sorted = sorted(int(p) for p in peaks)

        # Build LP: min t
        # s.t. 2*sum_i a[k-i]*a_i - t <= b[k]  for each peak k
        #      sum a_i = 1
        #      a_i >= 0
        # Variables: [a_0, ..., a_{n-1}, t]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        m = len(peaks_sorted)
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)

        for idx, k in enumerate(peaks_sorted):
            row = np.zeros(n)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                row[i] = 2.0 * a[k - i]
            A_ub[idx, :n] = row
            A_ub[idx, n] = -1.0
            b_ub[idx] = b[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break

            a_lp = np.maximum(res.x[:n], 0)
            if a_lp.sum() < 1e-15:
                break
            a_lp = a_lp / a_lp.sum()

            # Line search
            best_alpha = 0.0
            best_val = best_obj
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                a_trial = (1 - alpha) * a + alpha * a_lp
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() < 1e-15:
                    continue
                a_trial = a_trial / a_trial.sum()
                trial_max = fast_convolve(a_trial).max()
                if trial_max < best_val:
                    best_val = trial_max
                    best_alpha = alpha

            if best_alpha > 0:
                a = (1 - best_alpha) * a + best_alpha * a_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_val < best_obj - 1e-15:
                    best_obj = best_val
                    best_a = a.copy()
                    if it % 5 == 0:
                        print(f"    SLP iter {it}: best={best_obj:.8f}, alpha={best_alpha}")
            else:
                # No improvement from LP; try a small random step
                break

        except Exception:
            break

    return best_a, best_obj


def local_peak_refine(a0, time_limit, n_peaks=5):
    """Systematically reduce top peaks by adjusting contributing elements."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stagnation = 0

    while time.monotonic() - t0 < time_limit and stagnation < 50:
        b = fast_convolve(a)
        b_inner = b.copy()
        b_inner[0] = -np.inf
        b_inner[-1] = -np.inf
        peak_indices = np.argpartition(b_inner, -n_peaks)[-n_peaks:]
        peak_indices = sorted(peak_indices, key=lambda x: -b[x])

        any_improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            if b[k] <= best_obj * 1.0005:
                continue

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contribs = a[idx] * a[k - idx]
            top_contribs = idx[np.argsort(contribs)[::-1][:8]]

            for i in top_contribs:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[i]
                if old_val < 1e-12:
                    continue
                for factor in [0.6, 0.75, 0.85, 0.92]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    trial_max = fast_convolve(a_trial).max()
                    if trial_max < best_obj - 1e-14:
                        best_obj = trial_max
                        best_a = a_trial.copy()
                        a = a_trial
                        any_improved = True
                        break
                    # Also try: reduce a[k-i] instead (the partner)
                    j = k - i
                    if j != i and 0 <= j < n and a[j] > 1e-12:
                        a_trial2 = a.copy()
                        a_trial2[j] = a_trial2[j] * factor
                        diff2 = a_trial2[j] / factor - a_trial2[j]  # original - new
                        mask2 = np.ones(n, dtype=bool)
                        mask2[j] = False
                        others_sum2 = a_trial2[mask2].sum()
                        if others_sum2 > 1e-15:
                            a
