Actually for the auto-convolution, `b[k] = sum_i a[i]*a[k-i]`. Let's define `c_i = a[i]` and `d_i = a[n-1-i]` (reversed). Then `b[k] = (c*d)[k]` (linear cross-correlation). This is not a quadratic form in a single vector in the standard sense.

OK regardless of the mathematical structure, the key point is that `max_k b[k]` is the max of functions that are each a "quadratic" in `a` (bilinear, but since it's auto-correlation, it's actually quadratic). And the max of convex functions is convex. But are the individual `b[k]` convex?

`b[k] = sum_i a[i]*a[k-i]`. The Hessian is:
`H_{i,j} = db[k]/da[i]da[j] = delta(i+j, k) + delta(j+i, k) = 2*delta(i+j, k)` (for `i != j`) and `delta(i+j, k)` (for `i=j`).

Wait no. `db[k]/da[j] = a[k-j] + a[j-k]` (using the zero-padding convention). So:
`d^2b[k]/da[j]da[l] = da[k-j]/da[l] + da[j-k]/da[l] = delta(k-j, l) + delta(j-k, l)`

For `k >= j`: first term is `delta(k-j, l)`, second is `delta(j-k, l) = 0` (since `j-k < 0`).
For `k < j`: first term is 0, second is `delta(j-k, l)`.

So `H_{j,l} = delta(|k-j|, l)`... this is getting confusing. Let me just note that the Hessian of `b[k]` with respect to `a` is a Hankel matrix with entries `delta(|j-l|, k)`...

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse-constraint LP-FW,
FISTA with log-sum-exp smoothing, exact 1D coordinate descent,
and diverse shape initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_sparse_lp(f, M, n, top_k=8):
    """Solve LP with only top-k peak constraints."""
    conv_f = fast_convolve(f)
    # Get top-k peaks
    k_actual = min(top_k, len(conv_f))
    peak_indices = np.argpartition(conv_f, -k_actual)[-k_actual:]
    
    c = -np.ones(n)
    A_ub = np.zeros((len(peak_indices), n))
    for idx, k in enumerate(peak_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (f*g)[k] = sum_j f[k-j]*g[j] for j in [js, je]
        # = sum_j f[k-j]*g[j]
        A_ub[idx, js:je+1] = f[k-je:k-js+1][::-1]
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, top_k=8):
    """LP-based Frank-Wolfe with sparse constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        g = solve_sparse_lp(f, M * m_relax, n, top_k=top_k)
        if g is None:
            m_relax *= 1.2
            if m_relax > 10.0:
                m_relax = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_relax = min(m_relax * 1.03, 5.0)
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(30):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def fista_smooth(f0, time_limit, tau_init=0.01, tau_end=0.001, verbose=False):
    """FISTA with log-sum-exp smoothed max of autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    t = 1
    x = f.copy()
    y = f.copy()
    
    # Adapt tau over time
    total_iters_expected = time_limit * 500  # estimate
    tau = tau_init

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        t += 1
        
        # Update tau (geometric decay)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_end / tau_init) ** progress
        
        # Compute smooth max and its gradient at y
        conv_y = fast_convolve(y)
        # Log-sum-exp
        max_c = conv_y.max()
        scaled = (conv_y - max_c) / tau
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        
        # Weights for the peak lags
        w = exp_scaled / sum_exp  # weights summing to 1

        # d/d a[j] log(sum_k exp(b[k]/tau)) = (1/tau) * sum_k w[k] * db[k]/da[j]
        # db[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)

        for k in range(len(conv_y)):
            if w[k] < 1e-10:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # db[k]/da[j] for j in [js, je]:
            # = y[k-j] (from the first term, valid when 0<=k-j<n, i.e., j in [k-n+1, k])
            # + y[j-k] (from the second term, valid when 0<=j-k<n, i.e., j in [k, k+n-1])
            # Combined: for j in [js, je], db[k]/da[j] = y[k-j] + y[j-k]
            # But y[m] = 0 for m < 0 or m >= n

            #   term1 = y[k-j] if k-j >= 0 and k-j < n else 0
            #   term2 = y[j-k] if j-k >= 0 and j-k < n else 0

            # So for j < k: only term1 = y[k-j] (valid if k-j < n, i.e., j > k-n)
            # For j > k: only term2 = y[j-k] (valid if j-k < n, i.e., j < k+n)

            # For j in [js, min(je, k)]: k-j >= 0, so term1 = y[k-j], term2 = 0 (j-k < 0)
            # For j in [max(js, k+1), je]: j-k >= 1, so term2 = y[j-k], term1 = 0 (k-j < 0)
            # For j = k (if k < n): both = y[0]

            # For j in [js, je]:
            #   if j <= k: contrib = y[k-j] (k-j ranges from k-je to k-js, all >= 0)
            #   if j > k: contrib = y[j-k] (j-k ranges from 1 to je-k)

            # Similarly je = min(n-1, k+n-1) = min(n-1, k), so j-k <= min(n-1-k, 0) for j > k...

            # This is getting complicated. Let me just use a simpler formula:
            # For the auto-convolution, db[k]/da[j] = 2*y[k-j] if k != 2j, and y[k-j] if k == 2j
            # (assuming k-j is in [0, n-1])
            # And 0 otherwise.
            
            # Actually, I realize I should just use the formula directly for small n.
            # Let me just vectorize it:
            if k < n:

                j_hi1 = min(n - 1, k)
                if j_lo1 <= j_hi1:
                    js_ = np.arange(j_lo1, j_hi1 + 1)
                    grad[js_] += wk * y[k - js_]

        y_new = project_simplex(y_new)

        cur_obj = conv_x.max()

def coordinate_descent_exact(f0, time_limit, verbose=False):
    """Coordinate descent with exact 1D line search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            # Compute the "base" autocorrelation (without a[j]'s contribution)
            # b[k] with a[j] removed:
            # b[k] = sum_i a[i]*a[k-i]
            # Terms involving a[j]: a[j]*a[k-j] (if 0<=k-j<n) + a[k-j]*a[j] (if 0<=k-j<n)
            # These are the same product, so b[k] = a[j]*a[k-j] + D_k where D_k doesn't involve a[j]
            # But actually b[k] = a[j]*a[k-j] + a[k-j]*a[j] when both are valid and j != k-j
            # = 2*a[j]*a[k-j] when j != k-j and both valid
            # = a[j]^2 when j = k-j (i.e., k = 2j)
            
            # For simplicity, let's just try a few multipliers and pick the best
            # This is faster than exact optimization for large n
            
            multipliers = [0.3, 0.5, 0.7, 1.4, 1.7, 2.5, 3.5]
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def make_initial_shape(n, shape_type, rng):
    """Create different initial shapes."""
    x = np.linspace(0, 1, n)
    if shape_type == 'gaussian':
        center = 0.5
        width = 0.25 + 0.1 * rng.random()
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif shape_type == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** (1.0 + rng.random())
    elif shape_type == 'power_law':
        center = 0.5
        p = 1.5 + 2.0 * rng.random()
        a = 1.0 - np.abs(x - center) / 0.5
        a = np.maximum(a, 0) ** p
    elif shape_type == 'triangle':
        center = 0.5
        a = 1.0 - 2.0 * np.abs(x - center)
        a = np.maximum(a, 0)
    elif shape_type == 'uniform':
        a = np.ones(n)
    elif shape_type == 'bump':
        center = 0.5
        width = 0.3 + 0.15 * rng.random()
        a = np.exp((x - center)**2 / (2*width**2) - 0.5)
        a = np.maximum(a - 0.1, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n)
        ss = n
    return a / ss


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def sq_polish(f0, time_limit):
    """Subgradient polishing with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse-constraint LP-FW with
constraint generation, FISTA with log-sum-exp smoothing, exact 1D coordinate

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(peaks, weights, a, n):
    """Compute subgradient of max conv at peak lags with given weights."""
    grad = np.zeros(n)
    for k, wk in zip(peaks, weights):
        k = int(k)
        if wk < 1e-12:
            continue
        if k < n:
            # j ranges from 0 to k
            # db[k]/da[j] = a[k-j] for j < k, and 2*a[0] for j = k
            if k > 0:
                grad[0:k] += wk * a[k:0:-1]
            if k < n:
                grad[k] += 2.0 * wk * a[0]
        else:
            # k >= n: j ranges from k-n+1 to n-1
            # db[k]/da[j] = a[k-j]
            js = np.arange(k - n + 1, n)
            grad[js] += wk * a[k - js]
    return grad


def solve_lp_constraints(f, M, n, constraint_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in constraint_indices, g >= 0."""
    c = -np.ones(n)
    K = len(constraint_indices)
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[idx, js:je + 1] = f[k - je:k - js + 1][::-1]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 60})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Get active constraints (top peaks)
        conv_f = fast_convolve(f)
        M = conv_f.max()
        k_actual = min(top_k, len(conv_f))
        peak_idx = np.argpartition(conv_f, -k_actual)[-k_actual:]
        # Sort by value descending
        peak_idx = peak_idx[np.argsort(conv_f[peak_idx])[::-1]]

        # Solve LP with these constraints
        g = solve_lp_constraints(f, M * m_relax, n, peak_idx)
        if g is None:
            # Try with relaxed M
            m_relax *= 1.3
            if m_relax > 15.0:
                m_relax = 1.0
            continue

        # Check if solution violates other constraints
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check all lags
        conv_fg = fast_convolve(np.concatenate([f, g]))  # This is NOT right
        # Actually we need (f*g)[k] = sum_j f[k-j]*g[j]
        # Let's compute it properly
        fg = np.zeros(2 * n - 1)
        for j in range(n):
            if g[j] < 1e-15:
                continue
            fg[j:j+n] += g[j] * f

        # Find most violated constraint not in our set
        current_max = fg.max()
        if current_max > M * m_relax * 1.05:
            # Add the most violated constraint
            viol_idx = np.argmax(fg)
            if viol_idx not in peak_idx:
                peak_idx = np.append(peak_idx, viol_idx)
                # Re-solve with one more constraint
                g2 = solve_lp_constraints(f, M * m_relax, n, peak_idx)
                if g2 is not None:
                    g2_sum = g2.sum()
                    if g2_sum > 1e-15:
                        g = g2 / g2_sum
                    else:
                        g = g  # keep old
                else:
                    m_relax = min(m_relax * 1.1, 5.0)
                    continue

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_relax = min(m_relax * 1.03, 5.0)
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(25):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.65

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_relax = 1.0
            f = f_new

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    iter {it} ({elapsed:.0f}s): obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 60, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def fista_smooth(f0, time_limit, tau_init=0.02, tau_end=0.003, verbose=False):
    """FISTA with log-sum-exp smoothed max of autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    t = 1.0
    x = f.copy()
    y = f.copy()
    x_prev = f.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        t += 1.0

        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_end / tau_init) ** progress

        # Compute smooth max gradient at y
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        scaled = (conv_y - max_c) / tau
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 50)
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        w = exp_scaled / sum_exp

        # Get peaks for efficient gradient
        k_actual = min(max(20, n // 4), len(conv_y))
        peak_idx = np.argpartition(w, -k_actual)[-k_actual:]
        w_peaks = w[peak_idx]

        grad = np.zeros(n)
        for k, wk in zip(peak_idx, w_peaks):
            k = int(k)
            wk_eff = wk / tau
            if wk_eff < 1e-14:
                continue
            if k < n:
                if k > 0:
                    grad[0:k] += wk_eff * y[k:0:-1]
                if k < n:
                    grad[k] += 2.0 * wk_eff * y[0]
            else:
                js = np.arange(k - n + 1, n)
                grad[js] += wk_eff * y[k - js]

        # Step size: for smooth max with temperature tau, Lipschitz ~ 1/tau * L_qp
        # L_qp is bounded by ~4n (sum of |a| in convolution)
        # Use conservative step: lr ~ tau / (4*n)
        lr = tau / (4.0 * n) * 10.0  # 10x more aggressive
        lr = min(lr, 0.1)

        y_new = y - lr * grad
        y_new = project_simplex(y_new)

        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_next
        x_new = y_new
        y = (1.0 + beta) * x_new - beta * x_prev
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y /= ys
        else:
            y = x_new.copy()
        x_prev = x
        x = x_new
        t = t_next

        # Evaluate current x
        conv_x = fast_convolve(x)
        cur_obj = conv_x.max()
        if cur_obj < best_obj - 1e-16:
            best_obj = cur_obj
            best_f = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            rng = np.random.default_rng(int(t) * 31 + 7)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            x = f_pert
            y = f_pert.copy()
            x_prev = f_pert.copy()
            t = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates on each coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                continue

            # Try several multipliers
            multipliers = [0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0]
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.005, mom=0.92):
    """Subgradient polishing with softmax-weighted peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_mom = mom
    temp = 0.0001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        k_actual = min(max(15, n // 10), len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgrad(peaks, w, f, n)

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        temp *= 1.003
        if no_improve % 200 == 0:
            cur_mom = min(0.98, cur_mom + 0.001)

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_f, best_obj


def make_initial_shape(n, shape_type, rng):
    """Create different initial shapes."""
    x = np.linspace(0, 1, n)
    if shape_type == 'gaussian':
        center = 0.5
        width = 0.2 + 0.15 * rng.random()
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif shape_type == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** (0.8 + 0.4 * rng.random())
    elif shape_type == 'power_law':
        center = 0.5
        p = 1.0 + 2.0 * rng.random()
        a = 1.0 - np.abs(x - center) / 0.5
        a = np.maximum(a, 0.0) ** p
    elif shape_type == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif shape_type == 'bump':
        center = 0.5
        width = 0.25 + 0.1 * rng.random()
        t = (x - center) / width
        a = np.where(np.abs(t) < 1, np.cos(np.pi * t / 2) ** 2, 0.0)
    elif shape_type == 'uniform':
        a = np.ones(n)
    elif shape_type == 'log_cosh':
        # Smooth "plateau" shape
        center = 0.5
        a = 1.0 / np.cosh(3.0 * (x - center))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n)
        ss = n
    return a / ss


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n > 2 and new_n > old_n:
        from scipy.interpolate import CubicSpline
        try:
            cs = CubicSpline(old_x, f, bc_type='natural')
            f_interp = cs(new_x)
            f_interp = np.maximum(f_interp, 0)
        except Exception:
            f_interp = np.interp(new_x, old_x, f)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
