"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
spectral shaping, momentum-based projected gradient with adaptive temperature,
block-level Nelder-Mead refinement, length exploration, and coordinate descent polish.
Key innovations over previous approach: spectral re-flattening initialization,
Nesterov momentum with backtracking line search, block-level local optimization,
and adaptive temperature scheduling based on convergence diagnostics.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(p):
    n = len(p)
    c = fast_convolve(p)
    return 2.0 * n * c.max()


def smooth_max_and_weights(p, T):
    c = fast_convolve(p)
    Tc = T * c
    m = Tc.max()
    exp_Tc = np.exp(Tc - m)
    log_sum_exp = m + np.log(exp_Tc.sum())
    obj = (1.0 / T) * log_sum_exp
    w = exp_Tc / exp_Tc.sum()
    return obj, w, c


def smooth_max_gradient(p, T):
    n = len(p)
    c = fast_convolve(p)
    Tc = T * c
    m = Tc.max()
    exp_Tc = np.exp(Tc - m)
    w = exp_Tc / exp_Tc.sum()

    # grad[i] = 2 * sum_{m=0}^{n-1} w[m+i] * p[m]
    # This is a valid cross-correlation of w with p
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(w[i:i + n], p)

    obj = (1.0 / T) * (m + np.log(exp_Tc.sum()))
    return obj, grad, w


def spectral_shape(p, n_iters=200, flatten_gamma=0.3):
    n = len(p)
    p = p.copy()
    p = p / p.sum()

    for it in range(n_iters):
        c = fast_convolve(p)
        c_max = c.max()
        c_min = c[c > 0].min() if (c > 0).any() else c.min()

        # Target: make c more flat
        # Desired: c_target[k] proportional to 1 (flat)
        # Use gradient of sum(c^2) to reduce variance
        # grad sum(c^2)/grad p[i] = 4 * sum_k c[k] * p[k-i] (for valid k-i)
        # This pushes mass away from positions that contribute to high c values

        # Simpler: use the log-sum-exp gradient to push down peaks
        T = 50.0
        obj, w, _ = smooth_max_and_weights(p, T)
        obj_val, grad, _ = smooth_max_gradient(p, T)

        # Also add a "flatness" penalty: gradient of sum(c^2)
        # d(sum c^2)/dp[i] = 4 * sum_k c[k]*p[k-i]
        flat_grad = np.zeros(n)
        for i in range(n):
            flat_grad[i] = 4.0 * np.dot(c[:n], p[:n])  # approximation

        # Actually let's use a different approach:
        # Minimize the "excess" of c over its mean
        # Use weighted gradient where weights emphasize the peaks
        step = 0.01
        p_new = p - step * grad
        p_new = project_simplex(p_new)

        new_score = compute_score(p_new)
        old_score = compute_score(p)
        if new_score < old_score:
            p = p_new
        else:
            step *= 0.7

    return p


def momentum_projected_gradient(p0, T_start, T_end, n_iters, deadline, gamma=0.9, T_update=0.995):
    p = p0.copy()
    n = len(p)
    v = np.zeros(n)
    T = T_start
    best_p = p.copy()
    best_score = compute_score(p)
    step = 0.05
    no_improve = 0
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        obj, grad, w = smooth_max_gradient(p, T)

        # Nesterov momentum
        v = gamma * v - step * grad
        p_candidates = project_simplex(p + v)

        # Backtracking line search
        accepted = False
        s = step
        for bt in range(15):
            p_new = project_simplex(p + s * v / step)
            new_score = compute_score(p_new)
            if new_score < best_score:
                p = p_new
                best_p = p_new.copy()
                best_score = new_score
                no_improve = 0
                accepted = True
            elif new_score < compute_score(p) * 1.001:
                p = p_new
                accepted = True
            if accepted:
                break
            s *= 0.7

        if not accepted:
            no_improve += 1
            if no_improve > 100:
                # Reset momentum and reduce step
                v *= 0.5
                step *= 0.8
                # Occasional jump
                if no_improve > 200:
                    rng = np.random.default_rng(it * 7919 + 13)
                    p = project_simplex(best_p + rng.standard_normal(n) * 0.01)
                    v = np.zeros(n)
                    step = 0.05
                    no_improve = 0

        # Adapt temperature
        T = min(T_end, T * T_update)

        # Adapt step size
        if no_improve == 0:
            step = min(0.2, step * 1.02)
        elif no_improve > 50:
            step = max(0.005, step * 0.95)

    return best_p, best_score


def block_nelder_mead(p0, block_size, n_passes, deadline):
    p = p0.copy()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    n_blocks = (n + block_size - 1) // block_size

    for pass_num in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Shuffle block order
        order = np.random.RandomState(pass_num * 31 + 7).permutation(n_blocks)

        for bi in order:
            if time.monotonic() > deadline:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            bsize = end - start

            if bsize <= 1:
                continue

            block = p[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-15:
                continue

            # Normalize block to sum to block_sum for the local problem
            # We'll work in normalized space for the block
            block_norm = block / block_sum

            def block_obj(x):
                if time.monotonic() > deadline:
                    return 1e10
                p_trial = p.copy()
                p_trial[start:end] = x * block_sum
                p_trial = np.maximum(p_trial, 0)
                s = p_trial.sum()
                if s < 1e-15:
                    return 1e10
                p_trial = p_trial / s
                return compute_score(p_trial)

            # Nelder-Mead on the simplex-constrained block
            x0 = block_norm.copy()
            # Perturb for NM
            rtol = 0.01
            x0_perturbed = x0.copy()
            if bsize > 2:
                x0_perturbed[0] += rtol / bsize
                x0_perturbed[1] -= rtol / bsize
                x0_perturbed = np.maximum(x0_perturbed, 0)
                x0_perturbed = x0_perturbed / x0_perturbed.sum()

            try:
                res = minimize(
                    block_obj,
                    x0_perturbed,
                    method='Nelder-Mead',
                    options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8, 'adaptive': True}
                )
                x_opt = np.maximum(res.x, 0)
                xs = x_opt.sum()
                if xs > 1e-15:
                    x_opt = x_opt / xs
                    p_trial = p.copy()
                    p_trial[start:end] = x_opt * block_sum
                    p_trial = np.maximum(p_trial, 0)
                    s = p_trial.sum()
                    if s > 1e-15:
                        p_trial = p_trial / s
                        sc = compute_score(p_trial)
                        if sc < best_score:
                            p = p_trial
                            best_p = p.copy()
                            best_score = sc
            except Exception:
                pass

    return best_p, best_score


def coordinate_descent_polish(p0, deadline, multipliers=None):
    p = p0.copy()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.7])

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() < deadline:
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break

            old_val = p[j]
            if old_val < 1e-12:
                continue

            best_j_score = best_score
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                p_trial = p.copy()
                p_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                p_trial[mask] *= scale
                p_trial = np.maximum(p_trial, 0)
                s = p_trial.sum()
                if s < 1e-15:
                    continue
                p_trial = p_trial / s

                sc = compute_score(p_trial)
                if sc < best_j_score - 1e-14:
                    best_j_score = sc
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    p[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    p[mask] *= scale
                    p = np.maximum(p, 0)
                    s = p.sum()
                    if s > 1e-15:
                        p = p / s

                if best_j_score < best_score - 1e-14:
                    best_score = best_j_score
                    best_p = p.copy()
                    improved = True
                    no_improve = 0
                else:
                    no_improve += 1

        if not improved:
            no_improve += 10
            if no_improve > 100:
                # Try with finer multipliers
                multipliers = np.array([0.85, 0.92, 0.97, 1.0, 1.03, 1.08, 1.15])
                no_improve = 0

    return best_p, best_score


def simulated_annealing(p0, deadline, n_perturb_min=3, n_perturb_max=15):
    p = p0.copy()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    rng = np.random.default_rng(12345)

    # Temperature for Metropolis
    T_init = 0.01
    T_final = 0.0001
    n_steps_est = 5000
    cool_rate = (T_final / T_init) ** (1.0 / max(1, n_steps_est))
    T = T_init
    no_improve = 0

    t0 = time.monotonic()
    step_count = 0

    while time.monotonic() < deadline and T > T_final:
        step_count += 1
        # Randomly perturb a small block
        n_pert = rng.integers(n_perturb_min, n_perturb_max + 1)
        start = rng.integers(0, n - n_pert + 1)

        p_trial = p.copy()
        block = p_trial[start:start + n_pert]
        noise = rng.standard_normal(n_pert) * 0.02 * np.maximum(block, 0.01)
        block_new = np.maximum(block + noise, 0)
        if block_new.sum() < 1e-15:
            T *= cool_rate
            continue
        # Preserve block sum
        block_new = block_new * (block.sum() / block_new.sum())
        p_trial[start:start + n_pert] = block_new
        p_trial = p_trial / p_trial.sum()

        new_score = compute_score(p_trial)
        delta = new_score - compute_score(p)

        if delta < 0 or rng.random() < np.exp(-delta / T):
            p = p_trial
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cool_rate

        if no_improve > 500:
            # Restart from best
            p = best_p.copy()
            no_improve = 0
            T = max(T, T_init * 0.1)

    return best_p, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def two_phase_gradient(p0, deadline, T_start=5.0, T_end=200.0, n_iters=5000):
    """Two-phase: first reduce peaks (low T), then fine-tune (high T)."""
    p = p0.copy()
    n = len(p)
    v = np.zeros(n)
    best_p = p.copy()
    best_score = compute_score(p)
    gamma = 0.85
    step = 0.1
    t_phase_transition = 0.5  # fraction of time at low T

    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        elapsed_frac = (time.monotonic() - t0) / total_time
        if elapsed_frac < t_phase_transition:
            T = T_start
        else:
            T = T_start + (T_end - T_start) * (elapsed_frac - t_phase_transition) / (1 - t_phase_transition)

        obj, grad, w = smooth_max_gradient(p, T)

        v = gamma * v - step * grad
        p_new = project_simplex(p + v)
        new_score = compute_score(p_new)

        if new_score < best_score:
            p = p_new
            best_p = p_new.copy()
            best_score = new_score
            step = min(0.3, step * 1.01)
        elif new_score < compute_score(p) + 1e-10:
            p = p_new
            step = max(0.01, step * 0.95)
        else:
            step = max(0.005, step * 0.9)
            if step < 0.006:
                # Jump
                rng = np.random.default_rng(it * 6151 + 42)
                p = project_simplex(best_p + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                step = 0.1

    return best_p, best_score
