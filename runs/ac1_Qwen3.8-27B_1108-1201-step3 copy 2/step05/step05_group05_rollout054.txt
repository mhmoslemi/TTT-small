from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k in 0..2n-2"""
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)(k) <= M for k = 0, ..., 2n-2
    # (f*g)(k) = sum_{i} f_{k-i} * g_i for valid i
    # For each k, the valid i range is max(0, k-n+1) to min(n-1, k)
    
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = f[k - i]
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
        else:
            return None
    else:
        return None

def multiplicative_descent(f, time_limit, t0=None):
    """Mirror descent with multiplicative updates."""
    if t0 is None:
        t0 = time.monotonic()
    
    n = len(f)
    f = np.maximum(f.copy(), 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = 0.1  # step size
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Smooth gradient
        tau = M * 50  # temperature for log-sum-exp
        shifted = conv - M
        weights = np.exp(shifted / tau)
        weights /= weights.sum()
        
        # Gradient of LSE w.r.t. f
        grad = np.zeros(n)
        # For efficiency, only use top peaks
        top_k = min(100, 2*n-1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_weights = weights[peak_indices]
        peak_weights /= peak_weights.sum()
        
        for idx in range(len(peak_indices)):
            if peak_weights[idx] < 1e-12:
                continue
            k = int(peak_indices[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # grad[i] += 2 * w_k * f[k-i]
            i_range = np.arange(i_start, i_end + 1)
            grad[i_range] += 2.0 * peak_weights[idx] * f[k - i_range]
        
        # Multiplicative update: f_new = f * exp(-eta * grad_centered)
        # Center the gradient to preserve sum approximately
        grad_c = grad - grad.mean()
        
        f_new = f * np.exp(-eta * grad_c)
        f_new = np.maximum(f_new, 1e-300)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            # Decrease eta slowly
            eta = max(0.01, eta * 0.999)
        else:
            f = f_new  # still move, but track no improvement
            no_improve += 1
            if no_improve % 50 == 0:
                eta = max(0.005, eta * 0.95)
            if no_improve > 200:
                # Restart with noise
                rng = np.random.default_rng(no_improve)
                f = best_f + rng.standard_normal(n) * 0.003
                f = np.maximum(f, 0)
                f = f / f.sum()
                eta = 0.05
                no_improve = 0
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-dual-guided mirror descent,
with adaptive multiplicative updates and coordinate descent polishing.
Key innovation: Uses the dual of the peak-constraint LP to obtain optimally-weighted
subgradients, combined with entropy-regularized (multiplicative) updates for
natural non-negativity handling.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def solve_lp_dual(f, M, n):
    """
    Solve the dual LP: min M * sum(y) s.t. T_f^T y >= 1, y >= 0
    where (T_f^T y)_i = sum_k f(k-i) * y_k for valid k.
    
    Returns optimal y (normalized to sum 1) or None on failure.
    """
    # Dual: min M * 1^T y s.t. (T_f^T y)_i >= 1 for all i, y >= 0
    # (T_f^T y)_i = sum_k (T_f)_{k,i} * y_k = sum_k f(k-i) * y_k
    # Valid k for each i: k ranges from i to i+n-1 (since k-i must be in [0, n-1])
    
    m = 2 * n - 1  # number of y variables
    # c = M * ones(m)
    c = np.full(m, M)
    
    # Constraints: (T_f^T y)_i >= 1 for i = 0, ..., n-1
    # => -(T_f^T y)_i <= -1
    A_ub = np.zeros((n, m))
    b_ub = -np.ones(n)
    
    for i in range(n):
        for k in range(i, i + n):
            if k < m:
                A_ub[i, k] = -f[k - i]
    
    bounds = [(0, None)] * m
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        y = np.maximum(y, 0)

        if s > 1e-15:
            return y / s
    return None


def lp_guided_mirror_descent(f0, time_limit, verbose=False):
    """
    Mirror descent with LP-dual-guided gradients.
    Uses multiplicative updates for natural non-negativity.
    """
    n = len(f0)
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    eta = 0.08  # step size
    lp_interval = 50  # solve LP every 50 iterations
    no_improve = 0
    iter_count = 0
    lp_weights = None  # cached LP dual weights
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Periodically re-solve LP for fresh weights
        if iter_count % lp_interval == 1 or lp_weights is None:
            y = solve_lp_dual(f, M, n)
            if y is not None:
                lp_weights = y
        
        # Compute gradient using LP weights or fallback to log-sum-exp
        if lp_weights is not None:
            # grad[i] = 2 * sum_k lp_weights[k] * f[k-i] for valid (k,i)
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if lp_weights[k] < 1e-12:
                    continue
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                i_range = np.arange(i_start, i_end + 1)
                grad[i_range] += 2.0 * lp_weights[k] * f[k - i_range]
        else:
            # Fallback: log-sum-exp with adaptive temperature
            top_k = min(50, 2 * n - 1)
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            tau = M * 30
            shifted = conv[peak_indices] - conv[peak_indices].max()
            w = np.exp(shifted / tau)
            w /= w.sum()
            
            grad = np.zeros(n)
            for idx in range(len(peak_indices)):
                if w[idx] < 1e-12:
                    continue
                k = int(peak_indices[idx])
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                i_range = np.arange(i_start, i_end + 1)
                grad[i_range] += 2.0 * w[idx] * f[k - i_range]
        
        # Center gradient
        grad_c = grad - grad.mean()
        
        # Multiplicative update with clipping for numerical stability
        update = -eta * grad_c
        update = np.clip(update, -5, 5)  # prevent overflow
        f_new = f * np.exp(update)
        f_new = np.maximum(f_new, 1e-300)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            eta = max(0.02, eta * 0.9995)
        else:
            f = f_new
            no_improve += 1
            if no_improve > 100:
                eta = max(0.01, eta * 0.95)
                lp_weights = None  # force LP re-solve
            if no_improve > 300:
                # Perturb and restart
                rng = np.random.default_rng(iter_count * 17 + 3)
                f = best_f + rng.standard_normal(n) * 0.003
                f = np.maximum(f, 0)
                f = f / f.sum()
                eta = 0.06
                lp_weights = None
                no_improve = 0
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MD iter {iter_count}: obj={best_obj:.10f}, eta={eta:.4f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent with local search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.4, 0.6, 0.8, 1.25, 1.6, 2.5]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj


def pair_swap_search(a0, time_limit, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def triangular_init(n):
    """Triangular distribution initialization."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def gaussian_init(n):
    """Gaussian initialization."""
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / 2)
    return a / a.sum()


def solve_lp_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k in 0..2n-2.
    Returns (g, y) where g is the primal solution and y is the dual solution,
    both normalized. Returns (None, None) on failure.
    """
    m = 2 * n - 1
    
    # Primal: max 1^T g s.t. T_f * g <= M*1, g >= 0
    c_prim = -np.ones(n)
    A_prim = np.zeros((m, n))
    b_prim = np.full(m, M)
    
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_prim[k, i_start:i_end+1] = f[i_end:i_start-1:-1] if i_start > 0 else f[0:i_end+1][::-1]
        # Actually let me be more careful
        # (f*g)(k) = sum_i f[k-i] * g_i for valid i
        # For i in [i_start, i_end]: f[k-i]
        # So A_prim[k, i] = f[k-i]
        # Let me redo
        A_prim[k, :] = 0.0
    
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_prim[k, i] = f[k - i]
    
    bounds = [(0, None)] * n
    result_p = linprog(c_prim, A_ub=A_prim, b_ub=b_prim, bounds=bounds, method='highs',
                       options={'time_limit': 10})
    
    g = None
    if result_p.success:
        g = np.maximum(result_p.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        else:
            g = None
    
    # Dual: min M * sum(y) s.t. T_f^T * y >= 1, y >= 0
    # (T_f^T y)_i = sum_k f(k-i) * y_k for valid k (i <= k <= i+n-1, k < 2n-1)
    c_dual = np.full(m, M)
    A_dual = np.zeros((n, m))
    b_dual = -np.ones(n)
    
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, m - 1)
        for k in range(k_start, k_end + 1):
            A_dual[i, k] = -f[k - i]
    
    bounds_dual = [(0, None)] * m
    result_d = linprog(c_dual, A_ub=A_dual, b_ub=b_dual, bounds=bounds_dual, method='highs',
                       options={'time_limit': 10})
    
    y = None
    if result_d.success:
        y = np.maximum(result_d.x, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = None
    
    return g, y


def compute_grad_from_weights(f, weights):
    """Compute gradient: grad[i] = 2 * sum_k weights[k] * f[k-i] for valid (k,i)."""
    n = len(f)
    # Use cross-correlation: grad[i]/2 = sum_j weights[i+j] * f[j]
    # This is np.correlate(weights, f, mode='full')[n-1 : 2*n-1]
    full_corr = np.correlate(weights, f, mode='full')
    grad = 2.0 * full_corr[n-1:2*n-1]
    return grad


def compute_subgrad_lse(f, conv, top_k=80):
    """Compute subgradient using log-sum-exp smoothing."""
    n = len(f)
    k_actual = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    tau = max(pv.max() * 25, 1e-10)
    scaled = (pv - pv.max()) / tau
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Build full weight vector
    weights = np.zeros(2 * n - 1)
    weights[peaks] = w
    
    return compute_grad_from_weights(f, weights)


def lp_guided_optimize(f0, time_limit, verbose=False):
    """
    Main optimization: LP-guided blending + mirror descent refinement.
    Alternates between:
    1. LP solve to get direction g, blend f with g via line search
    2. Mirror descent steps using LP dual weights for gradient
    """

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    no_improve = 0
    iter_count = 0
    lp_interval = 30  # solve LP every 30 iterations
    mirror_interval = 5  # do mirror descent every 5 iterations
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        if iter_count % lp_interval == 1:
            # Solve LP for fresh direction
            g, y = solve_lp_direction(f, M, n)
            
            if g is not None and iter_count % mirror_interval == 1:
                # LP-guided blending with line search
                best_t = 0.0
                best_t_obj = best_obj
                for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    f_trial = (1.0 - t_val) * f + t_val * g
                    f_trial = np.maximum(f_trial, 0)
                    s_trial = f_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    f_trial = f_trial / s_trial
                    obj_trial = fast_convolve(f_trial).max()
                    if obj_trial < best_t_obj:
                        best_t_obj = obj_trial
                        best_t = t_val
                
                if best_t > 0:
                    f = (1.0 - best_t) * f + best_t * g
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    obj_new = fast_convolve(f).max()
                    if obj_new < best_obj - 1e-14:
                        best_obj = obj_new
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                        f = best_f.copy()
                    continue
            
            # Store y for use in gradient steps
            lp_y = y
        
        # Mirror descent step using current weights
        if 'lp_y' in dir() and lp_y is not None and iter_count % mirror_interval == 0:
            weights = lp_y
        else:
            weights = None
        
        if weights is not None:
            grad = compute_grad_from_weights(f, weights)
        else:
            grad = compute_subgrad_lse(f, conv, top_k=60)
        
        grad_c = grad - grad.mean()
        grad_std = np.std(grad_c)
        if grad_std < 1e-15:
            continue
        grad_c = grad_c / grad_std  # normalize
        
        # Adaptive step size
        eta = 0.05 * max(0.3, 1.0 - no_improve / 200.0)
        
        update = np.clip(-eta * grad_c, -3, 3)
        f_new = f * np.exp(update)
        f_new = np.maximum(f_new, 1e-300)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            f = f_new
            no_improve += 1
        
        if no_improve > 200:
            # Perturb and reset
            rng = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.003 + 0.003 * min(no_improve // 200, 3)
            f = best_f + rng.standard_normal(n) * noise_scale
            f = np.maximum(f, 0)
            f = f / f.sum()
            no_improve = 0
            lp_interval = 20  # solve LP more frequently after restart
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MD iter {iter_count}: obj={best_obj:.10f}, no_imp={no_improve} ({elapsed:.0f}s)")
    
    return best_f, best_obj
