def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_exact(f, K=1):

        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * f[k_star - i]
        return grad, conv, k_star
    else:
        # Use softmax weights over top-K lags

        grad, conv, active = subgradient_exact(f, K=K)
        
        # BB step size
        if prev_grad is not None:
            s_prev = f - prev_f
            y_prev = grad - prev_grad
            sy = np.dot(s_prev, y_prev)
            yy = np.dot(y_prev, y_prev)
            ss = np.dot(s_prev, s_prev)
            if yy > 1e-15:
                alpha_2 = sy / yy if sy > 1e-15 else alpha
                alpha_1 = ss / sy if sy > 1e-15 else alpha
                alpha = 0.5 * (alpha_1 + alpha_2)
                alpha = np.clip(alpha, 1e-6, 1.0)
        
        prev_f = f.copy()
        prev_grad = grad.copy()
        
        # Take step with backtracking
        best_trial_obj = best_obj
        best_trial_f = f
        trial_alpha = alpha
        for _ in range(8):
            f_trial = project_simplex(f - trial_alpha * grad)
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < f.max() * 0 + best_obj:  # any improvement
                best_trial_obj = obj_trial
                best_trial_f = f_trial
            trial_alpha *= 0.5
            if trial_alpha < 1e-8:
                break
        
        if best_trial_obj < best_obj - 1e-14:
            f = best_trial_f
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
        else:
            # Try a small step in the subgradient direction
            f_trial = project_simplex(f - 0.001 * grad)
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_obj - 1e-14:
                f = f_trial
                best_obj = obj_trial
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 200:
            # Perturb
            seed = int(elapsed * 1000) % (2**31)
            rng_p = np.random.default_rng(seed)
            noise = rng_p.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            prev_f = f.copy()
            prev_grad = None
            alpha = 0.1
            no_improve = 0
    
    return best_f, best_obj


def lpfw_optimize_v2(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with tighter relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = _lp_direction_cached(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                # Perturb
                seed = int(time.monotonic() * 1000) % (2**31)
                rng_p = np.random.default_rng(seed)
                f = project_simplex(best_f + rng_p.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            m_exp = 1.0
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 6.0)
    
    return best_f, best_obj


def _lp_direction_cached(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def pairwise_cd_v2(f0, time_limit, verbose=False):
    """Improved pairwise coordinate descent focusing on high-impact pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    rng = np.random.default_rng(42)
    
    ratios = np.linspace(0.05, 0.95, 19)
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        n_pairs = min(500 * (n // 250 + 1), 5000)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Choose pairs weighted by magnitude
            probs = f / f.sum()
            i = int(rng.choice(n, p=probs))
            j = int(rng.choice(n, p=probs))
            if i == j:
                continue
            
            old_i = f[i]
            old_j = f[j]
            total_ij = old_i + old_j
            
            if total_ij < 1e-12:
                continue
            
            cur_max = fast_convolve(f).max()
            best_obj_local = cur_max
            best_ratio = old_i / total_ij
            
            for ratio in ratios:
                if abs(ratio - best_ratio) < 0.03:
                    continue
                f_trial = f.copy()
                f_trial[i] = ratio * total_ij
                f_trial[j] = (1.0 - ratio) * total_ij
                val = fast_convolve(f_trial).max()
                if val < best_obj_local - 1e-14:
                    best_obj_local = val
                    best_ratio = ratio
            
            if abs(best_ratio - old_i / total_ij) > 1e-10:
                f[i] = best_ratio * total_ij
                f[j] = (1.0 - best_ratio) * total_ij
                if best_obj_local < best_obj - 1e-13:
                    best_obj = best_obj_local
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj


def make_window(n, window_type='cosine', power=2.0, asym=0.0):
    """Generate a windowed initialization."""
    x = np.linspace(0, 1, n)
    if window_type == 'cosine':
        a = np.sin(np.pi * x) ** power
    elif window_type == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 'geo':
        a = 0.995 ** np.arange(n, dtype=float)
    elif window_type == 'sym_geo':
        half = (n + 1) // 2
        asc = 0.99 ** np.arange(half, dtype=float)
        desc = 0.99 ** np.arange(half - 1, 0, -1, dtype=float)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            a = np.concatenate([a, np.ones(n - len(a))])
    else:
        a = np.ones(n)
    
    # Add asymmetry
    if asym != 0.0:
        x_shift = (1 - asym) * x
        x_shift = np.clip(x_shift, 0, 1)
        a = a * (0.5 + 0.5 * np.sin(np.pi * x_shift) ** power)
    
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method optimizer combining:
    1. Diverse windowed initializations
    2. Smooth LSE optimization with tuned temperature
    3. Exact subgradient with BB step
    4. LP-Frank-Wolfe refinement
    5. Pairwise coordinate descent
    6. Multi-length exploration with warm starts
    
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass
    
    n_parent = len(parent) if parent is not None else 768
    
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old:.8f}]")
    
    # === Phase 0: Diverse Initializations (60s) ===
    print("=== Phase 0: Diverse Initializations ===")
    t_start = time.monotonic()
    
    lengths = [n_parent]
    if n_parent != 512:
        lengths.append(512)
    if n_parent != 640:
        lengths.append(640)
    if n_parent != 768:
        lengths.append(768)
    if n_parent != 896:
        lengths.append(896)
    
    for n_try in lengths:
        if time.monotonic() - t_start > 60:
            break
        for wt in ['cosine', 'hann', 'blackman']:
            for pw in [1.5, 2.0, 3.0, 4.0]:
                g = make_window(n_try, wt, power=pw)
                _try(g, f"{wt}_p{pw}_n{n_try}")
        
        for asym in [-0.2, -0.1, 0.0, 0.1, 0.2, 0.3]:
            g = make_window(n_try, 'cosine', power=2.0, asym=asym)
            _try(g, f"asym_{asym}_n{n_try}")
        
        g = make_window(n_try, 'geo')
        _try(g, f"geo_n{n_try}")
    
    print(f"  After Phase 0: best={best_score:.8f}")
    
    # === Phase 1: Smooth Optimization (400s) ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem, 350)
        print(f"  Smooth opt n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_optimize_v2(best_seq, time_limit=t_alloc,
                                       T_start=1.0, T_end=0.003, verbose=True)
        _try(a_opt, "smooth_p1")
    
    # Secondary smooth with different params
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        t_alloc = min(rem, 200)
        f_init = 0.7 * best_seq + 0.3 * make_window(len(best_seq), 'cosine', 2.0)
        f_init = project_simplex(f_init)
        a_opt, _ = smooth_optimize_v2(f_init, time_limit=t_alloc,
                                       T_start=0.5, T_end=0.005)
        _try(a_opt, "smooth_blend")
    
    # === Phase 2: Subgradient + BB (200s) ===
    print(f"\n=== Phase 2: Subgradient BB (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem, 180)
        print(f"  BB subgradient n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = bb_subgradient_opt(best_seq, time_limit=t_alloc, K_max=3)
        _try(a_opt, "subgrad_bb")
    
    # === Phase 3: LP-FW (200s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        t_alloc = min(rem, 180)
        print(f"  LP-FW n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize_v2(best_seq, time_limit=t_alloc)
        _try(a_opt, "lpfw_p3")
    
    # LP-FW from perturbed
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        noise = rng.standard_normal(len(best_seq)) * 0.005
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(rem, 100)
        a_opt, _ = lpfw_optimize_v2(f_pert, time_limit=t_alloc)
        _try(a_opt, "lpfw_pert")
    
    # === Phase 4: Pairwise CD (150s) ===
    print(f"\n=== Phase 4: Pairwise CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 15:
        t_alloc = min(rem, 130)
        print(f"  Pairwise CD t={t_alloc:.0f}s")
        a_opt, _ = pairwise_cd_v2(best_seq, time_limit=t_alloc)
        _try(a_opt, "pairwise_cd")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for target_n in [n_best // 2, n_best + 64, n_best + 128]:
        if target_n < 64 or target_n > 2048:
            continue
        rem = deadline - time.monotonic() - 120
        if rem < 40:
            break
        f_interp = interpolate_seq(best_seq, target_n)
        
        t1 = min(rem * 0.4, 60)
        a_opt, _ = smooth_optimize_v2(f_interp, time_limit=t1, T_start=0.5, T_end=0.01)
        _try(a_opt, f"smooth_n{target_n}")
        
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = lpfw_optimize_v2(f_interp, time_limit=t2)
            _try(a_opt, f"lpfw_n{target_n}")
    
    # === Phase 6: Final Polish (200s) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"  Fine smooth t={t_alloc:.0f}s")
            a_opt, _ = smooth_optimize_v2(best_seq, time_limit=t_alloc,
                                           T_start=0.05, T_end=0.002)
            _try(a_opt, "smooth_fine")
        
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, _ = lpfw_optimize_v2(best_seq, time_limit=t_alloc)
            _try(a_opt, "lpfw_final")
        
        # Short BB subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, _ = bb_subgradient_opt(best_seq, time_limit=t_alloc, K_max=2)
            _try(a_opt, "bb_final")
        
        # Multiple short restarts
        for restart in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 20)
            a_opt, _ = smooth_optimize_v2(f_pert, time_limit=t_alloc,
                                           T_start=0.02, T_end=0.002)
            _try(a_opt, f"restart_{restart}")
    
    # Fallback
    if best_seq is None:
        best_seq = make_window(512, 'cosine', 2.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (deadline - budget_s + 20):.1f}s")
    return [float(x) for x in best_seq]
