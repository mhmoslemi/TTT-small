3. Peak-separation local search (structured pairwise moves)

import numpy as np

def fast_conv(a):
    """Linear auto-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, n_iter=300, alpha=0.6):
    """Spectral peak shaving: iteratively reduce peaks in |FFT(a)|^2."""
    n = len(a)
    fft_size = 2 * n - 1
    fft_size = 1 << (fft_size - 1).bit_length()
    a = normalize(a.copy())
    for i in range(n_iter):
        F = np.fft.rfft(a, fft_size)
        mag = np.abs(F)
        peak = mag.max()
        if peak < 1e-15:
            break
        # Soft threshold: compress values above threshold
        threshold = peak * (0.7 + 0.2 * (i / n_iter))
        new_mag = np.where(mag > threshold,
                           threshold + (mag - threshold) * alpha,
                           mag)
        new_F = new_mag * np.exp(1j * np.angle(F))
        new_a = np.fft.irfft(new_F, fft_size)[:n]
        a = np.maximum(new_a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a /= s
        else:
            break
    return a


def adam_subgradient(a, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                     eps=1e-8, top_k=40, temp_scale=0.01):
    """Adam-optimized subgradient descent in log-space with adaptive peak weighting."""
    n = len(a)
    a = normalize(a.copy())
    log_a = np.log(np.maximum(a, 1e-15))
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    step = 0
    patience = 0
    max_patience = 3000

    while time.monotonic() - t0 < time_limit:
        step += 1
        conv = fast_conv(a)

        # Adaptive top-k: use more peaks when stuck
        k_actual = min(top_k + patience // 500, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Weight by peak values (higher peaks get more attention)
        max_pv = pv.max()
        if max_pv < 1e-15:
            break
        w = pv / max_pv
        w = w ** 2  # Squared weighting for more focus on dominant peaks

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(k_actual):
            if w[i] < 1e-10:
                continue
            p = int(peaks[i])
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * a[p - idx]

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        else:
            break

        # Adam update in log-space
        m = beta1 * m + (1.0 - beta1) * sg
        v = beta2 * v + (1.0 - beta2) * sg * sg
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Adaptive learning rate (decay over time)
        lr_eff = lr / (1.0 + step * 1e-5)
        log_a = log_a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Restart with perturbation
            log_a = np.log(np.maximum(best_a, 1e-15)) + np.random.standard_normal(n) * 0.01
            a = np.exp(log_a)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            patience = 0
            lr = lr * 0.5

    return best_a, best_score

def block_coord_descent(a, time_limit, block_size=2, max_passes=50):
    """
    Block coordinate descent: optimize blocks of coordinates simultaneously
    using Nelder-Mead or Powell method.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    pass_count = 0
    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break

            block_idx = list(order[start:start + block_size])
            if len(block_idx) < block_size:
                break

            old_vals = a[block_idx].copy()
            other_sum = 1.0 - old_vals.sum()

            if other_sum < 1e-12:
                continue

            def block_obj(params):
                params = np.maximum(params, 0.0)
                s_params = params.sum()
                if s_params > 1.0 - 1e-12:
                    return float('inf')
                new_a = a.copy()
                # Scale other elements proportionally
                scale = (1.0 - s_params) / other_sum
                mask = np.ones(n, dtype=bool)
                mask[block_idx] = False
                new_a[mask] *= scale
                new_a[block_idx] = params
                conv = fast_conv(new_a)
                return conv.max()

            try:
                res = minimize(block_obj, old_vals, method='Nelder-Mead',
                               options={'xatol': 1e-10, 'fatol': 1e-14,
                                        'maxiter': 50, 'maxfev': 100})
                if res.fun < block_obj(old_vals) - 1e-13:
                    new_a = a.copy()
                    new_params = np.maximum(res.x, 0.0)
                    s_params = new_params.sum()
                    if s_params < 1.0 - 1e-12:
                        scale = (1.0 - s_params) / other_sum
                        mask = np.ones(n, dtype=bool)
                        mask[block_idx] = False
                        new_a[mask] *= scale
                        new_a[block_idx] = new_params
                        new_a = normalize(new_a)
                        sc = compute_score(new_a)
                        if sc < best_score - 1e-13:
                            a = new_a
                            best_score = sc
                            best_a = a.copy()
                            improved = True
            except Exception:
                pass

        if not improved and pass_count > 5:
            break

def cosine_bell(n):

def interpolate_sequence(f, new_n):
    """Interpolate to new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

    t3 = min(rem * 0.25, 100)
    if rem > t3 + 30:
        a_ps, sc_ps = peak_separation_search(a, time_limit=t3, n_trials=3000)
        if sc_ps < best_score:
            best_score = sc_ps
            best_a = a_ps.copy()
            a = a_ps.copy()
        if verbose:
            print(f"      After peak_sep: {best_score:.10f}")

    # Phase 4: Block coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    t4 = min(rem * 0.2, 80)
    if rem > t4 + 20:
        a_bc, sc_bc = block_coord_descent(a, time_limit=t4, block_size=2)
        if sc_bc < best_score:
            best_score = sc_bc
            best_a = a_bc.copy()
            a = a_bc.copy()
        if verbose:
            print(f"      After block_CD: {best_score:.10f}")

    # Phase 5: Precise 1D coordinate descent polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        t5 = min(rem * 0.8, 100)
        a_1d, sc_1d = precise_coord_1d(a, time_limit=t5)
        if sc_1d < best_score:
            best_score = sc_1d
            best_a = a_1d.copy()
        if verbose:
            print(f"      After 1D_CD: {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Tries multiple initializations and optimization
    strategies to find the best sequence minimizing 2*n*max(auto_conv)/sum^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_a = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01 or len(a) == 0:
            return float('inf')
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # === Collect starting points ===
    print("=== Collecting starting points ===")
    starts = []

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            starts.append((parent, "parent"))
            print(f"  Parent: n={len(parent)}, score={compute_score(parent):.10f}")
    except Exception:
        pass

    # Structured initializations at various lengths
    n_choices = [512, 640, 768, 896, 1024, 1152, 1280]

    for n in n_choices:
        # Geometric (decaying)
        for r in [0.998, 0.999, 0.9995, 0.9998]:
            starts.append((geometric_sequence(n, r), f"geom_r{r}_n{n}"))
            starts.append((reverse_geometric_sequence(n, r), f"rgeom_r{r}_n{n}"))

    # Cosine bell
    for n in [640, 768, 1024]:
        starts.append((cosine_bell(n), f"cosbell_n{n}"))

    # Two bumps
    for n in [768, 1024]:
        starts.append((two_bump_sequence(n), f"twobump_n{n}"))

    # Random smooth
    for n in [768, 1024]:
        a = rng.random(n)
        kernel_size = max(5, n // 30)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        starts.append((normalize(a), f"randsmooth_n{n}"))

    # Spectrally flattened versions of some starts
    for idx in [0, 1, 2]:
        if idx < len(starts):
            a, label = starts[idx]
            a_f = spectral_flatten(a, n_iter=100, alpha=0.5)
            starts.append((a_f, f"specflat_{label}"))

    # Quick evaluation of all starts
    print(f"  Evaluating {len(starts)} starts...")
    start_scores = []
    for a, label in starts:
        sc = compute_score(a)
        start_scores.append((sc, a, label))
    start_scores.sort(key=lambda x: x[0])

    for i, (sc, a, label) in enumerate(start_scores[:8]):
        print(f"    Start #{i+1} [{label}]: {sc:.10f}")
        try_seq(a, label)

    # === Phase 1: Optimize top diverse starts ===
    print(f"\n=== Phase 1: Optimize diverse starts (budget to {deadline - time.monotonic():.0f}s) ===")

    # Select diverse starts (different types/lengths)
    selected = []
    selected_n = set()
    selected_type = set()
    for sc, a, label in start_scores:
        n = len(a)
        base_type = label.split('_n')[0] if '_n' in label else label
        if base_type not in selected_type:
            selected.append((a, label, n))
            selected_type.add(base_type)
            selected_n.add(n)
        if len(selected) >= 8:
            break

    # Ensure best is included
    best_start = start_scores[0]
    if not any(label == best_start[2] for _, label, _ in selected):
        selected.append(best_start)

    # Also include parent if it's competitive
    if parent is not None:
        parent_sc = compute_score(parent)
        if parent_sc < best_score * 1.02:
            if not any(label == "parent" for _, label, _ in selected):
                selected.append((parent, "parent", len(parent)))

    n_sel = len(selected)
    # Allocate time: more time for promising starts
    total_time_available = deadline - time.monotonic() - 400
    print(f"  {n_sel} starts, {total_time_available:.0f}s available")

    for i, (a_init, label, n) in enumerate(selected):
        rem = deadline - time.monotonic() - 350
        if rem < 30:
            break
        # More time for better starts
        quality = start_scores[0][0] / max(start_scores[i][0] if i < len(start_scores) else best_score, 1e-10)
        t_alloc = min(rem / max(1, n_sel - i), 200)
        t_alloc = max(t_alloc, 30)

        print(f"  [{i+1}/{n_sel}] Optimizing [{label}] n={n}: t={t_alloc:.0f}s")
        a_opt, sc_opt = multi_phase_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, label + "_opt")

    # === Phase 2: Dimension exploration ===
    print(f"\n=== Phase 2: Dimension exploration (best={best_score:.10f}) ===")
    if best_a is not None:
        best_n = len(best_a)
        # Try nearby lengths
        nearby_ns = set()
        for delta in [-64, -32, 32, 64, 128, -128, 192]:
            target_n = best_n + delta
            if 256 <= target_n <= 2048:
                nearby_ns.add(target_n)

        for target_n in sorted(nearby_ns):
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 100)
            a_interp = interpolate_sequence(best_a, target_n)
            a_opt, sc = multi_phase_optimize(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"dim_{target_n}")

    # === Phase 3: Aggressive refinement on best ===
    print(f"\n=== Phase 3: Aggressive refinement (best={best_score:.10f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 100

        if rem > 30:
            # Extended peak separation
            t_ps = min(rem * 0.3, 150)
            print(f"  Extended peak sep: t={t_ps:.0f}s")
            a_ps, sc = peak_separation_search(best_a, time_limit=t_ps, n_trials=10000)
            try_seq(a_ps, "ext_peak_sep")

        rem = deadline - time.monotonic() - 80
        if rem > 30:
            # Adam subgradient with different params
            t_sg = min(rem * 0.3, 120)
            print(f"  Re-Adam SG: t={t_sg:.0f}s")
            a_sg, sc = adam_subgradient(best_a, time_limit=t_sg, lr=0.01, top_k=60)
            try_seq(a_sg, "re_adam_sg")

        rem = deadline - time.monotonic() - 60
        if rem > 20:
            # Final 1D polish
            t_1d = min(rem * 0.5, 80)
            print(f"  Final 1D CD: t={t_1d:.0f}s")
            a_1d, sc = precise_coord_1d(best_a, time_limit=t_1d, max_passes=50)
            try_seq(a_1d, "final_1d")

        rem = deadline - time.monotonic() - 30
        if rem > 15:
            # Tiny perturbations + quick re-optimize
            n_best = len(best_a)
            for trial in range(4):
                rem = deadline - time.monotonic() - 20
                if rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_a + noise)
                a_opt, sc = multi_phase_optimize(a_pert, time_limit=min(rem, 30))
                try_seq(a_opt, f"perturb_{trial}")

    # === Phase 4: Final spectral + local polish ===
    if best_a is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_f = spectral_flatten(best_a, n_iter=50, alpha=0.4)
            try_seq(a_f, "final_spec")
            if best_a is not None:
                a_1d, sc = precise_coord_1d(best_a, time_limit=max(5, rem - 5))
                try_seq(a_1d, "very_final_1d")

    # === Final verification ===
    if best_a is None:
        best_a = cosine_bell(768)
        best_score = compute_score(best_a)

    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in best_a]
    try:
        final_score = float(evaluate_sequence(final_list))
        if np.isfinite(final_score):
            if abs(final_score - best_score) > 1e-6:
                print(f"  NOTE: Internal score={best_score:.10f}, external={final_score:.10f}")
            best_score = min(best_score, final_score)
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_a)})")
    return final_list
