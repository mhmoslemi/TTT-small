Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a dual-ascent Frank-Wolfe approach
with adaptive peak weighting, targeted coordinate descent, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, lambda_vec):
    """Compute subgradient g_i = 2 * sum_k lambda_k * a[k-i] (valid indices only)."""
    n = len(a)
    g = np.zeros(n)
    # For each k with non-zero lambda, add lambda_k * 2 * a[k-i] for valid i
    for k in range(len(lambda_vec)):
        lk = lambda_vec[k]
        if lk < 1e-15:
            continue
        # Valid i: 0 <= i < n and 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            g[i_lo:i_hi+1] += 2.0 * lk * a[k - i_hi:k - i_lo - 1:-1]
    return g


def compute_subgradient_fast(a, conv, peak_indices, weights):
    """Fast subgradient using only specified peaks."""
    n = len(a)
    g = np.zeros(n)
    for idx, w in zip(peak_indices, weights):
        k = int(idx)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            # a[k-i] for i in [i_lo, i_hi] is a[k-i_hi : k-i_lo+1] reversed
            segment = a[k - i_hi : k - i_lo + 1][::-1]
            g[i_lo:i_hi+1] += 2.0 * w * segment
    return g


def subgradient_mirror_descent(f0, time_limit, verbose=False):
    """
    Mirror descent (multiplicative weights) for minimizing max_k (a*a)_k
    over the simplex. Uses adaptive peak weighting.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = 1.0
    eta_decay = 0.999
    it = 0
    no_improve = 0
    num_peaks = 5
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Select peaks with softmax weighting
        k_sel = min(num_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k_sel)[-k_sel:]
        peak_vals = conv[peak_idx].copy()
        
        # Adaptive temperature: start hot, cool down
        temp = max(0.01, 0.5 * M / (1 + it * 0.001))
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        g = compute_subgradient_fast(f, conv, peak_idx, w)
        
        # Mirror descent update
        f = f / (1.0 + eta * g)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        conv_new = fast_convolve(f)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: try smaller step
            if no_improve > 5:
                f = np.maximum(f0 if no_improve > 50 else best_f, 1e-15)
                # Actually, revert to best and try smaller eta
                f = best_f.copy()
                eta *= 0.7
                no_improve = 0
        
        eta *= eta_decay
        num_peaks = min(num_peaks + (1 if it % 20 == 0 else 0), 30)
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 1e-15)
            f = f / f.sum()
            eta = 0.5
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    SMD iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, peaks={num_peaks}")
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient mirror descent
with log-sum-exp smoothing, targeted peak shaving via coordinate descent,
LP Frank-Wolfe direction finding, and multi-length exploration.
Key improvements: adaptive temperature scheduling, dual ascent for peak weights,
and structured restarts to escape local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peaks(a, conv, peak_indices, peak_weights):
    """Compute subgradient g_i = 2 * sum_k w_k * a[k-i] for specified peaks."""
    n = len(a)
    g = np.zeros(n)
    for idx, w in zip(peak_indices, peak_weights):
        k = int(idx)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            segment = a[k - i_hi : k - i_lo + 1][::-1]
            g[i_lo:i_hi + 1] += 2.0 * w * segment
    return g


def smooth_max_grad(a, conv, temperature):
    """Compute smoothed gradient of log-sum-exp approximation to max(conv)."""
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    # Only use peaks with significant weight
    threshold = 1e-4
    active = np.where(w > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
        w_active = np.array([1.0])
    else:
        w_active = w[active] / w[active].sum()
    return subgradient_at_peaks(a, conv, active, w_active)


def mirror_descent_optimize(f0, time_limit, verbose=False, initial_temp=0.02,
                            temp_growth=1.003, initial_eta=0.5, eta_decay=0.997):
    """
    Mirror descent (multiplicative weights) with adaptive log-sum-exp temperature.
    Minimizes max_k (a*a)_k over the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    temp = initial_temp
    eta = initial_eta
    it = 0
    no_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        # Compute smoothed gradient
        g = smooth_max_grad(f, conv, temp)

        # Mirror descent update (multiplicative weights)
        # f_i *= exp(-eta * g_i) equivalently f_i /= (1 + eta*g_i) for small eta
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: increase to smooth out, decrease to sharpen
        if no_improve > 50:
            temp *= temp_growth
            eta *= 0.9
        elif no_improve == 0 and it > 100:
            temp = max(0.005, temp * 0.98)

        eta *= eta_decay

        # Periodic restart
        if no_improve > patience:
            rng = np.random.default_rng(it * 13 + 7)
            noise_level = 0.003 + 0.002 * np.random.default_rng(it).random()
            f = np.maximum(best_f + rng.standard_normal(n) * noise_level, 1e-12)
            f = f / f.sum()
            eta = initial_eta * 0.5
            temp = initial_temp * 2
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, temp={temp:.4f}, eta={eta:.4f}")

        it += 1

    return best_f, best_obj


def peak_shaving_optimize(f0, time_limit, verbose=False):
    """
    Targeted coordinate descent: find the peak-contributing coordinate
    and reduce it, redistributing mass to low-contribution coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find the coordinate that most contributes to the peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        idx = np.arange(i_lo, i_hi + 1)
        products = f[idx] * f[peak_k - idx]
        j_max = idx[np.argmax(products)]
        partner = peak_k - j_max

        # Find coordinates far from the peak (low contribution)
        # These are good targets for receiving redistributed mass
        dist_from_peak = np.abs(np.arange(n) - j_max)
        far_coords = np.where(dist_from_peak > n // 4)[0]
        far_coords = far_coords[far_coords != j_max]

        if len(far_coords) == 0:
            far_coords = np.arange(n)
            far_coords = far_coords[far_coords != j_max]

        # Try reducing f[j_max] and redistributing
        old_val = f[j_max]
        if old_val < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        improved = False
        best_trial = None
        best_trial_obj = M

        # Try multiple reduction levels
        for frac in [0.7, 0.8, 0.85, 0.9, 0.5, 0.3]:
            new_val = old_val * frac
            diff = old_val - new_val
            if diff < 1e-12:
                continue

            trial = f.copy()
            trial[j_max] = new_val

            # Redistribute: add to far coordinates proportionally to their current values
            far_weights = trial[far_coords].copy()
            far_weights[far_weights < 1e-10] = 1e-10
            far_weights = far_weights / far_weights.sum()
            trial[far_coords] += diff * far_weights

            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s

            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-12:
                best_trial_obj = obj
                best_trial = trial.copy()

        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-12:
                best_obj = best_trial_obj
                best_f = f.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Try increasing the partner coordinate
            partner_val = f[partner]
            for frac in [1.1, 1.2, 1.3, 1.5]:
                diff = partner_val * (frac - 1)
                trial = f.copy()
                trial[partner] += diff
                # Take from j_max
                trial[j_max] -= diff
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-10:
                    continue
                trial = trial / s
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-12:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try a different peak
                top_peaks = np.argsort(conv)[-3:]
                for pk in top_peaks:
                    pk = int(pk)
                    i_lo2 = max(0, pk - n + 1)
                    i_hi2 = min(n - 1, pk)
                    idx2 = np.arange(i_lo2, i_hi2 + 1)
                    prods2 = f[idx2] * f[pk - idx2]
                    j2 = idx2[np.argmax(prods2)]
                    old2 = f[j2]
                    if old2 < 1e-10:
                        continue
                    for frac2 in [0.6, 0.7, 0.8]:
                        new2 = old2 * frac2
                        diff2 = old2 - new2
                        trial = f.copy()
                        trial[j2] = new2
                        others = np.arange(n)
                        others = others[others != j2]
                        ow = trial[others].copy()
                        ow[ow < 1e-10] = 1e-10

                        trial[others] += diff2 * ow
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj2 = fast_convolve(trial).max()
                        if obj2 < best_obj - 1e-12:

                            best_f = f.copy()

                            break
                    if no_improve == 0:

                if no_improve > 80:

                    f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-12)

                    no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        # (f*g)[k] = sum_j f_j * g_{k-j}
        # This is: for each k, sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j} <= M*m_exp
        # Let's rewrite: for each k, sum_{j} f_j * g_{k-j} <= M*m_exp (valid j only)
        # In matrix form: A_ub @ g <= b_ub
        # A_ub has shape (2n-1, n), A_ub[k, j] = f_j if k-j is in [0,n-1], else 0
        # Wait, (f*g)[k] = sum_j f_j g_{k-j}. For this to be valid, k-j must be in [0,n-1].
        # So A_ub[k, j] = f_j if 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k.

        # For large n, this LP has 2n-1 constraints and n variables.
        # For n=768, that's 1535 constraints and 768 variables - might be slow with linprog.
        # Use a smaller set of constraints (only top peaks) for speed.
        top_k = min(50, 2 * n - 1)
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        # Sort by value descending
        peak_positions = peak_positions[np.argsort(conv[peak_positions])[::-1]]

        c = -np.ones(n)
        A_ub_rows = []
        b_ub_vals = []
        for k in peak_positions:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            row = np.zeros(n)
            row[j_lo:j_hi + 1] = f[j_lo:j_hi + 1]
            A_ub_rows.append(row)
            b_ub_vals.append(M * m_exp)

        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if not res.success or res.x is None:
            m_exp *= 1.2
            if m_exp > 10:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 150:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine line search
            lo = max(0, best_alpha - 0.1)
            hi = min(1, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2
                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def dual_ascent_optimize(f0, time_limit, verbose=False):
    """
    Primal-dual method: alternate between optimizing a (primal) and p (dual peak weights).
    Inner primal step uses multiplicative weights for the quadratic a^T P(p) a.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = np.ones(conv_len) / conv_len
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 2000:
        it += 1

        # Primal step: minimize a^T P(p) a via a few mirror descent iterations
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            # (P @ a)_j = sum_l p[j+l] * a_l
            Pa = np.array([np.dot(p[j:j + n], f) for j in range(n)])
            grad = 2.0 * Pa
            eta_inner = 0.05
            f = f * np.exp(-eta_inner * grad)
            f = np.maximum(f, 1e-12)
            f = f / f.sum()

        # Dual step: update p toward the current peaks
        conv = fast_convolve(f)
        M = conv.max()
        # Exponentiated gradient on p
        p = p * np.exp(0.5 * conv / max(M, 1e-10))
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 50 == 0:
                print(f"    Dual iter {it}: obj={best_obj:.10f}")

        # Periodic perturbation
        if it % 300 == 0:
            rng = np.random.default_rng(it + 42)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, width_frac=0.15):
    """Bimodal: mass at edges."""
    x = np.arange(n)
    center = n / 2.0
    w = n * width_frac
    a = np.exp(-((x - n * 0.1) / w) ** 2) + np.exp(-((x - n * 0.9) / w) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_window(n):
    """Raised cosine window (Hanning)."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        try_seq(a_norm, label)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(768, 0.3)

    print(f"  Best init score: {best_score:.6f}")

    # === Phase 2: Mirror descent on best init ===
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 200)
        print(f"  MD: allocating {t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(best_init, time_limit=t_alloc, verbose=True,
                                             initial_temp=0.015, temp_growth=1.004,
                                             initial_eta=0.8, eta_decay=0.997)
        try_seq(a_opt, "md_phase2")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Peak shaving: allocating {t_alloc:.0f}s")
            a_opt, obj = peak_shaving_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peak_phase3")

    # === Phase 4: LP Frank-Wolfe ===
    print(f"\n=== Phase 4: LP Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  LP-FW: allocating {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_phase4")

    # === Phase 5: Dual Ascent ===
    print(f"\n=== Phase 5: Dual Ascent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem, 150)
            print(f"  Dual ascent: allocating {t_alloc:.0f}s")
            a_opt, obj = dual_ascent_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "dual_phase5")

    # === Phase 6: Try longer sequences ===
    print(f"\n=== Phase 6: Longer Sequences ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_{target_n}")

        t_alloc = min(rem * 0.4, 120)
        if t_alloc > 20:
            print(f"  MD n={target_n}: {t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"md_{target_n}")

    # === Phase 7: Iterative refinement with alternating methods ===
    print(f"\n=== Phase 7: Alternating Refinement ===")
    if best_seq is not None:
        n_best = len(best_seq)
        methods = ['md', 'peak', 'md', 'lpfw', 'peak', 'md', 'dual', 'md']
        for mi, method in enumerate(methods):
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem / (len(methods) - mi), 80)

            if method == 'md':
                a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=False,
                                                     initial_temp=0.01, temp_growth=1.005,
                                                     initial_eta=0.3, eta_decay=0.998)
            elif method == 'peak':
                a_opt, obj = peak_shaving_optimize(best_seq, time_limit=t_alloc, verbose=False)
            elif method == 'lpfw':
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            else:
                a_opt, obj = dual_ascent_optimize(best_seq, time_limit=t_alloc, verbose=False)

            try_seq(a_opt, f"alt_{method}_{mi}")

    # === Phase 8: Structured restarts ===
    print(f"\n=== Phase 8: Structured Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("symmetric", lambda a: (a + a[::-1]) / 2),
            ("blended_uniform", lambda a: 0.85 * a + 0.15 * np.ones_like(a) / len(a)),
            ("noise_small", lambda a: np.maximum(a + np.random.default_rng(123).standard_normal(len(a)) * 0.002, 0)),
            ("noise_med", lambda a: np.maximum(a + np.random.default_rng(456).standard_normal(len(a)) * 0.005, 0)),
            ("spectral_smooth", lambda a: np.real(np.fft.irfft(np.fft.rfft(a) * np.exp(-0.01 * np.arange(np.fft.rfft(a).shape[0])**2), len(a)))),
        ]
        for name, transform in restarts:
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            f_restart = transform(best_seq.copy())
            f_restart = np.maximum(f_restart, 1e-12)
            f_restart = f_restart / f_restart.sum()
            try_seq(f_restart, f"restart_{name}")

            t_alloc = min(rem, 60)
            a_opt, obj = mirror_descent_optimize(f_restart, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"restart_{name}_opt")

    # === Phase 9: Final intensive MD polish ===
    print(f"\n=== Phase 9: Final MD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Final MD: {t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                                 initial_temp=0.005, temp_growth=1.002,
                                                 initial_eta=0.2, eta_decay=0.999)
            try_seq(a_opt, "final_md")

    # === Phase 10: Final peak shave ===
    print(f"\n=== Phase 10: Final Peak Shave ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt, obj = peak_shaving_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_peak")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
