"""

Uses: (1) smooth L-BFGS with annealed log-sum-exp, (2) Nesterov momentum
subgradient with adaptive peak weighting, (3) LP-based Frank-Wolfe,

Key improvements: correct FFT-based gradient for smooth max, Nesterov
acceleration, adaptive peak focusing, and multi-resolution search.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * np.conj(A), fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.flatnonzero(cond)[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def smooth_max_and_grad(conv, beta, a):
    """
    Compute smooth max (log-sum-exp) of conv and its gradient w.r.t. a.
    Returns (smooth_value, grad_a, weights).
    """
    n = len(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
    # Using FFT: N >= len(a) + len(w) - 1 = n + (2n-1) - 1 = 3n - 2
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(A) * W, N)[:n]
    grad_a = 2.0 * R

    return obj, grad_a, w


def nesterov_subgrad(a0, time_limit, verbose=False):
    """
    Nesterov accelerated projected subgradient on smooth approximation.
    Adapts beta (smoothness) and learning rate over time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = 20.0
    lr = 0.02
    mu = 0.9  # momentum
    v = np.zeros(n)
    a_prev = a.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        if it > 1:
            y = a + (mu - 1.0) * (a - a_prev)
        else:
            y = a.copy()
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = best_a.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)
        sm, grad_y, w = smooth_max_and_grad(conv_y, beta, y)

        # Update velocity and position
        v = mu * v - lr * grad_y
        a_new = project_simplex(y + v)

        # Evaluate
        obj = fast_convolve(a_new).max()
        a_prev = a.copy()
        a = a_new

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
            beta = min(beta * 1.005, 2000.0)
            if verbose and it % 500 == 0:
                print(f"    Nesterov it {it}: obj={best_obj:.10f}, beta={beta:.0f}, lr={lr:.4f}")
        else:
            no_improve += 1
            lr *= 0.995
            beta = min(beta * 1.01, 2000.0)
            if no_improve > 200:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                lr = 0.01
                no_improve = 0

        if no_improve > 800:
            break

    return best_a, best_obj


def lbfqg_optimize(a0, time_limit, verbose=False):
    """L-BFGS on smoothed log-sum-exp objective with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    eps = 1e-15
    t = np.log(a + eps)

    betas = [10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    t0_time = time.monotonic()

    def obj_fn(t_val, beta_val):
        a_val = softmax_fn(t_val)
        a_val = np.maximum(a_val, 0.0)
        a_val = a_val / a_val.sum()
        conv = fast_convolve(a_val)
        obj, grad_a, _ = smooth_max_and_grad(conv, beta_val, a_val)
        # Chain rule: grad_t = a * (grad_a - a.dot(grad_a))
        dot = np.dot(a_val, grad_a)
        grad_t = a_val * (grad_a - dot)
        return obj, grad_t

    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 30:
            break

        try:
            res = minimize(
                obj_fn, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 50, 'maxfun': 3000}
            )
            t = res.x
            a_new = softmax_fn(t)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                t = np.log(a_new + eps)
                if verbose:
                    print(f"    L-BFGS beta={beta:.0f}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        g = solve_lp_direction(a, M * m_factor, n)

            if m_factor > 8.0:
                m_factor = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_factor = min(m_factor * 1.05, 5.0)
            continue

        # Backtracking line search

        best_val = M
        alpha = 0.5
        for _ in range(30):

def peak_shave(a0, time_limit, verbose=False):
    """
    Targeted peak shaving: identify top peaks, compute weighted subgradient,
    and do projected gradient steps with adaptive learning rate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    lr = 0.01
    momentum = 0.85
    v = np.zeros(n)
    temp = 0.001
    top_k = max(5, n // 32)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        conv_max = conv.max()

        # Select top-k peaks with temperature weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = (pv - conv_max) / temp
        w = np.exp(scaled)
        w /= w.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # grad[m] += 2 * w[i] * a[k-m] for valid m
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            temp *= 1.01
            if verbose and best_obj < 1.51:
                print(f"    PeakShave: obj={best_obj:.10f}, temp={temp:.5f}")
        else:
            no_improve += 1
            a = a_new  # still move (might help escape)
            lr *= 0.998
            temp *= 1.02

            if no_improve > 300:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0

        if no_improve > 1000:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initialization(n, method, rng):
    """Create initial sequences with different shapes."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "edge_peak":
        # Peaked at edges: a[i] ~ 1/sqrt(min(i+1, n-i))
        i = np.arange(n)
        a = 1.0 / np.sqrt(np.minimum(i + 1, n - i) + 0.1)
        return a / a.sum()
    elif method == "center_dip":
        # Dipped at center
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x**2
        return a / a.sum()
    elif method == "random":
        a = rng.random(n) ** 0.5
        return a / a.sum()
    elif method == "inv_sqrt":
        i = np.arange(n)
        a = 1.0 / np.sqrt(i + 1.0)
        return a / a.sum()
    elif method == "bell_edges":
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x**2) + 0.3
        return a / a.sum()
    else:
        return np.ones(n) / n
