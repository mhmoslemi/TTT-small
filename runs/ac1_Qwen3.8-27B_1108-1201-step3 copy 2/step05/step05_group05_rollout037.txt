def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def log_sum_exp_objective(a, T):
    """Compute log-sum-exp of conv(a,a)/T and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    # Stabilize
    c_max = c.max()
    shifted = (c - c_max) / T
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    obj = T * (np.log(Z) + c_max / T)
    
    # Compute gradient
    p = exp_shifted / Z  # softmax weights
    grad = np.zeros(n)
    
    # Only process significant peaks for efficiency
    threshold = 1e-10
    significant = p > threshold
    if not significant.any():
        # All weight on one peak
        k_max = np.argmax(c)
        p = np.zeros(len(c))
        p[k_max] = 1.0
        significant = np.array([k_max])
    
    for k in range(len(c)):
        if p[k] < threshold:
            continue

        #   Contribution 2: a[j-k] if j >= k
        # So for j from 0 to n-1:
        #   grad[j] += p[k] * (a[j+k]*1[j<n-k] + a[j-k]*1[j>=k])
        
        if k == 0:

        else:

            # Contribution 2: for j = k to n-1, add p[k]*a[j-k]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def lp_direction(a, M, method='highs'):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns optimal b.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A_ub such that A_ub @ b <= M for all constraints
    # A_ub[k, j] = a[j] if 0 <= k-j < n, else 0
    
    # For efficiency, build the matrix
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        # conv(a,b)[k] = sum_j a[j] * b[k-j] where 0<=k-j<n
        # = sum_j a[j] * b[k-j] for j in [max(0, k-n+1), min(n-1, k)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[j]
    
    b_ub_vec = np.full(num_constraints, M)
    
    # Solve: max c^T b s.t. A_ub b <= b_ub, b >= 0
    # linprog minimizes, so negate objective
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_vec, bounds=[(0, None)] * n, method=method)
        if res.success:
            return res.x
        else:
            return None
    except Exception:
        return None

        if s > 1e-15:

        new_obj = fast_convolve(a_new).max()
        
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for t_small in [t * 0.5, t * 0.25, t * 0.1]:
                a_try = (1 - t_small) * a + t_small * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try /= s_try
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
    
    return best_a, best_obj


def lbfgs_smooth(a0, T_start=1.0, T_end=0.05, time_limit=100, max_iter_per_stage=50):
    """
    L-BFGS optimization on the log-sum-exp smooth objective with decreasing T.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a /= a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Decrease T geometrically
    T = T_start
    stages = int(np.log(T_start / T_end) / np.log(2))  # Rough number of doublings
    
    for stage in range(stages + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = T_start / (2 ** stage)
        if T < T_end:
            T = T_end
        
        stage_time = min(30, (time_limit - (time.monotonic() - t0)) / max(stages - stage, 1))
        
        # Use scipy minimize with L-BFGS-B
        # We optimize on the simplex, so we parameterize with a free vector
        # and project after each step... Actually, L-BFGS-B doesn't handle
        # simplex constraints well. Let's use a log-barrier or just penalize.
        
        # Alternative: optimize in log-space or use a penalty for sum != 1
        def obj_and_grad(x):
            # x is free, we project to simplex
            a_proj = project_simplex(x, 1.0)
            c = fast_convolve(a_proj)
            c_max = c.max()
            shifted = (c - c_max) / T
            exp_shifted = np.exp(shifted)
            Z = exp_shifted.sum()
            obj = T * (np.log(Z) + c_max / T)
            
            # Gradient through projection is complex; use subgradient
            # For simplicity, use the gradient of the smooth max
            p = exp_shifted / Z
            grad = np.zeros(n)
            threshold = 1e-10
            for k in range(len(c)):
                if p[k] < threshold:
                    continue
                if k == 0:
                    grad[:] += 2.0 * p[k] * a_proj
                else:
                    if n - k > 0:
                        grad[:n-k] += p[k] * a_proj[k:]
                    if k < n:
                        grad[k:] += p[k] * a_proj[:n-k]
            return obj, grad
        
        # Starting point
        x0 = a * 2  # Scale up so projection is mostly identity
        
        try:
            res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_stage, 'maxfun': 200,
                                   'gtol': 1e-8, 'maxls': 50})
            a_new = project_simplex(res.x, 1.0)
        except Exception:
            a_new = a
        
        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60):
    """Coordinate descent with per-coordinate optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 2:
            break
    
    return best_a, best_obj


def pair_swap_search(a0, time_limit=60, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit=100):
    """Subgradient optimization with momentum (improved version)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    tau = 50.0
    lr = 0.03
    momentum = 0.85
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive parameters
        if no_improve > 300:
            momentum = min(0.95, momentum + 0.005)
            lr = max(0.002, lr * 0.95)
        elif no_improve > 100:
            tau = min(300.0, tau * 1.02)
            momentum = min(0.93, momentum + 0.002)
        elif no_improve > 30:
            tau = max(20.0, tau * 0.99)
            lr = min(0.06, lr * 1.01)
        else:
            tau = min(200.0, tau * 1.008)
        
        # Compute gradient
        c = fast_convolve(a)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        Z = exp_shifted.sum()
        p = exp_shifted / Z
        
        grad = np.zeros(n)
        threshold = 1e-8
        for k in range(len(c)):
            if p[k] < threshold:
                continue
            if k == 0:
                grad[:] += 2.0 * p[k] * a
            else:
                if n - k > 0:
                    grad[:n-k] += p[k] * a[k:]
                if k < n:
                    grad[k:] += p[k] * a[:n-k]
        
        # Center gradient
        grad_centered = grad - grad.mean()
        
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, 1.0)
        
        obj_new = fast_convolve(a_new).max()
        
        # Line search
        if obj_new < best_obj:
            a_ext = project_simplex(a + 2.0 * v, 1.0)
            obj_ext = fast_convolve(a_ext).max()
            if obj_ext < obj_new:
                a_new = a_ext
                obj_new = obj_ext
        
        a = a_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 400:
            rng_r = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.005 + 0.005 * min(no_improve // 400, 3)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, 1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.02
            tau = 40.0
            momentum = 0.85
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative direction finding,
smooth subgradient optimization with adaptive parameters, and coordinate descent polish.
Key innovation: Uses the LP direction-finding from the SOTA gradient pursuit approach
combined with multi-phase refinement strategies and adaptive step sizes.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def smooth_power_init(n, rng, alpha=2.0):
    """Smooth power-law initialization."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** alpha + (1 - x) ** alpha
    noise = rng.standard_normal(n) * 0.01
    a = np.maximum(a + noise, 0)
    a /= a.sum()
    return a


def build_lp_matrix(a):
    """Build LP constraint matrix for conv(a,b) <= M."""
    n = len(a)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A[k, idx] = a[k - idx]
    return A


def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    A_ub = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iterate(a0, max_iters=60, t_start=0.35, t_end=0.1, time_limit=200):

    Iterative LP-based optimization with adaptive blending.
    At each step: find b maximizing sum(b) s.t. conv(a,b)<=M, then blend.
    """

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    last_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if b is None:

        else:
            b_sum = b.sum()
            if b_sum < 1e-10:
                break
            b = b / b_sum
        
        # Adaptive step size with backtracking
        progress = it / max(max_iters - 1, 1)
        t = t_start * (t_end / t_start) ** progress
        
        # Try a range of step sizes
        best_new_obj = best_obj
        best_new_a = a
        for t_try in [t, t * 0.5, t * 1.5, t * 0.25]:
            if t_try <= 0 or t_try > 1:
                continue
            a_try = (1 - t_try) * a + t_try * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try /= s_try
            obj_try = fast_convolve(a_try).max()
            if obj_try < best_new_obj:
                best_new_obj = obj_try
                best_new_a = a_try
        
        if best_new_obj < best_obj - 1e-14:
            best_obj = best_new_obj
            best_a = best_new_a.copy()
            a = best_new_a
            last_improve = it
        else:
            # No improvement: add small perturbation to escape
            if it - last_improve > 5:
                rng_p = np.random.default_rng(it * 7 + 42)
                a = project_simplex(a + rng_p.standard_normal(n) * 0.005, 1.0)

def subgradient_optimize(a0, time_limit=100, tau_init=40.0, lr_init=0.025):
    """Subgradient optimization with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    momentum = 0.84
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Adaptive parameter scheduling
        if no_improve > 500:
            momentum = min(0.96, momentum + 0.003)
            lr = max(0.001, lr * 0.93)
        elif no_improve > 200:
            tau = min(400.0, tau * 1.015)
            momentum = min(0.93, momentum + 0.002)
        elif no_improve > 50:
            tau = max(15.0, tau * 0.992)
            lr = min(0.05, lr * 1.008)
        else:
            tau = min(250.0, tau * 1.006)
        
        # Compute smoothed gradient
        c = fast_convolve(a)
        c_max = c.max()
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        Z = exp_shifted.sum()
        if Z < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            p = exp_shifted / Z
        
        grad = np.zeros(n)
        threshold = 1e-9
        for k in range(len(c)):
            if p[k] < threshold:
                continue
            if k == 0:
                grad[:] += 2.0 * p[k] * a
            else:
                if n - k > 0:
                    grad[:n-k] += p[k] * a[k:]
                if k < n:
                    grad[k:] += p[k] * a[:n-k]
        
        # Center and apply momentum
        grad_centered = grad - grad.mean()
        v = momentum * v - lr * grad_centered
        a_new = project_simplex(a + v, 1.0)
        
        obj_new = fast_convolve(a_new).max()
        
        # Extended line search
        if obj_new < best_obj:
            for ext in [1.5, 2.0, 3.0]:
                a_ext = project_simplex(a + ext * v, 1.0)
                obj_ext = fast_convolve(a_ext).max()
                if obj_ext < obj_new:
                    a_new = a_ext
                    obj_new = obj_ext
                else:
                    break
        
        a = a_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            rng_r = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.004 + 0.004 * min(no_improve // 500, 4)
            a = project_simplex(best_a + rng_r.standard_normal(n) * noise_scale, 1.0)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.02
            tau = 35.0
            momentum = 0.84
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit=60):
    """Coordinate descent with per-coordinate line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            multipliers = [0.4, 0.6, 0.8, 1.0, 1.25, 1.5, 2.0, 0.2]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                old_remaining = 1.0 - old_val
                if old_remaining < 1e-10:
                    continue
                scale = (1.0 - new_val) / old_remaining
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and pass_count > 3:
            break
    
    return best_a, best_obj


def pair_swap_search(a0, time_limit=50, seed=0):
    """Randomized pair-swap local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)

    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    support = np.where(a > 1e-8)[0]
    if len(support) == 0:
        support = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            idx = rng.integers(0, len(support), size=2)
            i, j = int(support[idx[0]]), int(support[idx[1]])
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                for direction in [1, -1]:
                    transfer = direction * frac * min(a[i], a[j])
                    new_i = a[i] + transfer
                    new_j = a[j] - transfer
                    if new_i < 0 or new_j < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_i
                    a_trial[j] = new_j
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a_trial.copy()
                        a = a_trial.copy()
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
