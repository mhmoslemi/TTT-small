    max_b = np.max(conv)
    sum_a = s
    
    return float(2 * n * max_b / (sum_a ** 2))

def fast_convolve(seq):
    """Efficient full convolution using FFT."""
    seq_np = np.array(seq, dtype=np.float64)
    n = len(seq_np)
    # Zero-padded to avoid circular effects and make length power of two
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Compute FFT of the sequence and its square to convolve
    A = np.fft.rfft(np.append(seq_np, np.zeros(fft_size - n)))
    convolved = np.fft.irfft(A * A, fft_size)[:size]
    
    return convolved
