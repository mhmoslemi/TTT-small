import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_subgrad(a, T):
    """Compute subgradient of smooth max of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(T * conv_shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / w_sum
    # Subgradient: g_m = 2 * sum_j w[m+j] * a[j]
    corr = np.correlate(w, a, 'full')
    g = 2.0 * corr[n-1 : 2*n-1]
    return g, conv

def smooth_max_optimize(a0, time_limit, T_start=2.0, T_end=200.0, lr=0.005, momentum=0.9):

        a = a_new

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: structured initializations, smooth-max projected gradient with
adaptive temperature and Nesterov momentum, LP-based Frank-Wolfe,
Fourier-parameterized global search, multi-length exploration, and
final micro-perturbation search.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_subgrad(a, T):
    """Compute subgradient of (1/T)*log(sum_k exp(T*(a*a)[k]))."""
    n = len(a)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(T * conv_shifted)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / w_sum
    # g_m = 2 * sum_j w[m+j] * a[j]  (correlation)
    corr = np.correlate(w, a, 'full')
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return g, conv


def smooth_max_optimize(a0, time_limit, T_start=2.0, T_end=200.0,
                        lr=0.005, momentum=0.9, seed=0):

        a = a / s

        a_new = project_simplex(a + v)
        obj = fast_convolve(a_new).max()
        a = a_new

            best_a = a.copy()

        else:

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': 30, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def fourier_sequence(n, coeffs):
    """Build non-negative sequence from cosine coefficients."""
    j = np.arange(n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * np.cos(2.0 * np.pi * k * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def fourier_search(n, time_limit, seed=0):

        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def coordinate_descent_quick(f0, time_limit, seed=0):
    """Quick coordinate descent with multiplicative updates (few passes)."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)

        # Random subset of coordinates if n is large
        if n > 500:
            indices = rng.choice(n, size=min(200, n), replace=False)
            np.random.seed(indices)
        else:
            indices = range(n)

        for j in indices:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            multipliers = [0.5, 0.7, 0.9, 1.1, 1.3, 1.8]
            best_j_val = old_val
            best_j_obj = cur_conv.max()

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj

            best_seq = a_opt.copy()
            print(f"  NEW BEST [SMO {label}]: {sc2:.6f}")

    # === Phase 3: LP-FW on current best ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_lp = min(300, rem * 0.5)
        print(f"  n={n_best}, t={t_lp:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, verbose=True, seed=seed)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [LP-FW]: {sc:.6f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coord descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_cd = min(100, rem * 0.4)
        a_opt, _ = coordinate_descent_quick(best_seq, time_limit=t_cd, seed=seed + 1)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [CD]: {sc:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = sorted(set([
        n_best - 128, n_best - 64, n_best + 64, n_best + 128,
        n_best + 256, n_best - 256,
        512, 640, 896, 1024, 1152, 1280
    ]))
    candidate_lengths = [l for l in candidate_lengths
                         if 128 <= l <= 2048 and l != n_best][:6]

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        f_interp = interpolate_sequence(best_seq, target_n)

        # Quick SMO
        t_smo = min(50, rem * 0.3)
        if t_smo < 10:
            continue
        a_opt, _ = smooth_max_optimize(f_interp, time_limit=t_smo,
                                       T_start=3.0, T_end=100.0,
                                       lr=0.005, momentum=0.9,
                                       seed=seed + target_n)

        # LP-FW polish
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 15:
            t_lp = min(70, rem2 * 0.4)
            if t_lp > 5:
                a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_lp, seed=seed + target_n)

        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [n={target_n}]: {sc:.6f}")
        else:
            print(f"  n={target_n}: {sc:.6f}")

    # === Phase 6: Second round LP-FW + CD on best ===
    print(f"\n=== Phase 6: Second polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20:
        t_lp2 = min(120, rem * 0.5)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp2, seed=seed + 100)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [LP-FW2]: {sc:.6f}")

    # === Phase 7: Micro perturbations ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(20):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.00008
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick SMO polish on the improved point
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = smooth_max_optimize(best_seq, time_limit=min(8, rem2 * 0.3),
                                               T_start=50.0, T_end=200.0,
                                               lr=0.001, momentum=0.95,
                                               seed=seed + trial)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_smo]: {sc2:.6f}")

    # === Final LP-FW squeeze ===
    rem = deadline - time.monotonic()
    if rem > 8:
        t_final = min(rem * 0.85, 60)
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, seed=seed + 200)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL LP-FW: {sc:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
