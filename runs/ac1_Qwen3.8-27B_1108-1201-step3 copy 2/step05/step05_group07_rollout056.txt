"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated log-space subgradient descent with exact max subgradient
2. Multi-length exploration with interpolation
3. Coordinate descent polishing
4. Frank-Wolfe as alternative optimizer
5. Multiple restarts for escaping local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def compute_subgrad(a, b, top_k=1):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    # Find top-k peaks
    if top_k == 1:
        k_max = np.argmax(b)
        peaks = [k_max]
    else:
        # Use peaks that are within a small factor of the max
        threshold = b.max() * (1 - 1e-6)
        peaks = np.where(b >= threshold)[0].tolist()
        if len(peaks) > top_k:
            peaks = np.argpartition(b, -top_k)[-top_k:].tolist()
    
    g = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je+1] += 2.0 * a[k-je:k-js+1][::-1]
    return g

def log_space_gradient(a, g):
    """Chain rule: gradient in log-space."""
    S = np.dot(g, a)
    return a * (g - S)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_logspace(a0, time_limit, eta0=0.5, beta=0.9, top_k=1):
    """Nesterov-accelerated log-space optimization with exact subgradient."""
    t0 = time.monotonic()
    n = len(a0)
    a0 = a0 / a0.sum()
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.mean()
    v = np.zeros(n)  # velocity in log-space
    best_a = a0.copy()
    best_val = fast_convolve(a0).max()
    no_improve = 0
    cur_eta = eta0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = softmax(z)
        b = fast_convolve(a)
        val = b.max()
        
        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        g = compute_subgrad(a, b, top_k=top_k)
        dz = log_space_gradient(a, g)
        
        # Nesterov update
        v = beta * v - cur_eta * dz
        z = z + v
        z = z - z.mean()  # normalize
        
        # Adaptive step size
        if no_improve > 100:
            cur_eta *= 0.9
            v = v * 0.5  # damp momentum
            no_improve = 0
        elif no_improve > 50:
            cur_eta *= 0.95
            no_improve = 0
        
        # Periodic restart to avoid oscillation
        if t % 5000 == 0 and no_improve > 1000:
            v = np.zeros(n)
            cur_eta = eta0 * 0.5
            no_improve = 0
    
    return best_a, best_val

def frank_wolfe(a0, time_limit, top_k=1):
    """Frank-Wolfe algorithm with exact subgradient and line search."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_convolve(a)
        val = b.max()
        
        if val < best_val - 1e-15:
            best_val = val
            best_a = a.copy()
        
        # Subgradient
        g = compute_subgrad(a, b, top_k=top_k)
        
        # FW step: s = e_{argmin g_i}
        j = np.argmin(g)
        s = np.zeros(n)
        s[j] = 1.0
        
        # Line search: minimize f((1-gamma)*a + gamma*s)
        # Use bisection or Armijo
        best_gamma = 0.0
        best_f = val
        
        for _ in range(20):
            gamma = best_gamma / 2 if best_gamma > 0 else 1.0
            if gamma >= 1.0:
                gamma = 1.0
            a_trial = (1.0 - gamma) * a + gamma * s
            f_trial = fast_convolve(a_trial).max()
            if f_trial < best_f:
                best_f = f_trial
                best_gamma = gamma
            if best_gamma > 0:
                try:
                    gamma2 = min(1.0, best_gamma * 1.5)
                    a_trial2 = (1.0 - gamma2) * a + gamma2 * s
                    f_trial2 = fast_convolve(a_trial2).max()
                    if f_trial2 < best_f:
                        best_f = f_trial2
                        best_gamma = gamma2
                except:
                    pass
        
        if best_gamma < 1e-10:
            break
        
        a = (1.0 - best_gamma) * a + best_gamma * s
        
        # FW gap
        gap = np.dot(g, a - s)
        if gap < 1e-10:
            break
    
    return best_a, best_val

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent for the normalized problem."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            best_j_val = old_val
            best_j_obj = best_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                
                if best_j_obj < best_val - 1e-15:
                    best_val = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_val

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated log-space

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_vec(z):
    """Softmax of a vector (numerically stable)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def compute_subgrad(a, b, rel_tol=1e-10):
    """
    Compute subgradient of max(b) where b = conv(a,a).
    Uses all indices where b_k >= b_max * (1 - rel_tol).
    Returns array g of same length as a.
    """
    n = len(a)
    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n)

    threshold = b_max * (1.0 - rel_tol)
    peak_mask = b >= threshold
    peaks = np.where(peak_mask)[0]

    # Limit number of peaks to avoid excessive computation
    max_peaks = 50
    if len(peaks) > max_peaks:
        peaks = np.argpartition(b, -max_peaks)[-max_peaks:]

    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je + 1] += 2.0 * a[k - je:k - js + 1][::-1]

    return g


def log_space_grad(a, g):
    """Chain rule: gradient of f(softmax(z)) w.r.t. z."""
    S = np.dot(g, a)
    return a * (g - S)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_logspace(a0, time_limit, eta0=0.3, beta=0.9, rel_tol=1e-10):
    """Nesterov-accelerated log-space optimization with exact subgradient."""
    t0 = time.monotonic()
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a0.copy()
    best_val = fast_convolve(a0).max()
    no_improve = 0
    cur_eta = eta0

        b = fast_convolve(a)

def frank_wolfe(a0, time_limit, rel_tol=1e-10):
    """Frank-Wolfe with exact subgradient and backtracking line search."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        val = b.max()

        if val < best_val - 1e-16:
            best_val = val
            best_a = a.copy()

        g = compute_subgrad(a, b, rel_tol=rel_tol)
        j = int(np.argmin(g))

        # FW direction: toward vertex e_j
        s = np.zeros(n)
        s[j] = 1.0

        # Backtracking line search
        gamma = 1.0
        for _ in range(40):
            a_trial = (1.0 - gamma) * a + gamma * s
            f_trial = fast_convolve(a_trial).max()
            if f_trial < val - gamma * np.dot(g, a - s) * 0.5:
                break
            gamma *= 0.5

        if gamma < 1e-12:
            break

        a = (1.0 - gamma) * a + gamma * s

        # Check FW gap
        gap = np.dot(g, a - s)
        if gap < 1e-12 * max(1.0, best_val):
            break

    return best_a, best_val


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent cycling through multiplier steps."""
    t0 = time.monotonic()
    n = len(a0)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0])
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                continue

            best_j_val = old_val
            best_j_obj = best_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-14:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-14:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()

                if best_j_obj < best_val - 1e-16:
                    best_val = best_j_obj
                    best_a = a.copy()
                    improved = True
                    stall = 0
                else:
                    stall += 1

        if not improved:
            if stall > 3:
                break
            stall += 1

    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n=512, score={best_score:.6f}")

    # === Phase 1: Nesterov log-space at parent length ===
    print(f"\n=== Phase 1: Nesterov log-space (n={len(parent)}) ===")
    t1 = min(250.0, (deadline - time.monotonic()) * 0.2)
    if t1 > 20:
        a1, v1 = nesterov_logspace(parent, t1, eta0=0.4, beta=0.9, rel_tol=1e-10)
        best_seq, best_score, _ = try_and_update(a1, best_seq, best_score, "nesterov_parent")

    # === Phase 2: Frank-Wolfe ===
    print(f"\n=== Phase 2: Frank-Wolfe (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 20:
            t2 = min(rem * 0.4, 80.0)
            a2, v2 = frank_wolfe(best_seq, t2, rel_tol=1e-10)
            best_seq, best_score, _ = try_and_update(a2, best_seq, best_score, "fw")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [1024, 1536, 2048, 3072]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)

        # Nesterov at new length
        t3 = min(rem * 0.25, 80.0)
        if t3 < 15:
            continue
        print(f"  Nesterov n={target_n}, t={t3:.0f}s")
        a3, _ = nesterov_logspace(f_interp, t3, eta0=0.3, beta=0.9, rel_tol=1e-10)
        best_seq, best_score, _ = try_and_update(a3, best_seq, best_score, f"nest_n{target_n}")

        # Quick CD on best if time permits
        rem2 = deadline - time.monotonic() - 200
        if rem2 > 10 and best_seq is not None:
            t3b = min(rem2 * 0.2, 25.0)
            a3b, _ = coordinate_descent(best_seq, t3b)
            best_seq, best_score, _ = try_and_update(a3b, best_seq, best_score, f"cd_n{len(best_seq)}")

    # === Phase 4: Deep refinement on best ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Long Nesterov with smaller eta for fine-tuning
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t4 = min(rem * 0.4, 150.0)
            print(f"  Deep Nesterov n={n_best}, eta=0.1, t={t4:.0f}s")
            a4, _ = nesterov_logspace(best_seq, t4, eta0=0.1, beta=0.92, rel_tol=1e-12)
            best_seq, best_score, _ = try_and_update(a4, best_seq, best_score, "deep_nest")

        # Coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t4b = min(rem * 0.4, 50.0)
            a4b, _ = coordinate_descent(best_seq, t4b)
            best_seq, best_score, _ = try_and_update(a4b, best_seq, best_score, "cd_deep")

    # === Phase 5: Multiple restarts ===
    print(f"\n=== Phase 5: Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(15):
            rem = deadline - time.monotonic() - 40
            if rem < 15:
                break

            noise_scale = 0.005 * (1.0 + 0.5 * attempt)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_seq + noise)

            t5 = min(rem * 0.2, 35.0)
            if t5 < 5:
                break
            a5, _ = nesterov_logspace(a_pert, t5, eta0=0.2, beta=0.9, rel_tol=1e-10)
            best_seq, best_score, _ = try_and_update(a5, best_seq, best_score, f"restart_{attempt}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10 and best_seq is not None:
            a6, _ = coordinate_descent(best_seq, rem)
            best_seq, best_score, _ = try_and_update(a6, best_seq, best_score, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
