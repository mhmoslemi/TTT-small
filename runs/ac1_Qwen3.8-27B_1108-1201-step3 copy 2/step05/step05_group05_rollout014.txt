"""
LP-direction-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal descent directions at each step,
combined with line search, multiple initializations, and local polish.

import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ b = cross_convolve(a, b)."""
    n = len(a)
    c_col = np.concatenate([a, np.zeros(n - 1)])
    r_row = np.concatenate([[a[0]], np.zeros(n - 1)])
    return toeplitz(c_col, r_row)


def solve_lp_direction(a, M, upper_bound=10.0):
    """
    Solve: maximize sum(b) subject to b >= 0, C@a @ b <= M, b <= upper_bound.
    Returns the optimal b or None if infeasible.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
        elif res.status == 1:
            # Optimal but no improvement possible
            return None
        else:
            return None
    except Exception:
        return None

def subgrad_direction(a, M):
    """
    Fallback: compute a subgradient-based direction.
    Returns a normalized non-negative direction or None.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Find peak positions
    peak_idx = np.argmax(conv)
    # Build direction: reduce mass at positions contributing to peak
    k = peak_idx
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    # Direction: decrease a at contributing positions, increase elsewhere
    direction = np.zeros(n)
    for i in range(js, je + 1):
        direction[i] -= a[k - i]
    # Make it non-negative by projecting
    direction = np.maximum(direction + direction.max() / 2, 0)
    if direction.sum() < 1e-10:
        return None
    return direction / direction.sum()


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    return a / a.sum()


def cosine_init(n):
    """Smooth cosine-bell initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.4, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent_fast(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 4: Pair transfer search ===
    print(f"\n=== Phase 4: Pair Transfer (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_ps = min(rem * 0.5, 120)
            print(f"  PS t={t_ps:.0f}s")
            a_opt, obj = pair_transfer_search(best_seq, time_limit=t_ps, seed=seed)
            try_seq(a_opt, "pair_transfer")

    # === Phase 5: Refine with LP from perturbed best ===
    print(f"\n=== Phase 5: LP Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            t_alloc = min(rem / (5 - p), 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 0)
                f_init = f_init / f_init.sum()
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif p == 3:
                # Frequency-domain perturbation
                A = np.fft.rfft(f_init)
                phases = rng.uniform(-0.1, 0.1, len(A))
                A[1:] *= np.exp(1j * phases[1:])
                A[1:] = A[1:] * np.maximum(np.abs(A[1:]), 0.5)
                f_init = np.fft.irfft(A, n_best)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                # Blend with uniform
                f_init = 0.85 * best_seq + 0.15 * np.ones(n_best) / n_best
                f_init = f_init / f_init.sum()

            a_opt, obj = lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"lp_refine_{p}")

    # === Phase 6: Try longer sequence ===
    print(f"\n=== Phase 6: Longer Sequence (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for mult in [1.5, 2.0]:
            target_n = int(n_best * mult)
            if target_n <= n_best or target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 40:
                continue

            f_interp = interpolate_sequence(best_seq, target_n)
            # Quick LP run
            t_alloc = min(rem * 0.5, 100)
            print(f"  Longer n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lp_optimize(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"long_n{target_n}")

            rem = deadline - time.monotonic() - 30
            if rem > 20 and best_score < 1.504:
                a_opt2, obj2 = coordinate_descent_fast(a_opt, time_limit=min(rem, 50))
                try_seq(a_opt2, f"cd_long_n{target_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, obj = coordinate_descent_fast(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

        # Quick pair transfer
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = pair_transfer_search(best_seq, time_limit=min(rem, 30), seed=seed + 100)
            try_seq(a_opt, "final_ps")

        # One more LP round on the best
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_opt, obj = lp_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
