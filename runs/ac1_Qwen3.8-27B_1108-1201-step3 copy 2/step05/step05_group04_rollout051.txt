- Frequency-domain designed initializations
- Adam with adaptive soft-max subgradients
- Structured sinusoidal perturbations
- Frequency-domain optimization (optimize DFT, project to non-negative)

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, weights):
    """Compute subgradient of max(conv) using given weights on conv values."""
    n = len(f)

    # conv[k] = sum_i f[i]*f[k-i]
    # d(conv[k])/d(f[j]) = f[k-j] + f[j-(k-j)] = f[k-j] + f[2j-k] (if valid)
    # More precisely: d(conv[k])/d(f[j]) = f[k-j] (if 0<=k-j<n) + f[j-(k-j)]... 
    # Actually: conv[k] = sum_i f[i]*f[k-i]
    # d/d(f[j]) conv[k] = f[k-j] (from the term where i=j) + f[k-j] (from the term where k-i=j, i.e., i=k-j)
    # But these are the same term if k-j = j, i.e., k=2j.
    # So d/d(f[j]) conv[k] = f[k-j] + f[k-j] if k != 2j (two separate terms)
    #                         = f[j] if k = 2j (one term, derivative of f[j]^2 is 2f[j])
    # Wait no. conv[k] = sum_i f[i]*f[k-i]. The terms involving f[j] are:
    # - i=j: f[j]*f[k-j]
    # - k-i=j, i.e., i=k-j: f[k-j]*f[j]
    # These are the same if j = k-j, i.e., k=2j.
    # So d(conv[k])/d(f[j]) = f[k-j] + f[k-j] = 2*f[k-j] if k != 2j and 0<=k-j<n
    #                        = 2*f[j] if k = 2j
    # In both cases: d(conv[k])/d(f[j]) = 2*f[k-j] if 0<=k-j<n, else 0
    # Wait, when k=2j: the term is f[j]*f[j] = f[j]^2, derivative is 2*f[j] = 2*f[k-j]. ✓
    # So in general: d(conv[k])/d(f[j]) = 2*f[k-j] if 0 <= k-j < n, else 0.
    
    # So subgrad[j] = sum_k weights[k] * 2*f[max(0,min(n-1,k-j))] * (1 if 0<=k-j<n else 0)
    # = 2 * sum_{k: 0<=k-j<n} weights[k] * f[k-j]
    # = 2 * (weights * reverse(f))[j] ... hmm, let me think more carefully.
    
    # subgrad[j] = 2 * sum_{k=j}^{j+n-1} weights[k] * f[k-j]  (where weights[k]=0 for k outside [0,2n-2])
    # Let m = k-j, then m goes from 0 to n-1:
    # subgrad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]
    # This is the cross-correlation of weights (shifted) and f.
    
    # More efficiently: subgrad = 2 * correlate(weights, f, 'valid') ... 
    # Actually, correlate(a, b, 'valid') gives sum_i a[i+len(b)-i-1]*b[i] for i=0..len(a)-len(b)
    # That's not quite what we want.
    
    # Let me think of it as: subgrad[j] = 2 * sum_m weights[j+m] * f[m]
    # This is like a "soft" cross-correlation.
    
    # Using FFT: if we define W = weights (length 2n-1) and F = f (length n),
    # then subgrad[j] = 2 * sum_m W[j+m] * F[m] = 2 * (W * F_rev)[j+n-1] ... 
    # Hmm, this is getting complicated. Let me just use np.correlate.
    
    # Actually, the simplest way:
    # subgrad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]
    # This is the "valid" correlation of weights (length 2n-1) and reversed f (length n).
    # np.correlate(weights, f[::-1], 'valid') gives for j=0..n-1: sum_m weights[j+m]*f[n-1-m]
    # That's not quite right either.
    
    # Let me just do it directly with a loop or use a clever FFT trick.
    # For n up to ~2048, a direct computation might be fast enough.
    
    # Actually, the cleanest way:
    # subgrad = 2 * np.array([np.dot(weights[j:j+n], f) for j in range(n)])
    # But this is O(n^2). For n=768, that's ~590K operations per subgradient computation.
    # With FFT, we can do it in O(n log n).
    
    # FFT approach:
    # subgrad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * f[m]
    # Let's define g[m] = f[m] for m=0..n-1, 0 otherwise.
    # And h[k] = weights[k] for k=0..2n-2, 0 otherwise.
    # Then subgrad[j]/2 = sum_m h[j+m]*g[m] = (h * g_flipped)[j+n-1] where g_flipped[m] = g[n-1-m]
    # This is a cross-correlation, computable via FFT.
    
    # Actually, let me just use the correlation formulation:
    # np.correlate(h, g, 'valid') where h has length 2n-1 and g has length n
    # gives array of length n: result[j] = sum_m h[j+m]*g[m] for j=0..n-1
    # Wait, np.correlate definition: c_k = sum_n a_{n+k} * conj(v_n)
    # For 'valid': only returns output where there's no partial overlap
    # If len(a) >= len(v), output length is len(a) - len(v) + 1
    # So np.correlate(weights, f, 'valid') has length (2n-1) - n + 1 = n
    # And result[j] = sum_{m=0}^{n-1} weights[j+m] * f[m] (for real inputs)
    
    # That's exactly what we want!
    return 2.0 * np.correlate(weights, f, 'valid')

def soft_max_weights(conv, temperature, top_k=None):
    """Compute softmax weights over convolution values."""
    shifted = conv / temperature
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(w))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w

def adam_optimize(f0, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_init=0.008, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)

        w = soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * (0.9999 ** t)
        cur_lr = max(cur_lr, lr * 0.1)

        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 100 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj

def freq_domain_optimize(f0, time_limit, n_harmonics=8, verbose=False):
    """
    Optimize in the frequency domain: represent f as a sum of cosine harmonics,
    optimize the amplitudes and phases, then project to non-negative simplex.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Get initial DFT coefficients
    F = np.fft.rfft(f)
    
    # We'll optimize the real and imaginary parts of F[1:k] (keeping F[0] fixed)
    # F[0] is the DC component = sum(f) = 1 (after normalization)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    lr = 0.01
    
    # Parameters: amplitudes and phases for harmonics 1..n_harmonics
    # Represent as complex numbers F[1], F[2], ..., F[n_harmonics]
    # and their conjugates F[n-1], F[n-2], ..., F[n-n_harmonics]
    
    # Actually, let's just directly optimize the DFT coefficients.
    # Keep F[0] fixed (= 1 for normalized f).
    # Optimize F[1..min(n_harmonics, n//2-1)] and their conjugates.
    
    k_max = min(n_harmonics, n // 2 - 1)
    
    # Current DFT
    F_curr = np.fft.rfft(f)
    
    m = np.zeros(2 * k_max)  # for real and imag parts
    v = np.zeros(2 * k_max)
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute gradient w.r.t. DFT coefficients
        conv = fast_convolve(f)
        w = soft_max_weights(conv, 0.005)
        
        # Gradient of max(conv) w.r.t. f is the subgradient we computed before
        sg_f = compute_subgradient(f, w)
        
        # Gradient w.r.t. F: sg_F[k] = sum_j sg_f[j] * exp(-2πijk/n) / n (approximately)
        # Actually, f = IDFT(F), so df/dF[k] = exp(2πijk/n)/n
        # sg_F[k] = sum_j sg_f[j] * conj(df/dF[k]) = sum_j sg_f[j] * exp(-2πijk/n) / n
        # This is just (1/n) * DFT(sg_f)[k]
        
        sg_F = np.fft.rfft(sg_f) / n
        
        # Update F[1..k_max]
        params = np.concatenate([sg_F[1:k_max+1].real, sg_F[1:k_max+1].imag])
        
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * params
        v = beta2 * v + (1 - beta2) * params * params
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        delta_real = -lr * m_hat[:k_max] / (np.sqrt(v_hat[:k_max]) + eps)
        delta_imag = -lr * m_hat[k_max:] / (np.sqrt(v_hat[k_max:]) + eps)
        
        F_new = F_curr.copy()
        F_new[1:k_max+1] += delta_real + 1j * delta_imag
        # Keep F[0] fixed
        F_new[0] = np.abs(F_new[0])  # real and positive
        
        # Convert back to time domain
        f_new = np.fft.irfft(F_new, n)
        
        # Project to non-negative simplex
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum
        
        f = f_new
        F_curr = np.fft.rfft(f)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            if verbose and t % 50 == 0:
                print(f"    Freq iter {t}: obj={best_obj:.10f}")
        
        lr = max(lr * 0.999, 0.001)
    
    return best_f, best_obj

def sinusoidal_perturb(f0, n_harmonics=5, n_trials=20, verbose=False):
    """
    Try adding sinusoidal perturbations to the sequence and keep the best.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(123)
    
    for trial in range(n_trials):
        # Pick a random harmonic and phase
        k = rng.integers(1, min(n_harmonics + 1, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.005, 0.05)
        
        x = np.arange(n)
        perturbation = amplitude * np.cos(2 * np.pi * k * x / n + phase)
        
        f_new = f + perturbation
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            continue
        f_new = f_new / f_new_sum
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
    
    return best_f, best_obj

def coordinate_descent_fast(f0, time_limit, n_trials_per_coord=5, verbose=False):
    """
    Fast coordinate descent: for each coordinate, try several values
    using a quadratic approximation of the objective.
    """
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            # Try several multipliers
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

def make_sinusoidal_init(n, amplitude=0.3, freq=1, phase=0):
    """Create a sinusoidal initialization: 1 + a*cos(2*pi*freq*x + phase)."""
    x = np.arange(n) / n
    f = 1.0 + amplitude * np.cos(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_multi_sinusoidal_init(n, components=None, rng=None):
    """Create a multi-sinusoidal initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    if components is None:
        components = [(1, 0.2, 0.0), (2, 0.1, np.pi/4), (3, 0.05, np.pi/3)]
    
    x = np.arange(n) / n
    f = np.ones(n)
    for freq, amp, phase in components:
        f += amp * np.cos(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_skewed_init(n, skew=0.3):
    """Create a skewed (non-symmetric) initialization."""
    x = np.arange(n) / n
    # Asymmetric bump
    f = np.exp(-((x - 0.3) / 0.2) ** 2) + 0.5 * np.exp(-((x - 0.7) / 0.15) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def make_geometric_init(n, ratio=0.98):
    """Create a geometric progression initialization."""
    f = np.array([ratio ** i for i in range(n)])
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def fast_evaluate(f):
    """Quick evaluation using our own convolution (should match evaluate_sequence)."""
    f = np.maximum(f, 0.0)
    n = len(f)
    conv = fast_convolve(f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))
