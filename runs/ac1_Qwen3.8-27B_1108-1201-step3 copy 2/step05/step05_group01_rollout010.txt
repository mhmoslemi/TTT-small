where `W_f = FFT(w, N)` and `A_f = FFT(a, N)`.

So `grad_a = 2 * IDFT(FFT(w) * conj(FFT(a)))[0:n]`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Diverse parametric initializations with screening

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def compute_smooth_obj_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. a.
    Returns (obj, grad_a) where obj approximates max(conv(a,a))
    and grad_a is the gradient of obj w.r.t. a.
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    # grad_a[i] = 2 * sum_m w[m] * a[m-i] (cross-correlation)
    N = 1 << (3 * n - 2).bit_length()
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(w, N)
    corr = np.fft.irfft(W_f * np.conj(A_f), N)[:n]
    grad_a = 2.0 * corr

    return obj, grad_a


def compute_t_grad(t, beta):
    """
    Compute smoothed objective and gradient w.r.t. t (log-softmax parameterization).
    a = softmax(t)
    """
    a = softmax_fn(t)
    obj, grad_a = compute_smooth_obj_grad(a, beta)
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    return obj, grad_t


def adam_smooth_optimize(f0, time_limit, beta_schedule=None, lr=0.01, verbose=False):
    """
    Adam optimization on the log-sum-exp smoothed objective in softmax parameterization.
    f0: initial point on simplex
    Returns (best_f, best_exact_obj)
    """
    n = len(f0)
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    t = np.log(f + 1e-300)
    # Center t for numerical stability
    t = t - t.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_exact_obj = fast_convolve(f).max()
    best_f = f.copy()
    current_lr = lr

    if beta_schedule is None:
        beta_schedule = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]

    t0 = time.monotonic()
    last_beta_idx = 0
    last_check = 0.0

    while time.monotonic() - t0 < time_limit:
        step += 1

        # Anneal beta
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        target_idx = min(int(progress * len(beta_schedule)), len(beta_schedule) - 1)
        if target_idx > last_beta_idx:
            last_beta_idx = target_idx
        beta = beta_schedule[last_beta_idx]

        # Compute gradient
        obj_smooth, grad_t = compute_t_grad(t, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * (grad_t ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        t = t - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Recover a and check exact objective periodically
        if step % 50 == 0 or (time.monotonic() - last_check > 0.5):
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            exact_obj = fast_convolve(a).max()
            if exact_obj < best_exact_obj - 1e-14:
                best_exact_obj = exact_obj
                best_f = a.copy()
                if verbose and step % 500 == 0:
                    print(f"      Adam step {step}: obj={best_exact_obj:.10f}, beta={beta}")
            last_check = time.monotonic()

        # Anneal learning rate
        current_lr = lr * (1.0 - 0.3 * progress)

    # Final recovery
    a = softmax_fn(t)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    exact_obj = fast_convolve(a).max()
    if exact_obj < best_exact_obj:
        best_exact_obj = exact_obj
        best_f = a.copy()

    return best_f, best_exact_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 80:
                # Try to escape
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.005 + 0.003 * min(no_improve // 80, 5)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.08, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 120:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            scale = 0.003 + 0.003 * min(no_improve // 120, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def equioscillation_optimize(f0, time_limit, verbose=False):
    """
    Optimize by penalizing variance of top convolution values.
    Encourages the convolution to be "flat" at the peak level.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Use iterative approach: find active set, then try to equalize
    for outer in range(20):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        conv_max = conv.max()
        mean_conv = conv.mean()

        # Find active lags (within 5% of max)
        threshold = conv_max - 0.05 * (conv_max - mean_conv)
        active = np.where(conv >= threshold)[0]

        if len(active) < 2:
            break

        # Try to adjust f to equalize active convolution values
        # Use small perturbations weighted by how far each active value is from the mean
        active_vals = conv[active]
        active_mean = active_vals.mean()
        errors = active_vals - active_mean

        # Compute gradient contribution from active set
        grad = np.zeros(n)
        for i, k in enumerate(active):
            weight = errors[i] / (np.abs(errors).sum() + 1e-30)
            # d(conv[k])/d(f[j]) = 2*f[k-j] (if 0 <= k-j < n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * weight * f[k - j]

        # Small step in negative gradient direction (projected)
        step_size = 0.01
        f_new = f - step_size * grad
        f_new = project_simplex(f_new)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            step_size = min(step_size * 1.1, 0.05)
        else:
            step_size *= 0.8
            if step_size < 1e-5:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    from scipy.interpolate import interp1d
    if old_n >= 4:
        interp = interp1d(old_x, f, kind='cubic', fill_value="extrapolate")
        f_interp = interp(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.abs(vals) ** 2

        parent = np.ones(512) / 512
        p_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        best_score = p_score
        print(f"Default parent: n=512, score={p_score:.8f}")

    parent_n = len(parent)

    # === Phase 1: Diverse initialization screening (60s) ===
    print(f"\n=== Phase 1: Diverse Init (best={best_score:.8f}) ===")
    t1 = min(60, deadline - time.monotonic() - 1400)
    if t1 > 10:
        inits = generate_initializations(parent_n, rng)
        top_inits = []
        for i, init in enumerate(inits):
            score = try_seq(init, f"init_{i}")
            top_inits.append((score, init))
        top_inits.sort(key=lambda x: x[0])
        top_inits = top_inits[:5]
        print(f"  Best init score: {top_inits[0][0]:.8f}")
        # Add parent to top inits if good
        if best_score < top_inits[-1][0]:
            top_inits[-1] = (best_score, best_seq)
    else:
        top_inits = [(best_score, best_seq)]

    # === Phase 2: Adam optimization (700s) ===
    print(f"\n=== Phase 2: Adam Opt (best={best_score:.8f}) ===")
    t2_total = min(700, deadline - time.monotonic() - 700)
    n_inits = max(1, len(top_inits))
    t_per_init = max(20, t2_total // n_inits)

    for idx, (init_score, init_f) in enumerate(top_inits):
        remaining = deadline - time.monotonic() - 700
        if remaining < 30:
            break
        t_alloc = min(t_per_init, remaining - 50 * (n_inits - 1 - idx))
        if t_alloc < 20:
            t_alloc = min(remaining, 20)
        print(f"  Adam init {idx}/{n_inits-1}, t={t_alloc:.0f}s, start={init_score:.8f}")

        # Adam from this initialization
        a_opt, obj = adam_smooth_optimize(init_f, time_limit=t_alloc,
                                           beta_schedule=[5, 10, 20, 40, 80, 160, 320, 640],
                                           lr=0.008, verbose=True)
        try_seq(a_opt, f"adam_{idx}")

        # Also try Adam with different lr
        rem = deadline - time.monotonic() - 650
        if rem > 30 and t_alloc > 40:
            a_opt2, _ = adam_smooth_optimize(init_f, time_limit=min(rem, t_alloc // 3),
                                              beta_schedule=[10, 30, 60, 120, 240],
                                              lr=0.02, verbose=False)
            try_seq(a_opt2, f"adam_fast_{idx}")

    # === Phase 3: LP-FW refinement (350s) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        t3 = min(350, deadline - time.monotonic() - 350)
        if t3 > 30:
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw_main")

            # Equioscillation refinement
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                a_opt2, _ = equioscillation_optimize(a_opt if obj < fast_convolve(best_seq).max() else best_seq,
                                                     time_limit=min(rem, t3 // 4))
                try_seq(a_opt2, "equiosc")

    # === Phase 4: Multi-length exploration (350s) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        target_lengths = [512, 640, 1024, 1280, 1536]
        # Filter out lengths too close to current
        target_lengths = [tl for tl in target_lengths if abs(tl - len(base)) > 50]
        target_lengths.sort(key=lambda x: abs(x - len(base)))
        target_lengths = target_lengths[:4]

        t4_per = min(80, (deadline - time.monotonic() - 250) // max(1, len(target_lengths)))
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  Trying n={target_n}, t={min(t4_per, rem):.0f}s")

            # Adam on interpolated
            t_adam = min(t4_per // 2, rem - 20)
            if t_adam > 10:
                a_opt, _ = adam_smooth_optimize(f_interp, time_limit=t_adam,
                                                 beta_schedule=[10, 30, 60, 120, 240, 480],
                                                 lr=0.01, verbose=False)
                try_seq(a_opt, f"len_{target_n}_adam")

            # LP-FW on interpolated
            rem2 = deadline - time.monotonic() - 200
            if rem2 > 20:
                best_len = best_seq if best_seq is not None else a_opt
                if len(a_opt) == target_n:
                    best_len = a_opt
                a_opt2, _ = lp_fw_optimize(best_len if len(best_len) == target_n else f_interp,
                                           time_limit=min(t4_per // 2, rem2 - 10), verbose=False)
                try_seq(a_opt2, f"len_{target_n}_lpfw")

    # === Phase 5: Deep optimization on best (250s) ===
    print(f"\n=== Phase 5: Deep Opt (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # 5a: Adam with very high beta (near-exact)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 80)
            # Start from best, use high beta schedule
            a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_alloc,
                                             beta_schedule=[100, 200, 400, 800, 1600],
                                             lr=0.003, verbose=True)
            try_seq(a_opt, "deep_adam_highbeta")

        # 5b: LP-FW deep
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 100)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_lpfw")

        # 5c: Coordinate descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_cd")

        # 5d: Equioscillation
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = equioscillation_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "deep_equip")

    # === Phase 6: Final polish with restarts (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempt = 0
        while deadline - time.monotonic() > 15:
            attempt += 1
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break

            if attempt % 3 == 1:
                # Small perturbation + Adam
                noise_scale = 0.002 + 0.001 * min(attempt // 3, 5)
                noise = rng.standard_normal(n_best) * noise_scale
                f_start = project_simplex(best_seq + noise)
                t_alloc = min(rem, 40)
                a_opt, _ = adam_smooth_optimize(f_start, time_limit=t_alloc,
                                                 beta_schedule=[50, 100, 200, 400, 800],
                                                 lr=0.005, verbose=False)
                try_seq(a_opt, f"polish_adam_{attempt}")
            elif attempt % 3 == 2:
                # LP-FW short burst
                t_alloc = min(rem, 30)
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
                try_seq(a_opt, f"polish_lpfw_{attempt}")
            else:
                # Coordinate descent
                t_alloc = min(rem, 20)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
