    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(w, b, N):
    """
    Compute gradient of sum_k w_k*(b*b)_k w.r.t. b.
    grad_j = 2 * sum_m w_{j+m} * b_m
    Using FFT-based circular correlation.
    """
    n = len(b)
    W = np.fft.rfft(w, N)
    B = np.fft.rfft(b, N)
    corr = np.fft.irfft(W * np.conj(B), N)
    return 2.0 * corr[:n]


def verify_gradient(b, tau, N):
    """Verify gradient via finite differences (for debugging)."""
    conv = fast_convolve(b)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    w_scaled = tau * w
    w_scaled -= w_scaled.max()
    vw = np.exp(w_scaled)
    vw /= vw.sum()
    
    # Objective: (1/tau) * log(sum exp(tau * conv_k))
    # Gradient of this w.r.t. b_j = sum_k vw_k * 2 * b_{k-j} (valid)
    grad_analytical = compute_conv_gradient(vw, b, N)
    
    # Finite difference check for a few components
    eps = 1e-7
    for j in [0, len(b)//4, len(b)//2, 3*len(b)//4, len(b)-1]:
        b_plus = b.copy()
        b_plus[j] += eps
        conv_p = fast_convolve(b_plus)
        sp = tau * conv_p
        sp -= sp.max()
        fp = np.log(np.exp(sp).sum()) / tau
        
        b_minus = b.copy()
        b_minus[j] -= eps
        conv_m = fast_convolve(b_minus)
        sm = tau * conv_m
        sm -= sm.max()
        fm = np.log(np.exp(sm).sum()) / tau
        
        fd = (fp - fm) / (2 * eps)
        if abs(fd) > 1e-10:
            rel_err = abs(grad_analytical[j] - fd) / (abs(fd) + 1e-15)
            if rel_err > 0.01:
                # print(f"  Grad check j={j}: analytical={grad_analytical[j]:.6e}, FD={fd:.6e}, err={rel_err:.4f}")
                pass
    return grad_analytical


def nesterov_smoothed_gradient(b_init, time_limit, tau_init=50.0, tau_min=1000.0,
                                lr_init=0.5, verbose=False):
    """
    Nesterov-accelerated proximal gradient on the simplex.
    Minimizes max_k (b*b)_k using log-sum-exp smoothing with adaptive temperature.
    """
    b = np.maximum(b_init.astype(np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(len(b_init)) / len(b_init)
    else:
        b = b / s
    n = len(b)
    N = 1 << (2 * n - 1 - 1).bit_length()
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    t_mom = 1.0  # Nesterov momentum parameter
    
    # Nesterov: y is the "lookahead" point
    y = b.copy()
    prev_b = b.copy()
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute smoothed objective and gradient at y
        conv_y = fast_convolve(y)
        scaled = tau * conv_y
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        grad = compute_conv_gradient(w, y, N)
        
        # Backtracking line search
        best_alpha = lr
        best_val = conv_y.max()
        
        # Try several step sizes
        for i in range(20):
            alpha = lr * (0.5 ** i)
            if alpha < 1e-12:
                break
            z = y - alpha * grad
            b_trial = project_simplex(z)
            conv_trial = fast_convolve(b_trial)
            val = conv_trial.max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            else:
                break
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        b_new = project_simplex(y - best_alpha * grad)
        y = b_new + (t_mom / t_new - 1.0) * (b_new - prev_b)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_new.copy()
        else:
            y = y / sy
        
        prev_b = b_new
        b = b_new
        t_mom = t_new
        
        # Check improvement
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_b = b.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    NSGD iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        # Adapt temperature: increase tau (sharper approximation to max)
        tau = min(tau * 1.02, tau_min)
        
        # Adapt learning rate
        if no_improve > 20:
            lr *= 0.8
            if no_improve > 60:
                # Perturb to escape local minimum
                rng = np.random.default_rng(it * 7 + 13)
                b = project_simplex(best_b + rng.standard_normal(n) * 0.001)
                y = b.copy()
                prev_b = b.copy()
                t_mom = 1.0
                lr = lr_init * 0.3
                no_improve = 0
        
        if lr < 1e-10:
            break
    
    return best_b, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses scipy linprog with HiGHS."""
    from scipy.optimize import linprog
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For efficiency, use sparse matrix
    try:
        from scipy import sparse
        rows = []
        cols = []
        vals = []
        for j in range(n):
            if f[j] < 1e-15:
                continue
            k_start = j
            k_end = j + n
            ks = np.arange(k_start, min(k_end, 2*n-1))
            rows.extend(ks.tolist())
            cols.extend([j] * len(ks))
            vals.extend([f[j]] * len(ks))
        A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
            
        g = solve_lp_direction(f, M * 1.0, n)
        if g is None:
            # Try relaxed M
            g = solve_lp_direction(f, M * 1.05, n)
            if g is None:
                no_improve += 5
                if no_improve > 10:
                    break
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 10
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = np.maximum(f + best_alpha * (g - f), 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def hankel_solve(w, n):
    """
    Solve the KKT system: A * x = 1 where A_{j,m} = 2*w_{j+m}.
    This is a Hankel matrix. Use direct solve for moderate n.
    """
    # Build Hankel matrix: A[j,m] = 2 * w[j+m]
    # w has length 2n-1, indices 0 to 2n-2
    # j+m ranges from 0 to 2n-2
    idx_j = np.arange(n)[:, None]
    idx_m = np.arange(n)[None, :]
    idx = idx_j + idx_m
    A = 2.0 * w[idx]
    
    # Solve A * x = ones
    try:
        x = np.linalg.solve(A, np.ones(n))
        return x
    except np.linalg.LinAlgError:
        # Fallback: least squares
        x, _, _, _ = np.linalg.lstsq(A, np.ones(n), rcond=None)
        return x


def kkt_iterate(b_init, time_limit, tau_init=10.0, verbose=False):
    """
    Iterative KKT-based method:
    1. Compute weights from smoothed max of convolution
    2. Solve linear system for the "ideal" b
    3. Project to simplex
    4. Iterate
    """
    b = np.maximum(b_init.astype(np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(len(b_init)) / len(b_init)
    else:
        b = b / s
    n = len(b)
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        conv = fast_convolve(b)
        scaled = tau * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Try to solve the KKT linear system
        x = hankel_solve(w, n)
        
        # Project to simplex
        b_new = project_simplex(x)
        
        # Blend with current to ensure progress
        b_trial = 0.5 * b + 0.5 * b_new
        b_trial = project_simplex(b_trial)
        
        obj_trial = fast_convolve(b_trial).max()
        
        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_b = b_trial.copy()
            b = b_trial
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    KKT iter {it}: obj={best_obj:.10f}")
        else:
            # Try pure step
            b_trial2 = b_new
            obj2 = fast_convolve(b_trial2).max()
            if obj2 < best_obj - 1e-15:
                best_obj = obj2
                best_b = b_trial2.copy()
                b = b_trial2
                no_improve = 0
            else:
                no_improve += 1
                # Gradually increase tau
                tau *= 1.05
        
        if no_improve > 50:
            # Perturb
            rng = np.random.default_rng(it * 3 + 1)
            b = project_simplex(best_b + rng.standard_normal(n) * 0.002)
            tau = tau_init
            no_improve = 0
    
    return best_b, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_bell(n):
    """Cosine bell: (1+cos(2*pi*x/n))/2 shape."""
    x = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tukey(n, r=0.3):
    """Tukey (tapered cosine) window."""
    n_half = n // 2
    x = np.zeros(n)
    # Left taper
    left = int(n * r / 2)
    right = n - left
    if left > 0:
        for i in range(left):
            x[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / left - 1)))
        x[left:right] = 1.0
        for i in range(left):
            x[n - 1 - i] = x[i]
    else:
        x[:] = 1.0
    a = np.maximum(x, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10 and len(best_seq) <= 768:
            t_alloc = min(rem * 0.9, 30)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_lpfw")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
