"""
Minimizes 2*n*max(convolve(f,f))/(sum(f))^2 using active-set LP Frank-Wolfe
with aggressive length exploration, fast line search, coordinate descent,
and subgradient polishing. Key improvements: active-set LP for speed,
better length exploration, precise 1D optimization, and multi-phase refinement.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_active_lp(f, M, n, K=60):
    """
    Solve LP with only top-K active constraints for speed.
    max sum(g) s.t. (f*g)_k <= M for top-K peaks, g >= 0.
    """
    conv = fast_convolve(f)
    # Get top K constraint indices
    k_actual = min(K, 2 * n - 1)
    top_k = np.argpartition(conv, -k_actual)[-k_actual:]
    top_k = top_k[np.argsort(conv[top_k])[::-1]]  # sort by decreasing value

    # Only keep constraints that are close to M (within 5%)
    threshold = M * 0.95
    active = top_k[conv[top_k] >= threshold]
    if len(active) < 5:
        active = top_k[:min(20, k_actual)]

    # Build constraint matrix for active constraints only
    m = len(active)
    A_ub = np.zeros((m, n))
    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]

    b_ub = np.full(m, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def solve_full_lp(f, M, n):
    """Solve the full LP with all constraints."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': 45, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def line_search_exact(f, g, M, n, iters=40):
    """
    Find best alpha in [0,1] minimizing max(conv(f+alpha*(g-f), f+alpha*(g-f))).
    Uses golden-section search on the 1D non-convex function.
    """
    def obj_at_alpha(alpha):
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return M * 2
        ft = ft / st
        return fast_convolve(ft).max()

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at_alpha(c_pt)
    fd = obj_at_alpha(d_pt)

    for _ in range(iters):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at_alpha(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_alpha(d_pt)

    best_alpha = (lo + hi) / 2.0
    best_val = obj_at_alpha(best_alpha)

    # Also check alpha=0 and alpha=1
    v0 = M
    v1 = obj_at_alpha(1.0)
    if v0 < best_val:
        best_alpha = 0.0
        best_val = v0
    if v1 < best_val:
        best_alpha = 1.0
        best_val = v1

    return best_alpha, best_val

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplicative updates and bisection refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Try several multipliers
            best_j_val = old_val
            best_j_obj = cur_max

            for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.7, 2.5]:
                new_val = old_val * mult
                if new_val <= 0 or new_val >= 0.9999:
                    continue
                scale = denom / (1.0 - new_val)
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = denom / (1.0 - best_j_val)
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 6:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=80, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0002)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True, max_K=80)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: Explore different lengths ===
    print(f"\n=== Phase 2: Length exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)

    # Try a range of lengths
    length_candidates = []
    for delta in [-192, -128, -96, -64, -48, -32, -16, 16, 32, 48, 64, 96, 128, 192, 256, 384]:
        nl = n_best + delta
        if 64 <= nl <= 3000 and nl != n_best:
            length_candidates.append(nl)

    # Also try some "round" lengths
    for nl in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1152, 1280, 1536, 2048]:
        if nl != n_best and 64 <= nl <= 3000 and nl not in length_candidates:
            length_candidates.append(nl)

    length_candidates = sorted(set(length_candidates))

    for target_n in length_candidates:
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.15)
        if t_alloc < 12:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, max_K=60)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")

    # === Phase 3: Deep LP-FW with diverse initializations ===
    print(f"\n=== Phase 3: Deep optimization (best={best_score:.6f}, n={len(best_seq)}) ===")
    n_best = len(best_seq)

    inits = []
    inits.append(("direct", best_seq.copy()))

    # Small noise
    noise = rng.standard_normal(n_best) * 0.003
    inits.append(("noise3k", project_simplex(best_seq + noise)))

    # Medium noise
    noise = rng.standard_normal(n_best) * 0.008
    inits.append(("noise8k", project_simplex(best_seq + noise)))

    # Reverse
    f_rev = 0.5 * best_seq + 0.5 * best_seq[::-1]
    inits.append(("reverse", project_simplex(f_rev)))

    # Mix with uniform
    f_mix = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
    inits.append(("mix_uniform", project_simplex(f_mix)))

    # Concentrated center
    center = np.zeros(n_best)
    center_center = n_best // 2
    width = max(5, n_best // 8)
    center[center_center - width:center_center + width] = 1.0
    f_conc = 0.7 * best_seq + 0.3 * (center / center.sum())
    inits.append(("concentrate", project_simplex(f_conc)))

    # Edge emphasis
    f_edge = np.zeros(n_best)
    f_edge[:n_best//4] = 0.5
    f_edge[-n_best//4:] = 0.5
    f_edge = f_edge / f_edge.sum()
    f_edge_mix = 0.7 * best_seq + 0.3 * f_edge
    inits.append(("edge_mix", project_simplex(f_edge_mix)))

    for label, f_init in inits:
        rem = deadline - time.monotonic() - 150
        if rem < 20:
            break
        t_alloc = min(120, rem * 0.22)
        if t_alloc < 12:
            continue
        print(f"  {label}: t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, max_K=70)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        t_cd = min(rem * 0.4, 180)
        print(f"  t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [CD]: {sc:.6f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_sg = min(rem * 0.5, 100)
        a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                             lr=0.002, mom=0.94,
                             top_k=max(30, n_best // 5),
                             temp=0.00003)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [SG]: {sc:.6f}")

    # === Phase 6: Another LP-FW round on the polished result ===
    print(f"\n=== Phase 6: Final LP-FW (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 40
    if rem > 20:
        t_final = min(rem * 0.6, 120)
        print(f"  t={t_final:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, max_K=90)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [Final LP]: {sc:.6f}")

    # === Phase 7: Micro-search with many tiny perturbations ===
    print(f"\n=== Phase 7: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(20):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            # Quick SG polish
            rem2 = deadline - time.monotonic()
            if rem2 > 8:
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem2 * 0.3, 15),
                                     lr=0.0004, mom=0.95,
                                     top_k=max(15, n_best // 4),
                                     temp=0.00001)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sg]: {sc2:.6f}")

    # === Final verification ===
    if best_seq is None:
        best_seq = np.ones(384) / 384

    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = np.ones(384) / 384
        fs = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {fs:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
