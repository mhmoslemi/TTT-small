def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""

    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'time_limit': time_limit, 'presolve': True}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success and res.x is not None:
        return res.x
    return None

    This provides richer gradient signal than max-only methods.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    lr = 0.0005
    momentum = 0.9
    v = np.zeros(n)
    top_k = 50

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()

        # Blend max-gradient and flat-target gradient
        alpha_blend = min(1.0, it / 200.0)
        sg_max = compute_sg_max(f, conv, temperature=max(0.001, 0.01 / (1 + it * 0.01)),
                                top_k=top_k)
        sg_flat = compute_sg_flat(f, conv, top_k=top_k)
        sg = (1 - alpha_blend) * sg_max + alpha_blend * sg_flat

        # Normalize gradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        f = f_new

        conv = fast_convolve(f)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Flat-target iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.9
            if lr < 1e-6:
                # Restart with noise
                rng = np.random.default_rng(it * 271 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = max(0.0003, 0.0005)
                no_improve = 0
        elif no_improve < 5:
            lr *= 1.01

        top_k = min(80, 30 + it // 20)

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=40, temp0=0.0001):
    """Subgradient polishing with geometric temperature cooling."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        sg = compute_sg_max(f, conv, temperature=cur_temp, top_k=top_k)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_temp *= 0.998
        cur_lr *= 0.9997
        if no_improve > elli := 300:
            rng = np.random.default_rng(it * 41 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temp0 * 0.5
            no_improve = 0
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with adaptive multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_conv = fast_convolve(f)
        cur_max = cur_conv.max()

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = cur_max
            multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved and passes > 3:
            break
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, seed):
    """Create diverse initialization using spectral methods."""
    rng = np.random.default_rng(seed)
    # Random PSD -> autoconvolution is positive definite
    freqs = np.fft.rfftfreq(n)
    # Create a smooth random spectrum
    coeffs = rng.standard_normal(len(freqs)) * np.exp(-5 * freqs**2 * n)
    coeffs += rng.standard_normal(len(freqs)) * np.exp(-2 * freqs**2 * n) * 0.5
    spec = coeffs ** 2 + 0.1
    # Inverse FFT to get a sequence
    f = np.fft.irfft(spec * n, n)
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-10:
        f /= ss
    else:
        f = np.ones(n) / n
    return f


def make_skewed_beta(n, alpha=1.5, beta=3.0, seed=42):
    """Create a skewed Beta distribution as initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.01, 0.99, n)
    f = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    f += rng.standard_normal(n) * 0.01 * f.max()
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def squeezed_optimize(f0, time_limit):
    """Aggressive local optimization with very tight LP steps + CD alternation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        M = fast_convolve(f).max()

        if it % 5 == 0:
            # CD step
            f_cd, obj_cd = coordinate_descent(f, time_limit=min(3, rem * 0.2))
            if obj_cd < best_obj - 1e-14:
                best_obj = obj_cd
                best_f = f_cd.copy()
                f = f_cd
            continue

        # Tight LP
        g = solve_lp_direction(f, M * 1.001, n, time_limit=min(10, rem * 0.3))
        if g is None:
            g = solve_lp_direction(f, M * 1.02, n, time_limit=min(10, rem * 0.3))
        if g is None:
            g = solve_lp_direction(f, M * 1.05, n, time_limit=min(15, rem * 0.3))
        if g is None:
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            continue

        for alpha in [0.03, 0.07, 0.15, 0.3]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < M - 1e-14:
                f = ft
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining flat-target gradients, LP-FW,
    adaptive subgradient, multi-length search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Quick LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent n={n_parent} ===")
    rem = deadline - time.monotonic()
    t1 = min(150, rem * 0.12)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 2: Flat-target gradient (NEW METHOD) ===
    print(f"\n=== Phase 2: Flat-target from parent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.15)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        # Run from parent AND from LP-FW result
        a_ft1, _ = flat_target_optimize(parent, time_limit=t2 * 0.5, verbose=True)
        sc1 = float(evaluate_sequence(a_ft1.tolist()))
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_seq = a_ft1.copy()
            print(f"  NEW BEST (flat from parent): {sc1:.6f}")

        a_ft2, _ = flat_target_optimize(best_seq, time_limit=t2 * 0.5, verbose=True)
        sc2 = float(evaluate_sequence(a_ft2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = a_ft2.copy()
            print(f"  NEW BEST (flat from best): {sc2:.6f}")

    # === Phase 3: Deep LP-FW on best ===
    print(f"\n=== Phase 3: Deep LP-FW (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    t3 = min(350, rem * 0.25)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        # Multiple inits
        inits = []
        inits.append(("direct", best_seq.copy()))
        noise = rng.standard_normal(n_best) * 0.003
        inits.append(("noise_small", project_simplex(best_seq + noise)))
        f_rev = 0.5 * best_seq + 0.5 * best_seq[::-1]
        inits.append(("symmetrized", project_simplex(f_rev)))
        f_shift = np.roll(best_seq, n_best // 64)
        inits.append(("shift", project_simplex(f_shift)))

        for label, f_init in inits:
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(150, rem * 0.3)
            if t_alloc < 10:
                continue
            print(f"  [{label}] t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(label == "direct"))
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [{label}]: {sc:.6f}")

    # === Phase 4: SG polish with adaptive temperature ===
    print(f"\n=== Phase 4: SG Polish (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_sg = min(180, rem * 0.4)
        print(f"  Budget: {t_sg:.0f}s")
        a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                             lr=0.0015, mom=0.93,
                             top_k=max(20, n_best // 8),

        sc = float(evaluate_sequence(a_opt.tolist()))

            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (SG): {sc:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    candidate_lengths = set()
    for delta in [-128, -64, -32, 32, 64, 128, 256]:
        nl = n_best + delta
        if 256 <= nl <= 2048 and nl != n_best:
            candidate_lengths.add(nl)

    for target_n in sorted(candidate_lengths, key=lambda x: abs(x - n_best)):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(80, rem * 0.25)
        if t_alloc < 10:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        # Quick LP-FW
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (n={target_n}): {sc:.6f}")
        else:
            # Also try flat-target
            a_ft, _ = flat_target_optimize(f_interp, time_limit=min(30, t_alloc * 0.4))
            sc2 = float(evaluate_sequence(a_ft.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_ft.copy()
                print(f"  NEW BEST (n={target_n}, flat): {sc2:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: CD (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 120
    if rem > 20:
        t_cd = min(120, rem * 0.5)
        print(f"  Budget: {t_cd:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (CD): {sc:.6f}")

    # === Phase 7: Squeeze (alternating LP + CD) ===
    print(f"\n=== Phase 7: Squeeze (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_sq = min(150, rem * 0.7)
        print(f"  Budget: {t_sq:.0f}s")
        a_opt, _ = squeezed_optimize(best_seq, time_limit=t_sq)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST (squeeze): {sc:.6f}")

    # === Phase 8: Final micro-search ===
    print(f"\n=== Phase 8: Micro-search (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for trial in range(12):
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        scale = 0.0002 + trial * 0.00015
        noise = rng.standard_normal(n_best) * scale
        f_pert = project_simplex(best_seq + noise)
        sc = float(evaluate_sequence(f_pert.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f_pert.copy()
            print(f"  NEW BEST [pert{trial}]: {sc:.6f}")
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem2 * 0.3, 8),
                                     lr=0.0004, mom=0.95,
                                     top_k=max(10, n_best // 6),
                                     temp0=0.00003)
                sc2 = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [pert{trial}_sg]: {sc2:.6f}")

    # === Phase 9: Final LP-FW squeeze ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 9: Final LP-FW (rem={rem:.0f}s) ===")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.8, 60))
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  FINAL: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
