def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):

    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / s
    
    obj = conv_max + np.log(max(s, 1e-300)) / beta
    
    # grad_a[i] = 2 * sum_m a[m] * w[m+i]
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)
    
    return obj, grad_t

    eps = 1e-15

        try:
            from scipy.optimize import minimize
            maxiter = min(500, max(50, int(sub_time * 5)))
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 50,
                         'maxfun': maxiter * 5}
            )
            t = np.clip(t, -50, 50)
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:

def peak_focused_optimize(f0, time_limit, verbose=False):
    """
    Peak-focused optimization: identify top peaks in convolution,
    compute gradient contribution from each peak, and update.
    Uses adaptive temperature and momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    mom = 0.9
    v = np.zeros(n)
    temp = 0.001
    no_improve = 0
    top_k = max(10, n // 8)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)

def adaptive_mwu(f0, time_limit, verbose=False):
    """
    Adaptive MWU with multiple step sizes and gradient clipping.
    Uses the smooth gradient but with adaptive learning rate based on curvature.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = np.log(f + 1e-15)
    v = np.zeros(n)  # momentum for t-space
    beta = 50.0
    lr = 0.01
    beta_mom = 0.99
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()
        
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        s_exp = exp_shifted.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_shifted / s_exp
        
        # Gradient in a-space
        N = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(a, N)
        W = np.fft.rfft(w, N)
        R = np.fft.irfft(W * np.conj(A), N)[:n]
        grad_a = 2.0 * R
        
        # Convert to t-space
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad_t)
        if grad_norm > 1e-30:
            # Normalize and scale
            grad_t_scaled = grad_t / grad_norm
        else:
            grad_t_scaled = grad_t
        
        v = 0.95 * v - lr * grad_t_scaled
        t = t + v
        t = np.clip(t, -50, 50)
        
        a_new = softmax_fn(t)
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = a_new.copy()
            no_improve = 0
            f = a_new
        else:
            no_improve += 1
            lr *= 0.98
            if no_improve > 150:

        beta = beta * 1.001 + 0.1

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def block_coordinate_descent(f0, time_limit):
    """
    Block coordinate descent: partition into blocks, optimize each block
    while fixing others. Uses LP for each block optimization.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Partition into blocks of size ~32
    block_size = max(8, min(64, n // 8))
    n_blocks = (n + block_size - 1) // block_size
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < n_blocks * 3:
        improved = False
        for bi in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            
            start = bi * block_size
            end = min(n, start + block_size)
            bsize = end - start
            
            if bsize <= 1:
                continue
            
            # For this block, try to optimize by LP
            # Fix other coordinates, find best values for block
            # (f*g)(k) = sum_j f_j * g_{k-j}
            # For block coordinates, this is complex. Instead, use local search.
            
            # Try random perturbations of the block
            rng = np.random.default_rng(bi * 1000 + int(time.monotonic() * 100) % 10000)
            best_block_obj = best_obj
            
            for trial in range(5):
                f_trial = f.copy()
                delta = rng.standard_normal(bsize) * 0.02
                f_trial[start:end] = np.maximum(f_trial[start:end] + delta, 0)
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_block_obj - 1e-14:
                    best_block_obj = obj
                    f = f_trial
            
            if best_block_obj < best_obj - 1e-14:
                best_obj = best_block_obj
                best_f = f.copy()
                improved = True
        
        if not improved:
            no_improve += block_size  # count as multiple iterations
        else:
            no_improve = 0
    
    return best_f, best_obj

4. L-BFGS with annealed log-sum-exp smoothing
5. Adaptive momentum method in softmax parameterization
6. Multi-length exploration with interpolation
7. Aggressive restart and perturbation strategy
Key improvements over previous: sparse LP for speed, structured perturbations,
adaptive temperature in peak-focused method, and more aggressive multi-M exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s


def try_evaluate(a, state):
    """Evaluate sequence a and update state [best_score, best_seq]."""
    a = np.maximum(a, 0.0)
    a_list = [float(x) for x in a]
    score = float(evaluate_sequence(a_list))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a.copy()
        print(f"  NEW BEST: {score:.6f} (n={len(a)})")
    return score


def gen_initializations(n, seed=42):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    rng = np.random.default_rng(seed)

    # Uniform
    inits.append(np.ones(n) / n)

    # Power-law with various exponents
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** (-alpha)
        inits.append(a / a.sum())

    # Cosine window
    x = np.linspace(0, np.pi / 2, n)
    a = np.cos(x)
    inits.append(a / a.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    a = np.exp(-x ** 2 / 2)
    inits.append(a / a.sum())

    # Chaotic (logistic map)
    x = 0.5
    a = np.zeros(n)
    for i in range(n):
        x = 3.99 * x * (1 - x)
        a[i] = x
    inits.append(a / a.sum())

    # Random exponential
    a = rng.exponential(1, n)
    inits.append(a / a.sum())

    # Two-peak
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    inits.append(a / a.sum())

    # Sine squared
    x = np.linspace(0, 1, n)
    a = np.abs(np.sin(2 * np.pi * x)) ** 2
    inits.append(a / a.sum())

    # Triangle
    x = np.linspace(-1, 1, n)
    a = np.maximum(0, 1 - np.abs(x))
    inits.append(a / a.sum())

    # Bimodal power
    x = np.arange(n, dtype=np.float64)
    mid = n / 2
    a = np.maximum(0.01, 1.0 / (1.0 + np.abs(x - mid) ** 1.5))
    inits.append(a / a.sum())

    # Random with heavy tail
    a = rng.pareto(1.5, n)
    inits.append(a / a.sum())

    return inits


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, using sparse matrix."""
    c = -np.ones(n)
    A_ub = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub = A_ub.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        return None
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, seed=0):
    """LP-based Frank-Wolfe with multi-M exploration and backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(seed + 999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Try multiple M values for direction finding
        best_direction = None
        best_dir_obj = best_obj

        m_candidates = [M * m_exp, M * 1.3, M * 0.85, M * 1.0]
        seen = set()
        for m_val in m_candidates:
            key = round(m_val, 10)
            if key in seen or m_val <= 0:
                continue
            seen.add(key)
            g = solve_lp_direction(f, m_val, n)
            if g is None:
                continue
            g_sum = g.sum()
            if g_sum < 1e-15:
                continue
            g = g / g_sum

            # Quick check with alpha=0.5
            ft = 0.5 * f + 0.5 * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_dir_obj - 1e-15:
                best_dir_obj = val
                best_direction = g

        if best_direction is None:
            no_improve += 1
            if no_improve > 40:
                # Structured perturbation
                scale = 0.005 + 0.004 * min(no_improve // 40, 5)
                # Mix Gaussian and structured noise
                noise = rng.standard_normal(n) * scale
                # Add a small "ripple"
                ripple = np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n) * scale * 0.5
                f = project_simplex(best_f + noise + ripple)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.08, 3.0)
            continue

        g = best_direction
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 2.0)
            if no_improve > 80:
                scale = 0.008
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(25):
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha > 1e-10:
            f_new = (1 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            new_conv = fast_convolve(f_new)
            M = new_conv.max()
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                m_exp = max(m_exp * 0.97, 0.7)
                if verbose and it % 200 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 3.0)

        if no_improve > 120:
            scale = 0.003 + 0.003 * min(no_improve // 120, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)

    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / s

    obj = conv_max + np.log(max(s, 1e-300)) / beta

    # grad_a[i] = 2 * sum_m a[m] * w[m+i] (cross-correlation)
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R

    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS with annealed beta on smooth approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    eps = 1e-15
    t = np.log(f + eps)

    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 15:
            break
        sub_time = remaining / (len(betas) - bi)

        try:
            maxiter = min(400, max(30, int(sub_time * 4)))
            res = minimize(
                _smooth_obj_grad, t, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 40,
                         'maxfun': maxiter * 4}
            )
            t = np.clip(t, -50, 50)
            a = softmax_fn(t)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a = a / a_sum
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = a.copy()
                if verbose:
                    print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")
        except Exception:
            break

    return best_f, best_obj


def peak_focused_optimize(f0, time_limit, verbose=False):
    """
    Peak-focused optimization: identify top peaks, compute weighted gradient,
    update with momentum and adaptive step size.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.008
    mom = 0.92
    v = np.zeros(n)
    temp = 0.0008
    no_improve = 0
    top_k = max(16, n // 6)
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Normalize gradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-30:
            sg = sg / sg_norm

        v = mom * v - lr * sg
        f_new = project_simplex(f + v)

        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and best_obj < 1.51:
                print(f"    PeakFocus: obj={best_obj:.10f}")
        else:
            no_improve += 1
            lr *= 0.985
            temp *= 1.005

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            lr = max(0.004, min(lr, 0.01))
            no_improve = 0

        lr = max(lr, 0.0005)
        temp = min(temp, 0.1)

    return best_f, best_obj


def adaptive_momentum_optimize(f0, time_limit, verbose=False):
    """
    Adaptive momentum in softmax parameterization.
    Uses smooth gradient with adaptive learning rate based on gradient norm.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    t = np.log(f + 1e-15)
    v = np.zeros(n)
    beta = 30.0
    lr = 0.005
    beta_adapt = 1.002
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        a = softmax_fn(t)
        conv = fast_convolve(a)
        conv_max = conv.max()

        shifted = beta * (conv - conv_max)
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        s_exp = exp_shifted.sum()
        if s_exp < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_shifted / s_exp

        # Gradient in a-space
        N = 1 << (3 * n - 1).bit_length()
        A_ft = np.fft.rfft(a, N)
        W_ft = np.fft.rfft(w, N)
        R = np.fft.irfft(W_ft * np.conj(A_ft), N)[:n]
        grad_a = 2.0 * R

        # Convert to t-space
        dot = np.dot(a, grad_a)
        grad_t = a * (grad_a - dot)

        # Adaptive step
        gn = np.linalg.norm(grad_t)
        if gn > 1e-30:
            grad_t = grad_t / gn

        v = 0.93 * v - lr * grad_t
        t = t + v
        t = np.clip(t, -50, 50)

        a_new = softmax_fn(t)
        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = a_new.copy()
            no_improve = 0
            f = a_new
            if verbose and best_obj < 1.51:
                print(f"    AdaptiveMom: obj={best_obj:.10f}")
        else:
            no_improve += 1
            lr *= 0.985
            if no_improve > 150:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.006)
                t = np.log(f + 1e-15)
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0

        lr = max(lr, 0.0003)
        beta = min(beta * beta_adapt, 200.0)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def two_stage_lp_fw(f0, time_limit):
    """
    Two-stage LP-FW: First use relaxed M to find a good direction,
    then tighten M iteratively.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    # Stage 1: Relaxed M (1.5x) for 30% of time
    # Stage 2: Tight M (1.0x) for remaining

    t_split = (time.monotonic() - t0) + time_limit * 0.3

    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        if time.monotonic() < t_split:
            m_factor = 1.4
        else:
            m_factor = 1.0 + 0.1 * max(0, no_improve / 50)

        M_use = M * m_factor
        g = solve_lp_direction(f, M_use, n)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = (1 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining:
    1. Multi-start with structured initializations
    2. Sparse LP-Frank-Wolfe (primary)
    3. Two-stage LP-FW
    4. Peak-focused subgradient
    5. L-BFGS with annealed smoothing
    6. Adaptive momentum
    7. Multi-length exploration
    8. Final polishing
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    state = [float('inf'), None]  # [best_score, best_seq]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 0: Multi-start (120s) ===
    print("\n=== Phase 0: Multi-start ===")
    t0_phase = time.monotonic()
    inits = gen_initializations(len(parent), seed=seed)
    inits.insert(0, parent.copy())

    n_inits = len(inits)
    t_per_init = min(15, 120.0 / n_inits)

    for i, init in enumerate(inits):
        rem = deadline - time.monotonic() - (n_inits - i) * 3
        if rem < 5:
            break
        t_alloc = min(t_per_init, rem)
        if t_alloc < 5:
            try_evaluate(init, state)
            continue
        a_opt, _ = lp_fw_optimize(init, time_limit=t_alloc, seed=seed + i)
        try_evaluate(a_opt, state)

    # === Phase 1: Main LP-FW on best (400s) ===
    print(f"\n=== Phase 1: Main LP-FW (best={state[0]:.6f}) ===")
    base = state[1] if state[1] is not None else parent
    t1 = min(400, deadline - time.monotonic() - 1000)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(base, time_limit=t1, verbose=True, seed=seed + 100)
        try_evaluate(a_opt, state)

    # Reversed
    t1r = min(80, deadline - time.monotonic() - 900)
    if t1r > 20:
        a_rev = base[::-1].copy()
        a_opt, _ = lp_fw_optimize(a_rev, time_limit=t1r, seed=seed + 200)
        try_evaluate(a_opt, state)

    # === Phase 2: Two-stage LP-FW (150s) ===
    print(f"\n=== Phase 2: Two-stage LP-FW (best={state[0]:.6f}) ===")
    base = state[1] if state[1] is not None else parent
    t2 = min(150, deadline - time.monotonic() - 800)
    if t2 > 20:
        a_opt, _ = two_stage_lp_fw(base, time_limit=t2)
        try_evaluate(a_opt, state)

    # === Phase 3: Peak-focused (150s) ===
    print(f"\n=== Phase 3: Peak-focused (best={state[0]:.6f}) ===")
    base = state[1] if state[1] is not None else parent
