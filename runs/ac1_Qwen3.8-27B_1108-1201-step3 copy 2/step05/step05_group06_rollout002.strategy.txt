def smooth_optimize(initial_seq, time_limit=600, **kwargs):
    """
    Perform gradient-based optimization with Adam using smoothed convolution metrics.
    """
    n = len(initial_seq)
    seq = initial_seq.copy() / np.sum(initial_seq)  # unit-norm
    
    # Adam with variant for sequence-level optimization
    from scipy.optimize import minimize
    
    def cost_function(x):
        conv = np.convolve(x, x, mode='full')
        max_b = np.max(conv)
        sum_x = np.sum(x)
        if sum_x < 0.01:
            return np.inf
        return (2 * n * max_b) / (sum_x ** 2)
    
    # Set up bounds and constraints
    bounds = [(0, 1000) for i in range(n)]
    # Convolve optimization with Adam-style step
    scl = Adam(
        fun=cost_function,
        x0=seq,
        method='L-BFGS-B',
        bounds=bounds,
        maxfev=1000,
        tol=1e-8
    )
    return scl.x
