"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of LP-based
direction finding (linearizing the cross-correlation constraint), 
soft-max smoothed gradient optimization in log-space, and coordinate
descent polishing. The LP step provides provably good descent directions
by solving a linear program that maximizes the new sequence's sum while
keeping the cross-correlation with the current sequence bounded.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_fft(w, a):
    """
    Compute cross-correlation (w ⊛ a)[i] = sum_j w[j+i]*a[j] for i = 0..len(a)-1.
    w has length 2n-1, a has length n.
    """
    n_w = len(w)
    n_a = len(a)
    size = n_w + n_a - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(W) * A, fft_size)
    return result[:n_a]

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_cross_corr_matrix(f, n):
    """Build matrix A (2n-1 x n) where (A @ g)[i] = sum_k f[i-k]*g[k] (valid)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_solve_direction(f, M, n, epsilon=1e-4):
    """
    Solve LP: max sum(g) + epsilon * dot(f, g)
    s.t. A @ g <= M * 1, g >= 0
    """
    A = build_cross_corr_matrix(f, n)
    c = -(np.ones(n) + epsilon * f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None

def soft_max_obj(z, T, n):
    """Soft-max smoothed objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp

def soft_max_grad(z, T, n):
    """Gradient of soft-max objective w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a = 2n * cross_corr(w, a)
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    
    # Chain rule through softmax
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return grad_z

def soft_max_obj_grad(z, T, n):
    """Combined objective and gradient computation."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z

def lp_refine(f0, n, time_limit=300.0, max_iters=500):
    """
    LP-based iterative refinement.
    f0: initial sequence (will be normalized)
    Returns best sequence found.
    """
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    base_t = 0.1
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        g = lp_solve_direction(f, M, n)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Line search over step sizes
        best_trial_obj = best_obj
        best_trial_f = None
        
        for t in [base_t, base_t * 0.5, base_t * 2, base_t * 0.25, base_t * 4]:
            if t <= 0 or t > 1:
                continue
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            ts = f_trial.sum()
            if ts < 1e-15:
                continue
            f_trial = f_trial / ts
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj - 1e-14:
                best_trial_obj = obj
                best_trial_f = f_trial.copy()
        
        if best_trial_f is not None:
            f = best_trial_f
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            base_t *= 0.5
            if base_t < 1e-4 or no_improve > 30:
                # Try perturbing
                rng = np.random.default_rng(it)
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                no_improve = 0
                base_t = 0.1
    
    return best_f, best_obj

def softmax_lbgfs(z0, T, n, time_limit=60.0, maxiter=5000):
    """L-BFGS optimization with soft-max objective."""
    t_start = time.monotonic()
    
    def func(z):
        if time.monotonic() - t_start > time_limit:
            return 1e15
        return soft_max_obj(z, T, n)
    
    def jac(z):
        if time.monotonic() - t_start > time_limit:
            return np.zeros_like(z)
        return soft_max_grad(z, T, n)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        return res.x
    except Exception:
        return z0

def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent on the simplex (cycling through multipliers)."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.18, 1.4, 1.7])
    
    rounds = 0
    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting it to a small value
                candidates = [1e-4, 1e-3, 0.01]
            else:
                candidates = [old_val * m for m in multipliers]
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    scale = 0
                else:
                    scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 10:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True
        
        if not improved and rounds > 3:
            break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(linearizing cross-correlation constraints), soft-max smoothed gradient
optimization via L-BFGS, and coordinate descent polishing. The LP step
solves a linear program to find a descent direction that provably bounds
the cross-correlation, providing a strong initialization for gradient methods.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """Cross-correlation (w * a)[i] = sum_j w[j+i]*a[j] for i=0..len(a)-1."""
    n_w = len(w)
    n_a = len(a)
    size = n_w + n_a - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(W) * A, fft_size)
    return result[:n_a]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_cross_corr_matrix(f, n):
    """Build matrix A (2n-1 x n) where (A@g)[i] = sum_k f[i-k]*g[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve_direction(f, M, n, epsilon=1e-4):
    """Solve LP: max sum(g) + eps*dot(f,g) s.t. A@g <= M, g >= 0."""
    A = build_cross_corr_matrix(f, n)
    c = -(np.ones(n) + epsilon * f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return None


def sm_obj(z, T, n):
    """Soft-max smoothed objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def sm_jac(z, T, n):
    """Gradient of soft-max objective w.r.t. log-domain z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return grad_z


def run_sm_lbgfs(z0, T, n, time_limit=60.0, maxiter=5000):
    """L-BFGS with soft-max objective, time-limited."""
    t_start = time.monotonic()
    elapsed = [t_start]

    def _obj(z):

        return sm_obj(z, T, n)

        return sm_jac(z, T, n)

    try:

def lp_refine(f0, n, time_limit=300.0, max_iters=500):
    """LP-based iterative refinement with line search."""
    t0 = time.monotonic()
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    base_t = 0.15

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()
        g = lp_solve_direction(f, M, n)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        best_trial_obj = best_obj
        best_trial_f = None

        step_sizes = [base_t, base_t * 0.5, base_t * 2.0, base_t * 0.25,
                      base_t * 4.0, 0.05, 0.3, 0.5]
        for t in step_sizes:
            if t <= 0 or t > 0.95:
                continue
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            ts = f_trial.sum()
            if ts < 1e-15:
                continue
            f_trial = f_trial / ts
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj - 1e-14:
                best_trial_obj = obj
                best_trial_f = f_trial.copy()

        if best_trial_f is not None:
            f = best_trial_f
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            base_t *= 0.7
            if base_t < 1e-5 or no_improve > 40:
                rng = np.random.default_rng(12345 + it)
                f = project_simplex(f + rng.standard_normal(n) * 0.003)
                no_improve = 0
                base_t = 0.1

    return best_f, best_obj


def coordinate_descent(a0, time_limit=60.0):
    """Coordinate descent cycling through log-spaced multipliers."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.18, 1.4, 1.7, 2.5])
    rounds = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-15:
                candidates = [1e-5, 1e-4, 1e-3, 5e-3]
            else:
                candidates = [old_val * m for m in multipliers]

            best_j_obj = best_obj
            best_j_val = old_val

            for new_val in candidates:
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.01 or scale > 100:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 5:
                break
            rng = np.random.default_rng(rounds * 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
        else:
            stall = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        f_best = np.maximum(best_seq, 0)
        fs = f_best.sum()
        if fs > 1e-15:
            f_best = f_best / fs

        for attempt in range(8):
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            noise_scale = max(0.0005, 0.002 / (attempt + 1))
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(f_best + noise)
            z = z_from_a(a_pert)
            t_alloc = min(rem * 0.25, 25.0)
            if t_alloc < 5:
                break
            z = run_sm_lbgfs(z, 500, n_best, time_limit=t_alloc, maxiter=3000)
            a_fin = softmax_from_z(z)
            record(a_fin, f"pert{attempt}")
            f_best = np.maximum(best_seq, 0)
            fs2 = f_best.sum()
            if fs2 > 1e-15:
                f_best = f_best / fs2

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs_final = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs_final):
        best_score = fs_final

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
