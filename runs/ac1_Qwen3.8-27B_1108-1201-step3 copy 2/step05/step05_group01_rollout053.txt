"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Improved LP-based Frank-Wolfe with coupled directions and momentum
2. L-BFGS on smoothed log-sum-exp objective
3. Peak-shaving subgradient method
4. Multi-length exploration with Fourier-modulated initializations
5. Coordinate descent and final polishing
Key innovations: coupled LP directions, momentum FW, structured restarts

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_fn(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lpfw_optimize(f0, time_limit, verbose=False, coupled=True, momentum_beta=0.0):
    """Improved LP-based Frank-Wolfe with coupled directions and optional momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    v_mom = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1

        if coupled:
            # Solve LP with multiple M values and combine
            directions = []
            for m_scale in [0.97, 1.0, 1.03]:
                g_i = solve_lp_direction(f, M * m_factor * m_scale, n)
                if g_i is not None and g_i.sum() > 1e-10:
                    g_i = g_i / g_i.sum()
                    directions.append(g_i)
            
            if not directions:
                m_factor = min(m_factor * 1.15, 10.0)
                continue
            
            g = np.mean(directions, axis=0)
            gs = g.sum()
            if gs < 1e-15:
                m_factor = min(m_factor * 1.15, 10.0)
                continue
            g = g / gs
        else:
            g = solve_lp_direction(f, M * m_factor, n)
            if g is None or g.sum() < 1e-15:
                m_factor = min(m_factor * 1.15, 10.0)
                continue
            g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            m_factor = min(m_factor * 1.05, 5.0)
            continue

        # Apply momentum
        if momentum_beta > 0:
            v_mom = momentum_beta * v_mom + (1 - momentum_beta) * g
            v_mom = project_simplex(v_mom)
            direction = v_mom
        else:
            direction = g

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-15

        for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * direction
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(18):
                mid = (lo + hi) / 2.0
                ft = (1 - mid) * f + mid * direction
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = (1 - best_alpha) * f + best_alpha * direction
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_factor = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.1, 10.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 1000003 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 60, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_factor = 1.0
            no_improve = 0
            v_mom = np.zeros(n)

    return best_f, best_obj


def _smooth_obj_grad(t, beta):
    """Compute smooth log-sum-exp objective and gradient in t-space."""
    n = len(t)
    a = softmax_fn(t)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    obj = conv_max + np.log(exp_shifted.sum()) / beta

    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    dot = np.dot(a, grad_a)
    grad_t = a * (grad_a - dot)

    return obj, grad_t


def lbfqg_optimize(f0, time_limit, verbose=False):
    """L-BFGS on smooth log-sum-exp with annealed beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    eps = 1e-15
    t = np.log(f + eps)

    betas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    t0_time = time.monotonic()

    for bi, beta in enumerate(betas):
        remaining = time_limit - (time.monotonic() - t0_time)
        if remaining < 20:
            break
        res = minimize(
            _smooth_obj_grad, t, args=(beta,),
            jac=True, method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-10, 'maxls': 40, 'maxfun': 1500}
        )
        t = res.x
        a = softmax_fn(t)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a / a_sum
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = a.copy()
            if verbose:
                print(f"    L-BFGS beta={beta}: obj={best_obj:.10f}")

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Peak-shaving subgradient method with BB step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    prev_f = f.copy()
    prev_sg = None
    
    top_k = max(10, n // 10)
    temp = 0.0001

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()

        # Get top-K peaks with soft weights
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # BB step size
        if prev_sg is not None:
            y = f - prev_f
            r = sg - prev_sg
            yr = np.dot(y, r)
            yy = np.dot(y, y)
            if yr > 1e-18 and yy > 1e-18:
                step = yr / yy
                step = np.clip(step, 1e-4, 0.1)
            else:
                step = 0.001
        else:
            step = 0.001

        # Projected gradient step
        f_new = project_simplex(f - step * sg)
        
        if np.max(np.abs(f_new - f)) < 1e-14:
            no_improve += 1
            if no_improve > 100:
                break
            temp *= 1.01
            prev_f = f.copy()
            prev_sg = sg.copy()
            continue

        prev_f = f.copy()
        prev_sg = sg.copy()
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and best_obj < 1.55:
                print(f"    PeakShave: obj={best_obj:.10f}, step={step:.6f}")
        else:
            no_improve += 1

        temp *= 1.005

        if no_improve > 80:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            prev_sg = None
            no_improve = 0
            temp = 0.0001

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def fourier_modulated_init(n, seed, n_modes=8):
    """Create a Fourier-modulated non-negative initialization."""
    rng = np.random.default_rng(seed)
    j = np.arange(n)
    a = np.ones(n)
    for m in range(1, n_modes + 1):
        angle = 2 * np.pi * m * j / n
        c = rng.uniform(-0.3, 0.3)
        phi = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(angle + phi)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss
    return a
