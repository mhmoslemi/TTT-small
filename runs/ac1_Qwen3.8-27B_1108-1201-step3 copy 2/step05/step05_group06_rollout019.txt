def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fourier_flat_convolve(a, n_iters=1000):
    """
    Use iterative Fourier method to flatten the autocorrelation.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    for it in range(n_iters):
        A = np.fft.rfft(a_padded)
        power = np.abs(A) ** 2
        mean_power = np.sum(power) / len(power)
        
        # Flatten the power spectrum
        # New power: interpolate between current and flat
        gamma = 0.1  # mixing parameter
        new_power = (1 - gamma) * power + gamma * mean_power
        new_power = np.maximum(new_power, 0)
        
        # Keep the phase, adjust the magnitude
        phase = np.angle(A)
        new_mag = np.sqrt(new_power)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_padded = np.fft.irfft(A_new, N)
        
        # Project to non-negative, zero-pad back to length n
        a_padded[:n] = np.maximum(a_padded[:n], 0)
        a_padded[n:] = 0
        
        # Normalize to sum = 1
        s = a_padded.sum()
        if s > 1e-15:

    return a_padded[:n] / a_padded[:n].sum()

Combines smooth Adam optimization, active-set local optimization,
and coordinate descent with improved cooling schedules.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w

        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()

        it += 1

def single_cd(f0, time_limit, verbose=False):
    """Single coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            lo = 0.0
            hi = min(1.0 - 1e-10, old_val * 3)  # Don't search too far
            if hi <= lo:
                continue
            
            def obj_at(x):
                denom = 1.0 - old_val
                if denom < 1e-10:
                    return 1e10
                scale = (1.0 - x) / denom
                f_trial = f.copy()
                f_trial[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                return fast_convolve(f_trial).max()
            
            # Golden section
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d)
            
            for _ in range(16):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = obj_at(d)
            
            opt_val = (lo + hi) / 2.0
            opt_obj = obj_at(opt_val)
            
            if opt_obj < best_obj - 1e-14:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                best_obj = opt_obj
                best_f = f.copy()
                improved = True
        
        if not improved and passes > 5:
            break
    
    return best_f, best_obj

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved smooth Adam with slow cooling, active-set local polishing,
fast coordinate descent, and multi-length exploration.
Key improvements: logarithmic cooling to T=0.001, adaptive restarts,
active-set identification for targeted local optimization, and
efficient FFT-based coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))
    N = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(f, N)
    W_fft = np.fft.rfft(w, N)
    corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    grad = (2.0 / T) * corr
    return grad, lse_val, conv, w


def smooth_optimize(f0, time_limit, T_start=0.4, T_end=0.002, alpha=0.012, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Very slow exponential cooling
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.5)
        T = max(T, T_end)

        lr = alpha * (0.4 ** (progress ** 0.3))

        grad, lse_val, conv, w = smooth_grad(f, T)

        t_adam += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t_adam)
        v_hat = v_state / (1 - beta2 ** t_adam)

        f = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f)

        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 5e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Smooth iter {it}: obj={best_obj:.10f}, T={T:.5f}")
        else:
            no_improve += 1

        if no_improve > 1500:
            seed_val = int(elapsed * 7777) % 2147483647
            rng_p = np.random.default_rng(seed_val)
            noise_scale = min(0.015, 0.003 * (1.0 + no_improve / 400.0))
            f = project_simplex(best_f + rng_p.standard_normal(n) * noise_scale)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t_adam = 0
            no_improve = 0

    return best_f, best_obj


def active_polish(f0, time_limit, verbose=False):
    """
    Active-set polishing: iteratively identify top convolution entries
    and minimize their max using projected gradient with line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    K = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        # Identify active entries
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:K]
        active_tol = M * 0.15
        active_mask = conv >= M - active_tol
        active_all = np.unique(np.concatenate([active, np.where(active_mask)[0]]))
        active_all = active_all[active_all < len(conv)]

        # Use log-sum-exp of active entries
        T_act = 0.005 + 0.02 * (1.0 - min((time.monotonic() - t0) / time_limit, 1.0))
        active_vals = conv[active_all]
        scaled = active_vals / T_act
        scaled -= scaled.max()
        e = np.exp(scaled)
        w = e / e.sum()

        # Gradient of log-sum-exp of active conv entries w.r.t. f
        N = 1 << (3 * n - 1).bit_length()
        A_fft = np.fft.rfft(f, N)
        W_full = np.zeros(len(A_fft))
        # w is defined on active_all indices of conv (length 2n-1)
        w_arr = np.zeros(len(conv))
        w_arr[active_all] = w
        W_fft = np.fft.rfft(w_arr, N)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
        grad = (2.0 / T_act) * corr

        # Project to tangent space of simplex
        support = f > 1e-12
        if support.sum() > 1:
            grad_t = grad - np.mean(grad[support]) * support
        else:
            grad_t = grad

        # Adaptive line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
            f_trial = project_simplex(f - alpha * grad_t)
            conv_trial = fast_convolve(f_trial)
            val_trial = conv_trial[active_all].max()
            if val_trial < best_val:
                best_val = val_trial
                best_alpha = alpha

        if best_alpha > 1e-8:
            f = project_simplex(f - best_alpha * grad_t)
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    print(f"    Active-polish iter {it}: obj={best_obj:.10f}")
        else:
            # Perturb
            seed = int(time.monotonic() * 10000) % 2147483647
            rng_p = np.random.default_rng(seed)
            f = project_simplex(best_f + rng_p.standard_normal(n) * 0.004)

    return best_f, best_obj


def full_cd(f0, time_limit, verbose=False):
    """
    Full coordinate descent with golden section search over all coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    ORDER = 0.6180339887

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            lo = 0.0
            hi = min(1.0 - 1e-10, old_val * 4.0)
            if hi - lo < 1e-12:
                continue

            def _eval(x):
                scale = (1.0 - x) / denom
                ft = f.copy()
                ft[j] = x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                return fast_convolve(ft).max()

            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)
            fc = _eval(c)
            fd = _eval(d)

            for _ in range(14):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = _eval(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = _eval(d)

            opt_val = (lo + hi) / 2.0
            opt_obj = _eval(opt_val)

            if opt_obj < best_obj - 1e-14:
                scale = (1.0 - opt_val) / denom
                f[j] = opt_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                best_obj = opt_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 3:
            break

    return best_f, best_obj


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = _lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.1, 3.0)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 80:
            seed = it * 13 + 5
            rng_p = np.random.default_rng(seed)
            scale = 0.003 + 0.004 * min(no_improve // 50, 5)
            noise = rng_p.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def _lp_direction(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_symmetric_geo(n, r):
    half = (n + 1) // 2
    asc = r ** np.arange(half, dtype=np.float64)
    desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
    a = np.concatenate([asc, desc])
    if len(a) < n:
        a = np.concatenate([a, np.ones(n - len(a))])
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** power + 0.01
    return a / a.sum()


def make_sine(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) + 0.01
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
