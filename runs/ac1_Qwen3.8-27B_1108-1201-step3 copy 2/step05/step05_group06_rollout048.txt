"""
Optimized step function search combining parametric initialization,
log-sum-exp smooth Adam optimization with cosine annealing,
pair-swap exploration, and multi-length search to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(f, T):
    """Gradient of logsumexp(conv(f,f)/T) using FFT cross-correlation."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = T * (scaled.max() + np.log(e.sum()))

    # g[i] = sum_j f[j] * w[i+j] for i=0..n-1
    # Use FFT: g[i] = (f * w_rev)[2n-2-i] where w_rev = w[::-1]
    v = w[::-1]
    N = 1 << (3 * n).bit_length()
    F_fft = np.fft.rfft(f, N)
    V_fft = np.fft.rfft(v, N)
    conv_fv = np.fft.irfft(F_fft * V_fft, N)
    g = conv_fv[n - 1:2 * n - 1][::-1]
    grad = (2.0 / T) * g
    return grad, lse_val, conv, w

def pair_swap_explore(f0, time_limit, n_swaps_per_sec=500, verbose=False):
    """Explore via random pair swaps (move mass between two positions)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(67890)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        # Try moving a fraction of mass from i to j (or vice versa)
        total_ij = f[i] + f[j]
        if total_ij < 1e-12:
            continue
        cur_ratio = f[i] / total_ij
        # Perturb the ratio

        f_trial[i] = new_ratio * total_ij
        f_trial[j] = (1.0 - new_ratio) * total_ij
        val = fast_convolve(f_trial).max()

        if val < best_obj - 1e-14:
            f = f_trial
            if val < best_obj:
                best_obj = val
                best_f = f.copy()
                if verbose and it % 5000 == 0:
                    print(f"    PairSwap iter {it}: obj={best_obj:.10f}")

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        if g_sum < 1e-15:

            m_exp = min(m_exp * 1.05, 3.0)

            ft = ft / st

    t0 = time.monotonic()

        passes += 1
        improved = False

            if time.monotonic() - t0 > time_limit:

            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Golden section on f[j], scaling others to maintain sum=1
            lo = 0.0
            hi = 0.999
            c = hi - ORDER * (hi - lo)
            d = lo + ORDER * (hi - lo)

            def eval_at(x):
                denom = 1.0 - old_val
                if denom < 1e-10:
                    return 1e10
                scale = (1.0 - x) / denom
                f_trial = f.copy()
                f_trial[j] = x

                f_trial[mask] *= scale
                return fast_convolve(f_trial).max()

            fd = eval_at(d)
            for _ in range(12):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - ORDER * (hi - lo)
                    fc = eval_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + ORDER * (hi - lo)
                    fd = eval_at(d)

                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def make_gaussian(n, width=0.3):
    """Gaussian window normalized to sum=1."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_raised_cosine(n, power=2.0):
    """Raised cosine window: (cos(pi*(x-0.5)))^power clipped to positive."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** power
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_gaussian_skewed(n, center=0.5, width=0.3):
    """Asymmetric Gaussian (skewed normal-like)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    # Add slight asymmetry
    a *= (1 + 0.3 * (x - center))
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_poly_window(n, p=2.0):
    """Polynomial window: (1 - 4*(x-0.5)^2)^p."""
    x = np.linspace(0, 1, n)
    a = (1 - 4 * (x - 0.5) ** 2) ** p
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def make_geometric(n, r):
    """Geometric sequence normalized to sum=1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_sequence(a, best_state, label, verbose=True):
    """Try a sequence, update best if improved. Returns (is_better, score)."""
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return False, float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return False, float('inf')
    if np.isfinite(score) and score < best_state[0]:
        best_state[0] = score
        best_state[1] = a.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return True, score
    return False, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer: parametric initialization, smooth Adam optimization,
    LP-FW, pair-swap exploration, and coordinate descent across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    best_state = [best_score, best_seq]

    # Helper to try a sequence
    def _try(a, label):
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            best_state[0] = score
            best_state[1] = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old:.8f}]")

    # === Phase 0: Diverse parametric initializations ===
    print("=== Phase 0: Parametric Initializations ===")
    n_base = len(parent) if parent is not None else 768

    init_candidates = []
    for n_try in [n_base, 512, 1024]:
        # Gaussian with various widths
        for width in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
            init_candidates.append((make_gaussian(n_try, width), f"gauss_w{width}_n{n_try}"))
        # Raised cosine
        for p in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0]:
            init_candidates.append((make_raised_cosine(n_try, p), f"cos_p{p}_n{n_try}"))
        # Polynomial windows
        for p in [1.5, 2.0, 3.0, 4.0, 5.0]:
            init_candidates.append((make_poly_window(n_try, p), f"poly_p{p}_n{n_try}"))
        # Skewed Gaussian
        for c in [0.4, 0.5, 0.6]:
            for w in [0.2, 0.25, 0.3]:
                init_candidates.append((make_gaussian_skewed(n_try, c, w), f"skew_c{c}_w{w}_n{n_try}"))
        # Geometric
        for r in [0.97, 0.98, 0.99, 0.995]:
            init_candidates.append((make_geometric(n_try, r), f"geo_r{r}_n{n_try}"))
        # Uniform
        init_candidates.append((np.ones(n_try) / n_try, f"unif_n{n_try}"))

    for a, label in init_candidates:
        _try(a, label)

    print(f"  After Phase 0: best={best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 1: Smooth optimization on best candidates ===
    print(f"\n=== Phase 1: Smooth Optimization (best={best_score:.8f}) ===")

    # Run smooth optimization on the best and a few diverse candidates
    n_smooth_runs = 3
    candidates_to_optimize = [best_seq.copy()]

    # Add some diverse candidates for diversification
    if parent is not None:
        candidates_to_optimize.append(parent.copy())
    for width in [0.25, 0.35]:
        candidates_to_optimize.append(make_gaussian(n_base, width))

    for idx, cand in enumerate(candidates_to_optimize[:n_smooth_runs]):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem, 300)
        print(f"  Smooth run {idx+1}/{n_smooth_runs} (n={len(cand)}, t={t_alloc:.0f}s)")
        a_opt, _ = smooth_optimize(cand, time_limit=t_alloc, T_start=0.4, T_end=0.003,
                                   alpha=0.015, verbose=True, restart_interval=300)
        _try(a_opt, f"smooth_{idx}_n{len(a_opt)}")

    # Second round of smooth optimization with different params
    rem = deadline - time.monotonic() - 500
    if rem > 60 and best_seq is not None:
        t_alloc = min(rem, 250)
        print(f"  Smooth run (fine, t={t_alloc:.0f}s)")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t_alloc, T_start=0.1, T_end=0.002,
                                   alpha=0.008, verbose=True, restart_interval=200)
        _try(a_opt, "smooth_fine")

    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  LP-FW (n={len(best_seq)}, t={t_alloc:.0f}s)")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "lpfw_main")

        # LP-FW with perturbation
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            noise = rng.standard_normal(len(best_seq)) * 0.005
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 100)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t_alloc, verbose=True)
            _try(a_opt, "lpfw_pert")

    # === Phase 3: Pair swap exploration ===
    print(f"\n=== Phase 3: Pair Swap (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  PairSwap (t={t_alloc:.0f}s)")
            a_opt, _ = pair_swap_explore(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "pairswap")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  CoordDesc (t={t_alloc:.0f}s)")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "coord")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Different Lengths (best={best_score:.8f}) ===")
    if best_seq is not None:
        for target_n in [512, 1024, 1536]:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = smooth_optimize(f_interp, time_limit=t_alloc, T_start=0.2, T_end=0.005,
                                       alpha=0.01, verbose=False)
            _try(a_opt, f"len_{target_n}")

    # === Phase 6: Final polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for _ in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            # Small perturbation + fine smooth
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem, 40)
            a_opt, _ = smooth_optimize(f_pert, time_limit=t_alloc, T_start=0.05, T_end=0.002,
                                       alpha=0.005, verbose=False, restart_interval=150)
            _try(a_opt, "polish")

        # LP-FW final
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 30), verbose=False)
            _try(a_opt, "lpfw_final")

        # Coordinate descent final
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20), verbose=False)
            _try(a_opt, "coord_final")

    # Fallback
    if best_seq is None:
        fallback_n = n_base if 'n_base' in dir() else 512
        best_seq = np.ones(fallback_n) / fallback_n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        if not np.isfinite(best_score):
            best_seq = make_gaussian(512, 0.3)
            best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
