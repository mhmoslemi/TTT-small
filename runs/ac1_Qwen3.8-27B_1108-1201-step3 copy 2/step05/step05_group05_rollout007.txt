- sum_k w_k * (a*a)_k = sum_i sum_j a_i * a_j * w_{i+j}

- Projected gradient: grad = 2*M*a, step in -grad direction, project to simplex

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def reweight_gradient(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k.
    grad_m = 2 * sum_j a_j * w_{m+j}
    """
    # np.correlate(w, a, mode='valid') gives sum_j w[m+j]*a[j] for m=0..n-1
    corr = np.correlate(w, a, mode='valid')[:len(a)]
    return 2.0 * corr

def flat_spectrum_init(n, rng, seed_offset=0):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        A[k] = np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def random_smooth_init(n, rng):
    """Random initialization with smoothness (low-pass filtered random)."""
    a = rng.exponential(1.0, n)
    # Apply a low-pass filter in frequency domain
    A = np.fft.rfft(a)
    cutoff = int(n * 0.3)
    if cutoff > 0:
        A[cutoff:] *= 0.3
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def triangular_init(n):
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    return a / a.sum()


def gaussian_init(n, width=0.3):
    x = np.linspace(-3, 3, n)
    a = np.exp(-x**2 / (2 * width**2))

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
