"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced multi-scale optimization:
1. Spectral initialization (flat-spectrum random phase signals)
2. Smooth gradient descent with adaptive top-K log-sum-exp (fast, no LP needed)
3. LP-Frank-Wolfe with constraint generation (reduced constraints for speed)
4. Multi-length exploration (768, 1024, 1280, 1536, 2048)
5. Iterative peak-elimination via subgradient on active set
6. Coordinate descent polishing
Key innovations: top-K constraint LP (avoids full 2n-1 constraints), spectral init,
and adaptive time allocation favoring longer sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_w_weighted(f, w):
    """Compute grad = 2 * cross-correlation of f with w (w has length 2n-1)."""
    n = len(f)
    N = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(F) * W, N)[:n]
    return 2.0 * R


def smooth_obj_grad(a, beta, top_k=None):
    """Compute smooth approximation of max auto-conv and gradient using top-K peaks."""
    conv = fast_convolve(a)
    n = len(a)
    
    if top_k is not None and top_k < len(conv):
        # Only use top-K peaks for the smooth approximation
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        conv_active = conv[peaks]
        conv_max = conv_active.max()
        shifted = beta * (conv_active - conv_max)
        exp_shifted = np.exp(shifted)
        w_active = exp_shifted / exp_shifted.sum()
        obj = conv_max + np.log(exp_shifted.sum()) / beta
        
        # Build full w vector (zeros except at peaks)
        w = np.zeros(len(conv))
        w[peaks] = w_active
    else:
        conv_max = conv.max()
        shifted = beta * (conv - conv_max)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        obj = conv_max + np.log(exp_shifted.sum()) / beta
    
    grad = cross_corr_w_weighted(a, w)
    return obj, grad


def spectral_init(n, seed=42):
    """Initialize with a flat-spectrum random phase signal (projected to simplex)."""
    rng = np.random.default_rng(seed)
    # Random phases in frequency domain
    freq_size = 1 << (n - 1).bit_length()
    phases = rng.uniform(-np.pi, np.pi, freq_size // 2 + 1)
    # Flat amplitude spectrum
    spec = np.exp(1j * phases)
    # Inverse FFT to get time-domain signal
    sig = np.fft.irfft(spec, n)
    # Project to simplex (make non-negative, normalize)
    sig = np.maximum(sig, 0.0)
    s = sig.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return sig / s


def lp_fw_optimize(f0, time_limit, verbose=False, top_k_constraints=200):
    """LP-based Frank-Wolfe with constraint generation (only top-K constraints)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP with reduced constraints (top-K where f*g is closest to M)
        # First, solve with all constraints if n is small enough
        if n <= 1024:
            c = -np.ones(n)
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M)
            bounds = [(0, None)] * n
        else:
            # Use only top-K constraints for large n
            # Find where the "dual" is most active
            # Approximate: use constraints where the current f*f is highest
            k_actual = min(top_k_constraints, 2 * n - 1)
            active = np.argpartition(conv, -k_actual)[-k_actual:]
            c = -np.ones(n)
            A_ub = np.zeros((k_actual, n))
            for idx, k in enumerate(active):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                A_ub[idx, js:je+1] = f[k - je:k - js + 1][::-1]
            b_ub = np.full(k_actual, M)
            bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.003 + 0.002 * min(no_improve // 80, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Binary refinement
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def smooth_gradient_optimize(f0, time_limit, verbose=False, initial_beta=20.0, top_k=None):
    """Smooth gradient descent with adaptive beta and top-K log-sum-exp."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    beta = initial_beta
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        obj, grad = smooth_obj_grad(f, beta, top_k=top_k)
        
        # Adam-like update
        v = momentum * v + (1 - momentum) * grad
        f_new = f - lr * v
        f_new = project_simplex(f_new)
        
        true_obj = fast_convolve(f_new).max()
        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr *= 0.999
            if verbose and (time.monotonic() - t0) % 30 < 1.0:
                print(f"    SmoothGrad: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.98
            if no_improve > 100:
                beta = min(beta * 1.5, 200.0)
                if no_improve > 300:
                    rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                    v = np.zeros(n)
                    lr = 0.005
                    no_improve = 0
                    beta = initial_beta
        
        lr = max(lr, 0.0005)

    return best_f, best_obj


def peak_elimination_optimize(f0, time_limit, verbose=False):
    """Iterative peak elimination: focus on the active set of maximum peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.005
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # Get the subgradient at the active set (peaks within 1% of max)
        threshold = conv_max * 0.99
        active = conv[conv >= threshold]
        active_count = len(active)
        
        # Subgradient weighted by proximity to max
        weights = (conv - threshold) / (active.sum() - threshold * active_count + 1e-30)
        w_full = np.zeros(len(conv))
        mask = conv >= threshold
        w_full[mask] = weights[mask]
        w_full = w_full / (w_full.sum() + 1e-30)
        
        grad = cross_corr_w_weighted(f, w_full)
        
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        true_obj = fast_convolve(f_new).max()
        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = max(lr * 0.999, 0.001)
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve > 200:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                lr = 0.003
                no_improve = 0
        
        if verbose and (best_obj < 1.51) and (time.monotonic() - t0) % 30 < 1.0:
            print(f"    PeakElim: obj={best_obj:.10f}, lr={lr:.5f}, active={active_count}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n < 2 or new_n < 2:
        return np.full(new_n, f.mean())
    
    from scipy.interpolate import CubicSpline
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    cs = CubicSpline(old_x, f, bc_type='not-a-knot')
    f_interp = cs(new_x)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def inflate_sequence(f, new_n):
    """Inflate sequence by nearest-neighbor (preserves structure better for optimization)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    indices = np.round(np.linspace(0, old_n - 1, new_n)).astype(int)
    f_new = f[indices].astype(np.float64)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new

                t_alloc = min(rem * 0.6, 50)
                a_opt, _ = peak_elimination_optimize(f_start, time_limit=t_alloc)
                try_seq(a_opt, f"final_peak_{attempt}")
            else:
                # LP-FW (if feasible)
                if n_best <= 1280:
                    f_start = best_seq.copy()
                    t_alloc = min(rem * 0.6, 50)
                    a_opt, _ = lp_fw_optimize(f_start, time_limit=t_alloc)
                    try_seq(a_opt, f"final_lpfw_{attempt}")
                else:
                    # Fallback: smooth gradient
                    f_start = best_seq.copy()
                    t_alloc = min(rem * 0.6, 50)
                    a_opt, _ = smooth_gradient_optimize(f_start, time_limit=t_alloc,
                                                         initial_beta=60.0,
                                                         top_k=min(80, n_best // 6))
                    try_seq(a_opt, f"final_sgf_{attempt}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
