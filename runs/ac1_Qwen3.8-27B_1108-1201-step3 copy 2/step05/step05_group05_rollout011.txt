"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with line search, subgradient fallback, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_constraint_matrix(f):
    """Build the (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n))
    mask = (idx >= 0) & (idx < n)
    A[mask] = f[idx[mask]]
    return A


def solve_lp_direction(f, M):
    """Solve the LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Returns the optimal g (normalized to sum 1) or None if infeasible."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs-ds')
    
    if res.status != 0:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_a = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:

        M = best_obj  # Use the current max as the constraint bound
        
        g = solve_lp_direction(f, M)
        if g is None:
            # LP infeasible or trivial, try with relaxed constraint
            g = solve_lp_direction(f, M * 1.001)
            if g is None:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with adaptive line search, top-k constraint relaxation, subgradient fallback,
and coordinate descent polish. Key innovation: solves a linear program to find
optimal descent directions, then uses line search for actual updates.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_lp_matrix(f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n))
    mask = (idx >= 0) & (idx < n)
    A[mask] = f[idx[mask]]
    return A


def solve_lp_direction(f, M, top_k=None):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M (optionally only top-k k's)."""
    n = len(f)
    A_full = build_lp_matrix(f)

    if top_k is not None and top_k < 2 * n - 1:
        conv = fast_convolve(f)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        A = A_full[top_idx]
        b = np.full(top_k, M)
    else:
        A = A_full
        b = np.full(2 * n - 1, M)

    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs-ds')
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        except Exception:
            return None

    if res.status != 0:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n + 1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, tau=80.0, top_k=100):
    """Compute smooth subgradient of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = tau * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * f[k - idx]
    return grad, conv

                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                old_remaining = 1.0 - old_val
                if old_remaining > 1e-10:
                    scale = (1.0 - best_j_val) / old_remaining
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved and pass_count > 2:
            break
    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Generate non-negative sequence with approximately flat cyclic spectrum."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.zeros(n, dtype=complex)
    A[0] = np.sqrt(n)
    half = n // 2
    for k in range(1, half + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < n - k:
            A[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A[n // 2] = 1.0
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization using LP-based direction finding as the core
    optimizer, with subgradient fallback, coordinate descent polish,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Quick LP exploration across lengths ===
    print("\n=== Phase 1: LP Exploration ===")
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [640, 768, 896, 1024, 1280]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths.sort()

    candidates = {}
    exploration_time = min(300, (deadline - time.monotonic()) * 0.25)
    per_candidate_time = exploration_time / len(target_lengths)

    for n_len in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        t_alloc = min(per_candidate_time, rem)
        if t_alloc < 10:
            break

        # Get best initialization for this length
        if parent is not None:
            init = interpolate_sequence(parent, n_len)
        else:
            init = flat_spectrum_init(n_len, rng)

        init = np.maximum(init, 0)
        init = init / init.sum()
        init_obj = fast_convolve(init).max()
        print(f"  n={n_len}: init_obj={init_obj:.8f}, t={t_alloc:.0f}s")

        a_opt, obj = lp_optimize(init, time_limit=t_alloc, verbose=(n_len == (len(parent) if parent is not None else 768)))
        score = try_seq(a_opt, f"lp_n{n_len}")
        candidates[n_len] = a_opt

    # === Phase 2: Deep LP optimization on best ===
    print(f"\n=== Phase 2: Deep LP Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        deep_time = min(500, (deadline - time.monotonic()) * 0.35)
        print(f"  Deep LP on n={len(best_seq)}, t={deep_time:.0f}s")
        a_opt, obj = lp_optimize(best_seq, time_limit=deep_time, verbose=True)
        try_seq(a_opt, "deep_lp")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sg = min(rem * 0.4, 120)
            print(f"  SG t={t_sg:.0f}s")
            a_opt, obj = subgrad_optimize(best_seq, time_limit=t_sg)
            try_seq(a_opt, "sg_refine")

            # LP again after subgradient
            rem = deadline - time.monotonic() - 100
            if rem > 30:
                t_lp2 = min(rem * 0.4, 100)
                a_opt, obj = lp_optimize(best_seq, time_limit=t_lp2)
                try_seq(a_opt, "lp_after_sg")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 5: LP after CD ===
    print(f"\n=== Phase 5: LP After CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_lp3 = min(rem * 0.5, 80)
            a_opt, obj = lp_optimize(best_seq, time_limit=t_lp3)
            try_seq(a_opt, "lp_after_cd")

    # === Phase 6: Perturbation restarts ===
    print(f"\n=== Phase 6: Perturbation Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_alloc = min(rem / (6 - p), 50)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise, target_sum=1.0)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init, target_sum=1.0)
            elif p == 3:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init, target_sum=1.0)
            else:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise, target_sum=1.0)

            a_opt, obj = lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{p}")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd2 = min(rem, 40)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd2)
            try_seq(a_opt, "final_cd")

            rem = deadline - time.monotonic() - 15
            if rem > 10:
                a_opt, obj = lp_optimize(best_seq, time_limit=min(rem, 20))
                try_seq(a_opt, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
