```python
import numpy as np

from scipy.optimize import linprog, minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(p):
    """Build the matrix A such that (A @ b)[k] = conv(p, b)[k]."""
    n = len(p)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        # (p*b)[k] = sum_j p[k-j] * b[j] where 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = p[k - j]
    return A

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

For k = j + n - 1:
`conv[j+n-1] = sum_i w[i] * p[n-1-(j+n-1)+i] = sum_i w[i] * p[i-j]`

So `conv[j+n-1] = sum_i w[i]*p[i-j]` where valid i means 0<=i<2n-1 and 0<=i-j<n, i.e., j<=i<=j+n-1.

So `conv[j+n-1] = sum_{i=j}^{j+n-1} w[i]*p[i-j]`.

And I want `grad[j]/2 = sum_{k} w[k]*p[k-j]` where k ranges over [j, j+n-1] (valid range since 0<=k-j<n).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided optimization,
L-BFGS polishing with log-sum-exp smoothing, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_convolution_matrix(p):
    """Build matrix A where (A @ b)[k] = conv(p, b)[k]."""
    n = len(p)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = p
    return A


def solve_lp_direction(p, M):
    """Solve LP: max sum(b) s.t. b>=0, conv(p,b)[k]<=M for all k."""
    n = len(p)
    A = build_convolution_matrix(p)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, 2.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(p, w):
    """Compute gradient of log-sum-exp of conv(p,p) wrt p."""
    n = len(p)
    p_rev = p[::-1]
    conv = np.convolve(w, p_rev, 'full')
    grad = 2.0 * conv[n-1 : 2*n-1]
    return grad


def softmax_weights(conv, temp):
    """Compute softmax weights over convolution values."""
    x = conv / temp
    x -= x.max()
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    return w / s


def lbfgs_optimize(p0, time_limit, temp_init=0.005, verbose=False):
    """L-BFGS-B optimization of log-sum-exp approximation."""
    n = len(p0)
    temp = temp_init
    t0 = time.monotonic()
    
    def obj_and_grad(p_flat):
        p = p_flat.copy()
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-15:
            p = np.ones(n) / n
        else:
            p = p / s
        conv = fast_convolve(p)
        x = conv / temp
        x -= x.max()
        log_sum_exp = temp * (x.max() + np.log(np.sum(np.exp(x - x.max()))))
        # Gradient
        w = softmax_weights(conv, temp)
        grad = compute_gradient(p, w)
        # Project gradient to simplex tangent space
        grad -= grad.mean()
        return log_sum_exp, grad
    
    # Initial point
    p = p0.copy()
    s = p.sum()
    p = p / s
    
    best_p = p.copy()
    best_val = obj_and_grad(p)[0]
    
    x0 = p.copy()
    result = minimize(obj_and_grad, x0, method='L-BFGS-B',
                      bounds=[(0, None)] * n,
                      options={'maxiter': 5000, 'maxfun': 50000,
                               'disp': False, 'gtol': 1e-10})
    
    if result.success or result.fun < best_val:
        p_new = np.maximum(result.x, 0)
        s = p_new.sum()
        if s > 1e-15:
            p_new = p_new / s
            conv = fast_convolve(p_new)
            if conv.max() < best_val / temp:  # rough check
                best_p = p_new
                best_val = result.fun
    
    return best_p, best_val


def adam_optimize(f0, time_limit, lr=0.005, temp_init=0.006, temp_decay=0.9995,
                  top_k=None, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = softmax_weights(conv, temp)
        if top_k is not None and top_k < len(conv):
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_m = np.zeros(len(conv))
            w_m[idx] = w[idx]
            w_m /= w_m.sum()
            w = w_m
        sg = compute_gradient(f, w)
        
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = max(lr * (0.9999 ** t), lr * 0.1)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(t * 17 + 3)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.7, 0.003)
            cur_lr = lr
            no_improve = 0
    
    return best_f, best_obj


def lp_guided_optimize(p0, time_limit, verbose=False):
    """LP-guided optimization: solve LP periodically, use Adam in between."""
    n = len(p0)
    p = p0.copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_M = fast_convolve(p).max()
    t0 = time.monotonic()
    
    # LP solver state
    lp_direction = None
    lp_countdown = 0
    lp_interval = 5  # solve LP every 5 "inner" iterations
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    temp = 0.005
    cur_lr = 0.004
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Periodically solve LP
        lp_countdown -= 1
        if lp_countdown <= 0:
            M = fast_convolve(p).max()
            b = solve_lp_direction(p, M)
            if b is not None and b.sum() > 1e-10:
                b = b / b.sum()
                lp_direction = b
            lp_countdown = lp_interval
            if verbose:
                print(f"    LP solved at t={time.monotonic()-t0:.1f}s, M={M:.8f}")
        
        # Adam step
        conv = fast_convolve(p)
        w = softmax_weights(conv, temp)
        sg = compute_gradient(p, w)
        
        # If we have an LP direction, blend the gradient with it
        if lp_direction is not None:
            # The LP direction suggests where to go
            # Use it as a "prior" by computing gradient of distance to lp_direction
            lp_grad = -2 * (lp_direction - p)
            sg = 0.5 * sg + 0.5 * lp_grad
        
        adam_t += 1
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** adam_t)
        v_hat = v / (1 - 0.999 ** adam_t)
        
        p_new = p - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        p = project_simplex(p_new)
        
        temp = max(temp * 0.9997, 1e-7)
        cur_lr = max(0.004 * (0.99995 ** adam_t), 0.0002)
        
        conv = fast_convolve(p)
        cur_M = conv.max()
        
        if cur_M < best_M - 1e-15:
            best_M = cur_M
            best_p = p.copy()
            no_improve = 0
            if verbose and no_improve == 0 and int(time.monotonic()-t0) % 30 == 0:
                print(f"    lp_guided: M={best_M:.10f} at t={time.monotonic()-t0:.1f}s")
        else:
            no_improve += 1
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            p = project_simplex(best_p + rng.standard_normal(n) * 0.008)
            m = np.zeros(n)
            v = np.zeros(n)
            adam_t = 0
            temp = 0.004
            cur_lr = 0.004
            no_improve = 0
            lp_countdown = 0  # force LP solve
    
    return best_p, best_M


def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = f0.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

Wait, I have a potential issue. The `lp_guided_optimize` function uses `solve_lp_direction` which builds a matrix of size (2n-1) x n. For n=768, that's 1535 x 768 ≈ 1.18M entries. This should be fine for memory and speed.

But the LP itself might take a while. Let me estimate: with 768 variables and 1535 inequality constraints + 768 bound constraints, HiGHS should solve this in well under a second. Let me add a time check.

Also, I want to make sure the LP is being called efficiently. In my `lp_guided_optimize`, I call it every 5 iterations. Each Adam iteration is fast (just a few FFT operations), so 5 iterations might take < 0.1s. The LP might take 0.1-1s. So the overhead is manageable.

Let me also think about whether the "blending" of LP direction and gradient makes sense. The LP direction is a point (a probability vector), not a direction. So I'm computing `lp_grad = -2*(lp_direction - p)` which is the negative gradient of `||p - lp_direction||^2/2`. This encourages p to move toward the LP direction. Combined with the original gradient (which pushes to reduce max convolution), this should work.

But actually, I think a better approach might be:

```python
def lp_guided_optimize(p0, time_limit, verbose=False):
    """LP-guided optimization: alternate between LP direction and Adam."""
    n = len(p0)
    p = p0.copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_M = fast_convolve(p).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    temp = 0.005
    cur_lr = 0.004
    
    no_improve = 0
    iteration = 0
    lp_step_size = 0.3
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Every 10 iterations, solve LP and try direct step
        if iteration % 10 == 1:
            M = fast_convolve(p).max()
            b = solve_lp_direction(p, M)
            if b is not None and b.sum() > 1e-10:
                b = b / b.sum()
                # Try mixing
                for t in [lp_step_size, lp_step_size*0.5, lp_step_size*0.25]:
                    p_trial = (1-t)*p + t*b
                    M_trial = fast_convolve(p_trial).max()
                    if M_trial < best_M - 1e-15:
                        p = p_trial
                        best_M = M_trial
                        best_p = p.copy()
                        no_improve = 0
                        lp_step_size = min(lp_step_size * 1.05, 0.7)
                        if verbose:
                            print(f"    LP step (t={t:.2f}): M={best_M:.10f} at {time.monotonic()-t0:.1f}s")
                        break
                else:
                    # No improvement from LP step
                    no_improve += 5
                    lp_step_size = max(lp_step_size * 0.8, 0.05)
        
        # Adam step
        conv = fast_convolve(p)
        w = softmax_weights(conv, temp)
        sg = compute_gradient(p, w)
        
        adam_t += 1
        m = 0.9 * m + 0.1 * sg
        v = 0.999 * v + 0.001 * sg * sg
        m_hat = m / (1 - 0.9 ** adam_t)
        v_hat = v / (1 - 0.999 ** adam_t)
        
        p_new = p - cur_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        p = project_simplex(p_new)
        
        temp = max(temp * 0.9998, 1e-7)
        cur_lr = max(0.004 * (0.99997 ** adam_t), 0.0003)
        
        conv = fast_convolve(p)
        cur_M = conv.max()
        
        if cur_M < best_M - 1e-15:
            best_M = cur_M
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng_local = np.random.default_rng(iteration * 31 + 7)
            p = project_simplex(best_p + rng_local.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            adam_t = 0
            temp = 0.004
            cur_lr = 0.004
            no_improve = 0
            lp_step_size = 0.3
    
    return best_p, best_M
```
