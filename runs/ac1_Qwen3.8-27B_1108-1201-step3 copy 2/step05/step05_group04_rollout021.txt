def sca_iterate(f_t, n):
    """One SCA iteration: solve the LP relaxation."""
    b_t = fast_convolve(f_t)  # length 2n-1
    M_t = b_t.max()
    
    # LP: minimize t
    # subject to: 2*(f_t * f)_k - b_t[k] <= t for k=0,...,2n-2
    #             sum(f) = 1
    #             f >= 0
    #             t >= 0 (optional, but t should be positive)
    
    # Variables: [f_0, ..., f_{n-1}, t]
    # Objective: minimize t => c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: 2*(f_t * f)_k - b_t[k] - t <= 0 for each k
    # (f_t * f)_k = sum_j f_t[k-j] * f_j (for valid j)
    # So constraint k: sum_j 2*f_t[k-j] * f_j - t <= b_t[k]
    
    # A_ub has shape (2n-1, n+1)
    # For row k: [2*f_t[n-1-k], 2*f_t[n-2-k], ..., 2*f_t[0], -1] (reversed f_t, padded)
    
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # (f_t * f)_k = sum_j f_t[k-j] * f_j where 0<=j<n and 0<=k-j<n
        # So the coefficient of f_j in constraint k is 2*f_t[k-j] (if 0<=k-j<n)
        for j in range(n):
            if 0 <= k-j < n:
                A_ub[k, j] = 2.0 * f_t[k-j]
        A_ub[k, -1] = -1.0  # coefficient of t
        b_ub[k] = b_t[k]
    
    # Equality constraint: sum(f) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: f_j >= 0, t >= 0 (or t unbounded below? Let's use t >= 0 for safety)
    bounds = [(0, None)] * n + [(None, None)]  # t can be anything
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return f_t.copy(), M_t

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using SCA (Successive Convex Approximation)
with LP subproblems, combined with correct subgradient methods, mirror descent,
and coordinate descent polishing. Multi-start with diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """
    Correct subgradient of max(conv(f,f)) w.r.t. f.
    w is the soft-max weight over convolution lags (length 2n-1).
    Returns array of length n.
    """
    n = len(f)
    # sg[j] = 2 * sum_k w[k] * f[k-j] (for valid k-j)
    # = 2 * correlate(w, f, 'full')[j]
    return 2.0 * np.correlate(w, f, 'full')[:n]


def get_soft_max_weights(conv, temp, top_k=None):
    """Get soft-max weights over convolution values."""
    shifted = conv / temp
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    w /= ws
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(len(conv))
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    return w


def sca_solve_lp(f_t):
    """
    Solve one SCA LP iteration.
    Returns (f_lp, t_lp) or (f_t, inf) if LP fails.
    """
    n = len(f_t)
    b_t = fast_convolve(f_t)
    conv_len = 2 * n - 1
    
    # LP: minimize t
    # subject to: 2*(f_t * f)_k - b_t[k] <= t for k=0,...,2n-2
    #             sum(f) = 1
    #             f >= 0
    # Variables: [f_0,...,f_{n-1}, t]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Build constraint matrix
    # A_ub[k, j] = 2 * f_t[k-j] if 0 <= k-j < n, else 0
    # A_ub[k, -1] = -1
    # b_ub[k] = b_t[k]
    
    A_ub = np.zeros((conv_len, n + 1))
    for k in range(conv_len):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = 2.0 * f_t[k - j]
        A_ub[k, -1] = -1.0
    b_ub = b_t.copy()
    
    # Equality: sum(f) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n].copy(), result.x[n]
    return f_t.copy(), b_t.max()


def sca_optimize(f0, time_limit, alpha=1.0, verbose=False):
    """
    SCA optimization: iteratively solve LP and update.
    f is on the simplex (sum=1, f>=0).
    """
    f = f0.copy()
    f = project_simplex(f)
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        f_lp, t_lp = sca_solve_lp(f)
        
        # Damped update
        f_new = (1 - alpha) * f + alpha * f_lp
        f_new = project_simplex(f_new)
        
        cur_obj = fast_convolve(f_new).max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    SCA iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # If no improvement, try smaller alpha
            if no_improve > 10:
                alpha = max(alpha * 0.9, 0.1)
            if no_improve > 50:
                # Restart from best
                f = best_f.copy()
                alpha = 1.0
                no_improve = 0
        
        f = f_new
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.01, temp_init=0.005,
                         temp_decay=0.999, top_k=None, verbose=False):
    """Subgradient method with CORRECT gradient and Polyak-like step size."""
    f = project_simplex(f0.copy())
    n = len(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    
    # Running average for Polyak step
    obj_history = []
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)
        
        # Adam-style update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        cur_lr = lr * min(1.0, 100 / t)
        
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)
        
        temp = max(temp * temp_decay, 1e-8)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(t * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.5, 0.002)
            no_improve = 0
    
    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.02, lr_decay=0.9995,
                            temp_init=0.005, temp_decay=0.999, top_k=None):
    """Mirror descent (multiplicative weights) with correct subgradient."""
    f = project_simplex(f0.copy())
    n = len(f)
    
    temp = temp_init
    cur_lr = lr_init
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        w = get_soft_max_weights(conv, temp, top_k)
        sg = compute_subgradient(f, w)
        
        # Multiplicative update
        log_update = -cur_lr * sg
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs
        f = f_new
        
        temp = max(temp * temp_decay, 1e-8)
        cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            rng = np.random.default_rng(t * 41 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            temp = max(temp_init * 0.5, 0.002)
            cur_lr = lr_init
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent: try scaling each coordinate."""
    f = project_simplex(f0.copy())
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.7]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-10:
                continue
            
            old_val = f[j]
            best_j_obj = best_obj  # current global best
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                
                # Scale others to maintain sum=1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                fs = f_trial.sum()
                if fs < 1e-10:
                    continue
                f_trial /= fs
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f = f / f.sum()
                improved = True
                if best_j_obj < best_obj:
                    best_obj = best_j_obj
                    best_f = f.copy()
        
        if not improved:
            break
    
    return best_f, best_obj


def make_asymmetric_init(n, rng, n_components=4):
    """Create an asymmetric random initialization."""
    x = np.linspace(0, 1, n)
    f = np.zeros(n)
    for _ in range(n_components):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.05, 0.3)
        a = rng.uniform(0.5, 2.0)
        f += a * np.exp(-0.5 * ((x - c) / s) ** 2)
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def make_bump_init(n, center=0.5, sigma=0.2, asymmetry=0.0):
    """Create a bump initialization with optional asymmetry."""
    x = np.linspace(0, 1, n)
    # Asymmetric Gaussian
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    if asymmetry != 0:
        f *= np.exp(asymmetry * (x - center))
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            base = best_seq if best_seq is not None else base
    
    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror descent (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    t_phase3 = min(150, (deadline - time.monotonic()) * 0.1)
    
    if t_phase3 > 15:
        configs = [
            {'lr_init': 0.02, 'temp_init': 0.004, 'top_k': max(20, n_base // 10)},
            {'lr_init': 0.04, 'temp_init': 0.008, 'top_k': None},
            {'lr_init': 0.01, 'temp_init': 0.003, 'top_k': max(15, n_base // 15)},
        ]
        for ci, cfg in enumerate(configs):
            rem = t_phase3 / len(configs)
            if rem < 10:
                break
            print(f"  Mirror config {ci}, t={rem:.0f}s")
            a_opt, _ = mirror_descent_optimize(base, time_limit=rem, **cfg)
            try_seq(a_opt, f"p3_md_{ci}")
            base = best_seq if best_seq is not None else base
    
    # === Phase 4: SCA from diverse initializations ===
    print(f"\n=== Phase 4: SCA diverse (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + min(200, (deadline - time.monotonic()) * 0.15)
    
    n_try = n_parent
    inits = [
        ('asym1', make_asymmetric_init(n_try, rng, 4)),
        ('bump_left', make_bump_init(n_try, 0.35, 0.2, asymmetry=-2.0)),
        ('bump_right', make_bump_init(n_try, 0.65, 0.2, asymmetry=2.0)),
        ('two_peak', 0.5 * make_bump_init(n_try, 0.3, 0.15) + 0.5 * make_bump_init(n_try, 0.7, 0.15)),
    ]
    
    for name, init in inits:
        rem = t_phase4_end - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.4, 40)
        print(f"  SCA '{name}', t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(init, time_limit=t_alloc, alpha=1.0)
        try_seq(a_opt, f"p4_sca_{name}")
        # Follow with subgradient
        rem = t_phase4_end - time.monotonic()
        if rem > 15:
            a_opt, _ = subgradient_optimize(a_opt, time_limit=min(rem * 0.5, 20),
                                             lr=0.005, temp_init=0.004, top_k=max(20, n_try//10))
            try_seq(a_opt, f"p4_sg_{name}")
    
    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    t_phase5_end = time.monotonic() + min(200, (deadline - time.monotonic()) * 0.15)
    
    base = best_seq if best_seq is not None else parent
    target_lengths = [384, 512, 640, 896, 1024, 1280]
    target_lengths = [t for t in target_lengths if t != len(base) and 128 <= t <= 2048]
    target_lengths.sort(key=lambda x: abs(x - len(base)))
    target_lengths = target_lengths[:3]
    
    for target_n in target_lengths:
        rem = t_phase5_end - time.monotonic()
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 50)
        print(f"  SCA n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(f_interp, time_limit=t_alloc, alpha=1.0)
        try_seq(a_opt, f"p5_n{target_n}")
        
        rem = t_phase5_end - time.monotonic()
        if rem > 15:
            a_opt, _ = subgradient_optimize(a_opt, time_limit=min(rem * 0.5, 20),
                                             lr=0.005, temp_init=0.004, top_k=max(20, target_n//10))
            try_seq(a_opt, f"p5_sg_n{target_n}")
    
    # === Phase 6: Deep SCA on best ===
    print(f"\n=== Phase 6: Deep SCA (best={best_score:.8f}) ===")
    t_phase6_end = time.monotonic() + min(150, (deadline - time.monotonic()) * 0.12)
    
    if best_seq is not None:
        # Multiple SCA runs with different alpha
        for alpha in [1.0, 0.7, 0.5]:
            rem = t_phase6_end - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem * 0.5, 40)
            print(f"  SCA alpha={alpha}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, alpha=alpha)
            try_seq(a_opt, f"p6_sca_a{alpha}")
    
    # === Phase 7: Coordinate descent polishing ===
    print(f"\n=== Phase 7: CD polish (best={best_score:.8f}) ===")
    t_phase7_end = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.1)
    
    if best_seq is not None:
        rem = t_phase7_end - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.8, 90)
            print(f"  CD, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "p7_cd")
            
            # CD from perturbed
            rem = t_phase7_end - time.monotonic()
            if rem > 10 and best_seq is not None:
                n_best = len(best_seq)
                perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                a_opt, _ = coordinate_descent_polish(perturbed, time_limit=min(rem * 0.7, 30))
                try_seq(a_opt, "p7_cd_pert")
    
    # === Phase 8: Final fine-tuning ===
    print(f"\n=== Phase 8: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            # SCA with small alpha for fine-tuning
            t_alloc = min(rem * 0.4, 40)
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, alpha=0.3)
            try_seq(a_opt, "p8_fine_sca")
            
            rem = deadline - time.monotonic() - 8
            if rem > 15:
                a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem * 0.5, 20),
                                                 lr=0.002, temp_init=0.002, top_k=max(10, len(best_seq)//20))
                try_seq(a_opt, "p8_fine_sg")
            
            # CD polish again
            rem = deadline - time.monotonic() - 5
