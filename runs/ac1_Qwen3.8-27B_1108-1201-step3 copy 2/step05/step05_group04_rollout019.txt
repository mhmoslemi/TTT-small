    n = len(a)

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):

Now, let me also think about a completely different approach: **iterative proportional fitting (IPF)** or **Iterated Conditional Modes (ICM)**.

The idea: we want a such that a*a ≈ c (constant). This is equivalent to: for each k, sum_i a[i]*a[k-i] = c.

This is a system of quadratic equations. We can solve it iteratively:
1. Start with a initial guess.
2. For each k, adjust a to make (a*a)[k] closer to c.
3. Repeat.

The adjustment for a given k: we want to reduce (a*a)[k] = sum_i a[i]*a[k-i]. The most effective way is to reduce the products where both a[i] and a[k-i] are large. This suggests a "decrease the large values in the pairs that sum to k" operation.

This is similar to the "Sinkhorn-Knopp" algorithm for scaling matrices to have given row and column sums. Here, we're "scaling" the quadratic form to have given "diagonal" sums.

Let me formalize: at each step, for the lag k where (a*a)[k] is the maximum, we want to reduce it. The gradient descent step in the log-domain (multiplicative update) would be:
a[i] ← a[i] * exp(-η * ∂(a*a)[k]/∂a[i]) = a[i] * exp(-2η * a[k-i])

This is the mirror descent step with KL divergence. Let me implement this.

the non-negativity constraint naturally, and diverse initializations (including
U-shaped and edge-concentrated) explore the non-convex landscape.
"""
import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplicityx {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_conv_matrix(a):
    """Build matrix C where (C @ b)[k] = (a*b_conv)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def subgradient(a, w):
    """Compute subgradient: sg[j] = sum_i w[i+j] * a[i]."""
    n = len(a)
    return np.correlate(w, a, 'valid')


def lp_optimize(a_init, time_limit, t_mix=0.3, verbose=False):
    """LP-based optimization following the SOTA approach."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, fast_convolve(a_init).max()
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP: max sum(b) s.t. (a*b) <= M, b >= 0
        C = make_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            # LP failed, try smaller M
            M *= 0.9
            continue
            
        b = res.x
        b_sum = b.sum()
        
        if b_sum < 1e-10:
            break
        
        g = b / b_sum  # Normalize to sum = 1
        
        # Mix and check improvement
        if no_improve < 50:
            t = t_mix
        else:
            t = max(0.1, t_mix * (50 / no_improve))
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        a = a_new
        conv_new = fast_convolve(a)
        obj_new = conv_new.max()
        
        iter_count += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter {iter_count}: obj={obj_new:.10f}, t={elapsed:.1f}s")
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and restart
                rng = np.random.default_rng(iter_count)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
    
    return best_a, best_obj


def mirror_descent_optimize(a_init, time_limit, lr_init=0.02, temp_init=0.008,
                            temp_decay=0.9995, verbose=False):
    """Multiplicative weights (mirror descent) with soft-max subgradients."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, fast_convolve(a_init).max()
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    temp = temp_init
    lr = lr_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Subgradient
        sg = subgradient(a, w)
        
        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        # Update parameters
        temp = max(temp * temp_decay, 1e-7)
        lr = max(lr * 0.9998, lr_init * 0.05)
        
        conv = fast_convolve(a)
        obj = conv.max()
        iter_count += 1
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and iter_count % 100 == 0:
                print(f"    Mirror iter {iter_count}: obj={obj:.10f}, t={time.monotonic()-t0:.1f}s")
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(iter_count * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            temp = temp_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def subgradient_adam_optimize(a_init, time_limit, lr=0.005, temp_init=0.006,
                              temp_decay=0.9994, verbose=False):
    """Adam optimizer with simplex projection and soft-max subgradients."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, fast_convolve(a_init).max()
    a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        
        # Soft-max weights
        shifted = conv / temp
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Subgradient
        sg = subgradient(a, w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)
        
        # Update temperature and lr
        temp = max(temp * temp_decay, 1e-7)
        cur_lr = lr * max(0.1, min(1.0, 5000 / t))
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            rng = np.random.default_rng(t * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = max(temp_init * 0.8, 0.003)
            no_improve = 0
    
    return best_a, best_obj
