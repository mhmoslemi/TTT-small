"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Frank-Wolfe iterations with LP-guided directions (breaks palindrome symmetry)
2. Log-softmax L-BFGS with temperature annealing for smooth local refinement
3. Coordinate descent for final polish
4. Multi-length exploration with diverse asymmetric initializations
Key insight: non-palindromic sequences can achieve R < 2 by making the center
autocorrelation small while balancing all lags.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value directly (scale-invariant)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return float(2.0 * len(a_norm) * b.max())


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_step(z0, T, n, time_limit, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max objective."""
    t0 = time.monotonic()
    expired = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            expired[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            expired[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 25, 'disp': False}
        )
        if expired[0]:
            return z0, func(z0)
        return res.x, res.fun
    except Exception:
        return z0, func(z0)


def solve_lp_direction(f, M, max_time=30.0):
    """
    Solve the LP to find Frank-Wolfe direction.
    Maximize sum(g) subject to ||f*g||_inf <= M, g >= 0.
    Returns normalized g (sum=1) or None if LP fails.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp

    n = len(f)
    n_cons = 2 * n - 1

    t0 = time.monotonic()

    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build row by row
    rows = []
    cols = []
    vals = []

    for k in range(n_cons):
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            if 0 <= i < n and f[i] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])

    if len(rows) == 0:
        return None

    A_sparse = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))

    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(
            c, A_ub=A_sparse, b_ub=b_ub,
            bounds=bounds, method='highs',
            options={'time_limit': max_time, 'presolve': True}
        )
        if time.monotonic() - t0 > max_time:
            return None
        if res.success and res.fun > -1e-10:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass

    return None


def frank_wolfe_iterate(f0, time_limit=120.0, verbose=True):
    """
    Frank-Wolfe iterations with LP-guided directions.
    f0: normalized probability vector (sum=1, non-negative)
    Returns best f found and its score.
    """
    f = f0.copy()
    n = len(f)
    M = fast_convolve(f).max()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        M = fast_convolve(f).max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_f = f.copy()
            if verbose and iteration % 5 == 0:
                print(f"    FW iter {iteration}: score={score:.6f}")

        # Solve LP for direction
        g = solve_lp_direction(f, M, max_time=min(20.0, time_limit - (time.monotonic() - t0)))
        if g is None:
            # Fallback: use uniform direction
            g = np.ones(n) / n

        # Line search for step size
        best_t = 0.0
        best_trial_score = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            fs = f_trial.sum()
            if fs < 1e-10:
                continue
            f_trial = f_trial / fs
            trial_score = compute_score(f_trial)
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_t = t

        if best_t < 0.01:
            # No improvement from FW direction, try a small random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            f_trial = f + rng.standard_normal(n) * 0.01
            f_trial = project_simplex(f_trial)
            trial_score = compute_score(f_trial)
            if trial_score < best_score:
                f = f_trial
                best_score = trial_score
                best_f = f.copy()
            else:
                break
        else:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()

        # Decay step sizes over iterations
        if iteration > 50:
            break

    return best_f, best_score


def lbgfs_anneal(a0, time_limit, max_T=200):
    """L-BFGS with temperature annealing from a probability vector."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()

    # Temperature schedule: start high (smooth), decrease (sharper)
    temps = [5, 10, 20, 50, 100, 200]
    # Filter to those within time budget
    per_T = min(60.0, (time_limit - 5.0) / len(temps))

    best_a = a0.copy()
    best_score = compute_score(a0)

    for T in temps:
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_alloc = min(per_T, time_limit - (time.monotonic() - t0) - 5)
        if t_alloc < 5:
            break

        z, _ = lbgfs_step(z, T, n, t_alloc, maxiter=2000)
        a_curr = softmax_from_z(z)
        score = compute_score(a_curr)
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
            print(f"    L-BFGS T={T}: score={score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit=30.0):
    """Coordinate descent on probability vector."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_score = best_score
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()

                val = 2.0 * n * fast_convolve(a_trial).max()
                if val < best_j_score - 1e-14:
                    best_j_score = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_score < best_score - 1e-14:
                    best_score = best_j_score
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_score


def subgradient_polish(a0, time_limit=45.0, lr=0.003, mom=0.9, top_k=50):
    """Subgradient method with softmax-weighted active peaks."""
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = 0.001

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        b = fast_convolve(a)
        co = 2.0 * n * b.max()

        if co < best_score - 1e-14:
            best_score = co
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.01

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse asymmetric initializations of length n."""
    inits = []

    # 1. Uniform (baseline, R=2)
    inits.append(('uniform', np.ones(n) / n))

    # 2. Linear ramp (strongly asymmetric)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))

    # 3. Reverse ramp
    inits.append(('rev_ramp', ramp[::-1] / ramp.sum()))

    # 4. Two-block (first half high, second half low)
    two_block = np.ones(n)
    two_block[:n // 2] = 2.0
    inits.append(('two_block', two_block / two_block.sum()))

    # 5. Sine perturbation (non-palindromic)
    x = np.linspace(0, 1, n)
    sine = 1.0 + 0.8 * np.sin(2 * np.pi * x + 0.5)
    sine = np.maximum(sine, 0.01)
    inits.append(('sine', sine / sine.sum()))

    # 6. Cosine half-wave (asymmetric)
    cos_init = 0.5 + 0.5 * np.cos(np.pi * x)
    cos_init = np.maximum(cos_init, 0.01)
    inits.append(('cos_asym', cos_init / cos_init.sum()))

    # 7. Random with monotone bias
    base = rng.random(n)
    bias = np.linspace(0.5, 1.5, n)
    rand_biased = base * bias
    inits.append(('rand_bias', rand_biased / rand_biased.sum()))

    # 8. Exponential decay
    exp_init = np.exp(-np.linspace(0, 2, n))
    inits.append(('exp_decay', exp_init / exp_init.sum()))

    # 9. Triangle (asymmetric peak)
    peak_pos = n // 3
    tri = np.zeros(n)
    for i in range(n):
        tri[i] = 1.0 - abs(i - peak_pos) / max(peak_pos, n - peak_pos - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(('tri_asym', tri / tri.sum()))

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    Phase 3: Multi-length exploration (20% time)
    Phase 4: Deep L-BFGS refinement (20% time)
    Phase 5: Coordinate descent + subgradient polish (10% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Diverse initializations + quick L-BFGS ===
    print("\n=== Phase 1: Diverse initializations ===", flush=True)
    n_base = len(parent)
    t1_budget = min(270.0, (deadline - time.monotonic()) * 0.15)

    candidates = []

    # Try parent first
    try_seq(parent, "parent")
    candidates.append(parent.copy())

    # Generate diverse initializations at parent length
    inits = make_initializations(n_base, rng)
    for name, init_a in inits:
        score = try_seq(init_a, f"init_{name}_n{n_base}")
        if np.isfinite(score):
            candidates.append(init_a.copy())

    # Quick L-BFGS on best candidates
    top_candidates = sorted(candidates, key=lambda x: compute_score(x))[:4]
    per_init_time = t1_budget / max(1, len(top_candidates))

    for i, cand in enumerate(top_candidates):
        rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.7
        if rem < 10:
            break
        t_alloc = min(per_init_time, rem * 0.4)
        if t_alloc < 5:
            break
        a_opt, _ = lbgfs_anneal(cand, t_alloc, max_T=100)
        try_seq(a_opt, f"lbgfs_init{i}")
        if best_seq is not None:
            candidates.append(best_seq.copy())

    # === Phase 2: Frank-Wolfe LP-guided iterations ===
    print(f"\n=== Phase 2: Frank-Wolfe (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        t2_budget = min(630.0, (deadline - time.monotonic()) * 0.35)
        rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.4
        t_alloc = min(t2_budget, rem)
        if t_alloc > 20:
            print(f"  Running FW for {t_alloc:.0f}s on n={len(best_seq)}", flush=True)
            a_fw, _ = frank_wolfe_iterate(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_fw, "frank_wolfe")

            # Also try FW from a few other initializations
            for name, init_a in inits[:3]:
                rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.35
                if rem < 15:
                    break
                t_fw = min(30.0, rem * 0.3)
                if t_fw < 10:
                    break
                a_fw2, _ = frank_wolfe_iterate(init_a, time_limit=t_fw, verbose=False)
                try_seq(a_fw2, f"fw_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        base = best_seq
        target_lengths = [512, 768, 1024, 1536, 2048, 3072]
        # Sort by proximity to current best length
        target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - len(base)))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.3
            if rem < 30:
                break
            if target_n == len(base):
                continue

            f_interp = interpolate_sequence(base, target_n)
            # Also add asymmetric perturbation
            rng2 = np.random.default_rng(target_n + seed)
            f_interp = f_interp + rng2.standard_normal(target_n) * 0.02
            f_interp = project_simplex(f_interp)

            t_alloc = min(60.0, rem * 0.2)
            if t_alloc < 10:
                continue

            print(f"  Explored n={target_n}, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = lbgfs_anneal(f_interp, t_alloc, max_T=100)
            try_seq(a_opt, f"explore_n{target_n}")

            # Quick FW on this length
            rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.25
            if rem > 20 and best_seq is not None:
                t_fw = min(30.0, rem * 0.3)
                a_fw, _ = frank_wolfe_iterate(a_opt, time_limit=t_fw, verbose=False)
                try_seq(a_fw, f"fw_n{target_n}")

    # === Phase 4: Deep L-BFGS refinement ===
    print(f"\n=== Phase 4: Deep refinement (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        t4_budget = min(360.0, (deadline - time.monotonic()) * 0.2)

        # High-precision L-BFGS with very high T (near-exact max)
        z0 = z_from_a(best_seq)
        temps_deep = [50, 100, 200, 500]
        per_T_deep = t4_budget / len(temps_deep)

        z_curr = z0
        for T in temps_deep:
            rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.15
            if rem < 10:
                break
            t_alloc = min(per_T_deep, rem * 0.3)
            if t_alloc < 5:
                break
            print(f"  Deep L-BFGS n={n_best}, T={T}, t={t_alloc:.0f}s", flush=True)
            z_curr, _ = lbgfs_step(z_curr, T, n_best, t_alloc, maxiter=3000)
            a_curr = softmax_from_z(z_curr)
            try_seq(a_curr, f"deep_T{T}")

        # Try perturbation + re-optimize
        rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.1
        if rem > 20:
            for attempt in range(3):
                rem = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.05
                if rem < 10:
                    break
                noise_scale = 0.01 / (attempt + 1)
                noise = rng.standard_normal(n_best) * noise_scale
                a_pert = project_simplex(best_seq + noise)
                z_pert = z_from_a(a_pert)
                t_alloc = min(30.0, rem * 0.3)
                z_p, _ = lbgfs_step(z_pert, 200, n_best, t_alloc, maxiter=2000)
                a_p = softmax_from_z(z_p)
                try_seq(a_p, f"perturb_{attempt}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(45.0, rem * 0.4)
            print(f"  Coordinate descent, t={t_cd:.0f}s", flush=True)
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "coord_descent")

        # Subgradient polish
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_sg = min(45.0, rem * 0.5)
            print(f"  Subgradient polish, t={t_sg:.0f}s", flush=True)
            a_sg, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                         lr=0.002, mom=0.92,
                                         top_k=max(20, n_best // 10))
            try_seq(a_sg, "sg_polish")

        # One more round of FW + L-BFGS if time permits
        rem = deadline - time.monotonic() - 15
        if rem > 15 and best_seq is not None:
            t_fw_final = min(25.0, rem * 0.3)
            a_fw_f, _ = frank_wolfe_iterate(best_seq, time_limit=t_fw_final, verbose=False)
            try_seq(a_fw_f, "fw_final")

            rem = deadline - time.monotonic() - 10
            if rem > 10 and best_seq is not None:
                z_final = z_from_a(best_seq)
                z_final, _ = lbgfs_step(z_final, 500, len(best_seq), min(20.0, rem * 0.5), maxiter=2000)
                a_final = softmax_from_z(z_final)
                try_seq(a_final, "final_lbgfs")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
