def project_simplex(v):

            if opt_obj < best_obj - 1e-15:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - opt_val) / denom
                    f[j] = opt_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                best_obj = opt_obj
                best_f = f.copy()
                improved = True

        if not improved and passes > 2:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.12f}")

    return best_f, best_obj


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and warm restart."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Build LP constraint matrix
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                m_exp *= 1.2
                if m_exp > 5.0:
                    break
                continue
            g = res.x
        except Exception:
            m_exp *= 1.2
            if m_exp > 5.0:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine
        if best_alpha > 0:
            lo_a = max(0.0, best_alpha - 0.1)
            hi_a = min(1.0, best_alpha + 0.1)
            for _ in range(15):
                mid = (lo_a + hi_a) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo_a = mid
                else:
                    hi_a = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 80:
            rng_p = np.random.default_rng(it * 17 + 3)
            scale = 0.002 + 0.003 * min(no_improve // 50, 4)
            f = project_simplex(best_f + rng_p.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Linear interpolation to new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_init(n, seed=0):
    """Create diverse initial sequences for exploration."""
    rng = np.random.default_rng(seed)
    inits = []

    # Geometric
    for r in [0.97, 0.985, 0.995]:
        a = r ** np.arange(n, dtype=np.float64)
        inits.append(a / a.sum())

    # Cosine powers
    x = np.linspace(0, np.pi, n)
    for pw in [1.0, 2.0, 3.0]:
        a = np.sin(x) ** pw + 0.01
        inits.append(a / a.sum())

    # Random smooth
    for _ in range(3):
        base = rng.standard_normal(n)
        # Smooth with a moving average
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        base = np.convolve(base, kernel, mode='same')
        a = np.exp(base)
        inits.append(a / a.sum())

    # Symmetric geometric
    half = (n + 1) // 2
    for r in [0.96, 0.98]:
        asc = r ** np.arange(half, dtype=np.float64)
        desc = r ** np.arange(half - 1, 0, -1, dtype=np.float64)
        a = np.concatenate([asc, desc])
        if len(a) < n:
            a = np.concatenate([a, np.ones(n - len(a))])
        inits.append(a[:n] / a[:n].sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method fine optimizer combining smooth Adam, entropy descent,
    LP-Frank-Wolfe, and coordinate descent with aggressive hyperparameter
    tuning. Designed to improve upon a near-optimal starting sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
    except Exception as e:
        print(f"Parent loading failed: {e}")

    n_parent = len(parent) if parent is not None else 768

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            old_score = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) [was {old_score:.8f}]")

    # === Phase 1: Smooth Adam on parent with aggressive cooling ===
    print("=== Phase 1: Smooth Adam (aggressive) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 30 and parent is not None:
        t_alloc = min(rem, 400)
        print(f"  n={len(parent)}, t={t_alloc:.0f}s, T: 0.05 -> 0.0005")
        a_opt, _ = adam_smooth_optimize(parent, time_limit=t_alloc,
                                         T_start=0.05, T_end=0.0005, verbose=True)
        _try(a_opt, "adam_aggressive")

    # === Phase 2: Entropy descent ===
    print("=== Phase 2: Entropy Descent ===")
    base = best_seq.copy() if best_seq is not None else parent
    rem = deadline - time.monotonic() - 600
    if rem > 30 and base is not None:
        t_alloc = min(rem, 350)
        print(f"  n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = entropy_descent(base, time_limit=t_alloc, verbose=True)
        _try(a_opt, "entropy_p2")

    # === Phase 3: LP-FW on best ===
    print("=== Phase 3: LP-Frank-Wolfe ===")
    base = best_seq.copy() if best_seq is not None else parent
    rem = deadline - time.monotonic() - 400
    if rem > 20 and base is not None:
        t_alloc = min(rem, 250)
        print(f"  n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(base, time_limit=t_alloc, verbose=True)
        _try(a_opt, "lpfw_p3")

    # === Phase 4: Fine coordinate descent ===
    print("=== Phase 4: Fine Single CD ===")
    base = best_seq.copy() if best_seq is not None else parent
    rem = deadline - time.monotonic() - 250
    if rem > 15 and base is not None:
        t_alloc = min(rem, 200)
        print(f"  n={len(base)}, t={t_alloc:.0f}s")
        a_opt, _ = fine_single_cd(base, time_limit=t_alloc, verbose=True)
        _try(a_opt, "fine_cd")

    # === Phase 5: Second round of smooth + entropy on best ===
    print("=== Phase 5: Second Round Polish ===")
    base = best_seq.copy() if best_seq is not None else parent
    rem = deadline - time.monotonic() - 150
    if rem > 20 and base is not None:
        t_alloc = min(rem * 0.5, 100)
        a_opt, _ = adam_smooth_optimize(base, time_limit=t_alloc,
                                         T_start=0.02, T_end=0.0003, verbose=True)
        _try(a_opt, "adam_p5")

    rem = deadline - time.monotonic() - 100
    if rem > 15 and best_seq is not None:
        t_alloc = min(rem * 0.5, 80)
        a_opt, _ = entropy_descent(best_seq, time_limit=t_alloc)
        _try(a_opt, "entropy_p5")

    # === Phase 6: Try different lengths ===
    print("=== Phase 6: Length Exploration ===")
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 80
        if rem < 40 or best_seq is None:
            break
        f_interp = interpolate_seq(best_seq, target_n)

        t1 = min(rem * 0.4, 60)
        a_opt, _ = adam_smooth_optimize(f_interp, time_limit=t1,
                                         T_start=0.1, T_end=0.005)
        _try(a_opt, f"length_{target_n}")

        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t2 = min(rem * 0.5, 50)
            a_opt, _ = entropy_descent(f_interp, time_limit=t2)
            _try(a_opt, f"entropy_{target_n}")

    # === Phase 7: Perturbation-based exploration ===
    print("=== Phase 7: Perturbation Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for trial in range(5):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            rng_p = np.random.default_rng(seed + 1000 + trial * 7)
            # Different perturbation strategies
            if trial == 0:
                noise = rng_p.standard_normal(n_best) * 0.003
                f_pert = project_simplex(best_seq + noise)
            elif trial == 1:
                # Remove a random few elements
                f_pert = best_seq.copy()
                mask = rng_p.choice(n_best, size=max(3, n_best // 100), replace=False)
                f_pert[mask] = 0
                f_pert = project_simplex(f_pert)
            elif trial == 2:
                # Blend with uniform
                f_pert = 0.95 * best_seq + 0.05 * (np.ones(n_best) / n_best)
                f_pert = project_simplex(f_pert)
            elif trial == 3:
                # Reverse
                f_pert = best_seq[::-1].copy()
            else:
                # Local perturbation around peak
                R = fast_convolve(best_seq)
                peak_idx = np.argmax(R)
                # Perturb elements that contribute to the peak
                contrib = np.zeros(n_best)
                lo = max(0, peak_idx - n_best + 1)
                hi = min(n_best - 1, peak_idx)
                contrib[lo:hi+1] = best_seq[peak_idx - hi:peak_idx - lo + 1]
                noise = rng_p.standard_normal(n_best) * 0.002
                noise[lo:hi+1] *= 2.0  # more noise where it matters
                f_pert = project_simplex(best_seq + noise)

            t_alloc = min(rem * 0.6, 30)
            a_opt, _ = adam_smooth_optimize(f_pert, time_limit=t_alloc,
                                             T_start=0.02, T_end=0.001)
            _try(a_opt, f"perturb_{trial}")

    # === Phase 8: Final ultra-fine polish ===
    print("=== Phase 8: Final Ultra-Fine Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem, 40)
            a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_alloc,
                                             T_start=0.005, T_end=0.0001)
            _try(a_opt, "ultra_fine")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 20)
            a_opt, _ = fine_single_cd(best_seq, time_limit=t_alloc)
            _try(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
