"""
Hybrid subgradient + LP optimization for minimizing autocorrelation peak
of non-negative step functions.

Strategy:
  1. FFT-based projected subgradient method for fast iterations
  2. LP-based structured refinement (max sum(g) s.t. (f*g) <= M)
  3. Diverse initializations (trapezoidal, cosine, random, bimodal)
  4. Momentum-accelerated subgradient + adaptive step sizes
  5. Final coordinate-level polish via simulated annealing
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len
import time


def fft_convolve(f):
    """Fast full linear convolution via FFT."""
    n = len(f)
    m = next_fast_len(2 * n - 1)
    F = fft(f, m)
    conv = ifft(F * F).real
    return conv[:2 * n - 1]


def compute_subgrad(f, k_star):
    """Vectorized subgradient of max_k (f*f)[k] at argmax k_star."""
    n = len(f)
    subgrad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        idx = k_star - np.arange(j_start, j_end + 1)
        subgrad[j_start:j_end + 1] = 2.0 * f[idx]
    return subgrad


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = fft_convolve(f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) where A[k,j] = f[k-j] if valid."""
    i_idx = np.repeat(np.arange(n), n)
    j_idx = np.tile(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_for_g(f, M, time_limit_s=8.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    try:
        A = build_conv_matrix(n, f)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_step(f, t, lp_time=8.0):
    """One LP-driven update step."""
    conv = fft_convolve(f)
    M = float(conv.max())
    if M < 1e-15:
        return f
    g = solve_lp_for_g(f, M, lp_time)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-10:
        return None
    g = g / gs
    if np.linalg.norm(g - f) < 1e-10:
        return None
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return None
    return f_new / s


def subgradient_optimize(f, time_budget, eta_init=0.05, eta_min=0.001,
                         momentum=0.9, stall_limit=200):
    """Momentum-accelerated projected subgradient method."""
    f = f.copy()
    n = len(f)
    v = np.zeros(n)
    eta = eta_init
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        conv = fft_convolve(f)
        k_star = int(np.argmax(conv))
        subgrad = compute_subgrad(f, k_star)

        v = momentum * v + (1.0 - momentum) * subgrad
        f_new = f - eta * v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f
            s = 1.0
        f_new = f_new / s
        f = f_new

        it += 1
        eta = max(eta_min, eta_init / (1.0 + 0.001 * it))

        if it % 50 == 0:
            sc = fast_score(f)
            if np.isfinite(sc) and sc < best_sc - 1e-8:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 50
            if stall > stall_limit:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                f = np.maximum(f + rng.normal(0, 0.02, n), 0.0)
                f = f / f.sum()
                v = np.zeros(n)
                stall = 0

    return best, best_sc


def lp_optimize(f, time_budget, t_init=0.3, t_min=0.05, lp_time=8.0):
    """LP-based iterative optimization."""
    f = f.copy()
    t = t_init
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        f_new = lp_step(f, t, lp_time)
        if f_new is None:
            # Perturb and retry
            rng = np.random.default_rng(10000 + it)
            w = max(2, len(f) // 16)
            start = int(rng.integers(0, max(1, len(f) - w)))
            f_tmp = f.copy()
            f_tmp[start:start + w] *= np.exp(rng.normal(0, 0.3, w))
            f_tmp = np.maximum(f_tmp, 0.0)
            f = f_tmp / f_tmp.sum()
            it += 1
            continue
        f = f_new
        t = max(t_min, t * 0.98)
        it += 1

        if it % 2 == 0:
            sc = fast_score(f)
            if np.isfinite(sc) and sc < best_sc - 1e-9:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 2
            if stall > 80:
                rng = np.random.default_rng(20000 + it)
                w = max(2, len(f) // 10)
                start = int(rng.integers(0, max(1, len(f) - w)))
                f = f.copy()
                f[start:start + w] *= np.exp(rng.normal(0, 0.5, w))
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stall = 0

    return best, best_sc


def combined_optimize(f, time_budget):
    """Interleave subgradient (fast) and LP (structured) steps."""
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    t0 = time.monotonic()
    use_lp_every = 10  # do LP every 10 subgradient bursts

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Fast subgradient burst
        sg_time = min(remaining * 0.3, 10.0)
        f_sg, sc_sg = subgradient_optimize(f, sg_time,
                                           eta_init=0.03, eta_min=0.001,
                                           momentum=0.85, stall_limit=100)
        if sc_sg < best_sc:
            best_sc = sc_sg
            best = f_sg.copy()
        f = f_sg

        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        # LP step
        lp_time = min(remaining * 0.15, 10.0)
        if use_lp_every <= 0:
            f_lp, sc_lp = lp_optimize(f, max(1.0, remaining * 0.4),
                                      t_init=0.2, t_min=0.05,
                                      lp_time=max(2.0, lp_time))
            if sc_lp < best_sc:
                best_sc = sc_lp
                best = f_lp.copy()
            f = f_lp
            use_lp_every = 10
        else:
            use_lp_every -= 1

    return best, best_sc


def make_initial(n, rng, method):
    """Generate diverse non-negative starting vectors (sum=1)."""
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1.0 + np.cos(x))
    elif method == 'sine_sq':
        x = np.linspace(0, np.pi, n)
        f = np.sin(x) ** 2
    elif method == 'trapezoid':
        x = np.arange(n, dtype=np.float64)
        edge = max(1, n // 8)
        f = np.ones(n)
        f[:edge] = np.linspace(0.3, 1.0, edge)
        f[-edge:] = np.linspace(1.0, 0.3, edge)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        sigma = max(1, n // 20)
        f = np.exp(-0.5 * ((x - n * 0.15) / sigma) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.85) / sigma) ** 2) + 0.1 * np.ones(n)
    elif method == 'gaussian_mid':
        x = np.arange(n, dtype=np.float64)
        sigma = max(1, n // 6)
        f = np.exp(-0.5 * ((x - n // 2) / sigma) ** 2) + 0.05 * np.ones(n)
    elif method == 'random':
        f = rng.random(n) + 0.1
    elif method == 'perturbed_uniform':
        f = np.ones(n) + rng.normal(0, 0.3, n)
    elif method == 'linear_ramp':
        f = np.linspace(0.2, 1.0, n)
    elif method == 'v_shape':
        x = np.arange(n, dtype=np.float64)
        mid = n // 2
        f = 0.5 + 0.5 * np.abs(x - mid) / mid
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f


def coordinate_polish(f, time_budget, rng=None):
    """Fine-tune via random coordinate perturbations (simulated annealing)."""
    if rng is None:
        rng = np.random.default_rng(12345)
    f = f.copy()
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    T = 0.05  # annealing temperature
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        # Perturb a random block
        block_size = int(rng.integers(1, max(2, min(8, n // 10))))
        start = int(rng.integers(0, max(1, n - block_size)))
        f_new = f.copy()
        f_new[start:start + block_size] *= np.exp(rng.normal(0, 0.1, block_size))
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        sc_new = fast_score(f_new)
        if np.isfinite(sc_new):
            delta = sc_new - best_sc
            if delta < -1e-10:
                f = f_new
                best = f_new.copy()
                best_sc = sc_new
            elif rng.random() < np.exp(-delta / T):
                f = f_new
        T *= 0.9995

    return best, best_sc

        pass
    print("Baseline score = %.6f (n=%d)" % (bs, best.size), flush=True)

    def try_update(cand, sc):
        nonlocal best, bs
        if np.isfinite(sc) and sc < bs - 1e-10:
            bs = sc
            best = np.maximum(cand, 0.0).copy()
            print("  IMPROVEMENT: %.6f (n=%d)" % (bs, best.size), flush=True)

    # Phase 1: Subgradient optimization from diverse starts at n=512
    n_base = int(best.size)
    methods = ['perturbed_uniform', 'cosine', 'sine_sq', 'trapezoid',
               'bimodal', 'gaussian_mid', 'random', 'linear_ramp', 'v_shape']
    rng = np.random.default_rng(42)

    print("\n=== Phase 1: Subgradient from diverse starts (n=%d) ===" % n_base, flush=True)
    per_method_time = max(3.0, min(30.0, 300.0 / len(methods)))

    for m in methods:
        if time.monotonic() > deadline:
            break
        start = make_initial(n_base, rng, m)
        # If this method gives same as parent, add more perturbation
        if m == 'perturbed_uniform':
            start = 0.7 * start + 0.3 * (best / best.sum())
        t_start = time.monotonic()
        f_res, sc_res = subgradient_optimize(start, per_method_time,
                                             eta_init=0.04, eta_min=0.0005,
                                             momentum=0.88, stall_limit=150)
        try_update(f_res, sc_res)

    # Phase 2: Try parent with subgradient
    print("\n=== Phase 2: Subgradient from parent ===", flush=True)
    f_res, sc_res = subgradient_optimize(best, 60.0,
                                         eta_init=0.02, eta_min=0.0003,
                                         momentum=0.9, stall_limit=100)
    try_update(f_res, sc_res)

    # Phase 3: LP refinement on current best
    print("\n=== Phase 3: LP refinement (n=%d) ===" % best.size, flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        f_res, sc_res = lp_optimize(best, min(remaining * 0.5, 300.0),
                                    t_init=0.25, t_min=0.04, lp_time=8.0)
        try_update(f_res, sc_res)

    # Phase 4: Combined optimization from multiple starts at larger n
    print("\n=== Phase 4: Combined opt at larger n ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for n_try in [1024, 2048]:
            if time.monotonic() > deadline:
                break
            if remaining < 60:
                break
            for m in ['cosine', 'bimodal', 'trapezoid']:
                if time.monotonic() > deadline:
                    break
                if remaining < 30:
                    break
                start = make_initial(n_try, rng, m)
                t_budget = min(remaining * 0.15, 120.0)
                print("  n=%d method=%s budget=%.0fs" % (n_try, m, t_budget), flush=True)
                f_res, sc_res = combined_optimize(start, t_budget)
                try_update(f_res, sc_res)

    # Phase 5: LP refinement on the overall best
    print("\n=== Phase 5: Final LP on best (n=%d) ===" % best.size, flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        f_res, sc_res = lp_optimize(best, min(remaining * 0.5, 200.0),
                                    t_init=0.15, t_min=0.03, lp_time=10.0)
        try_update(f_res, sc_res)

    # Phase 6: Coordinate polish
    print("\n=== Phase 6: Coordinate polish ===", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        rng_pol = np.random.default_rng(99999)
        f_res, sc_res = coordinate_polish(best, min(remaining * 0.6, 120.0), rng_pol)
        try_update(f_res, sc_res)

    # Phase 7: One more subgradient pass on the best
    remaining = deadline - time.monotonic()
    if remaining > 5:
        f_res, sc_res = subgradient_optimize(best, min(remaining * 0.5, 60.0),
                                             eta_init=0.01, eta_min=0.0001,
                                             momentum=0.92, stall_limit=50)
        try_update(f_res, sc_res)

    # Phase 8: Final coordinate polish
    remaining = deadline - time.monotonic()
    if remaining > 5:
        rng_pol2 = np.random.default_rng(77777)
        f_res, sc_res = coordinate_polish(best, min(remaining - 2.0, 30.0), rng_pol2)
        try_update(f_res, sc_res)

    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512)

    # Verify with authoritative evaluator
    try:
        final_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        final_score = fast_score(best)

    print("\nFINAL score = %.6f (n=%d)" % (final_score, best.size), flush=True)
    return [float(x) for x in best]
