```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def smooth_gradient(a, T):
    """Compute gradient of 2n * softmax_max(T * c) where c = a * a (self-conv)."""
    n = len(a)
    c = fft_convolve_self(a)
    Tc = T * (c - c.max())  # shift for stability
    W = np.exp(Tc)
    W /= W.sum()
    # grad[m] = 2 * sum_j W[m+j] * a[j]
    # = correlation of W and a, first n elements
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    W_flipped = W[::-1]
    conv = np.real(ifft(fft(W_flipped, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[2*n-2 : 2*n-2+n]
    return grad, W, c

def hard_subgradient(a, top_k=100):
    """Compute subgradient using top-k active lags."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    W = top_vals / total
    # Build sparse W-like array
    W_full = np.zeros(2*n-1)
    W_full[top_idx] = W
    # Same correlation computation
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    W_flipped = W_full[::-1]
    conv = np.real(ifft(fft(W_flipped, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[2*n-2 : 2*n-2+n]
    return grad, c
```

```python
"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm:
1. Normalize to sum = sqrt(2n) so objective = max of self-convolution.
2. Use smoothed max (softmax) with temperature annealing for a smooth objective.
3. Compute the gradient efficiently via FFT-based cross-correlation.
4. Use Adam optimizer with projection onto the non-negative simplex.
5. Multiple phases: smooth -> semi-hard -> hard (top-k subgradient) -> polish.
6. Periodic perturbations and restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (w * idx > cumsum - S)
    if not np.any(cond):
        theta = cumsum[-1] / n
    else:
        rho = int(np.max(np.where(cond)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, T):
    """
    Compute gradient of softmax-smoothed max of self-convolution.
    Returns (gradient, convolution, max_conv).
    """
    n = len(a)
    c = fft_convolve_self(a)
    # Softmax weights (shifted for stability)
    c_shift = c - c.max()
    Tc = T * c_shift
    W = np.exp(Tc)
    W_sum = W.sum()
    if W_sum < 1e-300:
        W = np.zeros(2 * n - 1)
        W[np.argmax(c)] = 1.0
    else:
        W /= W_sum
    # Gradient: grad[m] = 2 * sum_j W[m+j] * a[j]
    # = 2 * (cross-correlation of W and a)[m]
    # Cross-correlation via FFT:
    # corr[m] = sum_j W[m+j] * a[j]
    # = conv(W_flipped, a)[m + len(W) - 1]
    W_flipped = W[::-1]
    size = (2 * n - 1) + n - 1  # 3n - 2
    fsize = 1 << (size - 1).bit_length()
    conv = np.real(ifft(fft(W_flipped, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[(2 * n - 2):(2 * n - 2) + n]
    return grad, c, c.max()


def compute_hard_grad(a, top_k):
    """Compute subgradient using top-k active lags. Returns (gradient, max_conv)."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-300:
        return np.zeros(n), c.max()
    W = top_vals / total
    # Build full W array
    W_full = np.zeros(2 * n - 1)
    W_full[top_idx] = W
    # Same correlation computation
    W_flipped = W_full[::-1]
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    conv = np.real(ifft(fft(W_flipped, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[(2 * n - 2):(2 * n - 2) + n]
    return grad, c.max()


def run_smooth_phase(a, n, S, T_start, T_end, max_time, rng, lr_scale=1.0):
    """Adam with smoothed max objective and temperature annealing."""
    a = project_simplex(a.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # Learning rate schedule
    base_lr = lr_scale * S / n
    
    while time.monotonic() - start < max_time:
        t += 1
        # Temperature schedule (exponential interpolation)
        frac = min(1.0, t / max(1, int(max_time * 500)))  # ~500 iterations per second
        T = T_start * (T_end / T_start) ** frac
        
        grad, c, cur_max = compute_smooth_grad(a, T)
        
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
        
        # Adam update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm  # normalize gradient
        
        lr = base_lr / (1 + 0.01 * t)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Periodic perturbation
        if t % 5000 == 0 and rng.random() < 0.3:
            sigma = 0.05 * S / n
            a = project_simplex(a + rng.normal(0, sigma, n), S)
    
    return best_a, best_score


def run_hard_phase(a, n, S, max_time, rng, top_k=50, lr_scale=0.5):
    """Adam with hard subgradient (top-k lags)."""
    a = project_simplex(a.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    base_lr = lr_scale * S / n
    
    while time.monotonic() - start < max_time:
        t += 1
        grad, cur_max = compute_hard_grad(a, top_k)
        
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        lr = base_lr / (1 + 0.005 * t)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

            a = project_simplex(a + rng.normal(0, sigma, n), S)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences of length ~8192.
Uses smoothed-max Adam with FFT-based gradients, temperature annealing, and
multiple phases (smooth -> hard -> polish).
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (w * idx > cs - S)
    if not np.any(cond):
        theta = cs[-1] / n
    else:
        rho = int(np.max(np.where(cond)[0])) + 1
        theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, T):
    """Returns (grad, c, max_c) for the softmax-smoothed max objective."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    c_shift = c - max_c
    Tc = T * c_shift
    W = np.exp(Tc)
    Ws = W.sum()
    if Ws < 1e-300:
        W = np.zeros(2 * n - 1)
        W[int(np.argmax(c))] = 1.0
    else:
        W /= Ws
    Wf = W[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    conv = np.real(ifft(fft(Wf, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, c, max_c


def hard_subgradient(a, top_k):
    """Returns (grad, max_c) using top-k active lags."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    vals = c[top_idx]
    total = vals.sum()
    W = np.zeros(2 * n - 1)
    W[top_idx] = vals / total
    Wf = W[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    conv = np.real(ifft(fft(Wf, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, max_c


def adam_smooth(a, n, S, T0, T1, dur, rng):
    """Adam with smoothed max, T annealed from T0 to T1 over dur seconds."""
    a = project_simplex(a.copy(), S)
    mx = np.zeros(n)
    vw = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-9
    t0 = time.monotonic()
    best = np.inf
    best_a = a.copy()
    steps = 0
    while time.monotonic() - t0 < dur:
        t += 1
        frac = min(1.0, (time.monotonic() - t0) / dur)
        T = T0 * (T1 / T0) ** frac
        g, c, cur = smooth_gradient(a, T)
        if cur < best - 1e-10:
            best = cur
            best_a = a.copy()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        lr = (S / n) * 0.5 / (1.0 + 0.002 * t)
        mx = b1 * mx + (1 - b1) * g
        vw = b2 * vw + (1 - b2) * g * g
        mh = mx / (1 - b1 ** t)
        vh = vw / (1 - b2 ** t)
        a = project_simplex(a - lr * mh / (np.sqrt(vh) + eps), S)
        steps += 1
        if t % 10000 == 0:
            if rng.random() < 0.2:
                a = project_simplex(a + rng.normal(0, 0.03 * S / n, n), S)
    return best_a, best, steps


def adam_hard(a, n, S, dur, rng, top_k=50, lr_mult=0.3):
    """Adam with hard subgradient."""
    a = project_simplex(a.copy(), S)
    mx = np.zeros(n)
    vw = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-9
    t0 = time.monotonic()
    best = np.inf
    best_a = a.copy()
    steps = 0
    while time.monotonic() - t0 < dur:
        t += 1
        g, cur = hard_subgradient(a, top_k)
        if cur < best - 1e-10:
            best = cur
            best_a = a.copy()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        lr = (S / n) * lr_mult / (1.0 + 0.003 * t)
        mx = b1 * mx + (1 - b1) * g
        vw = b2 * vw + (1 - b2) * g * g
        mh = mx / (1 - b1 ** t)
        vh = vw / (1 - b2 ** t)
        a = project_simplex(a - lr * mh / (np.sqrt(vh) + eps), S)
        steps += 1
        if t % 20000 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.01 * S / n, n), S)
    return best_a, best, steps
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm:
- Normalize to sum=sqrt(2n) so objective = max of self-convolution.
- Use softmax-smoothed max with temperature annealing for a smooth objective.
- Compute gradient via FFT-based cross-correlation in O(n log n).
- Adam optimizer with Euclidean projection onto non-negative simplex.
- Four phases: smooth(low T) -> smooth(high T) -> hard(top-k) -> polish.
- Multiple initializations and periodic perturbations for exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    if v.sum() <= S and np.all(v >= 0):
        return v.copy()
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (w * idx > cs - S)
    if not np.any(cond):
        theta = cs[-1] / n
    else:
        rho = int(np.max(np.where(cond)[0])) + 1
        theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, T):
    """
    Compute gradient of softened max: (1/T)*log(sum exp(T*c_k)) where c = a*a.
    Returns (grad, c, max_c).
    Gradient of the objective 2n*(1/T)*log(sum exp(T*c)) is 2n*sum_k W_k * dc_k/da.
    We return the gradient without the 2n factor (it's a constant scaler for Adam).
    """
    n = len(a)
    c = fft_convolve_self(a)
    max_c = float(c.max())
    c_shift = c - max_c
    Tc = T * c_shift
    W = np.exp(Tc)
    Ws = W.sum()
    if Ws < 1e-300:
        W = np.zeros(2 * n - 1)
        W[int(np.argmax(c))] = 1.0
    else:
        W /= Ws
    # grad[m] = 2 * sum_j W[m+j] * a[j] = 2 * conv(W_flipped, a)[2n-2-m]
    Wf = W[::-1]
    conv_len = (2 * n - 1) + n - 1  # 3n-2
    fsize = 1 << (conv_len - 1).bit_length()
    conv = np.real(ifft(fft(Wf, fsize) * fft(a, fsize)))
    # Extract: grad[m] = 2 * conv[2n-2-m] for m=0..n-1
    # i.e., conv indices from 2n-2 down to n-1
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, c, max_c


def hard_subgradient(a, top_k):
    """Compute subgradient using top-k active lags. Returns (grad, max_c)."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = float(c.max())
    if top_k >= 2 * n - 1:
        top_k = 2 * n - 1
    # Use argpartition for efficiency
    part = np.argpartition(c, -top_k)[-top_k:]
    top_idx = part[np.argsort(c[part])[::-1]]  # sort by value descending
    vals = c[top_idx]
    total = vals.sum()
    if total < 1e-300:
        return np.zeros(n), max_c
    W = np.zeros(2 * n - 1)
    W[top_idx] = vals / total
    Wf = W[::-1]
    conv_len = 3 * n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    conv = np.real(ifft(fft(Wf, fsize) * fft(a, fsize)))
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, max_c


def adam_smooth(a_in, n, S, T0, T1, dur, rng):
    """Adam with smoothed max objective, T annealed from T0 to T1 over dur seconds."""
    a = project_simplex(a_in.copy(), S)
    mx = np.zeros(n)
    vw = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-9
    t0 = time.monotonic()
    best = np.inf
    best_a = a.copy()
    steps = 0
    while time.monotonic() - t0 < dur:
        t += 1
        frac = min(1.0, (time.monotonic() - t0) / dur)
        T = T0 * (T1 / T0) ** frac
        g, c, cur = smooth_gradient(a, T)
        if cur < best - 1e-10:
            best = cur
            best_a = a.copy()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        lr = (S / n) * 0.5 / (1.0 + 0.002 * t)
        mx = b1 * mx + (1.0 - b1) * g
        vw = b2 * vw + (1.0 - b2) * g * g
        mh = mx / (1.0 - b1 ** t)
        vh = vw / (1.0 - b2 ** t)
        a = project_simplex(a - lr * mh / (np.sqrt(vh) + eps), S)
        steps += 1
        if t % 8000 == 0 and rng.random() < 0.25:
            sigma = 0.04 * S / n
            a = project_simplex(a + rng.normal(0.0, sigma, n), S)
    return best_a, best, steps


def adam_hard(a_in, n, S, dur, rng, top_k=100, lr_mult=0.3):
    """Adam with hard subgradient (top-k lags)."""
    a = project_simplex(a_in.copy(), S)
    mx = np.zeros(n)
    vw = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-9
    t0 = time.monotonic()
    best = np.inf
    best_a = a.copy()
    steps = 0
    while time.monotonic() - t0 < dur:
        t += 1
        g, cur = hard_subgradient(a, top_k)
        if cur < best - 1e-10:
            best = cur
            best_a = a.copy()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        lr = (S / n) * lr_mult / (1.0 + 0.003 * t)
        mx = b1 * mx + (1.0 - b1) * g
        vw = b2 * vw + (1.0 - b2) * g * g
        mh = mx / (1.0 - b1 ** t)
        vh = vw / (1.0 - b2 ** t)
        a = project_simplex(a - lr * mh / (np.sqrt(vh) + eps), S)
        steps += 1
        if t % 15000 == 0 and rng.random() < 0.15:
            sigma = 0.015 * S / n
            a = project_simplex(a + rng.normal(0.0, sigma, n), S)
    return best_a, best, steps


def mirror_descent(a_in, n, S, dur, rng, eta0=0.5):
    """Mirror descent (multiplicative weights) variant."""
    a = project_simplex(a_in.copy(), S)
    t0 = time.monotonic()
    best = np.inf
    best_a = a.copy()
    t = 0
    steps = 0
    while time.monotonic() - t0 < dur:
        t += 1
        g, c, cur = smooth_gradient(a, 50.0)
        if cur < best - 1e-10:
            best = cur
            best_a = a.copy()
        gmax = np.abs(g).max()
        if gmax < 1e-15:
            break
        g = g / gmax
        eta = eta0 * 0.5 ** (t / 5000.0)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new * (S / s)
        else:
            a = np.full(n, S / n)
        steps += 1
        if t % 10000 == 0 and rng.random() < 0.2:
            a = a * (1 + rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
    return best_a, best, steps
```
