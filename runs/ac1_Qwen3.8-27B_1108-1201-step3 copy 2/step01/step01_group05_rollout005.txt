import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_sq(p):
    """Full self-convolution via FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def _score(p):
    """Score = 2*n*max(p*p)/sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = _conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    return float(2.0 * n * cmax / (s * s))


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _normalize(p):
    """Normalize to sum 1, nonnegative."""
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(int(p.size), 1.0 / max(1, p.size), dtype=np.float64)
    return p / s

    """
    LP power iteration: given p (sum=1), find best response g that maximizes
    sum(g) subject to p*g <= M componentwise, g >= 0.
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n < 2 or n > 2000:
        return p, _score(p)

    best_p = p.copy()
    best_score = _score(best_p)
    t_start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t_start > time_limit:
            break

        c = _conv_sq(best_p)
        M = float(np.max(c))

        if M <= 0:
            break

        # Build LP: max sum(g) s.t. (p*g)_k <= M, g >= 0
        # (p*g)_k = sum_j p_j * g_{k+j} for valid indices
        # Rows: k = 0, 1, ..., 2n-2
        # (p*g)_k = sum_{j: 0<=j<n, 0<=k-j<n} p_j * g_{k-j}
        # = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} p_j * g_{k-j}
        # Let i = k-j, so j = k-i: sum_i p_{k-i} * g_i where 0<=i<n and 0<=k-i<n

        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                i = k - j  # index in g
                rows.append(k)
                cols.append(i)
                data.append(best_p[j])

        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                          method='highs', options={'disp': False})
        except Exception:
            break

        if res is None or res.status not in (0, 2):
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-12:
            break
        g = g / s

        # Try several mixing ratios
        best_local = best_score
        best_local_p = best_p.copy()
        for t in (1.0, 0.7, 0.5, 0.3, 0.1):
            q = _project_simplex((1.0 - t) * best_p + t * g)
            sc = _score(q)
            if sc < best_local:
                best_local = sc
                best_local_p = q

        if best_local < best_score - 1e-14:
            improvement = best_score - best_local
            best_score = best_local
            best_p = best_local_p
            if verbose:
                print("  LP iter %d: score=%.10f (improved by %.2e)" % (it, best_score, improvement))
        else:
            # No improvement, check if we're at a fixed point
            if verbose and it % 10 == 0:
                print("  LP iter %d: no improvement, score=%.10f" % (it, best_score))
            # Try harder: use a slightly relaxed bound
            M_relaxed = M * 1.001
            try:
                res2 = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_relaxed),
                               bounds=(0, None), method='highs', options={'disp': False})
                if res2 is not None and res2.status == 0:
                    g2 = np.asarray(res2.x, dtype=np.float64)
                    g2 = np.maximum(g2, 0.0)
                    s2 = float(g2.sum())
                    if s2 > 1e-12:
                        g2 = g2 / s2
                        for t in (0.5, 0.3, 0.1, 0.05):
                            q = _project_simplex((1.0 - t) * best_p + t * g2)
                            sc = _score(q)
                            if sc < best_score - 1e-14:
                                best_score = sc
                                best_p = q
                                break
            except Exception:
                pass

    return best_p, best_score


def _gradient_optimize(p0, time_limit=60.0, verbose=False):
    """
    Gradient-based optimization using smoothed max surrogate.
    """
    p = _project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n < 2:
        return p, _score(p)

    best_p = p.copy()
    best_score = _score(best_p)
    t_start = time.monotonic()
    factor = 0.1
    since_improve = 0
    prev_p = p.copy()

    while time.monotonic() - t_start < time_limit:
        c = _conv_sq(p)
        cmax = float(np.max(c))
        if cmax <= 0:
            break

        T = max(factor * cmax, 1e-10)
        diff = (c - cmax) / T
        e = np.exp(diff - diff.max())
        w = e / e.sum()

        # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (autocorrelation of w with p)
        L = next_fast_len(3 * n - 2)
        Fp = fft(p, L)
        Fw = fft(w[::-1], L)
        G = np.real(ifft(Fp * Fw))
        # Extract the right part
        g = 2.0 * G[2 * n - 2:n - 2:-1] if n > 1 else 2.0 * G[:1]
        g = np.asarray(g, dtype=np.float64)
        if g.size != n:
            tmp = np.zeros(n, dtype=np.float64)
            m = min(n, g.size)
            tmp[:m] = g[:m]
            g = tmp
        g = np.maximum(g, 0.0)

        # Gradient descent with projection
        accepted = False
        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        p_scale = max(1.0 / n, 0.05 * float(np.max(p)))
        if gabs > 1e-300:
            base = p_scale / gabs
        else:
            base = 0.1

        for lr_mult in (0.5, 0.1, 0.02, 0.004, 0.001):
            lr = lr_mult * base
            v = p - lr * g
            v = v + 0.3 * (p - prev_p)  # momentum
            pt = _project_simplex(v)
            sc = _score(pt)
            if sc < best_score - 1e-14 or sc < _score(p) - 1e-12:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            factor = max(0.001, factor * 0.5)
            since_improve += 1
            if since_improve > 100:
                # Random perturbation
                sigma = 0.5 / n
                p = _project_simplex(best_p + np.random.normal(0, sigma, n))
                since_improve = 0
                factor = 0.1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for nonnegative step function minimizing the evaluation score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 30.0)

    best_p = None
    best_score = np.inf

    def try_record(p, sc=None):
        nonlocal best_p, best_score
        if p is None:
            return
        p = _normalize(np.asarray(p, dtype=np.float64))
        if p.size == 0:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  NEW BEST: score=%.10f, n=%d" % (sc, p.size), flush=True)

    print("Starting search with budget %.0fs" % budget_s, flush=True)

    # Try to load provided sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 16 and parent_arr.sum() > 0:
            parent = _normalize(parent_arr)
            print("Loaded parent sequence: n=%d" % parent.size, flush=True)
    except Exception:
        pass

    # 1) Try uniform sequence (proven optimal by Cauchy-Schwarz)
    n_unif = 700
    p_unif = np.ones(n_unif, dtype=np.float64) / n_unif
    sc_unif = _score(p_unif)
    try_record(p_unif, sc_unif)
    print("Uniform n=%d: score=%.10f" % (n_unif, sc_unif), flush=True)

    # Try different lengths of uniform
    for n_try in (100, 200, 350, 500, 700, 1000, 1400, 2000):
        pt = np.ones(n_try, dtype=np.float64) / n_try
        sc = _score(pt)
        try_record(pt, sc)

    # 2) Try the parent sequence if available
    if parent is not None:
        sc_parent = _score(parent)
        try_record(parent, sc_parent)
        print("Parent sequence: score=%.10f, n=%d" % (sc_parent, parent.size), flush=True)

    # 3) LP power iteration from multiple starting points
    n_lp = 400  # Keep LP tractable

    if deadline - time.monotonic() > 60.0:
        lp_starts = []
        lp_starts.append(np.ones(n_lp, dtype=np.float64) / n_lp)

        if parent is not None and parent.size == n_lp:
            lp_starts.append(parent.copy())
        elif parent is not None:
            # Resample to n_lp
            idx = (np.arange(n_lp) * parent.size / n_lp).astype(int)
            lp_starts.append(_normalize(parent[idx]))

        # Some non-uniform starts
        x = np.linspace(0, 1, n_lp)
        for s in (0.5, 1.0, 2.0, 4.0, 8.0):
            lp_starts.append(_normalize(np.exp(-s * x)))
            lp_starts.append(_normalize(np.exp(s * (x - 1))))
        for a in (0.5, 1.0, 2.0, 4.0):
            lp_starts.append(_normalize(1.0 + a * (2 * x - 1)**2))
            lp_starts.append(_normalize(np.maximum(1.0 - a * (2 * x - 1)**2, 0.1)))
        lp_starts.append(_normalize(1.0 + 0.8 * np.sin(2 * np.pi * 3 * x)))
        lp_starts.append(_normalize(1.0 + 0.8 * np.cos(2 * np.pi * 5 * x)))
        lp_starts.append(_normalize(np.maximum(1.0 + 1.5 * np.sin(2 * np.pi * 7 * x + 0.5), 0.05)))

        # Score and sort starting points
        scored = []
        for ps in lp_starts:
            ps = _normalize(ps)
            sc = _score(ps)
            scored.append((sc, ps))
        scored.sort(key=lambda x: x[0])

        # Run LP from top-5 starts
        lp_time_per = min(120.0, max(20.0, (deadline - time.monotonic() - 120.0) / 5.0))
        for i, (sc, ps) in enumerate(scored[:5]):
            if time.monotonic() > deadline - 60.0:
                break
            remaining_lp = min(lp_time_per, deadline - time.monotonic() - 60.0)
            if remaining_lp < 10.0:
                break
            print("LP start %d (initial score=%.6f), time_limit=%.0fs" % (i, sc, remaining_lp), flush=True)
            p_lp, sc_lp = _lp_best_response(ps, max_iter=200, time_limit=remaining_lp, verbose=True)
            try_record(p_lp, sc_lp)
            # Also try gradient polish
            if time.monotonic() < deadline - 30.0:
                p_grad, sc_grad = _gradient_optimize(p_lp, time_limit=min(30.0, deadline - time.monotonic() - 30.0))
                try_record(p_grad, sc_grad)

    # 4) If parent available and longer, try LP on a truncated/resampled version
    if parent is not None and parent.size > 200 and time.monotonic() < deadline - 60.0:
        n_parent_lp = min(parent.size, 500)
        if n_parent_lp > 200:
            idx = (np.arange(n_parent_lp) * parent.size / n_parent_lp).astype(int)
            p_parent_resamp = _normalize(parent[idx])
            sc_pr = _score(p_parent_resamp)
            try_record(p_parent_resamp, sc_pr)
            if time.monotonic() < deadline - 30.0:
                p_lp2, sc_lp2 = _lp_best_response(p_parent_resamp, max_iter=100,
                                                   time_limit=min(120.0, deadline - time.monotonic() - 30.0))
                try_record(p_lp2, sc_lp2)

    # 5) Final: try gradient optimization from the best found
    if best_p is not None and time.monotonic() < deadline - 15.0:
        limit = max(5.0, deadline - time.monotonic() - 10.0)
        p_final, sc_final = _gradient_optimize(best_p, time_limit=limit)
        try_record(p_final, sc_final)

    # 6) Sanity check: verify with evaluate_sequence
    if best_p is None:
        best_p = np.ones(700, dtype=np.float64)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.ones(700, dtype=np.float64)
        s = 700.0
    # Scale to be in [0, 1000]
    best_p = np.minimum(best_p * (1.0 / s), 1000.0)
    # Actually, just return as-is since score is scale-invariant
    # But make sure sum is reasonable
    best_p = np.asarray(best_p, dtype=np.float64)
    sum_final = float(np.sum(best_p))
    if sum_final < 0.01:
        best_p = np.ones(700, dtype=np.float64)

    print("FINAL: n=%d, score=%.10f, sum=%.4f" % (best_p.size, _score(best_p), sum_final), flush=True)

    # Verify with the actual evaluator
    try:
        actual = evaluate_sequence([float(x) for x in best_p])
        print("Evaluator confirms: %.10f" % actual, flush=True)
    except Exception as e:
        print("Evaluator check failed: %s" % e, flush=True)

    return [float(x) for x in best_p]
