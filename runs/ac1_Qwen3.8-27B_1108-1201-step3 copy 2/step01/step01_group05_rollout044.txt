def conv_sq(p):
    """Self-convolution via FFT."""
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def internal_score(p):
    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = np.max(c)
    return 2*n*cmax/(s*s)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, float)
    n = len(v)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = (u * idx) > (cssum - 1.0)
    k = np.count_nonzero(cond) - 1
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    return out / out.sum()

def lp_best_response(p, eps=1e-10):
    """
    Solve LP: max sum(g) s.t. (p+eps)*g <= M, g >= 0.
    Returns the optimal g (normalized to sum 1).
    """
    n = len(p)
    f = p + eps
    f = f / f.sum()
    c = conv_sq(f)
    M = np.max(c)
    
    # Build constraint matrix A where A[k, j] = sum_i f[i] * delta[i + j == k]
    # Actually, (f*g)_k = sum_j f[k-j] * g_j for valid k-j

    cols = []
    data = []
    for k in range(2*n - 1):

            if 0 <= idx < n:

                cols.append(j)

        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        return np.full(max(1, int(p.size)), 1.0 / max(1, int(p.size)))
    return p / s


def build_conv_matrix(f):
    n = int(f.size)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(p, rng, eps_scale=1e-8):
    n = int(p.size)
    c = conv_sq(p)
    M = float(np.max(c))
    if M < 1e-300:
        return None, 0.0
    pert = eps_scale * M * rng.standard_normal(n)
    f = p + pert
    f = np.maximum(f, 0.0)
    fs = float(f.sum())
    if fs < 1e-15:
        return None, 0.0
    f = f / fs
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method='highs',
                      options={'disp': False, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None, 0.0
    if res.status != 0 or res.x is None:
        return None, 0.0
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return None, 0.0
    return g / s, s


def gradient_lse(p, T_factor):
    n = int(p.size)
    c = conv_sq(p)
    cmax = float(np.max(c))
    if cmax < 1e-300:
        return np.zeros(n), c, 0.0
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if raw < 1e-300 or not np.isfinite(raw):
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, c.astype(np.float64), surr


def resample_to_length(p, n_target):
    n_src = int(p.size)
    if n_src == n_target:
        return p.copy()
    x_src = np.linspace(0.0, 1.0, n_src)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    p_interp = np.interp(x_tgt, x_src, p)
    return safe_normalize(p_interp)


def make_uniform(n):
    return np.full(n, 1.0 / n)


def make_beta(n, a, b):
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    p = np.power(x, a) * np.power(1.0 - x, b)
    return safe_normalize(np.maximum(p, 0.0))


def make_cheb(n, coeffs):
    x = np.linspace(-1.0, 1.0, n)
    logp = np.zeros(n)
    c = np.asarray(coeffs, dtype=np.float64)
    T0 = np.ones(n)
    T1 = x.copy()
    if c.size > 0:
        logp += c[0] * T1
    for d in range(2, min(c.size, 15)):
        T2 = 2.0 * x * T1 - T0
        logp += c[d - 1] * T2
        T0 = T1
        T1 = T2
    eps = 1e-4
    if c.size > 14:
        xc = np.linspace(0.0, 1.0, n)
        logp += c[14] * np.log(xc + eps) + c[15] * np.log(1.0 - xc + eps) if c.size > 15 else c[14] * np.log(xc + eps)
    m = float(np.max(logp))
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)


def optimize_lp_gradient(p0, rng, time_limit, label=""):
    p = safe_normalize(p0)
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    end = time.monotonic() + time_limit
    no_improve = 0
    iter_no = 0
    T_factor = 0.1

    print(label + " start: score=%.10f n=%d" % (best_score, n), flush=True)

    while time.monotonic() < end:
        iter_no += 1
        remaining = end - time.monotonic()
        if remaining < 3.0:
            break

        # LP best response (every iteration for small n, every 2-3 for larger)
        do_lp = True
        if n > 1500 and iter_no % 3 != 0:
            do_lp = False
        elif n > 800 and iter_no % 2 != 0:
            do_lp = False

        accepted = False

        if do_lp:
            g, lp_val = lp_best_response(p, rng)
            if g is not None:
                for t in (0.5, 0.3, 0.7, 0.2, 0.85, 0.1, 1.0, 0.4, 0.6):
                    q = project_simplex((1.0 - t) * p + t * g)
                    sc = internal_score(q)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_p = q.copy()
                        p = q.copy()
                        current_score = sc
                        no_improve = 0
                        accepted = True
                        break
                    elif sc < current_score - 1e-14:
                        if not accepted:
                            p = q.copy()
                            current_score = sc
                            accepted = True

        if not accepted:
            # Gradient descent with log-sum-exp
            T_factor = max(0.005, T_factor * 0.985)
            g, c_cur, surr = gradient_lse(p, T_factor)
            if np.all(np.isfinite(g)):
                gabs = float(np.max(np.abs(g)))
                if gabs > 1e-300:
                    pmax = float(np.max(p))
                    p_scale = max(1.0 / n, 0.1 * pmax)
                    base = p_scale / gabs
                    for lr in (0.5 * base, 0.1 * base, 0.02 * base, 0.005 * base):
                        q = project_simplex(p - lr * g)
                        sc = internal_score(q)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_p = q.copy()
                            p = q.copy()
                            current_score = sc
                            no_improve = 0
                            accepted = True
                            break
                        elif sc < current_score - 1e-14 and not accepted:
                            p = q.copy()
                            current_score = sc
                            accepted = True
                            break

        if not accepted:
            no_improve += 1
            if no_improve > 80:
                # Perturb and restart from best
                sigma = 0.3 / n
                p = project_simplex(0.7 * best_p + 0.3 * p + rng.normal(0, sigma, n))
                current_score = internal_score(p)
                no_improve = 0
                T_factor = 0.15
            elif no_improve > 30:
                sigma = 0.1 / n
                p = project_simplex(0.9 * best_p + 0.1 * p + rng.normal(0, sigma, n))
                current_score = internal_score(p)
                no_improve = 25

        if iter_no % 200 == 0:
            print(label + " iter=%d best=%.10f cur=%.10f T=%.4f remain=%.0fs" % (
                iter_no, best_score, current_score, T_factor, remaining), flush=True)

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 15.0)

    # Get parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = safe_normalize(parent_arr)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d" % (best_score, p.size), flush=True)

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pn = int(parent.size)
        lengths.append(pn)
    for n_try in [512, 700, 1024, 1536, 2048, 3072, 4096]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))

    # Filter to reasonable range
    lengths = [n for n in lengths if 256 <= n <= 6000]

    # Phase 1: Quick evaluation of initial candidates at each length
    print("Phase 1: Quick initialization", flush=True)
    candidates = []  # (score, p, length)

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        inits = []
        if parent is not None:
            inits.append(resample_to_length(parent, n))
        inits.append(make_uniform(n))
        for a, b in [(0.5, 0.5), (1.0, 1.0), (0.3, 1.5), (1.5, 0.3), (0.7, 1.2), (1.2, 0.7), (2.0, 0.5), (0.5, 2.0)]:
            inits.append(make_beta(n, a, b))
        # Chebyshev shapes
        for _ in range(4):
            coeffs = rng.uniform(-1.5, 1.5, 8)
            inits.append(make_cheb(n, coeffs))
        # Symmetric shapes
        for a in [0.3, 0.7, 1.5, 3.0]:
            x = np.linspace(0, 1, n)
            p = 1.0 + a * np.cos(np.pi * (2 * x - 1))
            inits.append(safe_normalize(np.maximum(p, 0.01)))
        for a in [0.3, 0.8, 2.0]:
            x = np.linspace(0, 1, n)
            p = 1.0 + a * np.cos(2 * np.pi * (2 * x - 1))
            inits.append(safe_normalize(np.maximum(p, 0.01)))

        for p_init in inits:
            p_init = safe_normalize(p_init)
            sc = internal_score(p_init)
            if np.isfinite(sc):
                candidates.append((sc, p_init, n))

    candidates.sort(key=lambda x: x[0])
    for sc, p, n in candidates[:10]:
        try_record(p, sc)

    print("Phase 1 done. Best so far: %.10f" % best_score, flush=True)

    # Phase 2: LP-based optimization on top candidates
    print("Phase 2: LP optimization", flush=True)

    # Group by length, take top 2-3 per length
    by_length = {}
    for sc, p, n in candidates:
        if n not in by_length:
            by_length[n] = []
        if len(by_length[n]) < 3:
            by_length[n].append((sc, p))

    # Sort lengths by initial best score, spend more time on better lengths
    length_order = sorted(by_length.keys(), key=lambda n: by_length[n][0])

    # Time allocation: spend most time on best 3-4 lengths
    n_lengths = len(length_order)
    time_alloc = {}
    total_time = deadline - time.monotonic()
    if total_time < 60:
        time_alloc = {n: max(10, total_time / n_lengths) for n in length_order}
    else:
        # Weighted: top lengths get more time
        weights = []
        for i, n in enumerate(length_order):
            w = max(0.3, 1.0 - 0.15 * i)
            weights.append(w)
        wsum = sum(weights)
        for i, n in enumerate(length_order):
            time_alloc[n] = max(30.0, total_time * weights[i] / wsum)

    # Cap LP for very long sequences
    LP_MAX_N = 2500

    for n in length_order:
        if time.monotonic() > deadline - 30:
            break
        alloc = time_alloc.get(n, 60.0)
        alloc = min(alloc, deadline - time.monotonic() - 20.0)
        if alloc < 15.0:
            continue

        init_list = by_length[n][:3]
        for sc_init, p_init in init_list:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic() - 25.0
            if remaining < 15.0:
                break
            t_limit = min(alloc / len(init_list) * 1.5, remaining)
            if n <= LP_MAX_N:
                label = "LP-n%d" % n
            else:
                label = "GD-n%d" % n
            p_best, sc_best = optimize_lp_gradient(p_init, rng, t_limit, label)
            try_record(p_best, sc_best)

    print("Phase 2 done. Best: %.10f" % best_score, flush=True)

    # Phase 3: Final polish on the best solution found
    if best_p is not None and time.monotonic() < deadline - 20:
        print("Phase 3: Final polish", flush=True)
        n_best = int(best_p.size)
        t_limit = deadline - time.monotonic() - 10.0
        # Try a few different perturbations of the best
        for attempt in range(3):
            if time.monotonic() > deadline - 10:
                break
            p_try = best_p.copy()
            if attempt > 0:
                sigma = 0.15 / n_best
                p_try = project_simplex(p_try + rng.normal(0, sigma, n_best))
            t_sub = min(60.0, t_limit / (3 - attempt))
            p_pol, sc_pol = optimize_lp_gradient(p_try, rng, t_sub, "POLISH%d-" % attempt)
            try_record(p_pol, sc_pol)

    # Fallback
    if best_p is None:
        best_p = make_uniform(700)
        best_score = internal_score(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = make_uniform(700)
        best_p = best_p * 1.0
    best_p = np.minimum(best_p, 1000.0)

    # Verify with actual evaluator
    try:
        test_score = evaluate_sequence([float(x) for x in best_p])
        print("Final verification score: %.10f" % test_score, flush=True)
        if not np.isfinite(test_score) or test_score > best_score * 1.001:
            # Something wrong, use a safe fallback
            print("WARNING: score mismatch, using fallback", flush=True)
            best_p = make_uniform(700) * 1.0
            best_score = internal_score(best_p)
    except Exception:
        pass

    print("DONE. Best score: %.10f, length: %d" % (best_score, len(best_p)), flush=True)
    return [float(x) for x in best_p]
