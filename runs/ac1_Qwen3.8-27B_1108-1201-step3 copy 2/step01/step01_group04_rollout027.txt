    Solve LP: maximize sum(g) subject to conv(f,g) <= M (elementwise), g >= 0.

        if time.monotonic() > deadline - 5.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = lp_solve(a, M)
        if g is None:
            break
        
        S_g = float(g.sum())
        S_a = float(a.sum())
        
        if S_g > S_a * (1.0 + 1e-9):
            # Improvement possible: blend
            improved = False
            for t in [1.0, 0.9, 0.7, 0.5, 0.3]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_to(a_new, S_target)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    stuck_count = 0
                    break
            
            if not improved:
                stuck_count += 1
                # Small perturbation
                a = a + 0.005 * rng_local.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = normalize_to(a, S_target)
        else:
            stuck_count += 1
        
        if stuck_count > 15:
            # Use gradient to escape
            beta = 10.0 + 10.0 * min(stuck_count / 10.0, 5.0)
            grad = smooth_max_grad(a, beta)
            grad -= grad.mean()
            lr = 0.02 * (1.0 + 0.1 * stuck_count)
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S_target)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
            stuck_count = 0
    
    return best, best_score


def grad_optimize(a, deadline, max_iters=3000):
    """Smoothed-max gradient descent."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = normalize_to(a.copy(), S_target)
    best = a.copy()
    best_score = compute_obj(a)
    rng_local = np.random.default_rng(999)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        frac = it / max_iters
        beta = 5.0 + 45.0 * frac
        lr = max(0.001, 0.1 * (1.0 - frac) ** 1.5)
        
        grad = smooth_max_grad(a, beta)
        grad -= grad.mean()
        
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = normalize_to(a, S_target)
        
        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
        
        # Periodic random restart for diversity
        if (it + 1) % 500 == 0 and frac > 0.3:
            a = a + 0.01 * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S_target)
    
    s = compute_obj(a)
    if s < best_score:
        best = a.copy()
        best_score = s
    
    return best, best_score


def combined_optimize(a0, deadline, lp_iters=3000, grad_iters=1500):
    """Combined gradient + LP optimization."""
    n = len(a0)
    
    # Phase 1: Gradient to get to a good region
    if time.monotonic() < deadline - 60.0:
        grad_deadline = min(deadline - max(60.0, 0.4 * (deadline - time.monotonic())), 
                           time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a, s = grad_optimize(a0, grad_deadline, max_iters=grad_iters)
    else:
        a = a0.copy()
        s = compute_obj(a)
    
    # Phase 2: LP refinement (the main workhorse)
    if time.monotonic() < deadline - 10.0:
        a, s = lp_optimize(a, deadline, max_iters=lp_iters)
    
    return a, s


def make_starting_sequences(rng):
    """Generate diverse starting sequences of various lengths."""
    starts = []
    
    # Use global if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in [256, 512, 768]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Perturbed versions
            if len(gs) >= 512:
                starts.append(gs[:512] * (1.0 + 0.05 * rng.standard_normal(512)))
                starts.append(gs[:512] * (1.0 + 0.1 * rng.standard_normal(512)))
    except NameError:
        pass
    
    # Random sequences
    for n in [256, 512, 1024]:
        starts.append(rng.random(n) + 0.05)
        starts.append(rng.exponential(0.5, size=n) + 0.01)
    
    # Structured sequences
    for n in [512, 1024]:
        i = np.arange(n, dtype=np.float64)
        # Chirps
        for freq in [2, 3, 5]:
            starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * freq / n))
        # Gaussian
        starts.append(np.exp(-0.5 * ((i - n/2) / (n * 0.25)) ** 2) + 0.01)
        # Two bumps
        starts.append(np.exp(-0.5*((i-n*0.3)/(n*0.08))**2) + 
                      np.exp(-0.5*((i-n*0.7)/(n*0.08))**2) + 0.01)
        # Ramp
        starts.append(np.linspace(0.5, 1.5, n))
        # Sine wave
        starts.append(1.0 + 0.3 * np.sin(2 * np.pi * 4.0 * i / n))
    
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    main_optimize = make_starting_sequences
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 8.0)
    
    best_seq = None
    best_score = float('inf')
    
    starts = main_optimize(rng)
    starts.sort(key=lambda x: len(x))
    
    print(f"Starting optimization: {len(starts)} candidates, budget={budget_s}s")
    
    # Phase A: Quick evaluation of all starts
    time_per_quick = max(5.0, (deadline - t_start) * 0.15 / max(1, len(starts)))
    candidates = []
    
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 120.0:
            break
        t0 = time.monotonic()
        # Quick gradient optimization
        a_quick, s_quick = grad_optimize(a0, t0 + min(time_per_quick, 30.0), max_iters=300)
        candidates.append((s_quick, a_quick, len(a0)))
        if s_quick < best_score:
            best_score = s_quick
            best_seq = a_quick.copy()
            print(f"  Quick {idx}: n={len(a0)}, score={s_quick:.6f}")
    
    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    
    print(f"Top 5 candidates: {[f'{c[0]:.4f} (n={c[2]})' for c in candidates[:5]]}")
    
    # Phase B: Deep optimization on top candidates
    n_deep = min(5, len(candidates))
    remaining = deadline - time.monotonic() - 30.0
    time_per_deep = max(30.0, remaining / max(1, n_deep))
    
    for idx in range(n_deep):
        if time.monotonic() > deadline - 30.0:
            break
        s_c, a_c, n_c = candidates[idx]
        t0 = time.monotonic()
        alloc = min(time_per_deep, deadline - time.monotonic() - 15.0)
        if alloc < 10.0:
            break
        
        a_opt, s_opt = combined_optimize(a_c, t0 + alloc, 
                                          lp_iters=2000, grad_iters=1000)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  Deep {idx}: n={n_c}, score={s_opt:.6f} (was {s_c:.4f})")
        else:
            print(f"  Deep {idx}: n={n_c}, score={s_opt:.6f} (no improvement)")
    
    # Phase C: Final intensive LP refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        print(f"Final LP refinement (current best={best_score:.6f})...")
        a_final, s_final = lp_optimize(best_seq, deadline - 5.0, max_iters=5000)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"  After final LP: score={best_score:.6f}")
    
    # Phase D: One more gradient+LP combo on the best
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        a_final2, s_final2 = combined_optimize(best_seq, deadline - 5.0,
                                                lp_iters=2000, grad_iters=500)
        if s_final2 < best_score:
            best_score = s_final2
            best_seq = a_final2.copy()
            print(f"  After final combo: score={best_score:.6f}")
    
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"Final result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = evaluate_sequence([float(x) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]
    
    print("Falling back to uniform sequence")
    return [1.0] * 256
