    """Build the row of the convolution matrix for position k."""
    n = len(a)
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # For the LP, we need conv(a, b)[k] as a linear function of b
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # So the coefficient of b[i] is a[k-i] (when k-i is valid)
    row = np.zeros(n)

        if 0 <= i < n:

    """

    """
    n = len(a)
    n_constraints = 2 * n - 1
    
    # Build constraint matrix A_ub where A_ub[b] gives conv(a, b)

            if 0 <= j < n:
                grad[i] += a[j]
                if i != j:
                    grad[i] += a[i]
                else:
                    grad[i] += a[i]  # total 2*a[i]
        
        # Learning rate
        lr = 1.0 / (1.0 + 0.01 * t)
        
        # Update
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        score = score_sequence(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        t += 1
    
    if verbose:
        print(f"  Subgrad done: best={best_score:.8f}")
    return best_a, best_score

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build sparse matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    n_rows = 2 * n - 1
    rows, cols, data = [], [], []
    for k in range(n_rows):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                data.append(a[j])
    return csr_matrix((data, (np.array(rows), np.array(cols))), shape=(n_rows, n))

def solve_lp(a, M, time_limit=20):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.maximum(0, res.x)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except:
        pass
    return None

def lp_refine(a, time_limit_s=600, verbose=True):
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  LP start: score={best_score:.8f}, n={n}")
    
    stagnant = 0
    for it in range(2000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 5):
                print(f"  LP iter {it:4d}: score={score:.8f}")
        else:
            stagnant += 1
        
        if stagnant > 60:
            if verbose:
                print(f"  LP stagnant at iter {it}, best={best_score:.8f}")
            break
        
        remaining = time_limit_s - elapsed
        if remaining < 5:
            break
        
        timeout = max(2, min(30, remaining * 0.15))
        
        # Try multiple M values
        best_b = None
        for m_factor in [1.0, 1.02, 1.05, 1.1, 1.2]:
            b = solve_lp(a, M * m_factor, time_limit=timeout)
            if b is not None:
                s = b.sum()
                if s > 1.0 + 1e-8:
                    if best_b is None or s > best_b[1]:
                        best_b = (b, s)
        
        if best_b is None:
            if verbose:
                print(f"  LP: no improvement direction at iter {it}")
            break
        
        b, s = best_b
        
        # Line search
        best_alpha = 0.5
        best_new_score = score_sequence(normalize(a + best_alpha * (b - a)))
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
            candidate = normalize(a + alpha * (b - a))
            sc = score_sequence(candidate)
            if sc < best_new_score:
                best_new_score = sc
                best_alpha = alpha
        
        a = normalize(a + best_alpha * (b - a))
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  LP done: best={best_score:.8f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score

def subgrad_refine(a, time_limit_s=120, verbose=True):
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  SGD start: score={best_score:.8f}")
    
    t = 0
    for it in range(50000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of conv(a,a)[k_star]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = a[j] + a[i]
        
        lr = 1.0 / (1.0 + 0.005 * t) ** 0.7
        a_new = project_simplex(a - lr * grad)
        
        sc = score_sequence(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        t += 1
    
    if verbose:
        print(f"  SGD done: best={best_score:.8f}")
    return best_a, best_score

def random_perturb_refine(a, time_limit_s=120, rng=None, verbose=True):
    """Local search with random perturbations."""

        rng = np.random.default_rng(0)

    best_score = score_sequence(a)
    
    if verbose:
        print(f"  RPS start: score={best_score:.8f}")

        if elapsed > time_limit_s:

        width = max(2, int(rng.integers(n // 100, n // 5)))

            best_a = candidate.copy()
            a = candidate.copy()

            step_size = max(0.01, step_size * 0.99)

                print(f"  RPS accept {accepts}: score={best_score:.8f}")

            step_size = min(0.5, step_size * 1.01)
    
    if verbose:
        print(f"  RPS done: best={best_score:.8f}")
    return best_a, best_score

def resize_sequence(a, new_n):
    """Resize sequence to new length using interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    old_t = np.linspace(0, 1, n)
    new_t = np.linspace(0, 1, new_n)
    new_a = np.interp(new_t, old_t, a)
    return normalize(np.maximum(0, new_a))
