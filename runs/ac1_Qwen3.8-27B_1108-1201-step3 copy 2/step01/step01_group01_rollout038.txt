Multi-phase optimization to minimize the peak-to-average ratio of the
auto-convolution of a non-negative sequence. Combines spectral shaping,
smooth (softmax) gradient descent, and L-BFGS refinement.

import numpy as np

def _fft_full_convolve(a, b=None):
    """Full convolution of a with b (or with itself if b is None) using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = fft(a, n=size)
    fb = fft(b, n=size)
    return np.real(ifft(fa * fb, n=size))[:n]

def _compute_b(a):
    """Compute the full auto-convolution of a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, n=size)
    b = np.real(ifft(fa * fa, n=size))[:2*n-1]
    # Clean up tiny negative values from FFT roundoff
    b = np.maximum(b, 0.0)
    return b

def _score(a):
    """Compute the evaluation score."""
    n = len(a)
    b = _compute_b(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * n * np.max(b) / (s * s)

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_objective_and_grad(a, alpha):
    """
    Compute soft-max objective and its gradient w.r.t. a.
    F(a) = (1/alpha) * log(sum_k exp(alpha * b_k))
    grad_j = 2 * sum_k w_k * a_{k-j}
    """
    n = len(a)
    b = _compute_b(a)
    b_max = np.max(b)
    exp_ab = np.exp(alpha * (b - b_max))
    Z = np.sum(exp_ab)
    F = b_max + np.log(Z) / alpha
    w = exp_ab / Z
    
    # Gradient: grad_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * correlate(w, a, 'full')[:n]
    # Using FFT: correlate(w, a) = ifft(fft(w) * conj(fft(a))) for circular, but we need linear
    # Use convolution: correlate(w, a, 'full')[j] = convolve(w, a[::-1], 'full')[j + n - 1]
    c = _fft_full_convolve(w, a[::-1])[n-1:2*n-1]
    grad = 2.0 * c
    
    return F, grad, b

def _optimize_softmax(a_init, deadline, alpha_start=1.0, alpha_end=50.0, 
                       lr_start=0.5, lr_end=0.01, n_max_iter=50000):
    """
    Softmax gradient descent with annealing.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    vel = np.zeros(n)
    momentum = 0.9
    total_iter = n_max_iter
    
    for it in range(total_iter):
        if time.monotonic() > deadline:
            break
        
        # Annealing schedule
        t = it / total_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        
        F, grad, b = _softmax_objective_and_grad(a, alpha)
        
        # Center the gradient (for sum constraint)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        vel = momentum * vel + (1.0 - momentum) * grad_centered
        a = a - lr * vel / (np.linalg.norm(vel) + 1e-15)
        
        # Project
        a = _project_simplex(a, 1.0)
        
        # Check improvement periodically
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_score - 1e-9:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def _spectral_shaping(a_init, deadline, n_iters=500, beta=0.3):
    """
    Spectral shaping: iterate to flatten |A(omega)|^2.
    """
    n = len(a_init)
    a = a_init.copy()
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        A = fft(a, n=fft_size)
        power = np.abs(A) ** 2 + 1e-30
        
        # Log-domain: target flat log-power
        log_p = np.log(power)
        mean_log_p = np.mean(log_p)
        
        # Reduce peaks in log-spectrum
        correction = np.exp(-beta * (log_p - mean_log_p))
        A_new = A * correction
        
        a = np.real(ifft(A_new, n=fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-15:
            break
        a /= s
    
    return a

def _lbfgs_optimize(a_init, deadline, alpha=10.0, maxiter=500):

    Use scipy L-BFGS-B to optimize the softmax objective in v-space.

    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    a = a / s
    
    # Initialize v from a
    v0 = np.log(a + 1e-30)
    v0 -= np.max(v0)
    
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1
    n_b = 2 * n - 1
    
    call_count = [0]
    
    def _obj_grad(v):
        call_count[0] += 1
        if time.monotonic() > deadline:
            raise ValueError("Timeout")
        
        # Compute a from v
        v_shifted = v - np.max(v)
        exp_v = np.exp(v_shifted)
        a = exp_v / np.sum(exp_v)
        
        # Compute b
        fa = fft(a, n=fft_size)
        b = np.real(ifft(fa * fa, n=fft_size))[:n_b]
        b = np.maximum(b, 0.0)
        
        # Softmax
        b_max = np.max(b)
        exp_ab = np.exp(alpha * (b - b_max))
        Z = np.sum(exp_ab)
        F = b_max + np.log(Z) / alpha
        w = exp_ab / Z
        
        # Gradient w.r.t. a
        c = _fft_full_convolve(w, a[::-1])[n-1:2*n-1]
        grad_a = 2.0 * c
        
        # Gradient w.r.t. v
        weighted_sum = np.dot(grad_a, a)
        grad_v = a * (grad_a - weighted_sum)
        
        return F, grad_v
    
    try:
        result = minimize(_obj_grad, v0, method='L-BFGS-B', 
                         jac=True, options={'maxiter': maxiter, 'maxfun': maxiter*10,
                                           'ftol': 1e-12, 'gtol': 1e-8})
        v_opt = result.x
        v_opt -= np.max(v_opt)
        a_opt = np.exp(v_opt)
        a_opt /= np.sum(a_opt)
    except (ValueError, Exception) as e:
        a_opt = a
    
    return a_opt, _score(a_opt)

def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 4), size=min(10, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, np.sum(a))
    return a_new

Multi-phase optimization to minimize the peak auto-convolution of a non-negative sequence.
Combines spectral shaping (fast FFT-based), softmax gradient descent with annealing,
L-BFGS refinement, and diversification to escape local minima.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def _fft_full_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = fft(a, n=size)
    fb = fft(b, n=size)
    return np.real(ifft(fa * fb, n=size))[:n]


def _compute_b(a):
    """Compute full auto-convolution of a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, n=size)
    b = np.real(ifft(fa * fa, n=size))[:2 * n - 1]
    return np.maximum(b, 0.0)


def _score(a):
    """Compute the evaluation score: 2*n*max(b)/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    b = _compute_b(a)
    return 2.0 * n * np.max(b) / (s * s)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    r = idx[-1]
    theta = (cssv[r] - target) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj_grad_a(a, alpha):
    """
    Compute soft-max objective and gradient w.r.t. a (unconstrained).
    F = (1/alpha)*log(sum_k exp(alpha*b_k))
    dF/da_j = 2 * sum_k w_k * a_{k-j} = 2 * correlate(w,a,'full')[j]
    """
    n = len(a)
    b = _compute_b(a)
    b_max = np.max(b)
    exp_ab = np.exp(alpha * (b - b_max))
    Z = np.sum(exp_ab)
    F = b_max + np.log(Z) / alpha
    w = exp_ab / Z
    # c_j = correlate(w, a, 'full')[j] = convolve(w, a[::-1], 'full')[j]
    c = _fft_full_convolve(w, a[::-1])[:n]
    grad = 2.0 * c
    return F, grad, b, w


def _phase_spectral_shaping(a_init, deadline, n_iters=300, beta=0.3):
    """
    Iterative spectral shaping: modify |A(omega)| to be flatter.
    Each iteration is O(n log n).
    """
    n = len(a_init)
    a = a_init.copy()
    fft_size = 1
    while fft_size < 2 * n:
        fft_size <<= 1

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = fft(a, n=fft_size)
        power = np.abs(A) ** 2 + 1e-30
        log_p = np.log(power)
        mean_lp = np.mean(log_p)

        # Flatten the log-power spectrum
        correction = np.exp(-beta * (log_p - mean_lp))
        A_new = A * correction

        a = np.real(ifft(A_new, n=fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-15:
            break
        a /= s

    return a


def _phase_softmax_gd(a_init, deadline, alpha_start=0.5, alpha_end=30.0,
                      lr_start=0.3, lr_end=0.005, momentum=0.9):
    """
    Softmax gradient descent with temperature annealing and momentum.
    Projects onto simplex after each step.
    """
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_score = _score(best_a)

    vel = np.zeros(n)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1
        t = min(1.0, iter_count / 20000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        F, grad, b, w = _softmax_obj_grad_a(a, alpha)

        # Center gradient for sum constraint
        grad_c = grad - np.mean(grad)

        # Momentum
        vel = momentum * vel + (1.0 - momentum) * grad_c

        # Adaptive step: normalize by gradient magnitude
        gnorm = np.linalg.norm(vel)
        if gnorm < 1e-15:
            break
        step = lr / gnorm

        a_new = a - step * vel
        a_new = _project_simplex(a_new, 1.0)

        if np.sum(a_new) < 1e-15:
            break

        # Check improvement every 200 iterations
        if iter_count % 200 == 0:
            sc = _score(a_new)
            if sc < best_score - 1e-9:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            else:
                no_improve += 1
                a = a_new
                if no_improve >= 30:
                    # Cool down
                    lr *= 0.7
                    vel *= 0.5
                    no_improve = 0
                    if lr < 1e-10:
                        break

    return best_a, best_score

def _phase_peak_reduction(a_init, deadline):
    """
    Targeted peak reduction: identify top peaks in b, and reduce them
    by adjusting a at contributing positions.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(best_a)
    lr = 0.05 / np.sqrt(n)

    for outer in range(200):
        if time.monotonic() > deadline:
            break

        b = _compute_b(a)
        b_max = np.max(b)

        # Get top 20 peak locations
        k_top = np.argpartition(b, -20)[-20:]
        k_top = k_top[np.argsort(b[k_top])[::-1]]

        # Compute weighted gradient focusing on top peaks
        b_range = b_max - b.min()
        wts = np.exp(-3.0 * (b_max - b[k_top]) / max(b_range, 1e-15))
        wts /= np.sum(wts)

        # Build weight vector on full b grid
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(k_top):
            w_full[k] = wts[i]

        # Gradient: grad_j = 2 * correlate(w_full, a, 'full')[j]
        c = _fft_full_convolve(w_full, a[::-1])[:n]
        grad = 2.0 * c
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a = a - lr * grad / gnorm
        a = _project_simplex(a, 1.0)

        sc = _score(a)
        if sc < best_score - 1e-9:
            best_score = sc
            best_a = a.copy()

        #anneal lr
        if outer % 50 == 0:
            lr *= 0.8
            if lr < 1e-10:
                break

    return best_a, best_score


def _perturb_smooth(a, rng, scale=0.03):
    """Add smooth low-frequency random perturbation."""
    n = len(a)
    n_freqs = min(8, n // 2)
    freqs = rng.integers(1, max(2, n // 3), size=n_freqs)
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    max_n = np.max(np.abs(noise))
    if max_n > 1e-15:
        noise /= max_n
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, np.sum(a))
    return a_new


def _make_two_level(n, eps):
    """Create a two-level anti-symmetric sequence."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)


def _make_four_level(n, eps):
    """Four-level anti-symmetric sequence."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    return a / np.sum(a)
