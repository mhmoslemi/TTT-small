"""
Iterated LP best-response with active-set refinement and multi-length exploration
for minimizing the peak of self-convolution of non-negative sequences.

Key ideas:
- The score 2*n*max(conv(a,a))/sum(a)^2 is scale-invariant; normalize to sum=1.
- Iterated LP best-response: given a, solve max sum(b) s.t. conv(a,b)<=M, b>=0.
- Active-set QP refinement on binding lags.
- Multi-length exploration via interpolation.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_len_conv(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_self(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0)
    L = _fft_len_conv(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def conv_cross(a, b):
    """Compute conv(a, b) using FFT."""
    n = max(int(a.size), int(b.size))
    L = _fft_len_conv(n)
    Fa = fft(a, L)
    Fb = fft(b, L)
    return np.real(ifft(Fa * Fb))[:int(a.size) + int(b.size) - 1]


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = conv_self(p)
    cmax = float(np.max(c)) if c.size else 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(max(1, p.size), 1.0 / max(1, p.size), dtype=np.float64)
    return p / s


def build_conv_matrix(a, n):
    """
    Build the (2n-1) x n sparse convolution matrix A such that A @ b = conv(a, b).
    A[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0.
    """
    a = np.asarray(a, dtype=np.float64)
    rows = np.empty(0, dtype=np.int32)
    cols = np.empty(0, dtype=np.int32)
    data = np.empty(0, dtype=np.float64)

    for k in range(2 * n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            i_vals = np.arange(i_lo, i_hi + 1, dtype=np.int32)
            r_vals = np.full(i_hi - i_lo + 1, k, dtype=np.int32)
            d_vals = a[k - i_vals].astype(np.float64)
            rows = np.concatenate([rows, r_vals])
            cols = np.concatenate([cols, i_vals])
            data = np.concatenate([data, d_vals])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_response(a, M, time_budget=5.0):
    """
    Solve LP: max sum(b) s.t. A*b <= M, b >= 0.
    Returns b (normalized to sum 1) and dual variables y.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0 or M <= 0:
        return None, None

    A = build_conv_matrix(a, n)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"time_limit": time_budget, "disp": False},
        )
    except Exception:
        return None, None

    if res is None or res.status not in (0, 3):
        return None, None

    b = np.asarray(res.x, dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-12:
        return None, None
    b = b / s

    # Extract dual variables (y) from the LP result
    y = None
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        y = np.asarray(res.ineqlin, dtype=np.float64)
    elif hasattr(res, 'dualv') and res.dualv is not None:
        y = np.asarray(res.dualv, dtype=np.float64)

    return b, y


def smooth_max_gradient(p, T_factor=0.1):
    """
    Compute gradient of smooth max of conv(p,p) using logsumexp surrogate.
    Returns (gradient, convolution, cmax, temperature).
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 0.0, 1.0

    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if cmax <= 0:
        return np.zeros(n), c, 0.0, 1.0

    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if raw <= 0 or not np.isfinite(raw):
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = e / raw

    # Gradient: grad_j = 2 * sum_k weights[k] * p[k-j] (for valid k-j)
    # This is 2 * conv(weights_reversed, p) sliced appropriately
    Fw = fft(weights[::-1], L)
    h = np.real(ifft(Fw * Fp))
    # h has length L; we need h[n-1-n : n-1+1] = h[0 : n] reversed? 
    # Actually conv(weights, p) at lag k = sum_i weights[i] * p[k-i]
    # We need: for each j, sum_k weights[k] * p[k-j]
    # = (weights * p_reversed) at the appropriate lag
    # Let's use: grad_j = 2 * sum_k w[k] * p[k-j] for 0<=k-j<n
    # This is the cross-correlation of w and p
    # cross_corr[j] = sum_k w[k] * p[k-j]
    # In FFT terms: ifft(fft(w) * conj(fft(p))) gives cross-correlation
    # But we have full convolution on a finite support.
    
    # Simpler: just compute directly using the convolution structure
    # grad_j = 2 * sum_{k: 0<=k-j<n} w[k] * p[k-j]
    # Let m = k-j, so k = j+m, m in [0, n-1], k in [j, j+n-1] ∩ [0, 2n-2]
    # grad_j = 2 * sum_{m=0}^{n-1} w[j+m] * p[m]  (where w[j+m]=0 if j+m >= 2n-1)
    
    g = np.zeros(n, dtype=np.float64)
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo > k_hi:
            continue
        m_range = np.arange(0, k_hi - k_lo + 1)
        g[j] = 2.0 * float(np.dot(weights[k_lo:k_hi+1], p[m_range[:k_hi-k_lo+1]]))
    
    return g, c, cmax, T

        a_try = project_simplex(a_try)
        sc = internal_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc, y


def active_set_refine(a, best_score, time_budget=30.0):
    """
    Refine using active-set: identify top lags and do gradient steps to equalize them.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    c = conv_self(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return a, best_score

    # Identify active set: lags where c[k] > 0.95 * cmax
    threshold = 0.90 * cmax
    active = np.where(c > threshold)[0]
    if active.size == 0:
        active = np.where(c > 0.8 * cmax)[0]
    if active.size == 0:
        return a, best_score

    # Gradient of max over active set (subgradient = uniform over active set)
    # grad_j = (2/|S|) * sum_{k in S} [a[k-j] + a[j-k] (if valid)]
    # Actually for c_k = sum_i a[i]*a[k-i], dc_k/da_j = a[k-j] + a[j-k] (with a[i]=0 if i out of range)
    
    end = time.monotonic() + time_budget
    current_a = a.copy()
    current_sc = best_score
    lr_base = 2.0 / (n * n)  # small learning rate
    
    for it in range(2000):
        if time.monotonic() > end:
            break
        
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            # dc_k/da_j for each j
            # c_k = sum_i a[i] a[k-i], valid i: max(0, k-n+1) <= i <= min(k, n-1)
            # dc_k/da_j = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            for j in range(n):
                val = 0.0
                km = k - j
                if 0 <= km < n:
                    val += current_a[km]
                jk = j - k
                if 0 <= jk < n:
                    val += current_a[jk]
                g[j] += val
        g /= float(len(active))
        
        # Try a few learning rates
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = lr_base * lr_mult
            a_try = current_a - lr * g
            a_try = project_simplex(a_try)
            sc = internal_score(a_try)
            if sc < current_sc - 1e-12:
                current_a = a_try
                current_sc = sc
                break
        else:
            # Decrease learning rate or break
            lr_base *= 0.5
            if lr_base < 1e-15:
                break
    
    if current_sc < best_score:
        best_score = current_sc
        a = current_a
    return a, best_score


def gradient_polish(a, time_budget=60.0, T_start=0.3, T_end=0.02):
    """
    Polish using smoothed-max gradient descent.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    best_a = a.copy()
    best_sc = internal_score(a)
    end = time.monotonic() + time_budget
    it = 0
    T = T_start
    lr = 5.0 / n
    
    while time.monotonic() < end:
        it += 1
        T = T_start * (T_end / T_start) ** (it / max(1, int(time_budget * 2)))
        T = max(T, T_end)
        
        g, c, cmax, _ = smooth_max_gradient(a, T)
        if not np.all(np.isfinite(g)):
            break
        
        accepted = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_try = a - lr_try * g
            a_try = project_simplex(a_try)
            sc = internal_score(a_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_try.copy()
                a = a_try
                accepted = True
                break
            elif sc < internal_score(a):
                a = a_try
                accepted = True
                break
        if not accepted:
            lr *= 0.5
            if lr < 1e-12:
                break
        
        if it % 50 == 0:
            print("  grad_polish it=%d T=%.4f best=%.10f" % (it, T, best_sc), flush=True)
    
    return best_a, best_sc


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    a = np.asarray(a, dtype=np.float64)
    old_n = int(a.size)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return safe_normalize(a_new)


def make_initial_candidates(n, parent=None):
    """Generate diverse initial candidates."""
    candidates = []
    if parent is not None and parent.size > 10:
        candidates.append(safe_normalize(parent))
    
    # Uniform
    candidates.append(np.full(n, 1.0 / n))
    
    # Linear
    x = np.linspace(0, 1, n)
    for a in [0.5, 1.0, 2.0, 5.0]:
        candidates.append(safe_normalize(1.0 + a * x))
        candidates.append(safe_normalize(1.0 - a * x))
    
    # Power
    for a in [0.5, 1.0, 2.0, 4.0]:
        candidates.append(safe_normalize(np.power(x + 1e-10, a)))
        candidates.append(safe_normalize(np.power(1.0 - x + 1e-10, a)))
    
    # U-shaped (arcsine-like)
    eps = 1e-5
    candidates.append(safe_normalize(1.0 / np.sqrt(x + eps) * 1.0 / np.sqrt(1.0 - x + eps)))
    
    # Gaussian centered
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        candidates.append(safe_normalize(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)))
    
    # Twopeaked
    for mu in [0.2, 0.3, 0.35]:
        p = np.exp(-0.5 * ((x - mu) / 0.1) ** 2) + np.exp(-0.5 * ((x - (1-mu)) / 0.1) ** 2)
        candidates.append(safe_normalize(p))
    
    # Semicircle
    candidates.append(safe_normalize(np.sqrt(1.0 - (2*x - 1.0)**2 + 1e-10)))
    
    return candidates


def propose_candidate():
    """Search for a nonnegative step function with low evaluate_sequence score."""
    start_time = time.monotonic()
    total_budget = 1800.0
    deadline = start_time + total_budget - 15.0  # reserve 15 sec for cleanup
    
    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32:
            parent = safe_normalize(parent_arr)
            print("Parent loaded: n=%d, score=%.10f" % (parent.size, internal_score(parent)), flush=True)
    except Exception:
        pass
    
    best_p = None
    best_score = np.inf
    
    # Determine working length
    if parent is not None:
        base_n = int(parent.size)
    else:
        base_n = 700
    
    # Generate candidates at base length
    candidates = make_initial_candidates(base_n, parent)
    
    # Score all candidates
    scored = []
    for c in candidates:
        c = safe_normalize(c)
        sc = internal_score(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])
    
    print("Initial candidates scored. Best: %.10f" % scored[0][0], flush=True)
    
    for sc, c in scored[:3]:
        if sc < best_score:
            best_score = sc
            best_p = c.copy()
    
    # Phase 1: Quick gradient polish on the best candidate
    print("Phase 1: Gradient polish (60s)", flush=True)
    if deadline - time.monotonic() > 300.0:
        p_polished, sc_polished = gradient_polish(best_p, time_budget=60.0)
        if sc_polished < best_score:
            best_score = sc_polished
            best_p = p_polished
        print("  After gradient polish: %.10f" % best_score, flush=True)
    
    # Phase 2: Iterated LP best-response (main workhorse)
    print("Phase 2: Iterated LP best-response", flush=True)
    a = best_p.copy()
    current_score = best_score
    n = int(a.size)
    
    # Adaptive LP budget based on sequence length
    if n <= 500:
        lp_time = 3.0
    elif n <= 1000:
        lp_time = 5.0
    elif n <= 2000:
        lp_time = 8.0
    else:
        lp_time = 12.0
    
    lp_count = 0
    no_improve_count = 0
    stalled_its = 0
    
    while time.monotonic() < deadline - 180.0:
        c_cur = conv_self(a)
        M = float(np.max(c_cur))
        
        new_a, new_sc, y_dual = lp_iteration(a, current_score, time_budget=lp_time)
        lp_count += 1
        
        if new_sc < current_score:
            current_score = new_sc
            a = new_a
            no_improve_count = 0
            if current_score < best_score:
                best_score = current_score
                best_p = a.copy()
        else:
            no_improve_count += 1
            stalled_its += 1
        
        # Occasionally try gradient polish to escape
        if no_improve_count > 15 and time.monotonic() < deadline - 180.0:
            p_pol, sc_pol = gradient_polish(a, time_budget=20.0, T_start=0.2, T_end=0.05)
            if sc_pol < current_score:
                current_score = sc_pol
                a = p_pol
                no_improve_count = 0
                stalled_its = 0
                if current_score < best_score:
                    best_score = current_score
                    best_p = a.copy()
            else:
                # Try random perturbation
                sig = 0.3 / n
                a_pert = project_simplex(a + np.random.default_rng(lp_count * 7919).normal(0, sig, n))
                sc_pert = internal_score(a_pert)
                if sc_pert < current_score:
                    current_score = sc_pert
                    a = a_pert
                    no_improve_count = 0
                    stalled_its = 0
                    if current_score < best_score:
                        best_score = current_score
                        best_p = a.copy()
        
        # Active set refinement every 20 LP iterations
        if lp_count % 20 == 0 and time.monotonic() < deadline - 180.0:
            a_act, sc_act = active_set_refine(a, current_score, time_budget=10.0)
            if sc_act < current_score:
                current_score = sc_act
                a = a_act
                if current_score < best_score:
                    best_score = current_score
                    best_p = a.copy()
        
        if lp_count % 10 == 0:
            print("  LP it=%d M=%.6e score=%.10f best=%.10f" % (lp_count, M, current_score, best_score), flush=True)
        
        # Early stopping if score is very good
        if best_score < 1.5050 and no_improve_count > 30:
            print("  Early stop: score=%.10f" % best_score, flush=True)
            break
    
    print("Phase 2 done. best=%.10f after %d LP its" % (best_score, lp_count), flush=True)
    
    # Phase 3: Final gradient polish
    print("Phase 3: Final gradient polish", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        polish_time = min(120.0, remaining * 0.5)
        p_fin, sc_fin = gradient_polish(best_p, time_budget=polish_time, T_start=0.15, T_end=0.01)
        if sc_fin < best_score:
            best_score = sc_fin
            best_p = p_fin
        print("  After final polish: %.10f" % best_score, flush=True)
    
    # Phase 4: Multi-length exploration
    print("Phase 4: Multi-length exploration", flush=True)
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        test_lengths = [500, 800, 1000, 1200, 1500, 2000]
        # Filter to reasonable lengths
        test_lengths = [L for L in test_lengths if L != base_n and L >= 100]
        
        for test_n in test_lengths:
            if time.monotonic() > deadline - 30.0:
                break
            t_rem = deadline - time.monotonic()
            if t_rem < 20.0:
                break
            
            # Interpolate to new length
            a_interp = interpolate_sequence(best_p, test_n)
            sc_interp = internal_score(a_interp)
            
            # Quick gradient polish + few LP iters
            time_per_len = min(40.0, t_rem * 0.5 / max(1, len(test_lengths)))
            a_pol, sc_pol = gradient_polish(a_interp, time_budget=min(20.0, time_per_len * 0.5))
            
            # A few LP iterations
            a_lp = a_pol.copy()
            sc_lp = sc_pol
            if time.monotonic() < deadline - 30.0:
                for _ in range(5):
                    a_lp, sc_lp, _ = lp_iteration(a_lp, sc_lp, time_budget=3.0)
                    if time.monotonic() > deadline - 30.0:
                        break
            
            sc_best_len = min(sc_pol, sc_lp)
            if sc_best_len < best_score:
                best_score = sc_best_len
                a_final = a_lp if sc_lp < sc_pol else a_pol
                best_p = a_final
                print("  New best at n=%d: %.10f" % (test_n, best_score), flush=True)
            else:
                print("  n=%d: %.10f (no improvement)" % (test_n, sc_best_len), flush=True)
    
    # Phase 5: Final LP burst
    remaining = deadline - time.monotonic()
    if remaining > 60.0 and best_p is not None:
        print("Phase 5: Final LP burst", flush=True)
        a = best_p.copy()
        sc = best_score
        for _ in range(20):
            if time.monotonic() > deadline - 15.0:
                break
            a, sc, _ = lp_iteration(a, sc, time_budget=3.0)
            if sc < best_score:
                best_score = sc
                best_p = a
        print("  Final score: %.10f" % best_score, flush=True)
    
    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(max(1, base_n), 1.0 / max(1, base_n))
    
    # Scale to a reasonable range (avoid very small numbers that might cause issues)
    best_p = np.minimum(best_p, 1000.0)
    
    print("Final: n=%d score=%.10f" % (best_p.size, internal_score(best_p)), flush=True)
    print("Total time: %.1f sec" % (time.monotonic() - start_time), flush=True)
    
    return [float(x) for x in best_p]
