Multi-phase optimization for minimizing peak autocorrelation of non-negative step functions.
Uses scipy L-BFGS-B with annealed log-sum-exp smoothing on softmax-parameterized sequences.
Employs multi-stage temperature cooling with refinement loops, diverse initializations,

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)

    # Score gradient w.r.t. a
    score_grad_a = 2.0 * n * grad_a

    # Chain rule through softmax: d(score)/d(theta[k]) = a[k]*(score_grad_a[k] - dot(score_grad_a, a))
    inner = np.dot(score_grad_a, a)
    grad_theta = a * (score_grad_a - inner)

    return score, grad_theta


def optimize_sequence(theta_init, tau_start, tau_end, time_budget, max_iter_per_stage=200, n_refinements=3):
    """
    Multi-stage L-BFGS-B optimization with annealed temperature and refinement loops.
    """
    t0 = time.monotonic()
    best_score = np.inf
    best_theta = theta_init.copy()
    n = len(theta_init)

    # Generate tau schedule (log-spaced from tau_start to tau_end)
    base_stages = 8
    n_stages = base_stages * n_refinements
    tau_schedule = np.exp(np.linspace(np.log(tau_start), np.log(tau_end), n_stages))

    theta = theta_init.copy()

    for i in range(n_stages):
        if time.monotonic() - t0 > time_budget:
            break
        tau = tau_schedule[i]

        def obj(t):
            val, _ = smooth_objective(t, tau)
            return val

        def grad(t):
            _, g = smooth_objective(t, tau)
            return g

        try:
            result = minimize(obj, theta, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter_per_stage, 'gtol': 1e-10,
                                       'ftol': 1e-12, 'maxfun': max_iter_per_stage * 2})
            theta = result.x
        except Exception:
            pass

        # Check actual score
        a = softmax_normalize(theta)
        actual = compute_score(a)
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            best_theta = theta.copy()

    return best_theta, best_score


def make_init_from_sequence(seq, n, rng):
    """Resample a sequence to length n."""
    if len(seq) == n:
        return seq.copy().astype(np.float64)
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq).astype(np.float64)


def make_random_init(n, rng):
    """Random positive initialization."""
    return np.clip(rng.standard_normal(n) * 0.3 + 1.0, 0.05, None)


def make_chirp_init(n, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.4 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_edge_init(n, rng):
    """Edge-weighted initialization (more mass at edges)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    spread = n * (0.3 + 0.4 * rng.random())
    a = 1.0 + 0.5 * np.cosh((t - center) / spread)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * (0.15 + 0.3 * rng.random())
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 0.1
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multi_tone_init(n, rng):
    """Multi-tone sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_tones = rng.integers(3, 8)
    for _ in range(n_tones):
        k = rng.integers(1, max(2, n // 4))
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    return np.clip(a, 0.05, None)


def make_geometric_init(n, rng):
    """Geometric (exponential decay) initialization."""
    r = 1.0 + rng.uniform(-0.05, 0.08)
    t = np.arange(n, dtype=np.float64)
    a = r ** t
    # Also try reversed
    if rng.random() < 0.5:
        a = a[::-1]
    a /= np.max(a)
    a += 0.05
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def local_polish(a, time_budget, rng):
    """
    Local search polish with targeted peak reduction and coordinate descent.
    Works directly on the non-negative sequence (not log-parametrized).
    """
    current = a.astype(np.float64).copy()
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    # Simulated annealing with targeted perturbations
    temperature = 0.001
    cooling = 0.9995
    iteration = 0

    while time.monotonic() - t0 < time_budget:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted: reduce elements contributing most to peak
            C = fft_convolve(current, current)
            peak_idx = int(np.argmax(C))
            # Contributions to C[peak_idx]: a[i]*a[peak_idx-i]
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(2, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Increase some non-peak contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 2:
                n_inc = min(5, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))

        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))

        elif r < 0.65:
            # Coordinate descent step on a random subset
            n_coords = min(n // 4, 50)
            coords = rng.choice(n, size=n_coords, replace=False)
            for j in coords:
                # Try small multiplicative changes
                for factor in [0.95, 0.97, 1.03, 1.05]:
                    test = candidate.copy()
                    test[j] *= factor
                    if np.max(test) < 1000 and np.min(test) >= 0:
                        s = compute_score(test)
                        if s < candidate_score if 'candidate_score' in dir() else compute_score(candidate):
                            candidate = test
                            break

        elif r < 0.80:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.005, size=n))

        elif r < 0.90:
            # Smooth perturbation (Gaussian kernel)
            noise = rng.standard_normal(n)
            # Smooth it
            from scipy.ndimage import gaussian_filter1d
            noise = gaussian_filter1d(noise, sigma=max(1.0, n / 20.0))
            candidate = current * np.exp(0.01 * noise)

        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        iteration += 1

        if iteration % 1000 == 0:
            print(f"    polish iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, t={time.monotonic()-t0:.1f}s")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Multi-start L-BFGS-B optimization
    print("\n=== Phase 1: L-BFGS-B Optimization ===")

    # Configuration: (length, num_starts, initialization_types)
    configs = [
        (256, 4, ['provided', 'random', 'chirp', 'edge']),
        (384, 3, ['provided', 'random', 'multi_tone']),
        (512, 3, ['provided', 'chirp', 'edge']),
        (768, 2, ['provided', 'random']),
        (1024, 2, ['provided', 'chirp']),
        (128, 2, ['random', 'geometric']),
    ]

    for n, n_inits, init_types in configs:
        if time.monotonic() > deadline - 60:
            break
        time_left = deadline - time.monotonic()
        time_per_init = max(5.0, time_left / (n_inits * 1.5))

        for init_i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            init_type = init_types[min(init_i, len(init_types) - 1)]

            if init_type == 'provided' and start_seq is not None:
                a0 = make_init_from_sequence(start_seq, n, rng)
            elif init_type == 'chirp':
                a0 = make_chirp_init(n, rng)
            elif init_type == 'edge':
                a0 = make_edge_init(n, rng)
            elif init_type == 'gaussian':
                a0 = make_gaussian_init(n, rng)
            elif init_type == 'multi_tone':
                a0 = make_multi_tone_init(n, rng)
            elif init_type == 'geometric':
                a0 = make_geometric_init(n, rng)
            else:
                a0 = make_random_init(n, rng)

            a0 = np.clip(a0, 0.01, 1000.0)
            theta_init = np.log(a0 / np.sum(a0) + 1e-10)

            # Temperature schedule: start smooth, end sharp
            tau_start = 50.0 / n
            tau_end = 0.2 / n

            t_start = time.monotonic()
            theta_best, grad_score = optimize_sequence(
                theta_init, tau_start, tau_end,
                time_budget=min(time_per_init, time_left),
                max_iter_per_stage=150,
                n_refinements=2
            )
            elapsed = time.monotonic() - t_start

            a_best = softmax_normalize(theta_best)
            # Scale doesn't matter for score, keep sum=1 for now
            actual_score = compute_score(a_best)
            print(f"  n={n}, init='{init_type}', score={actual_score:.6f}, time={elapsed:.1f}s")

            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Refine best candidates with lower temperature
    print(f"\n=== Phase 2: Refinement ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)
        theta_best = np.log(best_seq / np.sum(best_seq) + 1e-10)

        # Refine with much lower temperature
        tau_start = 5.0 / n_best
        tau_end = 0.05 / n_best
        refine_time = min(180.0, max(20.0, deadline - time.monotonic() - 60))

        t_start = time.monotonic()
        theta_refined, refine_score = optimize_sequence(
            theta_best, tau_start, tau_end,
            time_budget=refine_time,
            max_iter_per_stage=300,
            n_refinements=4
        )
        a_refined = softmax_normalize(theta_refined)
        actual_refine_score = compute_score(a_refined)
        print(f"  After refinement: {actual_refine_score:.6f} (time={time.monotonic()-t_start:.1f}s)")

        if actual_refine_score < best_score:
            best_score = actual_refine_score
            best_seq = a_refined.copy()
            print(f"    *** New best after refinement: {best_score:.6f} ***")

    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local Search Polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, deadline - time.monotonic() - 10))
        print(f"  Starting polish, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = local_polish(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve: {polish_score:.6f}")

    # Phase 4: Try to improve with direct coordinate optimization
    print(f"\n=== Phase 4: Coordinate Optimization ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a = best_seq.copy()
        n = len(a)
        coord_time = min(120.0, deadline - time.monotonic() - 10)
        t_start = time.monotonic()
        improved = True
        while improved and time.monotonic() - t_start < coord_time:
            improved = False
            current_score = compute_score(a)
            # Try adjusting each element
            for j in range(n):
                if time.monotonic() - t_start > coord_time:
                    break
                orig = a[j]
                for factor in [0.9, 0.95, 1.05, 1.1]:
                    a[j] = orig * factor
                    if a[j] < 0 or a[j] > 1000:
                        a[j] = orig
                        continue
                    s = compute_score(a)
                    if s < current_score - 1e-10:
                        current_score = s
                        improved = True
                        break
                else:
                    a[j] = orig
            if improved:
                if current_score < best_score:
                    best_score = current_score
                    best_seq = a.copy()
                    print(f"    coord opt improved: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL RESULTS ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"  Best score: {final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
