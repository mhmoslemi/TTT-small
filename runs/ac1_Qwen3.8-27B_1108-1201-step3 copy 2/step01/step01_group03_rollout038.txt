LP-based iterative optimization with adaptive step size, diverse initializations,

import numpy as np

from scipy.optimize import linprog

def fast_score(f):
    """Compute the objective score quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * np.max(conv) / (s * s))


def build_A(n, f):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] for valid k-j."""
    # For each j, the valid k range is [j, j+n-1] (since k-j must be in [0, n-1])
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # k-j in [0, n-1] means k in [j, j+n-1]
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        r = np.arange(k_start, k_end + 1)
        rows.extend(r)
        cols.extend([j] * len(r))
        vals.extend(f[0:n - k_start + k_end - j + 1])  # f[k-j] for k in range
    # Hmm, let me redo this more carefully
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # For fixed j: k-j ranges from 0 to n-1, so k ranges from j to j+n-1
    # But k must also be in [0, 2n-2]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_lo = max(0, j)  # k >= j (since k-j >= 0)
        k_hi = min(2*n-2, j+n-1)  # k <= j+n-1 (since k-j <= n-1)
        for k in range(k_lo, k_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=2.0):
    """Solve max sum(g) s.t. A @ g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_A(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-6,
                               'primal_feasibility_tolerance': 1e-6})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_step(f, t, time_limit=2.0):
    """Perform one LP-based step. Returns new f or None if no improvement."""
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    if M < 1e-15:
        return None
    g = solve_lp(f, M, time_limit)
    if g is None:
        return None
    g = g / g.sum()
    if np.linalg.norm(g - f) < 1e-10:
        return None
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return None
    return f_new / s


def subgradient_step(f, eta=0.05):
    """Subgradient descent step for max conv."""
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    n = len(f)
    # Subgradient: g[m] = 2*f[k_star - m] if 0 <= k_star-m < n, else 0
    g = np.zeros(n)
    for m in range(n):
        r = k_star - m
        if 0 <= r < n:
            g[m] = 2.0 * f[r]
    f_new = f - eta * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def make_start(n, method, rng):
    """Create initial sequence of length n."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x/2)**2  # cos^2 from 1 to 0
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)**2  # 0 to 1 to 0, peak in middle
    elif method == 'ramp':
        a = np.arange(1, n+1, dtype=np.float64)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n//8)
        a = np.exp(-0.5*((x - n//4)/s)**2) + np.exp(-0.5*((x - 3*n//4)/s)**2)
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'perturbed_uniform':
        a = 1.0 + rng.normal(0, 0.3, n)
    elif method == 'geometric':
        r = 0.95
        a = r ** np.arange(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def block_perturb(f, rng, block_width=None, sigma=0.3):
    """Random block multiplicative perturbation."""
    n = len(f)
    if block_width is None:
        block_width = max(1, min(n//4, int(rng.integers(1, n//2+1))))
    start = int(rng.integers(0, max(1, n - block_width)))
    end = min(n, start + block_width)
    factor = np.exp(rng.normal(0, sigma, end - start))
    f_new = f.copy()
    f_new[start:end] *= factor
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s

    t = t_init
    best = f.copy()

    it = 0
    rng = np.random.default_rng(12345)

        f_new = lp_step(f, t, lp_time)

            f_new = subgradient_step(f, eta=0.01 * t)
        
        if it % perturb_every == 0:

            f_pert = block_perturb(f, rng, sigma=perturb_sigma)
            sc_pert = fast_score(f_pert)

We want: conv(f, g)[k] <= M for all k.

"""
LP-based iterative optimization with adaptive step size, subgradient fallback,
diverse initializations, and line search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses precomputed sparsity patterns for efficiency.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute the objective score quickly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * np.max(conv) / (s * s))


def build_A_pattern(n):
    """Build sparsity pattern for convolution matrix A[k,j]=f[k-j]."""
    rows = []
    cols = []
    for j in range(n):
        k_lo = j
        k_hi = j + n - 1  # always <= 2n-2 since j <= n-1
        rows.extend(range(k_lo, k_hi + 1))
        cols.extend([j] * n)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_A_from_data(n, f, pattern):
    """Build sparse conv matrix using precomputed pattern."""
    rows, cols = pattern
    data = f[rows - cols]
    return sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def solve_lp(f, M, pattern, time_limit=2.0):
    """Solve max sum(g) s.t. A@ g <= M, g>=0."""
    n = len(f)
    A = build_A_from_data(n, f, pattern)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def subgradient_step(f, eta):
    """Subgradient descent step on max(conv(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    g = np.zeros(n)
    for m in range(n):
        r = k_star - m
        if 0 <= r < n:
            g[m] = 2.0 * f[r]
    f_new = f - eta * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def make_start(n, method, rng):
    """Create initial normalized non-negative sequence of length n."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif method == 'sin2':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 8)
        a = np.exp(-0.5 * ((x - n // 4) / s) ** 2) + np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2)
    elif method == 'random':
        a = rng.random(n) + 0.05
    elif method == 'perturbed_uniform':
        a = 1.0 + rng.normal(0, 0.4, n)
    elif method == 'geometric':
        a = 0.97 ** np.arange(n)
    elif method == 'three_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 12)
        a = (np.exp(-0.5 * ((x - n // 6) / s) ** 2) +
             np.exp(-0.5 * ((x - n // 2) / s) ** 2) +
             np.exp(-0.5 * ((x - 5 * n // 6) / s) ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.005)
    a = a / a.sum()
    return a


def block_perturb(f, rng, sigma=0.25, max_frac=0.25):
    """Random block multiplicative perturbation."""
    n = len(f)
    max_w = max(1, int(n * max_frac))
    w = int(rng.integers(1, max_w + 1))
    start = int(rng.integers(0, max(1, n - w)))
    end = min(n, start + w)
    factor = np.exp(rng.normal(0, sigma, end - start))
    f_new = f.copy()
    f_new[start:end] *= factor
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def lp_optimize(f, deadline, t_init=0.3, t_min=0.05, t_decay=0.9995,
                lp_time=2.0, perturb_every=15, perturb_sigma=0.2,
                line_search=True, seed=0):
    """Main LP optimization loop with adaptive step and perturbations."""
    n = len(f)
    pattern = build_A_pattern(n)
    f = f.copy()
    t = t_init
    best = f.copy()
    best_sc = fast_score(f)
    it = 0
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        g = solve_lp(f, M, pattern, time_limit=lp_time)

        if g is not None:
            g = g / g.sum()
            if np.linalg.norm(g - f) < 1e-10:
                f = block_perturb(f, rng, sigma=perturb_sigma)
            elif line_search:
                # Line search: try several step sizes
                best_t = t
                best_f = (1 - t) * f + t * g
                best_f = np.maximum(best_f, 0.0)
                s = best_f.sum()
                if s > 1e-10:
                    best_f = best_f / s
                best_trial_sc = fast_score(best_f)
                for t_try in [t * 0.5, t * 2.0, t * 0.1]:
                    if t_try < 1e-4:
                        continue
                    f_try = (1 - t_try) * f + t_try * g
                    f_try = np.maximum(f_try, 0.0)
                    s_try = f_try.sum()
                    if s_try < 1e-10:
                        continue
                    f_try = f_try / s_try
                    sc_try = fast_score(f_try)
                    if sc_try < best_trial_sc:
                        best_trial_sc = sc_try
                        best_f = f_try
                        best_t = t_try
                f = best_f
                t = best_t
            else:
                f = (1 - t) * f + t * g
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
        else:
            # Subgradient fallback
            f = subgradient_step(f, eta=0.005 * max(t, 0.1))

        # Periodic perturbation to escape local minima
        if it % perturb_every == 0:
            f_pert = block_perturb(f, rng, sigma=perturb_sigma)
            sc_pert = fast_score(f_pert)
            sc_cur = fast_score(f)
            if sc_pert < sc_cur:
                f = f_pert

        # Cool down step size
        t = max(t_min, t * t_decay)

        # Track best
        sc = fast_score(f)
        if np.isfinite(sc) and sc < best_sc:
            if sc < best_sc - 1e-8:
                stall = 0
            else:
                stall += 1
            best_sc = sc
            best = f.copy()
            if it % 200 == 1:
                print("    it=%d score=%.8f t=%.4f" % (it, best_sc, t), flush=True)
        else:
            stall += 1

        # If stalled for a long time, increase perturbation
        if stall > 80:
            f = block_perturb(f, rng, sigma=0.4, max_frac=0.4)
            stall = 0

    return best, best_sc


def polish(f, deadline, seed=999):
    """Final polish with very small perturbations and LP steps."""
    n = len(f)
    pattern = build_A_pattern(n)
    f = f.copy()
