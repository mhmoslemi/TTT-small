def normalize_sum1(a):
    """Normalize non-negative array to sum to 1."""
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score(a):
    """Compute the objective score."""
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1} using water-filling."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    # Bisection for the water level
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(0.0, a - mid))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.maximum(0.0, a - tau)
    s = proj.sum()
    if s > 0:
        proj /= s
    return proj


def compute_gradient_smooth(a, tau=50.0):
    """Compute gradient of smooth max of auto-convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    # Softmax weights with temperature tau
    conv_shifted = conv - conv.max()  # for numerical stability
    w = np.exp(tau * conv_shifted)
    w /= w.sum()
    # Gradient: grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # = correlate(w, a, 'full')[j] + convolve(a, w)[j]
    grad1 = np.correlate(w, a, mode='full')[:n]
    grad2 = np.convolve(a, w)[:n]
    return grad1 + grad2


def compute_gradient_subgrad(a):
    """Compute subgradient using argmax position."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    # grad[j] = a[k_star - j] + a[j - k_star] (with 0 for out of range)
    grad = np.zeros(n)
    # First term: a[k_star - j] for j where 0 <= k_star - j < n
    j_start = max(0, k_star - n + 1)

    # Second term: a[j - k_star] for j where 0 <= j - k_star < n

    for j in range(j_start2, j_end2):
        idx = j - k_star
        if 0 <= idx < n:
            grad[j] += a[idx]
    return grad

"""
Gradient-based optimization with smooth-max for minimizing peak auto-convolution
of non-negative sequences. Combines Nesterov momentum, adaptive step size,
multi-start search across lengths, and LP-based polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def normalize_sum1(a):
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_seq(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def project_simplex(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.maximum(0.0, a - mid))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    proj = np.maximum(0.0, a - tau)
    s = proj.sum()
    if s > 0:
        proj /= s
    else:
        proj = np.full(n, 1.0 / n)
    return proj


def grad_smooth_max(a, tau=100.0):
    n = len(a)
    conv = np.convolve(a, a)
    conv_s = conv - conv.max()
    w = np.exp(tau * conv_s)
    w /= w.sum()
    grad1 = np.correlate(w, a, mode='full')[:n]
    grad2 = np.convolve(a, w)[:n]
    return grad1 + grad2


def grad_subgrad(a):
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    if j_lo <= j_hi:
        vals = a[k_star - j_hi : k_star - j_lo + 1][::-1]
        grad[j_lo : j_hi + 1] += vals
    j_lo2 = max(0, k_star)
    j_hi2 = min(n - 1, k_star + n - 1)
    if j_lo2 <= j_hi2:
        vals = a[0 : j_hi2 - j_lo2 + 1]
        grad[j_lo2 : j_hi2 + 1] += vals
    return grad


def init_raised_cosine(n, shift=0.0):
    t = np.arange(n, dtype=np.float64)
    x = (t + shift) / n
    return np.maximum(0.0, 0.5 * (1.0 + np.cos(2.0 * np.pi * x)))


def init_bell(n, center_frac=0.5, width_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = width_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_power_law(n, alpha=1.0, side=0.5):
    t = np.arange(n, dtype=np.float64)
    center = side * n
    return np.maximum(0.01, 1.0 - 0.9 * np.abs(t - center) / (n / 2.0)) ** alpha


def init_random_smooth(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 15), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return a / s


def gradient_optimize(a, time_limit, rng, use_smooth=True, tau_init=200.0,
                      tau_final=20.0, lr_init=0.05, max_lr=0.5, verbose=True):
    start_time = time.monotonic()
    a = normalize_sum1(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.9
    stagnant = 0
    it = 0
    max_stagnant = 200

    while time.monotonic() - start_time < time_limit:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_final / tau_init) ** progress

        if use_smooth:
            grad = grad_smooth_max(a, tau=tau)
        else:
            grad = grad_subgrad(a)

        v = momentum * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)

        new_score = score_seq(a_new)
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            lr = min(lr * 1.02, max_lr)
            momentum = min(0.95, momentum * 0.999)
        else:
            a = a_new
            stagnant += 1
            lr *= 0.85

        if verbose and it % 200 == 0:
            cur_score = score_seq(a)
            print(f"      grad iter {it}: cur={cur_score:.6f}, best={best_score:.6f}, "
                  f"lr={lr:.5f}, tau={tau:.1f}, t={elapsed:.0f}s")

        if lr < 1e-10:
            v = np.zeros(n)
            lr = lr_init * 0.5
            a = a + rng.normal(0, 0.01 / n, n)
            a = project_simplex(a)
            stagnant = 0
        if stagnant > max_stagnant:
            v = np.zeros(n)
            lr = lr_init
            if rng.random() < 0.5:
                a = a + rng.normal(0, 0.02 / n, n)
            else:
                perm = rng.permutation(n)
                a = a[perm]
            a = project_simplex(a)
            stagnant = 0

    final_score = score_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"      grad done: best={best_score:.6f}, iters={it}, "
              f"time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def lp_polish(a, time_limit=60.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_sum1(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)

    for _ in range(20):
        if time.monotonic() - start_time > time_limit:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 2:
            break
        timeout = max(1.0, min(20.0, remaining * 0.4))
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if res.success and res.x is not None:
                g = np.maximum(0.0, res.x)
                sum_g = g.sum()
                if sum_g > 1.0 + 1e-10:
                    g = g / sum_g
                    for alpha in [0.3, 0.15, 0.05, 0.01]:
                        a_try = (1 - alpha) * a + alpha * g
                        a_try = normalize_sum1(a_try)
                        s_try = score_seq(a_try)
                        if s_try < best_score:
                            best_score = s_try
                            best_a = a_try.copy()
                            a = a_try
                            break
                else:
                    break
            else:
                break
        except Exception:
            break

    if verbose:
        print(f"      LP polish: best={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_polish(a, time_limit=60.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_sum1(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    step_size = 0.1

    while time.monotonic() - start_time < time_limit:
        for _ in range(10):
            if time.monotonic() - start_time > time_limit:
                break
            width = max(2, int(n * rng.uniform(0.005, 0.1)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate = a.copy()
            candidate[start:stop] *= np.exp(rng.normal(0, step_size, stop - start))
            candidate = normalize_sum1(candidate)
            sc = score_seq(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                a = candidate
                step_size = min(0.3, step_size * 1.05)
            else:
                step_size = max(0.01, step_size * 0.9)
        if verbose:
            print(f"      local: score={best_score:.6f}, t={time.monotonic()-start_time:.0f}s")

    return best_a, best_score

                polished2, s_lp = lp_polish(best_seq, time_limit=t_lp, verbose=True)
                if s_lp < best_score:
                    best_score = s_lp
                    best_seq = polished2
                    print(f"  After LP: {best_score:.6f}")

            # Final gradient polish with lower tau
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10.0:
                polished3, s_grad = gradient_optimize(
                    best_seq, time_limit=remaining3 * 0.8, rng=rng,
                    use_smooth=True, tau_init=50.0, tau_final=10.0,
                    lr_init=0.02, max_lr=0.2, verbose=True
                )
                if s_grad < best_score:
                    best_score = s_grad
                    best_seq = polished3
                    print(f"  After final grad: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_seq = normalize_sum1(best_seq)
        best_score = score_seq(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    result_list = [float(x) for x in best_seq]
    verify_score = evaluate_sequence(result_list)
    print(f"\n=== Final: n={len(result_list)}, score={best_score:.6f}, "
          f"verified={verify_score:.6f} ===")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return result_list
