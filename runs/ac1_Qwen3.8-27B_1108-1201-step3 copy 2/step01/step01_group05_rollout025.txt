def conv_full(p):
    """Full self-convolution using FFT."""

def score_fn(p):
    """Compute the evaluation score."""

    if s < 0.01:

    c = conv_full(p)

    return float(2.0 * n * cmax / (s * s))

    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    return out / s
