Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based iterative refinement (Subbotin-like), soft-max gradient descent,

import numpy as np
import time
from scipy.optimize import linprog

# Global variables to track best
_global_best_seq = None
_global_best_score = float('inf')

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def lp_solve_direction(f, M):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns the optimal g or None if infeasible.
    """
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            A[m, lo:hi+1] = f[m-hi:m-lo+1][::-1]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    LP-based refinement loop.

    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:

def soft_gradient_step(a, lr, tau=30.0):
    """Soft-max gradient step for peak autocorrelation minimization."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = normalize(a, S_target)
    
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Gradient: d(max approx)/da[j] = 2 * sum_k a[k] * w[j+k]
    grad = np.zeros(n)
    for j in range(n):
        k_max = min(n, 2 * n - 1 - j)
        if k_max > 0:
            grad[j] = 2.0 * np.dot(a[:k_max], w[j:j + k_max])
    
    # Project out mean (fixed sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize(a, S_target)
    return a

def gradient_refine(a, deadline, max_iters=1000):

    n = len(a)

    a = a.copy()

            if s < best_score:

    return best, best_score

def fourier_sequence(n, K, params):
    """Generate a sequence from Fourier parameters."""
    j = np.arange(n, dtype=np.float64)
    a = np.full(n, params[0])  # DC component
    for k in range(1, K + 1):
        a += params[2*k-1] * np.cos(2 * np.pi * k * j / n)
        a += params[2*k] * np.sin(2 * np.pi * k * j / n)
    return np.maximum(a, 0.0)

def coordinate_descent_step(a, n_coords=None):
    """One pass of coordinate descent: optimize each coordinate in turn."""
    n = len(a)
    a = a.copy()
    S_target = np.sqrt(2.0 * n)
    
    if n_coords is None:
        indices = list(range(n))
    else:
        indices = list(range(n))
        np.random.shuffle(indices)
        indices = indices[:n_coords]
    
    for j in indices:
        # Optimize a[j] via golden section search
        a_j = a[j]
        S_rest = a.sum() - a_j
        
        # The objective as function of x = a[j]:
        # num(x) = 2*n*max_k conv_k(x)
        # den(x) = (S_rest + x)^2
        # We want to minimize num/den
        
        # Sample a few values
        candidates = [max(0, a_j - 0.5), max(0, a_j - 0.2), a_j, 
                      a_j + 0.2, a_j + 0.5, max(0, a_j * 0.5), a_j * 1.5]
        best_val = a_j
        best_obj = compute_obj(a)
        
        for x in candidates:
            a[j] = max(0.0, x)
            s = compute_obj(a)
            if s < best_obj:
                best_obj = s
                best_val = x
        
        a[j] = max(0.0, best_val)
    
    return a

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP-based iterative refinement (Subbotin method) with soft-max gradient
polishing, Fourier parameterization for diverse starts, and coordinate descent.
Uses multiple restarts and adaptive time management.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_seq(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def build_lp_matrix(f):
    """Build the LP constraint matrix for conv(f, g) <= M."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            A[m, lo:hi+1] = f[m-hi : m-lo+1][::-1]
    return A


def solve_lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(a, deadline, max_iters=300, eps=0.0):
    """
    LP-based iterative refinement.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S_target = np.sqrt(2.0 * n)
    a = normalize_seq(a, S_target)
    
    best = a.copy()
    best_score = compute_obj(a)
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        f = np.maximum(a, 1e-10)  # Ensure positive for LP
        c = np.convolve(f, f)
        M = float(c.max()) * (1.0 + eps)
        
        g = solve_lp_direction(f, M)
        if g is None:
            break
        
        S_g = g.sum()
        if S_g < 1e-12:
            break
        g = g * (S_target / S_g)
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.7, 0.1, 0.9, 0.2, 0.4]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, S_target)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:

                best = a.copy()

def soft_grad_step(a, lr, tau=30.0):
    """Soft-max gradient step."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = normalize_seq(a, S_target)

    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for j in range(n):
        k_max = min(n, 2 * n - 1 - j)
        if k_max > 0:
            grad[j] = 2.0 * np.dot(a[:k_max], w[j:j + k_max])
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize_seq(a, S_target)
    return a


def gradient_refine(a, deadline, max_iters=1000):
    """Soft-max gradient refinement with annealing."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S_target = np.sqrt(2.0 * n)
    a = normalize_seq(a, S_target)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        progress = it / max_iters
        tau = 15.0 + 50.0 * progress
        lr = 0.08 * (1.0 - 0.9 * progress)
        
        a = soft_grad_step(a, lr, tau)
        
        if (it + 1) % 100 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    return best, best_score


def hard_grad_step(a, lr):
    """Hard max gradient step (fast, less smooth)."""
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = normalize_seq(a, S_target)
    
    c = np.convolve(a, a)
    m_star = int(c.argmax())
    
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = normalize_seq(a, S_target)
    return a


def hard_gradient_refine(a, deadline, max_iters=2000):
    """Hard max gradient refinement."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()

def fourier_seq(n, K, params):
    """Generate sequence from Fourier parameters."""
    j = np.arange(n, dtype=np.float64)
    a = np.full(n, params[0])
    for k in range(1, K + 1):
        if 2*k - 1 < len(params):
            a += params[2*k-1] * np.cos(2 * np.pi * k * j / n)
        if 2*k < len(params):
            a += params[2*k] * np.sin(2 * np.pi * k * j / n)
    return np.maximum(a, 0.0)


def fourier_evolve(n, K, deadline, rng, pop_size=40, max_gens=100):
    """Evolutionary search over Fourier parameters."""
    n_params = 2 * K + 1
    best_a = None
    best_score = float('inf')
    
    population = rng.uniform(0.1, 2.0, size=(pop_size, n_params))
    
    for gen in range(max_gens):
        if time.monotonic() > deadline - 30.0:
            break
        
        scores = np.full(pop_size, float('inf'))
        for i in range(pop_size):
            a = fourier_seq(n, K, population[i])
            if a.sum() > 0.01:
                scores[i] = compute_obj(a)
        
        best_idx = int(np.argmin(scores))
        if scores[best_idx] < best_score:
            best_score = scores[best_idx]
            best_a = fourier_seq(n, K, population[best_idx]).copy()
        
        # Tournament selection + mutation
        order = np.argsort(scores)
        new_pop = np.zeros_like(population)
        
        # Elitism
        n_elite = max(1, pop_size // 4)
        for i in range(n_elite):
            new_pop[i] = population[order[i]]
        
        for i in range(n_elite, pop_size):
            # Crossover
            p1 = population[order[rng.integers(0, n_elite + 5)]]
            p2 = population[order[rng.integers(0, n_elite + 5)]]
            mask = rng.random(n_params) < 0.5
            child = np.where(mask, p1, p2)
            # Mutation (adaptive)
            scale = 0.15 * (1.0 - gen / max_gens) + 0.02
            child += rng.normal(0, scale, n_params)
            child = np.clip(child, 0.0, 5.0)
            new_pop[i] = child
        
        population = new_pop
    
    return best_a, best_score


def cd_pass(a, rng, n_subset=None):
    """One pass of coordinate descent on a random subset of coordinates."""
    n = len(a)
    a = a.copy()
    
    if n_subset is None or n_subset >= n:
        indices = list(range(n))
        rng.shuffle(indices)
    else:
        indices = rng.choice(n, size=n_subset, replace=False)
    
    for j in indices:
        a_j = a[j]
        best_x = a_j
        best_obj = float('inf')
        
        # Try a few candidates around current value
        candidates = [a_j * 0.5, a_j * 0.75, a_j * 1.25, a_j * 1.5,
                      a_j - 0.1, a_j + 0.1]
        candidates.append(0.0)
        candidates.append(2.0)
        
        for x in candidates:
            if x < 0:
                x = 0.0
            a[j] = x
            s = compute_obj(a)
            if s < best_obj:
                best_obj = s
                best_x = x
        
        a[j] = best_x
    
    return a
