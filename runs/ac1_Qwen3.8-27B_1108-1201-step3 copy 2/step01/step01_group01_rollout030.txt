def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)

def _smooth_grad(f, T):
    """Compute gradient of T*log(sum(exp(b/T))) w.r.t. f, where b=conv(f,f) and sum(f)=1."""
    n = len(f)
    b = np.convolve(f, f)
    b_max = np.max(b)
    p = np.exp((b - b_max) / T)
    s = np.sum(p)
    if s < 1e-300:
        # Degenerate case: all mass on the max
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p /= s

    grad_base = correlate(f, p, mode='valid')

Hybrid Fourier-parameter + projected-gradient optimizer for minimizing
the normalized autocorrelation peak of non-negative sequences.
Uses soft-max smoothing, Nesterov momentum, and temperature annealing.
"""

import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time


def _score(f):
    """Compute 2*n*max(conv(f,f)) for f with sum=1."""
    n = len(f)
    b = np.convolve(f, f)
    return 2.0 * n * np.max(b)


def _smooth_loss_and_grad(f, T):
    """
    Compute smooth approximation of max(conv(f,f)) and its gradient.
    Returns (smooth_value, gradient_wrt_f).
    """
    n = len(f)
    b = np.convolve(f, f)
    b_max = np.max(b)
    shifted = (b - b_max) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    sum_exp = np.sum(exp_b)
    if sum_exp < 1e-300:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_b / sum_exp
    smooth_val = T * (b_max / T + np.log(sum_exp))
    # grad_j = (2/T) * sum_m p[j+m] * f[m] for j=0..n-1
    grad_base = correlate(f, p, mode='valid')
    grad = (2.0 / T) * grad_base
    return smooth_val, grad


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fourier_to_seq(params, n):
    """Convert Fourier log-amplitude parameters to a normalized non-negative sequence."""
    i = np.arange(n, dtype=np.float64)
    log_f = np.full(n, params[0], dtype=np.float64)
    n_harm = (len(params) - 1) // 2
    for k in range(1, n_harm + 1):
        c = params[2 * k - 1]
        d = params[2 * k]
        log_f += c * np.cos(2.0 * np.pi * k * i / n) + d * np.sin(2.0 * np.pi * k * n_harm * 0 + 2.0 * np.pi * k * i / n)
    # Wait, that's redundant. Let me redo.
    log_f = np.full(n, params[0], dtype=np.float64)

        d = params[2 * k]
        angle = 2.0 * np.pi * k * i / n
        log_f += c * np.cos(angle) + d * np.sin(angle)
    log_f -= np.max(log_f)
    f = np.exp(log_f)
    s = np.sum(f)
    if s < 1e-300:
        return np.ones(n) / n
    return f / s


def _fourier_score_and_grad(params, n, T=0.1):
    """Compute smooth score and gradient w.r.t. Fourier parameters."""
    f = _fourier_to_seq(params, n)
    smooth_val, grad_f = _smooth_loss_and_grad(f, T)
    
    # d(score)/d(params_m) = sum_j grad_f[j] * df[j]/d(params_m)
    # df[j]/d(params_m) = f[j] * (basis_m[j] - sum_l f[l]*basis_m[l])
    i = np.arange(n, dtype=np.float64)
    n_harm = (len(params) - 1) // 2
    grad_params = np.zeros(len(params))
    
    # Precompute weighted sums
    wf = grad_f * f  # grad_f[j] * f[j]
    
    for m in range(len(params)):
        if m == 0:
            basis = np.ones(n)
        else:
            k = (m + 1) // 2
            if m % 2 == 1:  # cos
                basis = np.cos(2.0 * np.pi * k * i / n)
            else:  # sin
                basis = np.sin(2.0 * np.pi * k * i / n)
        basis_mean = np.dot(f, basis)
        grad_params[m] = np.dot(wf, basis) - basis_mean * np.sum(wf)
    
    return 2.0 * n * smooth_val, 2.0 * n * grad_params


def _lbfqs_fourier(params_init, n, T, maxiter=200, time_limit=30.0):
    """Run L-BFGS-B on Fourier parameters."""
    start_time = time.monotonic()
    
    def obj(x):
        if time.monotonic() - start_time > time_limit:
            raise RuntimeError("timeout")
        val, _ = _fourier_score_and_grad(x, n, T)
        return val
    
    def jac(x):
        if time.monotonic() - start_time > time_limit:
            raise RuntimeError("timeout")
        _, g = _fourier_score_and_grad(x, n, T)
        return g
    
    try:
        result = minimize(obj, params_init, method='L-BFGS-B', jac=jac,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        if result.success or result.fun < np.inf:
            return result.x, result.fun
    except (RuntimeError, Exception):
        pass
    return params_init, _score(_fourier_to_seq(params_init, n))


def _projected_gradient(f_init, deadline, T_init=0.5, T_end=0.005,
                        lr_init=0.005, lr_min=1e-7,
                        momentum_end=0.85, patience=300,
                        reheat_every=500):
    """
    Nesterov-accelerated projected gradient with soft-max smoothing.
    f_init must be on the simplex (sum=1, f>=0).
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _project_simplex(f, 1.0)
    n = len(f)
    
    best_f = f.copy()
    best_score = _score(f)
    
    T = T_init
    lr = lr_init
    momentum = 0.0
    prev_f = f.copy()
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() < deadline:
        total_iters += 1
        
        # Nesterov lookahead
        if momentum > 0:
            y = f + momentum * (f - prev_f)
            y = _project_simplex(y, 1.0)
        else:
            y = f.copy()
        
        # Compute smooth gradient at y
        _, grad = _smooth_loss_and_grad(y, T)
        
        # Center gradient (tangent to simplex)
        grad -= np.mean(grad)
        
        # Adaptive step: normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Step
        f_new = _project_simplex(y - lr * gnorm * (grad / gnorm) if gnorm > 0 else y, 1.0)
        # Actually let's just do f_new = project(y - lr * grad) with unscaled grad
        f_new = _project_simplex(y - lr * grad, 1.0)
        
        # Check improvement (use true score, not smooth)
        s_new = _score(f_new)
        if s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update
        prev_f = f
        f = f_new
        
        # Anneal
        momentum = min(momentum + 0.002, momentum_end)
        lr *= 0.9995
        T = max(T * 0.9997, T_end)
        
        if no_improve >= patience:
            # Reheat: increase lr, decrease T slightly
            lr = max(lr * 3.0, lr_min * 10)
            T = max(T * 2.0, T_end)
            momentum = min(momentum * 0.5, momentum_end)
            no_improve = 0
            # Add small random perturbation
            n_pert = int(0.1 * n)
            idx = np.random.choice(n, size=min(n_pert, n), replace=False)
            pert = np.random.randn(len(idx)) * 0.01
            f[idx] += pert
            f = _project_simplex(f, 1.0)
            prev_f = f.copy()
            if lr < lr_min * 2:
                break
    
    return best_f, best_score


def _make_anti_symmetric(n, eps):
    """Anti-symmetric two-level: first half 1+eps, second half 1-eps."""
    f = np.ones(n)
    half = n // 2
    f[:half] = 1.0 + eps
    f[half:] = 1.0 - eps
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def _make_smooth_anti(n, eps, sharpness=1.0):
    """Smooth anti-symmetric using tanh transition."""
    i = np.arange(n, dtype=np.float64)
    # Smooth step from +1 to -1
    x = (i - n / 2.0) / (n / 2.0)  # ranges from -1 to 1
    f = 1.0 - eps * np.tanh(sharpness * x)
    f = np.maximum(f, 0.01)
    s = np.sum(f)
    return f / s


def _make_cosine_anti(n, eps):
    """Cosine anti-symmetric: 1 + eps*cos(pi*i/(n-1)) shifted."""
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    f = np.maximum(f, 0.01)
    s = np.sum(f)
    return f / s


def _make_quarter_sym(n, eps):
    """Quarter-symmetric: first quarter high, rest low."""
    f = np.ones(n)
    q = n // 4
    f[:q] = 1.0 + eps
    f[q:] = 1.0 - eps * 0.5
    f = np.maximum(f, 0.01)
    s = np.sum(f)
    return f / s
