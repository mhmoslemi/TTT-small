def build_conv_matrix_sparse(a):
    """Build sparse matrix for LP: A @ g gives the convolution a * g."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0

    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    for it in range(2000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break

        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 50:
            break

        M = conv.max()
        timeout = max(1.0, min(15.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g

        # Adaptive step size
        alpha = min(0.5, max(0.03, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            alpha2 = alpha * 0.2
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it} iters")
    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    while time.monotonic() - start_time < time_limit:
        width = max(2, int(rng.integers(max(1, n // 200), max(2, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start))
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        sc = score_sequence(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}")
    return best_a, best_score


def combined_refine(a, time_limit=300.0, verbose=True, rng=None):
    """Alternate between multiplicative and LP refinement."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    Combined start: score={best_score:.6f}, n={len(a)}")

    phase = 0
    while time.monotonic() - start_time < time_limit:
        elapsed = time.monotonic() - start_time
        remaining = time_limit - elapsed
        if remaining < 10.0:
            break

        chunk_time = min(60.0, remaining * 0.8)

        if phase % 3 == 0:
            a, sc = multiplicative_refine(a, time_limit=chunk_time, verbose=False,
                                          T_init=0.01, T_min=0.0005,
                                          eta_init=0.4, rng=rng)
        elif phase % 3 == 1:
            a, sc = lp_refine(a, time_limit=chunk_time, verbose=False)
        else:
            a, sc = local_search(a, time_limit=min(30.0, chunk_time), rng=rng, verbose=False)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose:
                print(f"    Combined phase {phase}: score={sc:.6f}, elapsed={time.monotonic()-start_time:.0f}s")

        phase += 1

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    Combined done: best={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid approach: multiplicative subgradient (FFT-accelerated),
    LP refinement, local search, across multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Try to use parent sequence if available
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(0, parent)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
            ps = score_sequence(parent)
            print(f"Parent sequence: n={len(parent)}, score={ps:.6f}")
            if ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except (NameError, UnboundLocalError):
        parent = None

    # Define search configurations: (n, init_type, time_for_this)
    configs = [
        (300, 'raised_cosine', 0.10),
        (500, 'raised_cosine', 0.12),
        (500, 'tukey', 0.10),
        (700, 'raised_cosine', 0.12),
        (1000, 'raised_cosine', 0.15),
        (500, 'gaussian', 0.08),
        (700, 'gaussian', 0.10),
        (1000, 'blackman', 0.12),
        (1500, 'raised_cosine', 0.15),
        (1000, 'random', 0.10),
        (2000, 'raised_cosine', 0.15),
    ]

    # If we have a parent, add a config to refine it
    if parent is not None:
        configs.insert(0, (len(parent), 'parent', 0.15))

    for cfg_idx, (n, init_type, time_frac) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.9)
        if time_limit < 15.0:
            continue

        print(f"\n=== Config {cfg_idx}: n={n}, init={init_type}, time={time_limit:.0f}s ===")

        if init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.3)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'tukey':
            a = initial_tukey(n, r=0.5)
        elif init_type == 'random':
            a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        elif init_type == 'parent' and parent is not None:
            a = parent.copy()
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        # Use combined refinement
        a_refined, score = combined_refine(a, time_limit=time_limit, verbose=True, rng=rng)

        if score < best_score:
            best_score = score
            best_seq = a_refined.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST: score={score:.6f}, n={n}, elapsed={elapsed:.0f}s")

    # Final polishing phase
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            # Do another round of combined refinement on the best
            polished, score = combined_refine(best_seq, time_limit=remaining * 0.6,
                                             verbose=True, rng=rng)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** After polish: score={score:.6f}")

            # Final local search
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15.0:
                p2, s2 = local_search(best_seq, time_limit=remaining2 * 0.5, rng=rng, verbose=True)
                if s2 < best_score:
                    best_score = s2
                    best_seq = p2.copy()
                    print(f"  *** After final LS: score={s2:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_seq]
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence(final_list)
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return final_list
