"""
Optimized LP + coordinate descent approach to minimize the autocorrelation peak
score = 2*n*max(conv(f,f)) / (sum(f))^2 for non-negative sequences.

Strategy:
  Phase 1: Quick multi-start exploration with LP
  Phase 2: Focused LP optimization with adaptive step size and line search
  Phase 3: Coordinate descent polish (1D golden section per coordinate)
  Phase 4: Final verification
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Fast evaluation of the score: 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for valid k-j in [0,n-1]."""
    j_idx = np.repeat(np.arange(n), n)
    m_idx = np.tile(np.arange(n), n)
    row_idx = j_idx + m_idx
    data = f[m_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=5.0):
    """Solve max sum(g) s.t. A_g <= M, g >= 0 where A_g[k,j] = f[k-j]."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_step_with_line_search(f, t_init=0.5, t_min=0.01, max_trials=8,
                             lp_time=5.0):
    """LP step with backtracking line search. Returns (new_f, improved)."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    M_base = float(f.copy().sum())
    score_base = 2.0 * len(f) * M / (f.sum() ** 2)
    
    if M < 1e-15:
        return f, False
    
    g = solve_lp(f, M, time_limit=lp_time)
    if g is None:
        return f, False
    
    gs = g.sum()
    if gs < 1e-10:
        return f, False
    g = g / gs
    
    # Backtracking line search
    t = t_init
    best_f = f.copy()
    best_sc = score_base
    for _ in range(max_trials):
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            t *= 0.5
            if t < t_min:
                break
            continue
        f_new = f_new / s
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f_new.copy()
        t *= 0.6
        if t < t_min:
            break
    
    improved = (not np.array_equal(best_f, f))
    return best_f, improved


def coordinate_descent_pass(f, rng, max_iters=3, tol=1e-8):
    """One pass of coordinate descent with golden section search."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    score = fast_score(f)
    
    for _iter in range(max_iters):
        improved = False
        order = rng.permutation(n)
        for i in order:
            # Golden section search for f[i]
            lo = 0.0
            hi = max(4.0 / n, 5.0 * f.mean()) * 3
            # Check if current value is near boundary
            cur = f[i]
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            
            def obj_1d(x):
                f[i] = x
                s = f.sum()
                if s < 0.01:
                    return 1e10
                return 2.0 * n * np.convolve(f, f).max() / (s * s)
            
            # Golden section
            a, b = lo, hi
            c_ = b - (b - a) / phi
            d_ = a + (b - a) / phi
            fc = obj_1d(c_)
            fd = obj_1d(d_)
            
            for _gs in range(30):
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc
                    c_ = b - (b - a) / phi
                    fc = obj_1d(c_)
                else:
                    a = c_
                    c_ = d_
                    fc = fd
                    d_ = a + (b - a) / phi
                    fd = obj_1d(d_)
                if abs(b - a) < tol:
                    break
            
            best_x = (a + b) / 2.0
            best_val = obj_1d(best_x)
            
            # Also check current value
            cur_val = obj_1d(cur)
            if cur_val < best_val:
                best_x = cur
                best_val = cur_val
            
            if best_val < score - 1e-10:
                f[i] = best_x
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 0.01:
                    f = f / s
                    score = fast_score(f)
                improved = True
            f[i] = cur
        if not improved:
            break
    
    return f, score


def coordinate_descent_full(f, rng, time_budget=60.0):
    """Run coordinate descent until time budget."""
    start_time = time.monotonic()
    n = len(f)
    f = f.copy() / f.sum()
    score = fast_score(f)
    
    while time.monotonic() - start_time < time_budget:
        f, score = coordinate_descent_pass(f, rng, max_iters=1)
        if time.monotonic() - start_time > time_budget * 0.8:
            break
    return f, score


def block_perturb(f, rng, low=2, high=30):
    """Multiplicative random block perturbation, renormalised."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(low, min(high + 1, n)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.3)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def make_initial(n, rng, method='uniform'):
    """Create a normalized non-negative initial sequence."""
    if method == 'uniform':
        a = np.ones(n) + rng.normal(0, 0.3, n)
    elif method == 'inverse_sqrt':
        x = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(x)
        a += rng.normal(0, 0.1, n)
    elif method == 'geometric':
        r = 0.98 + 0.02 * rng.random()
        a = r ** np.arange(n, dtype=np.float64)
        a += 0.01 * rng.random(n)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(2, n // 10)
        a = (np.exp(-0.5 * ((x - n * 0.25) / s) ** 2) +
             np.exp(-0.5 * ((x - n * 0.75) / s) ** 2))
        a += 0.01 * rng.random(n)
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a += 0.01 * rng.random(n)
    elif method == 'lowpass':
        a = 1.0 / (1.0 + np.arange(n, dtype=np.float64) * 0.01)
        a += 0.01 * rng.random(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

    while time.monotonic() - start_time < time_budget:

        f_new, improved = lp_step_with_line_search(

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()
            stall = 0
            # Slightly increase step on success
            t = min(0.5, t * 1.05)

            stall += 1
            t = max(0.02, t * 0.9)

    # Final check on best
