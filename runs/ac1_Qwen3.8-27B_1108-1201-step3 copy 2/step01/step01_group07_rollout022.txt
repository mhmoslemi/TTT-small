```python
            best_obj = obj
        a = a_new
        
        # Decay learning rate
        lr *= 0.999
    
    return best_a
```

```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    inits['bump_right'] = bump_r

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, obj={obj_opt:.6f}, score={score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on top configs
    print("=== Phase 2: Deep Optimization ===")
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 8:
            break
    
    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    # Also add some fresh starts at the best length
    best_n = len(best_sequence) if best_sequence is not None else 256
    for extra_init in ['random', 'ramp_inc', 'bump_left']:
        inits_b = generate_initializations(best_n, parent, rng)
        if extra_init in inits_b:
            top_configs.append((best_n, f'extra_{extra_init}', inits_b[extra_init]))
    
    remaining = deadline - time.monotonic() - 60.0
    time_per_deep = max(15.0, remaining / max(1, len(top_configs) * 2))
    
    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 8:
            break
        
        # Try both subgradient and mirror descent
        for method in ['subgrad', 'mirror']:
            if time.monotonic() > deadline - 30:
                break
            t_single = t_alloc / 2.0
            if t_single < 5:
                break
            
            if method == 'subgrad':
                a_opt, obj_opt = optimize_subgrad(
                    a0, time_limit=t_single, lr_init=0.02, momentum=0.9,
                    top_k=max(20, n // 30), temp_init=0.0002,
                    decay=0.9998, perturb_every=2000, perturb_scale=0.003)
            else:
                a_opt, obj_opt = optimize_mirror(
                    a0, time_limit=t_single, t_param=80.0, eta_init=0.3)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep: n={n}, {name}, {method}, score={score:.6f}")
    
    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        n_best = len(best_sequence)
        
        # Multiple polish attempts with different parameters
        polish_configs = [
            {'lr_init': 0.01, 'momentum': 0.92, 'top_k': n_best // 20, 'temp_init': 0.0001},
            {'lr_init': 0.005, 'momentum': 0.95, 'top_k': n_best // 10, 'temp_init': 0.00005},
            {'lr_init': 0.02, 'momentum': 0.88, 'top_k': max(10, n_best // 50), 'temp_init': 0.0003},
        ]
        
        t_per_polish = remaining / (len(polish_configs) + 2)
        
        for cfg in polish_configs:
            if remaining < 5:
                break
            a_opt, obj_opt = optimize_subgrad(
                best_sequence, time_limit=t_per_polish, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish: score={score:.6f}")
            remaining = deadline - time.monotonic() - 10.0
        
        # Mirror polish
        if remaining > 10:
            a_opt, obj_opt = optimize_mirror(
                best_sequence, time_limit=remaining / 2, t_param=150.0, eta_init=0.2)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Mirror polish: score={score:.6f}")
        
        # Random perturbation + re-optimize
        if remaining > 10:
            for _ in range(3):
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.002
                a_pert = project_simplex(best_sequence + noise)
                a_opt, obj_opt = optimize_subgrad(
                    a_pert, time_limit=min(remaining / 3, 30),
                    lr_init=0.01, momentum=0.9, top_k=n_best // 20, temp_init=0.0001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb+opt: score={score:.6f}")
                remaining = deadline - time.monotonic() - 10.0
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Verify the score one more time
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
```
