"""
Hybrid optimization: LP-based block coordinate descent + L-BFGS gradient
descent with log-sum-exp smoothing, to minimize the autocorrelation peak
of a non-negative step function.

Objective: score = 2*n*max(conv(f,f)) / (sum(f))^2
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def fast_score(f):
    """Compute the objective value exactly."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()


def lse_objective_and_grad(z, alpha):
    """Compute J(z) = (1/alpha)*log(sum_k exp(alpha*C[k])) and its gradient."""
    f = softmax(z)
    n = len(f)
    C = np.convolve(f, f)  # length 2n-1
    
    # J = max_C + (1/alpha)*log(sum exp(alpha*(C - max_C)))
    max_C = np.max(C)
    shifted = alpha * (C - max_C)
    log_sum = np.log(np.sum(np.exp(shifted)))
    J = max_C + log_sum / alpha
    
    # p[k] = exp(alpha*C[k]) / sum(exp(alpha*C[m]))
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # P[i] = sum_{m=0}^{n-1} f[m] * p[i+m]
    P = np.zeros(n)
    for m in range(n):
        P += f[m] * p[m:m+n]
    
    E = np.dot(p, C)
    grad = 2.0 * f * (P - E)
    
    return J, grad


def gradient_refine(f, alpha_start=5.0, alpha_end=200.0, max_iter=200,
                    time_limit=60.0):
    """Refine f using L-BFGS with increasing alpha."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    z0 = np.log(f)
    
    best_z = z0.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    
    n_alphas = 5
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        if time.monotonic() - start_time > time_limit:
            break
        try:
            res = minimize(
                lse_objective_and_grad,
                z0,
                args=(alpha,),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'gtol': 1e-8}
            )
            f_new = softmax(res.x)
            f_new = np.maximum(f_new, 1e-15)
            f_new = f_new / f_new.sum()
            sc = fast_score(f_new)
            if sc < best_score:
                best_score = sc
                best_z = res.x.copy()
                z0 = res.x.copy()
        except Exception:
            pass
    
    f_best = softmax(best_z)
    f_best = np.maximum(f_best, 1e-15)
    f_best = f_best / f_best.sum()
    return f_best, best_score

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def lse_obj_grad(z, alpha):
    f = softmax_fn(z)
    n = len(f)
    C = np.convolve(f, f)
    max_C = np.max(C)
    shifted = alpha * (C - max_C)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    J = max_C + np.log(sum_exp) / alpha
    p = exp_shifted / sum_exp
    P = np.zeros(n)
    for m in range(n):
        P += f[m] * p[m:m+n]
    E = np.dot(p, C)
    grad = 2.0 * f * (P - E)
    return J, grad

    row_idx = i_idx + j_idx
    data = f[i_idx]
    A = coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    return A

def lp_step(f, M, time_limit_s=5.0):
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': time_limit_s})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def lp_optimize(f, time_limit=120.0, t_init=0.3, t_min=0.02):

    t0 = time.monotonic()
    best = f.copy()

            best = f_new.copy()
            best_sc = sc
        f = f_new
        t = max(t_min, t * 0.99)
    return best, best_sc

"""
Hybrid optimization combining LP block-coordinate descent and L-BFGS
gradient descent (with log-sum-exp smoothing) to minimize the peak
autocorrelation of a non-negative step function.

Score = 2*n*max(conv(f,f)) / (sum(f))^2, minimized over f >= 0.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def fast_score(f):
    """Compute score = 2*n*max(conv(f,f))/sum(f)^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))
