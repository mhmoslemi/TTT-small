def _local_search(a_init, deadline, rng, step_scale=0.02):
    """
    Coordinate-style local search: randomly perturb pairs of coordinates
    and accept if the score improves.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score(a)
    accepting = True
    cur_score = best_score

    while time.monotonic() < deadline:
        # Pick two random coordinates to swap mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Transfer delta from i to j (or vice versa)
        delta = rng.uniform(-1, 1) * step_scale * a[i]
        if delta > 0 and a[i] - delta < 0:
            delta = a[i] * 0.5
        if delta < 0 and a[j] + delta < 0:
            delta = -a[j] * 0.5

        if abs(delta) < 1e-15:
            continue

        a[i] -= delta
        a[j] += delta

        b = np.convolve(a, a)
        new_score = 2.0 * n * np.max(b)

        if new_score < cur_score:
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
        else:
            # Reject
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1) / f
        noise += amp * np.cos(2.0 * np.pi * f * i / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    a_new = _proj_simplex(a_new, 1.0)
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse starts, FISTA main loop,
    local search polishing, and diversification restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except (NameError, Exception):
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Normalize parent
    S0 = np.sum(parent)
    if S0 > 1e-10:
        a0 = parent / S0
    else:
        a0 = np.ones(n) / n

    # Global best tracking
    global_best = a0.copy()
    global_best_score = _score(a0)
    print(f"Initial score: {global_best_score:.6f} (n={n})")

    # Phase 1: Quick evaluation of diverse starting points (first 10% of time)
    phase1_deadline = t_start + 0.1 * (deadline - t_start)

    starts = [a0]
    # Cosine-modulated starts
    for freq in [1, 2, 3]:
        for eps in [0.3, 0.5, 0.7]:
            starts.append(_make_cosine(n, freq, eps))
    # Multi-cosine
    for _ in range(3):
        starts.append(_make_multicosine(n, rng, n_freqs=3, max_freq=4))
    # Anti-symmetric
    for mode in range(3):
        for eps in [0.3, 0.5]:
            starts.append(_make_antisymmetric_cosine(n, eps, mode))
    # Two-level
    for eps in [0.3, 0.5, 0.7]:
        starts.append(_make_two_level(n, eps))
    # Smooth random
    for _ in range(3):
        starts.append(_make_smooth_random(n, rng, n_modes=5, eps=0.3))

    # Quick FISTA from each start (very short time each)
    time_per_start = max(1.0, (phase1_deadline - time.monotonic()) / max(1, len(starts) - 1))

    # Always optimize the parent first with a slightly longer budget
    parent_deadline = min(time.monotonic() + 3.0 * time_per_start, phase1_deadline)
    best_a, best_s = _fista(a0, parent_deadline, beta_start=5.0, beta_end=100.0,
                            M_start=10, M_end=50, eta_init=2.0 / np.sqrt(n))
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"  Parent optimized: {global_best_score:.6f}")

    for i, s in enumerate(starts[1:]):
        if time.monotonic() >= phase1_deadline:
            break
        sd = min(time.monotonic() + time_per_start, phase1_deadline)
        ba, bs = _fista(s, sd, beta_start=5.0, beta_end=50.0,
                        M_start=10, M_end=30, eta_init=2.0 / np.sqrt(n))
        if bs < global_best_score:
            global_best = ba.copy()
            global_best_score = bs
            print(f"  Start {i+2}: {global_best_score:.6f}")

    print(f"Phase 1 best: {global_best_score:.6f}")

    # Phase 2: Main FISTA optimization (60% of time)
    phase2_deadline = t_start + 0.7 * (deadline - t_start)
    remaining2 = phase2_deadline - time.monotonic()

    # Run FISTA from the global best with longer budget and more aggressive params
    best_a, best_s = _fista(global_best, phase2_deadline,
                            beta_start=10.0, beta_end=500.0,
                            M_start=20, M_end=300,
                            eta_init=1.5 / np.sqrt(n))
    if best_s < global_best_score:
        global_best = best_a.copy()
        global_best_score = best_s
        print(f"Phase 2 main FISTA: {global_best_score:.6f}")
    else:
        print(f"Phase 2 main FISTA: {best_s:.6f} (no improvement)")

    # Phase 3: Diversification (20% of time)
    phase3_deadline = t_start + 0.9 * (deadline - t_start)
    remaining3 = phase3_deadline - time.monotonic()

    if remaining3 > 10:
        n_div = max(2, int(remaining3 / 60.0))
        per_div = remaining3 / n_div

        for d in range(n_div):
            if time.monotonic() >= phase3_deadline:
                break
            # Perturb the best
            a_pert = _perturb(global_best, rng, scale=0.08)
            div_deadline = min(time.monotonic() + per_div, phase3_deadline)
            ba, bs = _fista(a_pert, div_deadline,
                           beta_start=20.0, beta_end=400.0,
                           M_start=30, M_end=200,
                           eta_init=1.0 / np.sqrt(n))
            if bs < global_best_score:
                global_best = ba.copy()
                global_best_score = bs
                print(f"  Diversify {d+1}: NEW BEST {global_best_score:.6f}")

    # Phase 4: Local search polishing (last 10%)
    phase4_deadline = deadline
    remaining4 = phase4_deadline - time.monotonic()

    if remaining4 > 5:
        ba, bs = _local_search(global_best, phase4_deadline, rng, step_scale=0.01)
        if bs < global_best_score:
            global_best = ba.copy()
            global_best_score = bs
            print(f"  Local search: NEW BEST {global_best_score:.6f}")

    # Final: one more FISTA polish with very small step
    final_deadline = deadline
    remaining_f = final_deadline - time.monotonic()
    if remaining_f > 3:
        ba, bs = _fista(global_best, final_deadline,
                        beta_start=200.0, beta_end=1000.0,
                        M_start=50, M_end=500,
                        eta_init=0.1 / np.sqrt(n))
        if bs < global_best_score:
            global_best = ba.copy()
            global_best_score = bs
            print(f"  Final polish: NEW BEST {global_best_score:.6f}")

    # Ensure sum is reasonable for the evaluator (sum >= 0.01)
    result = [float(max(0.0, x)) for x in global_best]
    total = sum(result)
    if total < 0.01:
        result = [max(0.0, x) * (1.0 / total) for x in global_best]

    final_score = float(evaluate_sequence(result))
    print(f"FINAL SCORE: {final_score:.6f}")
    return result
