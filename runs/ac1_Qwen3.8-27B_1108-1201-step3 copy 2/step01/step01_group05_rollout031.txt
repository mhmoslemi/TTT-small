"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog, differential_evolution
from scipy.sparse import csr_matrix

LP_NMAX = 450


def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def score_and_gradient(p, T_factor):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.zeros(0), np.empty(0), 0.0, np.inf, 0.0

    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]

    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n), c.astype(np.float64, copy=False), 0.0, np.inf, 0.0

    T = max(float(T_factor) * cmax, 1e-300)
    if not np.isfinite(T) or T <= 0.0:
        T = 1e-300

    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c, dtype=np.float64)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0

    w = e / raw
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]

    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    G = np.maximum(G, 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    s = float(np.sum(p))
    sc = float(2.0 * n * cmax / (s * s)) if s > 1e-12 else np.inf
    return g, c.astype(np.float64, copy=False), T, sc, surr


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def fourier_to_sequence(params, n):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    logp = np.zeros(n, dtype=np.float64)
    K = min(20, int((len(params) - 4) // 2))
    for k in range(1, K + 1):
        a = float(params[2 * (k - 1)])
        b = float(params[2 * (k - 1) + 1])
        logp += a * np.cos(2.0 * np.pi * k * x) + b * np.sin(2.0 * np.pi * k * x)
    if len(params) > 2 * K:
        taper_left = float(params[2 * K])
        taper_right = float(params[2 * K + 1])
    else:
        taper_left = 0.0
        taper_right = 0.0
    eps = 1e-6
    logp += taper_left * np.log(x + eps) + taper_right * np.log(1.0 - x + eps)
    if not np.all(np.isfinite(logp)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    m = float(np.max(logp))
    if not np.isfinite(m):
        return np.full(n, 1.0 / n, dtype=np.float64)
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)


def fourier_objective(params, n):

        p = fourier_to_sequence(params, n)
        sc = internal_score(p)
        if not np.isfinite(sc):

        return float(sc)
    except Exception:
        return 1e10


def make_geometric(n, s, reverse=False):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(s) * x)
    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_power(n, a, reverse=False):
    n = int(max(2, n))
    x = np.linspace(1e-6, 1.0, n)
    p = np.power(x, float(a))
    if reverse:
        p = p[::-1]
    return safe_normalize(np.maximum(p, 0.0))


def make_beta_dist(n, alpha, beta):
    n = int(max(2, n))
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    p = np.power(x, float(alpha)) * np.power(1.0 - x, float(beta))
    return safe_normalize(np.maximum(p, 0.0))


def make_gaussian(n, mu, sigma, skew=0.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    dx = x - float(mu)
    p = np.exp(-0.5 * (dx / max(1e-4, float(sigma))) ** 2 + float(skew) * dx)
    return safe_normalize(p)


def make_two_peak(n, mu1, s1, w1, mu2, s2, w2):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = float(w1) * np.exp(-0.5 * ((x - float(mu1)) / max(1e-4, float(s1))) ** 2)
    p += float(w2) * np.exp(-0.5 * ((x - float(mu2)) / max(1e-4, float(s2))) ** 2)
    return safe_normalize(np.maximum(p, 0.0))

def lp_best_response(p, budget_s):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > LP_NMAX or budget_s <= 1.0:
        return None
    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None
    f = p + 1e-14
    fs = float(f.sum())
    if not np.isfinite(fs) or fs <= 0.0:
        return None
    f = f / fs
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64, copy=False)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False, "time_limit": max(1.0, budget_s - 2.0)})
    except Exception:
        return None
    if res is None:
        return None
    g = getattr(res, "x", None)
    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g / s

                            p = pt
                            accepted = True

            if not accepted:
                since_improve += 1
                if since_improve > 800:
                    mix = float(rng.uniform(0.3, 0.6))
                    p = project_simplex((1.0 - mix) * best_p + mix * p + rng.normal(0.0, 0.15 / n, n))
                    current_score = internal_score(p)
                    prev_p = p.copy()
                    T_factor = 0.4
                    since_improve = 0
                elif since_improve > 300:
                    p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0.0, 0.04 / n, n))
                    current_score = internal_score(p)
                    prev_p = p.copy()

        if allow_lp and n <= n_lp and iter_no % 1500 == 0:
            remaining = end - time.monotonic()
            if remaining > 10.0:
                g = lp_best_response(p, min(30.0, remaining))
                if g is not None:
                    for t in (1.0, 0.6, 0.3, 0.1):
                        q = project_simplex((1.0 - t) * p + t * g)
                        sc = internal_score(q)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            current_score = sc
                            best_p = q.copy()
                            p = q.copy()
                            prev_p = q.copy()
                            since_improve = 0
                            break

        if iter_no % 3000 == 0:
            print(label + " iter=%d best=%.10f cur=%.10f Tf=%.5f" % (
                iter_no, best_score, current_score, T_factor), flush=True)

        if best_score < 1.4950 and since_improve > 8000:
            break

    best_p = project_simplex(best_p)
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and float(parent_arr.sum()) > 0.0:
            parent = safe_normalize(parent_arr)
            print("Parent loaded: n=%d score=%.10f" % (parent.size, internal_score(parent)), flush=True)
    except Exception:
        parent = None

    if parent is None:
        parent_n = 700
    else:
        parent_n = int(parent.size)

    target_n = parent_n
    if target_n < 64:
        target_n = 700
    if target_n > 8000:
        target_n = 8000

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("New best: score=%.10f n=%d t=%.1fs" % (sc, p.size, time.monotonic() - start_time), flush=True)

    # Phase 1: Quick evaluation of parent and simple shapes
    if parent is not None:
        try_record(parent, internal_score(parent))

    if deadline - time.monotonic() < 60.0:
        if best_p is None:
            best_p = make_geometric(target_n, 1.0)
            best_score = internal_score(best_p)
        best_p = np.asarray(best_p, dtype=np.float64)
        best_p = np.maximum(best_p, 0.0)
        best_p = np.minimum(best_p, 1000.0)
        return [float(x) for x in best_p]

    # Phase 2: Global search via differential evolution on Fourier params
    fourier_n = target_n
    K_four = 15
    dim = 2 * K_four + 2
    bounds = [(-2.0, 2.0)] * (2 * K_four) + [(-4.0, 6.0), (-4.0, 6.0)]

    de_budget = min(180.0, (deadline - time.monotonic()) * 0.12)
    if de_budget > 30.0:
        print("DE search: dim=%d n=%d budget=%.1fs" % (dim, fourier_n, de_budget), flush=True)

        def de_obj(params):
            return fourier_objective(params, fourier_n)

        try:
            res = differential_evolution(
                de_obj, bounds,
                maxiter=30, popsize=15, tol=0.3,
                seed=int(rng.integers(0, 2**31 - 1)),
                polish=False, init="latinhypercube",
                workers=1
            )
            if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                p_de = fourier_to_sequence(res.x, fourier_n)
                sc_de = internal_score(p_de)
                try_record(p_de, sc_de)
                print("DE result: score=%.10f" % sc_de, flush=True)
        except Exception as e:
            print("DE failed: %s" % str(e), flush=True)

    # Phase 3: Generate and score initial candidates
    inits = generate_initials(target_n, rng, parent)
    scored_inits = []
    for p in inits:
        p = safe_normalize(p)
        if int(p.size) != target_n:
            continue
        sc = internal_score(p)
        if np.isfinite(sc):
            scored_inits.append((sc, p))
            try_record(p, sc)
    scored_inits.sort(key=lambda x: x[0])

    # Phase 4: Local optimization on top candidates
    if deadline - time.monotonic() > 90.0 and len(scored_inits) > 0:
        final_reserve = min(300.0, max(90.0, (deadline - time.monotonic()) * 0.25))
        n_jobs = min(5, len(scored_inits))
        available = max(0.0, (deadline - final_reserve) - time.monotonic())
        time_per = max(40.0, min(240.0, available / float(n_jobs)))

        for idx in range(n_jobs):
            sc, p = scored_inits[idx]
            remaining = max(0.0, deadline - final_reserve - time.monotonic())
            if remaining < 30.0:
                break
            limit = max(30.0, min(time_per, remaining))
            allow_lp = (target_n <= LP_NMAX)
            label = "JOB%d(n=%d)" % (idx, target_n)
            p_best, sc_best = optimize_local(p, rng, limit, label + ":", allow_lp, target_n)
            try_record(p_best, sc_best)
            print("Job %d done: best=%.10f global=%.10f" % (idx, sc_best, best_score), flush=True)

    # Phase 5: Try a different length if time allows
    if best_p is not None and deadline - time.monotonic() > 180.0:
        alt_n = target_n * 2 if target_n < 2048 else target_n // 2
        if alt_n >= 64 and alt_n != target_n:
            alt_budget = min(120.0, (deadline - time.monotonic()) * 0.3)
            if alt_budget > 60.0:
                print("Alt length search: n=%d budget=%.1fs" % (alt_n, alt_budget), flush=True)
                alt_inits = generate_initials(alt_n, rng, None)
                if best_p is not None:
                    from scipy.interpolate import interp1d
                    x_old = np.linspace(0, 1, int(best_p.size))
                    x_new = np.linspace(0, 1, alt_n)
                    f_interp = interp1d(x_old, best_p, kind='linear', fill_value='extrapolate')
                    alt_inits.insert(0, safe_normalize(np.maximum(f_interp(x_new), 0)))
                scored_alt = []
                for p in alt_inits[:8]:
                    p = safe_normalize(p)
                    if int(p.size) != alt_n:
                        continue
                    sc = internal_score(p)
                    if np.isfinite(sc):
                        scored_alt.append((sc, p))
                        try_record(p, sc)
                scored_alt.sort(key=lambda x: x[0])
                if scored_alt:
                    p_best, sc_best = optimize_local(scored_alt[0][1], rng, min(100.0, alt_budget),
                                                     "ALT(n=%d):" % alt_n, alt_n <= LP_NMAX, alt_n)
                    try_record(p_best, sc_best)

    # Phase 6: Final polish
    if best_p is not None and time.monotonic() < deadline - 20.0:
        limit = max(5.0, deadline - 5.0 - time.monotonic())
        p_best, sc_best = optimize_local(best_p, rng, limit, "FINAL:", target_n <= LP_NMAX, target_n)
        try_record(p_best, sc_best)

    # Ensure contract
    if best_p is None:
        best_p = np.full(max(1, target_n), 1.0 / max(1, target_n), dtype=np.float64)
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 0.01:
        best_p = best_p * (0.05 / max(s, 1e-12))
    print("Final: n=%d score=%.10f time=%.1fs" % (len(best_p), internal_score(best_p), time.monotonic() - start_time), flush=True)
    return [float(x) for x in best_p]
