Optimization of non-negative step functions to minimize the normalized peak
autocorrelation. Uses direct gradient descent with soft-max smoothing,
non-negativity projection, and multiple asymmetric initializations.
Final polishing via coordinate descent and targeted peak reduction.
"""

import numpy as np
import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_score(a):
    """Compute the score: 2*n*max(conv)/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (S * S)

def soft_max_weights(conv, tau):
    """Compute soft-max weights over convolution elements."""
    c = conv / tau
    c = c - np.max(c)
    w = np.exp(c)
    w /= np.sum(w)
    return w

def gradient_soft_max(a, tau):
    """
    Compute gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    w = soft_max_weights(conv, tau)
    # g[j] = 2 * sum_l a[l] * w[l+j] for valid l, l+j
    # = 2 * correlate(w, a)[:n]
    g = 2.0 * np.correlate(w, a, mode='full')[:n]
    value = tau * (np.log(np.sum(np.exp((conv - np.max(conv)) / tau))) + np.max(conv) / tau)
    # Actually: soft_max = max(conv) + tau * log(sum(exp((conv - max)/tau)))
    c_shifted = (conv - np.max(conv)) / tau
    value = np.max(conv) + tau * np.log(np.sum(np.exp(c_shifted)))
    return value, g

    best_score = compute_score(a)

Let me also consider: what if I use a **randomized coordinate descent** approach? At each step, pick a random index j and adjust a[j] to minimize the peak. This is simpler and might work well.

For a fixed j, conv[k] depends on a[j] through:
- conv[k] contains terms a[j]*a[k-j] (if 0≤k-j<n) and a[k-j]*a[j] (same thing)
- So ∂conv[k]/∂a[j] = 2*a[k-j] (if 0≤k-j<n)

If I increase a[j] by δ, the change in conv[k] is approximately 2*a[k-j]*δ.
To keep all conv[k] ≤ T, I need 2*a[k-j]*δ ≤ T - conv[k] for all k with 0≤k-j<n.
So δ ≤ min_k (T - conv[k]) / (2*a[k-j]) (over k where a[k-j] > 0).

This gives the maximum increase for a[j]. Similarly for decrease.

This is essentially a coordinate descent / greedy approach. It's O(n) per coordinate update (to find the bottleneck constraint).

Let me implement this:

```python
def coordinate_descent_optimize(a, T, max_iter=10000):
    """
    Greedy coordinate descent: try to increase each a[j] as much as possible
    while keeping all conv[k] <= T.
    """
    n = len(a)
    a = a.copy()
    
    for iter in range(max_iter):
        improved = False
        # Try increasing each coordinate
        for j in range(n):
            if a[j] == 0:
                # Starting from zero, try a small increase
                delta = 1e-6
            else:
                # Find max allowed increase
                # conv[k] + 2*a[k-j]*delta <= T for all k with 0<=k-j<n
                max_delta = np.inf
                for k in range(j, j + n):
                    if 0 <= k < 2*n - 1:
                        marginal = 2 * a[k - j]
                        if marginal > 0:
                            room = T - compute_conv(a)[k]
                            if room < 0:
                                max_delta = 0
                                break
                            max_delta = min(max_delta, room / marginal)
                delta = 0.5 * max_delta if np.isfinite(max_delta) else 1e-3
            
            if delta > 1e-10:
                a[j] += delta
                improved = True
        
        if not improved:
            break
    
    return a
```
