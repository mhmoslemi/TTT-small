"""
Multi-phase optimization to minimize autocorrelation peak of non-negative
step functions. Combines spectral initialization, subgradient descent,
LP-based refinement, and coordinate descent.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    """Return 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j]."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    return A

def subgradient_step(f, eta, n_iter=1):
    """
    Subgradient-like step to reduce the peak of conv(f,f).
    The subgradient of max_k conv_k w.r.t. f[i] is proportional to
    the "responsibility" of f[i] for the peak.
    """
    f = f.copy()
    s = f.sum()
    f = f / s  # normalize
    
    for _ in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        k_star = np.argmax(conv)
        
        # Responsibility of each f[i] for the peak at k_star
        # conv[k_star] = sum_j f[j] f[k_star - j]
        # d(conv[k_star])/df[i] = f[k_star - i] (if valid) + f[k_star - i] (if valid)
        # Actually: d/df[i] sum_j f[j]f[k-j] = f[k-i] + f[k-i] = 2f[k-i] (interior)
        # At boundaries it's just f[k-i]
        
        # For the peak at k_star:
        # responsibility[i] = f[k_star - i] if 0 <= k_star - i < n else 0
        # But also, reducing f[i] reduces conv at OTHER lags where f[i] participates
        
        # Simple approach: reduce f[i] proportionally to how much it
        # contributes to the peak
        resp = np.zeros(n)
        # f[i] contributes to conv[k] for k = i to i+n-1 (as f[i]*f[k-i])
        # and also as f[k-i]*f[i] for k = i to i+n-1
        # At the peak k_star, f[i] contributes f[i]*f[k_star-i]
        for i in range(len(f)):
            ki = k_star - i
            if 0 <= ki < len(f):
                resp[i] = f[i] * f[ki]
        
        # Also consider secondary peaks
        # Sort conv values and consider top few
        top_k = np.argsort(conv)[-min(3, len(conv)):]
        for k in top_k:
            for i in range(len(f)):
                ki = k - i
                if 0 <= ki < len(f):
                    resp[i] += 0.5 * f[i] * f[ki]
        
        # Multiplicative update: reduce where responsibility is high
        max_resp = resp.max()
        if max_resp > 0:
            f = f / (1.0 + eta * resp / max_resp)
        
        # Renormalize
        s = f.sum()
        if s < 1e-10:
            return f / max(f.sum(), 1e-10)
        f = f / s
    
    return f

def mul_weight_update(f, eta, n_peaks=3):
    """
    Multiplicative weight update: reduce f[i] where it contributes
    to the peaks of the autocorrelation.
    """
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    
    conv = np.convolve(f, f)
    # Get the top n_peaks positions
    n_conv = len(conv)
    top_idx = np.argsort(conv)[-n_peaks:]
    
    # Responsibility vector
    resp = np.zeros(len(f))
    for k in top_idx:
        weight = conv[k] / conv.max()  # weight by how close to max
        for i in range(len(f)):
            ki = int(k) - i
            if 0 <= ki < len(f):
                resp[i] += weight * f[ki]
    
    # Update
    max_r = resp.max()
    if max_r > 0:
        f = f / (1.0 + eta * resp / max_r)
    
    s = f.sum()
    if s < 1e-10:
        return f
    return f / s

def lp_step(f, M_factor=1.0, time_limit=5.0):
    """
    One LP step: find g that maximizes sum(g) subject to conv(f,g) <= M.
    Return normalized g or None.
    """
    n = len(f)
    conv = np.convolve(f, f)
    M = M_factor * conv.max()
    
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def make_flat_spectrum_candidate(n, rng, n_fft=None):
    """
    Generate a candidate by starting from a flat-ish spectrum and
    taking the inverse FFT, keeping only the real positive part.
    """
    if n_fft is None:
        n_fft = max(2*n, 256)
    
    # Start with a random phase (flat magnitude)
    phases = rng.uniform(0, 2*np.pi, n_fft)
    # Make it conjugate symmetric for real output
    if n_fft % 2 == 0:
        F = np.zeros(n_fft, dtype=np.complex128)
        F[0] = 1.0
        F[n_fft//2] = 1.0
        for k in range(1, n_fft//2):
            F[k] = np.exp(1j * phases[k])
            F[n_fft - k] = np.exp(-1j * phases[k])
    else:
        F = np.zeros(n_fft, dtype=np.complex128)
        F[0] = 1.0
        for k in range(1, (n_fft+1)//2):
            F[k] = np.exp(1j * phases[k])
            F[n_fft - k] = np.exp(-1j * phases[k])
    
    # Inverse FFT
    f_full = np.real(np.fft.ifft(F))
    # Take the first n samples
    f = f_full[:n]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s

def make_windowed_candidate(n, rng, window_type='hann'):

def iterative_peak_shaving(f, n_iters=200, eta=0.1):
    """
    iteratively reduce the peaks of the autocorrelation using
    multiplicative weights.
    """
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best = f.copy()
    best_sc = fast_score(f)
    
    for it in range(n_iters):
        f_new = mul_weight_update(f, eta, n_peaks=3)
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f_new.copy()
        f = f_new
        # Decay eta
        eta = max(eta * 0.995, 0.001)
    
    return best / best.sum()

def coordinate_descent(f, budget=10.0, n_sweeps=50):
    """
    Coordinate descent: for each position i, try to optimally adjust f[i]
    to reduce the score (keeping other positions fixed).
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    deadline = time.monotonic() + budget
    
    best_sc = fast_score(f)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                    break
            old_val = f[i]
            # Try reducing f[i] and adding its mass uniformly to others
            # Or: try a small perturbation
            # Binary search for optimal f[i] is expensive; use a simpler approach
            # Try: f[i] * (1 +/- delta)
            for delta in [0.1, 0.05, 0.01]:
                f_trial = f.copy()
                f_trial[i] = old_val * (1 - delta)
                # Distribute the removed mass to the position with lowest "responsibility"
                removed = old_val * delta
                # Add to the position that least contributes to the peak
                # Simple heuristic: add to the minimum-value position
                min_idx = np.argmin(f_trial)
                f_trial[min_idx] += removed
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                f_trial = f_trial / s
                
                sc = fast_score(f_trial)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    f = f_trial.copy()
                    old_val = f[i]
    
    return f

def run_local_opt(f, budget_s, use_lp=True, lp_time=5.0):

    """
    f = f.copy().astype(np.float64)
    best = f.copy()
    best_sc = fast_score(f)
    deadline = time.monotonic() + budget_s
    
    eta = 0.15
    step = 0
    
    while time.monotonic() < deadline:
        step += 1
        
        # Every 50 steps, try an LP step
        if use_lp and step % 50 == 0 and time.monotonic() < deadline - lp_time - 1.0:
            g = lp_step(f, M_factor=1.0, time_limit=min(lp_time, max(1.0, deadline - time.monotonic() - 1.0)))
            if g is not None:
                f_new = g
            else:
                f_new = mul_weight_update(f, eta, n_peaks=3)
        else:
            f_new = mul_weight_update(f, eta, n_peaks=3)
        
        # Check for improvement
        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f_new.copy()
        
        f = f_new
        eta = max(eta * 0.999, 0.005)
        
        # Occasionally perturb to escape local minima
        if step % 200 == 0:
            rng = np.random.default_rng(step)
            f = f * np.exp(rng.normal(0, 0.1, len(f)))
            f = np.maximum(f, 0)
            f = f / f.sum()
    
    return best, best_sc
