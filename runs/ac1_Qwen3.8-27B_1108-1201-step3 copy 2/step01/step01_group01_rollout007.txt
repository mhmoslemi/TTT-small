import numpy as np
import time

def _fast_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    L = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]

def _compute_b(a):
    """Compute autocorrelation b = convolve(a, a)."""
    return _fast_convolve(a, a)

def _compute_score(a):
    """Compute 2*n*max(b) for sum(a)=1."""
    n = len(a)
    b = _compute_b(a)
    return 2.0 * n * np.max(b)

def _softmax_weights(b, T):
    """Compute softmax weights for the peaks."""
    b_shifted = T * (b - np.max(b))
    w = np.exp(b_shifted)
    w /= w.sum()
    return w

def _compute_gradient(a, w):
    """
    Compute gradient of sum_k w_k * b_k with respect to a.
    grad[j] = 2 * sum_k w_k * a[k-j] (with bounds)
    = 2 * convolve(w, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    # Use FFT
    conv_len = 2 * n - 1
    L = 1 << (conv_len + n - 2).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a[::-1], L)
    conv = np.fft.irfft(W * A, L)
    return 2.0 * conv[n - 1 : 2 * n - 1]

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0])
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smoothed_score(a, T):
    """Compute smoothed score: 2*n*(1/T)*log(sum(exp(T*b_k)))."""
    n = len(a)
    b = _compute_b(a)
    b_max = np.max(b)
    s = np.sum(np.exp(T * (b - b_max)))
    return 2.0 * n * (b_max + np.log(s) / T)

def _fw_step(a, obj_func, line_search=True, n_ls=15):
    """
    One Frank-Wolfe step.
    obj_func: function that takes a and returns scalar.
    """
    n = len(a)
    b = _compute_b(a)
    # Gradient of the max (or smoothed) objective
    # For now, use the subgradient at the argmax
    k_star = np.argmax(b)
    grad = _compute_gradient(a, np.zeros(2*n-1))
    # Actually for the max, the gradient is just the gradient at k_star
    # Let me recompute
    # grad[j] = 2 * a[k_star - j] (with bounds)
    # This is 2 * reverse of a shifted
    # Actually: grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n
    # So grad = 2 * a[k_star-n+1 : k_star+1] reversed... 
    # Let me just use the general formula with a one-hot w
    w = np.zeros(2*n-1)
    w[k_star] = 1.0
    grad = _compute_gradient(a, w)
    
    # Center for sum constraint
    grad -= np.mean(grad)
    
    # Find the FW direction: s = e_i where i = argmin(grad)
    i = np.argmin(grad)
    s = np.zeros(n)
    s[i] = 1.0
    
    if not line_search:
        # Greedy step size
        return i, None
    
    # Line search: minimize obj_func((1-t)*a + t*s) for t in [0,1]
    # Use golden section
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    a_c = (1-c)*a + c*s
    a_d = (1-d)*a + d*s
    f_c = obj_func(a_c)
    f_d = obj_func(a_d)
    for _ in range(n_ls - 2):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - golden * (hi - lo)
            a_c = (1-c)*a + c*s
            f_c = obj_func(a_c)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + golden * (hi - lo)
            a_d = (1-d)*a + d*s
            f_d = obj_func(a_d)
    t = (lo + hi) / 2.0
    return i, t

    """
    Frank-Wolfe optimization.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = _compute_score(a)
    print(f"  Initial score: {best_score:.6f}")
    
    if max_iter is None:
        max_iter = 10**6
    
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        
        # Compute objective and gradient
        b = _compute_b(a)
        cur_score = 2.0 * n * np.max(b)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if it % 100 == 0:
                print(f"    Iter {it}: score={best_score:.6f}")
        
        # Softmax weights
        w = _softmax_weights(b, T)
        grad = _compute_gradient(a, w)
        grad -= np.mean(grad)
        
        # FW direction
        i = np.argmin(grad)
        gap = np.max(grad) - grad[i]  # FW gap
        if gap < 1e-10:
            break
        
        s = np.zeros(n)
        s[i] = 1.0
        
        # Line search (use smoothed for speed, or direct)
        if use_smooth_ls:
            # Use smoothed objective for line search
            T_ls = max(T, 100.0)
            def obj_func(a_new):
                n_new = len(a_new)
                b_new = _compute_b(a_new)
                b_max = np.max(b_new)
                s_val = np.sum(np.exp(T_ls * (b_new - b_max)))
                return 2.0 * n_new * (b_max + np.log(s_val) / T_ls)
        else:
            def obj_func(a_new):
                return _compute_score(a_new)
        
        # Golden section line search
        golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c = hi - golden_ratio * (hi - lo)
        d = lo + golden_ratio * (hi - lo)
        f_c = obj_func((1-c)*a + c*s)
        f_d = obj_func((1-d)*a + d*s)
        for _ in range(8):  # 8 iterations of golden section

                d, f_d = c, f_c
                c = hi - golden_ratio * (hi - lo)
                f_c = obj_func((1-c)*a + c*s)
            else:

                c, f_c = d, f_d
                d = lo + golden_ratio * (hi - lo)
                f_d = obj_func((1-d)*a + d*s)
        t = (lo + hi) / 2.0
        
        a = (1 - t) * a + t * s
    
    # Final check
    final_score = _compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score
