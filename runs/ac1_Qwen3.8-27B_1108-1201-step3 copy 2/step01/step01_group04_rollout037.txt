Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax refinement with line search, subgradient descent with
Nesterov momentum, and coordinate descent for final polish.

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target):
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)


def _solve_lp_minpeak(f, S):
    """Solve LP: min M s.t. f*g <= M (elementwise), sum(g) = S, g >= 0.
    Returns (g, M) or (None, None) on failure."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}, M
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # f*g <= M for each k -> A row: [f_{k-j} for j, -1]
    # -sum(g) <= -S -> A row: [-1, ..., -1, 0]
    # sum(g) <= S -> A row: [1, ..., 1, 0]
    # (the last two together give sum(g) = S)
    
    n_constraints = 2 * n - 1 + 2
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, n] = -1.0  # -M
        b_ub[k] = 0.0
    
    # sum(g) <= S
    A_ub[2*n-1, :n] = 1.0
    b_ub[2*n-1] = S
    
    # -sum(g) <= -S
    A_ub[2*n, :n] = -1.0
    b_ub[2*n] = -S
    
    bounds = [(0, None)] * n + [(None, None)]  # g_j >= 0, M free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            M_val = res.x[n]
            return g, M_val
    except Exception:
        pass
    return None, None


def _solve_lp_fixed_M(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0.
    Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]

    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, S, max_iters=200, deadline=None):
    """Iterative LP-based refinement. Returns (best_a, best_score)."""
    n = len(a)
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = _solve_lp_fixed_M(a, M)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum() * S
        
        # Try full update first, then bisection
        a_new = (1.0 - 1.0) * a + 1.0 * g  # = g
        score_new = _compute_obj(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Bisection on t
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                a_try = (1.0 - mid) * a + mid * g
                s_try = _compute_obj(a_try)
                if s_try < best_score:
                    lo = mid
                else:
                    hi = mid
            
            if lo > 0.01:
                a_new = (1.0 - lo) * a + lo * g
                a_new = _normalize(a_new, S)
                score_new = _compute_obj(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break
            else:
                break
        
        if (it + 1) % 20 == 0:
            print(f"    LP iter {it+1}: score={best_score:.6f}, M={M:.4f}")
    
    return best_a, best_score


def _subgrad_descent(a, S, max_iters=5000, deadline=None):
    """Subgradient descent with Nesterov momentum on peak autocorrelation."""
    n = len(a)
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    lr = 0.1
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        a_look = a - beta * v
        a_look = np.maximum(a_look, 0)
        
        c = np.convolve(a_look, a_look)
        m_star = int(c.argmax())
        
        # Subgradient at argmax
        grad = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * a_look[m_star - j]
        
        # Project gradient (remove mean to preserve sum)
        grad -= grad.mean()
        
        # Update
        v = beta * v + lr * grad
        a = a - v
        a = np.maximum(a, 0)
        a = _normalize(a, S)
        
        # Adaptive learning rate
        if (it + 1) % 100 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            lr = 0.1 * (0.95 ** (it // 100))
            lr = max(lr, 0.005)
    
    return best_a, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based minimax refinement (min peak of f*g with fixed sum), adaptive
line search, geometric blending, subgradient descent with soft-max smoothing,
and coordinate descent for final polish. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)


def _build_constraint_matrix(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A[k, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]
    return A


def _solve_lp_min_peak(f, S):
    """LP: min M s.t. f*g <= M (elementwise), sum(g) = S, g >= 0.
    Returns (g, M_opt) or (None, None)."""
    n = len(f)
    A_f = _build_constraint_matrix(f)
    # Full constraint matrix: [A_f, -1] for each row (size (2n-1) x (n+1))
    # Plus sum(g) <= S and -sum(g) <= -S
    n_rows = 2 * n - 1 + 2
    A = np.zeros((n_rows, n + 1))
    b = np.zeros(n_rows)

    A[:2 * n - 1, :n] = A_f
    A[:2 * n - 1, n] = -1.0
    b[:2 * n - 1] = 0.0

    A[2 * n - 1, :n] = 1.0
    b[2 * n - 1] = S

    A[2 * n, :n] = -1.0
    b[2 * n] = -S

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            g = res.x[:n]
            M_val = float(res.x[n])
            return g, M_val
    except Exception:
        pass
    return None, None


def _solve_lp_fixed_M(f, M):
    """LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0. Returns g or None."""
    n = len(f)
    A_f = _build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_f, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _line_search(a, g, S, best_score, max_bisect=25):
    """Find best t in [0,1] for a_new = (1-t)*a + t*g that improves score."""
    n = len(a)
    # Try t=1 first
    a_try = g.copy()
    s_try = _compute_obj(a_try)
    if s_try < best_score:
        return 1.0, a_try, s_try

    # Bisection
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_s = best_score
    best_a = None
    for _ in range(max_bisect):
        mid = (lo + hi) / 2.0
        a_try = (1.0 - mid) * a + mid * g
        a_try = _normalize(a_try, S)
        s_try = _compute_obj(a_try)
        if s_try < best_s:
            best_s = s_try
            best_t = mid
            best_a = a_try
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-8:
            break
    return best_t, best_a, best_s


def _geometric_blend(a, g, t, S):
    """Geometric (log-space) blend: a_new = a^(1-t) * g^t, normalized."""
    a_safe = np.maximum(a, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    a_new = a_safe ** (1.0 - t) * g_safe ** t
    a_new = _normalize(a_new, S)
    return a_new


def _lp_refine(a0, S, deadline, max_iters=500, verbose=True):
    """Main LP-based refinement loop. Returns (best_a, best_score)."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    best_a = a.copy()
    best_score = _compute_obj(a)
    stall_count = 0

    if verbose:
        print(f"  LP refine: n={n}, initial score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        # Solve LP to find best direction
        g, M_lp = _solve_lp_min_peak(a, S)
        if g is None:
            # Fallback: fixed-M LP
            g = _solve_lp_fixed_M(a, M)
            if g is not None and g.sum() > 1e-10:
                g = _normalize(g, S)
            else:
                break

        if g.sum() < 1e-10:
            break

        g = _normalize(g, S)

        # Try linear blend with line search
        t, a_new, s_new = _line_search(a, g, S, best_score)

        # Also try geometric blend
        for tg in [0.5, 0.8, 1.0]:
            a_geo = _geometric_blend(a, g, tg, S)
            s_geo = _compute_obj(a_geo)
            if s_geo < s_new:
                a_new = a_geo
                s_new = s_geo
                t = tg

        if a_new is not None and s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 10:
                # Reheat: add small perturbation
                rng = np.random.default_rng(it)
                a = a + 0.01 * S * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, S)
                stall_count = 0
                if time.monotonic() > deadline:
                    break
            elif stall_count >= 3:
                # Try smaller step
                a_try = (1.0 - 0.1) * a + 0.1 * g
                a_try = _normalize(a_try, S)
                s_try = _compute_obj(a_try)
                if s_try < best_score:
                    best_score = s_try
                    best_a = a_try.copy()
                    a = a_try
                    stall_count = 0
                else:
                    break

        if verbose and (it + 1) % 50 == 0:
            elapsed = time.monotonic()
            print(f"    LP iter {it+1}: score={best_score:.6f}, M={M:.4f}, "
                  f"stall={stall_count}, time={elapsed - (deadline - (1800-10)):.0f}s")

    return best_a, best_score


def _subgrad_soft(a0, S, deadline, max_iters=3000):
    """Subgradient descent with soft-max smoothing and Nesterov momentum."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    best_a = a.copy()
    best_score = _compute_obj(a)

    v = np.zeros(n)
    beta = 0.85
    lr = 0.1
    tau = 10.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal parameters
        frac = it / max_iters
        tau_cur = 10.0 + 40.0 * frac
        lr_cur = 0.1 * (1.0 + np.cos(np.pi * frac)) * 0.5
        lr_cur = max(lr_cur, 0.005)

        # Nesterov lookahead
        a_look = a - beta * v
        a_look = np.maximum(a_look, 0.0)

        c = np.convolve(a_look, a_look)

        # Soft-max gradient
        c_shifted = c - c.max()
        w = np.exp(tau_cur * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(c.argmax())] = 1.0
        else:
            w /= w_sum

        # Gradient: grad[j] = 2 * sum_k a[k] * w[j+k]
        grad = np.zeros(n)
        for j in range(n):
            k_lo = 0
            k_hi = min(2 * n - 2, j + n - 1)
            if k_hi >= k_lo:
                grad[j] = 2.0 * np.dot(a_look[:k_hi - j + 1], w[j:k_hi + 1])

        # Project (remove mean for fixed-sum constraint)
        grad -= grad.mean()

        # Momentum update
        v = beta * v + lr_cur * grad
        a = a - v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S)

        if (it + 1) % 200 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()

    # Final check
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def _coord_descent(a0, S, deadline, max_rounds=50):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    best_a = a.copy()
    best_score = _compute_obj(a)
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        indices = rng.permutation(n)

        for j in indices:
            v = a[j]
            c = np.convolve(a, a)
            current_max = c.max()

            # Try several perturbations
            scale = S / n
            best_d = 0.0
            best_peak = current_max

            for d_frac in [-2.0, -1.0, -0.5, -0.2, 0.2, 0.5, 1.0, 2.0]:
                d = d_frac * scale
                new_v = v + d
                if new_v < 0:
                    continue

                # Compute new max efficiently
                # delta_k = 2*d*a[k-j] for valid k
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)
                a_seg = a[k_lo - j:k_hi - j + 1]
                c_seg = c[k_lo:k_hi + 1].copy()
                c_seg += 2.0 * d * a_seg
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    c_seg[2 * j - k_lo] += d * d
                new_max = c_seg.max()

                if new_max < best_peak:
                    best_peak = new_max
                    best_d = d

            if best_d != 0.0:
                a[j] = v + best_d
                a = np.maximum(a, 0.0)
                a = _normalize(a, S)
                improved = True

        # Check improvement
        s = _compute_obj(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        elif not improved and rnd > 10:
            break

    return best_a, best_score

            if time.monotonic() > deadline - 10.0:
                break
            a_r, s_r = _lp_refine(best_seq, S_best, deadline - 5.0,
                                  max_iters=100, verbose=True)
            if s_r < best_score:
                best_score = s_r
                best_seq = a_r.copy()
                print(f"  Final LP pass {pass_i+1}: {best_score:.6f}")
            else:
                break

        # Final coordinate descent
        if time.monotonic() < deadline - 5.0:
            a_f, s_f = _coord_descent(best_seq, S_best, deadline - 2.0, max_rounds=20)
            if s_f < best_score:
                best_score = s_f
                best_seq = a_f.copy()
                print(f"  Final CD: {best_score:.6f}")

    # --- Final evaluation and return ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        # Use the best we have
        seq_to_return = best_seq if final_score <= eval_score + 1e-9 else best_seq
        return [float(x) for x in seq_to_return]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
