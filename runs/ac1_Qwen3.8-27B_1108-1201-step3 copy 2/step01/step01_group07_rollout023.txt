import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def evaluate_objective(a):
    """Compute 2n * max(conv) / (sum)^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def fw_optimize(a0, time_limit, gamma=None, top_k_frac=0.05, seed=0):
    """
    Frank-Wolfe style optimization.
    At each step, compute subgradient from top-k peaks,
    then move toward the coordinate with minimum gradient.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_iter = 0
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Find top-k peaks
        k = max(1, int(top_k_frac * (2*n - 1)))
        threshold = np.partition(conv, (2*n-1) - k)[(2*n-1) - k]  # k-th largest
        active = np.where(conv >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for ki in active:
            k_int = int(ki)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                grad[j_start:j_end+1] += a[k_int - j_end : k_int - j_start + 1][::-1]
        grad /= len(active)
        
        # Frank-Wolfe step: move toward argmin grad
        j_star = np.argmin(grad)
        
        # Step size
        if gamma is None:
            gamma_t = 2.0 / (t_iter + 2.0)
        else:
            gamma_t = gamma
        
        a_new = (1 - gamma_t) * a + gamma_t * np.zeros(n)
        a_new[j_star] = gamma_t  # add mass to j_star
        
        # Actually, FW update is: a_new = (1-γ)*a + γ*s where s = e_{j_star}
        # So a_new[j_star] = (1-γ)*a[j_star] + γ
        a_new = (1 - gamma_t) * a
        a_new[j_star] += gamma_t
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            a = a_new
            conv = conv_new
        else:
            patience += 1
            a = a_new  # still move (FW theory says this)
            conv = conv_new
        
        if patience >= max_patience:
            break
    
    return best_a, best_obj / 1.0  # already normalized since sum(a)=1

import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective_normalized(a):
    """a is normalized (sum=1). Returns 2n * max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def subgradient_topk(conv, a, top_k_frac=0.05):
    """Compute subgradient using top-k convolution peaks."""
    n = len(a)
    size = 2 * n - 1
    k = max(1, int(top_k_frac * size))
    # Find k-th largest value
    if k >= size:
        active = np.arange(size)
    else:
        threshold = np.partition(conv, size - 1 - k)[size - 1 - k]
        active = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for ki in active:
        k_int = int(ki)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:

    return grad

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def method_subgrad(a0, time_limit, top_k_frac=0.05, eta0=0.1, seed=0):
    """Projected subgradient descent with top-k peaks."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    t0 = time.monotonic()
    t_iter = 0
    patience = 0
    max_patience = 100
    
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        grad = subgradient_topk(conv, a, top_k_frac)
        
        # Projected gradient step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        conv = conv_new
        
        # Adapt step size
        if patience >= max_patience:
            eta *= 0.7
            patience = 0
            if eta < 0.001:
                # Random restart within the current basin
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                eta = eta0 * 0.5
                max_patience = 50
    
    return best_a

def method_fw_simple(a0, time_limit, top_k_frac=0.05, seed=0):
    """Simple Frank-Wolfe: move toward argmin gradient coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_iter = 0
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        grad = subgradient_topk(conv, a, top_k_frac)
        j_star = np.argmin(grad)
        
        gamma = 2.0 / (t_iter + 2.0)
        a_new = (1 - gamma) * a
        a_new[j_star] += gamma
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        conv = conv_new
        
        if patience >= max_patience:
            break
    
    return best_a

def method_fw_lp(a0, time_limit, seed=0, max_iters=5000):
    """LP-based Frank-Wolfe."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_iter = 0
    patience = 0
    max_patience = 100
    
    # Build constraint matrix A_g: (2n-1) x n, A_g[k,j] = a[k-j] if valid
    size = 2 * n - 1
    A_g = np.zeros((size, n))
    for j in range(n):
        for k in range(j, j + n):
            A_g[k, j] = a[k - j]
    
    while time.monotonic() - t0 < time_limit and t_iter < max_iters:
        t_iter += 1
        M = conv.max()
        
        # Solve LP: max sum(g) s.t. A_g @ g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        A_ub = A_g
        b_ub = np.full(size, M)
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g /= g_sum
        
        # FW update
        gamma = 2.0 / (t_iter + 2.0)
        a_new = (1 - gamma) * a + gamma * g
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        conv = conv_new
        
        # Update A_g (it depends on a)
        for j in range(n):
            for k in range(j, j + n):
                A_g[k, j] = a[k - j]
        
        if patience >= max_patience:
            break
    
    return best_a
