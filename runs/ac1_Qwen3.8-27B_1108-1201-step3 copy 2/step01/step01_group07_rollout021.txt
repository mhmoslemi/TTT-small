import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_score(a):
    """Compute 2n*max(conv)/sum^2 assuming a is non-negative, sum > 0."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S)

def mirror_descent_peak(a0, time_limit, eta0=1.0, peak_sharpen=1.0):
    """
    Mirror descent focusing on the actual max peak (not soft-max).
    Uses the subgradient of the max function directly.
    """

    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    patience = 0
    max_patience = 100
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        if max_val < 1e-15:
            break
        
        # Subgradient of conv[max_idx] w.r.t. a
        k = max_idx
        sg = np.zeros(n)
        for i in range(n):
            j1 = k - i
            j2 = i - k
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            # If k == 2*i, the term a[i]*a[i] has derivative 2*a[i] not a[i]+a[i]

                        conv = conv_new

                    else:

                    break

        a_new[i_src] -= amount
        a_new[i_dst] += amount
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-15:

        a_new /= S_new
        
        conv_new = fast_convolve(a_new)

def generate_initializations(n, seed, parent=None):
    """Generate diverse initializations for a given length."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Two Gaussians
    g1 = np.exp(-0.5 * ((x - n * 0.25) / (0.15 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - n * 0.75) / (0.15 * n)) ** 2)
    bi = g1 + g2
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Quadratic phase (related to chirp / Golomb ruler ideas)
    for alpha in [0.618, 0.7, 0.5 + 0.1 * np.sqrt(2)]:
        q = (x * alpha) % 1.0
        q = q + 0.1  # Keep positive
        q /= q.sum()
        inits[f'quad_{alpha:.3f}'] = q

    # Cosine with irrational frequency
    for freq in [1.618, 1.414]:
        c = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        c = c + 0.05
        c /= c.sum()
        inits[f'cos_{freq:.3f}'] = c

    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Random with structure (blocks)
    block_size = max(1, n // 10)
    blocks = rng.uniform(0.5, 2.0, size=(n // block_size + 1,))
    r2 = np.repeat(blocks, block_size)[:n]
    r2 = np.append(r2, blocks[-1] * rng.uniform(0.5, 2.0, size=(n - len(r2),))) if len(r2) < n else r2
    r2 = np.maximum(r2, 0)
    r2 /= r2.sum()
    inits['blocks'] = r2

    # Parent interpolation
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.size >= n:
            pp = p[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    return inits

        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 10:
            break

    # Ensure best is included
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 40
    time_per_deep = max(15.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 30:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 30)
        if t_alloc < 5:
            break

        # Mirror descent (exact subgradient)
        t_md = t_alloc * 0.3
        a_md, obj_md = mirror_descent(a0, time_limit=t_md, eta0=0.6)

        # Local search
        t_ls = t_alloc * 0.3
        a_ls, obj_ls = local_search_peak(a_md, time_limit=t_ls)
        if obj_ls < obj_md:
            a_md = a_ls
            obj_md = obj_ls

        # ILS for the remainder
        t_ils = t_alloc * 0.4
        a_ils, obj_ils = iterated_local_search(a_md, time_limit=t_ils, num_restarts=8)
        if obj_ils < obj_md:
            a_md = a_ils
            obj_md = obj_ils

        score = float(evaluate_sequence(a_md.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_md.copy()
                print(f"  Deep [{n}, {name}]: {score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Final intensive polish on the best
    print("Phase 3: Final polish...")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            n_best = len(best_sequence)

            # Round 1: Extended local search
            t1 = remaining * 0.3
            a1, o1 = local_search_peak(best_sequence, time_limit=t1)
            if o1 < obj_normalized(best_sequence):
                best_sequence = a1

            # Round 2: ILS with more restarts
            t2 = remaining * 0.4
            a2, o2 = iterated_local_search(best_sequence, time_limit=t2, num_restarts=30)
            if o2 < obj_normalized(a1) if best_sequence is a1 else o2 < obj_normalized(best_sequence):
                best_sequence = a2

            # Round 3: Mirror descent with very small eta for final polish
            t3 = remaining * 0.2
            a3, o3 = mirror_descent(best_sequence, time_limit=t3, eta0=0.2)
            if o3 < obj_normalized(best_sequence):
                best_sequence = a3

            # Round 4: One more local search
            t4 = remaining * 0.1
            a4, o4 = local_search_peak(best_sequence, time_limit=t4)
            if o4 < obj_normalized(best_sequence):
                best_sequence = a4

            score = float(evaluate_sequence(best_sequence.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score

    print(f"  Phase 3 done. Best: {best_score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        print(f"FINAL SCORE: {final_score:.6f} (n={len(best_sequence)})")
        return best_sequence.tolist()
    else:
        fallback = np.ones(128)
        print(f"FALLBACK used. Score: {evaluate_sequence(fallback.tolist()):.6f}")
        return fallback.tolist()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: multi-phase optimization combining exact-subgradient
    mirror descent, peak-targeted local search, and iterated local search.
    Explores multiple sequence lengths and diverse initializations.
    """
    t_start = time.monotonic()
    print(f"Starting optimization with budget={budget_s}s")

    result = run_full_optimization(seed=seed, budget_s=budget_s)

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")

    # Ensure valid output
    if not isinstance(result, list) or len(result) == 0:
        result = [1.0] * 128

    final_val = float(evaluate_sequence(result))
    print(f"Returning sequence of length {len(result)}, score={final_val:.6f}")

    return result
