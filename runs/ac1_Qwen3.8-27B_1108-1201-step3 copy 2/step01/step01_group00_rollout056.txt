```python
from scipy.optimize import linprog

def lp_direction(f, M):
    """Find g maximizing sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    # c = -1 (we want to maximize sum(g), linprog minimizes)
    c = -np.ones(n)
    # A_ub: each row is a constraint (f*g)[m] <= M
    # (f*g)[m] = sum_i f[i] * g[m-i] (with bounds)
    # For constraint m: sum over valid i of f[i] * g[m-i] <= M
    # Let j = m-i, so i = m-j
    # sum over j of f[m-j] * g[j] <= M (where 0 <= m-j < n and 0 <= j < n)
    
    A_rows = []
    for m in range(2*n-1):
        row = np.zeros(n)
        for j in range(n):
            i = m - j
            if 0 <= i < n:
                row[j] = f[i]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return -res.x
    return None
```

```python
def fft_convolve(a, b):
    n = len(a)

    size = 1
    while size < n + m - 1:

    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]
```

```python
"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines LP-based directional optimization, L-BFGS-B gradient descent with smooth max,
annealed Adam with adaptive temperature, and structured local search.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    """Fast score: 2*n*max(conv)/(sum)^2"""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[m] = (f*g)[m] for m=0..2n-2."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def solve_lp_direction(f, M):
    """Solve: max 1^T g s.t. ConvMatrix(f) @ g <= M*1, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(f_init, time_limit, rng, verbose=True):
    """LP-based iterative optimization with adaptive step size."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sqrt(2.0 * n)
    f = f / np.sum(f) * S

    best_f = f.copy()
    best_score = local_score_fast(f)
    t0 = time.monotonic()
    t = 0.3
    stall_count = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp_direction(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        g = g / sum_g * S

        improved = False
        for t_try in [t, 0.5 * t, 0.25 * t, 0.1 * t, 0.05, 0.02, 0.01, 0.005]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            score_new = local_score_fast(f_new)
            if score_new < best_score * (1.0 - 1e-9):
                f = f_new
                best_score = score_new
                best_f = f.copy()
                improved = True
                break

        if improved:
            stall_count = 0
            t = max(0.005, t * 0.97)
        else:
            stall_count += 1
            if stall_count > 30:
                f = f * np.exp(rng.normal(0, 0.01, n))
                f = np.maximum(f, 0.0)
                stall_count = 0
            t = max(0.005, t * 0.9)

        it += 1
        if verbose and it % 50 == 0:
            print(f"    LP it={it}, score={best_score:.6f}, t={t:.4f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def gradient_adam(f_init, time_limit, rng, verbose=True):
    """Annealed Adam on softmax parameterization with smooth max."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sqrt(2.0 * n)
    theta = np.log(np.maximum(f, 1e-10))

    best_score = local_score_fast(f)
    best_theta = theta.copy()
    t0 = time.monotonic()

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr_base = 1.5

    tau_phases = [(20.0 / n, 0.25), (5.0 / n, 0.25), (1.0 / n, 0.25), (0.2 / n, 0.25)]

    for tau, frac in tau_phases:
        phase_end = t0 + frac * time_limit
        while time.monotonic() < phase_end:
            e = np.exp(theta - np.max(theta))
            a = e / np.sum(e) * S
            conv = fft_convolve(a, a)
            c_max = np.max(conv)
            shifted = np.clip((conv - c_max) / tau, -50, 50)
            w = np.exp(shifted)
            w = w / np.sum(w)

            # Gradient via FFT cross-correlation
            size = 4 * n
            A_fft = np.fft.rfft(a, size)
            W_fft = np.fft.rfft(w, size)
            corr = np.fft.irfft(A_fft * W_fft, size)[:3 * n - 1]
            grad_a = 2.0 * corr[n - 1:2 * n - 1]

            g_theta = a * (grad_a - np.dot(a, grad_a) / S)

            step += 1
            momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
            momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
            m1_hat = momentum1 / (1.0 - beta1 ** step)
            m2_hat = momentum2 / (1.0 - beta2 ** step)
            lr = lr_base / np.sqrt(step + 1)
            theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

            if step % 200 == 0:
                e_ch = np.exp(theta - np.max(theta))
                a_ch = e_ch / np.sum(e_ch) * S
                score = local_score_fast(a_ch)
                if score < best_score:
                    best_score = score
                    best_theta = theta.copy()
                    if verbose:
                        print(f"      Adam tau={tau:.2e}, step={step}, score={score:.6f}")

    a_best = np.exp(best_theta - np.max(best_theta))
    a_best = a_best / np.sum(a_best) * S
    return a_best, best_score


def lbfgs_optimize(f_init, time_limit, rng, verbose=True):
    """L-BFGS-B with smooth max on softmax parameterization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sqrt(2.0 * n)
    theta = np.log(np.maximum(f, 1e-10))

    best_score = local_score_fast(f)
    best_theta = theta.copy()
    t0 = time.monotonic()

    def obj_grad(theta_in):
        e = np.exp(theta_in - np.max(theta_in))
        a = e / np.sum(e) * S
        conv = fft_convolve(a, a)
        tau = 2.0 / n
        c_max = np.max(conv)
        shifted = np.clip((conv - c_max) / tau, -50, 50)
        w = np.exp(shifted)
        w = w / np.sum(w)
        val = c_max + tau * np.log(np.sum(w))

        # Gradient
        size = 4 * n
        A_fft = np.fft.rfft(a, size)
        W_fft = np.fft.rfft(w, size)
        corr = np.fft.irfft(A_fft * W_fft, size)[:3 * n - 1]
        grad_a = 2.0 * corr[n - 1:2 * n - 1]
        g_theta = a * (grad_a - np.dot(a, grad_a) / S)
        return val, g_theta

    class CB:
        def __init__(self):
            self.count = 0

        def __call__(self, th):
            self.count += 1
            if self.count % 20 == 0:
                e_c = np.exp(th - np.max(th))
                a_c = e_c / np.sum(e_c) * S
                sc = local_score_fast(a_c)
                if sc < best_score:
                    nonlocal best_score, best_theta
                    best_score = sc
                    best_theta = th.copy()
                    if verbose:
                        print(f"      L-BFGS cb={self.count}, score={sc:.6f}, elapsed={time.monotonic()-t0:.1f}s")
            if time.monotonic() - t0 > time_limit:
                return True
            return False

    cb = CB()
    try:
        res = minimize(obj_grad, theta, jac=True, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 5000},
                       callback=cb)
    except Exception as ex:
        if verbose:
            print(f"      L-BFGS exception: {ex}")

    a_best = np.exp(best_theta - np.max(best_theta))
    a_best = a_best / np.sum(a_best) * S
    return a_best, best_score


def local_search_polish(a_init, time_limit, rng, verbose=True):
    """Structured local search with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted peak reduction: reduce top contributors to peak
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.55:
            # Anti-symmetric perturbation (reduces central peak)
            half = n // 2
            idx1 = int(rng.integers(0, half))
            idx2 = n - 1 - idx1
            delta = rng.normal(0, 0.03)
            candidate[idx1] *= np.exp(delta)
            candidate[idx2] *= np.exp(-delta)
        elif r < 0.70:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.85:
            # Partial symmetrization with noise
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.015, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
                    print(f"      LS score={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):

def make_fourier_init(n, rng):

                best_score = score_ls
                best_seq = best_ls.copy()
                print(f"  Final LS: {best_score:.6f}")

    # Return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
