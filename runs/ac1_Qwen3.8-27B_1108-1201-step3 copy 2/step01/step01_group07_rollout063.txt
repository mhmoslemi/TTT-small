```python
    return conv_result[n-1:2*n-1][::-1].copy()

    if s < 1e-30:
        a = np.ones(n) / n

        a /= s

    best_a = a.copy()
```

Now `dc[k]/dx[j] = sum_i (da[i]/dx[j]) * ∂c[k]/∂a[i] = sum_i a[i](δ_{ij} - a[j]) * ∂c[k]/∂a[i]`

And `∂c[k]/∂a[i] = a[k-i]` (with the appropriate bounds, same as before).

So `dc[k]/dx[j] = sum_i a[i](δ_{ij} - a[j]) * a[k-i]`
`= sum_i δ_{ij} * a[i] * a[k-i] - a[j] * sum_i a[i] * a[k-i]`
`= a[j] * a[k-j] - a[j] * c[k]`
`= a[j] * (a[k-j] - c[k])`

(where `a[k-j] = 0` if `k-j` is out of bounds).

So `dF/dx[j] = 2n * sum_k w[k] * a[j] * (a[k-j] - c[k])`
`= 2n * a[j] * (sum_k w[k] * a[k-j] - sum_k w[k] * c[k])`
`= 2n * a[j] * (sg[j] - w·c)`

where `sg[j] = sum_k w[k] * a[k-j]` (the same subgradient as before!) and `w·c = sum_k w[k] * c[k]`.

So the gradient through the softmax reparameterization is:
`grad_x[j] = 2n * a[j] * (sg[j] - w·c)`

This is nice! It's the same computation as the subgradient (just one additional scalar `w·c`), times `a[j]`.

```python
    """
    x: parameter vector (length n)
    Returns (objective, gradient) for the log-sum-exp smoothed objective.
    a = softmax(x)
    objective = 2n * (1/temp) * log(sum exp(temp * conv_k))

    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()

    conv = fast_convolve(a)
    
    # Log-sum-exp
    tc = temp * conv
    tc -= tc.max()
    lse = tc.max() + np.log(np.exp(tc).sum())
    obj = 2.0 * n * lse / temp

    sg = compute_subgradient_fft(a, scaled_exp)
    
    grad = 2.0 * n * a * (sg - w_dot_c)
    
    return obj, grad
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with softmax reparameterization
2. FFT-accelerated subgradients for O(n log n) per iteration

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_subgradient_fft(a, w):
    n = len(a)
    w_flip = w[::-1].copy()
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:conv_len]
    return conv_result[n-1:2*n-1][::-1].copy()

def lse_objective_and_grad(x, temp, n):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a /= a_sum
    
    conv = fast_convolve(a)
    
    tc = temp * conv
    tc_max = tc.max()
    tc_shifted = tc - tc_max
    exp_tc = np.exp(tc_shifted)
    log_sum = tc_max + np.log(exp_tc.sum())
    obj = 2.0 * n * log_sum / temp
    
    w = exp_tc / exp_tc.sum()
    w_dot_c = np.dot(w, conv)
    sg = compute_subgradient_fft(a, w)
    
    grad = 2.0 * n * a * (sg - w_dot_c)
    
    return obj, grad

def lbfgs_optimize(a0, time_limit, temp=200.0):

        obj, grad = lse_objective_and_grad(x, temp, n)
        conv = fast_convolve(np.exp(x - x.max()) / np.exp(x - x.max()).sum())
        true_obj = 2.0 * n * conv.max()
        if true_obj < best_obj:
            nonlocal best_obj, best_a  # Can't use nonlocal in top-level
        return obj, grad
    
    # Actually, can't use nonlocal. Let me use a different approach.
    # Just run L-BFGS and evaluate at the end.
    
    def objective_and_grad(x):
        return lse_objective_and_grad(x, temp, n)
    
    result = minimize(
        objective_and_grad, x0, method='L-BFGS-B',
        jac=True,
        options={'maxiter': 5000, 'maxfun': 5000, 'ftol': 1e-15, 'gtol': 1e-10}
    )

    a_opt = np.exp(x_opt - x_opt.max())

        best_obj = obj
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on log-sum-exp
smoothed objective with softmax reparameterization, combined with
FFT-accelerated mirror descent for final polishing.
Explores multiple sequence lengths and diverse initializations.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgradient_fft(a, w):
    """
    Compute sg[j] = sum_i a[i] * w[j+i] for j=0,...,n-1.
    a has length n, w has length 2n-1. Uses FFT.
    """
    n = len(a)
    w_flip = w[::-1].copy()
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:conv_len]
    return conv_result[n-1:2*n-1][::-1].copy()


def lse_func(x, n, temp):
    """Compute (LSE objective, gradient) for softmax parameterization."""
    x_s = x - x.max()
    a = np.exp(x_s)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a /= a_sum

    conv = fast_convolve(a)

    tc = temp * conv
    tc_max = tc.max()
    tc_s = tc - tc_max
    e = np.exp(tc_s)
    lse = tc_max + np.log(e.sum()) / 1.0  # (1/temp)*log(sum exp(t*c)) = (1/temp)*(tc_max + log(sum exp(tc_s)))
    # Wait: LSE = (1/temp) * log(sum exp(temp*c[k]))
    # = (1/temp) * (temp*max(c) + log(sum exp(temp*(c[k]-max(c)))))
    # = max(c) + (1/temp) * log(sum exp(temp*(c[k]-max(c))))
    lse_val = conv.max() + np.log(e.sum()) / temp
    obj = lse_val

    w = e / e.sum()
    w_dot_c = float(np.dot(w, conv))
    sg = compute_subgradient_fft(a, w)

    # Gradient of LSE w.r.t. x[j] = 2 * a[j] * (sg[j] - w·c)
    grad = 2.0 * a * (sg - w_dot_c)

    return obj, grad


def run_lbfgs(a0, temp, max_iter=5000):
    """Run L-BFGS to minimize LSE objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a /= a.sum()
    x0 = np.log(a)

    def func(x):
        return lse_func(x, n, temp)

    res = minimize(func, x0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxls': 50})
    x_opt = res.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    a_opt = np.maximum(a_opt, 0.0)
    return a_opt


def mirror_descent_polish(a0, time_limit):
    """Polish using exact subgradient at the peak (hard max)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.5
    patience = 0

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-18:
            break

        # Find peaks (within 0.1% of max)
        thresh = max_c * 0.99
        peak_idx = np.where(conv >= thresh)[0]
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0 / len(peak_idx)

        sg = compute_subgradient_fft(a, w)
        sg_c = sg - sg.mean()

        lf = -eta * sg_c
        lf -= lf.max()
        a = a * np.exp(lf)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        conv = fast_convolve(a)
        cur = 2.0 * n * conv.max()

        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= 40:
            eta *= 0.5
            if eta < 0.001:
                break
            patience = 0

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = n * 0.5

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))

    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    bi /= bi.sum()
    inits.append(('bimodal', bi))

    # Cosine
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits.append(('cosine', cos_b))

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('random', r))

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(('ramp', ramp))

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent', pp))
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent_interp', pp))

    return inits
```

`LSE(c, t) = (1/t) * log(sum_k exp(t * c[k]))`

This satisfies `LSE(c, t) >= max(c)` and `LSE(c, t) → max(c)` as `t → ∞`.

```

tc_max = tc.max()
lse = tc_max / t + log(sum(exp(tc - tc_max))) / t

```

```python

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on log-sum-exp
smoothed objective with softmax parameterization (FFT-accelerated),
combined with mirror descent polish. Explores multiple sequence lengths
and diverse initializations with temperature continuation.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgradient_fft(a, w):
    """
    Compute sg[j] = sum_i a[i] * w[j+i] for j=0,...,n-1.
    a has length n, w has length 2n-1. Uses FFT for O(n log n).
    """
    n = len(a)
    w_flip = w[::-1].copy()
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)[:conv_len]
    return conv_result[n-1:2*n-1][::-1].copy()


def lse_func(x, n, temp):
    """
    Compute (LSE objective, gradient) for softmax parameterization.
    a = softmax(x), minimize (1/temp)*log(sum exp(temp*conv_k)).
    """
    x_s = x - x.max()
    a = np.exp(x_s)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(n)
    a /= a_sum

    conv = fast_convolve(a)

    # Numerically stable log-sum-exp
    tc = temp * (conv - conv.max())
    e = np.exp(tc)
    lse_val = conv.max() + np.log(e.sum()) / temp
    obj = lse_val

    w = e / e.sum()
    w_dot_c = float(np.dot(w, conv))
    sg = compute_subgradient_fft(a, w)

    # Gradient: dF/dx[j] = 2*a[j]*(sg[j] - w·c)
    grad = 2.0 * a * (sg - w_dot_c)

    return obj, grad


def run_lbfgs(a0, temp, max_iter=5000):
    """Run L-BFGS to minimize LSE objective. Returns optimal a."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a /= a.sum()
    x0 = np.log(a)

    def func(x):
        return lse_func(x, n, temp)

    res = minimize(func, x0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'ftol': 1e-18,
                            'gtol': 1e-14, 'maxls': 50})
    x_opt = res.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    a_opt = np.maximum(a_opt, 0.0)
    return a_opt


def mirror_descent_polish(a0, time_limit):
    """Polish using exact subgradient at peaks (hard max) with multiplicative updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.5
    patience = 0

    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-18:
            break

        # Find peaks within 0.5% of max
        thresh = max_c * 0.995
        peak_idx = np.where(conv >= thresh)[0]
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0 / len(peak_idx)

        sg = compute_subgradient_fft(a, w)
        sg_c = sg - sg.mean()

        lf = -eta * sg_c
        lf -= lf.max()
        a = a * np.exp(lf)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        conv = fast_convolve(a)
        cur = 2.0 * n * conv.max()

        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= 40:
            eta *= 0.5
            if eta < 0.001:
                break
            patience = 0

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = n * 0.5

    inits.append(('uniform', np.ones(n) / n))

    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits.append((f'gauss_{sf}', g))

    bi = np.zeros(n)
    w_id = max(1, n // 5)
    bi[:w_id] = 1.0
    bi[-w_id:] = 1.0
    bi /= bi.sum()
    inits.append(('bimodal', bi))

    cos_b = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cos_b /= cos_b.sum()
    inits.append(('cosine', cos_b))

    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('random', r))

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits.append(('ramp', ramp))

    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent', pp))
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits.append(('parent_interp', pp))

    return inits
```
