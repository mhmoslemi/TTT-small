import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize
from scipy.optimize import minimize as scipy_minimize

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient_weighted(a, top_indices, c, top_vals):
    """Compute weighted subgradient using top lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        # (a*g)[k] = sum_i a[i] * g[k-i], so d/dg = a[k-i] reversed
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * a[k - hi:k - lo + 1][::-1]
    return grad

def lp_direction(a, top_indices, top_vals):
    """Solve LP to find the steepest descent direction for L∞ norm of conv.
    
    Minimize t
    Subject to: (a*g)[k] <= t for k in top_indices
                sum(g) = 1
                g >= 0
    """
    n = len(a)
    m = len(top_indices)
    
    # Variables: g[0], ..., g[n-1], t
    # (a*g)[k] = sum_i a[i] * g[k-i] <= t
    # For each k, the coefficient of g[j] is a[k-j] if 0 <= k-j < n, else 0.
    
    # Objective: minimize t => c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: (a*g)[k] - t <= 0 for each k in top_indices
    # A_ub has m rows, n+1 columns
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    weights = top_vals / top_vals.sum()
    
    for idx, k in enumerate(top_indices):
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[idx, j] = weights[idx] * a[k - j]
        A_ub[idx, n] = -weights[idx]
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    else:
        # Fallback: uniform direction
        return np.ones(n) / n

def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)

def run_adam_smoothed(a_init, n, max_time, rng, lr_init, tau, top_k):
    """Adam with smoothed max (softmax) for better gradients."""
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Softmax weights for smoothing
        c_shifted = c - c.max()
        e = np.exp(tau * c_shifted)
        w = e / e.sum()
        
        # Only use top-k for efficiency
        top_k_actual = min(top_k, 2*n-1)
        top_indices = np.argsort(c)[::-1][:top_k_actual]
        top_w = w[top_indices]
        top_w = top_w / top_w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w_val, k in zip(top_w, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1**t)
        v_hat = v_adam / (1 - beta2**t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        lr = lr_init * (0.7 ** (float(t) / 500.0))

            noise = rng.normal(0, 0.005 * S / n, n)
            a = project_simplex(a + noise, S)
        
        # Adapt temperature
        tau = max(0.1, tau * 0.999)
    
    return best_local, best_score

def run_lbbfgs_smoothed(a_init, n, max_time, tau_initial=10.0):
    """L-BFGS on the smoothed objective using log-parameterization."""
    n = len(a_init)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    
    # Log-parameterization: a = S * softmax(b)
    # Handle zeros by adding small epsilon
    a_safe = np.maximum(a, 1e-10)
    a_safe = a_safe / a_safe.sum() * S
    b = np.log(a_safe / S)  # b_j = log(a_j / S), so a_j = S * exp(b_j) / sum(exp(b_j))
    
    tau = tau_initial
    
    def objective_and_grad(b_param):
        b_param = np.asarray(b_param, dtype=np.float64)
        # softmax
        b_shifted = b_param - b_param.max()
        exp_b = np.exp(b_shifted)
        a_local = S * exp_b / exp_b.sum()

        c = fft_convolve_self(a_local)
        
        # Smoothed max
        c_shifted = c - c.max()
        e = np.exp(tau * c_shifted)
        w = e / e.sum()
        
        # smooth_max = max(c) + log(sum(exp(tau * (c - max(c))))) / tau
        smooth_val = c.max() + np.log(e.sum()) / tau
        
        # Gradient w.r.t. a
        # d smooth_val / d a[j] = sum_k w[k] * d c[k] / d a[j]
        # d c[k] / d a[j] = 2 * a[k-j] (for valid indices)
        grad_a = np.zeros(n)

                grad_a[lo:hi + 1] += 2.0 * w[k] * a_local[k - hi:k - lo + 1][::-1]
        
        # Gradient w.r.t. b
        # a_j = S * exp(b_j) / sum(exp(b_i))
        # d a_j / d b_m = a_j * (delta_jm - a_m / S)
        grad_b = np.zeros(n)
        for j in range(n):
            grad_b[j] = grad_a[j] * a_local[j] - (a_local[j] / S) * np.dot(grad_a, a_local)
        
        return smooth_val, grad_b
    
    best_b = b.copy()
    best_val, _ = objective_and_grad(best_b)
    start = time.monotonic()
    
    current_b = b.copy()
    current_val = best_val
    
    while time.monotonic() - start < max_time:
        result = scipy_minimize(objective_and_grad, current_b, method='L-BFGS-B',

        if result.fun < best_val:

        if b_val < best_val:
            best_val = b_val
            best_b = current_b.copy()
        
        # Reduce temperature
        tau *= 0.5
        if tau < 0.01:
            break
        
        # Check time

            break
    
    # Convert back to a
    b_final = best_b - best_b.max()
    exp_b = np.exp(b_final)
    a_final = S * exp_b / exp_b.sum()
    a_final = np.clip(a_final, 0, 1000)
    
    c_final = fft_convolve_self(a_final)
    final_score = c_final.max()

Minimize 2*n*max(self-convolution)/sum^2 over non-negative sequences.

Algorithm:
  1. Initialize from known best (height_sequence_1) or random.
  2. Multi-stage optimization:
     - Coarse Adam with FFT-based subgradient
     - Smoothed L-BFGS (log-parameterization) 
     - Fine Adam with adaptive top-k
     - Ultra-fine Adam (top-k=1, just the peak)
  3. Periodic perturbation to escape local minima.
  4. Return best sequence found.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_fft(a, w):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    a has length n, w has length 2n-1 (the softmax weights on convolution values).
    """
    n = len(a)
    # We need: grad[j] = 2 * sum_m a[m] * w[m+j]
    # = 2 * ifft(FFT(a) * conj(FFT(w)))[j]  (with zero-padding)
    # Linear cross-correlation requires size >= n + (2n-1) - 1 = 3n-2
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    r = np.real(ifft(A * np.conj(W)))
    return 2.0 * r[:n]


def softmax_weights(c, tau, top_k=None):
    """Compute softmax weights over convolution values."""
    if top_k is not None and top_k < len(c):
        # Only keep top-k for efficiency
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        c_shifted = c_top - c_top.max()
        e = np.exp(tau * c_shifted)
        w = np.zeros(len(c))
        w[idx] = e / e.sum()
        return w
    else:
        c_shifted = c - c.max()
        e = np.exp(tau * c_shifted)
        return e / e.sum()


def run_adam_fft(a_init, n, max_time, rng, lr_init, tau_init, top_k_init):
    """Adam optimizer with FFT-based gradient computation."""
    S = np.sqrt(2.0 * float(n))
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    else:
        a = np.full(n, S / n)
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    tau = tau_init
    top_k = top_k_init
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Weighted subgradient via softmax
        w = softmax_weights(c, tau, top_k)
        grad = compute_grad_fft(a, w)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1**t)
        v_hat = v_adam / (1 - beta2**t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Decay learning rate
        lr = lr_init * (0.5 ** (float(t) / 1000.0))
        
        # Adapt temperature (decrease for sharper weights)
        tau = max(0.5, tau_init * (0.9995 ** t))
        
        # Adapt top-k (increase over time for finer resolution)
        top_k = min(2*n-1, top_k_init + t // 200)
        
        # Occasional perturbation
        if t % 3000 == 0 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_local, best_score


def run_lbbfgs(a_init, n, max_time, tau_init=5.0):
    """L-BFGS on smoothed objective with log-parameterization."""
    S = np.sqrt(2.0 * float(n))
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    else:
        a = np.full(n, S / n)
    
    # Log-parameterization
    a_safe = np.maximum(a, 1e-10)
    a_norm = a_safe / a_safe.sum()  # probability distribution
    b0 = np.log(a_norm)  # b_j = log(a_j / sum(a))
    
    tau = tau_init
    best_score = np.inf
    best_b = b0.copy()
    
    def obj_and_grad(b_param):
        b_param = np.asarray(b_param, dtype=np.float64)
        # Softmax to get normalized a
        b_shifted = b_param - b_param.max()
        exp_b = np.exp(b_shifted)
        a_local = S * exp_b / exp_b.sum()
        
        # Convolution
        c = fft_convolve_self(a_local)
        
        # Smoothed max
        c_shifted = c - c.max()
        e = np.exp(tau * c_shifted)
        smooth_val = c.max() + np.log(e.sum()) / tau
        w = e / e.sum()
        
        # Gradient w.r.t. a
        grad_a = compute_grad_fft(a_local, w)
        
        # Gradient w.r.t. b
        # a_j = S * exp(b_j) / sum(exp(b_i))
        # Let p_j = a_j / S (normalized)
        p_j = a_local / S
        # d a_j / d b_m = a_j * (δ_jm - p_m) * ... 
        # Actually: a_j = S * p_j where p_j = exp(b_j)/sum(exp(b_i))
        # d p_j / d b_m = p_j * (δ_jm - p_m)
        # d a_j / d b_m = S * p_j * (δ_jm - p_m) = a_j * (δ_jm - p_m)
        # d f / d b_m = sum_j (d f / d a_j) * (d a_j / d b_m)
        #              = sum_j grad_a[j] * a_j * (δ_jm - p_m)
        #              = grad_a[m] * a_m - p_m * sum_j grad_a[j] * a_j
        
        dot_grad_a = np.dot(grad_a, a_local)
        grad_b = grad_a * a_local - p_j * dot_grad_a
        
        return smooth_val, grad_b
    
    start = time.monotonic()
    current_b = b0.copy()
    current_val, _ = obj_and_grad(current_b)
    
    while time.monotonic() - start < max_time:
        result = scipy_minimize(obj_and_grad, current_b, method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': 30, 'maxfun': 100,
                                         'ftol': 1e-12, 'gtol': 1e-9})
        if result.success and result.fun < current_val:
            current_val = result.fun
            current_b = result.x.copy()
        
        # Convert to a and check actual (unsmoothed) score
        b_shifted = current_b - current_b.max()
        exp_b = np.exp(b_shifted)
        a_check = S * exp_b / exp_b.sum()
        c_check = fft_convolve_self(a_check)
        actual_score = c_check.max()
        
        if actual_score < best_score:
            best_score = actual_score
            best_b = current_b.copy()
        
        # Reduce temperature
        tau *= 0.4
        if tau < 0.1:
            break
        
        # Check time
        elapsed = time.monotonic() - start
        if elapsed > max_time * 0.7:
            break
    
    # Return best
    b_final = best_b - best_b.max()
    exp_b = np.exp(b_final)
    a_final = S * exp_b / exp_b.sum()
    a_final = np.clip(a_final, 0
