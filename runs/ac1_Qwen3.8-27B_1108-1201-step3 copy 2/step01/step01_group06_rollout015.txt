```python
    """Compute gradient of smooth max approximation of self-convolution."""
    c = fft_convolve_self(a)
    # Softmax weights
    Tc = T * c
    Tc = Tc - Tc.max()  # numerical stability
    exp_Tc = np.exp(Tc)
    weights = exp_Tc / exp_Tc.sum()
    
    # Gradient is weighted sum of individual lag gradients
    grad = np.zeros(n, dtype=np.float64)
    # Only use lags with significant weight
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for k in active:
        k = int(k)
        w = weights[k] * T
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    return grad
```

Let me refine the pairwise swap. For a given pair (i, j), we want to find the optimal transfer of mass from i to j (or vice versa) that minimizes the max convolution.

If we transfer δ from position i to position j (so a[i] decreases by δ and a[j] increases by δ), the change in b[k] is:
db[k] = -δ * (2*a[k-i]) + δ * (2*a[k-j]) = 2δ * (a[k-j] - a[k-i])

where a[m] = 0 for invalid m.

So the new max is max_k (b[k] + 2δ * (a[k-j] - a[k-i])).

We want to find δ that minimizes this, subject to a[i] - δ >= 0 (so δ <= a[i]) and δ can be negative (meaning transfer from j to i, with |δ| <= a[j]).

This is a 1D optimization problem! We can solve it by bisection or by finding the point where the max of the linear functions is minimized.

Actually, for a given pair (i,j), the function `f(δ) = max_k (b[k] + 2δ * d[k])` where `d[k] = a[k-j] - a[k-i]` is a convex function of δ (max of linear functions). We can minimize it with ternary search or by finding the point where the subgradient contains 0.

This is attractive because:
- It's exact (no step size issues)
- It's fast (O(n) per evaluation of f(δ) if we precompute the relevant values)
- It naturally respects non-negativity (δ is bounded)

However, doing this for all pairs O(n^2) is too expensive for n=8192. But we can be smart:
- Only consider pairs where the "direction" d = a[·-j] - a[·-i] has a max that's close to the current max
- Or: only consider pairs that are "hot" (where a[i] or a[j] contributes significantly to the max lags)

A simpler approach: random pairwise swaps. Pick random pairs and try to optimize the transfer. This is like a form of simulated annealing or random local search.

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  * Start from provided height_sequence_1 (n=8192)
  * Normalize to sum = sqrt(2n) so objective = max self-convolution
  * Main phase: Adam on smooth-max (log-sum-exp) objective with
    temperature annealing from low T (broad) to high T (focused)
  * Interleave pairwise swap local search (exact 1D optimization
    over mass transfer between two positions)
  * Periodic verification with evaluate_sequence
  * Return best verified sequence
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(a, c, T, n):
    """Gradient of (1/T)*log(sum exp(T*c_k)) w.r.t. a."""
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_vals = np.exp(Tc_shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    # Only process lags with non-negligible weight
    threshold = 1e-12
    active = np.where(weights > threshold)[0]
    for k in active:
        k = int(k)
        w = T * weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def hard_topk_grad(a, c, n, k):
    """Weighted subgradient using top-k lags by value."""
    top = np.argsort(c)[::-1][:k]
    vals = c[top]
    total = vals.sum()
    if total < 1e-15:
        return np.zeros(n)
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(k):
        weight = vals[idx] / total
        kk = int(top[idx])
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            grad[lo:hi + 1] += weight * 2.0 * a[kk - hi:kk - lo + 1][::-1]
    return grad


def optimize_transfer(a, i, j, c, n):
    """Find optimal mass transfer δ from position i to position j.
    f(δ) = max_k (c[k] + 2δ*(a[k-j] - a[k-i]))
    subject to -a[j] <= δ <= a[i]."""
    # Build d[k] = 2*(a[k-j] - a[k-i])
    d = np.zeros(2 * n - 1, dtype=np.float64)
    # a[k-j] valid for k in [j, j+n-1]
    lo_j = j
    hi_j = min(2 * n - 1, j + n)
    if lo_j < hi_j:
        d[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]
    # a[k-i] valid for k in [i, i+n-1]
    lo_i = i
    hi_i = min(2 * n - 1, i + n)
    if lo_i < hi_i:
        d[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]
    
    lo_bound = -a[j]
    hi_bound = a[i]
    if hi_bound - lo_bound < 1e-15:
        return a, c.max()
    
    # Coarse grid search
    n_evals = 30
    deltas = np.linspace(lo_bound, hi_bound, n_evals)
    best_delta = 0.0
    best_val = c.max()
    for delta in deltas:
        val = np.max(c + delta * d)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    if abs(best_delta) < 1e-15:
        return a, c.max()
    
    # Refine
    step = (hi_bound - lo_bound) / n_evals
    lo2 = max(lo_bound, best_delta - step)
    hi2 = min(hi_bound, best_delta + step)
    for delta in np.linspace(lo2, hi2, 50):
        val = np.max(c + delta * d)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    if abs(best_delta) < 1e-15:
        return a, c.max()
    
    a_new = a.copy()
    a_new[i] -= best_delta
    a_new[j] += best_delta
    return a_new, best_val


def run_smooth_adam(a, n, S, max_time, rng, T_start, T_end, lr_init, switch_T):
    """Adam on smooth-max with temperature annealing.
    Returns (best_a, best_score)."""
    a = project_simplex(a.copy(), S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-10
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Animate T
        progress = min(1.0, float(t) / switch_T)
        T = T_start + (T_end - T_start) * progress
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = smooth_max_grad(a, c, T, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.5 ** (float(t) / (switch_T * 2.0)))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        # Occasional random perturbation to escape local minima
        if t % 5000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0.0, 0.005 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_hard_subgrad(a, n, S, max_time, top_k_start, top_k_end, lr_init):
    """Hard subgradient with growing top-k."""
    a = project_simplex(a.copy(), S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        progress = min(1.0, float(t) / 10000)
        k = int(top_k_start + (top_k_end - top_k_start) * progress)
        k = max(1, k)
        
        grad = hard_topk_grad(a, c, n, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        
        lr = lr_init * (1.0 / (1.0 + 0.001 * t))
        a_new = a - lr * grad
        a = project_simplex(a_new, S)
    
    return best_local, best_score


def pairwise_local_search(a, c, n, budget, rng, n_swaps):
    """Perform n_swaps random and directed pairwise swaps."""
    a = a.copy()
    c = c.copy()
    best_score = c.max()
    start = time.monotonic()
    improved = 0
    
    for swap_i in range(n_swaps):
        if time.monotonic() - start > budget:
            break
        
        # Mix of directed and random swaps
        if rng.random() < 0.6:
            # Directed: use the max lag's gradient
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            grad_slice = a[k_star - hi:k_star - lo + 1][::-1]
            # i = position with highest a[m]*grad contribution
            weights = a[lo:hi + 1] * grad_slice
            i = lo + int(np.argmax(weights))
            # j = position with lowest gradient (preferably non-zero)
            grad_full = np.zeros(n)
            grad_full[lo:hi + 1] = grad_slice
            # Find j with min grad but a[j] > 0
            candidates = np.where(a > 1e-10)[0]
            if len(candidates) == 0:
                continue
            j = candidates[np.argmin(grad_full[candidates])]
        else:
            # Random
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
        
        c = fft_convolve_self(a)
        new_a, new_score = optimize_transfer(a, i, j, c, n)
        if new_score < best_score - 1e-12:
            a = new_a
            c = fft_convolve_self(a)
            best_score = new_score
            improved += 1
    
    return a, best_score, improved
```

```python
    d = np.zeros(2 * n - 1, dtype=np.float64)

        else:

        return a, c.max()
    
    a_new = a.copy()
    a_new[i] -= best_delta
    a_new[j] += best_delta
    return a_new, best_val
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  * Start from provided height_sequence_1 (n=8192)
  * Normalize to sum = sqrt(2n) so objective = max self-convolution
  * Phase 1: Adam on smooth-max (log-sum-exp) objective with temperature annealing
  * Phase 2: Hard subgradient with growing top-k
  * Phase 3: Extensive pairwise swap local search with exact 1D bisection
  * Phase 4: Final refinement and verification

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(a, c, T, n):
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_vals = np.exp(Tc_shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n, dtype=np.float64)
    threshold = 1e-12
    active = np.where(weights > threshold)[0]
    for k in active:
        k = int(k)
        w = T * weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def hard_topk_grad(a, c, n, k):
    top = np.argsort(c)[::-1][:k]
    vals = c[top]
    total = vals.sum()
    if total < 1e-15:
        return np.zeros(n, dtype=np.float64)
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(k):
        weight = vals[idx] / total
        kk = int(top[idx])
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            grad[lo:hi + 1] += weight * 2.0 * a[kk - hi:kk - lo + 1][::-1]
    return grad


def optimize_transfer(a, i, j, c, n):
    d = np.zeros(2 * n - 1, dtype=np.float64)
    lo_j = j
    hi_j = min(2 * n - 1, j + n)
    if lo_j < hi_j:
        d[lo_j:hi_j] += 2.0 * a[:hi_j - lo_j]

    hi_i = min(2 * n - 1, i + n)
    if lo_i < hi_i:
        d[lo_i:hi_i] -= 2.0 * a[:hi_i - lo_i]

    lo_bound = -a[j]
    hi_bound = a[i]
    if hi_bound - lo_bound < 1e-15:
        return a, c.max()

    lo = lo_bound
    hi = hi_bound
    best_delta = 0.0
    best_val = c.max()

    a_new[i] -= best_delta

    return a_new, best_val


def run_smooth_adam(a, n, S, max_time, rng, T_start, T_end, lr_init, switch_T):
    a = project_simplex(a.copy(), S)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-10
    start = time.monotonic()
    prev_best = best_score

    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, float(t) / switch_T)
        T = T_start + (T_end - T_start) * progress

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            prev_best = cur

        grad = smooth_max_grad(a, c, T, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.5 ** (float(t) / (switch_T * 2.0)))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

            noise = rng.normal(0.0, noise_scale, n)
            a = project_simplex(a + noise, S)
            prev_best = c.max()

def pairwise_local_search(a, n, budget, rng, max_swaps):
    a = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    start = time.monotonic()
    improved = 0
    stuck_count = 0

    for _ in range(max_swaps):
        if time.monotonic() - start > budget:
            break

        r = rng.random()
        if r < 0.55:
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            grad_slice = a[k_star - hi:k_star - lo + 1][::-1]
            weights_i = a[lo:hi + 1] * grad_slice
            i = lo + int(np.argmax(weights_i))
            grad_full = np.zeros(n, dtype=np.float64)
            grad_full[lo:hi + 1] = grad_slice
            candidates = np.where(a > 1e-10)[0]
            if len(candidates) < 2:
                j = int(rng.integers(0, n))
            else:
                j = candidates[np.argmin(grad_full[candidates])]
        elif r < 0.8:
            i = int(rng.integers(0, n))
            offset = int(rng.integers(1, max(2, n // 10)))
            j = (i + offset) % n
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

        if i == j:
            continue

        new_a, new_score = optimize_transfer(a, i, j, c, n)
        if new_score < best_score - 1e-13:
            a = new_a
            c = fft_convolve_self(a)
            best_score = new_score
            improved += 1
            stuck_count = 0
        else:
            stuck_count += 1

    return a, best_score, improved

    a3, s3, n_imp3 = pairwise_local_search(a3, n, phase3_budget, rng, max_swaps=5000000)
    if s3 < global_best_score:
        global_best_score = s3
        global_best_a = a3.copy()
    print("Phase 3 (pairwise): max_c=%.8f, improvements=%d" % (s3, n_imp3))

    # Phase 4: Another smooth Adam pass on best (10% of time)
    left = deadline - time.monotonic()
    if left > 5.0:
        a4, s4 = run_smooth_adam(global_best_a, n, S, left * 0.5, rng,
                                 T_start=20.0, T_end=100.0,
                                 lr_init=0.4 * S / n, switch_T=8000)
        if s4 < global_best_score:
            global_best_score = s4
            global_best_a = a4.copy()
        print("Phase 4 (refine adam): max_c=%.8f" % s4)

        left2 = deadline - time.monotonic()
        if left2 > 3.0:
            a5, s5, n_imp5 = pairwise_local_search(global_best_a, n, left2 * 0.7, rng, max_swaps=3000000)
            if s5 < global_best_score:
                global_best_score = s5
                global_best_a = a5.copy()
            print("Phase 4b (pairwise polish): max_c=%.8f, improvements=%d" % (s5, n_imp5))

    # Try a shorter length as alternative
    left3 = deadline - time.monotonic()
    if left3 > 30.0:
        a_short, s_short = try_new_length(4096, left3 * 0.8, rng, 999)
        if s_short < global_best_score:
            global_best_score = s_short
            global_best_a = a_short.copy()
            print("Short length 4096: max_c=%.8f (NEW BEST)" % s_short)
        else:
            print("Short length 4096: max_c=%.8f (not better)" % s_short)

    # Final verification
    result = np.clip(global_best_a, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)
    final_score = evaluate_sequence(result.tolist())
    print("FINAL: n=%d, max_c=%.8f, verified=%.6f" % (len(result), global_best_score, final_score))

    return [float(x) for x in result]
```
