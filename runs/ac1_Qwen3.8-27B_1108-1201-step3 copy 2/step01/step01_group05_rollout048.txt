```python
def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return out / s

def chebyshev_shape(n, coeffs, taper_pow=0.0):
    """Generate shape from Chebyshev coefficients + power taper."""
    x = np.linspace(0.0, 1.0, n)
    z = 2*x - 1
    p = np.ones(n)

        T_prev = np.ones(n)
        T_curr = z.copy()
        p += coeffs[0] * T_curr

            T_next = 2*z*T_curr - T_prev
            T_prev, T_curr = T_curr, T_next
            p += coeffs[d] * T_next

        p *= np.power(np.maximum(x, 1e-6), taper_pow) * np.power(np.maximum(1-x, 1e-6), taper_pow)
    return safe_normalize(np.maximum(p, 0.01))
```

**Alternative approach: Direct optimization of the power spectrum**

The linear autocorrelation c_k is related to the DFT of the sequence. Specifically, if we zero-pad a to length N >= 2n-1, then the DFT of the zero-padded convolution is |DFT(a)|^2.

So max_k c_k = max of the inverse DFT of |DFT(a)|^2.

We want to minimize this max. In the frequency domain, this is equivalent to making |DFT(a)|^2 as "flat" as possible (since a flat spectrum gives a delta-like autocorrelation, minimizing the peak).

But we have the constraint a >= 0, which is a constraint in the time domain.

This suggests an alternating approach:
1. In the time domain: project onto {a >= 0, sum(a) = 1}
2. In the frequency domain: try to flatten |DFT(a)|^2

For step 2, we could:
- Compute the current power spectrum P(ω) = |A(ω)|^2
- Find the frequencies where P(ω) is above its mean
- Adjust A(ω) at those frequencies to reduce P(ω)
- Transform back to time domain
- Project onto the constraint set

This is similar to the "spectral equalization" approach used in some signal processing applications.

Let me implement this:

```python
def spectral_shaping(p, rng, n_iter=50):
    """Try to flatten the power spectrum."""
    n = int(p.size)
    pad_len = next_fast_len(2*n - 1)
    p = project_simplex(p)
    
    for _ in range(n_iter):
        F = fft(p, pad_len)
        P = np.abs(F)**2
        target = float(P.mean())
        
        # Find frequencies where P > target
        above = P > target * 1.01
        
        if not np.any(above):
            break
        
        # Reduce amplitude at those frequencies
        F_new = F.copy()
        for i in np.where(above)[0]:
            F_new[i] *= 0.99  # Slightly reduce
            # Maintain conjugate symmetry
            conj_i = (pad_len - i) % pad_len
            if conj_i != i:
                F_new[conj_i] = np.conj(F_new[i])
        
        p_new = np.real(ifft(F_new, pad_len)[:n])
        p_new = project_simplex(p_new)
        
        if internal_score(p_new) < internal_score(p):
            p = p_new
    
    return p
```

Actually, a faster approach: instead of iterating, we can compute the "ideal" flat spectrum and take a step towards it:

```python
def spectral_shaping_fast(p, alpha=0.1):
    """One step of spectral flattening."""
    n = int(p.size)
    pad_len = next_fast_len(2*n - 1)
    F = fft(p, pad_len)
    P = np.abs(F)**2
    target = float(np.median(P))  # Use median to be robust
    
    # Adjust amplitude: new_amp = sqrt(alpha * P + (1-alpha) * target)
    # This moves the spectrum towards flat
    new_P = alpha * P + (1 - alpha) * target
    new_amp = np.sqrt(new_P)
    
    # Preserve phase
    phase = np.angle(F)
    F_new = new_amp * np.exp(1j * phase)
    
    # Maintain conjugate symmetry
    F_new[0] = float(F_new[0])  # Real
    if pad_len % 2 == 0:
        F_new[pad_len//2] = float(F_new[pad_len//2])  # Real
    for i in range(1, (pad_len+1)//2):
        F_new[pad_len - i] = np.conj(F_new[i])
    
    p_new = np.real(ifft(F_new, pad_len)[:n])
    return project_simplex(p_new)
```
