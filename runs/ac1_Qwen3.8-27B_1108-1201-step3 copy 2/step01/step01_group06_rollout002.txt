"""
Minimize 2*n*max_conv/sum^2 using log-sum-exp smooth approximation
with temperature annealing and Adam optimizer.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_lse_grad(a, c, t):
    """Gradient of (1/t)*log(sum exp(t*c)) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    # Numerically stable softmax of t*c
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    w = exp_tc / exp_tc.sum()
    
    # grad[m] = 2 * sum_j w[m+j] * a[j]
    # = 2 * cross-correlation of w (length 2n-1) with a (length n)
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * corr[:n]
    return grad

def run_lse_anneal(a_init, n, S, max_time, rng, 
                    t_start=0.5, t_end=100.0, n_stages=20,
                    steps_per_stage=500, lr_init=0.05, decay=0.9):
    """Log-sum-exp annealing with Adam."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / s)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_maxc = np.inf
    best_a = a.copy()
    start_time = time.monotonic()
    
    # Temperature schedule
    n_total_steps = n_stages * steps_per_stage
    lr = lr_init
    
    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        t = t_start * (t_end / t_start) ** frac
        
        for _ in range(steps_per_stage):
            if time.monotonic() - start_time > max_time:
                break
            step += 1
            
            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best_maxc:
                best_maxc = cur_max
                best_a = a.copy()
            
            grad = compute_lse_grad(a, c, t)
            
            # Adam update
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-15:
                grad = grad  # don't normalize, Adam handles scaling
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new, S)
            
            # Learning rate decay
            lr = lr_init * (decay ** (step / 200.0))
        
        # Print progress
        elapsed = time.monotonic() - start_time
        print(f"  stage {stage}/{n_stages} t={t:.2f} lr={lr:.6f} "
              f"maxc={cur_max:.6f} best={best_maxc:.6f} time={elapsed:.1f}s")
    
    return best_a, best_maxc

```python
"""
Minimize 2*n*max_conv/sum^2 using log-sum-exp smooth approximation with
temperature annealing and Adam optimizer with simplex projection.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c[k])) with respect to a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    # Numerically stable softmax
    tc = t * c
    tc_shifted = tc - tc.max()
    exp_tc = np.exp(tc_shifted)
    w = exp_tc / exp_tc.sum()
    
    # grad[m] = 2 * sum_j w[m+j] * a[j] for j=0..n-1
    # This is 2 * cross-correlation of w (length 2n-1) with a (length n)
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * corr[:n]
    return grad


def run_lse_anneal(a_init, n, S, max_time, rng,
                   t_start=0.5, t_end=100.0, n_stages=20,
                   steps_per_stage=10000, lr_init=0.1, lr_decay=0.95):
    """
    Log-sum-exp annealing with Adam optimizer and simplex projection.
    Returns (best_a, best_maxc).
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    a = a[:n]
    if len(a) < n:
        a = np.pad(a, (0, n - len(a)))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / s)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_maxc = np.inf
    best_a = a.copy()
    start_time = time.monotonic()
    
    c = fft_convolve_self(a)
    best_maxc = c.max()
    print(f"    Init maxc = {best_maxc:.6f}")
    
    for stage in range(n_stages):
        frac = stage / max(1, n_stages - 1)
        t = t_start * (t_end / t_start) ** frac
        lr = lr_init * (lr_decay ** (step / 500.0))
        
        stall_count = 0
        local_best = best_maxc
        
        for _ in range(steps_per_stage):
            if time.monotonic() - start_time > max_time:
                break
            
            step += 1
            c = fft_convolve_self(a)
            cur_max = c.max()
            
            if cur_max < best_maxc:
                best_maxc = cur_max
                best_a = a.copy()
                stall_count = 0
            elif cur_max < local_best:
                local_best = cur_max
            
            grad = compute_lse_grad(a, c, t)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new, S)
            
            # Random restart if stalled
            if step % 5000 == 0:
                if cur_max - best_maxc > 0.01 and rng.random() < 0.3:
                    noise_scale = 0.05 * S / n
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    m = np.zeros(n)
                    v = np.zeros(n)
            
        elapsed = time.monotonic() - start_time
        if stage % 5 == 0:
            print(f"    stage {stage:2d}/{n_stages} t={t:8.2f} lr={lr:.6f} "
                  f"maxc={cur_max:.6f} best={best_maxc:.6f} time={elapsed:.1f}s")
    
    return best_a, best_maxc


def run_subgradient_polish(a_init, n, S, max_time, rng, top_k=10, lr_init=0.01):
    """
    Fine-tuning with top-k subgradient and very small learning rate.
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    a = a[:n]
    if len(a) < n:
        a = np.pad(a, (0, n - len(a)))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / s)
    
    best_maxc = a.max()  # placeholder
    c = fft_convolve_self(a)
    best_maxc = c.max()
    best_a = a.copy()
    start_time = time.monotonic()
    step = 0
    
    while time.monotonic() - start_time < max_time:
        step += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_maxc:
            best_maxc = cur_max
            best_a = a.copy()
        
        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-15:
            break
        weights = top_vals / total
        
        grad = np.zeros(n)
        for w_val, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
        
        lr = lr_init * (0.999 ** step)
        if lr < 1e-10:
            break
        a = project_simplex(a - lr * grad, S)
    
    return best_a, best_maxc
