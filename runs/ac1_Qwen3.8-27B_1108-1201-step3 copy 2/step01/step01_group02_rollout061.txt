    n_a = len(a)
    n_p = len(p)
    out_len = n_p + n_a - 1
    N = 1 << (out_len - 1).bit_length()
    P = fft(p, N)
    A = fft(a, N)
    # h = ifft(FFT(p) * conj(FFT(a)))
    result = ifft(P * np.conj(A)).real
    return result[:n_a]


def compute_gradient(a, T):
    """
    Compute gradient of f_T(a) = (2n/T) * log(sum(exp(T*c))) where c = conv(a,a).
    Returns (gradient, c).
    """
    n = len(a)
    c = conv_auto_fft(a)
    # Stable softmax
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shifted)
    p = exp_Tc / exp_Tc.sum()
    # Gradient: 2n * cross_correlate(p, a)
    h = cross_corr_fft(p, a)
    grad = 2.0 * n * h
    return grad, c


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} (probability simplex)."""
    n = len(v)
    x = np.sort(v)[::-1]
    cssv = np.cumsum(x)
    cond = x * np.arange(1, n + 1) > (cssv - 1.0)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    return a / a.sum()


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a / a.sum()


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n <= 2:
        return np.ones(n) / n
    x = t / (n - 1)
    a = np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_smooth_init(n, rng, n_bumps=None):
    a = np.zeros(n)
    if n_bumps is None:
        n_bumps = int(rng.integers(2, 5))
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 10), max(3, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def fista_optimize(a, time_limit, init_T=2.0, final_T=100.0, n_phases=8, verbose=True):
    """
    FISTA with soft-max smoothing and temperature annealing.
    """
    start_time = time.monotonic()
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"      FISTA init: score={best_score:.6f}, n={n}")

    T_values = np.geomspace(init_T, final_T, n_phases)

    for phase, T in enumerate(T_values):
        phase_start = time.monotonic()
        remaining = time_limit - (time.monotonic() - start_time)
        if remaining < 5.0:
            break
        phase_budget = min(remaining * 0.4, 120.0)

        # Learning rate: use backtracking-free approach with estimated L
        # L ~ 4n * T * max(a) roughly, for normalized a, max(a) <= 1
        # Be conservative: lr = 1 / (4 * n * T / n + 1) = n / (4n + T)
        # Actually for f_T = (2n/T)*logsumexp(T*c), the Hessian eigenvalue is bounded by
        # T * 4 * max_contribution. Let's use L = 4 * T * (2n) / n = 8T as rough bound
        lr = 1.0 / (4.0 * T + 2.0 * n / max(1, T))
        # Simpler: lr = min(1.0, 2.0 / (T + n))
        lr = 2.0 / (T + 1.0)

        # FISTA parameters
        a_curr = a.copy()
        a_prev = a.copy()
        t_fista = 1.0

        for it in range(50000):
            if time.monotonic() - start_time > time_limit:
                break
            if time.monotonic() - phase_start > phase_budget:
                break

            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
            momentum = (t_fista - 1.0) / t_new
            y = a_curr + momentum * (a_curr - a_prev)
            y = np.maximum(0.0, y)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

            # Gradient step
            grad_y, c = compute_gradient(y, T)
            z = y - lr * grad_y
            a_new = project_simplex(z)

            a_prev = a_curr
            a_curr = a_new
            t_fista = t_new

            # Update best
            if it % 50 == 0:
                sc = score_sequence(a_curr)
                if sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    if verbose and it % 500 == 0:
                        print(f"      FISTA ph{phase}(T={T:.1f}) it{it}: score={sc:.6f}")

        a = a_curr.copy()

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      FISTA done: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    """Build sparse matrix for convolution with a."""
    n = len(a)
    n_rows = 2 * n - 1
    # Matrix M such that M @ x = conv(a, x)
    # M[k, j] = a[k-j] if 0 <= k-j < n, else 0
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        for k in range(j, j + n):
            row_idx.append(k)
            col_idx.append(j)
            data.append(a[k - j])
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    data = np.array(data, dtype=np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=200, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = project_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"      LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        c = conv_auto_fft(a)
        score = 2.0 * n * c.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 3):
                print(f"      LP iter {it:3d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 30:
            if verbose:
                print(f"      LP stagnant at iter {it}, best={best_score:.6f}")
            break

        M = c.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"      LP converged (sum_g={sum_g:.10f}) at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = project_simplex(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            alpha2 = alpha * 0.3
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a = project_simplex(a_new)

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      LP done: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = project_simplex(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"      LS start: score={best_score:.6f}")

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(3, int(rng.integers(n // 200, max(4, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        noise_scale = 0.05 + 0.1 * (1 - (time.monotonic() - start_time) / time_limit_s)
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))
        if rng.random() < 0.2:
            # Global shift
            shift = int(rng.integers(-n // 10, n // 10 + 1))
            candidate = np.roll(candidate, shift)
            candidate = np.maximum(0, candidate)
        candidate = project_simplex(candidate)
        score = score_sequence(candidate)
        if score < best_score - 1e-14:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"      LS done: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses FISTA with soft-max + LP refinement + local search.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Check if we have a parent sequence to start from
    parent_available = False
    parent_seq = None
    try:
        global height_sequence_1
        if 'height_sequence_1' in dir() or True:
            hs = height_sequence_1
            if hs is not None and len(hs) > 0:
                parent_seq = np.array([float(x) for x in hs])
                parent_available = True
                print(f"Parent sequence available, length={len(parent_seq)}")
    except Exception as e:
        print(f"Could not access parent: {e}")

    # Configuration: (n, init_type, time_fraction)
    # We want to explore different lengths and initializations
    configs = []

    if parent_available and parent_seq is not None:
        n_parent = len(parent_seq)
        configs.append((n_parent, 'parent', 0.20))
        # Also try nearby lengths
        configs.append((max(100, n_parent - 100), 'raised_cosine', 0.12))
        configs.append((n_parent + 100, 'raised_cosine', 0.12))
        configs.append((n_parent + 200, 'gaussian', 0.10))
        configs.append((max(200, n_parent // 2), 'blackman', 0.08))
        configs.append((n_parent * 2, 'raised_cosine', 0.10))
        configs.append((n_parent * 2, 'gaussian', 0.08))
        configs.append((max(300, n_parent - 50), 'random', 0.08))
        configs.append((n_parent + 300, 'random', 0.08))
        configs.append((n_parent * 3, 'raised_cosine', 0.08))
    else:
        configs = [
            (500, 'raised_cosine', 0.15),
            (1000, 'raised_cosine', 0.15),
            (1500, 'raised_cosine', 0.12),
            (2000, 'raised_cosine', 0.12),
            (1000, 'gaussian', 0.10),
            (2000, 'gaussian', 0.10),
            (500, 'blackman', 0.08),
            (1000, 'blackman', 0.08),
            (1000, 'random', 0.08),
            (2000, 'random', 0.05),
            (3000, 'raised_cosine', 0.05),
        ]

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.8)
        if time_limit < 20.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time_budget={time_limit:.0f}s ===")

        if init_type == 'parent' and parent_seq is not None and len(parent_seq) == n:
            a = parent_seq.copy()
            a = project_simplex(a)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'random':
            a = random_smooth_init(n, rng)
        else:
            a = initial_raised_cosine(n)

        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        # Phase 1: FISTA optimization
        fista_budget = time_limit * 0.55
        a_fista, score_fista = fista_optimize(a, time_limit=fista_budget,
                                              init_T=2.0, final_T=80.0,
                                              n_phases=8, verbose=True)

        # Phase 2: LP refinement
        lp_budget = time_limit * 0.30
        a_lp, score_lp = lp_refine(a_fista, time_limit_s=lp_budget,
                                   max_iters=100, verbose=True)

        # Phase 3: Local search polish
        ls_budget = time_limit * 0.10
        a_ls, score_ls = local_search_refine(a_lp, time_limit_s=ls_budget,
                                             rng=rng, verbose=False)

        # Pick the best from all phases
        candidates = [(a_fista, score_fista), (a_lp, score_lp), (a_ls, score_ls)]
        phase_best_a, phase_best_score = min(candidates, key=lambda x: x[1])

        if phase_best_score < best_score:
            best_score = phase_best_score
            best_seq = phase_best_a.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={n}, "
                  f"elapsed={time.monotonic()-total_start:.1f}s")

    # Final polishing on the best sequence found
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            # LP polish
            lp_final_budget = remaining * 0.4
            a_p, score_p = lp_refine(best_seq, time_limit_s=lp_final_budget,
                                     max_iters=50, verbose=True)
            if score_p < best_score:
                best_score = score_p
                best_seq = a_p.copy()

            # FISTA polish with high T
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                a_f, score_f = fista_optimize(best_seq, time_limit=remaining2 * 0.4,
                                              init_T=30.0, final_T=150.0,
                                              n_phases=4, verbose=True)
                if score_f < best_score:
                    best_score = score_f
                    best_seq = a_f.copy()

            # Local search
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10.0:
                a_s, score_s = local_search_refine(best_seq, time_limit_s=remaining3 * 0.5,
                                                   rng=rng, verbose=True)
                if score_s < best_score:
                    best_score = score_s
                    best_seq = a_s.copy()

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Ensure all values are valid
    best_seq = np.maximum(0.0, best_seq)
    best_seq = np.minimum(1000.0, best_seq)
    if best_seq.sum() < 0.01:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)

    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
