import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_smooth_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a) in O(n log n)."""
    n = len(a)
    b = fft_convolve_self(a)
    b_max = b.max()
    weights = np.exp((b - b_max) / tau)
    weights /= weights.sum()
    r = a[::-1]
    size = len(weights) + len(r) - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    R = fft(r, fsize)
    conv_wr = np.real(ifft(W * R))[:size]
    grad = 2.0 * conv_wr[n-1:2*n-1]
    return grad

def soft_max_value(b, tau):
    b_max = b.max()
    return b_max + tau * np.log(np.sum(np.exp((b - b_max) / tau)))

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[rho-1] - S) / rho
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * (S / s)
    
    x = a.copy()
    y = a.copy()
    t = 1.0
    step = 0.1
    best_a = a.copy()
    best_val = np.inf
    start = time.monotonic()
    
    # Estimate initial Lipschitz constant
    b = fft_convolve_self(y)
    f_val = soft_max_value(b, tau)
    
    for it in range(max_iter):
        if time.monotonic() - start > max_time:
            break
        if time.monotonic() - start > max_time - 2:
            break
            
        # Compute gradient at y
        grad = compute_smooth_grad(y, tau)
        
        # Nesterov update
        z = y - step * grad
        x_new = project_simplex(z, S)
        
        # Backtracking line search
        b_new = fft_convolve_self(x_new)
        f_new = soft_max_value(b_new, tau)
        
        # Check descent (with tolerance for numerical issues)
        # Simple: if f_new > f_val + 0.5*step*||grad||^2, reduce step
        gnorm2 = np.dot(grad, grad)
        if f_new > f_val + 0.5 * step * gnorm2 + 1e-10:
            step *= 0.5
            continue
        
        # Update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        x = x_new
        t = t_new
        step = min(step * 1.05, 1.0)  # slowly increase step
        
        b_val = fft_convolve_self(x)
        hard_max = b_val.max()
        if hard_max < best_val:
            best_val = hard_max
            best_a = x.copy()
            if it % 1000 == 0:
                print(f"  FISTA iter {it}: max_conv={hard_max:.8f}, tau={tau}, step={step:.4f}, time={time.monotonic()-start:.1f}s")
    
    return best_a, best_val

"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Key innovations:
1. Smooth max (log-sum-exp) with O(n log n) FFT-based gradient - smoother than top-k subgradient.
2. FISTA (Nesterov-accelerated proximal gradient) with backtracking line search.
3. Multiplicative weights (mirror descent in log-space) as complementary method.
4. Adaptive temperature schedule for the smooth max.
5. Structured initializations from low-autocorrelation sequence theory (alternating, multi-sinusoidal).
6. Multi-length exploration (8192, 16384).
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_smooth_grad(a, tau):
    """Compute gradient of soft-max(conv(a,a)) in O(n log n) via FFT."""
    n = len(a)
    b = fft_convolve_self(a)
    b_max = b.max()
    exponents = (b - b_max) / tau
    # Clip to avoid underflow issues
    exponents = np.maximum(exponents, -700.0)
    weights = np.exp(exponents)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(b)
        weights[int(b_max is not np.inf and np.argmax(b))] = 1.0
        wsum = 1.0
    weights /= wsum
    r = a[::-1]
    size = len(weights) + len(r) - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    R = fft(r, fsize)
    conv_wr = np.real(ifft(W * R))[:size]
    grad = 2.0 * conv_wr[n - 1:2 * n - 1]
    return grad


def soft_max_value(b, tau):
    """Compute log-sum-exp based smooth max."""
    b_max = b.max()
    exponents = (b - b_max) / tau
    exponents = np.maximum(exponents, -700.0)
    return b_max + tau * np.log(np.sum(np.exp(exponents)))


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def normalize_to_sum(a, S):
    """Clamp and normalize to sum S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fista_optimize(a_init, n, S, max_time, tau_start=0.5, tau_end=0.01,
                   tau_decay_period=500):
    """FISTA with adaptive tau and backtracking line search."""
    a = normalize_to_sum(a_init, S)
    x = a.copy()
    y = a.copy()
    t = 1.0
    step = 1.0 / (2.0 * np.sqrt(n))  # Initial step
    best_a = a.copy()
    best_hardmax = fft_convolve_self(x).max()
    start = time.monotonic()
    it = 0
    stalled = 0

    while time.monotonic() - start < max_time - 1.0:
        if it % 200 == 0:
            elapsed = time.monotonic() - start
            tau = tau_start * (tau_end / tau_start) ** (min(1.0, float(it) / tau_decay_period))
            if best_hardmax < 1.52:
                print("  FISTA it=%d tau=%.4f step=%.5f max=%.8f time=%.1fs" %
                      (it, tau, step, best_hardmax, elapsed))
            it = it

        tau = tau_start * (tau_end / tau_start) ** (min(1.0, float(it) / tau_decay_period))
        grad = compute_smooth_grad(y, tau)
        z = y - step * grad
        x_new = project_simplex(z, S)

        # Backtracking line search
        b_new = fft_convolve_self(x_new)
        f_new = soft_max_value(b_new, tau)
        b_y = fft_convolve_self(y)
        f_y = soft_max_value(b_y, tau)
        gnorm2 = np.dot(grad, grad)

        backtracks = 0
        while f_new > f_y + 0.5 * step * gnorm2 + 1e-10 and backtracks < 15:
            step *= 0.5
            z = y - step * grad
            x_new = project_simplex(z, S)
            b_new = fft_convolve_self(x_new)
            f_new = soft_max_value(b_new, tau)
            backtracks += 1

        if backtracks >= 15:
            break

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        x = x_new
        t = t_new
        step = min(step * 1.1, 1.0)
        it += 1

        hardmax = b_new.max()
        if hardmax < best_hardmax - 1e-12:
            best_hardmax = hardmax
            best_a = x_new.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled > 2000 and tau > tau_end * 1.01:
            # Force faster tau decay
            tau_decay_period = max(50, tau_decay_period // 2)

    return best_a, best_hardmax


def multiplicative_weights_optimize(a_init, n, S, max_time, eta_start=0.5,
                                    tau_start=0.5, tau_end=0.01, tau_decay_period=500):
    """Mirror descent / multiplicative weights in log-space with smooth gradient."""
    a = normalize_to_sum(a_init, S)
    x = np.log(a + 1e-15)  # Log parameters
    best_a = a.copy()
    best_hardmax = fft_convolve_self(a).max()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time - 1.0:
        tau = tau_start * (tau_end / tau_start) ** (min(1.0, float(it) / tau_decay_period))
        a = np.exp(x - x.max())
        a /= a.sum()
        a *= S

        grad_p = compute_smooth_grad(a, tau)
        # Mirror descent in log-space: x_new[i] = x[i] - eta * grad_p[i] * a[i] / S
        eta = eta_start / (1.0 + 0.01 * it)
        x = x - eta * grad_p * (a / S)
        # Renormalize
        x = x - x.max()
        it += 1

        if it % 100 == 0:
            a = np.exp(x)
            a /= a.sum()
            a *= S
            hardmax = fft_convolve_self(a).max()
            if hardmax < best_hardmax:
                best_hardmax = hardmax
                best_a = a.copy()
                if best_hardmax < 1.52:
                    print("  MW it=%d tau=%.4f max=%.8f time=%.1fs" %
                          (it, tau, best_hardmax, time.monotonic() - start))

    return best_a, best_hardmax


def generate_alternating_init(n, eps=0.5):
    """Generate alternating (low-autocorrelation) initialization."""
    p = 1.0 + eps * (-1.0) ** np.arange(n)
    p = np.maximum(p, 0.01)
    return p


def generate_multi_sinusoidal_init(n, rng):
    """Generate multi-sinusoidal initialization."""
    p = np.ones(n)
    n_modes = rng.integers(2, 8)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 2 + 1)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    p = np.maximum(p, 0.01)
    return p


def generate_gaussian_init(n, center_ratio=0.5, width_ratio=0.3):
    """Generate Gaussian-shaped initialization."""
    center = int(n * center_ratio)
    width = max(1.0, n * width_ratio)
    i = np.arange(n, dtype=np.float64)
    p = np.exp(-0.5 * ((i - center) / width) ** 2)
    p = np.maximum(p, 0.01)
    return p


def generate_uniform_init(n):
    return np.ones(n)


def get_initializations(n, rng, given_seq=None):
    """Generate diverse initializations for given length n."""
    inits = []
    inits.append(("uniform", generate_uniform_init(n)))
    inits.append(("alternating", generate_alternating_init(n, 0.5)))
    inits.append(("alt_0.3", generate_alternating_init(n, 0.3)))
    inits.append(("alt_0.7", generate_alternating_init(n, 0.7)))
    inits.append(("gaussian", generate_gaussian_init(n)))
    inits.append(("multi_sin", generate_multi_sinusoidal_init(n, rng)))
    inits.append(("multi_sin2", generate_multi_sinusoidal_init(n, rng)))
    inits.append(("random", rng.random(n) + 0.1))
    inits.append(("random2", rng.random(n) + 0.1))

    if given_seq is not None:
        given = np.asarray(given_seq, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0.0, 1000.0)
        if given.size != n:
            # Resample to length n
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, given.size)
            x_new = np.linspace(0, 1, n)
            f = interp1d(x_old, given, kind='cubic', fill_value='extrapolate')
            given_resized = np.maximum(f(x_new), 0.0)
        else:
            given_resized = given
        if given_resized.sum() > 0.01:
            inits.append(("given", given_resized))
        inits.append(("given_perturbed", given_resized + rng.normal(0, 0.1, n)))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0  # Reserve 10s for finalization

    # Get given sequence if available
    given_seq = None
    try:
        given_seq = height_sequence_1
    except NameError:
        pass

    global_best_score = np.inf
    global_best_seq = None

    print("Starting optimization. Given seq length: %s" %
          (len(given_seq) if given_seq is not None else "N/A"))

    # Phase 1: Global exploration across lengths and initializations
    lengths_to_try = [8192, 16384]
    if given_seq is not None and len(given_seq) not in lengths_to_try:
        lengths_to_try.insert(0, len(given_seq))

    phase1_budget = 0.4 * (budget_s - 10.0)
    phase1_deadline = start_time + phase1_budget
    n_total_inits = sum(len(get_initializations(n, rng)) for n in lengths_to_try)
    time_per_init = phase1_budget / max(1, n_total_inits)
    time_per_init = max(5.0, min(40.0, time_per_init))

    print("Phase 1: Global exploration (%.0fs, ~%d inits)" % (phase1_budget, n_total_inits))

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        S = np.sqrt(2.0 * float(n))
        inits = get_initializations(n, rng, given_seq)
        print("  Length n=%d, %d initializations, <%s each" %
              (n, len(inits), time_per_init))

        for name, arr in inits:
            if time.monotonic() > phase1_deadline - 5.0:
                break
            arr = np.clip(arr, 0.0, 10000.0)
            if arr.sum() < 0.01:
                continue

            # Quick FISTA run
            t0 = time.monotonic()
            a_opt, score = fista_optimize(
                arr, n, S,
                max_time=time_per_init,
                tau_start=0.5,
                tau_end=0.05,
                tau_decay_period=200
            )
            elapsed = time.monotonic() - t0

            if score < global_best_score:
                global_best_score = score
                global_best_seq = a_opt.copy()
                print("  NEW BEST: %s n=%d score=%.8f time=%.1f" %
                      (name, n, score, elapsed))
                # Verify with authoritative evaluator
                ver = evaluate_sequence(np.clip(a_opt, 0, 1000).tolist())
                if np.isfinite(ver) and ver < global_best_score:
                    global_best_score = ver
                    print("  Verified: %.8f" % ver)
                elif np.isfinite(ver):
                    print("  Verified (worse due to normalization): %.8f" % ver)
            else:
                if score < 1.55:
                    print("  %s n=%d score=%.8f (not best)" % (name, n, score))

    # Phase 2: Focused optimization on the best found
    phase2_budget = 0.35 * (budget_s - 10.0)
    phase2_deadline = start_time + phase1_budget + phase2_budget
    print("\nPhase 2: Focused optimization (%.0fs)" % phase2_budget)

    if global_best_seq is not None:
        n_best = len(global_best_seq)
        S_best = np.sqrt(2.0 * float(n_best))

        # Run FISTA with finer tau
        a2, score2 = fista_optimize(
            global_best_seq, n_best, S_best,
            max_time=phase2_budget * 0.6,
            tau_start=0.2,
            tau_end=0.01,
            tau_decay_period=800
        )
        if score2 < global_best_score:
            global_best_score = score2
            global_best_seq = a2.copy()
            print("  FISTA refined: %.8f" % score2)

        # Run multiplicative weights
        a3, score3 = multiplicative_weights_optimize(
            global_best_seq, n_best, S_best,
            max_time=phase2_budget * 0.4,
            eta_start=0.3,
            tau_start=0.2,
            tau_end=0.01,
            tau_decay_period=400
        )
        if score3 < global_best_score:
            global_best_score = score3
            global_best_seq = a3.copy()
            print("  MW refined: %.8f" % score3)

    # Phase 3: Multi-restart polishing
    phase3_deadline = deadline
    phase3_budget = max(5.0, phase3_deadline - time.monotonic())
    print("\nPhase 3: Polishing (%.0fs)" % phase3_budget)

    if global_best_seq is not None and phase3_budget > 10:
        n_polish = len(global_best_seq)
        S_polish = np.sqrt(2.0 * float(n_polish))
        n_restarts = max(2, int(phase3_budget / 30.0))
        time_per_restart = phase3_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > phase3_deadline - 2.0:
                break
            # Perturb the best
            noise_scale = 0.05 * S_polish / n_polish
            perturbed = global_best_seq + rng.normal(0, noise_scale, n_polish)
            perturbed = np.maximum(perturbed, 0.0)

            a_r, score_r = fista_optimize(
                perturbed, n_polish, S_polish,
                max_time=time_per_restart,
                tau_start=0.1,
                tau_end=0.005,
                tau_decay_period=300
            )
            if score_r < global_best_score:
                global_best_score = score_r
                global_best_seq = a_r.copy()
                print("  Restart %d: NEW BEST %.8f" % (r, score_r))
            else:
                print("  Restart %d: %.8f" % (r, score_r))

    # Final: try one more polish with very small tau
    if global_best_seq is not None and time.monotonic() < deadline - 5.0:
        n_final = len(global_best_seq)
        S_final = np.sqrt(2.0 * float(n_final))
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 10:
            a_final, score_final = fista_optimize(
                global_best_seq, n_final, S_final,
                max_time=remaining,
                tau_start=0.05,
                tau_end=0.001,
                tau_decay_period=1000
            )
            if score_final < global_best_score:
                global_best_score = score_final
                global_best_seq = a_final.copy()
                print("  Final polish: %.8f" % score_final)

    # Verify final result
    if global_best_seq is None:
        global_best_seq = np.ones(256)

    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)
    if global_best_seq.sum() < 0.01:
        global_best_seq = np.ones(256)

    final_result = evaluate_sequence(global_best_seq.tolist())
    print("\nFinal score (verified): %.8f" % final_result)
    print("Sequence length: %d, sum: %.4f, time used: %.1fs" %
          (len(global_best_seq), global_best_seq.sum(), time.monotonic() - start_time))

    return [float(x) for x in global_best_seq]
