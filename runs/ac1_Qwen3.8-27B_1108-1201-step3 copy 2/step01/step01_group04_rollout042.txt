```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses adaptive soft-max gradient descent, LP-based peak reduction, and
frequency-domain analysis. Multiple starting points of varying lengths explored.

import numpy as np

from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (S * S))


def normalize_to(a, target):
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros_like(a)
    return a * (target / S)


def soft_grad_step(a, lr, tau):
    """Gradient step minimizing soft-max of convolution."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient via FFT cross-correlation
    N = 1
    while N < 4 * n:
        N *= 2
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    cross = np.fft.ifft(W_fft * np.conj(A_fft)).real[:2 * n - 1]
    grad = 2.0 * cross[:n]
    
    # Project out mean (fixed sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    return a


def hard_grad_step(a, lr):
    """Gradient step minimizing the hard peak."""
    n = len(a)
    c = np.convolve(a, a)
    m_star = int(np.argmax(c))
    
    grad = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    return a


def solve_lp_peak(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1024:
        return None
    
    # Build convolution matrix A (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = a[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(a, deadline, max_iters=500):
    """Iterative LP-based refinement."""
    n = len(a)
    if n > 1024:
        return a
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        # Try slightly reduced M to be more aggressive
        M_try = M * 0.995
        
        g = solve_lp_peak(a, M_try)
        if g is None:
            g = solve_lp_peak(a, M)
            if g is None:
                break
        
        g_sum = float(np.sum(g))
        if g_sum < 1e-6:
            break
        
        g = normalize_to(g, S_t)
        
        # Try blending
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new, S_t)
            s = compute_obj(a_new)
            if s < best_score:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                break
        
        if not improved:
            # Small gradient kick
            a = soft_grad_step(a, 0.01, 50.0)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
    
    return best


def optimize_sequence(a0, deadline, phase1_iters=5000, phase2_iters=300):
    """Full optimization: gradient descent then LP refinement."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    
    best = a.copy()
    best_score = compute_obj(a)
    
    # Phase 1: Soft-max gradient descent
    tau_start = 5.0
    tau_end = 100.0
    lr_start = 0.2
    lr_end = 0.005
    
    for it in range(phase1_iters):
        if time.monotonic() > deadline - 60.0:
            break
        
        progress = it / phase1_iters
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)
        lr = lr_start + (lr_end - lr_start) * (0.5 * (1 + np.cos(np.pi * progress)))
        lr = max(lr, lr_end)
        
        a = soft_grad_step(a, lr, tau)
        
        if (it + 1) % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Final score after phase 1
    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    # Phase 2: LP refinement
    if time.monotonic() < deadline - 20.0:
        refined = lp_refine(best, deadline - 10.0, max_iters=phase2_iters)
        s = compute_obj(refined)
        if s < best_score:
            best_score = s
            best = refined
    
    return best, best_score
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses adaptive soft-max gradient descent with FFT-based gradient computation,
combined with LP-based peak reduction (maximize sum(g) s.t. conv(a,g) <= M),
and simulated annealing perturbations. Multiple starting points of varying
lengths are explored within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (S * S))


def normalize_to(a, target):
    """Normalize sequence to have a specific sum."""
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros_like(a)
    return a * (target / S)


def soft_grad_step(a, lr, tau):
    """One projected gradient step using soft-max weighted gradient (FFT-based)."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # Compute gradient via FFT: grad[j] = 2 * sum_i a[i] * w[j+i]
    N = 1
    while N < 4 * n:
        N *= 2
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    cross = np.fft.ifft(W_fft * np.conj(A_fft)).real[:n]
    grad = 2.0 * cross

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    return a


def multi_peak_grad_step(a, lr, gamma):
    """Gradient step using top-k peaks with exponential weighting."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(-gamma * (c_max - c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    N = 1
    while N < 4 * n:
        N *= 2
    A_fft = np.fft.fft(a, N)
    W_fft = np.fft.fft(w, N)
    cross = np.fft.ifft(W_fft * np.conj(A_fft)).real[:n]
    grad = 2.0 * cross

    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    return a


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0."""
    n = len(a)
    if n > 1100:
        return None

    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(a, deadline, max_iters=400):
    """Iterative LP-based refinement with adaptive M and blending."""
    n = len(a)
    if n > 1024:
        return a
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        # Try progressively tighter M values
        improved = False
        for m_factor in [0.99, 0.97, 0.95, 0.90, 1.0]:
            M_try = M * m_factor
            g = solve_lp(a, M_try)
            if g is None:
                continue

            g_sum = float(np.sum(g))
            if g_sum < 1e-6:
                break
            g = normalize_to(g, S_t)

            # Try blending factors
            for t in [0.8, 0.6, 0.4, 0.2, 0.1]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_to(a_new, S_t)
                s = compute_obj(a_new)
                if s < best_score - 1e-10:
                    a = a_new
                    best = a.copy()
                    best_score = s
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Small gradient kick to escape
            a = soft_grad_step(a, 0.015, 80.0)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s

        # Periodic larger perturbation
        if (it + 1) % 50 == 0 and not improved:
            n_perturb = max(5, n // 20)
            idx = np.random.choice(n, n_perturb, replace=False)
            a[idx] *= (1.0 + 0.1 * np.random.randn(n_perturb))
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S_t)

    return best


def optimize_sequence(a0, deadline, phase1_iters=4000, phase2_iters=300):
    """Full optimization: gradient descent then LP refinement."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    # Phase 1: Gradient descent with adaptive parameters
    tau_start = 3.0
    tau_end = 150.0
    lr_start = 0.25
    lr_end = 0.003

    for it in range(phase1_iters):
        if time.monotonic() > deadline - 80.0:
            break

        progress = it / phase1_iters
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.5)
        lr = lr_start + (lr_end - lr_start) * (0.5 * (1.0 + np.cos(np.pi * progress)))
        lr = max(lr, lr_end)

        if it % 3 == 0:
            a = soft_grad_step(a, lr, tau)
        else:
            a = multi_peak_grad_step(a, lr, 50.0 + 100.0 * progress)

        if (it + 1) % 300 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    # Phase 2: LP refinement
    if n <= 1024 and time.monotonic() < deadline - 20.0:
        lp_deadline = deadline - 10.0
        refined = lp_refine(best, lp_deadline, max_iters=phase2_iters)
        s = compute_obj(refined)
        if s < best_score:
            best_score = s
            best = refined

    return best, best_score


def simulated_annealing_pass(a, deadline, n_iter=2000):
    """Simple simulated annealing to escape local minima."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize_to(a, S_t)
    best = a.copy()
    best_score = compute_obj(a)

    T_start = 0.05
    T_end = 0.001
    rng_local = np.random.default_rng(12345)

    for it in range(n_iter):
        if time.monotonic() > deadline - 5.0:
            break

        progress = it / n_iter
        T = T_start + (T_end - T_start) * progress
        n_perturb = max(3, int(n * (0.05 + 0.1 * (1 - progress))))

        idx = rng_local.choice(n, n_perturb, replace=False)
        noise = rng_local.standard_normal(n_perturb) * T
        a_perturbed = a.copy()
        a_perturbed[idx] += noise
        a_perturbed = np.maximum(a_perturbed, 0.0)
        a_perturbed = normalize_to(a_perturbed, S_t)

        s_new = compute_obj(a_perturbed)
        delta = s_new - best_score

        if delta < 0 or rng_local.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_perturbed
            if s_new < best_score:
                best_score = s_new
                best = a.copy()

    return best

        best_seq, best_score = try_optimize(best_seq.copy(), deadline, best_score, best_seq, "deep_refine")

    # Phase C: Try to improve with different strategies on the best
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\n--- Phase C: Alternative strategies (best={best_score:.6f}) ---")
        n_best = len(best_seq)

        # Simulated annealing
        if time.monotonic() < deadline - 30.0:
            t0 = time.monotonic()
            sa_result = simulated_annealing_pass(best_seq, deadline - 20.0, n_iter=3000)
            sa_score = compute_obj(sa_result)
            elapsed = time.monotonic() - t0
            if sa_score < best_score:
                best_score = sa_score
                best_seq = sa_result.copy()
                print(f"  SA improved: {best_score:.6f}, t={elapsed:.1f}s")
            else:
                print(f"  SA: {sa_score:.6f} (no improvement), t={elapsed:.1f}s")

        # Try slightly different lengths by truncation/padding
        for n_try in [n_best - 64, n_best + 64, n_best + 128]:
            if n_try < 128 or n_try > 1024:
                continue
            if time.monotonic() > deadline - 25.0:
                break
            if n_try < n_best:
                a0 = best_seq[:n_try].copy()
            else:
                extra = n_try - n_best
                pad = np.linspace(best_seq[-1] * 0.6, best_seq[-1] * 0.1, extra)
                a0 = np.concatenate([best_seq, pad])
            best_seq, best_score = try_optimize(a0, deadline, best_score, best_seq, f"len_{n_try}")

    # Phase D: Final LP-only polish
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print(f"\n--- Phase D: Final LP polish (best={best_score:.6f}) ---")
        n = len(best_seq)
        if n <= 1024:
            polished = lp_refine(best_seq, deadline - 5.0, max_iters=200)
            p_score = compute_obj(polished)
            if p_score < best_score:
                best_score = p_score
                best_seq = polished.copy()
                print(f"  LP polish improved: {best_score:.6f}")
            else:
                print(f"  LP polish: {p_score:.6f} (no improvement)")

    # Return best result
    if best_seq is not None:
        final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_obj(np.array(final))
        print(f"\n=== FINAL: n={len(final)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence(final))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return final

    print("Falling back to uniform sequence.")
    return [1.0] * 256
```
