import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def conv_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    L = next_fast_len(2*n - 1)
    F = fft(a, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def conv_cross(a, b):
    """Cross-convolution a * b via FFT (full)."""
    n = len(a)
    L = next_fast_len(2*n - 1)
    Fa = fft(a, L)
    Fb = fft(b, L)
    c = np.real(ifft(Fa * Fb))
    return c[:2*n-1]

def gradient_smoothed_max(q, T):
    """
    Gradient of the smoothed max of q*q.
    Returns (grad, c, max_c, T_effective)
    """
    n = len(q)
    c = conv_full(q)
    max_c = np.max(c)
    
    # Softmax weights
    exponent = (c - max_c) / T
    w = np.exp(exponent)

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = (u * idx) > (cssum - 1.0)
    
    if not np.any(cond):
        return np.full(n, 1.0/n)
    
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0/n)
    return out / s
