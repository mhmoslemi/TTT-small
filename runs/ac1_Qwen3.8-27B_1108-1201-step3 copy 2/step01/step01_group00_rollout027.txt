import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    # Constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Size: (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    # Actually, let me vectorize this
    # A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1)
    # This is a banded matrix
    
    # Better: use the fact that A is a "convolution matrix"
    # Row k has f[n-1-k], f[n-2-k], ... starting at position k-n+1
    # Let me just construct it directly for clarity
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (f*b)[k] = sum_j f[k-j] * b[j] where 0 <= k-j < n and 0 <= j < n
        # So j ranges from max(0, k-n+1) to min(k, n-1)

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)

        A_ub[k, n] = -1.0  # -t

    A_eq[0, :n] = 1.0  # sum(b) = 1
    b_eq = [1.0]

                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]  # b, t
    else:
        return None, None

"""
LP-based iterative optimization for minimizing the peak of linear autocorrelation
of non-negative step functions. Uses a linear programming approach to find directions
that reduce the cross-correlation peak, combined with line search, multiple initializations,
and local search polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def build_conv_matrix(f, n):
    """Build the matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return A


def solve_lp_min_cross(f, n):
    """
    Solve LP: min t s.t. (f*b)[k] <= t for all k, sum(b) = 1, b >= 0.
    Returns (b_opt, t_opt) or (None, None) if infeasible.
    """
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A_ub[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], float(res.x[n])
    return None, None


def lp_optimize(f_init, n, time_budget, rng, verbose=True):
    """
    LP-based iterative peak reduction.
    """
    t0 = time.monotonic()
    f = f_init.astype(np.float64).copy()
    s = np.sum(f)
    if s < 0.01:
        return f_init, np.inf
    f = f / s  # normalize to sum = 1
    
    M = float(np.max(fft_convolve(f, f)))
    best_f = f.copy()
    best_M = M
    iter_count = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_budget:
        iter_count += 1
        # Solve LP
        b_opt, t_opt = solve_lp_min_cross(f, n)
        if b_opt is None:
            stuck_count += 1
            if stuck_count > 20:
                break
            # Perturb and retry
            f = f * np.exp(rng.normal(0, 0.02, n))
            f = np.clip(f, 1e-6, None)
            f = f / np.sum(f)
            continue
        
        # Check if LP found a useful direction
        if t_opt >= M * (1 - 1e-4):
            # LP can't find a much better direction
            stuck_count += 1
            if stuck_count > 20:
                break
            # Perturb
            f = f * np.exp(rng.normal(0, 0.02, n))
            f = np.clip(f, 1e-6, None)
            f = f / np.sum(f)
            M = float(np.max(fft_convolve(f, f)))
            continue
        
        stuck_count = 0
        
        # Line search for step size
        best_t_step = 0.01
        best_M_new = np.inf
        for t_step in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            f_new = (1 - t_step) * f + t_step * b_opt
            f_new = np.maximum(f_new, 0.0)
            s_new = np.sum(f_new)
            if s_new < 0.01:
                continue
            f_new = f_new / s_new
            M_new = float(np.max(fft_convolve(f_new, f_new)))
            if M_new < best_M_new:
                best_M_new = M_new
                best_t_step = t_step
        
        if best_M_new >= M:
            # Even the best step doesn't improve; use smallest step anyway
            f_new = (1 - 0.01) * f + 0.01 * b_opt
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / np.sum(f_new)
            M = float(np.max(fft_convolve(f_new, f_new)))
            f = f_new
            continue
        
        f = (1 - best_t_step) * f + best_t_step * b_opt
        f = np.maximum(f, 0.0)
        f = f / np.sum(f)
        M = float(np.max(fft_convolve(f, f)))
        
        if M < best_M:
            best_M = M
            best_f = f.copy()
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, M={M:.8f}, t_opt={t_opt:.8f}, t/M={t_opt/M:.4f}, time={elapsed:.1f}s")
    
    score = 2.0 * n * best_M
    return best_f, score


def local_polish(f_init, time_budget, rng):
    """
    Local search polish using targeted peak reduction.
    """
    f = f_init.astype(np.float64).copy()
    n = len(f)
    score = compute_score(f)
    best_f = f.copy()
    best_score = score
    t0 = time.monotonic()
    temperature = 0.005
    cooling_rate = 0.9995
    
    while time.monotonic() - t0 < time_budget:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted peak reduction
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            # Decrease top contributors
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            # Increase some non-contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif r < 0.8:
            # Symmetrize with noise
            sym = 0.5 * (f + f[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.02, n))
        else:
            # Small global perturbation
            candidate = f * np.exp(rng.normal(0, 0.01, n))
        
        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score = compute_score(candidate)
        
        if new_score < score:
            f = candidate
            score = new_score
            if score < best_score:
                best_score = score
                best_f = f.copy()
        elif rng.random() < np.exp(-(new_score - score) / max(temperature, 1e-12)):
            f = candidate
            score = new_score
        
        temperature *= cooling_rate
    
    return best_f, best_score


def make_init(n, rng, init_type='random'):
    """Generate an initial sequence of length n."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        return np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.4 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n + rng.uniform(0, np.pi))
        return np.clip(a, 0.01, None)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for k, amp in [(1, 0.3), (2, 0.2), (3, 0.15), (5, 0.1), (7, 0.08)]:
            a += amp * np.cos(2*np.pi*k*t/n + rng.uniform(0, 2*np.pi))
        return np.clip(a, 0.01, None)
    elif init_type == 'smooth_random':
        # Generate smooth random sequence using low-frequency noise
        freqs = rng.integers(1, 6, size=5)
        amps = rng.uniform(0.1, 0.4, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        a = np.ones(n)
        for f_, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2*np.pi*f_*t/n + ph)
        return np.clip(a, 0.01, None)
    elif init_type == 'tapered':
        # Tapered (Hann-window-like) sequence
        w = 0.5 + 0.5 * np.cos(2*np.pi*t/(n-1))
        a = 0.5 + 0.5 * w
        a += 0.05 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'bimodal':
        # Two peaks
        center1 = int(n * rng.uniform(0.2, 0.4))
        center2 = int(n * rng.uniform(0.6, 0.8))
        sigma = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5*((t-center1)/sigma)**2) + np.exp(-0.5*((t-center2)/sigma)**2)
        a += 0.05
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
