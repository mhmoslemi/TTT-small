def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    except Exception:

    exp_z = np.exp(z_shifted)

    if s < 1e-300:

        return b_val / n, None
    return None, None


def nesterov_lp_refine(a0, time_limit, n_max_iters=200000):
    """
    Nesterov accelerated LP-based alternating optimization.
    Minimizes max_k (a*a)[k] over the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response_fast(y)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Perturb to escape
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=2000000):
    """
    Projected subgradient method for refining the sequence.
    Much faster per iteration than LP, useful for bulk optimization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Initial learning rate
    eta = 0.5 / np.sqrt(n)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active set (peaks within 2% of max)
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient: average of gradients over active peaks
        g = np.zeros(n)
        for k in active:
            # (a*a)[k] = sum_j a[j] * a[k-j]
            # d/da[j] (a*a)[k] = a[k-j] (if valid) + a[k-j] (if k-j valid and j != k-j)
            # More carefully: d/da[j] sum_i a[i]*a[k-i]
            # = a[k-j] (from i=j term: a[j]*a[k-j])
            #   + (if k-j != j and 0 <= k-j <= n-1) a[k-j] (from i=k-j term: a[k-j]*a[j])
            # Wait, I need to be more careful.
            # (a*a)[k] = sum_i a[i] * a[k-i]
            # d/da[j] = a[k-j] (from the term where i=j, derivative of a[j] is 1 times a[k-j])
            #          + sum_{i != j} a[i] * d/da[j] a[k-i]
            # The second term is nonzero only when k-i = j, i.e., i = k-j.
            # So d/da[j] = a[k-j] + a[k-j] (if k-j != j and k-j in [0,n-1])
            #            = a[k-j] (if k-j == j, i.e., k = 2j)
            #            = a[k-j] (if k-j not in [0,n-1])
            # Actually, let me just use: d/da[j] (a*a)[k] = a[k-j] + a[j] * (delta_{k-j, j} correction)
            # No, the correct formula is:
            # If we write (a*a)[k] = sum_i a[i] * a[k-i], then:
            # d/da[j] = sum_i (delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j})
            #         = a[k-j] + sum_i a[i] * delta_{i,k-j}
            #         = a[k-j] + a[k-j] (if k-j in [0,n-1])
            #         = 2*a[k-j] (if 0 <= k-j <= n-1)
            # But wait, if k-j == j (i.e., k = 2j), then we're double-counting:
            # (a*a)[k] contains the term a[j]*a[j] = a[j]^2, whose derivative is 2*a[j].
            # Using the formula: 2*a[k-j] = 2*a[j]. ✓
            # If k-j != j: both the i=j and i=k-j terms contribute, giving 2*a[k-j]. ✓
            # If k-j not in [0,n-1]: only the i=j term contributes (if j in [0,n-1], which it is).
            #   Wait, if k-j < 0 or k-j >= n, then a[k-j] = 0, so the formula gives 0.
            #   But is this correct? If k-j < 0, then in the sum sum_i a[i]*a[k-i],
            #   the term i=j gives a[j]*a[k-j] = 0 (since k-j < 0).
            #   And there's no i such that k-i = j (since k-j < 0 means i = k-j < 0).
            #   So the derivative is 0. ✓
            #   Similarly if k-j >= n.
            # So the formula is simply: d/da[j] (a*a)[k] = 2*a[k-j] (with a[k-j]=0 if out of bounds)
            # Actually wait, that's not right either. Let me redo this.
            # (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
            # d/da[j] (a*a)[k] = (if 0<=j<n and 0<=k-j<n) a[k-j]  [from the term i=j]
            #                  + (if 0<=j<n and 0<=k-j<n) a[k-j]  [from the term i=k-j, only if k-j != j]
            # If k-j == j (k=2j): d/da[j] = a[j] + a[j] = 2*a[j] [the term a[j]*a[j] is counted once in the sum, but its derivative is 2*a[j]]
            # Wait, I'm confusing myself. Let me use a clearer approach.
            #
            # (a*a)[k] = sum_i a[i] * a[k-i] (summing over valid i)
            # This is a function of the vector a. Its gradient w.r.t. a[j] is:
            # For the term i=j: contributes a[k-j] (if k-j is valid)
            # For the term i=k-j: contributes a[k-j] * 1 = a[j]... 
            # NO WAIT. Let me be very explicit.
            #
            # Let f(a) = (a*a)[k] = sum_i a[i] * a[k-i]
            # ∂f/∂a[j] = sum_i [δ(i,j) * a[k-i] + a[i] * δ(k-i, j)]
            #          = a[k-j] (from first term, if k-j valid)
            #          + a[k-j] (from second term, if i=k-j is valid, i.e., k-j in [0,n-1])
            #
            # So ∂f/∂a[j] = a[k-j] * (number of times j appears)
            # If 0 <= k-j <= n-1: j appears in both roles (as i and as k-i), so coefficient is 2 (unless k-j=j, in which case it's still 2 because a[j]*a[j] has derivative 2*a[j]).
            # Wait no. If k-j = j, then the term is a[j]*a[j] = a[j]^2, and d/da[j] = 2*a[j].
            # Using the formula: a[k-j] + a[k-j] = a[j] + a[j] = 2*a[j]. ✓
            #
            # If 0 <= k-j <= n-1 and k-j != j: the two terms a[j]*a[k-j] (i=j) and a[k-j]*a[j] (i=k-j) are separate terms in the sum, each contributing a[k-j] to the derivative. Total: 2*a[k-j].
            #
            # If k-j < 0 or k-j >= n: a[k-j] = 0 (by convention), so derivative is 0.
            #
            # CONCLUSION: ∂(a*a)[k]/∂a[j] = 2 * a[k-j] if 0 <= k-j <= n-1, else 0.
            # This can be written as: 2 * a[k-j] * 1(0 <= k-j <= n-1)
            
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        
        g /= len(active)
        
        # Projected subgradient step
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve >= 100:
            eta *= 0.9
            no_improve = 0
        # Decay step size
        eta = max(0.5 / np.sqrt(step + 1) / np.sqrt(n), 1e-10)
        
        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} eta={eta:.2e}")
    
    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=8, maxiter_per_stage=3000):
    """
    L-BFGS with log-sum-exp smoothing and geometric temperature annealing.
    """
    from scipy.optimize import minimize as sp_minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_obj(T_val):
        def obj(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return obj
    
    def make_grad(T_val):
        def grad(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return grad
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        maxiter = max(20, min(maxiter_per_stage, int(remaining * 50)))
        
        try:
            res = sp_minimize(
                fun=make_obj(stage_T),
                x0=z,
                jac=make_grad(stage_T),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            break
    
    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    else:
        a = np.ones(n)
        a = a / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def active_set_polish(a0, time_limit, n_tol=0.01):
    """
    Polish using active set identification.
    Find the active constraints and solve the KKT system.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Iterate: identify active set, solve QP with only active constraints
    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: peaks within n_tol of max
        active = np.where(b >= max_b * (1 - n_tol))[0]
        
        # Solve: min max_{k in active} (a*a)[k] s.t. a>=0, sum(a)=1
        # Using the LP best response restricted to active set
        n_active_per_iter = 20
        local_best = a.copy()
        local_best_score = best_score
        
        for inner in range(n_active_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            
            # LP: min M s.t. (a_old * b)[k] <= M for k in active, sum(b)=1, b>=0
            # Actually, let's just do the full LP but track the active set
            b_new, M = solve_lp_best_response_fast(a)
            if b_new is None:
                break
            
            # Nesterov-like update
            t = outer * n_active_per_iter + inner
            alpha = 2.0 / (t + 2)
            a = (1 - alpha) * a + alpha * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            sc = 2.0 * n * fast_convolve(a).max()
            if sc < local_best_score:
                local_best_score = sc
                local_best = a.copy()
        
        if local_best_score < best_score:
            best_score = local_best_score
            best_a = local_best.copy()
            a = local_best.copy()
        else:
            break
    
    return best_a, best_score
