import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve_max(a):
    """Compute max of auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    return conv.max()

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based convolutions, projected gradient descent on the simplex,
LP-based direction finding, coordinate descent, and multi-length exploration.
Key insight: for non-negative sequences, max(conv(a,a)) = sum(a[i]^2), so the objective
is 2*n*sum(a^2)/sum(a)^2 >= 2 by Cauchy-Schwarz. We verify empirically and optimize
to find the best achievable score.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_autoconv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_autoconv_max(a):
    """Max of auto-convolution via FFT."""
    return fast_autoconv(a).max()


def objective_val(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * fast_autoconv_max(a) / (s * s)


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for LP: row i gives the convolution at lag indices[i]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M):
    """Solve LP: max sum(g) s.t. g>=0, conv constraints. Returns g or None."""
    n = len(f)
    K = len(indices)
    if K == 0 or n == 0:
        return None
    A = build_constraint_matrix(f, indices, n)

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    try:
        if K <= 200 and n <= 3000:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-7, eps_rel=1e-7)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def gradient_descent_simplex(f_init, time_limit, lr_init=0.05, momentum=0.9, n_top=100):
    """
    Projected gradient descent on the simplex with softmax-weighted subgradient.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:

    v = np.zeros(n)
    lr = lr_init
    temp = 0.01
    best_f = f.copy()
    best_obj = fast_autoconv_max(f)
    no_improve = 0
    it = 0
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autoconv(f)
        M = b.max()

        # Softmax weights on peaks
        if full_len <= 2000:
            scaled = (b - M) / temp
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(full_len)
                w[int(np.argmax(b))] = 1.0
            else:
                w = w / ws
        else:
            # Use top-k only
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            sub_b = b[top_idx]
            scaled = (sub_b - sub_b.max()) / temp
            w_sel = np.exp(scaled)
            ws = w_sel.sum()
            if ws < 1e-30:
                w = np.zeros(full_len)
                w[int(np.argmax(b))] = 1.0
            else:
                w = np.zeros(full_len)
                w[top_idx] = w_sel / ws

        # Subgradient: g[j] = sum_k w[k] * f[k-j] (for valid k-j)
        grad = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[j] += w[k] * f[k - j] for j in [lo, hi]
            # f[k-j] for j in [lo, hi] means indices [k-hi, k-lo] of f
            f_slice = f[k - hi:k - lo + 1]
            grad[lo:hi + 1] += w[k] * f_slice[np.arange(hi - lo + 1)][::-1]

        f = project_simplex(f + v)

def lp_based_optimize(f_init, time_limit, top_k_init=50, top_k_max=400):
    """LP-based direction finding with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:

    it = 0
    stuck = 0
    top_k = top_k_init
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)

        b = fast_autoconv(f)
        M = b.max()

        # Select peak indices

        else:
            indices = np.argpartition(b, -top_k)[-top_k:]

        # Solve LP
        g = solve_lp_direction(f, indices, M)

        if g is None or g.sum() <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 50:
                break
            continue

        h = g / g.sum()

        # Line search
        best_t = 0.0
        best_trial_obj = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            trial = (1.0 - t) * f + t * h
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_autoconv_max(trial)
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        cur_obj = fast_autoconv_max(f)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break

        if it % 50 == 0:
            print(f"    LP it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} k={len(indices)} t={time.monotonic()-t0:.0f}s")

    return best_f, 2.0 * n * best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent: try scaling each coordinate by various multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_obj = fast_autoconv_max(f)
    best_f = f.copy()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_obj = best_obj
            best_j_val = f[j]
            for mult in multipliers:
                trial = f.copy()
                trial[j] = f[j] * mult
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial = trial / ts
                obj = fast_autoconv_max(trial)
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0

    return best_f, 2.0 * n * best_obj

    # Phase 0: Try uniform sequences (theoretical minimum = 2.0)
    print("\n=== Phase 0: Uniform baselines ===")
    for n_try in [128, 256, 512, 1024, 2048, 4096]:
        u = np.ones(n_try) / n_try
        update_best(u, f"uniform n={n_try}")

    # If parent is better, use it
    if parent is not None and p_score < best_score:
        best_score = p_score
        best_seq = parent.copy()
        print(f"  Parent is best: score={best_score:.6f}, n={len(parent)}")

    print(f"  Current best: {best_score:.6f}")

    # If we're at the theoretical minimum, we're done
    if best_score <= 2.0 + 1e-9:
        print("  At theoretical minimum (2.0).")
        return [float(x) for x in best_seq]

    # Phase 1: Try structured sequences
    print("\n=== Phase 1: Structured sequences ===")
    structured_starts = []
    n_base = len(best_seq) if best_seq is not None else 256

    # Sine-based
    for freq in [1, 2, 3, 5]:
        x = np.linspace(0, 2 * np.pi * freq, n_base)
        a = np.maximum(np.cos(x), 0)
        a = a / a.sum()
        structured_starts.append(a)

    # Gaussian
    x = np.linspace(-3, 3, n_base)
    a = np.exp(-x**2 / 4)
    a = a / a.sum()
    structured_starts.append(a)

    # Triangle
    x = np.linspace(0, 1, n_base)
    a = 1 - 2 * np.abs(x - 0.5)
    a = a / a.sum()
    structured_starts.append(a)

    # Random
    a = rng.random(n_base)
    a = a / a.sum()
    structured_starts.append(a)

    for i, a in enumerate(structured_starts):
        update_best(a, f"structured[{i}] n={n_base}")

    # Phase 2: Gradient descent on best so far
    print(f"\n=== Phase 2: Gradient descent (best: {best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 400.0)
    if t2 > 20:
        n_b = len(best_seq)
        print(f"  GD n={n_b}, budget={t2:.0f}s")
        f_res, _ = gradient_descent_simplex(best_seq, time_limit=t2,
                                             lr_init=0.05, momentum=0.9,
                                             n_top=max(50, n_b // 4))
        if not update_best(f_res, "gd_best"):
            pass

        # Also try GD on a different length
        for n_try in [n_b // 2, n_b * 2]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.25, 150.0)
            if t_sub < 15:
                break
            f_init = interp_sequence(best_seq, n_try)
            print(f"  GD n={n_try}, budget={t_sub:.0f}s")
            f_res, _ = gradient_descent_simplex(f_init, time_limit=t_sub,
                                                 lr_init=0.04, momentum=0.85,
                                                 n_top=max(40, n_try // 4))
            update_best(f_res, f"gd n={n_try}")

    # Phase 3: LP-based optimization
    print(f"\n=== Phase 3: LP optimization (best: {best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 400.0)
    if t3 > 20:
        n_b = len(best_seq)
        # Try LP on current best
        t_sub = min(t3 * 0.5, rem * 0.4)
        if t_sub > 15:
            print(f"  LP n={n_b}, budget={t_sub:.0f}s")
            f_res, _ = lp_based_optimize(best_seq, time_limit=t_sub,
                                         top_k_init=30, top_k_max=min(200, 2 * n_b - 1))
            update_best(f_res, f"lp n={n_b}")

        # Try LP on different lengths
        for factor in [0.5, 1.5, 2.0]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            n_try = max(64, int(n_b * factor))
            n_try = min(n_try, 5000)
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 15:
                break
            f_init = interp_sequence(best_seq, n_try)
            print(f"  LP n={n_try}, budget={t_sub:.0f}s")
            f_res, _ = lp_based_optimize(f_init, time_limit=t_sub,
                                         top_k_init=30, top_k_max=min(150, 2 * n_try - 1))
            update_best(f_res, f"lp n={n_try}")

    # Phase 4: Deep optimization on current best
    print(f"\n=== Phase 4: Deep optimization (best: {best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 250.0)
    if t4 > 15:
        n_b = len(best_seq)
        # Alternate between GD and LP
        f_cur = best_seq.copy()
        t_sub = t4 / 3
        for method in ['gd', 'lp', 'gd']:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_sub = min(t_sub, rem * 0.5)
            if t_sub < 10:
                break
            if method == 'gd':
                print(f"  Deep GD n={n_b}, budget={t_sub:.0f}s")
                f_cur, _ = gradient_descent_simplex(f_cur, time_limit=t_sub,
                                                     lr_init=0.03, momentum=0.9,
                                                     n_top=max(50, n_b // 4))
            else:
                print(f"  Deep LP n={n_b}, budget={t_sub:.0f}s")
                f_cur, _ = lp_based_optimize(f_cur, time_limit=t_sub,
                                             top_k_init=50, top_k_max=min(300, 2 * n_b - 1))
            update_best(f_cur, f"deep_{method}")

    # Phase 5: Coordinate descent polish
    print(f"\n=== Phase 5: CD polish (best: {best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 150.0)
    if t5 > 10:
        n_b = len(best_seq)
        if n_b <= 2000:
            print(f"  CD n={n_b}, budget={t5:.0f}s")
            f_res, _ = coordinate_descent(best_seq, time_limit=t5)
            update_best(f_res, "cd")
        else:
            print(f"  Skipping CD (n={n_b} too large)")

    # Phase 6: Final burst
    print(f"\n=== Phase 6: Final burst (best: {best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        n_b = len(best_seq)
        t6 = rem * 0.7
        f_res, _ = gradient_descent_simplex(best_seq, time_limit=t6,
                                             lr_init=0.02, momentum=0.95,
                                             n_top=max(80, n_b // 5))
        update_best(f_res, "final_gd")

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
    else:
        best_seq = np.ones(256) / 256.0
        best_score = 2.0

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
