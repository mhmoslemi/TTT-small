def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                else:
                    # Perturb and continue
                    f = f + rng.standard_normal(n) * 0.0001
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    stuck = 0
                    continue
            else:
                # Try more peaks
                n_try = min(n_active * 2, max_active)
                if n_try > n_active:
                    active2 = np.argpartition(b, -n_try)[-n_try:]
                    g2 = solve_direction(f, active2, M)
                    if g2 is not None and g2.sum() > 1.0 + 1e-8:
                        g = g2
                        S_g = g2.sum()
                        stuck = 0
                    else:
                        continue
                else:
                    continue

        # Line search
        h = g / S_g
        best_t, best_obj = line_search(f, h, M)

        if best_t <= 0:
            stuck += 1
            if stuck > 200:
                break
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-12:
            f /= s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-15:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 200 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                  f"active={len(active)} t={time.monotonic()-t0:.1f}s")

    return best_f, best_b


def subgradient_polish(f_init, time_limit, label="sg"):
    """Subgradient method with exponential tilt for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = 0.02
    temp = 0.001
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on peaks
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Focus on top peaks
        n_top = min(80, 2 * n - 1)
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_sel

        # Compute subgradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        grad = np.zeros(n)
        for k in top_idx:
            wk = w[k]
            if wk < 1e-12:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            grad[js] += 2.0 * wk * f[k - js]

        # Mirror descent update
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.05)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.95
                    no_improve = 0
                if no_improve > 20:
                    temp = min(temp * 1.1, 0.05)
                if no_improve > 50:
                    f = best_f + rng.standard_normal(n) * 0.001
                    f = np.maximum(f, 1e-15)
                    f /= f.sum()
                    lr = 0.02
                    no_improve = 0

            if it % 500 == 0:
                score = 2.0 * n * best_b
                print(f"  {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} t={time.monotonic()-t0:.1f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent: try scaling each element by various factors."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3, 1.5, 2.0]

    for epoch in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if abs(best_j_val - f[j]) > 1e-15:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
