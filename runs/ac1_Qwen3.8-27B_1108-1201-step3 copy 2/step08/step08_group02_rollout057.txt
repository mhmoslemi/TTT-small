Multi-phase optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Combines smooth-max Nesterov descent, LP direction finding, and coordinate descent.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(f, T, n_top=150):
    n = len(f)
    b = fast_convolve(f)
    T_b = T * b
    T_b -= T_b.max()
    w = np.exp(T_b)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), b.max()
    w /= w_sum
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sub = w[top_idx].copy()
        ws = w_sub.sum()
        if ws > 1e-30:
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sub / ws
            w = w_new
    grad = np.zeros(n)
    sig = w > 1e-12
    idx = np.where(sig)[0]
    for k in idx:
        wk = w[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        grad[lo:hi+1] += wk * f[k - j]
    return grad, b.max()

def solve_lp_dir(f, n, peaks, M, eps=1e-8):
    K = len(peaks)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[hi:lo-1:-1]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=200000, eps=eps)
            if g.value is not None:
                gv = np.maximum(g.value, 0)
                if not np.any(np.isnan(gv)):

    return None

"""
Multi-phase optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Combines smooth-max Nesterov accelerated gradient descent, LP-based direction finding,
and coordinate descent with multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(f, T, n_top=150):
    """Compute gradient of smooth max approximation of max(conv(f,f)) at f."""
    n = len(f)
    b = fast_convolve(f)
    T_b = T * b
    T_b -= T_b.max()
    w = np.exp(T_b)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), b.max()
    w /= w_sum
    # Focus on top peaks for efficiency
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sub = w[top_idx].copy()
        ws = w_sub.sum()
        if ws > 1e-30:
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sub / ws
            w = w_new
    grad = np.zeros(n)
    sig = w > 1e-12
    idx = np.where(sig)[0]
    for k in idx:
        wk = w[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1)
        grad[lo:hi + 1] += wk * f[k - j]
    return grad, b.max()


def nesterov_descent(f_init, time_limit, label="nd", lr_init=0.005, T_init=80.0, momentum=0.9):
    """Nesterov accelerated gradient descent on smooth-max approximation."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    T = T_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        a_y = a + momentum * v
        a_y = project_simplex(a_y)
        grad, _ = smooth_max_grad(a_y, T)
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        v = a_new - a
        a = a_new

        if it % 15 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 12:
                    lr *= 0.88
                    T = max(T * 0.92, 20.0)
                    no_improve = 0
                if no_improve > 60:
                    a = best_a + rng.standard_normal(n) * 0.003 * best_a
                    a = project_simplex(a)
                    v = np.zeros(n)
                    lr = lr_init
                    T = T_init
                    no_improve = 0

        if it % 500 == 0:
            score = 2.0 * n * best_obj
            print(f"  {label} it={it} obj={best_obj:.10f} score={score:.6f} lr={lr:.6f} T={T:.1f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def solve_lp_dir(f, n, peaks, M, eps=1e-8):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in peaks."""
    K = len(peaks)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[hi:lo - 1:-1]
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=300000, eps=eps)
            if g.value is not None:
                gv = np.maximum(g.value, 0)
                if not np.any(np.isnan(gv)) and not np.any(np.isinf(gv)):
                    return gv
        except Exception:
            continue
    return None


def lp_optimize(f_init, time_limit, label="lp", max_peaks=250, patience=15,
                peak_frac_init=0.93, peak_frac_min=0.80):
    """LP-based direction finding with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_init
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        thresh = peak_frac * M
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -max(10, len(b) // 20))[-max(10, len(b) // 20):]

        g = solve_lp_dir(f, n, peaks, M)
        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > patience:
                break
            continue

        h = g / g.sum()
        # Line search
        best_t = 0.0
        best_obj = M
        for t in [0.05, 0.10, 0.15, 0.20, 0.30, 0.50]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
        # Refine
        if best_t > 0:
            for t in [best_t * 0.7, best_t * 1.3]:
                if t <= 0 or t > 0.9:
                    continue
                f_trial = (1 - t) * f + t * h
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj:
                    best_obj = obj
                    best_t = t

        if best_t <= 0:
            stuck += 1
            if stuck > patience:
                break
            continue

        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        f /= f.sum()
        cur = fast_convolve(f).max()
        if cur < best_b - 1e-14:
            best_b = cur
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 10 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} it={it} obj={best_b:.10f} score={score:.6f} peaks={len(peaks)} t={time.monotonic()-t0:.0f}s")

        if it % 20 == 0:
            peak_frac = max(peak_frac_min, peak_frac - 0.003)

        if stuck > patience // 2 and stuck % 5 == 0:
            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0)
            f /= f.sum()

    return best_f, best_b


def coord_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.7, 0.85, 0.93, 1.07, 1.15, 1.3, 1.5]
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        sweep += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] *= mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                f /= f.sum()
                best_b = best_obj
                improved = True
        if not improved:
            break
        if sweep % 3 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} sweep={sweep} obj={best_b:.10f} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return f, best_b


def interpolate(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def gaussian_init(n, sigma_frac=0.4):
    """Gaussian window initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a /= a.sum()
    return a
