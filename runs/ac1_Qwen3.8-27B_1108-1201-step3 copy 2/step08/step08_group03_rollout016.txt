"""
Optimized step-function sequence search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP-based direction finding with exact quadratic line search,
adaptive cutting planes, multi-phase strategy, and gradient-based polishing.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix for active convolution indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, solver_pref=None):
    """Solve LP: max sum(g) s.t. g>=0, A@g <= M. Returns g or None."""
    K = len(active_idx)
    if K == 0 or K > n * 3:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solvers_to_try = []
    if solver_pref:
        solvers_to_try.append(solver_pref)
    if n <= 1500:
        solvers_to_try.extend(['ECOS', 'PDP', 'SCS'])
    else:
        solvers_to_try.extend(['PDP', 'SCS', 'ECOS'])
    
    # Deduplicate
    seen = set()
    unique_solvers = []
    for s in solvers_to_try:
        if s not in seen:
            seen.add(s)
            unique_solvers.append(s)
    
    g_val = None
    for solver in unique_solvers[:3]:
        try:
            if solver == 'ECOS':
                prob.solve(solver=cp.ECOS, max_iters=50000, abstol=1e-9, reltol=1e-9)
            elif solver == 'PDP':
                prob.solve(solver=cp.PDP, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
            elif solver == 'SCS':
                prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
            else:
                prob.solve(solver=solver)
        except Exception:

            g_val = np.maximum(g.value, 0.0)
            if not np.any(np.isnan(g_val)):

def exact_line_search(f, d, b_ff, b_fd, b_dd, t_max, n_scan=30, n_golden=12):
    """
    Exact line search: minimize max_k (c_k + 2t*p_k + t^2*q_k) over t in [0, t_max].
    Uses coarse scan + golden section refinement.
    """
    if t_max <= 1e-12:
        return 0.0
    
    def obj_at_t(t):
        vals = b_ff + 2.0 * t * b_fd + t * t * b_dd
        return vals.max()
    
    # Coarse scan
    ts = np.linspace(0.0, t_max, n_scan)
    objs = np.array([obj_at_t(t) for t in ts])
    best_idx = np.argmin(objs)
    best_t = ts[best_idx]
    best_obj = objs[best_idx]
    
    # Golden section refinement around best region
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_scan - 1, best_idx + 1)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d_pt)
    
    for _ in range(n_golden):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_t(d_pt)
    
    t_mid = (lo + hi) / 2.0
    obj_mid = obj_at_t(t_mid)
    if obj_mid < best_obj:
        best_t = t_mid
        best_obj = obj_mid
    
    return best_t

        cur_M = fast_convolve(f_new).max()
        
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            f_new = f + t_opt * 0.5 * d
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s > 1e-10:
                f_new /= s
                cur_M = fast_convolve(f_new).max()
                if cur_M < best_M - 1e-14:
                    best_M = cur_M
                    best_f = f_new.copy()
                    no_improve = 0
        
        f = best_f.copy() if no_improve > 5 else f_new
        b = fast_convolve(f)
        M = b.max()
        
        # Periodic constraint refresh
        if it % 20 == 0:
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep_n = min(initial_k * 2, len(active_idx))
                keep = active_idx[np.argsort(active_scores)[-keep_n:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
        
        if it % 30 == 0:
            score = 2.0 * n * M
            print(f"    {label} it={it} M={M:.10f} score={score:.6f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_M


def gradient_polish(f_init, time_limit, K=30, temp_init=0.0005, seed=42):
    """Gradient-based polishing with exact line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    best_M = b.max()
    temp = temp_init
    rng = np.random.default_rng(seed)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Softmax weights on peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Focus on top-K
        if K < len(b):
            top_idx = np.argpartition(b, -K)[-K:]
            w_other = w.sum() - w[top_idx].sum()
            if w[top_idx].sum() > 1e-15:
                w[top_idx] = w[top_idx] / w[top_idx].sum()
            else:
                w[top_idx] = np.ones(K) / K
            w = np.zeros(len(b))
            w[top_idx] = w[top_idx]
        
        # Compute gradient: grad_k[m] = 2*a[k-m] (if valid)
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            # conv(a,a)[k] = sum_j a[j]*a[k-j], grad w.r.t. a[m] = 2*a[k-m] (if 0<=k-m<n)
            km = k - np.arange(n)
            valid = (km >= 0) & (km < n)
            grad[valid] += w[k] * 2.0 * f[km[valid]]
        
        # Project to zero-sum
        grad -= grad.mean()
        d = -grad
        d -= d.mean()
        
        # Scale direction
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Line search with exact quadratic
        b_fd = convolve_cross(f, d)
        b_dd = fast_convolve(d)
        
        neg_mask = d < 0
        if neg_mask.any():
            t_max = (f[neg_mask] / (-d[neg_mask])).min()
        else:
            t_max = 1.0
        t_max = min(t_max * 0.9, 1.0)
        
        # Try a few step sizes (since exact line search might be expensive for long sequences)
        best_t = 0.0
        best_obj = M
        for t_try in [t_max * 0.1, t_max * 0.3, t_max * 0.5, t_max * 0.7, t_max * 0.9]:
            f_trial = f + t_try * d
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t_try
        
        if best_t < 1e-8:
            no_improve += 1
            if no_improve > 15:
                temp *= 1.3
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue
        
        f_new = f + best_t * d
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        
        cur_M = fast_convolve(f_new).max()
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f_new.copy()
            no_improve = 0
            temp = max(temp * 0.98, 1e-6)
        else:
            f = f_new  # Accept anyway (might help later)
            no_improve += 1
        
        f = f_new
        if it % 50 == 0:
            score = 2.0 * n * best_M
            print(f"    grad it={it} best_M={best_M:.10f} score={score:.6f} "
                  f"temp={temp:.2e} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_M


def diversify(f_init, rng, strength=0.01):
    """Create a diversified version of the sequence."""
    n = len(f_init)
    f = f_init.copy()
    # Random perturbation
    noise = rng.standard_normal(n) * strength
    f = f + noise
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = f_init.copy()
    return f


def iaa_diversify(f_init, n_iters=100, peak_fraction=0.3):
    """IAA-based spectral shaping."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for _ in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        peak_val = R.max()
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()
        
        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break
    
    return f


def coordinate_polish(f_init, time_limit):
    """Coordinate descent for final polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    step = max(1, n // 500)
    
    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        any_improve = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return best_f
            if f[j] < 1e-15:
                continue
            orig = f[j]
            best_val = orig
            best_obj_j = best_M
            for mult in [0.8, 0.9, 1.1, 1.2]:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = orig * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_M - 1e-14:
                best_M = best_obj_j
                best_f = f.copy()
                any_improve = True
        if not any_improve:
            break
    return best_f
