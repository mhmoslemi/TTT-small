Multi-track optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, Adam-based smooth optimization with log-sum-exp,
palindromic constraint exploitation, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]) * target_sum, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, w):
    """
    Compute gradient of sum_k w[k] * conv(f,f)[k] w.r.t. f.
    w has length 2n-1, f has length n.
    Returns array of length n.
    """
    # grad[i] = 2 * sum_{k=0}^{n-1} w[i+k] * f[k]
    # This is the valid cross-correlation of w and f
    return 2.0 * correlate(w, f, mode='valid')

        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if K <= 1000 and n <= 2000:
            prob.solve(solver=cp.PDLP, max_iter=50000)
        else:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps_abs=1e-7, eps_rel=1e-7)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
            except Exception:
                return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    if g_val.sum() <= 0:
        return None
    return g_val


def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=500,
                momentum_beta=0.3, max_stuck=40, label="lp"):
    """LP-based optimization with momentum and backtracking."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    top_k = top_k_init
    momentum_dir = np.zeros(n)
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 15 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        full_len = 2 * n - 1
        
        # Peak selection
        if top_k >= full_len:
            indices = np.arange(full_len)
        else:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            # Add some boundary and center peaks
            extra = set()
            for di in range(-3, 4):
                c = (full_len - 1) // 2 + di
                if 0 <= c < full_len:
                    extra.add(c)
            for di in range(0, 5):
                if di < full_len:
                    extra.add(di)
                if full_len - 1 - di >= 0:
                    extra.add(full_len - 1 - di)
            all_cands = np.unique(np.concatenate([top_idx, np.array(list(extra))]))
            all_cands = all_cands[np.argsort(b[all_cands])[::-1]][:top_k]
            indices = np.sort(all_cands)
        
        # Solve LP
        g = solve_lp_direction(f, indices, M)
        
        if g is None:
            if len(indices) > 10:
                top_10 = np.sort(np.argpartition(b, -10)[-10:])
                g = solve_lp_direction(f, top_10, M)
            if g is None:
                stuck += 2
                if stuck > max_stuck:
                    break
                continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > max_stuck // 2:
                break
            top_k = min(top_k * 2, max(top_k_max, full_len))
            continue
        
        h = g / S_g
        
        # Momentum
        momentum_dir = momentum_beta * momentum_dir + (1.0 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h
        
        # Line search
        best_obj_trial = M
        best_f_trial = f.copy()
        best_t = 0.0
        
        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_trial - 1e-14:
                    best_obj_trial = obj
                    best_t = t
                    best_f_trial = f_trial.copy()
        
        if best_t == 0.0:
            stuck += 1
            if stuck > max_stuck:
                break
            top_k = min(top_k * 2, max(top_k_max, full_len))
            if stuck > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                momentum_dir = np.zeros(n)
                stuck = 0
            continue
        
        f = best_f_trial
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break
        
        if it % 20 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={len(indices)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def adam_smooth_optimize(f_init, time_limit, tau_start=0.02, tau_end=0.002,
                         lr=0.005, restart_every=150, label="adam"):
    """
    Adam optimization on simplex with log-sum-exp smoothed max objective.
    Minimizes sum_k exp(b_k/tau) which upper-bounds max(b_k).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_start
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    rng = np.random.default_rng(999)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        tau = max(tau_end, tau * 0.998)
        
        b = fast_convolve(f)
        M = b.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_score = 2.0 * n * M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute softmax weights
        scaled = (b - M) / tau
        scaled = np.clip(scaled, -100, 0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Gradient
        grad = compute_smooth_grad(f, w)
        
        # Project gradient to tangent space of simplex (subtract mean of active coords)
        active = f > 1e-12
        if active.any():
            grad_projected = grad.copy()
            grad_projected[active] -= np.mean(grad[active])
            # Zero out gradient for inactive coords (don't push them negative)
            grad_projected[~active] = np.maximum(grad[~active], 0)
        else:
            grad_projected = grad - np.mean(grad)
        
        # Adam update
        m = 0.9 * m + 0.1 * grad_projected
        v = 0.999 * v + 0.001 * grad_projected ** 2
        m_hat = m / (1 - 0.9 ** (t + 1))
        v_hat = v / (1 - 0.999 ** (t + 1))
        
        # Adaptive learning rate
        lr_eff = lr / (1 + 0.1 * no_improve)
        
        f_new = f - lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new, target_sum=1.0)
        f = f_new
        
        # Restart if stuck
        if no_improve > restart_every:
            # Try different perturbation scales
            scale = 0.005 + 0.01 * (no_improve % 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale, target_sum=1.0)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            tau = min(tau * 2, tau_start)
        
        if t % 100 == 0:
            obj_now = fast_convolve(f).max()
            print(f"    {label} t={t} best={best_obj:.12f} cur={obj_now:.12f} "
                  f"tau={tau:.4f} score={best_score:.6f} "
                  f"no_imp={no_improve} elapsed={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def palindromic_optimize(f_init, time_limit, label="palin"):
    """
    Optimizes a palindromic sequence. Only optimizes the first half.
    f[i] = f[n-1-i] enforced.
    """
    t0 = time.monotonic()
    n = len(f_init)
    
    # Ensure palindromic
    f = (f_init + f_init[::-1]) / 2.0
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    half_n = (n + 1) // 2
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    # Use Adam on the half sequence
    h = f[:half_n].copy()
    m = np.zeros(half_n)
    v = np.zeros(half_n)
    t = 0
    tau = 0.02
    lr = 0.004
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        tau = max(0.003, tau * 0.998)
        
        # Reconstruct full sequence from half
        f = np.zeros(n)
        f[:half_n] = h
        f[half_n:] = h[:n - half_n][::-1]
        s = f.sum()
        if s < 1e-15:
            break
        f = f / s
        h = f[:half_n].copy()
        
        b = fast_convolve(f)
        M = b.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_score = 2.0 * n * M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # For palindromic f, only need to check k from 0 to n-1 (by symmetry)
        b_sym = b[:n]  # first half (including center)
        M_sym = b_sym.max()
        
        scaled = (b_sym - M_sym) / tau
        scaled = np.clip(scaled, -100, 0)
        w_sym = np.exp(scaled)
        w_sum = w_sym.sum()
        if w_sum < 1e-30:
            w_sym = np.zeros(n)
            w_sym[np.argmax(b_sym)] = 1.0
        else:
            w_sym = w_sym / w_sum
        
        # Full weight vector (palindromic)
        w = np.zeros(2 * n - 1)
        w[:n] = w_sym
        w[n:] = w_sym[:2*n-1-n][::-1]
        
        # Full gradient
        grad_full = compute_smooth_grad(f, w)
        # Only need half
        grad_half = grad_full[:half_n]
        
        # Project to tangent space
        active = h > 1e-12
        if active.any():
            grad_proj = grad_half.copy()
            grad_proj[active] -= np.mean(grad_half[active])
            grad_proj[~active] = np.maximum(grad_half[~active], 0)
        else:
            grad_proj = grad_half - np.mean(grad_half)
        
        # Adam
        m = 0.9 * m + 0.1 * grad_proj
        v = 0.999 * v + 0.001 * grad_proj ** 2
        m_hat = m / (1 - 0.9 ** (t + 1))
        v_hat = v / (1 - 0.999 ** (t + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = np.maximum(h_new, 0)
        hs = h_new.sum()
        if hs > 1e-15:
            h_new = h_new / hs
        else:
            h_new = np.ones(half_n) / half_n
        h = h_new
        
        if no_improve > 100:
            scale = 0.005 + 0.005 * (no_improve % 4)
            best_half = best_f[:half_n].copy()
            h = np.maximum(best_half + rng.standard_normal(half_n) * scale, 0)
            hs = h.sum()
            if hs > 1e-15:
                h = h / hs
            m = np.zeros(half_n)
            v = np.zeros(half_n)
            no_improve = 0
            tau = min(tau * 1.5, 0.05)
        
        if t % 100 == 0:
            print(f"    {label} t={t} best={best_obj:.12f} score={best_score:.6f} "
                  f"no_imp={no_improve} elapsed={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Simple coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.9, 1.0, 1.1, 1.15, 1.25, 1.4, 1.6, 2.0, 3.0]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a normalized sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final CD]: {best_score:.6f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
