def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v, target_sum=1.0):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    v *= target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_best_response_linalg(a_fixed):
    """
    Solve the best response LP using scipy linprog (HiGHS):
    minimize M
    subject to: cross_conv(a, b)[k] <= M for k=0,...,2n-2
                sum(b) = 1
                b >= 0
    """
    from scipy.optimize import linprog
    n = len(a_fixed)
    m = 2 * n - 1
    
    # Variables: [M, b_0, b_1, ..., b_{n-1}]
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    # Inequality constraints: cross_conv(a, b)[k] - M <= 0 for each k
    # cross_conv(a, b)[k] = sum_j a[k-j] * b[j] (where valid)
    # So: -M + sum_j a[k-j]*b[j] <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0  # -M
    for k in range(m):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j + 1] = a_fixed[idx]
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: M free, b >= 0
    bounds = [(None, None)] + [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    b_val = result.x[1:]
    M_val = result.x[0]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val

def smooth_subgradient(a0, time_limit, verbose=False, label=""):
    """
    Smooth subgradient method using log-sum-exp approximation.
    Uses adaptive beta and Nesterov-like acceleration.
    """

        a = a / s

    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    
    # Nesterov parameter
    t_param = 1.0
    lr = 0.1
    
    max_iters = 8000
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        t_param_next = (1.0 + math.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        momentum = (t_param - 1.0) / t_param_next
        
        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute smooth approximation gradient
        conv_y = fast_convolve(y)
        # Adaptive beta: start moderate, increase over time
        beta = 5.0 + 80.0 * (it / max_iters) ** 0.5
        
        # Softmax weights (numerically stable)
        cv_shift = conv_y - conv_y.max()
        exp_vals = np.exp(beta * cv_shift)
        w = exp_vals / exp_vals.sum()
        
        # Gradient computation: grad[i] = 2 * sum_k w[k] * a[k-i]
        grad = np.zeros(n)
        # Only process significant weights
        thresh = w.max() * 1e-5

            wk = w[k]

                continue
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += 2.0 * wk * y[k - idx]

        a_prev = a
        a = a_new
        t_param = t_param_next

def init_periodic(n, rng):
    """Initialize with a periodic pattern (low autocorrelation-like)."""
    period = int(rng.integers(max(8, n // 32), n // 4 + 1))
    pattern = rng.random(period)
    a = np.tile(pattern, n // period + 1)[:n]
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def peak_perturbation(a0, rng, magnitude=0.04):
    """Perturb coordinates contributing to convolution peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(8, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization + quick coordinate descent
    Phase 2: LP fictitious play (main workhorse)
    Phase 3: Smooth subgradient refinement
    Phase 4: Fine coordinate descent polishing
    Phase 5: Final intensive optimization
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    # === Phase 1: Multi-length diverse initialization ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()
    
    n_values = [768, 896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    
    candidates = []
    inits_per_n = 3
    
    for n in n_values[:4]:
        if time.monotonic() > deadline - 1200:
            break
        
        init_fns = [init_fourier, init_cosine, init_u_shape, init_periodic]
        for c in range(min(inits_per_n, len(init_fns))):
            itype = init_fns[c]
            cand = itype(n, rng)
            
            # Quick CD to get a reasonable starting point
            t_alloc = 5.0
            a_opt, s_opt, _ = fast_coord_descent(cand, t_alloc, n_trials=8, label=f"p1_n{n}_{c}")
            candidates.append((s_opt, a_opt, n, c))
    
    # Also include parent-based candidates
    if parent is not None:
        for alt_n in [768, 1024, 1280]:
            if time.monotonic() > deadline - 1200:
                break
            cand = interpolate_seq(parent, alt_n)
            t_alloc = 5.0
            a_opt, s_opt, _ = fast_coord_descent(cand, t_alloc, n_trials=8, label=f"p1_parent_{alt_n}")
            candidates.append((s_opt, a_opt, alt_n, 'parent'))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best: {candidates[0][0]:.6f}")
    
    for s_c, a_c, n_c, it_c in candidates[:4]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")
    
    # === Phase 2: LP Fictitious Play (main workhorse) ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    t2_budget = 600.0
    t2_start = time.monotonic()
    
    n_best = len(best_seq) if best_seq is not None else 1024
    
    # Run FP at the best length and one alternative
    fp_lengths = [n_best]
    for alt in [896, 1152]:
        if alt != n_best and 256 <= alt <= 2048:
            fp_lengths.append(alt)
    
    for n_fp in fp_lengths[:2]:
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break
        t_alloc = min(rem * 0.5, t2_budget / max(len(fp_lengths), 1))
        print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")
        
        initial = interpolate_seq(best_seq, n_fp) if best_seq is not None else None
        a_fp, score_fp, iters = fictitious_play_refined(
            n_fp, t_alloc, initial=initial, verbose=True, label=f"fp{n_fp}"
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")
        
        # Quick CD polish after FP
        rem = deadline - time.monotonic() - 400
        if rem > 30 and best_seq is not None:
            a_cd, s_cd, _ = fast_coord_descent(best_seq, min(rem * 0.3, 60.0), 
                                               n_trials=10, label=f"fp_cd_{n_fp}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"fp_cd_{n_fp}")
    
    # === Phase 3: Smooth Subgradient Refinement ===
    print(f"\n=== Phase 3: Smooth Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_sg = min(rem * 0.5, 200.0)
            print(f"  SmoothSG n={len(best_seq)}, budget={t_sg:.0f}s")
            a_sg, s_sg = smooth_subgradient(best_seq, t_sg, verbose=True, label="sg")
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "smooth_sg")
            
            # FP on the smooth SG result
            rem = deadline - time.monotonic() - 200
            if rem > 60:
                t_fp2 = min(rem * 0.4, 120.0)
                n_sg = len(best_seq)
                a_fp2, s_fp2, _ = fictitious_play_refined(
                    n_sg, t_fp2, initial=best_seq, verbose=True, label=f"fp2_{n_sg}"
                )
                best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, "fp2")
    
    # === Phase 4: Intensive Coordinate Descent ===
    print(f"\n=== Phase 4: Intensive CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            t_cd = min(rem * 0.6, 200.0)
            print(f"  FCD n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, s_cd, sweeps = fast_coord_descent(best_seq, t_cd, n_trials=12,
                                                     verbose=True, label="intensive")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "intensive_cd")
    
    # === Phase 5: Perturbation + Restart ===
    print(f"\n=== Phase 5: Perturbation + Restart (best={best_score:.6f}) ===")
    if best_seq is not None and best_score > 1.503:
        rem = deadline - time.monotonic() - 80
        if rem > 60:
            for attempt in range(3):
                rem = deadline - time.monotonic() - 40
                if rem < 30:
                    break
                perturbed = peak_perturbation(best_seq, rng, magnitude=0.03 + 0.02 * attempt)
                t_alloc = min(rem * 0.3, 80.0)
                
                # Quick CD on perturbed
                a_p, s_p, _ = fast_coord_descent(perturbed, t_alloc, n_trials=10, 
                                                  label=f"perturb_{attempt}")
                best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"perturb_{attempt}")
    
    # === Phase 6: Fine Polishing ===
    print(f"\n=== Phase 6: Fine Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_fine = min(rem * 0.6, 100.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, label="fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")
        
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_last = min(rem * 0.7, 50.0)
            a_last, s_last, _ = fast_coord_descent(best_seq, t_last, n_trials=16, label="last")
            best_seq, best_score = check_and_update(a_last, best_seq, best_score, "final_fcd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
