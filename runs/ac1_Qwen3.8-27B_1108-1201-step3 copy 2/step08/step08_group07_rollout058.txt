"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent subgradient, LP fictitious play, and coordinate descent.
Leverages the key insight that the subgradient of the autocorrelation peak
is simply 2*a[k*-j], enabling very fast iterations.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Full linear convolution of two arrays via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def mirror_descent(a0, time_limit, initial_eta=None, verbose=True, label="MD"):
    """Mirror descent with subgradient for the normalized problem."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = initial_eta if initial_eta is not None else 1.0 / n
    min_eta = 0.01 / n
    stall = 0
    stall_limit = 200

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Subgradient: grad[j] = 2*a[k*-j] if 0<=k*-j<n else 0
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            grad[idx] = 2.0 * a[k_star - idx]

        # Mirror descent update: a_new[j] = a[j] * exp(-eta * grad[j])
        scale = eta * grad
        a_new = a * np.exp(-scale)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adaptive step size
        if stall > stall_limit:
            eta = max(min_eta, eta * 0.7)
            stall = 0
        elif stall == 0 and eta < 1.0 / n:
            eta = min(1.0 / n, eta * 1.02)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.6f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_lp_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 100:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0003
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    stall_outer = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.03, 0.08, 0.15, 0.3, 0.5, 0.7]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.4
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            stall_outer += 1
            if stall_outer > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_outer = 0
        else:
            stall_outer = 0

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score

def ramp_init(n, power=1.0):
    """Create a monotonically increasing initialization."""
    x = np.arange(n, dtype=np.float64)
    a = (x + 1) ** power
    a = a / a.sum()
    return a


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    return a


def chirp_init(n, freq=5.0, seed=0):
    """Oscillatory initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def two_block_init(n, ratio=0.4):
    """Two-block initialization: high on one side, low on other."""
    split = int(n * ratio)
    a = np.zeros(n)
    a[:split] = 1.0 / split
    a[split:] = 0.1 / (n - split)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_norm.copy()

    # Phase 5: Coordinate descent
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 150.0)
    if rem > 30 and best_seq is not None:
        print(f"  Budget: {t5:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t5, n_pairs=200, verbose=True)
        a_norm, sc_eval = check_and_update(a_cd, "coord_desc")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_norm.copy()

    # Phase 6: Try nearby lengths
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    n_curr = len(best_seq) if best_seq is not None else 768
    for alt_n in [n_curr + 2, n_curr - 2, n_curr + 4, n_curr + 8, n_curr + 16]:
        if alt_n < 128 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t6 = min(rem * 0.12, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  MD n={alt_n}, budget={t6:.0f}s")
        a_alt, sc_alt = mirror_descent(init_alt, t6, initial_eta=1.0/alt_n, verbose=False, label=f"MD-n{alt_n}")
        a_norm, sc_eval = check_and_update(a_alt, f"n{alt_n}")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_norm.copy()

    # Phase 7: Final LP polish
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 100.0)
    if rem > 20 and best_seq is not None:
        print(f"  Budget: {t7:.0f}s")
        a_lp_final, sc_lp_final = lp_fictitious_play(best_seq, t7, step=0.2, verbose=True, label="LP-final")
        a_norm, sc_eval = check_and_update(a_lp_final, "lp_final")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_norm.copy()

    # Phase 8: Final coordinate descent
    print(f"\n=== Phase 8: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.3, 60.0)
    if rem > 15 and best_seq is not None:
        a_cd2, sc_cd2 = coordinate_descent(best_seq, t8, n_pairs=100, verbose=False)
        a_norm, sc_eval = check_and_update(a_cd2, "final_cd")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_norm.copy()

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
        return [float(x) for x in best_seq]

    # Fallback
    n_fb = 768
    fallback = ramp_init(n_fb)
    a_md, _ = mirror_descent(fallback, 10.0, verbose=False)
    a_norm, sc_eval = check_and_update(a_md, "fallback")
    if np.isfinite(sc_eval):
        return [float(x) for x in a_norm]
    return [1.0 / n_fb] * n_fb
