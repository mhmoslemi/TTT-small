                delta_mid = (delta_lo + delta_hi) / 2.0

                new_aff = affected + 2.0 * delta_mid * a_ki
                if 0 <= idx_2i < length:
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] += delta_mid * delta_mid

                g_mid = max(unaff_max, float(new_aff.max()))

                if g_mid < best_g - 1e-18:
                    best_g = g_mid
                    best_delta = delta_mid

                # Subgradient sign
                eps = max(1e-13, 1e-10 * g_mid)
                active = new_aff >= (g_mid - eps)
                unaff_active = (unaff_max >= g_mid - eps)

                if active.any():
                    slopes = 2.0 * a_ki[active]
                    min_s = float(slopes.min())
                    max_s = float(slopes.max())
                    if 0 <= idx_2i < length and active[idx_2i]:
                        qs = 2.0 * a_ki[idx_2i] + 2.0 * delta_mid
                        min_s = min(min_s, qs)
                        max_s = max(max_s, qs)
                else:
                    min_s = max_s = 0.0

                if unaff_active:
                    min_s = min(min_s, 0.0)
                    max_s = max(max_s, 0.0)

                if min_s > 1e-10:
                    delta_hi = delta_mid
                elif max_s < -1e-10:
                    delta_lo = delta_mid
                else:
                    break

            # Apply if improves
            if best_delta > delta_lo + 1e-18:
                new_aff = affected + 2.0 * best_delta * a_ki
                if 0 <= idx_2i < length:
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] += best_delta * best_delta
                new_M = max(unaff_max, float(new_aff.max()))

                if new_M < M - 1e-17:
                    a[i] = max(0.0, a[i] + best_delta)
                    # Update conv incrementally
                    changes = 2.0 * best_delta * a_ki.copy()
                    if 0 <= idx_2i < length:
                        changes[idx_2i] += best_delta * best_delta
                    conv[k_start:k_end + 1] += changes
                    M = new_M
                    improved = True

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
            conv = fast_convolve(a)
            M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-16:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not improved and no_improve > 20:
            break
        if improved:
            no_improve = 0

        if verbose and sweeps % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sw={sweeps} ({elapsed:.0f}s): {best_score:.6f}")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
        best_score = 2.0 * len(best_a) * float(fast_convolve(best_a).max())
    return best_a, best_score


def nesterov_subgrad(a0, time_limit, lr_init=0.003, top_k=20, verbose=False, label=""):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    x = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = x.sum()
    x = x / s if s > 1e-15 else np.ones(n) / n

    x_prev = x.copy()
    best_a = x.copy()
    best_score = 2.0 * n * float(fast_convolve(x).max())
    t0 = time.monotonic()
    t = 1
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov momentum
        momentum = (t - 1) / (t + 2)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())

        # Subgradient: weighted average of active peak gradients
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        temp = 0.02 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * y[k - j]

        # Update
        lr_t = lr / (1.0 + 0.01 * t)
        x_new = simplex_project(y - lr_t * grad)
        x_prev = x
        x = x_new
        t += 1

        # Evaluate
        cur_score = 2.0 * n * float(fast_convolve(x).max())
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-8:
                break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG {label} t={t} ({elapsed:.0f}s): {best_score:.6f} lr={lr_t:.2e}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_peaks=8, verbose=False, label=""):
    """
    Greedy peak shaving: identify top peaks, find coordinates contributing to them,
    and redistribute mass from peak-contributing coords to non-peak coords.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        # Find top peaks
        k_peaks = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # For each peak, find contributing coordinates
        # Coordinate i contributes to peak k via a[i]*a[k-i]
        # We want to reduce peaks by decreasing a[i] where it contributes to peaks
        # and increasing a[j] where it doesn't

        # Compute "peak load" for each coordinate
        peak_load = np.zeros(n)
        non_peak_load = np.zeros(n)

        peak_set = set(peaks.tolist())

        for k in peaks:
            weight = conv[k] / conv[peaks].sum()  # weight by peak magnitude
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib = a[i] * a[j]
                peak_load[i] += weight * contrib
                if j != i:
                    peak_load[j] += weight * contrib
                # Non-peak contribution: total contribution minus peak
                # (approximate: use uniform distribution over non-peak positions)

        # Total contribution of each coordinate
        total_contrib = np.zeros(n)
        for i in range(n):
            if a[i] < 1e-15:
                continue
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            for k in range(k_start, k_end + 1):
                j = k - i
                if 0 <= j < n:
                    total_contrib[i] += a[j]

        # Fraction of each coordinate's contribution that goes to peaks
        frac_peak = np.zeros(n)
        for i in range(n):
            if total_contrib[i] > 1e-15:
                frac_peak[i] = peak_load[i] / total_contrib[i]

        # Redistribute: decrease high-frac_peak coords, increase low-frac_peak coords
        # Use a smooth update
        target = np.ones(n) / n
        blend = 0.05 + 0.05 * (it % 10) / 10.0  # small blending

        # Decrease coords with high peak fraction
        for i in range(n):
            if frac_peak[i] > 0.5 and a[i] > 1e-10:
                a[i] *= (1.0 - blend * (frac_peak[i] - 0.5) * 2.0)

        # Increase coords with low peak fraction (and low current value)
        low_peak_mask = frac_peak < 0.3
        deficit = a.sum() - 1.0  # how much to add back
        if deficit > 1e-10:
            low_coords = np.where(low_peak_mask)[0]
            if len(low_coords) > 0:
                # Distribute deficit proportionally to (1 - a[i] * n)
                weights = np.maximum(1.0 - a[low_coords] * n, 0.1)
                weights = weights / weights.sum()
                a[low_coords] += deficit * weights

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS {label} it={it} ({elapsed:.0f}s): {best_score:.6f}")

    return best_a, best_score


def block_update(a0, time_limit, n_peaks=6, verbose=False, label=""):
    """
    Block coordinate update: simultaneously adjust a group of coordinates
    that all contribute to the current maximum peak.
    Uses a simple iterative water-filling approach.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        # Find the current max peak
        k_max = int(np.argmax(conv))

        # Find coordinates that contribute to this peak
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        contributors = np.arange(i_start, i_end + 1)

        # For each contributor i, the sensitivity is 2*a[k_max - i]
        # We want to decrease a[i] for high-sensitivity contributors
        # and increase a[j] for non-contributors

        # Simple approach: proportionally reduce contributors, add to non-contributors
        step = 0.02
        contrib_vals = a[contributors]
        sens = 2.0 * a[contributors]  # sensitivity of peak to each contributor
        # Reduce in proportion to sensitivity * value (larger reduction for bigger contributors)
        reduction = step * sens * contrib_vals / (sens * contrib_vals).sum()
        a[contributors] -= reduction

        # Add the removed mass to non-contributors
        removed = reduction.sum()
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) > 0:
            # Add to lowest-value non-contributors
            n_add = min(10, len(non_contrib))
            lowest = non_contrib[np.argsort(a[non_contrib])[:n_add]]
            a[lowest] += removed / len(lowest)

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Check for plateau
        if it > 100 and abs(cur_score - best_score) < 1e-8:
            # Try a bigger step
            step = 0.05

    return best_a, best_score


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harm = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_u_shape(n, rng):
    """U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_multi_peak(n, rng):
    """Multiple Gaussian peaks."""
    n_peaks = rng.integers(2, 6)
    a = np.zeros(n)
    for _ in range(n_peaks):
        center = rng.uniform(0, n)
        width = n * rng.uniform(0.05, 0.2)
        a += np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    return f_interp / ss if ss > 1e-15 else np.ones(new_n) / new_n


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():

    Multi-phase optimization:
    Phase 1: Diverse initialization + quick CD
    Phase 2: Intensive coordinate descent with exact bisection
    Phase 3: Nesterov subgradient for global exploration
    Phase 4: Peak shaving + block updates
    Phase 5: Final intensive polish
    Phase 6: Multi-length exploration
    """

    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick CD ===
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()

    n_values = [896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []
    init_types = ['parent', 'fourier', 'cosine', 'u_shape', 'multi_peak']

    for n in n_values:
        if time.monotonic() - t1_start > 250:
            break
        for c in range(len(init_types)):
            itype = init_types[c]
            if itype == 'parent' and parent is not None:
                cand = interpolate_seq(parent, n)
            elif itype == 'fourier':
                cand = init_fourier(n, rng)
            elif itype == 'cosine':
                cand = init_cosine(n, rng)
            elif itype == 'u_shape':
                cand = init_u_shape(n, rng)
            else:
                cand = init_multi_peak(n, rng)

            t_left = 250.0 - (time.monotonic() - t1_start)
            t_alloc = max(3.0, min(10.0, t_left / 8.0))
            a_opt, s_opt, = coord_descent_bisection(cand, t_alloc, label=f"p1_n{n}_{itype[:3]}")
            candidates.append((s_opt, a_opt, n, itype))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top scores: {[f'{c[0]:.6f}' for c in candidates[:5]]}")

    for s_c, a_c, n_c, it_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{it_c}")

    # === Phase 2: Intensive coordinate descent ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_budget = 500.0
        t2_start = time.monotonic()

        current = best_seq.copy()
        for rnd in range(4):
            rem = deadline - time.monotonic() - 700
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, t2_budget / 4.0 + 50)
            print(f"  CD round {rnd + 1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = coord_descent_bisection(current, t_alloc, verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_cd_r{rnd}")

            # Perturb to escape
            if rnd < 3:
                conv = fast_convolve(a_opt)
                top_k = min(5, len(conv))
                peaks = np.argpartition(conv, -top_k)[-top_k:]
                for k in peaks:
                    i_start = max(0, k - n_best + 1)
                    i_end = min(n_best - 1, k)
                    idx = np.arange(i_start, i_end + 1)
                    picks = rng.choice(idx, size=max(1, len(idx) // 5), replace=False)
                    for i in picks:
                        a_opt[i] *= (1.0 + rng.standard_normal() * 0.04)
                a_opt = np.maximum(a_opt, 0)
                a_opt = a_opt / a_opt.sum()
                current = a_opt

    # === Phase 3: Nesterov subgradient ===

            a_pert = best_seq + rng.standard_normal(len(best_seq)) * 0.02

    # === Phase 4: Peak shaving + block updates ===
    print(f"\n=== Phase 4: Peak Shave + Block (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_ps = min(rem * 0.4, 120.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shave(best_seq, t_ps, n_peaks=10, label="ps")

        rem = deadline - time.monotonic() - 300
        if rem > 30:

            print(f"  BlockUpdate budget={t_bu:.0f}s")
            a_bu, s_bu = block_update(best_seq, t_bu, n_peaks=6, label="bu")
            best_seq, best_score = check_and_update(a_bu, best_seq, best_score, "block")

    # === Phase 5: Final intensive CD polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_pol = min(rem * 0.6, 200.0)
            print(f"  Final CD n={len(best_seq)}, budget={t_pol:.0f}s")
            a_pol, s_pol = coord_descent_bisection(best_seq, t_pol, verbose=True, label="final_cd")
            best_seq, best_score = check_and_update(a_pol, best_seq, best_score, "final_cd")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_pol2 = min(rem * 0.7, 100.0)
            a_pol2, s_pol2 = coord_descent_bisection(best_seq, t_pol2, label="final_cd2")
            best_seq, best_score = check_and_update(a_pol2, best_seq, best_score, "final_cd2")

    # === Phase 6: Alternative length ===
    print(f"\n=== Phase 6: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 50:

            for alt_n in [n_best - 128, n_best + 128, n_best + 256]:
                if alt_n < 384 or alt_n > 2048:

                if rem < 30:

                t_alt = min(rem * 0.5, 60.0)
                init_alt = interpolate_seq(best_seq, alt_n)

    # === Final verification ===
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
