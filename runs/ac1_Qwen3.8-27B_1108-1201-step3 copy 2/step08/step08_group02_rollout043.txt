def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def solve_primal_lp(a, peak_indices, M):
    """
    Solve: max sum(g) s.t. (a*g)[m] <= M for m in peak_indices, g >= 0.
    Returns g or None.
    """
    n = len(a)
    K = len(peak_indices)
    
    # Build constraint matrix A (K x n): A[i, j] = a[m_i - j] if 0 <= m_i - j <= n-1
    # (a*g)[m] = sum_j a[m-j] * g[j] for valid j
    # So the coefficient of g[j] in (a*g)[m_i] is a[m_i - j] if 0 <= m_i - j <= n-1.

    cols = []
    vals = []

            cols.extend(js.tolist())

"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP-based direction finding via scipy HiGHS, exact quadratic line search,
multi-start with length adaptation, and coordinate descent polishing.
Key difference from previous approach: uses scipy linprog (HiGHS) directly instead of
cvxpy, exact quadratic line search exploiting the bilinear structure, and aggressive
multi-start/length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_primal_lp(a, peak_indices, M):
    """
    Solve: max sum(g) s.t. (a*g)[m] <= M for m in peak_indices, g >= 0.
    Returns g vector or None.
    """
    n = len(a)
    K = len(peak_indices)
    rows = []
    cols = []
    vals = []
    for i, m in enumerate(peak_indices):
        j_start = max(0, m - (n - 1))
        j_end = min(n - 1, m)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([i] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[m - js].tolist())
    if not rows:
        return None
    A_mat = csr_matrix((vals, (rows, cols)), shape=(K, n))
    res = linprog(
        c=-np.ones(n),
        A_ub=A_mat,
        b_ub=np.full(K, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 5}
    )
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    if not np.all(np.isfinite(g)) or g.sum() <= 0:
        return None
    return g


def quadratic_line_search(a, h, conv_a, conv_cross, conv_h, n_grid=200):
    """
    Find t in [0,1] minimizing max_m [(1-t)^2*conv_a[m] + 2t(1-t)*conv_cross[m] + t^2*conv_h[m]].
    This is exact since (1-t)*a + t*h >= 0 when a,h >= 0.
    """
    # f_m(t) = conv_a[m] + B[m]*t + C[m]*t^2
    B = 2.0 * (conv_cross - conv_a)
    C = conv_a - 2.0 * conv_cross + conv_h

    ts = np.linspace(0.005, 0.995, n_grid)
    best_f = np.inf
    best_t = 0.0
    for t in ts:
        f_vals = conv_a + B * t + C * t * t
        val = f_vals.max()
        if val < best_f:
            best_f = val
            best_t = t

    # Refine around best
    if best_t > 0.01:
        lo = max(0.001, best_t - 0.02)
        hi = min(0.999, best_t + 0.02)
        ts2 = np.linspace(lo, hi, 50)
        for t in ts2:
            f_vals = conv_a + B * t + C * t * t
            val = f_vals.max()
            if val < best_f:
                best_f = val
                best_t = t

    return best_t, best_f


def lp_optimize(a_init, time_limit, label="lp", peak_frac_start=0.96, peak_frac_end=0.82,
                max_peaks_start=150, max_peaks_end=500, patience=20, seed=42):
    """
    LP-based optimization with adaptive peaks and quadratic line search.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_a = fast_convolve(a)
        M = conv_a.max()

        # Adaptive parameters
        progress = min(1.0, it / (3.0 * patience))
        frac = peak_frac_start + (peak_frac_end - peak_frac_start) * progress
        max_k = int(max_peaks_start + (max_peaks_end - max_peaks_start) * progress)
        max_k = min(max_k, 2 * n - 1)

        # Select peaks
        threshold = frac * M
        peaks = np.where(conv_a >= threshold)[0]
        if len(peaks) > max_k:
            peaks = np.argpartition(conv_a, -max_k)[-max_k:]
        if len(peaks) < 8:
            peaks = np.argpartition(conv_a, -max(10, len(conv_a) // 5))[-max(10, len(conv_a) // 5):]

        # Solve LP
        g = solve_primal_lp(a, peaks, M)
        if g is None:
            stuck += 1
            if stuck > patience:
                a = a + rng.standard_normal(n) * 0.001 * (a + 0.1 / n)
                a = normalize(a)
                stuck = 0
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > patience // 2:
                # Try with more peaks
                n_more = min(2 * n - 1, max_k * 2)
                if len(peaks) < n_more:
                    more_peaks = np.argpartition(conv_a, -n_more)[-n_more:]
                    g2 = solve_primal_lp(a, more_peaks, M)
                    if g2 is not None and g2.sum() > 1.0 + 1e-6:
                        g = g2
                        S_g = g.sum()
                    else:
                        stuck += 1
                        if stuck > patience:
                            a = a + rng.standard_normal(n) * 0.002 * (a + 0.1 / n)
                            a = normalize(a)
                            stuck = 0
                        continue
                else:
                    if stuck > patience:
                        a = a + rng.standard_normal(n) * 0.002 * (a + 0.1 / n)
                        a = normalize(a)
                        stuck = 0
                    continue

        # Line search
        h = g / S_g
        conv_h = fast_convolve(h)
        conv_cross = fast_cross_convolve(a, h)
        best_t, best_M_trial = quadratic_line_search(a, h, conv_a, conv_cross, conv_h)

        if best_t <= 0.001:
            stuck += 1
            if stuck > patience:
                a = a + rng.standard_normal(n) * 0.002 * (a + 0.1 / n)
                a = normalize(a)
                stuck = 0
            continue

        # Update
        a = (1.0 - best_t) * a + best_t * h
        a = normalize(a)
        cur_M = fast_convolve(a).max()

        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 25 == 0:
            score = 2.0 * n * best_M
            print(f"  {label} it={it} M={best_M:.10f} score={score:.6f} K={len(peaks)} "
                  f"Sg={S_g:.4f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_M


def coordinate_descent(a_init, time_limit, label="cd"):
    """Coordinate descent: try scaling each element by a multiplier."""
    t0 = time.monotonic()
    n = len(a_init)
    a = normalize(a_init.copy())
    best_M = fast_convolve(a).max()
    multipliers = [0.6, 0.75, 0.85, 0.9, 1.1, 1.15, 1.25, 1.5, 2.0]
    no_round = 0

    while time.monotonic() - t0 < time_limit and no_round < 3:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue
            orig_val = a[j]
            best_j_M = best_M
            best_j_mult = 1.0
            for mult in multipliers:
                a[j] = orig_val * mult
                s = a.sum()
                if s < 1e-10:
                    a[j] = orig_val
                    continue
                a_norm = a / s
                M = fast_convolve(a_norm).max()
                if M < best_j_M - 1e-14:
                    best_j_M = M
                    best_j_mult = mult
            if best_j_mult != 1.0:
                a[j] = orig_val * best_j_mult
                a = normalize(a)
                best_M = best_j_M
                improved = True
        if not improved:
            no_round += 1
        else:
            no_round = 0

    return normalize(a), best_M


def interpolate_length(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a)
    return normalize(a_new)


def subgradient_polish(a_init, time_limit, label="sg", n_top=100, lr_init=0.005, seed=77):
    """Projected subgradient descent with softmax-weighted peaks."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    a = normalize(a_init.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    lr = lr_init
    temp = 0.003
    no_improve = 0
    it = 0

    # Precompute the gradient structure
    # grad[j] = sum_{m: active} w[m] * a[m-j] (when valid)
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Softmax weights on top peaks
        scaled = conv / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(conv):
            top_idx = np.argpartition(conv, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(conv))
            w[top_idx] = w_sel

        # Compute gradient via convolution
        # grad = cross_convolve(w, a_flipped) ... actually:
        # (a*w)[k] = sum_j a[j] * w[k-j], but we want sum_m w[m] * a[m-j] for each j
        # which is (w * a)[j] where * is convolution... let me think.
        # We want grad[j] = sum_m w[m] * A[m,j] where A[m,j] = a[m-j] if valid.
        # This is: grad = cross_convolve(w, a) ... no.
        # Actually grad[j] = sum_{m: valid} w[m] * a[m-j]
        # Let k = m-j, so m = j+k. grad[j] = sum_k w[j+k] * a[k] (for valid indices).
        # This is the cross-correlation of w and a, which is cross_convolve(w, a_flipped).
        # Simpler: just compute it directly for the top peaks.
        grad = np.zeros(n)
        for m in top_idx if n_top < len(conv) else range(len(conv)):
            if w.get(m, 0) < 1e-12:
                continue
            wm = w[m]
            j_start = max(0, m - (n - 1))
            j_end = min(n - 1, m)
            grad[j_start:j_end + 1] += wm * a[m - j_start:j_end + 1]

        # Nesterov momentum
        v = 0.85 * v - lr * grad
        a_new = a + v
        a = normalize(a_new)

        if it % 100 == 0:
            cur_M = fast_convolve(a).max()
            if cur_M < best_M - 1e-14:
                best_M = cur_M
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.04)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.93
                    no_improve = 0
                    if lr < 0.0005:
                        # Restart with perturbation
                        a = best_a + rng.standard_normal(n) * 0.003
                        a = normalize(a)
                        v = np.zeros(n)
                        lr = lr_init * 0.5
                        temp = min(temp * 1.1, 0.05)

        if it % 500 == 0:
            score = 2.0 * n * best_M
            print(f"  {label} it={it} M={best_M:.10f} score={score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_M
