import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_ah(a, w):
    """Compute (a * w)[j] = sum_i a[i]*w[i+j] for j=0..n-1 via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (n + size - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    return irfft(np.conj(A) * W, fft_size)[:n]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_z(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_obj_and_grad_z(z, T):
    """Compute smooth objective and gradient in log-space (z parameterization)."""
    a = softmax_z(z)
    n = len(a)
    conv = fast_convolve(a)  # length 2n-1
    # Log-sum-exp of T*conv
    tc = T * conv
    tc_max = tc.max()
    log_sum_exp = tc_max + np.log(np.exp(tc - tc_max).sum())
    obj = (2.0 * n / T) * log_sum_exp
    # Weights w_k = exp(T*c_k) / sum(exp(T*c_k))
    w = np.exp(tc - tc_max)
    w /= w.sum()
    # Gradient w.r.t. a: g[j] = 4n * sum_i a[i] * w[j+i] = 4n * cross_corr(a, w)[j]
    g = 4.0 * n * cross_corr_ah(a, w)
    # Project to log-space (simplex tangent): grad_z[j] = a[j]*(g[j] - dot(g,a))
    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj, grad_z


def smooth_obj_and_grad_linear(a, T):

    n = len(a)

    obj = (2.0 * n / T) * log_sum_exp
    w = np.exp(tc - tc_max)
    w /= w.sum()
    g = 4.0 * n * cross_corr_ah(a, w)
    return obj, g

        if sc < best_score:
            best_a[:] = a_tmp
            best_score_val = sc
        obj, _ = smooth_obj_and_grad_z(z, T)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        T = T_schedule[min(n_iter[0], maxiter - 1)]
        _, g = smooth_obj_and_grad_z(z, T)
        return g

    best_score_val = best_score
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_z(res.x)
        score = compute_score(a_out)
        if score < best_score_val:
            best_a = a_out
            best_score_val = score
    except Exception:
        pass

    # Final check
    final_score = compute_score(softmax_z(res.x)) if hasattr(res, 'x') else best_score_val
    if final_score < best_score_val:
        best_a = softmax_z(res.x)
        best_score_val = final_score

    return best_a, best_score_val


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n
        C[k_start:k_end, j] = f
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None, None
    g = np.maximum(res.x, 0)
    # Get dual values (shadow prices)
    duals = None
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        duals = np.maximum(res.ineqlin.marginals, 0)
    return g, duals


def greedy_lp(f, M):
    """Greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def linear_update(f, g, t):
    """Linear (convex) interpolation."""
    a = (1 - t) * f + t * g
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, use_geometric=True, verbose=False, momentum=0.9):
    """LP-based fictitious play with adaptive step size and optional momentum."""
    f = f_init.copy()
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    f_avg = f.copy()
    vel = np.zeros(n)

    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()

        try:
            g, duals = solve_lp(f, M)
        except Exception:
            g = greedy_lp(f, M)
            duals = None

        if g is None or g.sum() < 1e-15:
            g = greedy_lp(f, M)
        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Adaptive step size
        t_step = 2.0 / (k + 2)
        if k > 200:
            t_step = max(0.005, 2.0 / (k + 2))

        # Compute update direction
        if use_geometric:
            a_new = geometric_mean_update(f, g, t_step)
        else:
            a_new = linear_update(f, g, t_step)

        # Momentum
        vel = momentum * vel + (1 - momentum) * (a_new - f)
        a_mom = f + vel
        a_mom = np.maximum(a_mom, 0)
        a_mom = a_mom / a_mom.sum()

        # Score both
        score_new = compute_score(a_new)
        score_mom = compute_score(a_mom)

        f_avg = ((k * f_avg) + a_new) / (k + 1)
        score_avg = compute_score(f_avg)

        # Pick the best
        candidates = [a_new, a_mom, f_avg]
        scores = [score_new, score_mom, score_avg]
        best_idx = np.argmin(scores)
        f_new = candidates[best_idx]
        sc_new = scores[best_idx]

        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()

        f = f_new

        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, score={sc_new:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_f, best_score


def anti_peak_water_filling(a0, time_limit, verbose=False):
    """Target the highest peaks and redistribute mass away from contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        # Sort peaks by height
        peak_order = np.argsort(conv)[::-1]

        improved = False
        for pk in peak_order[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            # Contributing pairs: a[i] * a[pk-i] for valid i
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Find the top contributor
            top_ci = np.argmax(contribs)
            i_top = idx[top_ci]
            j_top = pk - i_top

            if a[i_top] < 1e-12:
                continue

            # Try transferring mass from i_top to a position that doesn't contribute to this peak
            # Find positions far from both i_top and j_top relative to peak
            noncontrib = np.setdiff1d(np.arange(n), idx)
            if len(noncontrib) == 0:
                continue

            # Pick a non-contributing position with relatively low value
            candidates = noncontrib[np.argsort(a[noncontrib])[:5]]
            best_trial = None
            best_trial_score = best_score

            for j_target in candidates:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = a[i_top] * frac
                    a_trial = a.copy()
                    a_trial[i_top] -= delta
                    a_trial[j_target] += delta
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_trial_score:
                        best_trial_score = sc
                        best_trial = a_trial.copy()

            if best_trial is not None and best_trial_score < best_score - 1e-13:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True

        if not improved:
            # Perturb
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and outer % 10 == 0:
            print(f"    WF iter={outer}, score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent with golden-section search for pair transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    gr = (np.sqrt(5) - 1) / 2

    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)

        # Generate candidate pairs
        n_pairs = min(n, 300)
        pairs = []
        # Peak-focused
        for _ in range(n_pairs // 2):
            i = rng.integers(js, je + 1)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        # Random
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(t):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                if a_trial[i] < -1e-12 or a_trial[j] < -1e-12:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score(a_trial)

            f1 = eval_t(x1)
            f2 = eval_t(x2)

            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)

            f_opt = min(f1, f2)
            t_opt = (t_lo + t_hi) / 2

            if f_opt < best_score - 1e-12:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and iters % 20 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def structured_init(n, seed=0):
    """Create a structured non-uniform initialization."""
    rng = np.random.default_rng(seed)
    # Start with a chirp-like structure
    t = np.linspace(0, 1, n)
    base = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * t) + 0.3 * np.cos(2 * np.pi * 7 * t)
    base = np.maximum(base, 0.1)
    base += rng.standard_normal(n) * 0.1
    base = np.maximum(base, 0.01)
    base /= base.sum()
    return base


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization for minimum peak autocorrelation.
    Uses smooth L-BFGS, LP fictitious play, anti-peak water filling,
    and coordinate descent with multi-start and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = structured_init(n_init, seed=42)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Smooth L-BFGS with multiple temperature schedules ===
    print(f"\n=== Phase 1: Smooth L-BFGS (best={best_score:.8f}) ===")
    T_schedules = [
        (5000, 500, 4000),   # (T_start, T_end, maxiter)
        (10000, 2000, 3000),
        (2000, 200, 3000),
    ]
    for idx, (T_s, T_e, mx) in enumerate(T_schedules):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 300.0)
        if t_alloc < 20:
            break
        print(f"  L-BFGS T={T_s}->{T_e}, maxiter={mx}, budget={t_alloc:.0f}s")
        a_out, sc_out = lbgfs_smooth(best_seq, t_alloc, T_s, T_e, maxiter=mx, verbose=False)
        best_seq, best_score = check_and_update(a_out, f"lbgfs_{idx}", best_seq, best_score)

    # === Phase 2: LP Fictitious Play (Geometric Mean) ===
    print(f"\n=== Phase 2: LP Fictitious Play [geo] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.2, 350.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_alloc, use_geometric=True, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_geo", best_seq, best_score)

    # === Phase 3: LP Fictitious Play (Linear) ===
    print(f"\n=== Phase 3: LP Fictitious Play [linear] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t_alloc = min(rem * 0.15, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t_alloc, use_geometric=False, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "lp_lin", best_seq, best_score)

    # === Phase 4: Anti-Peak Water Filling ===
    print(f"\n=== Phase 4: Anti-Peak Water Filling (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_wf, sc_wf = anti_peak_water_filling(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_wf, "waterfill", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.2, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd", best_seq, best_score)

    # === Phase 6: Second round of L-BFGS (finer) ===
    print(f"\n=== Phase 6: Fine L-BFGS (best={best_score:.8f}) ===")
    for T_s, T_e, mx in [(500, 100, 5000), (2000, 500, 4000)]:
        rem = deadline - time.monotonic() - 150
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 200.0)
        if t_alloc < 15:
            break
        print(f"  Fine L-BFGS T={T_s}->{T_e}, budget={t_alloc:.0f}s")

    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")

        n_parent - 3, n_parent - 2, n_parent - 1, n_parent + 1,

        if rem < 20:

        t_alloc = min(rem * 0.12, 80.0)

            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_lp", best_seq, best_score)

    # === Phase 8: Multi-start L-BFGS ===
    print(f"\n=== Phase 8: Multi-Start (best={best_score:.8f}) ===")
    for restart in range(3):
        rem = deadline - time.monotonic() - 40
        if rem < 15:
            break
        # Perturb best
        noise = rng.standard_normal(len(best_seq)) * (0.01 + 0.01 * restart)
        a_pert = best_seq + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        t_alloc = min(rem * 0.25, 100.0)
        if t_alloc < 10:
            break
        print(f"  Restart {restart}, budget={t_alloc:.0f}s")
        a_rs, sc_rs = lbgfs_smooth(a_pert, t_alloc, T_start=2000, T_end=300, maxiter=3000)
        best_seq, best_score = check_and_update(a_rs, f"restart_{restart}", best_seq, best_score)

    # === Phase 9: Final Polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.5, 60.0)
        a_final, sc_final = coordinate_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_final, "final_cd", best_seq, best_score)

    # One more L-BFGS with very low temperature
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_fin, sc_fin = lbgfs_smooth(best_seq, rem * 0.8, T_start=500, T_end=100, maxiter=2000)
        best_seq, best_score = check_and_update(a_fin, "final_lb", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = best_seq if best_seq is not None else np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
