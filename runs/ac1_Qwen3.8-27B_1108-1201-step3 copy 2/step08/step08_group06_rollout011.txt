"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive-temperature log-sum-exp gradient descent (smoothed objective)
2. LP-based fictitious play with adaptive M scheduling (layer-by-layer peeling)
3. Active-set-aware coordinate descent with priority-based pair selection
4. Edge-aware refinement for boundary peak reduction

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    # For convolution (f*g)[k] = sum_j f[k-j] * g[j]
    # C[k,j] = f[k-j] if 0 <= k-j < n else 0
    C = np.zeros((2 * n - 1, n))
    kf = np.arange(2 * n - 1)[:, None]
    jf = np.arange(n)[None, :]
    idx = kf - jf
    mask = (idx >= 0) & (idx < n)
    C[mask] = f[idx[mask]]
    return C

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)

def log_sum_exp_smooth_max(a, T):
    """Compute smoothed max of convolution and its gradient using log-sum-exp."""
    n = len(a)
    conv = fast_convolve(a)
    smax = conv.max()
    shifted = (conv - smax) / T
    w = np.exp(shifted)
    w = w / w.sum()
    smooth_max = smax + T * np.log(w.sum())
    
    # Gradient w.r.t. a: for each i, d/max/da_i = sum_k w_k * d(conv[k])/da_i
    # d(conv[k])/da_i = f[k-i] (the cross-correlation)
    # So grad[i] = sum_k w_k * C[k,i] = (C^T w)[i]
    C = build_C(a)
    grad_a = C.T @ w
    return smooth_max, grad_a

def smoothed_descent(a0, time_limit, T0=100.0, T_end=1.0, verbose=True):
    """Gradient descent on the smoothed objective with adaptive temperature."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    total_iters = 5000
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature: exponential decay
        frac = it / total_iters
        T = T0 * (T_end / T0) ** frac
        
        smooth_max, grad_a = log_sum_exp_smooth_max(a, T)
        
        # Project gradient onto simplex tangent space
        s = grad_a @ a
        grad_proj = a * (grad_a - s)
        
        # Adaptive step size
        eta = T / (n * 0.1)  # step proportional to temperature
        eta = min(eta, 0.5)
        
        a_new = a - eta * grad_proj
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"  SD iter={it}, T={T:.2f}, score={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    kf = np.arange(2 * n - 1)[:, None]
    jf = np.arange(n)[None, :]
    idx = kf - jf
    mask = (idx >= 0) & (idx < n)
    C = np.zeros((2 * n - 1, n))
    C[mask] = f[idx[mask]]
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*f*g <= M, g >= 0. Returns (g, dual_y)."""
    n = len(f)
    C = build_C(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_vec, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None, None
    g = np.maximum(res.x, 0)
    dual_y = np.zeros(2 * n - 1)
    if res.ineqlin is not None and len(res.ineqlin) > 0:
        dual_y = np.maximum(res.ineqlin, 0)
    return g, dual_y


def smoothed_max_and_grad(a, T):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    smax = conv.max()
    shifted = (conv - smax) / T
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    smooth_max = smax + T * np.log(w_sum)
    # Gradient: (C^T w) where C[k,i] = a[k-i]
    C = build_C(a)
    grad_a = C.T @ w
    return smooth_max, grad_a, w

    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    max_iters = 20000
    patience = 0
    max_patience = 500

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac

        smooth_max, grad_a, w = smoothed_max_and_grad(a, T)

        # Project gradient onto simplex tangent
        s_val = grad_a @ a
        grad_proj = a * (grad_a - s_val)

        # Adaptive step size
        gnorm = np.sqrt(np.sum(grad_proj ** 2)) + 1e-15
        eta = T / (n * 0.05)
        eta = min(eta, 0.3)

        a_new = a - eta * grad_proj
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience > max_patience:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SD iter={it}, T={T:.3f}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def lp_fictitious_play_adaptive(a0, time_limit, verbose=True):
    """LP fictitious play with adaptive M and dual-guided updates."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    current_M = fast_convolve(a).max()
    iterate_history = []
    max_history = 50
    stagnation = 0

    for cycle in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP
        g, dual_y = solve_lp(a, current_M)
        if g is None or g.sum() < 1e-15:
            current_M *= 1.002
            continue

        g = g / g.sum()

        # Geometric mean update with adaptive step
        f_safe = np.maximum(a, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        t_step = max(0.1, 0.5 * (0.95 ** cycle))
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        new_score = compute_score(a_new)

        # Track iterate history for averaging
        iterate_history.append(a_new.copy())
        if len(iterate_history) > max_history:
            iterate_history.pop(0)

        if new_score < best_score - 1e-13:
            best_score = new_score
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Check averaged solution
        if len(iterate_history) >= 10:
            a_avg = np.mean(iterate_history, axis=0)
            a_avg = np.maximum(a_avg, 0)
            a_avg = a_avg / a_avg.sum()
            avg_score = compute_score(a_avg)
            if avg_score < best_score - 1e-13:
                best_score = avg_score
                best_a = a_avg.copy()

        # Update a
        a = a_new

        # Adapt M
        a_max = fast_convolve(a).max()
        if stagnation > 5:
            # Try more aggressive M reduction
            current_M = max(a_max * 0.995, a_max - 1e-6)
            stagnation = 0
        else:
            current_M = max(a_max, current_M * 0.998)

        if verbose and cycle % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FA cycle={cycle}, M={current_M:.6e}, score={best_score:.8f}, "
                  f"stagn={stagnation}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def hybrid_lp_gradient(a0, time_limit, verbose=True):
    """Hybrid: alternate between LP solves and gradient steps."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    current_M = fast_convolve(a).max()
    T = 10.0

    phase = 0
    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        if step % 5 == 0:
            # LP step
            g, _ = solve_lp(a, current_M)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                f_safe = np.maximum(a, 1e-30)
                g_safe = np.maximum(g, 1e-30)
                t_step = max(0.1, 0.4 * (0.98 ** step))
                log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
                log_new -= log_new.max()
                a_new = np.exp(log_new)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                sc = compute_score(a_new)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
                a_max = fast_convolve(a).max()
                current_M = max(a_max * 0.999, a_max - 1e-7)
        else:
            # Gradient step
            T = max(0.5, 10.0 * (0.97 ** step))
            _, grad_a, _ = smoothed_max_and_grad(a, T)
            s_val = grad_a @ a
            grad_proj = a * (grad_a - s_val)
            eta = T / (n * 0.1)
            eta = min(eta, 0.2)
            a_new = a - eta * grad_proj
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  HYB step={step}, M={current_M:.6e}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def priority_cd(a0, time_limit, verbose=True):
    """Priority-based coordinate descent targeting top peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()

        # Find top peaks
        top_k = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]

        improved_this_iter = False
        pairs_done = 0
        max_pairs = min(300, n // 2)

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue
            contrib_vals = a[contrib_idx] * a[pk - contrib_idx]

            # Top contributors to reduce
            top_contrib_pos = contrib_idx[np.argsort(contrib_vals)[::-1][:5]]

            # Positions far from this peak (to receive mass)
            far_positions = np.setdiff1d(np.arange(n), contrib_idx)
            if len(far_positions) == 0:
                far_positions = np.arange(max(0, js - 5), min(n, je + 6))
            # Pick low-value positions among far ones
            far_sorted = far_positions[np.argsort(a[far_positions])[:min(5, len(far_positions))]]

            for i_idx in top_contrib_pos:
                if a[i_idx] < 1e-14:
                    continue
                for j_idx in far_sorted:
                    if i_idx == j_idx or a[i_idx] < 1e-14:
                        continue

                    # Binary search for best transfer
                    lo = 0.0
                    hi = a[i_idx] * 0.4
                    best_t = 0.0
                    best_sc = best_score

                    for _ in range(12):
                        mid = (lo + hi) / 2
                        a_trial = a.copy()
                        a_trial[i_idx] -= mid
                        a_trial[j_idx] += mid
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_sc:
                            best_sc = sc
                            best_t = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_t > 1e-14 and best_sc < best_score - 1e-13:
                        a[i_idx] -= best_t
                        a[j_idx] += best_t
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_score = best_sc
                        best_a = a.copy()
                        improved_this_iter = True

                    pairs_done += 1
                    if pairs_done >= max_pairs:
                        break
                if pairs_done >= max_pairs:
                    break
            if pairs_done >= max_pairs:
                break

        # Periodic perturbation
        if not improved_this_iter and iters % 30 == 0:
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 20 == 0:
            print(f"  PCD iter={iters}, score={best_score:.8f}, max_conv={max_val:.6e}")

    return best_a, best_score


def bimodal_init(n, edge_factor=5.0):
    """Create a bimodal initialization with boosted edges."""
    a = np.ones(n)
    # Boost edges with smooth falloff
    for i in range(n):
        dist = min(i, n - 1 - i)
        a[i] = 1.0 + (edge_factor - 1.0) * np.exp(-dist / 10.0)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def multi_start_search(a0, time_limit, verbose=True):
    """Try multiple initializations and run LP on the best."""
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(55)

    # Generate diverse starts
    starts = [a0.copy()]

    # Bimodal
    starts.append(bimodal_init(n, edge_factor=3.0))
    starts.append(bimodal_init(n, edge_factor=8.0))

    # Edge-only boost
    a_edge = a0.copy()
    a_edge[0] += 0.1
    a_edge[-1] += 0.1
    a_edge = a_edge / a_edge.sum()
    starts.append(a_edge)

    # Random perturbation
    a_rnd = a0 + rng.standard_normal(n) * 0.03
    a_rnd = np.maximum(a_rnd, 0)
    a_rnd = a_rnd / a_rnd.sum()
    starts.append(a_rnd)

    # Smoothed version (low-pass filter)
    from scipy.ndimage import uniform_filter1d
    a_smooth = uniform_filter1d(a0, size=5)
    a_smooth = np.maximum(a_smooth, 0)
    a_smooth = a_smooth / a_smooth.sum()
    starts.append(a_smooth)

    per_start = (time_limit - 20) / len(starts)

    for si, start in enumerate(starts):
        if time.monotonic() - t0 > time_limit:
            break

        sc_init = compute_score(start)
        if verbose:
            print(f"  MS start {si}: score={sc_init:.8f}")

        if sc_init < best_score:
            best_score = sc_init
            best_a = start.copy()

        # Run LP on this start
        a_out, sc_out = lp_fictitious_play_adaptive(start, per_start * 0.7, verbose=False)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            if verbose:
                print(f"  MS start {si}: LP improved to {sc_out:.8f}")

        # Quick gradient polish
        rem = time_limit - (time.monotonic() - t0) - 10
        if rem > 10:
            a_gr, sc_gr = smoothed_descent(a_out, min(rem * 0.3, 30.0), T0=5.0, T1=0.5, verbose=False)
            if sc_gr < best_score:
                best_score = sc_gr
                best_a = a_gr.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def final_polish(a0, time_limit, verbose=True):
    """Final polish: alternates between small LP steps and gradient steps."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    current_M = fast_convolve(a).max()
    T = 2.0

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        if step % 3 == 0:
            # Small LP step
            g, _ = solve_lp(a, current_M)
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                f_safe = np.maximum(a, 1e-30)
                g_safe = np.maximum(g, 1e-30)
                log_new = 0.7 * np.log(f_safe) + 0.3 * np.log(g_safe)
                log_new -= log_new.max()
                a_new = np.exp(log_new)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                sc = compute_score(a_new)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
                current_M = max(fast_convolve(a).max() * 0.9995, current_M * 0.999)
        else:
            # Small gradient step
            T = max(0.3, 2.0 * (0.995 ** step))
            _, grad_a, _ = smoothed_max_and_grad(a, T)
            s_val = grad_a @ a
            grad_proj = a * (grad_a - s_val)
            eta = T / (n * 0.2)
            eta = min(eta, 0.1)
            a_new = a - eta * grad_proj
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
            a = a_new

        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  POL step={step}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization pipeline:
    1. Smoothed gradient descent for quick initial improvement
    2. LP fictitious play with adaptive M (main workhorse)
    3. Hybrid LP + gradient alternating
    4. Multi-start with diverse initializations
    5. Priority coordinate descent
    6. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = bimodal_init(n_init, edge_factor=4.0)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized bimodal: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Smoothed gradient descent ===
    print(f"\n=== Phase 1: Smoothed Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 1400
    if rem > 30:
        t_alloc = min(rem * 0.25, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_s, sc_s = smoothed_descent(best_seq, t_alloc, T0=30.0, T1=0.5, verbose=True)
        best_seq, best_score = check_and_update(a_s, "smooth", best_seq, best_score)

    # === Phase 2: LP adaptive M (main) ===
    print(f"\n=== Phase 2: LP Adaptive M (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 1000
    if rem > 50:
        t_alloc = min(rem * 0.4, 450.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play_adaptive(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_adapt", best_seq, best_score)

    # === Phase 3: Hybrid LP + gradient ===
    print(f"\n=== Phase 3: Hybrid (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 50:
        t_alloc = min(rem * 0.35, 350.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_h, sc_h = hybrid_lp_gradient(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_h, "hybrid", best_seq, best_score)

    # === Phase 4: Multi-start ===
    print(f"\n=== Phase 4: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    if rem > 100:
        t_alloc = min(rem * 0.4, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_ms, sc_ms = multi_start_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_ms, "multi", best_seq, best_score)

    # === Phase 5: Priority coordinate descent ===
    print(f"\n=== Phase 5: Priority CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = priority_cd(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd", best_seq, best_score)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 50
    if rem > 20:
        t_alloc = min(rem * 0.6, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin, sc_fin = final_polish(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_fin, "polish", best_seq, best_score)

    # === Phase 7: Try nearby lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2, n_parent + 4]
    for alt_n in nearby_ns:
        if alt_n < 64 or alt_n > 2000:
            continue
        rem = deadline - time.monotonic() - 40
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 80.0)
        if t_alloc < 15:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, sc_alt = lp_fictitious_play_adaptive(init_alt, t_alloc * 0.7, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
