from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def log_sum_exp(x, tau):
    """Compute tau * log(sum(exp(x/tau))) using the log-sum-exp trick."""
    scaled = x / tau
    m = scaled.max()
    return tau * (m + np.log(np.sum(np.exp(scaled - m))))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

"""
Smooth L-BFGS with log-sum-exp continuation for minimum peak self-convolution.
"""

from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_smooth_score_and_grad_x(x, n, tau):
    """
    Compute the smooth score and its gradient w.r.t. x.
    a[i] = exp(x[i]) / sum(exp(x[j])), so sum(a)=1, a>0.
    Score = 2n * log_sum_exp(conv(a,a), tau)
    """
    # Softmax to get a
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    a = exp_x / Z
    
    # Convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp
    scaled = conv / tau
    s_max = scaled.max()
    lse = tau * (s_max + np.log(np.sum(np.exp(scaled - s_max))))
    
    score = 2.0 * n * lse
    
    # Softmax weights
    w = np.exp(scaled - s_max)
    w /= w.sum()
    
    # Cross-correlation c[i] = sum_j w[i+j] * a[j]
    c = np.correlate(w, a, 'valid')
    
    # f_smooth = lse (since score = 2n * lse)
    # grad_x[j] = 4n * a[j] * (c[j] - lse)  (derived earlier)
    grad_x = 4.0 * n * a * (c - lse)
    
    return score, grad_x
