    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    from scipy.linalg import toeplitz
    return toeplitz(first_col, first_row)


def solve_lp_best_response(f, M, prev_g=None, time_budget=5.0):
    """Solve: max sum(g) s.t. conv(f,g) <= M (componentwise), g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None and len(prev_g) == n:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None),
                   time_limit=time_budget)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=(prev_g is not None))
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"step={current_step:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, eta=0.5, verbose=True):
    """
    Multiplicative weights / mirror descent on the constraint satisfaction problem.
    Update: g_k proportional to exp(-eta * max(0, conv(a,g)_k / M - 1))
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv_aa = fast_convolve(a)
        M = conv_aa.max()
        if M < 1e-15:
            break

        # Compute the "violation" for each coordinate using the current a
        # As a proxy for the LP solution, use: g_j proportional to
        # exp(-eta * sum_k max(0, conv(a, e_j)_k / M - 1))
        # conv(a, e_j) is just a shifted version of a
        # So for coordinate j: violation_j = max(0, max_k a_{k-j} / M - 1) (roughly)
        # More precisely: conv(a, e_j)_k = a_{k-j} (zero-padded)
        # So max_k conv(a,e_j)_k = max(a) (always the same for all j)
        # This doesn't differentiate... let me use a different approach.

        # Alternative: use the dual variable approach
        # y_k = exp(eta * conv(a,a)_k / M) normalized
        # Then g is the "best response" which satisfies C^T y ~ const
        # Approximate: g_j = 1 / (C^T y)_j where (C^T y)_j = sum_k C_{k,j} * y_k = sum_k a_{k-j} * y_k
        y = np.exp(eta * conv_aa / M)
        y = y / y.sum()

        # (C^T y)_j = sum_k a_{k-j} * y_k = cross-correlation of a and y at lag j
        conv_size = 2 * n - 1
        fft_size = 1 << (conv_size + n - 2).bit_length()
        Y = rfft(y, fft_size)
        A_flip = a[::-1]
        Af = rfft(A_flip, fft_size)
        cross = irfft(Y * Af, fft_size)[n - 1:2 * n - 1]
        cross = np.maximum(cross, 1e-10)

        g = 1.0 / cross
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Blend
        a_new = (1 - 0.3) * a + 0.3 * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MW iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_targeted(a0, time_limit, verbose=True):
    """
    Targeted coordinate descent: focus on reducing the lag(s) where
    the max autocorrelation occurs.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_lags = np.where(conv == conv.max())[0]
        # Pick a random max lag
        k_star = rng.choice(max_lags)

        # Try to reduce conv[k_star] by redistributing mass
        # conv[k_star] = sum_i a_i * a_{i+k_star}
        # To reduce it, we want to reduce the product at the "hot" spots
        # Find the index i where a_i * a_{i+k_star} is largest
        if k_star == 0:
            # Diagonal: sum a_i^2, reduce by equalizing
            idx = np.argmax(a)
            # Move mass from idx to a random low-value coordinate
            candidates = np.where(a < a.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            j = rng.choice(candidates)
            transfer = 0.1 * a[idx]
            a_trial = a.copy()
            a_trial[idx] -= transfer
            a_trial[j] += transfer
        else:
            # Off-diagonal: reduce a_i * a_{k_star - i} for the hot i
            i_range = max(0, n - k_star)
            if i_range == 0:
                continue
            products = a[:i_range] * a[k_star:k_star + i_range]
            hot_i = np.argmax(products)
            # Reduce either a[hot_i] or a[hot_i + k_star]
            if rng.random() < 0.5:
                idx = hot_i
            else:
                idx = hot_i + k_star
            candidates = np.where(a < a.mean() * 0.5)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            j = rng.choice(candidates)
            if j == idx:
                continue
            transfer = 0.05 * a[idx]
            a_trial = a.copy()
            a_trial[idx] -= transfer
            a_trial[j] += transfer

        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial / s_trial
        sc = compute_score(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        elif iters % 100 == 0:
            # Occasionally accept worse solutions for exploration
            if rng.random() < 0.1:
                a = a_trial

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD-T iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def structured_perturbation(a0, n_perturbations=5, scale=0.01):
    """Generate structured perturbations (smooth, oscillatory, etc)."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    rng = np.random.default_rng(42)
    candidates = []

    for i in range(n_perturbations):
        a_pert = a.copy()
        pert_type = i % 4
        if pert_type == 0:
            # Smooth Gaussian perturbation
            center = rng.integers(0, n)
            sigma = rng.uniform(n/20, n/5)
            x = np.arange(n)
            pert = scale * np.exp(-0.5 * ((x - center) / sigma) ** 2)
            a_pert += pert * rng.choice([-1, 1])
        elif pert_type == 1:
            # Sinusoidal perturbation
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            pert = scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_pert += pert
        elif pert_type == 2:
            # Block perturbation
            block_size = rng.integers(n//20, n//5)
            start = rng.integers(0, n - block_size)
            a_pert[start:start+block_size] += scale
            a_pert[(start+block_size)//2:(start+block_size)//2+block_size] -= scale
        else:
            # Random sparse perturbation
            idx = rng.choice(n, size=n//10, replace=False)
            a_pert[idx] += scale * rng.standard_normal(len(idx))
            a_pert = np.maximum(a_pert, 0)

        ss = a_pert.sum()
        if ss > 1e-10:
            a_pert = a_pert / ss
            candidates.append(a_pert)
    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization:
    1. LP fictitious play (arithmetic blend) - main driver
    2. Adam smoothed gradient optimization - fast progress
    3. LP fictitious play (geometric blend) - different geometry
    4. Targeted coordinate descent - precise adjustments
    5. Multiplicative weights - different divergence
    6. Multi-length exploration
    7. Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP Fictitious Play - arithmetic blend, large step
    print(f"\n=== Phase 1: LP-FP Arithmetic Step=0.5 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 350.0)
    if rem > 60:
        print(f"  Budget: {t1:.0f}s, n={n_parent}")
        a1, s1 = lp_fictitious_play(best_seq, t1, step=0.5, label="LP-arith-0.5",
                                     geometric=False, lp_time_budget=4.0)
        best_seq, best_score = check_and_update(a1, "lp_arith_0.5", best_seq, best_score)

    # Phase 2: Adam smoothed gradient - fast per-iteration progress
    print(f"\n=== Phase 2: Adam LSE (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = adam_smoothed_optimize(best_seq, t2,
                                         T_values=[100, 50, 25, 15, 10, 7, 5, 3, 2, 1.5, 1.2, 1.0],
                                         lr=0.003)
        best_seq, best_score = check_and_update(a2, "adam_lse", best_seq, best_score)

    # Phase 3: LP Fictitious Play - geometric blend
    print(f"\n=== Phase 3: LP-FP Geometric Step=0.4 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 250.0)
    if rem > 60:
        print(f"  Budget: {t3:.0f}s")
        a3, s3 = lp_fictitious_play(best_seq, t3, step=0.4, label="LP-geom-0.4",
                                     geometric=True, lp_time_budget=4.0)
        best_seq, best_score = check_and_update(a3, "lp_geom_0.4", best_seq, best_score)

    # Phase 4: Targeted coordinate descent
    print(f"\n=== Phase 4: Targeted CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if rem > 40:
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = coordinate_descent_targeted(best_seq, t4)
        best_seq, best_score = check_and_update(a4, "cd_targ", best_seq, best_score)

    # Phase 5: Multiplicative weights
    print(f"\n=== Phase 5: MultiWeights (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if rem > 40:
        print(f"  Budget: {t5:.0f}s")
        a5, s5 = multiplicative_weights(best_seq, t5, eta=1.0)
        best_seq, best_score = check_and_update(a5, "multi_w", best_seq, best_score)

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = [n_parent + 4, n_parent - 4, n_parent + 8, n_parent + 16,
                   n_parent - 8, 1024, 1536]
    alt_lengths = [l for l in alt_lengths if 128 <= l <= 2048 and l != n_parent]
    alt_lengths = list(dict.fromkeys(alt_lengths))[:4]  # unique, max 4

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t6 = min(rem * 0.15, 100.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Adam LSE n={alt_n}, budget={t6:.0f}s")
        a6, s6 = adam_smoothed_optimize(init_alt, t6,
                                         T_values=[50, 25, 10, 5, 3, 2, 1.5, 1.0],
                                         lr=0.003)
        best_seq, best_score = check_and_update(a6, f"n{alt_n}", best_seq, best_score)

    # Phase 7: Final LP polish on best
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.25, 120.0)
    if rem > 30:
        print(f"  Budget: {t7:.0f}s")
        a7, s7 = lp_fictitious_play(best_seq, t7, step=0.2, label="LP-polish",
                                     geometric=False, lp_time_budget=5.0)
        best_seq, best_score = check_and_update(a7, "lp_polish", best_seq, best_score)

    # Phase 8: Short Adam polish
    print(f"\n=== Phase 8: Final Adam Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.3, 60.0)
    if rem > 20:
        print(f"  Budget: {t8:.0f}s")
        a8, s8 = adam_smoothed_optimize(best_seq, t8,
                                         T_values=[5, 3, 2, 1.5, 1.2, 1.0, 0.8],
                                         lr=0.001)
        best_seq, best_score = check_and_update(a8, "adam_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
