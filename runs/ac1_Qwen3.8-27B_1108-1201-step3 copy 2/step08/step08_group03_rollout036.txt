Let me think about this more carefully. If we move from $f$ to $(1-t)f + tg$, then:
$((1-t)f + tg) * ((1-t)f + tg) = (1-t)^2 f*f + t(1-t)(f*g + g*f) + t^2 g*g$
$= (1-t)^2 f*f + 2t(1-t) f*g + t^2 g*g$

We want this to have max < $M = \max(f*f)$ for small $t$. For small $t$, the dominant change is:
$\Delta \approx 2t(f*g - f*f)$ (ignoring $t^2$ terms)

So we want $f*g \leq f*f$ element-wise, which is stricter than $f*g \leq M$. The LP constraint $f*g \leq M$ is a relaxation.

A better approach would be to use the actual subgradient of $F$ at $f$. The subgradient is:
$\partial F(f) = \{2 \sum_{k \in S} w_k P_k f : w_k \geq 0, \sum w_k = 1, S = \{k : (f*f)_k = M\}\}$

where $P_k f$ is the vector such that $(P_k f)_j = f_{k-j}$ (with appropriate boundary handling).

The Frank-Wolfe step would be:
1. Compute a subgradient $g = 2 \sum_{k \in S} w_k P_k f$.
2. Find $v = \arg\min_{v \geq 0, \sum v = 1} g^T v = e_{\arg\min_j g_j}$.
3. Update $f \leftarrow (1-t)f + tv$.

But this is very slow (it only changes one coordinate at a time).

A better approach is to use the "pairwise" Frank-Wolfe or to use a smoothing technique.

**Smoothing approach:**
Replace $F(b) = \max_k L_k(b)$ with $F_\epsilon(b) = \epsilon \log \sum_k e^{L_k(b)/\epsilon}$ where $L_k(b) = (b*b)_k$.

Then $F_\epsilon$ is smooth and we can use gradient descent with projection onto the simplex.

$\nabla_b F_\epsilon(b) = 2 \sum_k p_k P_k b$ where $p_k = e^{L_k(b)/\epsilon} / \sum_j e^{L_j(b)/\epsilon}$.

This is essentially the softmax-weighted subgradient! And it's smooth, so we can use accelerated gradient methods.

**Algorithm: Smoothed Frank-Wolfe / Mirror Descent**

1. Initialize $b$ (e.g., uniform or from parent).
2. For each iteration:
   a. Compute $c = b*b$, $M = \max c$.
   b. Compute softmax weights: $p_k \propto e^{c_k / \epsilon}$.
   c. Compute gradient: $g = 2 \sum_k p_k P_k b$.
   d. Take a step: $b^+ = b - \eta g$.
   e. Project to simplex: $b^+ = \Pi_{\Delta}(b^+)$.
   f. Adapt $\epsilon$ and $\eta$.
3. Return best $b$.

This is essentially what the "subgradient" phase in the previous code does, but with a smoother approximation (log-sum-exp instead of just top-K).

def frequency_init(n):
    """Initialize using frequency-domain square root."""
    # Target auto-convolution: uniform of length 2n-1
    c_target = np.ones(2*n-1) / (2*n-1)
    
    # FFT size
    fft_size = 1 << (4*n - 1).bit_length()  # oversample to avoid aliasing
    
    # Compute target spectrum
    C = np.fft.rfft(c_target, fft_size)
    
    # Take square root (magnitude)
    # B(omega)^2 = C(omega), so B(omega) = sqrt(C(omega))
    # We need to choose the phase carefully
    magnitude = np.sqrt(np.abs(C) + 1e-30)
    phase = np.angle(C) / 2  # half the phase
    B_target = magnitude * np.exp(1j * phase)
    
    # Inverse FFT
    b = np.fft.irfft(B_target, fft_size)[:n]
    
    # Project to non-negative simplex
    b = np.maximum(b, 0)
    s = b.sum()
    if s > 1e-10:
        b /= s
    else:
        b = np.ones(n) / n
    
    return b

"""
Optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based direction finding with HiGHS solver (scipy linprog)
- Cutting-plane constraint management
- Log-sum-exp smoothed gradient as secondary method
- Frequency-domain square-root initialization
- Multi-start with diverse initializations
- Coordinate descent polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, w):
    """
    Compute gradient of sum_k w_k * (f*f)_k with respect to f.
    Result: grad_j = 2 * sum_k w_k * f_{k-j}
    """
    n = len(f)
    N = 1 << (3 * n).bit_length()
    w_flip = w[::-1]
    Wf = rfft(w_flip, N)
    F = rfft(f, N)
    conv = irfft(Wf * F, N)[:3 * n]
    # grad_j = 2 * conv[2n-2-j] for j=0..n-1
    g = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * g


def solve_lp_efficient(f, n, M, indices):
    """
    Solve: max sum(g) s.t. g >= 0, A@g <= M
    using scipy linprog with HiGHS.
    """
    K = len(indices)
    if K == 0:
        return None

    # Build constraint matrix A (K x n)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': max(1.0, (time.monotonic() and 5.0))})
    if res.status not in (0, 3):  # 0=optimal, 3=optimal within tolerance
        return None
    g = np.maximum(res.x, 0.0)
    if np.any(np.isnan(g)):
        return None
    return g


def lp_optimize(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """
    LP-based optimization with cutting planes.
    Minimize max(f*f) over the simplex.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_M = M

    # Initialize active set with top peaks
    k_init = min(initial_k, len(b))
    active_idx = np.sort(np.argpartition(b, -k_init)[-k_init:])

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Solve LP
        g = solve_lp_efficient(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 8:
                break
            # Shrink active set
            if len(active_idx) > initial_k // 2:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S = g.sum()
        if S <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            # Try expanding
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                k_exp = min(max_k, len(b_cur))
                new_top = np.argpartition(b_cur, -k_exp)[-k_exp:]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
                active_idx = np.sort(active_idx)
            continue

        # Cutting plane: find violated constraints
        fg = cross_convolve(f, g / S)
        tol = M * 1e-8
        violated = fg > M + tol
        v_count = violated.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            mask = ~np.isin(v_idx, active_idx)
            new_v = v_idx[mask]
            if len(new_v) > 0:
                n_add = min(25, len(new_v))
                to_add = new_v[np.argsort(fg[new_v])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                active_idx = np.sort(active_idx)
                # Re-solve
                g = solve_lp_efficient(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S = g.sum()
                if S <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 10:
                        break
                    continue

        stuck = 0
        h = g / S

        # Line search: coarse + fine
        best_t = 0.1
        best_t_obj = M
        for t in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            ft = (1.0 - t) * f + t * h
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-10:
                continue
            ft /= s
            obj = fast_convolve(ft).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                ft1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                ft1 /= ft1.sum()
                ft2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                ft2 /= ft2.sum()
                o1 = fast_convolve(ft1).max()
                o2 = fast_convolve(ft2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            ft_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            ft_mid /= ft_mid.sum()
            o_mid = fast_convolve(ft_mid).max()
            if o_mid < best_t_obj:
                best_t_obj = o_mid
                best_t = t_mid

        # Update
        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        f /= f.sum()
        cur_M = fast_convolve(f).max()
        b = fast_convolve(f)
        M = cur_M

        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f.copy()

        # Periodically refresh active set
        if it % 15 == 0 and len(active_idx) > initial_k:
            # Keep top constraints by current b values
            keep_k = min(initial_k * 2, len(active_idx))
            top_active = active_idx[np.argsort(b[active_idx])[-keep_k:]]
            new_top = np.argpartition(b, -initial_k)[-initial_k:]
            active_idx = np.unique(np.concatenate([top_active, new_top]))
            active_idx = np.sort(active_idx)

        if it % 100 == 0:
            print(f"    {label} it={it} M={best_M:.12f} score={2.0*n*best_M:.8f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_M


def smoothed_gradient(f_init, time_limit, eps_init=0.005, label="sg"):
    """
    Smoothed gradient method using log-sum-exp approximation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    c = fast_convolve(f)
    best_M = c.max()
    eps = eps_init
    lr = 0.003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        c = fast_convolve(f)
        # Softmax weights
        scaled = c / eps
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient via FFT-based cross-correlation
        g = compute_gradient(f, w)

        # Gradient step with projection
        f_new = project_simplex(f - lr * g)

        # Check improvement
        obj = fast_convolve(f_new).max()
        f = f_new
        if obj < best_M - 1e-13:
            best_M = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.92
                eps *= 1.05
            if no_improve > 50:
                # Restart near best
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                lr = 0.003
                eps = eps_init
                no_improve = 0

        if it % 200 == 0:
            print(f"    {label} it={it} M={best_M:.12f} score={2.0*n*best_M:.8f} "
                  f"eps={eps:.5f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_M


def frequency_init(n):
    """Initialize using frequency-domain square root of flat auto-convolution."""
    c_target = np.ones(2 * n - 1) / (2 * n - 1)
    fft_size = 1 << (4 * n).bit_length()
    C = np.fft.rfft(c_target, fft_size)
    magnitude = np.sqrt(np.abs(C) + 1e-30)
    phase = np.angle(C) / 2.0
    B_target = magnitude * np.exp(1j * phase)
    b = np.fft.irfft(B_target, fft_size)[:n]
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 1e-10:
        b /= s
    else:
        b = np.ones(n) / n
    return b


def random_init(n, rng):
    """Random Dirichlet initialization."""
    alpha = 1.0
    b = rng.dirichlet(np.full(n, alpha))
    return b


def structured_init(n, rng):
    """Structured initialization using cosine basis."""
    b = np.ones(n)
    n_harmonics = rng.integers(1, 5)
    for _ in range(n_harmonics):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        b += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 1e-10:
        b /= s
    else:
        b = np.ones(n) / n
    return b


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    rng = np.random.default_rng(77)

            break
        improved = False
        # Randomize coordinate order
        order = np.arange(0, n, step)
        rng.shuffle(order)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = f[j] * s  # save the actual value before normalization
                    # Actually we need to save the un-normalized value
            # Restore best
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break

    return f, best_obj

        # Check violations
        fg = cross_convolve(f, g / S)
        violated = fg > M * (1.0 + 1e-9)
        if violated.any():
            v_idx = np.where(violated)[0]
            new_mask = ~np.isin(v_idx, active_idx)
            new_v = v_idx[new_mask]
            if len(new_v) > 0 and len(active_idx) < max_k:
                n_add = min(30, len(new_v))
                to_add = new_v[np.argsort(fg[new_v])[-n_add:]]
                active_idx = np.sort(np.unique(np.concatenate([active_idx, to_add])))
                g = solve_lp_efficient(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S = g.sum()
                if S <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 10:
                        break
                    continue

        stuck = 0
        h = g / S

        # Line search
        best_t = 0.1
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            ft = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = ft.sum()
            if s < 1e-10:
                continue
            ft /= s
            obj = fast_convolve(ft).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        f /= f.sum()
        cur_M = fast_convolve(f).max()
        b = fast_convolve(f)
        M = cur_M

        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f.copy()

        # Refresh active set periodically
        if it - last_refresh > 20:
            last_refresh = it
            if len(active_idx) > initial_k * 2:
                top_active = active_idx[np.argsort(b[active_idx])[-initial_k:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.sort(np.unique(np.concatenate([top_active, new_top])))

        if it % 200 == 0:
            print(f"    {label} it={it} M={best_M:.12f} score={2.0*n*best_M:.8f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining LP, smoothed gradient,
    frequency-domain init, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, budget_s - 40.0)
    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default uniform: n={len(parent)}, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    def try_update(f, label):
        nonlocal best_seq, best_score
        if len(f) == 0:
            return
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            return
        f = f / s
        sc = float(evaluate_sequence(f.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(f)})")

    # === Phase 1: LP optimization from parent (main) ===
    print(f"\n=== Phase 1: LP from parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = lp_optimize(parent, time_limit=t1, initial_k=50, max_k=350, label="p1")
        try_update(f1, "phase1")

    # === Phase 2: LP optimization from frequency-domain init ===
    print(f"\n=== Phase 2: Frequency init + LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 40:
        n_freq = n_parent
        f_freq = frequency_init(n_freq)
        print(f"  Frequency init n={n_freq}, budget={t2:.0f}s")
        f2, _ = lp_optimize(f_freq, time_limit=t2, initial_k=40, max_k=300, label="freq")
        try_update(f2, "freq")

    # === Phase 3: LP from structured random init ===
    print(f"\n=== Phase 3: Structured init + LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        n_str = n_parent
        f_str = structured_init(n_str, rng)
        print(f"  Structured init n={n_str}, budget={t3:.0f}s")
        f3, _ = lp_optimize(f_str, time_limit=t3, initial_k=40, max_k=300, label="struct")
        try_update(f3, "struct")

    # === Phase 4: Smoothed gradient on best ===
    print(f"\n=== Phase 4: Smoothed gradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 30:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t4:.0f}s")
        f4, _ = smoothed_gradient(best_seq, time_limit=t4, eps_init=0.003, label="sg4")
        try_update(f4, "sg4")

    # === Phase 5: Deep LP on best (more constraints) ===
    print(f"\n=== Phase 5: Deep LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 350.0)
    if t5 > 30:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t5:.0f}s, k=80->500")
        f5, _ = lp_optimize_v2(best_seq, time_limit=t5, initial_k=80, max_k=500, label="deep")
        try_update(f5, "deep")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Length variation (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    for factor in [0.9, 1.1]:
        if rem < 40:
            break
        tl = int(n_base * factor)
        tl = max(256, min(3000, tl))
        t6 = min(rem * 0.12, 120.0)
        if t6 < 20:
            break
        old_x = np.linspace(0.0, 1.0, len(best_seq))
        new_x = np.linspace(0.0, 1.0, tl)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(tl) / tl
        print(f"  n={tl}, budget={t6:.0f}s")
        f6, _ = lp_optimize(f_interp, time_limit=t6, initial_k=40, max_k=250, label=f"n{tl}")
        try_update(f6, f"n{tl}")
        rem = deadline - time.monotonic()

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coord descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.15, 120.0)
    if t7 > 15:
        print(f"  n={len(best_seq)}, budget={t7:.0f}s")
        f7, _ = coordinate_polish(best_seq, time_limit=t7)
        try_update(f7, "cd")

    # === Phase 8: Final LP burst ===
    print(f"\n=== Phase 8: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.4, 200.0)
    if t8 > 15:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t8:.0f}s")
        f8, _ = lp_optimize_v2(best_seq, time_limit=t8, initial_k=100, max_k=600, label="final")
        try_update(f8, "final")

    # === Phase 9: One more smoothed gradient pass ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.5, 100.0)
        print(f"\n=== Phase 9: Final SG (best={best_score:.8f}) ===")
        f9, _ = smoothed_gradient(best_seq, time_limit=t9, eps_init=0.002, label="sg9")
        try_update(f9, "sg9")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
