"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def mwu_optimize(a_init, time_limit, verbose=True):
    """
    Multiplicative Weight Updates (primal-dual subgradient method).
    Uses the cross-correlation structure for O(n log n) gradient computation.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Adaptive parameters
    temperature = 30.0  # controls sharpness of lambda
    eta_base = 0.3
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        A = rfft(a, fft_size)
        conv = irfft(A * A, fft_size)[:size]
        max_conv = conv.max()
        score = 2 * n * max_conv
        
        # Update dual variable (lambda)
        # Emphasize positions near the peak
        normalized_conv = conv / max_conv
        lambda_w = np.exp(temperature * (normalized_conv - 1.0))
        lambda_sum = lambda_w.sum()
        if lambda_sum < 1e-30:
            break
        lambda_w = lambda_w / lambda_sum
        
        # Compute gradient: g[i] = 2 * sum_k lambda[k] * a[k-i]
        # This is 2 * cross_correlation(lambda, a)
        L = rfft(lambda_w, fft_size)
        g = 2.0 * irfft(L * np.conj(A), fft_size)[:n]
        
        # Ensure g is real (numerical precision)
        g = np.real(g)
        
        # Adaptive step size: cap the maximum log-change
        g_max = np.abs(g).max()
        if g_max < 1e-15:
            break
        eta = eta_base / g_max
        
        # Multiplicative update
        log_change = -eta * g
        # Clip to prevent extreme changes
        log_change = np.clip(log_change, -2.0, 2.0)
        a = a * np.exp(log_change)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = best_a.copy()
            break
        a = a / a_sum
        
        # Check improvement
        new_score = compute_score(a)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
        
        # Occasionally try a different temperature
        if t % 200 == 199:
            temperature *= 1.1  # gradually increase sharpness
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MWU iter={t}, score={new_score:.8f}, best={best_score:.8f}, "
                  f"temp={temperature:.1f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, seed=42):
    """
    Coordinate descent with golden section search.
    Smart pair selection based on convolution peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Compute current convolution for pair selection
        conv = fast_convolve(a)
        n_pairs = min(n, 300)
        
        # Collect candidate pairs
        pairs = []
        
        # Peak-focused: positions contributing to top peaks
        top_k = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for _ in range(n_pairs // 4):
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Edge pairs
        edge = max(2, n // 20)
        for _ in range(n_pairs // 4):
            i = int(rng.integers(0, edge))
            j = int(rng.integers(n - edge, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Process pairs
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Golden section search for optimal transfer
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_transfer(t_val):
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score(a_trial)
            
            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Perturb
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def peak_shaving(a_init, time_limit, verbose=True, seed=123):
    """
    Targeted peak reduction: identify the hottest convolution positions
    and systematically reduce contributions from the top elements.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for iters in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        # Find top positions
        n_peaks = min(5, len(conv))
        peak_pos = np.argsort(conv)[::-1][:n_peaks]
        
        improved_this_iter = False
        
        for pk in peak_pos:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            
            # Contributing indices
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            
            # Contribution of each pair (j, pk-j)
            contribs = a[idx] * a[pk - idx]
            total_contrib = contribs.sum()
            
            if total_contrib < 1e-15:
                continue
            
            # Find top contributors
            top_c = min(5, len(idx))
            top_cc = np.argsort(contribs)[::-1][:top_c]
            
            for ci in top_cc:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                
                # Try reducing a[i] by small amounts
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-15:
                        continue
                    
                    # Redistribute delta proportionally to other elements
                    # that DON'T contribute to this peak
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Also exclude elements that contribute to this peak
                    for k_idx in idx:
                        other = pk - k_idx
                        if 0 <= other < n:
                            mask[other] = False
                    
                    if mask.sum() == 0:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                    
                    alpha = a[mask].sum()
                    if alpha < 1e-15:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[mask] += delta * (a_trial[mask] / alpha)
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved_this_iter = True
                        break
                if improved_this_iter:
                    break
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PS iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def simulated_annealing(a_init, time_limit, verbose=True, seed=999):
    """
    Simulated annealing for final polish.
    Uses very small steps and only accepts improvements (or near-improvements).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T = 1e-6  # very small temperature (almost greedy)
    T_min = 1e-10
    cooling = 0.999
    
    for iters in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_trial = a.copy()
        
        # Random move: pick 2-3 positions and redistribute
        n_move = rng.integers(2, 4)
        indices = rng.integers(0, n, n_move)
        
        # Get current values at these positions
        vals = a_trial[indices].copy()
        total = vals.sum()
        if total < 1e-15:
            continue
        
        # Perturb proportionally
        noise = rng.dirichlet(np.ones(n_move) * 50)  # nearly uniform perturbation
        new_vals = total * noise
        a_trial[indices] = new_vals
        
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        
        sc = compute_score(a_trial)
        delta = sc - best_score
        
        # Accept if improvement, or with small probability if near-improvement
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = a_trial
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        T = max(T * cooling, T_min)
        
        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA iter={iters}, score={best_score:.8f}, T={T:.2e}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Weight Updates (MWU) with backtracking line search

3. Peak shaving - targeted reduction of convolution peaks
4. Simulated annealing for final polish

6. Multi-round refinement combining all methods

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def cross_correlation_real(lambda_w, a, fft_size):
    """Compute 2 * sum_k lambda[k] * a[k-i] for i in [0, n-1] via FFT."""
    n = len(a)
    L = rfft(lambda_w, fft_size)
    A = rfft(a, fft_size)
    result = 2.0 * irfft(L * np.conj(A), fft_size)[:n]
    return np.real(result)


def mwu_step(a, fft_size, temperature, eta_max):
    """
    Single MWU step with backtracking.
    Returns updated a and new score.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Compute convolution
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    max_conv = conv.max()
    
    if max_conv < 1e-30:
        return a, 2.0 * n * max_conv
    
    # Compute dual variable
    norm_conv = conv / max_conv
    lambda_w = np.exp(temperature * (norm_conv - 1.0))
    lambda_sum = lambda_w.sum()
    if lambda_sum < 1e-30:
        return a, 2.0 * n * max_conv
    lambda_w = lambda_w / lambda_sum
    
    # Compute gradient via cross-correlation
    g = cross_correlation_real(lambda_w, a, fft_size)
    
    # Backtracking line search
    base_score = 2.0 * n * max_conv
    eta = eta_max
    best_trial = a
    best_trial_score = base_score
    
    for bt in range(20):
        if eta < 1e-10:
            break
        log_change = -eta * g
        log_change = np.clip(log_change, -3.0, 3.0)
        a_trial = a * np.exp(log_change)
        a_trial = np.maximum(a_trial, 1e-30)
        s = a_trial.sum()
        if s < 1e-15:
            eta /= 2.0
            continue
        a_trial = a_trial / s
        
        trial_score = compute_score(a_trial)
        if trial_score < best_trial_score:
            best_trial_score = trial_score
            best_trial = a_trial
        if trial_score < base_score * (1.0 - 1e-10):
            break
        eta /= 2.0
    
    return best_trial, best_trial_score

def eval_golden(a, i, j, t_val):
    """Evaluate score after transferring t_val from a[i] to a[j]."""
    a_trial = a.copy()
    a_trial[i] -= t_val
    a_trial[j] += t_val
    if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
        return 1e10
    a_trial = np.maximum(a_trial, 0)
    ss = a_trial.sum()
    if ss < 1e-15:
        return 1e10
    a_trial = a_trial / ss
    return compute_score(a_trial)

            je = min(n - 1, pk)
            count = n_pairs // (4 * top_k) + 1
            for _ in range(count):
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(n_pairs // 2):

                    f1 = f2

        T = max(T * cooling, T_min)
        
        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA iter={iters}, score={best_score:.8f}, T={T:.2e}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def subgradient_descent(a_init, time_limit, verbose=True, K=3):
    """
    Projected subgradient descent with Polyak step size.
    Uses top-K subgradient for a smoother direction.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Lower bound estimate (for Polyak step size)
    lb = 2.0 * n / (2 * n - 1)  # theoretical minimum (unachievable with a>=0)
    
    for t in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        
        # Find top-K positions
        top_k = min(K, len(conv))
        peak_idx = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient (average of gradients at top-K)
        g = np.zeros(n)
        for pk in peak_idx:
            pk = int(pk)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= top_k
        
        # Polyak step size
        score_gap = cur_score - lb
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        eta = score_gap / (g_norm_sq * (t + 1))
        eta = min(eta, 1.0)  # cap step size
        eta = max(eta, 1e-8)
        
        # Update
        a_trial = a - eta * g
        a_trial = np.maximum(a_trial, 0)
        s2 = a_trial.sum()
        if s2 < 1e-15:
            continue
        a_trial = a_trial / s2
        
        new_score = compute_score(a_trial)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_trial.copy()
            a = a_trial
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={t}, cur={cur_score:.8f}, best={best_score:.8f}, "
                  f"eta={eta:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    if a is None:
        return best_seq, best_score
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. MWU (Multiplicative Weight Updates) with backtracking
    2. Subgradient descent with Polyak step size
    3. Coordinate descent with golden section
    4. Peak shaving
    5. Simulated annealing
    6. Length exploration
    Multiple rounds of refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: MWU (main optimization) ===
    print(f"\n=== Phase 1: MWU (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(350.0, rem * 0.25)
    print(f"  Budget: {t_alloc:.0f}s")
    a1, s1 = mwu_optimize(best_seq, t_alloc, verbose=True, temp_init=20.0, eta_init=1.0)
    best_seq, best_score = check_and_update(a1, "mwu1", best_seq, best_score)
    
    # === Phase 2: Subgradient descent ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(200.0, rem * 0.15)
        print(f"  Budget: {t_alloc:.0f}s")
        a2, s2 = subgradient_descent(best_seq, t_alloc, verbose=True, K=3)
        best_seq, best_score = check_and_update(a2, "sgd", best_seq, best_score)
    
    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(250.0, rem * 0.2)
        print(f"  Budget: {t_alloc:.0f}s")
        a3, s3 = coordinate_descent(best_seq, t_alloc, verbose=True, seed=42)
        best_seq, best_score = check_and_update(a3, "cd1", best_seq, best_score)
    
    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(150.0, rem * 0.15)
        print(f"  Budget: {t_alloc:.0f}s")
        a4, s4 = peak_shaving(best_seq, t_alloc, verbose=True, seed=123)
        best_seq, best_score = check_and_update(a4, "peak", best_seq, best_score)
    
    # === Phase 5: MWU round 2 (on improved solution) ===
    print(f"\n=== Phase 5: MWU Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(180.0, rem * 0.15)
        print(f"  Budget: {t_alloc:.0f}s")
        a5, s5 = mwu_optimize(best_seq, t_alloc, verbose=True, temp_init=30.0, eta_init=0.5)
        best_seq, best_score = check_and_update(a5, "mwu2", best_seq, best_score)
    
    # === Phase 6: Coordinate Descent round 2 ===

    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(150.0, rem * 0.15)
        print(f"  Budget: {t_alloc:.0f}s")
        a6, s6 = coordinate_descent(best_seq, t_alloc, verbose=True, seed=77)
        best_seq, best_score = check_and_update(a6, "cd2", best_seq, best_score)
    
    # === Phase 7: Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    candidate_lengths = sorted(set([
        n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
        n_parent + 8, n_parent - 8, n_parent + 16, n_parent - 16
    ]))
    for alt_n in candidate_lengths:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(50.0, rem * 0.12)
        if t_alloc < 15:
            break

        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_mwu", best_seq, best_score)

    # === Phase 8: Simulated Annealing ===
    print(f"\n=== Phase 8: Simulated Annealing (best={best_score:.8f}) ===")

        t_alloc = min(rem * 0.35, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a8, s8 = simulated_annealing(best_seq, t_alloc, verbose=True, seed=999)
        best_seq, best_score = check_and_update(a8, "sa", best_seq, best_score)
    
    # === Phase 9: Final MWU polish ===
    print(f"\n=== Phase 9: Final MWU (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a9, s9 = mwu_optimize(best_seq, t_alloc, verbose=False, temp_init=40.0, eta_init=0.3)
        best_seq, best_score = check_and_update(a9, "mwu_final", best_seq, best_score)
    
    # === Phase 10: Final CD ===
    print(f"\n=== Phase 10: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.6, 40.0)
        a10, s10 = coordinate_descent(best_seq, t_alloc, verbose=False, seed=314)
        best_seq, best_score = check_and_update(a10, "cd_final", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
