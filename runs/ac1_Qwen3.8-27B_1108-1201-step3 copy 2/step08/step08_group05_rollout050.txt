"""
Optimized auto-convolution peak minimization using smoothed peak gradient descent
with adaptive temperature annealing, peak shaving heuristics, and LP-based polish.
Targets score < 1.5027 for the non-negative step function auto-convolution problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_peak_gradient(a, T):
    """
    Compute gradient of T*log(sum_k exp(b[k]/T)) where b = conv(a,a).
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    where w = softmax(b/T)
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Softmax weights (numerically stable)
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is cross-correlation of w and a
    # Using FFT: IFFT(FFT(w, N) * conj(FFT(a, N))) at indices 0..n-1
    # Need N >= len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    needed = 3 * n - 2
    N = 1 << (needed - 1).bit_length()
    
    W = rfft(w, N)
    A = rfft(a, N)
    # Cross-correlation: IFFT(FFT(w) * conj(FFT(a)))
    # For real signals: conj(FFT(a, N)) corresponds to circular cross-correlation
    corr = irfft(W * np.conj(A), N)
    # corr[j] = sum_k w[k] * a[(k-j) mod N]
    # For j in [0, n-1] and with proper zero-padding, this gives:
    # corr[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    grad = 2.0 * corr[:n]
    
    return grad


def smooth_peak_descent(a_init, time_limit, T_init=0.05, T_min=0.0005, label="smooth"):
    """
    Gradient descent with Adam on the smoothed peak objective.
    T is annealed from T_init to T_min.

    t0 = time.monotonic()
    n = len(a_init)
    f = a_init.copy()
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    u = np.zeros(n)
    t_step = 0
    
    T = T_init
    total_steps_planned = max(1000, int(time_limit * 200))  # estimate
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        grad = smoothed_peak_gradient(f, T)
        
        # Adam update
        t_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        u = beta2 * u + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_step)
        u_hat = u / (1.0 - beta2 ** t_step)
        step = lr * m_hat / (np.sqrt(u_hat) + eps)
        
        f_new = f - step
        f_new = project_simplex(f_new)
        
        new_peak = fast_convolve(f_new).max()
        f = f_new
        
        if new_peak < best_obj - 1e-12:
            best_obj = new_peak
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal T based on progress
        if no_improve == 0:
            # Making progress: decrease T (sharper)
            T = max(T * 0.998, T_min)
        elif no_improve > 200:
            # Stuck: increase T (smoother) and reduce lr
            T = min(T * 1.05, 0.1)
            lr = max(lr * 0.9, 0.0001)
            no_improve = 0
            #偶尔 restart from best with noise
            if no_improve > 0 and (it % 500 == 0):
                f = project_simplex(best_f + np.random.randn(n) * 0.005)
                m = np.zeros(n)
                u = np.zeros(n)
        
        # Periodic reporting

            print(f"    {label} iter={it} peak={best_obj:.10f} score={score:.6f} T={T:.6f} t={elapsed:.0f}s")

def peak_shaving(f, time_limit, n_peaks=3, label="shave"):
    """
    Direct peak shaving: find peaks, reduce contributing elements.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        peak = b.max()
        peak_idx = b.argmax()
        
        # Find top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))
        
        # For each peak, find the contributing pairs and reduce them
        reductions = np.zeros(n)
        for k in top_idx:
            weight = b[k] / (b[top_idx].max() + 1e-30)
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    # a[j] * a[m] contributes to b[k]
                    # Reduce the larger of the two
                    larger = max(j, m)
                    smaller = min(j, m)
                    reductions[larger] += weight * a[smaller]
        
        # Normalize reductions
        red_max = reductions.max()
        if red_max < 1e-15:
            break
        reductions = reductions / red_max
        
        # Apply reduction
        for alpha in [0.01, 0.005, 0.002, 0.001]:
            f_new = f - alpha * reductions
            f_new = project_simplex(f_new)
            new_peak = fast_convolve(f_new).max()
            if new_peak < peak:
                f = f_new
                if new_peak < best_obj:
                    best_obj = new_peak
                    best_f = f.copy()
                break
    
    return best_f, best_obj

    n = len(a)
    b = fast_convolve(a)
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w = w / w_sum
    needed = 3 * n - 2
    N = 1 << (needed - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    corr = irfft(W * np.conj(A), N)
    grad = 2.0 * corr[:n]
    return grad


def smooth_peak_descent(a_init, time_limit, T_init=0.05, T_min=0.0005, label="smooth",
                        lr_init=0.05, use_adam=True):
    """
    Adam gradient descent on smoothed peak objective with T annealing.
    """
    t0 = time.monotonic()
    n = len(a_init)
    f = a_init.copy()
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    u = np.zeros(n)
    t_step = 0
    T = T_init
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    grad_clip = 10.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = smoothed_peak_grad(f, T)
        # Clip gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > grad_clip:
            grad = grad * (grad_clip / gnorm)
        if use_adam:
            t_step += 1
            m = beta1 * m + (1.0 - beta1) * grad
            u = beta2 * u + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_step)
            u_hat = u / (1.0 - beta2 ** t_step)
            step = lr * m_hat / (np.sqrt(u_hat) + eps)
        else:
            step = lr * grad
        f_new = f - step
        f_new = project_simplex(f_new)
        new_peak = fast_convolve(f_new).max()
        f = f_new
        if new_peak < best_obj - 1e-14:
            best_obj = new_peak
            best_f = f.copy()
            no_improve = 0
            # Slightly decrease T when making progress
            T = max(T * 0.9995, T_min)
        else:
            no_improve += 1
        if no_improve > 500:
            # Stuck: adapt
            if no_improve % 200 == 0:
                T = min(T * 1.03, 0.15)
                lr = max(lr * 0.95, 0.001)
            if no_improve > 1500 and rng.random() < 0.1:
                # Restart with noise
                noise_scale = 0.002 * no_improve / 1500.0
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                m = np.zeros(n)
                u = np.zeros(n)
                t_step = 0
                no_improve = 0
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} peak={best_obj:.10f} score={score:.8f} "
                  f"T={T:.6f} lr={lr:.6f} t={elapsed:.0f}s")
    return best_f, best_obj


def spectral_init(n, n_iters=200, rng=None):
    """
    Gerchberg-Saxton style spectral initialization for flat auto-convolution.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    a = rng.random(n)
    a = project_simplex(a)
    for _ in range(n_iters):
        A = np.fft.fft(a)
        mag = np.abs(A)
        # Clip top 5% of magnitudes to flatten spectrum
        threshold = np.percentile(mag, 80)
        mag_clipped = np.minimum(mag, threshold * 1.1)
        phase = np.angle(A)
        A_new = mag_clipped * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
    return a


def peak_shave_descent(f_init, time_limit, label="shave"):
    """
    Iterative peak shaving: directly reduce elements contributing to peak.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    it = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        peak_val = b.max()
        peak_idx = int(b.argmax())
        # Find top 3 peaks
        if 3 < len(b):
            top_idx = np.argpartition(b, -3)[-3:]
        else:
            top_idx = np.arange(len(b))
        top_idx = sorted(top_idx)
        # Compute reduction direction
        reductions = np.zeros(n)
        for k in top_idx:
            w_k = b[k] / (b[top_idx[0]] + 1e-30)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                m2 = k - j
                # a[j] * a[m2] contributes; reduce the larger
                if f[j] >= f[m2]:
                    reductions[j] += w_k * f[m2]
                else:
                    reductions[m2] += w_k * f[j]
        rmax = reductions.max()
        if rmax < 1e-15:
            break
        reductions = reductions / rmax
        # Try several step sizes
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            f_new = f - alpha * reductions
            f_new = project_simplex(f_new)
            new_peak = fast_convolve(f_new).max()
            if new_peak < peak_val - 1e-14:
                f = f_new
                improved = True
                if new_peak < best_obj:
                    best_obj = new_peak
                    best_f = f.copy()
                break
        if not improved:
            # Try tiny random perturbations
            f = project_simplex(f + rng.standard_normal(n) * 0.001)
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} score={score:.8f} t={elapsed:.0f}s")
    return best_f, best_obj


def lp_polish(f_init, time_limit, label="lp"):
    """
    LP-based polish with constraint generation.
    max sum(g) s.t. g>=0, (f*g)[k] <= M for active k.
    """
    import cvxpy as cp
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break
        # Active constraints: top peaks
        top_k = min(30 + 5 * it, 100)
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        active_list = sorted(top_idx.tolist())

        # Build constraint matrix
        K = len(active_list)
        A_mat = np.zeros((K, n))
        for row_i in range(K):
            k = active_list[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A_mat @ g_var <= M]
        )
        g_val = None
        for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, max_iters=100000)
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
                    if not np.any(np.isnan(g_val)):
                        break
                    g_val = None
            except Exception:
                continue

        if g_val is None:
            break

        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            if it > 10:
                break
            continue

        h = g_val / S_g
        # Line search
        best_t = 0.1
        best_trial = best_obj
        for t_candidate in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t_candidate) * f + t_candidate * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t_candidate

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
        f = f_new
        cur = fast_convolve(f).max()
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_f = f.copy()
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} score={score:.8f} t={elapsed:.0f}s")
    return best_f, best_obj


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolation_init(src, target_n):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    old_x = np.linspace(0.0, 1.0, n_src)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, src)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return f_interp / s
