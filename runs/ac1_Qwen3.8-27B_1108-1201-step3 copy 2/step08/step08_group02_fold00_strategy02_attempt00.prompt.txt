def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def line_search(f, h, M_current, max_steps=15):
    """
    Backtracking line search. Finds best t in (0, 1] such that

    Returns (best_t, best_obj).

    best_obj = M_current

    for t in t_grid:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    if best_t > 0:

            t = dt

                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-15:

                best_t = t

    return best_t, best_obj

def mirror_descent_optimize(f_init, time_limit, lr_init=0.05, n_top=100, label="md"):
    """Mirror descent with negative entropy for polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on peaks
        temp = 0.002 * M
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[k] * f[k - idx]

        # Mirror descent update: f_new[i] proportional to f[i] * exp(-lr * grad[i])
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()  # numerical stability
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new /= s
        f = f_new

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.92
                    no_improve = 0

            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 1e-15)
                f /= f.sum()
                lr = lr_init
                no_improve = 0

    return best_f, best_b

        w = np.exp(scaled)

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
