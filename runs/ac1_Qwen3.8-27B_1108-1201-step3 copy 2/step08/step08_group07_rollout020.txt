    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e15
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def fast_cross_correlation(f, g, fft_size=None):
    """Compute cross-correlation (f ⋆ g)[n] = sum_k f[k] * g[k-n]."""
    len_f = len(f)
    len_g = len(g)
    if fft_size is None:
        fft_size = 1 << (len_f + len_g - 1 - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(np.conj(F) * G, fft_size)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def subgradient_method(a0, time_limit, eps_frac=0.01, eta_init=0.05,
                       momentum=0.9, verbose=True, label="SGD"):
    """
    Fast subgradient method using FFT for pressure computation.
    O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta_init
    iters = 0
    v = np.zeros(n)  # momentum accumulator
    stall = 0
    stall_threshold = 500
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Find active lags
        threshold = M * (1.0 - eps_frac)
        active_idx = np.where(conv >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([conv.argmax()])

        # Build indicator for active lags
        size = len(conv)
        S_mask = np.zeros(size)
        S_mask[active_idx] = 1.0 / len(active_idx)

        # Compute pressure via FFT cross-correlation
        fft_size = 1 << (size + n - 1 - 1).bit_length()
        p = fast_cross_correlation(S_mask, a, fft_size)[:n]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * p

        # Step
        a_new = np.maximum(0.0, a - eta * v)
        ss = a_new.sum()
        if ss < 1e-15:
            eta *= 0.5
            continue
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.02, 0.5)
        else:
            stall += 1
            eta *= 0.97

        a = a_new

        # Perturbation on stall
        if stall > stall_threshold:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            eta = eta_init
            stall = 0

        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"eta={eta:.4f}, |S|={len(active_idx)}, time={elapsed:.0f}s")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True, label="LP-FP"):
    """LP-based fictitious play with adaptive step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 800

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)

    def softmax_from_z(z):
        z_shift = z - z.max()
        exp_z = np.exp(z_shift)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def z_from_a(a):
        return np.log(np.maximum(a, 1e-30))

    def smoothed_objective(z, T_val, n_val):
        a_local = softmax_from_z(z)
        conv_local = fast_convolve(a_local)
        shifted = conv_local / T_val
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n_val / T_val) * np.exp(T_val * lse)

    def smoothed_gradient(z, T_val, n_val):
        a_local = softmax_from_z(z)
        conv_local = fast_convolve(a_local)
        shifted = conv_local / T_val
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size_local = 2 * n_val - 1
        fft_size_local = 1 << (size_local - 1).bit_length()
        W = rfft(w, fft_size_local)
        A = rfft(a_local, fft_size_local)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size_local)[:n_val]
        S = float(np.dot(grad_a, a_local))
        grad_z = a_local * (grad_a - S)
        return (2.0 * n_val / T_val) * grad_z

    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def cosine_taper_init(n, alpha=0.5):
    """Generate cosine-tapered initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 - alpha * (2 * x - 1) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def random_smooth_init(n, seed=42):
    """Generate random smooth initialization."""
    rng = np.random.default_rng(seed)
    a = rng.standard_normal(n)
    # Smooth with a small kernel
    kernel_size = max(3, n // 100)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 2)) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fast subgradient, LP fictitious play,
    and L-BFGS polishing. Tries multiple initializations and n values.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = cosine_taper_init(n_init, alpha=0.3)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized cosine taper: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Fast subgradient with cosine taper init (n=768)
    print(f"\n=== Phase 1: Subgradient cosine init, n=768 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 350.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1 = cosine_taper_init(n_parent, alpha=0.3)
        a1_res, sc1 = subgradient_method(a1, t1, eps_frac=0.005, eta_init=0.03,
                                         momentum=0.85, label="SGD-cos")
        best_seq, best_score = check_and_update(a1_res, "sgd_cos", best_seq, best_score)

    # Phase 2: Fast subgradient with random smooth init (n=768)
    print(f"\n=== Phase 2: Subgradient random init, n=768 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a2 = random_smooth_init(n_parent, seed=123)
        a2_res, sc2 = subgradient_method(a2, t2, eps_frac=0.005, eta_init=0.03,
                                         momentum=0.85, label="SGD-rand")
        best_seq, best_score = check_and_update(a2_res, "sgd_rand", best_seq, best_score)

    # Phase 3: LP fictitious play (medium step) on best so far
    print(f"\n=== Phase 3: LP-FP step=0.3 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 450.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a3_res, sc3 = lp_fictitious_play(best_seq, t3, step=0.3, label="LP-0.3")
        best_seq, best_score = check_and_update(a3_res, "lp_0.3", best_seq, best_score)

    # Phase 4: LP fictitious play (small step) for refinement
    print(f"\n=== Phase 4: LP-FP step=0.15 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 60:
        print(f"  Budget: {t4:.0f}s")
        a4_res, sc4 = lp_fictitious_play(best_seq, t4, step=0.15, label="LP-0.15")
        best_seq, best_score = check_and_update(a4_res, "lp_0.15", best_seq, best_score)

    # Phase 5: L-BFGS polish
    print(f"\n=== Phase 5: L-BFGS polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for T_val in [300, 800]:
        t5 = min(rem * 0.10, 80.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a5_res, sc5 = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a5_res, f"lbgfs_T{T_val}", best_seq, best_score)
        rem = deadline - time.monotonic()

    # Phase 6: Try a different n value
    print(f"\n=== Phase 6: Alternative n (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_n = n_parent + 128 if n_parent <= 1024 else n_parent - 64
    if rem > 120 and 64 <= alt_n <= 4096:
        t6 = min(rem * 0.20, 200.0)
        print(f"  Trying n={alt_n}, budget={t6:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick subgradient then LP
        t6a = t6 * 0.4
        a6a_res, _ = subgradient_method(init_alt, t6a, eps_frac=0.005,
                                         eta_init=0.03, momentum=0.85, label="SGD-alt")
        t6b = min(t6 * 0.6, deadline - time.monotonic() - 10)
        if t6b > 30:
            a6b_res, _ = lp_fictitious_play(a6a_res, t6b, step=0.25, label="LP-alt")
            best_seq, best_score = check_and_update(a6b_res, f"alt_n{alt_n}",
                                                     best_seq, best_score)

    # Phase 7: Final LP polish on overall best
    print(f"\n=== Phase 7: Final LP polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.3, 150.0)
        a7_res, sc7 = lp_fictitious_play(best_seq, t7, step=0.1, label="LP-final")
        best_seq, best_score = check_and_update(a7_res, "lp_final", best_seq, best_score)

    # Phase 8: Final subgradient burst
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.4, 100.0)
        a8_res, sc8 = subgradient_method(best_seq, t8, eps_frac=0.003,
                                         eta_init=0.01, momentum=0.9, label="SGD-final")
        best_seq, best_score = check_and_update(a8_res, "sgd_final", best_seq, best_score)

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL SCORE: {final_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
