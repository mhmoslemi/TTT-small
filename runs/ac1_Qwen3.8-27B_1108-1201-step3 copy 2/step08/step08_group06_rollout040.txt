"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play with "min peak" formulation + geometric mean + averaging
2. Adaptive step size selection (try multiple, pick best)
3. Subgradient descent with Nesterov momentum
4. Coordinate descent with peak-focused pairs and exact line search
5. Multi-start with diverse smooth initializations
6. Length exploration near parent
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_min_t_lp(f, max_constraints=None):
    """
    Solve LP: min t s.t. (f*g)(k) <= t for all k, sum(g)=1, g>=0.
    Optionally use only top max_constraints rows to speed up.
    Returns (g, t_opt) or (None, None).
    """
    n = len(f)
    C = build_conv_matrix(f)
    conv_f = fast_convolve(f)

    if max_constraints is not None and max_constraints < (2 * n - 1):
        # Use only the top constraints (highest f*f values)
        top_idx = np.argsort(conv_f)[::-1][:max_constraints]
        C_sub = C[top_idx, :]
        b_sub = np.zeros(len(top_idx))
    else:
        C_sub = C
        b_sub = np.zeros(2 * n - 1)

    m = C_sub.shape[0]
    # Variables: g_0,...,g_{n-1}, t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.hstack([C_sub, -np.ones((m, 1))])
    b_ub = b_sub

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None

    if res.status != 0:
        return None, None

    g = np.maximum(res.x[:n], 0)
    t_val = float(res.x[n])
    s = g.sum()
    if s < 1e-15:
        return None, None
    g = g / s
    return g, t_val


def solve_max_sum_lp(f, M, max_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M, g>=0."""
    n = len(f)
    C = build_conv_matrix(f)
    conv_f = fast_convolve(f)

    if max_constraints is not None and max_constraints < (2 * n - 1):
        top_idx = np.argsort(conv_f)[::-1][:max_constraints]
        C_sub = C[top_idx, :]
        b_sub = np.full(len(top_idx), M)
    else:
        C_sub = C
        b_sub = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=C_sub, b_ub=b_sub,
                      bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None

    if res.status != 0:
        return None

    g = np.maximum(res.x, 0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation: f^(1-t) * g^t, normalized."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1.0 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def linear_update(f, g, t):
    """Linear interpolation: (1-t)*f + t*g, normalized."""
    a = (1.0 - t) * f + t * g
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play_min_t(f_init, time_limit, use_reduced_lp=True, verbose=True):

    LP fictitious play using 'min t' formulation with geometric mean updates
    and running average. At each step, solve LP using the running average,

    """
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f_init, compute_score(f_init)
    f = f / s
    n = len(f)

    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    k = 0
    max_constraints = min(200, 2 * n - 1) if use_reduced_lp else None

    while time.monotonic() - t0 < time_limit:
        # Solve LP using f_avg (averaged fictitious play)
        g, t_opt = solve_min_t_lp(f_avg, max_constraints=max_constraints)
        if g is None:
            # Fallback: use max-sum LP
            M = fast_convolve(f_avg).max()
            g = solve_max_sum_lp(f_avg, M, max_constraints=max_constraints)
            if g is None:

                best_step_a = a_trial

        f = best_step_a
        f_avg = ((k * f_avg) + f) / (k + 1)

        if sc_cur < best_score:
            best_score = sc_cur
            best_f = f.copy()
        if sc_avg < best_score:

        k += 1
        if verbose and k % 50 == 0:

def subgradient_momentum(f_init, time_limit, verbose=True):
    """Subgradient descent with Nesterov momentum on the simplex."""
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f_init, compute_score(f_init)
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    k = 0
    momentum = np.zeros(n)
    learning_rate = 0.1

    while time.monotonic() - t0 < time_limit:
        # Convection (Nesterov lookahead)
        look = f - momentum * learning_rate * 0.9
        look = np.maximum(look, 0)
        ls = look.sum()
        if ls > 1e-15:
            look = look / ls
        else:
            look = f.copy()

        # Compute subgradient at peak positions
        conv = fast_convolve(look)
        peak_val = conv.max()
        # Use top positions within 1% of max
        threshold = peak_val * 0.995
        peak_mask = conv >= threshold
        S = np.where(peak_mask)[0]

        if len(S) == 0:
            S = np.array([int(np.argmax(conv))])

        # Compute subgradient: g_i = 2 * sum_{k in S} a_{k-i} (if valid)
        grad = np.zeros(n)
        for k_idx in S:
            i_start = max(0, k_idx - n + 1)
            i_end = min(n - 1, k_idx)
            lo = max(0, k_idx - n + 1)
            hi = min(k_idx, n - 1)
            # a[k-i] for i in [i_start, i_end]
            # k-i ranges from k-i_start to k-i_end = lo to hi
            # So a[k-i] = a[hi], a[hi-1], ..., a[lo] (reversed)
            grad[i_start:i_end + 1] += 2.0 * look[lo:hi + 1][::-1]

        # Project to tangent space of simplex
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean

        # Momentum update
        momentum = 0.9 * momentum + 0.1 * grad_proj
        direction = -momentum

        # Line search (try a few step sizes)
        step_candidates = [0.01, 0.02, 0.05, 0.1]
        best_trial_score = compute_score(f)
        best_trial_a = f.copy()
        for lr in step_candidates:
            a_trial = f + lr * direction
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial

        f = best_trial_a

        if best_trial_score < best_score:
            best_score = best_trial_score
            best_f = f.copy()

        # Decay learning rate
        learning_rate = max(0.005, 0.1 / (1.0 + k * 0.01))
        k += 1

        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-mom iter={k}, lr={learning_rate:.4f}, "
                  f"score={best_score:.8f}, n_peaks={len(S)}, elapsed={elapsed:.0f}s")

    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=True, rng_seed=42):
    """Coordinate descent with golden-section search, peak-focused."""
    n = len(f_init)
    a = np.maximum(f_init.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    gr = (np.sqrt(5) - 1) / 2

    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))

        n_pairs = min(n, 300)
        pairs_tried = 0

        while pairs_tried < n_pairs:
            if time.monotonic() - t0 > time_limit:
                break

            # Choose pair: mix of peak-focused, random, and edge
            r = rng.random()
            if r < 0.5:
                # Peak-focused: both indices near the top peak
                js = max(0, top_peak - n + 1)
                je = min(n - 1, top_peak)
                rng_width = je - js + 1
                if rng_width < 2:
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                else:
                    i = int(rng.integers(js, je + 1))
                    j = int(rng.integers(js, je + 1))
            elif r < 0.8:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
            else:
                # Edge: one near start, one near end
                edge_w = max(2, n // 10)
                i = int(rng.integers(0, edge_w))
                j = int(rng.integers(n - edge_w, n))

            if i == j:
                continue
            pairs_tried += 1

            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            # Golden section search for transfer amount
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            # Evaluate at x1, x2
            a_save = a.copy()
            a[i] -= x1
            a[j] += x1
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            f1 = compute_score(a)
            a[:] = a_save

            a[i] -= x2
            a[j] += x2
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss
            f2 = compute_score(a)
            a[:] = a_save

            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    a[i] -= x1
                    a[j] += x1
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
                    f1 = compute_score(a)
                    a[:] = a_save
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    a[i] -= x2
                    a[j] += x2
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
                    f2 = compute_score(a)
                    a[:] = a_save

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def smooth_initializations(n, rng):
    """Generate diverse smooth initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    hc = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(hc / hc.sum())

    # Gaussian-like
    center = n / 2
    sig = n / 4
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sig) ** 2)
    inits.append(gauss / gauss.sum())

    # Inverse cosine (U-shaped)
    uc = 0.5 * (1 - np.cos(2 * np.pi * t)) + 0.1
    inits.append(uc / uc.sum())

    # Power law
    pw = np.power(t + 0.01, 0.5)
    inits.append(pw / pw.sum())

    # Sine wave
    sw = 1.0 + 0.3 * np.sin(4 * np.pi * t)
    inits.append(np.maximum(sw, 0) / np.maximum(sw, 0).sum())

    # Random smooth
    freqs = rng.integers(1, 5, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    rs = np.ones(n)
    for fi, ph, am in zip(freqs, phases, amps):
        rs += am * np.sin(2 * np.pi * fi * t + ph)
    rs = np.maximum(rs, 0)
    inits.append(rs / rs.sum())

    return inits


def multi_start_search(f_best, time_limit, verbose=True):

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, normalized."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if sc_final < best_score:
            best_score = sc_final
            best_f = a_alt.copy() if sc_alt < sc_alt2 else a_alt2.copy()
            if verbose:
                print(f"    Length n={alt_n}: NEW BEST={best_score:.8f}")
        tested += 1

    return best_f, best_score


def final_polish(f_best, time_limit, verbose=True):
    """Alternating LP and CD for final squeezing."""
    a = f_best.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Alternate: 1 LP step, then 5 CD-like coordinate updates
    k = 0
    while time.monotonic() - t0 < time_limit:
        # One LP step
        t_lp = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if t_lp > 1:
            a_lp, sc_lp = lp_fictitious_play_min_t(a, t_lp, use_reduced_lp=True, verbose=False)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
                a = a_lp.copy()

        # A few coordinate descent steps
        t_cd = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if t_cd > 1:
            a_cd, sc_cd = coordinate_descent(a, t_cd, verbose=False, rng_seed=k % 100)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd.copy()
                a = a_cd.copy()

        k += 1
        if verbose and k % 3 == 0:
            print(f"    Polish iter={k}, score={best_score:.8f}")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate using evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing the step function score.
    Uses LP fictitious play (min-t formulation), subgradient with momentum,
    coordinate descent, multi-start, length exploration, and final polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        t_arr = np.linspace(0, 1, n_init)
        parent = 0.5 * (1 + np.cos(2 * np.pi * t_arr)) + 0.1
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized: n={n_init}, score={p_score:.8f}")
        best_seq = parent.copy()
        best_score = p_score

    n_parent = len(parent)
    print(f"Starting optimization. n={n_parent}, initial score={best_score:.8f}")

    # === Phase 1: LP Fictitious Play (min-t formulation) ===
    print(f"\n=== Phase 1: LP Fictitious Play [min-t] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(400.0, rem * 0.25)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, sc1 = lp_fictitious_play_min_t(best_seq, t_alloc, use_reduced_lp=True, verbose=True)
        best_seq, best_score = check_and_update(a1, "lp_min_t", best_seq, best_score)

    # === Phase 2: LP Fictitious Play (max-sum formulation) ===
    print(f"\n=== Phase 2: LP Fictitious Play [max-sum] (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(200.0, rem * 0.2)
        print(f"  Budget: {t_alloc:.0f}s")
        a2, sc2 = lp_fictitious_play_max_sum(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a2, "lp_max_sum", best_seq, best_score)

    # === Phase 3: Subgradient with momentum ===
    print(f"\n=== Phase 3: Subgradient Momentum (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(150.0, rem * 0.2)
        print(f"  Budget: {t_alloc:.0f}s")
        a3, sc3 = subgradient_momentum(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a3, "sg_mom", best_seq, best_score)

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(200.0, rem * 0.25)
        print(f"  Budget: {t_alloc:.0f}s")
        a4, sc4 = coordinate_descent(best_seq, t_alloc, verbose=True, rng_seed=seed)
        best_seq, best_score = check_and_update(a4, "cd", best_seq, best_score)

    # === Phase 5: Multi-start with diverse inits ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        t_alloc = min(250.0, rem * 0.4)
        print(f"  Budget: {t_alloc:.0f}s")
        a5, sc5 = multi_start_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a5, "multi_start", best_seq, best_score)

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 60:
        t_alloc = min(200.0, rem * 0.5)
        print(f"  Budget: {t_alloc:.0f}s")
        a6, sc6 = length_exploration(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a6, "length_exp", best_seq, best_score)

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_alloc = min(150.0, rem * 0.8)
        print(f"  Budget: {t_alloc:.0f}s")
        a7, sc7 = final_polish(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a7, "polish", best_seq, best_score)

    # === Phase 8: One more CD at the end ===
    rem = deadline - time.monotonic() - 10
    if rem > 10 and best_seq is not None:
        t_alloc = min(50.0, rem * 0.8)
        a8, sc8 = coordinate_descent(best_seq, t_alloc, verbose=False, rng_seed=seed + 999)
        best_seq, best_score = check_and_update(a8, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
