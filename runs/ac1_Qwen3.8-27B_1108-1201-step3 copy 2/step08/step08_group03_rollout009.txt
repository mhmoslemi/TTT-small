```python
"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using adaptive
log-sum-exp gradient descent with momentum, cutting-plane LP refinement,
and spectral diversification.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_smooth_gradient(w, a, fft_size):
    """
    Compute gradient of log-sum-exp(beta * b) w.r.t. a.
    w: softmax weights (length 2n-1)
    a: sequence (length n)
    Returns: gradient (length n) = 2 * beta * cross_corr(w, a)
    where cross_corr[k] = sum_{l=0}^{n-1} w[k+l] * a[l]
    """
    n = len(a)
    # cross_corr[k] = sum_{l=0}^{n-1} w[k+l] * a[l]
    # = ifft(fft(w) * conj(fft(a)))[:n]
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    cross = ifft(W * np.conj(A)).real[:n]
    return 2.0 * cross  # factor of 2 from the derivative; beta is multiplied into w already


def logsumexp_smooth_descent(f_init, time_limit, beta_0=1.0, beta_max=200.0,
                              lr_init=0.001, momentum_init=0.9,
                              beta_growth=1.02, lr_decay=0.95,
                              perturbation_interval=200,
                              perturbation_scale=0.01,
                              label="lse"):
    """
    Log-sum-exp gradient descent with adaptive temperature and momentum.
    Minimizes max(auto_convolve(a)) subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    a /= s
    
    # FFT size for the gradient computation
    # Need N >= (2n-1) + n - 1 = 3n - 2
    grad_fft_size = 1 << (3 * n - 3).bit_length()
    
    beta = beta_0
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_b = fast_convolve(a)
    best_obj = best_b.max()
    
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        
        # Compute softmax weights: w_k = exp(beta * (b_k - max_b)) / sum(...)
        scaled = beta * (b - b.max())
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(b)) / len(b)
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_smooth_gradient(w, a, grad_fft_size) * beta
        
        # Momentum update
        v = momentum_init * v - lr * grad
        a_new = a + v
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        a = a_new
        
        # Evaluate
        b_new = fast_convolve(a)
        obj = b_new.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 50:
            lr *= lr_decay
            no_improve = 0
        
        # Increase beta (annealing)
        if beta < beta_max:
            beta *= beta_growth
        
        # Perturbation to escape local minima
        if it % perturbation_interval == 0:
            a = a + rng.standard_normal(n) * perturbation_scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a /= s
            v = np.zeros(n)
        
        # Reporting
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} beta={beta:.2f} lr={lr:.2e} "
                  f"obj={best_obj:.8f} score={score:.6f} t={elapsed:.0f}s")
    
    return best_a, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    g = cp.Variable(n)
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g = None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_with_cutting_planes(f_init, time_limit, initial_k=50, max_k=400, label="cp"):

    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_lp_direction(f, n, M, active_idx, eps=1e-8)

        if g is None:
            stuck += 1
            if stuck > 8:
                break
            if len(active_idx) > initial_k:
                b_cur = fast_convolve(f)
                active_idx = np.argpartition(b_cur, -initial_k)[-initial_k:]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane: check for violated constraints
        fg = convolve_direction(f, g)
        violated_mask = fg > M * (1 + 1e-7)
        v_count = violated_mask.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated_mask)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(30, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx, eps=1e-8)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        scan_steps = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in scan_steps:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden-section refinement
        if best_t > 0.01:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(6):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically refresh active set
        if it % 10 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_obj


def spectral_perturb(f, scale=0.05, rng=None):
    """
    Apply a spectral perturbation: modify the phase of the FFT slightly.
    This can escape local minima while preserving the overall structure.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    F = rfft(f, fft_size)
    
    # Add small phase perturbation
    phase_perturbation = rng.standard_normal(F.shape).real * scale
    abs_F = np.abs(F)
    phase = F / (abs_F + 1e-30)
    F_new = abs_F * (phase + 1j * phase_perturbation)
    # Normalize phase
    abs_new = np.abs(F_new)
    phase_new = F_new / (abs_new + 1e-30)
    F_new = abs_F * phase_new
    
    f_new = irfft(F_new, fft_size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def diverse_initialization(base, n_target=None, rng_key=42):
    """
    Generate diverse initializations around the base sequence.
    """
    rng = np.random.default_rng(rng_key)
    candidates = [base.copy()]
    n = len(base)
    
    if n_target is not None and n_target != n:
        old_x = np.linspace(0.0, 1.0, n)
        new_x = np.linspace(0.0, 1.0, n_target)
        base_interp = np.interp(new_x, old_x, base)
        base_interp = np.maximum(base_interp, 0.0)
        s = base_interp.sum()
        if s > 1e-10:
            base_interp /= s
        candidates.append(base_interp)
        n = n_target
    
    # Spectral perturbations
    for _ in range(4):
        c = spectral_perturb(base, scale=0.1, rng=rng)
        candidates.append(c)
    
    # Add small noise
    for _ in range(3):
        c = base + rng.standard_normal(n) * 0.02
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-10:
            c /= s
        candidates.append(c)
    
    # Global random (to explore completely different structures)
    for _ in range(2):
        c = rng.random(n) ** 2  # biased toward 0
        s = c.sum()
        if s > 1e-10:
            c /= s
        candidates.append(c)
    
    return candidates


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)

    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: combines log-sum-exp gradient descent, cutting-plane LP,
    spectral diversification, and coordinate polish.
    
    Strategy:
    1. Generate diverse initializations
    2. Run log-sum-exp gradient descent on top candidates
    3. Refine with cutting-plane LP
    4. Polish with coordinate descent
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Diversified initialization ===
    print(f"\n=== Phase 1: Diverse init (best={best_score:.6f}) ===")
    rng = np.random.default_rng(seed)
    candidates = diverse_initialization(parent, n_target=None, rng_key=seed)
    
    # Evaluate all candidates
    cand_scores = []
    for i, c in enumerate(candidates):
        sc = float(evaluate_sequence(c.tolist()))
        cand_scores.append((sc, c))
        print(f"  Candidate {i}: score={sc:.6f}, n={len(c)}")
    
    # Sort by score
    cand_scores.sort(key=lambda x: x[0])
    
    # Update best if any candidate is better
    if cand_scores[0][0] < best_score:
        best_score = cand_scores[0][0]
        best_seq = cand_scores[0][1].copy()
        print(f"  NEW BEST from init: {best_score:.6f}")
    
    # === Phase 2: Log-sum-exp gradient descent on top candidates ===
    print(f"\n=== Phase 2: LSE Gradient Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    
    # Run on the best 2 candidates (if they're different enough)
    top_cands = cand_scores[:2]
    # Make sure they're different
    if len(top_cands) == 2:
        diff = np.abs(top_cands[0][1] - top_cands[1][1]).max()
        if diff < 1e-4:
            top_cands = top_cands[:1]
            # Add a perturbed version
            top_cands.append((None, spectral_perturb(parent, scale=0.15, rng=rng)))
    
    for i, (sc, cand) in enumerate(top_cands):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t2 = min(rem * 0.4, 400.0)
        if t2 < 30:
            break
        n_c = len(cand)
        print(f"  LSE-SD candidate {i}: n={n_c}, budget={t2:.0f}s")
        
        f2, obj2 = logsumexp_smooth_descent(
            cand, time_limit=t2,
            beta_0=0.5, beta_max=300.0,
            lr_init=0.0005,
            momentum_init=0.92,
            beta_growth=1.015,
            lr_decay=0.93,
            perturbation_interval=300,
            perturbation_scale=0.008,
            label=f"lse{i}"
        )
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lse{i}]: {best_score:.6f}")
    
    # === Phase 3: Cutting-plane LP refinement ===
    print(f"\n=== Phase 3: CP-LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    
    # Multiple LP rounds with increasing constraints
    for round_num, (ik, mk, frac) in enumerate([(50, 300, 0.25), (80, 500, 0.25), (100, 600, 0.20)]):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t3 = min(rem * frac, 350.0)
        if t3 < 30:
            break
        n_b = len(best_seq)
        print(f"  CP-LP round {round_num}: n={n_b}, k={ik}->{mk}, budget={t3:.0f}s")
        f3, obj3 = lp_with_cutting_planes(
            best_seq, time_limit=t3,
            initial_k=ik, max_k=mk,
            label=f"cp{round_num}"
        )
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [cp{round_num}]: {best_score:.6f}")
    
    # === Phase 4: Spectral diversification + LSE ===
    print(f"\n=== Phase 4: Spectral diversify + LSE (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    
    n_div = 3
    for d in range(n_div):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        f_div = spectral_perturb(best_seq, scale=0.08 + 0.04 * d, rng=rng)
        t4 = min(rem * 0.15, 120.0)
        if t4 < 20:
            break
        print(f"  Spectral div {d}: budget={t4:.0f}s")
        f4, obj4 = logsumexp_smooth_descent(
            f_div, time_limit=t4,
            beta_0=1.0, beta_max=200.0,
            lr_init=0.0008,
            momentum_init=0.90,
            beta_growth=1.02,
            lr_decay=0.94,
            perturbation_interval=200,
            perturbation_scale=0.01,
            label=f"div{d}"
        )
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [div{d}]: {best_score:.6f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    
    target_lengths = sorted(set([
        max(512, int(n_base * 0.9)),
        max(512, int(n_base * 1.1)),
        max(512, int(n_base * 1.25)),
    ]))
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if target_n == n_base:
            continue
        t5 = min(rem * 0.12, 100.0)
        if t5 < 20:
            break
        
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n
        
        print(f"  LSE n={target_n}, budget={t5:.0f}s")
        f5, _ = logsumexp_smooth_descent(
            f_interp, time_limit=t5,
            beta_0=0.5, beta_max=200.0,
            lr_init=0.0005,
            momentum_init=0.90,
            label=f"n{target_n}"
        )
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.6f}")
    
    # === Phase 6: Final LP + coordinate polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    
    # Final LP burst
    t6 = min(rem * 0.4, 150.0)
    if t6 > 20:
        n_b = len(best_seq)
