import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax(v, beta):
    v = v - v.max()
    e = np.exp(beta * v)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(v)
    return e / s

def logsumexp(v, beta):
    v_scaled = beta * v
    v_max = v_scaled.max()
    if v_max > 100:
        return v_max + np.log(np.exp(v_scaled - v_max).sum())
    return v_max + np.log(np.exp(v_scaled - v_max).sum()) / beta

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_gradient(p, active_indices):
    """Compute gradient of sum of (p*p)[k] for k in active_indices."""
    n = len(p)
    grad = np.zeros(n)
    for k in active_indices:
        # (p*p)[k] = sum_j p[j] * p[j+k] (with boundary)
        # d/dp[i] (p*p)[k] = p[i-k] + p[i+k] (with boundary zeros)
        # For i from 0 to n-1:
        #   p[i-k] is valid if i-k >= 0, i.e., i >= k
        #   p[i+k] is valid if i+k < n, i.e., i < n-k
        if k == 0:
            grad += 2.0 * p
        else:
            # p[i-k] for i >= k
            grad[k:] += p[:n-k]
            # p[i+k] for i < n-k
            grad[:n-k] += p[k:]
    return grad

            weights = softmax(conv, beta)

                    grad += w_full[k] * 2.0 * p
                else:
                    grad[k:] += w_full[k] * p[:n-k]
                    grad[:n-k] += w_full[k] * p[k:]
            
            # Adam update
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2

"""
Optimization of non-negative step function heights to minimize auto-convolution peak score.
Uses smooth max (log-sum-exp) surrogate with Adam optimizer on the probability simplex,
combined with subgradient refinement, multi-length search, and multiple initializations.
Key difference from LP approach: uses smooth optimization landscape to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(p, conv, beta, top_k=200):
    """Compute gradient of sum_k softmax(beta*conv)[k] * (p*p)[k] w.r.t. p.
    
    Uses the formula: d/dp[i] (p*p)[k] = 2*p[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(p)
    conv_len = len(conv)
    
    # Compute softmax weights on top-k values
    if top_k < conv_len:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_idx = np.arange(conv_len)
    
    max_val = conv[top_idx].max()
    exp_vals = np.exp(beta * (conv[top_idx] - max_val))
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / w_sum
    
    # Compute gradient: grad[i] = 2 * sum_k weights[k] * p[k-i] * [0<=k-i<n]
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        k = int(top_idx[idx_i])
        w = weights[idx_i]
        if w < 1e-15:
            continue
        # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # grad[i] += 2*w*p[k-i] for i in [lo, hi]
            # p[k-i] for i=lo is p[k-lo], for i=hi is p[k-hi]
            # So the slice is p[k-hi : k-lo+1 : -1]
            grad[lo:hi + 1] += 2.0 * w * p[k - hi:k - lo + 1:-1]
    
    return grad


def compute_subgradient(p, conv, n_peaks=50):
    """Compute subgradient at the current max (hard max)."""
    n = len(p)
    if n_peaks < len(conv):
        top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(len(conv))
    
    # Equal weight on all top peaks
    w = 1.0 / len(top_idx)
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w * p[k - hi:k - lo + 1:-1]
    return grad


def smooth_max_optimize(p_init, time_limit, beta_start=50.0, beta_end=5000.0,
                        n_steps=3000, top_k=200, lr=0.002, label="smooth"):
    """Optimize using smooth max with Adam on simplex."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    no_improve = 0
    increase_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        t += 1
        
        # Adaptive beta (annealing)
        frac = min(1.0, t / (n_steps * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        # Also adaptively increase beta if stuck
        if no_improve > 200:
            beta = min(beta * 1.5, beta_start * 100)
            increase_count += 1
            no_improve = 0
        
        conv = fast_convolve(p)
        cur_obj = conv.max()
        
        # Compute smooth gradient
        grad = compute_smooth_gradient(p, conv, beta, top_k)
        
        if np.all(grad == 0):
            break
        
        # Normalize gradient for stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        p_new = project_simplex(p - update)
        p = p_new
        
        # Check improvement
        if step % 5 == 0 or t < 50:
            new_obj = fast_convolve(p).max()
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_p = p.copy()
                no_improve = 0
                # Slightly increase lr on improvement
                lr = min(lr * 1.01, 0.01)
            else:
                no_improve += 1
                if no_improve > 500:
                    lr *= 0.9
                    no_improve = 0
                    # Small random perturbation
                    rng = np.random.default_rng(t)
                    p = project_simplex(p + rng.standard_normal(n) * 0.002)
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} step={t} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} beta={beta:.0f} lr={lr:.5f} t={elapsed:.0f}s")
    
    return best_p, best_obj


def subgradient_refine(p_init, time_limit, n_peaks=50, label="sg"):
    """Subgradient method with adaptive step size for final refinement."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    
    # Polyak-style step size
    obj_lb = 1.0 / (2 * n - 1)  # theoretical lower bound
    alpha = 0.1
    momentum = 0.7
    vel = np.zeros(n)
    no_improve = 0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(p)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_p = p.copy()
            no_improve = 0
            alpha = min(alpha * 1.005, 0.5)
        else:
            no_improve += 1
            if no_improve > 100:
                alpha *= 0.95
                no_improve = 0
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                p = project_simplex(best_p + rng.standard_normal(n) * 0.001)
                vel = np.zeros(n)
                no_improve = 0
                alpha = 0.05
        
        # Compute subgradient
        grad = compute_subgradient(p, conv, n_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        
        # Momentum update
        vel = momentum * vel - alpha * grad
        p_new = project_simplex(p + vel)
        p = p_new
    
    return best_p, best_obj


def flat_top_window(n):
    """Flat top window normalized to sum to 1."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i in range(len(a)):
        w -= a[i] * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Hann window normalized to sum to 1."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def gaussian_window(n, sigma_frac=0.4):
    """Gaussian window normalized to sum to 1."""
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    sigma = sigma_frac * (n - 1) / 2.0
    w = np.exp(-0.5 * (x / sigma) ** 2)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def cos_taper_window(n, power=1.0):
    """Cosine taper: w[i] = (cos(pi * x))^power where x in [-1,1]."""
    x = np.linspace(-1, 1, n)
    w = np.cos(np.pi * x / 2.0) ** power
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

            print(f"  fine_tune: {score_fin:.8f} (final best={best_score:.8f})")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Difference: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_seq]
