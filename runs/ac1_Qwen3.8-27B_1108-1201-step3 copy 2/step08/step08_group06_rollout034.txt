Multi-strategy optimization of 2*n*max(conv(a,a))/(sum(a))^2.
Combines:
1. LP fictitious play with min-peak LP direction and adaptive step sizes
2. Subgradient descent with simplex projection and line search

4. Coordinate descent with pairwise transfers

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.array(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    
    # Use the algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_max_sum(f, M):
    """Solve LP: max sum(g) s.t. C*f*g <= M, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None, M
    return np.maximum(res.x, 0), M


def solve_lp_min_peak(f):
    """Solve LP: min M s.t. C_f * g <= M, sum(g) = 1, g >= 0."""
    n = len(f)
    C = build_C(f)
    # Variables: g (n) + M (1) = n+1 variables
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: C_f * g - M <= 0 for each row (2n-1 rows)
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    g = np.maximum(res.x[:n], 0)
    M = res.x[-1]
    return g, M


def lp_fictitious_play_v2(f_init, time_limit, verbose=True):
    """Improved LP fictitious play with min-peak LP and adaptive steps."""
    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)
    
    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Use exponential moving average for better convergence
    ema_alpha = 0.1
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        # Solve LP to get direction
        g, M_lp = solve_lp_max_sum(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 1)
        
        # Geometric mean update
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        
        # EMA update
        f_avg = (1 - ema_alpha) * f_avg + ema_alpha * a_new
        
        score_avg = compute_score(f_avg)
        score_cur = compute_score(a_new)
        
        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_f = f_avg.copy()
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-v2 iter={k}, M={M:.6e}, score_avg={score_avg:.8f}, "
                  f"best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_f, best_score


def subgradient_descent(f_init, time_limit, verbose=True):
    """Subgradient descent with simplex projection and line search."""
    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Decay step size
    base_eta = 0.05
    decay = 0.999
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active set (within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute subgradient
        # grad[m] = (2/|S|) * sum_{k in S} a[k-m] * I(0 <= k-m < n)
        grad = np.zeros(n)
        for ki in active:
            ki = int(ki)
            for m in range(n):
                diff = ki - m
                if 0 <= diff < n:
                    grad[m] += 2.0 * f[diff]
        grad /= len(active)
        
        # Line search: try different step sizes
        eta = base_eta * (decay ** (k // 100))
        
        best_trial_score = best_score
        best_trial = None
        
        for eta_mult in [0.1, 0.25, 0.5, 1.0, 2.0, 4.0]:
            eta_try = eta * eta_mult
            f_trial = f - eta_try * grad
            f_trial = project_simplex(f_trial)
            sc = compute_score(f_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial = f_trial
        
        if best_trial is not None:
            f = best_trial
            if best_trial_score < best_score - 1e-14:
                best_score = best_trial_score
                best_f = f.copy()
        else:
            # No improvement, reduce step size
            base_eta *= 0.9
            if base_eta < 1e-6:
                base_eta = 0.01
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={k}, M={M:.6e}, score={best_score:.8f}, "
                  f"eta={eta:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_score

Hmm, the subgradient computation with the double loop is going to be very slow for large n. Let me optimize it.

`grad[m] = (2/|S|) * sum_{k in S} f[k-m] * I(0 <= k-m < n)`

This is equivalent to: for each m, sum f[k-m] over k in S such that 0 <= k-m < n.

Let `d = k - m`. Then `d` ranges over `k - m` for k in S, and we need `0 <= d < n`.
So `grad[m] = (2/|S|) * sum_{d: d in {k-m : k in S} and 0 <= d < n} f[d]`

This is like a cross-correlation of `f` with the indicator of the active set.

Let me define `S_indicator[k] = 1` if k is active, 0 otherwise. Then:
`grad[m] = (2/|S|) * sum_k S_indicator[k] * f[k-m] * I(0 <= k-m < n)`

This is the cross-correlation of `S_indicator` and `f` evaluated at lag `m`:
`grad[m] = (2/|S|) * (S_indicator * f)(m)` (cross-correlation)

Where `(S * f)(m) = sum_k S[k] * f[k-m]` with appropriate bounds.

This can be computed via FFT! The cross-correlation of two sequences can be computed as:
`corr = ifft(conj(FFT(S)) * FFT(f))`

But we need to be careful about the length. The active set S has indices in [0, 2n-2], and f has indices in [0, n-1]. The cross-correlation `sum_k S[k]*f[k-m]` for m in [0, n-1] involves k in [m, m+n-1] (since we need 0 <= k-m < n).

Actually, let me just use the direct computation but vectorize it:

```python
# Active set indices
active_diffs = active[:, None] - np.arange(n)[None, :]  # shape: (|S|, n)
valid = (active_diffs >= 0) & (active_diffs < n)  # shape: (|S|, n)
# grad[m] = (2/|S|) * sum_{i: valid[i,m]} f[active_diffs[i,m]]
indices = np.where(valid)
grad = np.zeros(n)
np.add.at(grad, indices[1], f[active_diffs[indices]])
grad *= 2.0 / len(active)
```

This is O(|S| * n) in the worst case but typically |S| is small (maybe 10-100 active peaks).

Actually, for large n (768), this is fine. |S| is at most 2n-1 = 1535, and n = 768, so |S|*n ≈ 1.2M operations per iteration. That's fast.

Let me also think about the **spectral domain approach** more carefully.

The key insight: we want to minimize `max_k R(k)` where `R(k) = conv(a,a)[k]`.

In the circular case (circular convolution), `R_circ(k) = (1/n) * IDFT(|A|^2)[k]` where `A = DFT(a)`.

For the linear convolution, it's approximately the same for the middle part.

If we could make `|A|^2` as flat as possible, `R(k)` would be as flat as possible. The constraint is `a >= 0`, which in the frequency domain means `IDFT(A) >= 0`, which is a non-convex constraint.

One approach: **iterative spectral flattening**
1. Compute `A = DFT(a)`
2. Compute `|A|^2`
3. Apply a "soft" flattening: `|A_new|^2 = |A|^2 / (1 + alpha * |A|^2 / mean(|A|^2))` or something similar
4. Update: `A_new = A * (|A_new| / |A|)` (keep the phase)
5. Update: `a_new = real(IDFT(A_new))`
6. Project to be non-negative and sum to 1

This is similar to the "peak reducing" algorithms used in OFDM systems.

Actually, a simpler approach: **iterative clipping in the frequency domain**
1. Compute `A = DFT(a)`
2. Find the frequency bins with the largest `|A|^2`
3. Reduce them slightly
4. Update `a` via IDFT
5. Project to non-negative simplex

Let me think about whether this would help... The issue is that reducing `|A|^2` at some frequencies might increase `R(k)` at some lags. It's a trade-off.

A better approach might be: **minimize the variance of R(k)** (not the max). If R(k) has low variance, its max must be close to its mean (= 1/(2n-1) approximately).

The variance of R(k) in terms of `|A|^2`:
`Var(R) = (1/(2n)) * sum_k (R(k) - mean(R))^2 = (1/(2n)) * sum_k R(k)^2 - mean(R)^2`

And `sum_k R(k)^2 = (1/n) * sum_m |A(m)|^4` (by Parseval applied to the autocorrelation).

So minimizing `sum_m |A(m)|^4` is equivalent to minimizing the variance of R(k)!

This gives us a convex (in `|A|^2`) objective: minimize `sum |A|^4` subject to `a >= 0, sum(a) = 1`.

But it's still non-convex due to the non-negativity constraint. However, we can use it as a heuristic.

**Spectral variance minimization algorithm:**
1. Compute `A = DFT(a)`
2. Compute `|A|^2`
3. The gradient of `sum |A|^4` w.r.t. `a` is:
   `d/d a_j [sum_m |A(m)|^4] = 4 * sum_m |A(m)|^2 * Re(A(m) * conj(dA(m)/da_j))`
   
   Now `A(m) = sum_j a(j) * exp(-2*pi*i*m*j/n)`
   `dA(m)/da_j = exp(-2*pi*i*m*j/n)`
   
   So the gradient is:
   `4 * sum_m |A(m)|^2 * Re(A(m) * exp(2*pi*i*m*j/n))`
   `= 4 * Re(sum_m |A(m)|^2 * A(m) * exp(2*pi*i*m*j/n))`
   `= 4 * Re(IDFT(|A|^2 * A)[j])`
   
   Where `IDFT(X)[j] = (1/n) * sum_m X(m) * exp(2*pi*i*m*j/n)`

4. Update: `a_new = a - eta * gradient`
5. Project to simplex.

    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):

        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]

        g[j] = max_gj

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.array(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_max_sum(f, M):
    """Solve LP: max sum(g) s.t. C_f * g <= M, g >= 0."""
    n = len(f)
    C = build_C(f)
    c_vec = -np.ones(n)
    res = linprog(c_vec, A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def greedy_lp_fast(f, M):
    """Fast greedy LP approximation - O(n^2)."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.min(slack[valid] / f_slice[valid])
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def lp_fictitious_play_v2(f_init, time_limit, verbose=True):
    """LP fictitious play with adaptive step sizes and EMA averaging."""
    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)

    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    # Adaptive EMA
    ema_alpha = 0.15
    iters_no_improve = 0

    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()

        # Use full LP for first 50 iters, then alternate
        if k % 3 == 0:
            g = solve_lp_max_sum(f, M)
        else:
            g = greedy_lp_fast(f, M)

        if g is None or g.sum() < 1e-15:
            g = greedy_lp_fast(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Adaptive step: larger when improving slowly
        if k < 20:
            t_step = 0.5
        elif k < 100:
            t_step = 0.3
        else:
            t_step = max(0.05, 2.0 / (k + 10))

        # Geometric mean update
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()

        # EMA
        f_avg = (1 - ema_alpha) * f_avg + ema_alpha * a_new

        score_avg = compute_score(f_avg)
        score_cur = compute_score(a_new)

        improved = False
        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_f = f_avg.copy()
            improved = True
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()
            improved = True

        if not improved:
            iters_no_improve += 1
            if iters_no_improve > 200:
                ema_alpha = max(0.05, ema_alpha * 0.95)
        else:
            iters_no_improve = 0

        f = a_new

        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-v2 iter={k}, M={M:.6e}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_f, best_score


def subgradient_descent(f_init, time_limit, verbose=True):
    """Subgradient descent with simplex projection and line search."""
    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    base_eta = 0.02
    no_improve = 0

    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find active set (top 5% of peaks)
        threshold = M * (1.0 - 0.02)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient efficiently
        grad = np.zeros(n)
        for ki in active:
            ki = int(ki)
            # grad[m] += 2*f[ki-m] for m where 0 <= ki-m < n
            m_start = max(0, ki - (n - 1))
            m_end = min(n - 1, ki)
            if m_start <= m_end:
                diff = ki - np.arange(m_start, m_end + 1)
                grad[m_start:m_end + 1] += 2.0 * f[diff]
        grad /= len(active)

        # Line search with multiple step sizes
        eta = base_eta * (0.99 ** (k // 50))
        best_trial_score = best_score
        best_trial = None

        for eta_mult in [0.05, 0.1, 0.25, 0.5, 1.0, 2.0]:
            eta_try = eta * eta_mult
            f_trial = f - eta_try * grad
            f_trial = project_simplex(f_trial)
            sc = compute_score(f_trial)
            if sc < best_trial_score - 1e-15:
                best_trial_score = sc
                best_trial = f_trial

        if best_trial is not None:
            f = best_trial.copy()
            if best_trial_score < best_score - 1e-14:
                best_score = best_trial_score
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            base_eta *= 0.9
            if base_eta < 1e-5:
                base_eta = 0.01
                # Perturb to escape
                rng = np.random.default_rng(k)
                f = best_f + rng.standard_normal(n) * 0.003
                f = np.maximum(f, 0)
                f = f / f.sum()
                no_improve = 0

        if no_improve > 500:
            f = best_f.copy()

        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={k}, M={M:.6e}, best={best_score:.8f}, eta={eta:.5f}, t={elapsed:.0f}s")

    return best_f, best_score


def spectral_optimization(f_init, time_limit, verbose=True):
    """Minimize sum(|A|^4) to flatten autocorrelation via spectral gradient descent."""
    f = np.maximum(f_init, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    eta = 0.01
    no_improve = 0

    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute DFT
        A = fft(f, n=n)
        abs_A2 = np.abs(A) ** 2

        # Gradient of sum(|A|^4) w.r.t. f[j]:
        # = 4 * Re(IDFT(|A|^2 * A)[j])
        grad_freq = abs_A2 * A
        grad = 4.0 * np.real(ifft(grad_freq))

        # Also add a small regularization to keep f away from 0
        # (helps maintain non-negativity)
        grad += 0.001 * np.maximum(0, -f)

        # Line search
        best_trial_score = best_score
        best_trial = None
        for eta_mult in [0.1, 0.25, 0.5, 1.0, 2.0]:
            eta_try = eta * eta_mult
            f_trial = f - eta_try * grad
            f_trial = project_simplex(f_trial)
            sc = compute_score(f_trial)
            if sc < best_trial_score - 1e-15:
                best_trial_score = sc
                best_trial = f_trial

        if best_trial is not None:
            f = best_trial.copy()
            if best_trial_score < best_score - 1e-14:
                best_score = best_trial_score
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            eta *= 0.9
            if eta < 1e-6:
                eta = 0.005
                rng = np.random.default_rng(k + 1000)
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                f = f / f.sum()
                no_improve = 0

        if no_improve > 300:
            f = best_f.copy()
            no_improve = 0

        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spec iter={k}, best={best_score:.8f}, eta={eta:.5f}, t={elapsed:.0f}s")

    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers and line search."""
    n = len(f_init)
    a = np.maximum(f_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Peak-focused pairs
        conv = fast_convolve(a)
        top_peak_idx = int(np.argmax(conv))
        js = max(0, top_peak_idx - n + 1)
        je = min(n - 1, top_peak_idx)

        n_pairs = min(n, 300)
        tried = set()

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            # Mix of peak-focused, random, and edge pairs
            r = rng.random()
            if r < 0.4:
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
            elif r < 0.7:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            else:
                i = rng.integers(0, max(2, n // 8))
                j = rng.integers(n - n // 8, n)

            if i == j:
                continue
            key = (min(i, j), max(i, j))
            if key in tried:
                continue
            tried.add(key)

            if a[i] < 1e-12 and a[j] < 1e-12:
                continue

            # Try several transfer amounts
            max_transfer = max(a[i], a[j])
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                for direction in [1, -1]:
                    t = direction * a[i] * frac if direction == 1 else -a[j] * frac
                    if direction == 1:
                        t = a[i] * frac  # move from i to j
                    else:
                        t = -a[j] * frac  # move from j to i

                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    if a_trial[i] < 0 or a_trial[j] < 0:
                        continue
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def make_chirp_init(n, seed=0):
    """Create a chirp-like initialization."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    # Quadratic chirp
    phase = 2.0 * np.pi * (t * t % n) / (2.0 * n)
    a = 1.0 + 0.3 * np.cos(phase)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_zadoff_chu_like(n, seed=0):
    """Create a Zadoff-Chu-like initialization."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    # Use multiple harmonics
    a = np.ones(n)
    for k in range(1, min(5, n // 2)):
        a += 0.1 / k * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def make_raised_cosine(n, seed=0):
    """Raised cosine window initialization."""
    t = np.arange(n)
    a = 0.5 * (1 + np.cos(2.0 * np.pi * t / n))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(250.0, rem * 0.15)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd", best_seq, best_score)

    # === Phase 5: Second LP round (on improved solution) ===
    print(f"\n=== Phase 5: LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(200.0, rem * 0.12)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play_v2(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "lp_polish", best_seq, best_score)

    # === Phase 6: Multi-start with diverse initializations ===
    print(f"\n=== Phase 6: Multi-Start (best={best_score:.8f}) ===")
    inits = [
        ("chirp1", make_chirp_init(n_parent, seed=1)),
        ("chirp2", make_chirp_init(n_parent, seed=7)),
        ("zadoff1", make_zadoff_chu_like(n_parent, seed=3)),
        ("raised_cos", make_raised_cosine(n_parent)),
    ]
    for name, init_seq in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(80.0, rem * 0.15)
        if t_alloc < 20:
            break
        # Quick LP from this init
        print(f"  Multi-start: {name}, budget={t_alloc:.0f}s")
        a_ms, sc_ms = lp_fictitious_play_v2(init_seq, t_alloc * 0.7, verbose=False)
        best_seq, best_score = check_and_update(a_ms, f"ms_{name}", best_seq, best_score)
        # Quick spectral from this init
        rem2 = deadline - time.monotonic()
        if rem2 > 30:
            a_ms2, sc_ms2 = spectral_optimization(init_seq, min(30, rem2 * 0.2), verbose=False)
            best_seq, best_score = check_and_update(a_ms2, f"ms_{name}_spec", best_seq, best_score)

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = []
    # Try nearby lengths
    for delta in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
        alt_n = n_parent + delta
        if 64 <= alt_n <= 3000 and alt_n != n_parent:
            nearby_ns.append(alt_n)
    # Also try some other lengths
    for alt_n in [512, 640, 680, 720, 750, 792, 800, 832, 896, 1024, 1280]:
        if alt_n != n_parent and 64 <= alt_n <= 3000:
            nearby_ns.append(alt_n)

    seen_ns = set()
    for alt_n in nearby_ns:
        if alt_n in seen_ns:
            continue
        seen_ns.add(alt_n)
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(60.0, rem * 0.12)
        if t_alloc < 15:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Length {alt_n}, budget={t_alloc:.0f}s")
        a_alt, sc_alt = lp_fictitious_play_v2(init_alt, t_alloc * 0.6, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_lp", best_seq, best_score)

        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_alt2, sc_alt2 = coordinate_descent(a_alt, min(20, rem2 * 0.4), verbose=False)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 8: Final polish with subgradient ===
    print(f"\n=== Phase 8: Final SGD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(120.0, rem * 0.4)
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin, sc_fin = subgradient_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin, "final_sgd", best_seq, best_score)

    # === Phase 9: Final coordinate descent ===
    print(f"\n=== Phase 9: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(80.0, rem * 0.5)
        a_fin2, sc_fin2 = coordinate_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin2, "final_cd", best_seq, best_score)

    # === Phase 10: Final LP ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(40.0, rem * 0.7)
        a_fin3, sc_fin3 = lp_fictitious_play_v2(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin3, "final_lp", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
