import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.PDLP)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def lp_fictitious_play_avg(a0, time_limit, step=None, verbose=True, label="LP-FP-avg"):
    """LP fictitious play with time-averaged iterate and optimal step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None

    # Time-averaged iterate (exponential weighting)
    avg_a = a.copy()
    # Optimal step size based on payoff range
    if step is None:
        step = 1.0 / np.sqrt(np.log(2 * n - 1))

    decay = 0.999  # exponential weight decay for averaging

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Update time-averaged iterate
        avg_a = decay * avg_a + (1 - decay) * g_norm
        avg_a = np.maximum(avg_a, 0)
        ss_avg = avg_a.sum()
        if ss_avg > 1e-15:
            avg_a = avg_a / ss_avg
            avg_score = compute_score(avg_a)
            if avg_score < best_score - 1e-12:
                best_score = avg_score
                best_a = avg_a.copy()

        # Current iterate update
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        # Adapt step size: reduce if oscillating
        if iters % 200 == 0:
            conv_check = fast_convolve(a)
            M_check = conv_check.max()
            if M_check > M * 1.05:
                step *= 0.85
            elif M_check < M * 0.95:
                step = min(step * 1.1, 0.8)

        if verbose and iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def subgradient_peak_reduction(a0, time_limit, verbose=True, label="SG-PR"):
    """Subgradient method targeting the peak of the convolution, with momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    eta = 0.02  # initial step size
    eta_min = 1e-6

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Find top-k peaks for a smoother subgradient
        k = min(5, 2 * n - 1)
        top_k = np.argsort(conv)[-k:]
        # Weight by exponential of (conv_i - M) / temperature
        temp = M * 0.1
        weights = np.exp((conv[top_k] - M) / temp)
        weights /= weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for w, idx in zip(weights, top_k):
            # grad of (a*a)_idx at position j is 2*a[idx-j] if 0<=idx-j<n
            j_start = max(0, idx - n + 1)
            j_end = min(n, idx + 1)
            for j in range(j_start, j_end):
                grad[j] += w * 2.0 * a[idx - j]

        # Momentum update
        v = beta * v + (1 - beta) * grad
        a_new = a - eta * v

        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Rejected: reduce step, reset momentum
            eta *= 0.9
            v *= 0.5
            if eta < eta_min:
                # Perturb to escape
                rng = np.random.default_rng(iters + 100000)
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                eta = 0.01

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score

def lbgfs_smoothed(a0, time_limit, T_start=100, T_end=10, verbose=True):
    """L-BFGS on smoothed objective with decreasing temperature."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    # Multiple passes with decreasing T
    for T in [T_start, T_start / 3, T_start / 9, T_end]:
        if time.monotonic() - t0 > time_limit:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = smoothed_objective_and_grad(z, T, n)
            return obj

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = smoothed_objective_and_grad(z, T, n)
            return g

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': 20000, 'disp': False}
            )
            a_out = softmax_from_z(res.x)
            score = compute_score(a_out)
            if score < best_score:
                best_score = score
                best_a = a_out.copy()
                z0 = res.x.copy()
        except Exception:
            pass

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS T={T:.0f}, score={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, verbose=True):
    """Improved coordinate descent with line search on pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved_any = False

        # Try several random pairs
        n_trials = min(500, n)
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue

            # Try transferring mass from i to j at various fractions
            base = a[i]
            best_local = best_score
            best_frac = 0.0

            for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-13:
                    best_local = sc
                    best_frac = t

            if best_frac != 0.0:
                a[i] -= best_frac
                a[j] += best_frac
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved_any = True
                no_improve = 0

        if not improved_any:
            no_improve += 1
            if no_improve > 10:
                # Add noise to escape
                noise = rng.standard_normal(n) * 0.005
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD v2 iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-FP with time averaging (main driver)
    2. Subgradient peak reduction with momentum
    3. L-BFGS smoothed objective polish
    4. Coordinate descent v2
    5. Multi-length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"\nStarting optimization. Initial score: {best_score:.8f}, n={n_parent}")

    # === Phase 1: LP-FP with time averaging (40% of budget) ===
    print(f"\n=== Phase 1: LP-FP with averaging (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 600.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_lp1, sc_lp1 = lp_fictitious_play_avg(best_seq, t1, step=0.4, label="LP-avg")
    best_seq, best_score = check_and_update(a_lp1, "lp_avg", best_seq, best_score)

    # === Phase 2: LP-FP with smaller step for convergence (15%) ===
    print(f"\n=== Phase 2: LP-FP fine-tune (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play_avg(best_seq, t2, step=0.15, label="LP-fine")
        best_seq, best_score = check_and_update(a_lp2, "lp_fine", best_seq, best_score)

    # === Phase 3: Subgradient peak reduction (15%) ===
    print(f"\n=== Phase 3: Subgradient Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if rem > 40:
        print(f"  Budget: {t3:.0f}s")
        a_sg, sc_sg = subgradient_peak_reduction(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_sg, "sg_pr", best_seq, best_score)

    # === Phase 4: L-BFGS smoothed (15%) ===
    print(f"\n=== Phase 4: L-BFGS Smoothed (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t4, T_start=200, T_end=5, verbose=True)
        best_seq, best_score = check_and_update(a_lb, "lbgfs", best_seq, best_score)

    # === Phase 5: Coordinate descent (10%) ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if rem > 30:
        print(f"  Budget: {t5:.0f}s")
        a_cd, sc_cd = coordinate_descent_v2(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd_v2", best_seq, best_score)

    # === Phase 6: Multi-length exploration (5%) ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.8f}) ===")
    for alt_n in [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4, n_parent + 8]:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t6 = min(rem * 0.25, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t6:.0f}s")
        # Quick LP-FP on new length
        a_alt, sc_alt = lp_fictitious_play_avg(init_alt, t6 * 0.6, step=0.3, verbose=False, label="alt")
        # Quick L-BFGS polish
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_alt2, sc_alt2 = lbgfs_smoothed(a_alt, rem2 * 0.8, T_start=100, T_end=5, verbose=False)
            a_alt = a_alt2
            sc_alt = sc_alt2
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 7: Final polish on best ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.5, 60.0)
        a_final, sc_final = lbgfs_smoothed(best_seq, t7, T_start=50, T_end=3, verbose=False)
        best_seq, best_score = check_and_update(a_final, "final_polish", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
