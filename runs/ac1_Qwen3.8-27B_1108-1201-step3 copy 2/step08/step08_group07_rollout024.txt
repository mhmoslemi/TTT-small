def fast_convolve(a):

    size = 2 * n - 1

def softmax_from_z(z):

    exp_z = np.exp(z_shifted)

def lse_smoothed_conv(conv, T):
    """Compute T * log(sum(exp(conv/T))) and the attention weights."""
    max_c = conv.max()
    shifted = (conv - max_c) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    lse_val = max_c + T * np.log(exp_shifted.sum())
    return lse_val, w


def compute_gradient_h(a, w, n):
    """
    Compute h[i] = 4n * sum_k w[k] * a[k-i] (cross-correlation of w and a).
    """
    size = 2 * n - 1
    fft_size = 1 << (2 * size).bit_length()
    V = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(conj(fft(w)) * fft(a)) gives sum_k w[k]*a[k+m] at position m
    # We want sum_k w[k]*a[k-i], which is cross-correlation at lag -i
    # = conj of cross-correlation at lag i
    # Let's use: corr[m] = ifft(conj(fft(w)) * fft(a))[m] = sum_k w[k]*a[(k+m) mod N]
    # We want h[i] = sum_k w[k]*a[k-i] = corr[-i] = corr[N-i]
    # For linear (no wrap), with sufficient padding:
    corr = irfft(np.conj(V) * A, fft_size)
    # corr[m] = sum_k w[k]*a[k+m] (for m in range, with wrap-around if not padded enough)
    # We want h[i] = sum_k w[k]*a[k-i] = corr[-i mod N]
    # With enough padding, h[i] = corr[fft_size - i] for i > 0, and corr[0] for i=0
    h = np.zeros(n)
    h[0] = corr[0]
    if n > 1:
        h[1:] = corr[fft_size - np.arange(1, n)]
    return 4.0 * n * h

"""
ADAM + Mirror Descent + LP Fictitious Play hybrid for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed objective in log-space with ADAM as primary driver, combined with
multiplicative weight updates and LP-based block coordinate methods for polish.
Tries multiple temperatures, learning rates, and nearby sequence lengths.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_smoothed_conv(conv, T):
    max_c = conv.max()
    shifted = (conv - max_c) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    lse_val = max_c + T * np.log(exp_shifted.sum())
    return lse_val, w


def compute_gradient_h(a, w, n):
    """Compute h[i] = 4n * sum_k w[k] * a[k-i] via FFT cross-correlation."""
    fft_size = 1 << (3 * n).bit_length()
    V = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(V) * A, fft_size)
    h = np.zeros(n)
    h[0] = corr[0]
    if n > 1:
        h[1:] = corr[fft_size - np.arange(1, n)]
    return 4.0 * n * h


def adam_optimize(a0, time_limit, T=100.0, lr=0.005, verbose=True):
    """ADAM optimization in log-space with LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t = 0

    t0 = time.monotonic()
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        _, w = lse_smoothed_conv(conv, T)
        h = compute_gradient_h(a, w, n)
        grad_z = a * (h - np.dot(h, a))

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_now = softmax_from_z(z)
        score = compute_score(a_now)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_now.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 300:
            lr *= 0.7
            stall_count = 0
            if T > 5.0:
                T *= 0.85

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ADAM t={t}, best={best_score:.8f}, lr={lr:.6f}, T={T:.1f}, time={elapsed:.0f}s")

    return best_a, best_score


def mirror_descent(a0, time_limit, eta0=0.02, verbose=True):
    """Multiplicative weight update (mirror descent on simplex)."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    eta = eta0
    t0 = time.monotonic()
    iters = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        T = max(0.01, 0.5 / (n * eta))
        _, w = lse_smoothed_conv(conv, T)
        h = compute_gradient_h(a, w, n)
        grad = h - np.dot(h, a)

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 200:
            eta *= 0.6
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iters}, best={best_score:.8f}, eta={eta:.6f}, time={elapsed:.0f}s")

    return best_a, best_score


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True):
    """LP-based fictitious play (block coordinate method)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1.0 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        if stall_count > 400:
            step = max(0.03, step * 0.6)
            stall_count = 0
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers and ternary search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        n_pairs = min(400, n)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-14:
                continue
            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]
            for frac in [0.03, 0.07, 0.15, 0.25, 0.4, 0.6]:
                t_val = frac * base
                if t_val >= a[i] - 1e-15:
                    t_val = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t_val
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        if not improved and iters > 15:
            noise = rng.standard_normal(n) * 0.0008
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        if verbose and iters % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            t8b = min(rem_sub * 0.4, t8 * 0.5)
            a_alt2, sc2 = lp_fictitious_play(a_alt, t8b, step=0.3, verbose=False)
            a_alt = a_alt2
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 9: Final coordinate descent
    print(f"\n=== Phase 9: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.4, 80.0)
        a_final_cd, sc = coordinate_descent(best_seq, t9, verbose=False)
        best_seq, best_score = check_and_update(a_final_cd, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
