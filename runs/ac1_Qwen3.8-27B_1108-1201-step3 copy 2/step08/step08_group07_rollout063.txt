Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent, LP fictitious play, simulated annealing,
and coordinate descent. Tries multiple starting points and lengths.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

        w = w / w.sum()

        # = sum_k w[k] * (a[k-i] + a[i-k]) (with a[m]=0 for m out of range)
        # This is a convolution of w with a (circular, but we need linear)
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad = 2.0 * irfft(W * A, fft_size)[:n]
        
        # Subtract mean to enforce sum constraint (approximately)
        grad = grad - np.mean(grad)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # Increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            # Decrease lr
            lr *= 0.95
        
        a = a_new
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SGD iter={iters}, best={best_score:.8f}, lr={lr:.5f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def multiplicative_update(a0, time_limit, T=50, lr_init=0.1, verbose=True):
    """Multiplicative weight update for convolution flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr = lr_init
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        c = fast_convolve(a)
        c_shifted = c / T
        c_max = c_shifted.max()
        w = np.exp(c_shifted - c_max)
        w = w / w.sum()
        
        # "Correlation" vector: h[i] = sum_k w[k] * (a[k-i] + a[i-k])
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        h = 2.0 * irfft(W * A, fft_size)[:n]
        
        # Multiplicative update: a[i] *= exp(-lr * (h[i] - mean(h)))
        h_centered = h - np.mean(h)
        a_new = a * np.exp(-lr * h_centered)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.97
        
        a = a_new
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MU iter={iters}, best={best_score:.8f}, lr={lr:.5f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def simulated_annealing(a0, time_limit, T_start=0.1, T_end=0.001, verbose=True):
    """Simulated annealing for the score function."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    total_iters_estimate = int(time_limit * 10)  # rough estimate
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Adaptive temperature
        frac = min(1.0, iters / total_iters_estimate)
        T = T_start * (T_end / T_start) ** frac
        
        # Random perturbation: move mass from one position to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Amount to move (relative to a[i])
        frac_move = rng.uniform(0.01, 0.2)
        amount = frac_move * a[i]
        
        a_trial = a.copy()
        a_trial[i] -= amount
        a_trial[j] += amount
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        
        score_trial = compute_score(a_trial)
        delta = score_trial - compute_score(a)
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
        
        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA iter={iters}, best={best_score:.8f}, T={T:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):

def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_pairs = 500
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            
            best_local = best_score
            best_t = 0.0
            base = min(a[i], 0.1)
            
            for frac in [0.01, 0.03, 0.1, 0.3, 0.5]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved:
            noise = rng.standard_normal(n) * 0.001
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, verbose=True):
    """L-BFGS on smoothed log-sum-exp objective."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func_and_grad(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_convolve(a)
        c_s = c / T
        c_max = c_s.max()
        w = np.exp(c_s - c_max)
        w = w / w.sum()
        val = 2.0 * n * T * np.log(np.sum(np.exp(c_s - c_max)) + c_max)
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(W * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return val, grad_z
    
    z0 = np.log(np.maximum(a0, 1e-30))
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        v, _ = func_and_grad(z)
        return v
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = func_and_grad(z)
        return g
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 50000, 'disp': False})
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    if verbose:
        print(f"  L-BFGS: best={best_score:.8f}")
    
    return best_a, best_score

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based smoothed gradient descent with simplex projection,
LP fictitious play with 2/(k+2) step schedule, L-BFGS polishing,
and coordinate descent. Multi-temperature annealing for the log-sum-exp
smoothing parameter. Combines these to escape local minima and converge.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_smoothing_weights(c, T):
    """Compute softmax weights over convolution values."""
    c_shifted = c / T
    c_max = c_shifted.max()
    w = np.exp(c_shifted - c_max)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s


def gradient_smoothed(a, w, fft_size):
    """
    Compute gradient of T*log(sum(exp(c/T))) w.r.t. a using FFT.
    Returns h where dF/da[i] = 2n * h[i].
    h[i] = sum_k w[k] * (a[k-i] + a[i-k])
    """
    n = len(a)
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    h = np.fft.ifft(np.conj(W) * A + W * A, fft_size).real[:n]
    return h


def smoothed_gd(a0, time_limit, T_start=200, T_end=5, lr_init=0.1,
                verbose=True, label="SGD"):
    """
    Smoothed gradient descent with simplex projection and
    adaptive temperature/step-size schedule.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr = lr_init
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    total_iters_est = 5000

    while time.monotonic() - t0 < time_limit:
        iters += 1
        frac = min(1.0, iters / total_iters_est)
        T = T_start * (T_end / T_start) ** frac
        lr_cur = lr * (T_start / T) ** 0.5

        c = fast_convolve(a)
        w = compute_smoothing_weights(c, T)
        h = gradient_smoothed(a, w, fft_size)
        grad = 2.0 * n * h

        a_new = a - lr_cur * grad
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            lr = min(lr * 1.03, 2.0)
        else:
            lr *= 0.97

        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"T={T:.1f}, lr={lr:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, verbose=True, label="LP-FP"):
    """LP fictitious play with optimal 2/(k+2) step schedule."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        step = 2.0 / (iters + 2.0)

        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 200:
            rng = np.random.default_rng(iters * 7 + 3)
            a = best_a + rng.standard_normal(n) * 0.0008
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
            iters = max(iters // 2, 1)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"step={step:.5f}, time={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=500, verbose=True):
    """L-BFGS on smoothed log-sum-exp in log-space."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_convolve(a)
        c_s = c / T
        c_max = c_s.max()
        return 2.0 * n * T * (c_max + np.log(np.exp(c_s - c_max).sum()))

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_convolve(a)
        w = compute_smoothing_weights(c, T)
        h = gradient_smoothed(a, w, fft_size)
        grad_a = 2.0 * n * h
        S = float(np.dot(grad_a, a))
        return a * (grad_a - S)

    z0 = np.log(np.maximum(a0, 1e-30))
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 60000, 'disp': False})
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    if verbose:
        print(f"  L-BFGS polish: best={best_score:.8f}")
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall_rounds = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_pairs = 400
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            if a[i] > 1e-14:
                base = a[i] * 0.5
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    t = frac * base
                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    a_trial = np.maximum(a_trial, 0)
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_t = t
            if a[j] > 1e-14:
                base = a[j] * 0.5
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    t = frac * base
                    a_trial = a.copy()
                    a_trial[j] -= t
                    a_trial[i] += t
                    a_trial = np.maximum(a_trial, 0)
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_t = -t  # negative means transfer j->i

            if best_t != 0.0:
                if best_t > 0:
                    a[i] -= best_t
                    a[j] += best_t
                else:
                    a[j] -= (-best_t)
                    a[i] += (-best_t)
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            stall_rounds += 1
            if stall_rounds > 3:
                noise = rng.standard_normal(n) * 0.001
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_rounds = 0

        if verbose and (iters := int((time.monotonic() - t0) * 2)) % 10 == 0:
            pass

    return best_a, best_score


def multiplicative_flatten(a0, time_limit, T=50, lr_init=0.1, verbose=True):
    """
    Multiplicative weight update to flatten the convolution.
    a[i] *= exp(-lr * (h[i] - mean(h)))
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr = lr_init
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        c = fast_convolve(a)
        w = compute_smoothing_weights(c, T)
        h = gradient_smoothed(a, w, fft_size)

        h_centered = h - h.mean()
        a_new = a * np.exp(-lr * h_centered)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.98

        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MulFlatten iter={iters}, best={best_score:.8f}, "
                  f"lr={lr:.4f}, time={elapsed:.0f}s")

    return best_a, best_score
