**Approach: Projected subgradient with Nesterov acceleration**

1. Normalize a to have sum = 1 (or sqrt(2n) for direct score computation)

```python
def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
```

```python
"""
Optimized subgradient + LP hybrid for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Fast Nesterov-averaged subgradient with FFT-based convolution (O(n log n) per iter)
2. Vectorized gradient computation via peak-weighted accumulation
3. Adaptive softmax temperature annealing for peak selection
4. Periodic LP refinement with small constraint sets for exact direction
5. Multi-n exploration with solution transfer via interpolation
6. Adaptive restart strategy with memory of best iterates
7. Final coordinate-descent polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length, keeping non-negativity."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def nesterov_subgradient(a_init, time_limit, top_k_init=30, top_k_max=250,
                         eps_init=0.005, eps_min=0.0005, lr_init=0.008,
                         momentum=0.85, label="NSG", restart_freq=60,
                         max_stuck=40):
    """
    Nesterov-averaged subgradient method with softmax peak weighting.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    eps = eps_init
    lr = lr_init
    top_k = top_k_init
    it = 0
    stuck = 0
    full_len = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_a = a.copy()
            stuck = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stuck += 1
        
        # Adaptive parameters
        if it % 30 == 0:
            top_k = min(int(top_k * 1.05), top_k_max, full_len)
            eps = max(eps * 0.997, eps_min)
            if stuck > 5:
                lr = max(lr * 0.97, 0.001)
        
        # Peak selection with softmax weights
        if top_k >= full_len:
            peak_idx = np.arange(full_len)
            weights = np.ones(full_len) / full_len
        else:
            peak_idx = np.argpartition(c, -top_k)[-top_k:]
            scaled = (c[peak_idx] - M) / eps
            weights = np.exp(scaled)
            ws = weights.sum()
            if ws > 1e-30:
                weights = weights / ws
            else:
                weights = np.zeros(top_k)
                weights[0] = 1.0
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for k, w in zip(peak_idx, weights):
            if w < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
        
        # Nesterov momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        # Restart if stuck
        if stuck > restart_freq and it % 10 == 0:
            noise_scale = 0.005 + 0.01 * (stuck / restart_freq)
            rng = np.random.default_rng(it * 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            stuck = 0
            lr = max(lr_init * 0.8, 0.003)
            eps = max(eps * 2.0, eps_init * 0.5)
        
        if stuck > max_stuck:
            break
        
        if it % 200 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"k={top_k} eps={eps:.5f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, 2.0 * n * best_obj


def lp_direction_fast(a, peak_indices, M, n):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns g or None.
    """
    K = len(peak_indices)
    
    # Build constraint matrix efficiently
    A = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = a[lo:hi+1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if K <= 200 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-6)
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.sum(g_val) <= 1.0 + 1e-6:
        return None
    return g_val / g_val.sum()


def lp_polish(a_init, time_limit, top_k=80, max_iters=50, label="LP"):
    """LP-based direction finding with line search."""
    t0 = time.monotonic()
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    full_len = 2 * n - 1
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        
        if M < best_obj - 1e-13:
            best_obj = M
            best_a = a.copy()
        
        # Select peaks
        k_sel = min(top_k, full_len)
        peak_idx = np.argpartition(c, -k_sel)[-k_sel:]
        
        # Solve LP
        g = lp_direction_fast(a, peak_idx, M, n)
        if g is None:
            if k_sel > 30:
                g = lp_direction_fast(a, peak_idx[:30], M, n)
            if g is None:
                break
        
        # Line search
        best_t = 0.0
        best_obj_trial = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            trial = (1 - t) * a + t * g
            trial = project_simplex(trial)
            obj = fast_convolve(trial).max()
            if obj < best_obj_trial:
                best_obj_trial = obj
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = project_simplex(a)
        
        if it % 10 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"t={best_t:.3f} elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, 2.0 * n * best_obj


def coordinate_polish(a_init, time_limit, label="CD"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0]
    
    no_improve_passes = 0
    while time.monotonic() - t0 < time_limit and no_improve_passes < 3:
        any_improved = False
        step = max(1, n // 200)
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = a[j]
            for m in multipliers:
                trial = a.copy()
                trial[j] = a[j] * m
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = a[j] * m
            if best_j_val != a[j]:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-12:
                    a = a / s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    any_improved = True
        
        if not any_improved:
            no_improve_passes += 1
            # Try finer step
            step = max(1, step // 3)
            if step == 1:
                # Try all coordinates
                for j in range(n):
                    if time.monotonic() - t0 > time_limit:
                        break
                    if a[j] < 1e-14:
                        continue
                    best_j_obj = best_obj
                    best_j_val = a[j]
                    for m in [0.7, 0.85, 1.0, 1.15, 1.3]:
                        trial = a.copy()
                        trial[j] = a[j] * m
                        ts = trial.sum()
                        if ts < 1e-12:
                            continue
                        trial = trial / ts
                        obj = fast_convolve(trial).max()
                        if obj < best_j_obj - 1e-12:
                            best_j_obj = obj
                            best_j_val = a[j] * m
                    if best_j_val != a[j]:
                        a[j] = best_j_val
                        s = a.sum()
                        if s > 1e-12:
                            a = a / s
                        if best_j_obj < best_obj - 1e-12:
                            best_obj = best_j_obj
                            best_a = a.copy()
                            any_improved = True
    
    return best_a, 2.0 * n * best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: fast subgradient -> LP refinement -> multi-n -> polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score=2.0")
    
    best_seq = parent.copy()
    best_score = p_score
    
    # ============================================================
    # Phase 1: Fast Nesterov subgradient on parent length
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Fast NSG on parent (n={len(best_seq)})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()

    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
```
