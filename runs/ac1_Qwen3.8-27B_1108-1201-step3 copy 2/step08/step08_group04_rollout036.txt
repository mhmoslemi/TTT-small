```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
```

```python
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
```

The dual problem is: `min M * sum(λ) s.t. A^T @ λ >= 1, λ >= 0`.

This dual is a linear program with variables λ (one per constraint) and the constraint A^T @ λ >= 1. The number of dual variables is K (number of constraints), and the number of dual constraints is n (one per coordinate of g).

For the primal-dual approach:
1. Maintain dual variables λ
2. Compute primal: `g_i = max(0, 1 - sum_k λ_k * A[k,i])`
3. Update dual: `λ_k <- λ_k * (1 + η * violation_k)` where `violation_k = A[k] @ g - M` (or similar multiplicative weights update)

This is the "multiplicative weights" or "follow-the-regularized-leader" approach, which converges at rate O(1/sqrt(T)) for the gap.

```python
def lp_direction_via_multiplicative_weights(f, indices, M, T=100, eta=0.1):
    """
    Approximate the LP max sum(g) s.t. A@g <= M, g >= 0
    using multiplicative weights on the dual.
    """
    n = len(f)
    K = len(indices)
    
    # Build constraint matrix A (K x n)
    # A[k, i] = f[i] if the constraint k involves a[i]
    # Constraint k: sum_{i: 0<=k-i<n} f[k-i] * g[i] <= M
    # So A[row_k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((K, n))
    for row in range(K):
        k = indices[row]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row, lo:hi+1] = f[k-hi:k-lo+1][::-1]
        # A[row, i] = f[k-i] for i in [lo, hi]
    
    # Dual: min M * sum(λ) s.t. A^T @ λ >= 1, λ >= 0
    # Using multiplicative weights:
    lam = np.ones(K) / K
    g = np.ones(n) / n
    
    for t in range(T):
        # Update dual: λ_k *= (1 + eta * (A[k] @ g - M) / M)
        violations = A @ g  # length K
        # If violation > 0, increase λ; if < 0, decrease
        # Scaling: use relative violation
        rel_viol = (violations - M) / M  # positive means constraint violated
        lam = lam * (1.0 + eta * np.maximum(rel_viol, -1.0))
        lam = np.maximum(lam, 1e-10)
        lam /= lam.sum()  # normalize (doesn't affect KKT direction)
        
        # Update primal: g_i = max(0, 1 - (A^T @ lam)_i / (sum(lam) * M / K))
        # Actually, the KKT condition for the LP is:
        # g_i = max(0, 1 - sum_k λ_k * A[k,i])
        # (with appropriate scaling of λ)
        # 
        # Let me think about this more carefully.
        # Primal: max 1^T g s.t. A g <= M, g >= 0
        # Lagrangian: 1^T g - λ^T (A g - M) - μ^T g = (1 - A^T λ - μ)^T g + M * 1^T λ
        # KKT for g_i > 0: 1 - (A^T λ)_i - μ_i = 0, i.e., (A^T λ)_i = 1 - μ_i <= 1
        # KKT for g_i = 0: (A^T λ)_i >= 1
        # So: g_i > 0 iff (A^T λ)_i < 1
        # And at optimum: g_i = max(0, ...) ... but we can't determine the exact value from λ alone.
        # 
        # Actually, for the LP, the optimum g is determined by the active constraints.
        # The multiplicative weights gives us approximate λ, from which we can determine
        # which constraints are active (λ_k > 0) and which variables are positive.
        # But we still need to solve a smaller LP for the exact values.
        # 
        # Alternatively, we can use the "FTRL" update directly on g:
        # g_i = exp(-η * (A^T @ λ)_{i-iter}) / sum_j exp(-η * (A^T @ λ)_{j-iter})
        # This is the "mirror descent" step.
        
        # Let me use the mirror descent (entropy) approach:
        # g_i = exp(-eta * (A^T @ lam)[i]) / Z
        # This automatically ensures g >= 0 and we can normalize.
        
        dual_grad = A.T @ lam  # length n
        g = np.exp(-eta * dual_grad)
        g_sum = g.sum()
        g = g / g_sum  # normalize to sum=1 for direction
    
    return g
```

```python
"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Uses a combination of:
1. Smoothed gradient descent (log-sum-exp) with Nesterov acceleration
2. Dual-based greedy direction (LP approximation without solving LP)
3. Random restarts and local search
4. Multi-resolution exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate as sp_correlate
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, epsilon, top_k=None):
    """
    Compute gradient of soft-max approximation of max(auto_convolution).
    Returns (gradient, max_value, weights).
    """
    n = len(a)
    full_len = 2 * n - 1
    b = fast_convolve(a)

    if top_k is not None and top_k < full_len:
        # Restrict to top-k peaks
        threshold_idx = np.argpartition(b, -top_k)[-top_k:]
        threshold = b[threshold_idx].min()
        mask = b >= threshold
        active_indices = np.where(mask)[0]
        b_active = b[active_indices]
        max_b = b_active.max()
        scaled = (b_active - max_b) / epsilon
        scaled = np.minimum(scaled, 50.0)
        p_active = np.exp(scaled)
        p_active /= p_active.sum()
        # Build full weight vector
        p_full = np.zeros(full_len)
        p_full[active_indices] = p_active
    else:
        p_full = np.zeros(full_len)
        max_b = b.max()
        scaled = (b - max_b) / epsilon
        scaled = np.minimum(scaled, 50.0)
        p_full = np.exp(scaled)
        p_full /= p_full.sum()

    # Compute gradient: grad[i] = sum_k p_full[k] * a[k-i]
    # Using cross-correlation
    result = sp_correlate(p_full, a, mode='full', method='fft')
    grad = result[n - 1:2 * n - 1].copy()

    return grad, b.max(), p_full


def dual_based_direction(f, top_k=50):
    """
    Compute a direction approximating the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Uses equal-weight dual: g[i] = max(0, 1 - (1/K) * sum_k f[k-i]).
    Returns normalized direction g (sum=1).
    """
    n = len(f)
    full_len = 2 * n - 1
    b = fast_convolve(f)
    M = b.max()

    if top_k < full_len:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(full_len)

    K = len(top_idx)
    # Compute sum_k f[k-i] for each i
    # f[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
    # For each i: sum_{k in top_idx, i <= k <= i+n-1} f[k-i]
    
    # Efficient computation using the same cross-correlation trick:
    # Create weight vector w with w[k] = 1 for k in top_idx, 0 otherwise
    w = np.zeros(full_len)
    w[top_idx] = 1.0 / K

    # grad[i] = sum_k w[k] * f[k-i] = correlate(w, f, 'full')[n-1:2*n-1]
    result = sp_correlate(w, f, mode='full', method='fft')
    dual_grad = result[n - 1:2 * n - 1].copy()

    # Direction: g[i] = max(0, 1 - dual_grad[i] / (M/n))
    # Normalize so that the constraint is tight: (f*g)[k] = M at active peaks
    # Scale: we want sum_k f[k-i] * g[i] / K = M, so g[i] proportional to max(0, C - dual_grad[i])
    # Choose C so that the constraint is tight: C = M / (average multiplier)
    # Simple choice: C = 1 (then scale g to make constraint tight)
    g = np.maximum(0.0, 1.0 - dual_grad * (n / M))
    g_sum = g.sum()
    if g_sum < 1e-15:
        return np.ones(n) / n
    g = g / g_sum
    return g


def smoothed_optimize(f_init, time_limit, top_k=80, eps_init=0.05, eps_min=0.005,
                      lr_init=0.02, momentum_init=0.7, label="smooth"):
    """
    Smoothed gradient descent with Nesterov acceleration.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    momentum = momentum_init
    lr = lr_init
    eps = eps_init

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    no_improve = 0
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov look-ahead
        f_y = f + momentum * v
        f_y = project_simplex(f_y)

        # Compute smoothed gradient at look-ahead point
        grad, cur_max, _ = compute_smoothed_gradient(f_y, eps, top_k)

        # Update
        v_new = momentum * v - lr * grad
        f_new = project_simplex(f + v_new)
        v = v_new

        # Evaluate
        obj = fast_convolve(f_new).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            best_score = 2.0 * n * best_obj
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.01, 0.1)
        else:
            f = f_new  # Still update f (momentum helps)
            no_improve += 1
            if no_improve > 30:
                lr *= 0.9
                no_improve = 0
            if no_improve > 100:
                # Restart from best
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0
                momentum = momentum_init * 0.8

        # Adapt parameters
        if it % 50 == 0:
            eps = max(eps * 0.97, eps_min)
            momentum = min(momentum + 0.001, 0.95)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"eps={eps:.5f} lr={lr:.5f} mom={momentum:.3f} t={elapsed:.0f}s")

    return best_f, best_score

                    no_improve = 0

            best_f = f.copy()
            best_score = 2.0 * n * best_obj
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.02)
                no_improve = 0

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_score


def coordinate_descent(f_init, time_limit, n_evals_per_coord=8):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.2, 1.5])

    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            best_j_obj = best_obj
            best_j_val = f[j]
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 1e-18:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_score


def random_search(f_init, time_limit, n_perturb=1000):
    """Random perturbation search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    rng = np.random.default_rng(42)

    it = 0
    while time.monotonic() - t0 < time_limit and it < n_perturb:
        it += 1
        # Random perturbation
        scale = 0.005 + 0.01 * (1.0 - (time.monotonic() - t0) / time_limit)
        f_pert = f + rng.standard_normal(n) * scale
        f_pert = project_simplex(f_pert)
        obj = fast_convolve(f_pert).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_pert.copy()
            best_score = 2.0 * n * best_obj
            f = f_pert.copy()

        if it % 200 == 0:
            print(f"    rand it={it} score={best_score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def symmetry_enhance(f):
    """Average with reverse to enforce symmetry."""
    return (f + f[::-1]) / 2.0
```
