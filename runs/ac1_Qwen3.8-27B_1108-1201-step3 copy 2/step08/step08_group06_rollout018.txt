"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent (entropy regularizer) with adaptive log-sum-exp smoothing
2. LP-based fictitious play with dual-variable-guided directional updates
3. Subgradient method with Polyak step size and geometric averaging
4. Coordinate descent with exact line search for final polish
5. Length exploration and multiple diverse starting points
Key innovations over previous approach:
- Nesterov momentum for faster convergence of mirror descent
- Dual-variable-guided gradient (focus on active constraints from LP dual)
- Geometric (log-space) averaging of iterates for better fictitious play
- Adaptive temperature schedule for log-sum-exp smoothing
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def gradient_lse(a, T):
    """
    Compute gradient of 2*n*LSE_T(conv(a,a)) where LSE_T(x)=T*log(sum(exp(x/T))).
    Returns (gradient, current_max).
    """
    n = len(a)
    c = fast_convolve(a)
    c_shifted = c / T
    c_max = c_shifted.max()
    w = np.exp(c_shifted - c_max)
    w = w / w.sum()
    # grad[m] = 4n * sum_j a[j] * w[m+j]
    # = 4n * conv(a, w_rev)[2n-2-m]
    size = 2 * n - 1
    # conv of a (len n) and w_rev (len 2n-1) has length 3n-2
    fft_size = 1 << (3 * n - 3).bit_length()
    A = rfft(a, fft_size)
    Wrev = rfft(w[::-1], fft_size)
    conv_result = irfft(A * Wrev, fft_size)[:3 * n - 2]
    grad = 4.0 * n * conv_result[::-1]
    grad = grad[:n]
    return grad, c.max()


def gradient_weighted(a, w_weights):
    """
    Compute gradient of 2n * sum_k w_weights[k] * c(k) / S^2 (S=1 assumed).
    w_weights should sum to 1.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 3).bit_length()
    A = rfft(a, fft_size)
    Wrev = rfft(w_weights[::-1], fft_size)
    conv_result = irfft(A * Wrev, fft_size)[:3 * n - 2]
    grad = 4.0 * n * conv_result[::-1]
    grad = grad[:n]
    return grad


def nesterov_mirror_descent(a0, time_limit, T_init=500.0, T_min=5.0, alpha=0.5, verbose=True):
    """Nesterov-accelerated mirror descent with adaptive log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).copy()
    a = a / a.sum()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    total_iters = int(max(100, time_limit * 50))  # estimate
    k = 0
    
    while time.monotonic() - t0 < time_limit:
        k += 1
        # Adaptive temperature: geometric decay
        frac = min(1.0, k / total_iters)
        T = T_init * (T_min / T_init) ** frac
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 1e-30)
        s = y.sum()
        if s < 1e-30:
            y = a.copy()
        else:
            y = y / s
        
        # Compute gradient at y
        grad, _ = gradient_lse(y, T)
        
        # Mirror descent step: a_new_j ∝ y_j * exp(-eta * grad_j)
        eta = alpha * T / (2.0 * n)
        log_update = np.log(y) - eta * grad
        log_max = log_update.max()
        a_new = np.exp(log_update - log_max)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = y.copy()
        else:
            a_new = a_new / a_new_sum
        
        a_prev = a
        a = a_new
        t = t_new
        
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NMD iter={k}, T={T:.1f}, score={cur_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def subgradient_mirror_descent(a0, time_limit, verbose=True):
    """Simple subgradient mirror descent with Polyak step size."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Geometric average (log-space)
    log_sum = np.zeros(n)
    K = 0
    
    # Estimate upper bound for Polyak step
    # Score upper bound: start with 2 (uniform)
    score_UB = 2.0
    
    k = 0
    while time.monotonic() - t0 < time_limit:
        k += 1
        c = fast_convolve(a)
        c_max = c.max()
        peak_idx = np.argmax(c)
        
        # Subgradient at the peak
        # grad[m] = 2 * c[peak_idx] derivative... 
        # Actually, subgradient of max_k c(k) is (dc(peak_idx)/da) / S^2 * 2n
        # dc(peak_idx)/da_m = 2*a[peak_idx - m] if 0<=peak_idx-m<n
        idx = peak_idx  # the peak position
        grad_raw = 2.0 * np.zeros(n)
        j_start = max(0, idx - n + 1)
        j_end = min(n - 1, idx)
        for j in range(j_start, j_end + 1):
            k_offset = idx - j  # 0 <= k_offset < n
            grad_raw[j] = 2.0 * a[k_offset]
        
        # Normalize: the actual subgradient of 2n*c_max/S^2 (S=1)
        grad = 2.0 * n * grad_raw
        
        # Mirror descent step with fixed step size
        # Use decreasing step size
        eta = 0.1 / (np.sqrt(k + 1) * np.sqrt(n))
        log_update = np.log(a) - eta * grad
        log_max = log_update.max()
        a_new = np.exp(log_update - log_max)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        K += 1
        log_sum += np.log(np.maximum(a, 1e-30))
        
        # Check geometric average
        if K % 10 == 0:
            geo_a = np.exp(log_sum / K)
            geo_a = geo_a / geo_a.sum()
            geo_score = compute_score(geo_a)
            if geo_score < best_score:
                best_score = geo_score
                best_a = geo_a.copy()
            
            # Also check current
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        
        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            cur_score = compute_score(a)
            print(f"  SGD iter={k}, score={cur_score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def build_C_matrix(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. C*f... wait, C*g <= M, g>=0.
    Also extract dual variables."""
    n = len(f)
    C = build_C_matrix(f)
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None, None
    g = np.maximum(res.x, 0)
    # Try to get dual variables (not directly available from linprog)
    # We'll estimate them from which constraints are tight
    tight_mask = (C @ g >= M * (1 - 1e-8))
    dual_est = np.zeros(2 * n - 1)
    dual_est[tight_mask] = 1.0
    return g, dual_est

def greedy_lp_direction(f, M):
    """Greedy approximation to the LP direction."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                if slack[idx] < 0:
                    max_gj = 0
                    break
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj) if np.isfinite(max_gj) else 0.0
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Peak-focused pairs
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        
        n_pairs = 200
        pairs = set()
        
        # Pairs involving peak contributors
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(n_pairs // 2):
            i = rng.integers(js, je + 1)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t_val):
                a[i] -= t_val
                a[j] += t_val
                if a[i] < 0 or a[j] < 0:
                    a[i] += t_val
                    a[j] -= t_val
                    return 1e10
                ss = a.sum()
                if ss < 1e-15:
                    a[i] += t_val
                    a[j] -= t_val
                    return 1e10
                a_n = a / ss
                sn = 2.0 * n * fast_convolve(a_n).max()
                a[i] += t_val
                a[j] -= t_val
                return sn
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f
