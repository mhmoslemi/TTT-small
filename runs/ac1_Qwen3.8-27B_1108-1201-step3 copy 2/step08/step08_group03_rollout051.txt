   ```python
   import numpy as np
   from scipy.fft import rfft, irfft
   import time

   def fast_convolve(a):
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = rfft(a, fft_size)
       return irfft(A * A, fft_size)[:size]

   def project_simplex(v):
       n = len(v)
       if n == 1:
           return np.array([max(float(v[0]), 0.0)])
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho_vals = np.arange(1, n + 1)
       cond = u * rho_vals > (cssv - 1.0)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)

   def compute_subgradient(a, active_indices):
       """Compute subgradient of max_{k in active} (a*a)_k."""
       n = len(a)
       g = np.zeros(n)
       for k in active_indices:
           # d(a*a)_k / da_j = a_{j+k} + a_{j-k} (with bounds)
           # For j+k in [0, n-1]: contribution a[j+k]
           for j in range(n):
               if j + k < n:
                   g[j] += a[j + k]
               if j - k >= 0:
                   g[j] += a[j - k]
       g /= len(active_indices)
       return 2.0 * g  # factor of 2 from the quadratic
   ```

   Fast subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
   Uses accelerated projected subgradient with adaptive active set,
   multi-restart, length exploration, and IAA diversification.
   """

   import numpy as np
   from scipy.fft import rfft, irfft
   import time


   def fast_convolve(a):
       """Linear auto-convolution via FFT."""
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = rfft(a, fft_size)
       return irfft(A * A, fft_size)[:size]


   def project_simplex(v):
       """Euclidean projection onto {x >= 0, sum(x) = 1}."""
       n = len(v)
       if n == 1:
           return np.array([max(float(v[0]), 0.0)])
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho_vals = np.arange(1, n + 1)
       cond = u * rho_vals > (cssv - 1.0)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)


   def compute_subgradient(a, indices):
       """
       Compute subgradient of max_{k in indices} (a*a)_k.
       Returns (1/|indices|) * sum_{k in indices} grad (a*a)_k
       """
       n = len(a)
       g = np.zeros(n)
       for k in indices:
           lo = max(0, k - n + 1)
           hi = min(n, k + 1)
           g[lo:hi] += 2.0 * a[k - (np.arange(lo, hi))][::-1]
       g /= len(indices)
       return g

"""
Fast subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated projected subgradient with adaptive active set, multi-restart
diversification, length exploration, and IAA-based smoothing. No LP solving.
Key advantages: ~100x faster iterations than LP-based methods, enabling far
more thorough exploration of the landscape within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, indices):
    """
    Compute (1/|indices|) * sum_{k in indices} grad (a*a)_k.
    Each grad (a*a)_k has components: 2*a_{k-j} if valid, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    ar = a[::-1]
    for k in indices:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx_start = n - 1 - k + lo
        idx_end = n - 1 - k + hi
        g[lo:hi] += 2.0 * ar[idx_start:idx_end]
    g /= len(indices)
    return g


def subgradient_optimize(a_init, time_limit, initial_lr=None,
                         initial_K=25, max_K=300, K_growth_per_100=3,
                         momentum_init=0.9, restart_patience=300,
                         label="sg", seed=0):
    """
    Accelerated projected subgradient with adaptive active set and restarts.
    """
    t0 = time.monotonic()
    n = len(a_init)
    if initial_lr is None:
        initial_lr = 1.0 / np.sqrt(n) * 0.5
    lr = initial_lr
    a = a_init.copy()
    v = np.zeros(n)
    beta = momentum_init
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    f_best_ever = best_f

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = fast_convolve(a)
        f_val = c.max()

        if f_val < best_f - 1e-14:
            best_f = f_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set size
        K = min(initial_K + (it * K_growth_per_100) // 100, max_K, len(c))
        if K >= len(c):
            active = np.arange(len(c))
        else:
            active = np.argpartition(c, -K)[-K:]

        g = compute_subgradient(a, active)

        # Adaptive learning rate
        if no_improve >= restart_patience:
            # Hard restart from best with perturbation
            noise_scale = 0.003 + 0.005 * (it % 5) / 5
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = initial_lr * (1.0 + 0.1 * (it % 3))
            beta = momentum_init
            no_improve = 0
        elif no_improve > 80:
            lr *= 0.93
        elif no_improve > 30:
            lr *= 0.97

        # Clamp lr
        lr = max(lr, 1e-8)
        lr = min(lr, 0.1)

        # Momentum update
        v = beta * v - lr * g
        a = project_simplex(a + v)

        # Slowly decay momentum
        if it % 2000 == 0:
            beta = max(0.82, beta * 0.995)

        if it % 10000 == 0:
            score = 2.0 * n * best_f
            rem = time_limit - (time.monotonic() - t0)
            print(f"    {label} it={it} f={best_f:.10f} score={score:.6f} "
                  f"K={len(active)} lr={lr:.6f} rem={rem:.0f}s")

    return best_a, best_f


def iaa_smooth(a, n_iters=100, peak_fraction=0.3):
    """
    IAA-based diversification: iteratively flatten the auto-convolution
    in the frequency domain while preserving non-negativity.
    """
    n = len(a)
    f = a.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for _ in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        peak_val = R.max()
        if peak_val < 1e-15:
            break
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-30:
            R_target *= energy / R_target.sum()

        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = np.ones_like(F)
        mask = abs_F > 1e-30
        phase[mask] = F[mask] / abs_F[mask]
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = a.copy()
            break

    return f


def coordinate_polish(a_init, time_limit, n_sweeps=5):
    """
    Coordinate descent with multiplicative updates for final polish.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_f = fast_convolve(a).max()
    step = max(1, n // 600)
    multipliers = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return a, best_f
            if a[j] < 1e-14:
                continue
            orig = a[j]
            best_val = orig
            best_obj_j = best_f
            for mult in multipliers:
                a[j] = orig * mult
                s = a.sum()
                if s < 1e-10:
                    a[j] = orig
                    continue
                a_norm = a / s
                obj = fast_convolve(a_norm).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = orig * mult
            a[j] = best_val
            s = a.sum()
            if s > 1e-10:
                a /= s
            if best_obj_j < best_f - 1e-14:
                best_f = best_obj_j
                improved = True
        if not improved:
            break
    return a, best_f


def make_initializations(n, seed=42):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    sigma = 0.2
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    g /= g.sum()
    inits.append(g)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    rc = np.maximum(rc, 0)
    s = rc.sum()
    if s > 0:
        inits.append(rc / s)

    # Hamming-like
    h = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    h = np.maximum(h, 0)
    s = h.sum()
    if s > 0:
        inits.append(h / s)

    # Triangular
    tri = 1 - np.abs(2 * x - 1)
    tri = np.maximum(tri, 1e-6)
    inits.append(tri / tri.sum())

    # Beta-distribution-like (skewed)
    rng = np.random.default_rng(seed)
    beta_a, beta_b = 2.0, 5.0
    rv = rng.beta(beta_a, beta_b, n)
    rv = np.maximum(rv, 1e-8)
    inits.append(rv / rv.sum())

    # Random smooth
    n_coarse = max(8, n // 50)
    coarse = rng.dirichlet(np.ones(n_coarse) * 2)
    from scipy.interpolate import interp1d
    new_x = np.linspace(0, 1, n)
    old_x = np.linspace(0, 1, n_coarse)
    smooth = np.interp(new_x, old_x, coarse)
    smooth = np.maximum(smooth, 0)
    s = smooth.sum()
    if s > 1e-10:
        inits.append(smooth / s)

    return inits
