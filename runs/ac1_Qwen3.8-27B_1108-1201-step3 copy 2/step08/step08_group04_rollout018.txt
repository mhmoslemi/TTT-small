Let me now think about the overall algorithm more carefully.

**Algorithm: Log-Sum-Exp Gradient Descent with Simplex Projection**

1. Initialize `a` (asymmetric, e.g., exponential decay, or random)
2. For each iteration:
   a. Compute `b = conv(a, a)`
   b. Compute softmax weights `w[k] = exp(beta * b[k]) / sum exp(beta * b[k])`
   c. Compute gradient `grad = correlate(w, a)[:n]` (times beta for the full log-sum-exp gradient)
   d. Update: `a = a - lr * grad`
   e. Project to simplex: `a = project_simplex(a)`
3. Adaptively increase `beta` over time
4. Use Adam or RMSProp for adaptive learning rate

Wait, I need to be more careful about the gradient. The objective is:
`f(a) = (1/beta) * log(sum_k exp(beta * b[k]))` where `b = conv(a,a)`

`df/d(a[j]) = (1/beta) * sum_k exp(beta*b[k]) * (db[k]/d(a[j])) / sum_k exp(beta*b[k])`
`= sum_k w[k] * (db[k]/d(a[j]))`
where `w[k] = exp(beta*b[k]) / sum exp(beta*b[k])`

And `db[k]/d(a[j]) = a[k-j]` (if `0 <= k-j < n`).

So `grad[j] = sum_k w[k] * a[k-j]` (summed over valid k).

This is exactly the correlation I described above. Good.

Now, the factor of `beta` cancels out in the gradient! The gradient of the log-sum-exp is the same regardless of `beta` (it's the softmax-weighted average of the individual gradients). `beta` only affects the "sharpness" of the weighting.

So the algorithm is:
1. Compute `b = conv(a,a)`
2. Compute `w = softmax(beta * b)`
3. Compute `grad = correlation(w, a)[:n]`
4. Update `a` using `grad`

For large `beta`, only the max peak matters (gradient descent on the max).
For small `beta`, all peaks matter equally (gradient of the sum).

**Key insight**: We should use a moderate `beta` that balances between focusing on the max peak and considering all peaks. As we get closer to the optimum, we can increase `beta` to focus more on the remaining peaks.

Let me also think about the computational cost:
- `fast_convolve`: O(n log n) via FFT
- `correlate`: O(n log n) via FFT (or O(n^2) directly for small n)
- `project_simplex`: O(n log n) for sorting

So each iteration is O(n log n), which is fast. We can do many iterations in the time budget.

For n=1000, each iteration might take ~1ms, so we can do ~1,000,000 iterations in 1000 seconds. That's more than enough.

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp gradient descent with simplex projection, combined with
LP-based polishing and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_correlation(w, a):
    """Compute correlation(w, a)[:len(a)] efficiently via FFT."""
    n = len(a)
    m = len(w)
    fft_size = 1 << (m + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # correlation(w, a)[k] = sum_n w[n+k] * a[n]
    # This equals ifft(conj(W) * A) in the circular case
    # For linear, we need sufficient padding
    result = irfft(np.conj(W) * A, fft_size)
    return result[:n]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(target_sum, 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(a, beta):
    """Compute gradient of log-sum-exp(conv(a,a))."""
    b = fast_convolve(a)
    max_b = b.max()
    scaled = beta * (b - max_b)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    grad = fast_correlation(w, a)
    return grad

def lse_gradient_descent(a_init, time_limit, beta_init=5.0, beta_final=200.0,
                          lr_init=0.01, momentum=0.9, label="lse"):
    """
    Log-sum-exp gradient descent with simplex projection.
    Returns (best_a, best_score).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
        s = 1.0

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj

    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr = lr_init
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_adam += 1

        # Adaptive beta: start moderate, increase over time
        progress = min(1.0, it / 5000)
        beta = beta_init + (beta_final - beta_init) * progress

        # Compute gradient
        grad = compute_grad(a, beta)

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - lr * update
        a_new = project_simplex(a_new)

        # Check for improvement
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1

        a = a_new

        # Adaptive learning rate
        if stuck > 50:
            lr *= 0.9
            stuck = 0
            if lr < 1e-6:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                lr = lr_init * 0.5

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} beta={beta:.1f} obj={best_obj:.12f} "
                  f"score={best_score:.6f} lr={lr:.6f} t={elapsed:.0f}s")

    return best_a, best_score

def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=500,

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()

    if s > 1e-15:

        s = 1.0

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    stuck = 0
    top_k = top_k_init
    full_len = 2 * n - 1
    momentum_dir = np.zeros(n)
    momentum_beta = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1

        if it % 10 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        b = fast_convolve(f)
        M = b.max()

        if top_k < full_len:
            indices = np.argpartition(b, -top_k)[-top_k:]
            indices = np.sort(indices)
        else:
            indices = np.arange(full_len)

        # Build constraint matrix
        K = len(indices)
        A_mat = np.zeros((K, n))
        for i in range(K):
            k = indices[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[i, lo:hi + 1] = f[lo:hi + 1]

        # Solve LP: max sum(g) s.t. g >= 0, A@g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A_mat @ g_var <= M])
        
        try:
            if K <= 1000 and n <= 2000:
                prob.solve(solver=cp.PDLP, max_iter=10000)
            else:
                prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
            except Exception:
                stuck += 2
                if stuck > 30:
                    break
                continue

        if g_var.value is None:
            stuck += 1
            if stuck > 30:
                break
            continue

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            stuck += 1
            if stuck > 30:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 20:
                break
            top_k = min(top_k + 20, max(top_k_max, full_len))
            continue

        h = g / S_g
        momentum_dir = momentum_beta * momentum_dir + (1 - momentum_beta) * h
        h_mom = momentum_dir / momentum_dir.sum()

        # Line search
        best_t = 0.0
        best_obj_trial = M
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * h_mom
            f_trial = np.maximum(f_trial, 0.0)
            s_t = f_trial.sum()
            if s_t < 1e-15:
                continue
            f_trial = f_trial / s_t
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = t
                f_best_trial = f_trial.copy()

        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue

        f = f_best_trial
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if stuck > 15 and it % 20 < 3:
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            stuck = 0
            momentum_dir = np.zeros(n)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0

def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7]
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0

    return best_f, best_score

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def random_asymmetric_init(n, seed=42):
    """Generate a random asymmetric initial sequence."""
    rng = np.random.default_rng(seed)
    # Exponential decay with noise
    t = np.linspace(0, 1, n)
    a = np.exp(-3 * t) + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.1)
    a = a / a.sum()
    return a

def fourier_init(n, seed=42):
    """Generate initial sequence using Fourier coefficients."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    # Add some Fourier components
    n_modes = min(10, n // 4)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
