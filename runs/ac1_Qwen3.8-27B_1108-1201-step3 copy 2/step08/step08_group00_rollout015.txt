"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive p-norm L-BFGS with
efficient gradient via correlation, LP-based Nesterov alternating optimization,
multi-scale length exploration, and aggressive local refinement.
Key innovations: p-norm smoothing (vs LogSumExp), efficient O(n log n) gradient
via FFT correlation, adaptive active-set LP, and structured multi-length search.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def p_norm_objective_value(a, p):
    """Compute (sum c_k^p)^(1/p) for c = convolve(a, a)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 0.0)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def p_norm_gradient_a(a, p):
    """Gradient of (sum c_k^p)^(1/p) w.r.t. a, where c = convolve(a,a).
    Returns g_a such that g_a[i] = d/d(a_i) (sum c_k^p)^(1/p)."""
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(a)
    # factor = (sum c_k^p)^(1/p - 1)
    factor = S ** (1.0 / p - 1.0)
    # w_k = c_k^(p-1)
    w = c_pos ** (p - 1.0)
    # g_a[i] = factor * sum_k w_k * d(c_k)/d(a_i) = factor * 2 * correlate(w, a)[i]
    # correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j]
    n = len(a)
    corr = np.correlate(w, a, mode='valid')
    # The 'valid' mode gives length len(w) - len(a) + 1 = (2n-1) - n + 1 = n
    g_a = 2.0 * factor * corr
    return g_a


def p_norm_smoothing_z(z, p):
    """Objective in log-domain: 2n * (sum c_k^p)^(1/p) where a = softmax(z)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return 2.0 * len(a) * val


def p_norm_gradient_z(z, p):
    """Gradient of 2n * (sum c_k^p)^(1/p) w.r.t. log-domain z (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    c = fast_convolve(a)
    c_pos = np.maximum(c, 1e-300)
    p_val = c_pos ** p
    S = np.sum(p_val)
    if S < 1e-300:
        return np.zeros_like(z)
    factor = S ** (1.0 / p - 1.0)
    w = c_pos ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    g_a = 4.0 * n * factor * corr
    # Gradient w.r.t. z: a * (g_a - dot(g_a, a))
    return a * (g_a - np.dot(g_a, a))


def p_norm_lbfgs_refine(a0, time_limit, p_start=20.0, p_end=500.0, n_stages=8):
    """L-BFGS with adaptive p-norm smoothing schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z = z - z.max()  # center for numerical stability
    t0 = time.monotonic()

    temps = np.geomspace(p_start, p_end, n_stages)
    iters_per_stage = max(100, 8000 // n_stages)

    for stage_p in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return p_norm_smoothing_z(z_val, stage_p)

        def jac(z_val):
            return p_norm_gradient_z(z_val, stage_p)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate current score
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        score = 2.0 * n * fast_convolve(a_cur).max()
        elapsed = time.monotonic() - t0
        print(f"      p-norm L-BFGS p={stage_p:.1f} ({elapsed:.0f}s): score={score:.6f}")

    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a


def build_conv_matrix_sparse(a):
    """Build the convolution matrix T where T @ b = convolve(a, b).
    Returns (size, n) matrix."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_stall_score = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_stall_score - 1e-7:
                stall_count = 0
                prev_stall_score = best_score
            else:
                stall_count += 1
            if stall_count >= 4:
                rng = np.random.default_rng(it + 42)
                noise_scale = 0.02 + 0.01 * stall_count
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
                prev_stall_score = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def local_refinement(a0, time_limit, n_steps=20000):
    """Aggressive local refinement: coordinate descent + pair adjustments + peak shaving."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf

        # Find top peaks
        peak_indices = np.argsort(b_work)[::-1][:5]
        peak_val = b_work[peak_indices[0]]
        near_threshold = peak_val - 0.05 * peak_val
        near_peaks = [k for k in range(len(b_work)) if b_work[k] >= near_threshold][:8]

        improved = False

        # Strategy 1: Single coordinate reduction on top contributors
        contrib_scores = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) > 0:
                contribs = a[idx] * a[k - idx]
                contrib_scores[idx] += contribs

        top_coords = np.argsort(contrib_scores)[::-1][:15]

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                if a_trial[i] < 0:
                    a_trial[i] = 0
                    diff = a[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Pair reduction at peak
        if not improved:
            peak_k = peak_indices[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(contribs)[::-1][:6]
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n or j < 0:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 0.01:
                            continue
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Small random perturbation
        if not improved and step % 100 == 0:
            rng = np.random.default_rng(step + 7)
            n_pert = rng.integers(1, 5)
            indices = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for i in indices:
                a_trial[i] *= (1.0 + rng.uniform(-0.05, 0.05))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalRefine step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng, variant=0):
    """Create structured initial sequence of length n."""
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif variant == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif variant == 3:
        # Bimodal: large at ends
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x) ** 2
    elif variant == 4:
        # Gaussian at center
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely using the actual evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_state, a, label):
    """Try to update the best state with a new candidate."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return True
    return False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining p-norm L-BFGS, Nesterov LP,
    local refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'seq': None}

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state['seq'] = parent.copy()
            best_state['score'] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        parent = structured_init(n_init, rng, variant=4)
        best_state['seq'] = parent.copy()
        best_state['score'] = evaluate_safe(parent)
        print(f"Default init: n={n_init}, score={best_state['score']:.6f}")

    n_best = len(best_state['seq'])

    # === Phase 1: p-norm L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: p-norm L-BFGS n={n_best} (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, p: 15 -> 400")
        a_pn = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=15.0, p_end=400.0, n_stages=9)
        try_update(best_state, a_pn, "p_norm_lbfgs_1")

    # === Phase 2: Nesterov LP ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=100000)
        try_update(best_state, a_lp, "nesterov_lp_1")

    # === Phase 3: p-norm L-BFGS round 2 (higher p) ===
    print(f"\n=== Phase 3: p-norm L-BFGS r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, p: 100 -> 1000")
        a_pn2 = p_norm_lbfgs_refine(best_state['seq'], t_alloc, p_start=100.0, p_end=1000.0, n_stages=6)
        try_update(best_state, a_pn2, "p_norm_lbfgs_2")

    # === Phase 4: Local refinement ===
    print(f"\n=== Phase 4: Local Refinement (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_lr, _ = local_refinement(best_state['seq'], t_alloc, n_steps=30000)
        try_update(best_state, a_lr, "local_refine_1")

    # === Phase 5: Nesterov LP round 2 ===
    print(f"\n=== Phase 5: Nesterov LP r2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_lp2, _ = nesterov_lp_refine(best_state['seq'], t_alloc, solver=solver, n_max_iters=60000)
        try_update(best_state, a_lp2, "nesterov_lp_2")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_state['score']:.6f}) ===")
    candidate_lengths = set()
    base = n_best
    for delta in [-96, -64, -32, 32, 64, 96, 128]:
        alt_n = base + delta
        if 512 <= alt_n <= 2048:
            candidate_lengths.add(alt_n)
    # Also try some round numbers
    for rn in [768, 896, 1024, 1152, 1280]:
        if 512 <= rn <= 2048 and abs(rn - base) > 32:
            candidate_lengths.add(rn)

    candidate_lengths = sorted(candidate_lengths)
    # Prioritize lengths close to current best
    candidate_lengths.sort(key=lambda x: abs(x - n_best))

    n_explored = 0
    for alt_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80 or n_explored >= 3:
            break
        n_explored += 1
        t_total = min(rem * 0.18, 180.0)
        if t_total < 40:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        # Initialize by interpolation
        init_alt = interpolate_sequence(best_state['seq'], alt_n)

        # Quick p-norm L-BFGS
        t_pn = t_total * 0.5
        a_alt = p_norm_lbfgs_refine(init_alt, t_pn, p_start=30.0, p_end=200.0, n_stages=5)
        improved = try_update(best_state, a_alt, f"pn_n{alt_n}")

        # If improved, do LP refinement
        if improved:
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.15, 100.0)
                a_alt2, _ = nesterov_lp_refine(best_state['seq'], t_lp, solver=solver, n_max_iters=30000)
                try_update(best_state, a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.15, 150.0)
        n_cur = len(best_state['seq'])

        for noise_scale in [0.03, 0.07, 0.12]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            perturbed = best_state['seq'] + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_sub = min(rem * 0.3, 60.0)
            a_pert = p_norm_lbfgs_refine(perturbed, t_sub, p_start=50.0, p_end=300.0, n_stages=5)
            try_update(best_state, a_pert, f"pert_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.25, 80.0)
        a_fin = p_norm_lbfgs_refine(best_state['seq'], t_fin, p_start=200.0, p_end=800.0, n_stages=4)
        try_update(best_state, a_fin, "final_pnorm")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lr = min(rem * 0.20, 50.0)
        a_lr2, _ = local_refinement(best_state['seq'], t_lr, n_steps=15000)
        try_update(best_state, a_lr2, "final_local")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.30, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_state['seq'], t_lp_fin, solver=solver, n_max_iters=20000)
        try_update(best_state, a_lp_fin, "final_lp")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    elapsed_total = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])}, time={elapsed_total:.0f}s)")
    return [float(x) for x in best_state['seq']]
