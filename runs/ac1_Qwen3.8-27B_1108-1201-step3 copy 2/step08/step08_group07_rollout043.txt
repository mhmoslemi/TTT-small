Enhanced multi-phase optimizer for minimum peak autocorrelation sequences.
Combines LP best-response (alternative formulation), Nesterov-accelerated
subgradient, adaptive-temperature L-BFGS, and spectral iterations.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_min_cross(f, time_budget=5.0):
    """Solve: min t s.t. C(f) @ g <= t, sum(g) = 1, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    C = build_conv_matrix(f)
    constraints = [C @ g <= t, cp.sum(g) == 1]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0)


def solve_lp_max_sum(f, M, time_budget=5.0):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

"""
Multi-phase optimizer for minimum peak linear autocorrelation.
Key innovations over prior approach:
1. Alternative LP best-response (minimize max cross-convolution)
2. Nesterov-accelerated projected subgradient on the hard max
3. Adaptive-temperature L-BFGS in log-space with warm restarts
4. Spectral amplitude iteration (generalized Bussgang)
5. Multi-length exploration and diverse restarts
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_max_sum(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else None
    if g.value is None:
        return prev_g if prev_g is not None else None
    return np.maximum(g.value, 0)


def solve_lp_min_cross(f):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    C = build_conv_matrix(f)
    constraints = [C @ g <= t, cp.sum(g) == 1]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None, None
    if g.value is None:
        return None, None
    t_val = t.value if t.value is not None else np.max(C @ g.value)
    return np.maximum(g.value, 0), t_val


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def hard_subgradient(a, conv=None):
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(min(n, k + 1)):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k


def nesterov_subgradient(a0, time_limit, num_iters=20000, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    a_prev = a.copy()
    base_alpha = 1.0 / np.sqrt(n)
    momentum = 0.0

    for t in range(num_iters):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_convolve(y)
        g, k = hard_subgradient(y, conv_y)
        # Projected gradient step
        alpha = base_alpha / np.sqrt(t + 1)
        a_new = project_simplex(y - alpha * g)
        # Nesterov update
        momentum = (t / (t + 3))
        y = a_new + momentum * (a_new - a_prev)
        y = project_simplex(y)
        a_prev = a_new
        a = a_new
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS iter={t}, best={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score


def smoothed_lse_objective(z, T, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    max_c = conv.max()
    lse = max_c + np.log(np.exp(conv - max_c).sum())
    return 2.0 * n / T * lse


def smoothed_lse_gradient(z, T, n):
    a = np.exp(z - z.max())
    a = a / a.sum()
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tc = Tconv.max()
    w = np.exp(Tconv - max_Tc)
    w = w / w.sum()
    # grad_a[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        if end > 0:
            grad_a[i] = 2.0 * np.dot(w[i:i + end], a[:end])
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T_seq=None, maxiter=3000):
    n = len(a0)
    if T_seq is None:
        T_seq = [1000, 500, 200, 100]
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    z = z0.copy()

    for T in T_seq:
        if time.monotonic() - t0 > time_limit:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return smoothed_lse_objective(z, T, n)

        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return smoothed_lse_gradient(z, T, n)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            a_out = np.exp(res.x - res.x.max())
            a_out = a_out / a_out.sum()
            score = compute_score(a_out)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_out.copy()
            z = res.x
        except Exception:
            pass

    return best_a, best_score


def spectral_iteration(a0, time_limit, tau=0.3, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, use_min_cross=False, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        if use_min_cross:
            g, _ = solve_lp_min_cross(a)
        else:
            g = solve_lp_max_sum(a, M, prev_g=prev_g)

        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        prev_g = g

        if stall_count > 300:
            step = max(0.05, step * 0.6)
            rng = np.random.default_rng(iters * 7 + 1)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            mode = "min_cross" if use_min_cross else "max_sum"
            print(f"    LP({mode}) iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]
            for frac in [0.1, 0.3, 0.5, 0.7]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def diverse_restart(propose_fn, a_best, n, try_count=5, time_per=10.0, seed=0):
    rng = np.random.default_rng(seed)
    best_a = a_best.copy()
    best_score = compute_score(a_best)
    for _ in range(try_count):
        noise_scale = 0.05 + 0.1 * rng.random()
        a_pert = a_best + rng.standard_normal(n) * noise_scale
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s
        a_new, sc_new = propose_fn(a_pert, time_per)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
    return best_a, best_score

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    total_budget = budget_s - 30.0
    print(f"Total budget: {total_budget:.0f}s, n={n_parent}")

    # === Phase 1: LP max-sum (fast progress) ===
    print(f"\n=== Phase 1: LP max-sum step=0.5 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    print(f"  Budget: {t1:.0f}s")
    a1, sc1 = lp_fictitious_play(best_seq, t1, step=0.5, use_min_cross=False)
    best_seq, best_score = check_and_update(a1, "lp_ms_0.5", best_seq, best_score)

    # === Phase 2: LP min-cross (alternative best-response) ===
    print(f"\n=== Phase 2: LP min-cross (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a2, sc2 = lp_fictitious_play(best_seq, t2, step=0.3, use_min_cross=True)
        best_seq, best_score = check_and_update(a2, "lp_min_cross", best_seq, best_score)

    # === Phase 3: Nesterov accelerated subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient (best={best_score:.8f}) ===")

    # === Phase 4: L-BFGS with adaptive temperature ===
    print(f"\n=== Phase 4: L-BFGS Adaptive T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 350.0)
    if rem > 60:
        print(f"  Budget: {t4:.0f}s")
        a4, sc4 = lbgfs_optimize(best_seq, t4, T_seq=[2000, 1000, 500, 200, 100], maxiter=2000)

    # === Phase 5: Spectral iteration ===
    print(f"\n=== Phase 5: Spectral (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    print(f"\n=== Phase 6: LP step=0.15 (best={best_score:.8f}) ===")

    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

        best_seq, best_score = check_and_update(a10, "final_polish", best_seq, best_score)

        # One more LP round

        if rem > 15:

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            print(f"\nFINAL VERIFIED SCORE: {fs:.8f} (n={len(best_seq)})")
            if fs < best_score:
                best_score = fs
