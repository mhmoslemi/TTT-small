import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def golden_section_search(f, direction, obj_current, max_iter=30):
    """
    Golden section search for best step size t in (0, 1].
    Minimize obj((1-t)*f + t*direction) over t in (0, 1].
    """
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    def obj_at(t):
        trial = (1.0 - t) * f + t * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return float('inf')
        trial = trial / s
        return fast_convolve(trial).max()
    
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_at(d)
    
    t_best = (a + b) / 2
    # Also check t=1 and t near endpoints
    for t_check in [0.01, 0.1, 0.5, 0.9, 0.99, 1.0]:
        obj_check = obj_at(t_check)
        obj_best = obj_at(t_best)
        if obj_check < obj_best:
            t_best = t_check
    
    return t_best, obj_at(t_best)

def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    """
    K = len(peak_indices)
    if K == 0:
        return np.ones(n) / n
    
    # Build constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n
    # (f*g)[k] = sum_j A[k,j] * g[j]
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=1000000, eps=eps)
            break
        except Exception:
            continue
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:

                stuck = 0

                stuck = 0

def coordinate_descent(f_init, time_limit, label="cd"):
    """Fast coordinate descent - only tries to decrease elements at peak positions."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_obj = fast_convolve(f).max()
    
    # Multipliers to try
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(f)
        
        # Focus on elements that contribute to peaks
        peak_thresh = 0.95 * b.max()
        peak_pos = np.where(b >= peak_thresh)[0]
        
        # Find which elements contribute to peaks
        active_j = set()
        for k in peak_pos:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                active_j.add(j)
        
        # Try modifying active elements first
        for j in sorted(active_j):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_val = f[j]
            best_obj_j = best_obj
            
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = new_val
            
            if best_val != f[j]:
                f[j] = best_val
                f = f / f.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    improved = True
        
        if not improved:
            break
    
    return f, best_obj

def simulated_annealing(f_init, time_limit, label="sa"):
    """Simulated annealing on the sequence."""
    t0 = time.monotonic()
    n = len(f_init)

    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    cur_obj = best_obj
    
    T = 0.01  # Initial temperature
    cooling = 0.9995
    rng = np.random.default_rng(77)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        n_perturb = rng.integers(1, max(2, n // 50))
        indices = rng.integers(0, n, size=n_perturb)
        
        f_trial = f.copy()
        for idx in indices:
            f_trial[idx] *= np.exp(rng.standard_normal() * 0.1)
        
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        
        trial_obj = fast_convolve(f_trial).max()
        delta = trial_obj - cur_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            f = f_trial
            cur_obj = trial_obj
            if trial_obj < best_obj - 1e-14:
                best_obj = trial_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cooling
        
        if no_improve > 200:
            T = max(T * 2, 0.001)
            no_improve = 0
        
        if it_equivalent := int((time.monotonic() - t0) * 100) % 500 == 0:
            pass  # just for reading

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP-based direction finding with golden-section line search,
adaptive peak selection with cutting-plane refinement, coordinate descent,
and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def golden_section_search(f, direction, max_iter=35):
    """Golden section search for best step t in (0,1] minimizing obj((1-t)*f + t*direction)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_at(t):
        trial = (1.0 - t) * f + t * direction
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return float('inf')
        trial = trial / s
        return fast_convolve(trial).max()

    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(max_iter):
        if b - a < 1e-6:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_at(d)

    t_best = (a + b) / 2.0
    obj_best = obj_at(t_best)

    # Check a few additional points
    for t_check in [0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 0.95, 1.0]:
        obj_check = obj_at(t_check)
        if obj_check < obj_best:
            obj_best = obj_check
            t_best = t_check

    return t_best, obj_best


def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    if K == 0:
        return np.ones(n) / n

    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver_name in [cp.ECOS, cp.SCS]:
        if solved:
            break
        try:
            prob.solve(solver=solver_name, max_iters=2000000, eps=eps)
            solved = True
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def solve_lp_with_cutting_planes(f, n, M, peak_frac_start, max_peaks, tol=1e-6, max_rounds=5):
    """
    LP with cutting plane refinement: start with top peaks, then add violated ones.
    """
    b = fast_convolve(f)
    threshold = peak_frac_start * M
    peaks = np.where(b >= threshold)[0]
    if len(peaks) > max_peaks:
        peaks = np.argpartition(b, -max_peaks)[-max_peaks:]

    g = None
    for round_i in range(max_rounds):
        g = solve_lp_direction(f, n, peaks, M, eps=1e-9)
        if g is None:
            break
        if g.sum() <= 1.0 + tol:
            break

        # Check if non-selected peaks are violated
        fg = fast_convolve(g)
        h = g / g.sum()
        fh = fast_convolve(h)
        violated = np.where(fh > M + tol)[0]
        # Also check the line search direction
        if len(violated) == 0:
            break
        # Add violated peaks (up to max_peaks total)
        new_peaks = np.unique(np.concatenate([peaks, violated]))
        if len(new_peaks) > max_peaks:
            # Keep the most important ones
            vals = np.zeros(len(fh))
            vals[new_peaks] = fh[new_peaks]
            new_peaks = np.argpartition(vals, -max_peaks)[-max_peaks:]
        peaks = new_peaks

    return g, peaks


def lp_optimize(f_init, time_limit, label="lp",
                peak_frac_start=0.88, peak_frac_end=0.65,
                max_peaks=400, patience=35):
    """LP-based optimization with adaptive peaks, cutting planes, and golden section search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_start
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Solve LP with adaptive peaks
        g, peaks_used = solve_lp_with_cutting_planes(
            f, n, M, peak_frac, max_peaks, tol=1e-7, max_rounds=4)

        if g is None:
            stuck += 1
            if stuck > patience:
                f = f * np.exp(rng.standard_normal(n) * 0.01)
                f = np.maximum(f, 0)
                f = f / f.sum()
                stuck = 0
                peak_frac = max(peak_frac_end, peak_frac - 0.05)
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > patience:
                # Geometric perturbation
                f = f * np.exp(rng.standard_normal(n) * 0.02)
                f = np.maximum(f, 0)
                f = f / f.sum()
                stuck = 0
                peak_frac = max(peak_frac_end, peak_frac - 0.03)
            continue

        h = g / S_g

        # Golden section line search
        t_best, obj_best = golden_section_search(f, h, max_iter=30)

        if t_best <= 0:
            stuck += 1
            if stuck > patience:
                f = f * np.exp(rng.standard_normal(n) * 0.01)
                f = np.maximum(f, 0)
                f = f / f.sum()
                stuck = 0
            continue

        # Update
        f = (1.0 - t_best) * f + t_best * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 5 == 0:
            score = 2.0 * n * best_obj
            print(f"  {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"peaks={len(peaks_used)} Sg={S_g:.4f} t={time.monotonic()-t0:.0f}s")

        # Adapt peak fraction (gradually include more constraints)
        if it % 8 == 0:
            peak_frac = max(peak_frac_end, peak_frac - 0.008)

        # Periodic geometric perturbation when stuck
        if stuck > patience // 2 and stuck % 10 == 0:
            f = f * np.exp(rng.standard_normal(n) * 0.004)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()

        # Reset stuck counter periodically with larger perturbation
        if stuck == patience:
            f = f * np.exp(rng.standard_normal(n) * 0.03)
            f = np.maximum(f, 0)
            f = f / f.sum()
            stuck = 0

    return best_f, best_obj


def peak_equalization(f_init, time_limit, label="pe"):
    """
    Targeted peak equalization: specifically try to reduce the gap
    between the top peaks by adjusting contributing elements.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        
        # Find top peaks
        top_k = min(10, len(b))
        top_indices = np.argsort(b)[-top_k:]
        
        # For each top peak, find the element that contributes most and try reducing it
        improved = False
        for k in top_indices:
            if time.monotonic() - t0 > time_limit:
                break
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contribution of each element j to b[k] is f[j] * f[k-j]
            contributions = np.zeros(hi - lo + 1)
            for ii in range(lo, hi + 1):
                contributions[ii - lo] = f[ii] * f[k - ii]
            
            # Find the dominant contributor
            max_idx = lo + np.argmax(contributions)
            
            # Try reducing this element
            for mult in [0.85, 0.9, 0.95]:
                f_trial = f.copy()
                f_trial[max_idx] *= mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            # Try increasing elements that don't contribute to peaks
            b = fast_convolve(f)
            low_thresh = 0.5 * b.max()
            safe_j = []
            for j in range(n):
                # Check if increasing f[j] would increase any peak
                would_violate = False
                for k in range(len(b)):
                    if b[k] < low_thresh:
                        continue
                    # f[j] contributes to b[k] via f[j]*f[k-j] if 0<=k-j<n
                    if 0 <= k - j < n:
                        # Increasing f[j] increases b[k] by delta * f[k-j]
                        # If b[k] is already high, this is bad
                        if b[k] > 0.9 * M:
                            would_violate = True
                            break
                if not would_violate:
                    safe_j.append(j)
            
            if safe_j:
                j = safe_j[rng.integers(0, len(safe_j))]
                f_trial = f.copy()
                f_trial[j] *= 1.1
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s > 1e-10:
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:
                        f = f_trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
        
        if not improved:
            # Small random perturbation
            f = f * np.exp(rng.standard_normal(n) * 0.005)
            f = np.maximum(f, 0)
            f = f / f.sum()
    
    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Fast coordinate descent focusing on elements near peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_obj = fast_convolve(f).max()
    multipliers = [0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(f)
        
        # Priority: elements contributing to top peaks first
        peak_thresh = 0.97 * b.max()
        peak_pos = np.where(b >= peak_thresh)[0]
        
        active_j = []
        for k in peak_pos:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            active_j.extend(range(lo, hi + 1))
        active_j = sorted(set(active_j))
        
        # Also include all elements in a second pass
        all_j = active_j + [j for j in range(n) if j not in set(active_j)]
        
        for j in all_j:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_val = f[j]
            best_val_obj = best_obj
            
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_val_obj - 1e-14:
                    best_val_obj = obj
                    best_val = new_val
            
            if best_val != f[j]:
                f[j] = best_val
                f = f / f.sum()
                if best_val_obj < best_obj - 1e-14:
                    best_obj = best_val_obj
                    improved = True
        
        if not improved:
            break

    return f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
