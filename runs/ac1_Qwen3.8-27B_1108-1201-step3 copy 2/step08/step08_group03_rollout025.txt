from scipy.optimize import minimize

    """Linear auto-convolution via FFT."""

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] as a linear function of g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, solver_attempts=('PDLP', 'ECOS', 'SCS')):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    for solver in solver_attempts:
        try:
            prob.solve(solver=solver, max_iters=500000)
        except Exception:
            continue
        if g_var.value is not None:
            g_val = np.maximum(g_var.value, 0.0)
            if not np.any(np.isnan(g_val)):
                return g_val
    return None

                break
            continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        scan_steps = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        for t in scan_steps:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden-section refinement
        if best_t > 0.01:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(8):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        if it % 15 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
            else:
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))

        if it % 30 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def smooth_obj_and_grad(z, beta, n):
    """
    Compute objective and gradient for smooth approximation using softmax parameterization.
    a = softmax(z), objective = 2*n*logsumexp(beta*b)/beta
    """
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    S = a.sum()
    a = a / S

    b = fast_convolve(a)

    beta_b = beta * b
    max_bb = beta_b.max()
    exps = np.exp(beta_b - max_bb)
    log_sum_exp = max_bb + np.log(exps.sum())
    obj = 2.0 * n * log_sum_exp / beta

    w = exps / exps.sum()

    # h[m] = 2 * sum_j a[j] * w[j+m]
    # Computed as 2 * cross_correlation(a, w)[m]
    # = 2 * convolve(flip(a), w)[n-1+m]
    flip_a = a[::-1]
    conv_len = n + (2 * n - 1)
    fft_sz = 1 << (conv_len - 1).bit_length()
    FA = rfft(flip_a, fft_sz)
    FW = rfft(w, fft_sz)
    conv = irfft(FA * FW, fft_sz)
    h = 2.0 * conv[n - 1:n - 1 + n]

    dot_ha = np.dot(h, a)
    grad = 2.0 * n / beta * a * (h - dot_ha)

    return obj, grad


def smooth_optimize(a_init, beta_start, beta_end, n_betas=8, maxiter_per_beta=300, label="smooth"):
    """
    Run L-BFGS-B smooth optimization with increasing beta.
    a_init: initial sequence (sum=1, non-negative)
    Returns best sequence found.
    """
    n = len(a_init)
    # Initialize z from a_init
    a_safe = np.maximum(a_init, 1e-15)
    a_safe = a_safe / a_safe.sum()
    z = np.log(a_safe)

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()

    alphas = np.linspace(0, 1, n_betas)
    for i in range(n_betas):
        beta = beta_start * (beta_end / beta_start) ** alphas[i]
        if beta < 1:
            beta = 1.0

        def fun_grad(zv):
            o, g = smooth_obj_and_grad(zv, beta, n)
            return o, g

        try:
            res = minimize(fun_grad, z, method='L-BFGS-B',
                           jac=True, options={'maxiter': maxiter_per_beta,
                                              'ftol': 1e-14, 'gtol': 1e-10,
                                              'maxfun': 5000})
            z = res.x
        except Exception:
            pass

        # Extract a
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        cur_score = 2.0 * n * fast_convolve(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            z = np.log(np.maximum(a, 1e-15))

        print(f"    {label} beta={beta:.1f} score={best_score:.8f}")

    return best_a


def generate_diverse_starts(a_best, n, rng, count=8):
    """Generate diverse starting points around a_best."""
    starts = [a_best.copy()]

    for i in range(count):
        start = a_best.copy()
        # Mix of perturbations
        mode = i % 4
        if mode == 0:
            # Gaussian noise
            noise = rng.standard_normal(n) * 0.05
            start = start + noise
        elif mode == 1:
            # Sinusoidal perturbation
            k = rng.integers(1, max(2, n // 10))
            phase = rng.uniform(0, 2 * np.pi)
            start = start + 0.05 * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
        elif mode == 2:
            # Block perturbation
            block_size = n // 4
            for _ in range(3):
                bs = rng.integers(block_size // 2, block_size)
                start_pos = rng.integers(0, n - bs)
                start[start_pos:start_pos + bs] *= rng.uniform(0.5, 1.5)
        else:
            # Sparse random change
            idx = rng.choice(n, size=n // 5, replace=False)
            start[idx] *= rng.uniform(0.5, 2.0, size=len(idx))

        start = np.maximum(start, 0.0)
        s = start.sum()
        if s > 1e-10:
            start /= s
        else:
            start = a_best.copy()
        starts.append(start)

    return starts


def adjust_length(seq, target_n):
    """Resample sequence to target length using interpolation."""
    old_n = len(seq)
    if old_n == target_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-10:
        new_seq /= s
    else:
        new_seq = np.ones(target_n) / target_n
    return new_seq


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: combines LP cutting planes, smooth L-BFGS-B optimization,
    multi-start diversification, and adaptive length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: LP optimization ===
    print(f"\n=== Phase 1: LP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = lp_cutting_planes(parent, time_limit=t1, initial_k=60, max_k=500, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {score1:.6f}")

    # === Phase 2: Smooth optimization (main Phase 2) ===
    print(f"\n=== Phase 2: Smooth L-BFGS-B (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        n_best = len(best_seq)
        # Multi-beta smooth optimization
        f2 = smooth_optimize(best_seq, beta_start=5.0, beta_end=200.0,
                             n_betas=6, maxiter_per_beta=200, label="sm1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [sm1]: {score2:.6f}")

    # === Phase 3: LP from smooth result ===
    print(f"\n=== Phase 3: LP from smooth (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        f3, _ = lp_cutting_planes(best_seq, time_limit=t3, initial_k=80, max_k=600, label="lp2")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp2]: {score3:.6f}")

    # === Phase 4: More smooth from LP result ===
    print(f"\n=== Phase 4: Smooth from LP (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4 = smooth_optimize(best_seq, beta_start=20.0, beta_end=500.0,
                             n_betas=5, maxiter_per_beta=150, label="sm2")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sm2]: {score4:.6f}")

    # === Phase 5: Multi-start diversification ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    n_starts = 6
    t5_per = max(15.0, rem * 0.20 / n_starts)
    if t5_per > 10:
        starts = generate_diverse_starts(best_seq, n_best, rng, count=n_starts)
        for si in range(1, n_starts):  # skip index 0 (original)
            rem = deadline - time.monotonic()
            if rem < t5_per + 5:
                break
            t5 = min(t5_per, rem - 5)
            if t5 < 10:
                break
            # Short smooth optimization from this start

                best_seq = f5.copy()

                # Update other starts to be based on new best

    print(f"\n=== Phase 6: Length exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    target_lengths = []
    for factor in [0.8, 0.9, 1.1, 1.2, 1.4]:
        tl = int(n_best * factor)

        if tl != n_best:
            target_lengths.append(tl)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_best))

        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [n{target_n}]: {score6:.6f}")
        # Quick smooth
        rem = deadline - time.monotonic()
        if rem > 15:
            f6s = smooth_optimize(f6, beta_start=10.0, beta_end=200.0,
                                  n_betas=3, maxiter_per_beta=80, label=f"n{target_n}s")
            score6s = float(evaluate_sequence(f6s.tolist()))
            if np.isfinite(score6s) and score6s < best_score:

    n_perts = 10
    t7_per = max(8.0, rem * 0.15 / n_perts)
    if t7_per > 5:
        for pi in range(n_perts):
            rem = deadline - time.monotonic()
            if rem < t7_per + 5:
                break
            # Random perturbation
            start = best_seq + rng.standard_normal(n_best) * 0.02
            start = np.maximum(start, 0.0)
            s = start.sum()
            if s > 1e-10:
                start /= s
            else:
                continue
            f7 = smooth_optimize(start, beta_start=20.0, beta_end=500.0,
                                 n_betas=3, maxiter_per_beta=60, label=f"pt{pi}")
            score7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(score7) and score7 < best_score:
                best_score = score7

                print(f"  NEW BEST [pt{pi}]: {score7:.6f}")

    # === Phase 8: Final LP + smooth burst ===

    rem = deadline - time.monotonic()
    n_best = len(best_seq)

    if rem > 30:

        f8, _ = lp_cutting_planes(best_seq, time_limit=t8a, initial_k=100, max_k=700, label="fin_lp")
        score8 = float(evaluate_sequence(f8.tolist()))
