"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses Nesterov-accelerated mirror descent on log-sum-exp approximation,
LP-based direction finding, and local search, with diverse initializations
and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, beta, top_k=None):
    """Compute softmax weights for peaks, optionally restricted to top_k."""
    if top_k is not None and top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_sub = b[top_idx]
        scaled = (b_sub - b_sub.max()) / beta
        w_sub = np.exp(scaled)
        w_sub = w_sub / w_sub.sum()
        w = np.zeros(len(b))
        w[top_idx] = w_sub
        return w
    else:
        scaled = (b - b.max()) / beta
        w = np.exp(scaled)
        w = w / w.sum()
        return w


def compute_subgradient(f, w, n):
    """Compute weighted subgradient of max autocorrelation."""
    g = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * f[int(k) - idx]
    return g


def nesterov_mirror_descent(f_init, time_limit, beta_init=0.01, beta_max=100.0,
                             beta_growth=1.005, lr_init=0.001, top_k=None,
                             label="nmd"):
    """
    Nesterov-accelerated mirror descent on log-sum-exp approximation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    momentum = 0.9
    
    beta = beta_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        b = fast_convolve(y)
        M = b.max()
        
        w = softmax_weights(b, beta, top_k)
        g = compute_subgradient(f, w, n)
        
        # Nesterov update
        momentum = min(momentum * 1.0001, 0.98)
        v = momentum * v - lr * g
        y_new = project_simplex(f + v)
        
        # Extrapolation
        t_curr = 2.0 / (it + 1 + momentum)
        y = (1 + t_curr) * y_new - t_curr * f
        
        f = y_new
        
        # Keep y on simplex
        y = project_simplex(y)
        
        # Grow beta (annealing)
        if it % 10 == 0:
            beta = min(beta * beta_growth, beta_max)
        
        # Adaptive learning rate
        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.9
                    no_improve = 0
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"beta={beta:.3f} lr={lr:.6f} t={elapsed:.0f}s")
        
        if no_improve > 40:
            # Reset momentum
            v = np.zeros(n)
            y = f.copy()
            no_improve = 0
            beta = max(0.005, beta * 0.5)
    
    return best_f, best_score


def lp_direction(f, peak_indices, M, solver='ECOS'):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    n = len(f)
    K = len(peak_indices)
    
    A = np.zeros((K, n))
    for i in range(K):
        k = int(peak_indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if K <= 500 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=20000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    
    return g_val


def lp_optimize(f_init, time_limit, top_k=100, label="lp", threshold_frac=0.9):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    full_len = 2 * n - 1
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Adaptive peak selection: use threshold + top-k
        threshold = M * threshold_frac
        above_thresh = np.where(b >= threshold)[0]
        
        if len(above_thresh) < 5:
            top_idx = np.argpartition(b, -max(5, min(top_k, full_len)))[-max(5, min(top_k, full_len)):]
            indices = np.sort(top_idx)
        elif len(above_thresh) > top_k * 3:
            # Too many above threshold, use top-k
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            indices = np.sort(top_idx)
        else:
            indices = np.sort(above_thresh)
        
        g = lp_direction(f, indices, M)
        
        if g is None:
            if len(indices) > 5:
                top_5 = np.argpartition(b, -5)[-5:]
                g = lp_direction(f, np.sort(top_5), M)
            if g is None:
                stuck += 2
                if stuck > 50:
                    break
                continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 40:
                break
            
            # Perturb and continue
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            stuck = 0
            continue
        
        h = g / S_g
        
        # Line search
        best_t = 0.0
        best_obj_trial = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            ts = f_trial.sum()
            if ts < 1e-15:
                continue
            f_trial = f_trial / ts
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = t
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 40:
                break
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 25 and it % 10 == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stuck = 0
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={len(indices)} t={elapsed:.0f}s")
    
    return best_f, best_score


def local_search(f_init, time_limit, n_perturb=100):
    """Local search with random perturbations and acceptance."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    rng = np.random.default_rng(456)
    
    # List of perturbation scales
    scales = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        scale = scales[it % len(scales)]
        
        # Random perturbation
        f_trial = f + rng.standard_normal(n) * scale
        f_trial = project_simplex(f_trial)
        
        obj_trial = fast_convolve(f_trial).max()
        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_score = 2.0 * n * obj_trial
            best_f = f_trial.copy()
            f = f_trial.copy()
        elif obj_trial < fast_convolve(f).max() + 1e-14:
            # Accept if not worse than current (simulated annealing-like)
            f = f_trial.copy()
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def make_cosine_window(n):
    """Create a cosine window."""
    j = np.arange(n)
    w = np.sin(np.pi * (j + 0.5) / n) ** 2
    w = w / w.sum()
    return w


def make_gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((j - center) / (sigma_frac * n / 2.0)) ** 2)
    w = w / w.sum()
    return w


def make_uniform(n):
    return np.ones(n) / n


def make_random_smooth(n, seed=0, smoothness=0.3):
    """Create a random smooth sequence."""
    rng = np.random.default_rng(seed)
    raw = rng.standard_normal(n)
    # Smooth via cumulative sum differences (random walk)
    from scipy.ndimage import gaussian_filter1d
    raw = gaussian_filter1d(raw, sigma=smoothness * n / 10.0)
    raw = np.maximum(raw, 0)
    s = raw.sum()
    if s > 1e-15:
        raw = raw / s
    else:
        raw = np.ones(n) / n
    return raw
