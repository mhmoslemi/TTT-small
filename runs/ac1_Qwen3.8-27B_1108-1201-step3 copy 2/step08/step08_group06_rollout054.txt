    max_val = conv.max()

    tol = 1e-10

    g = np.zeros(n)

        # conv(a,a)_k = sum_j a_j * a_{k-j}

    a_new = a * np.exp(-eta * g)

    if s > 0:

    return a_new, max_val

    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    for T in temps:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        z0 = np.log(np.maximum(a, 1e-30))

        def obj_func(z):
            a_tmp = softmax_from_z(z)
            conv = fast_convolve(a_tmp)
            cmax = conv.max()
            soft_max = T * np.log(np.sum(np.exp((conv - cmax) / T))) + cmax
            return 2.0 * n * soft_max

        def obj_grad(z):
            a_tmp = softmax_from_z(z)
            conv = fast_convolve(a_tmp)
            cmax = conv.max()
            w = np.exp((conv - cmax) / T)
            w = w / w.sum()
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a_tmp, fft_size)
            grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
            S = np.dot(grad_a, a_tmp)
            grad_z = a_tmp * (grad_a - S)
            return 2.0 * n * grad_z

        try:
            res = minimize(obj_func, z0, jac=obj_grad, method='L-BFGS-B',
                           options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-13,
                                    'maxfun': 50000})
            a_new = softmax_from_z(res.x)
            sc = compute_score(a_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=False):
    """Coordinate descent with golden-section line search on transfer amount."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    round_num = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        improved = False

        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)

        # Generate pairs
        pairs = []
        np_half = n_pairs // 2
        for _ in range(np_half):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        for _ in range(np_half):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            # Golden section search for optimal t (transfer from i to j)
            t_lo = -a[j] + 1e-14
            t_hi = a[i] - 1e-14
            if t_hi - t_lo < 1e-16:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            # Evaluate at x1
            trial_i = a[i] - x1
            trial_j = a[j] + x1
            if trial_i < -1e-14 or trial_j < -1e-14:
                f1 = 1e10
            else:
                a[i] = trial_i
                a[j] = trial_j
                s_tmp = a.sum()
                f1 = 2.0 * n * fast_convolve(a / s_tmp).max()
                a[i] = trial_i + x1
                a[j] = trial_j - x1

            # Evaluate at x2
            trial_i = a[i] - x2
            trial_j = a[j] + x2
            if trial_i < -1e-14 or trial_j < -1e-14:
                f2 = 1e10
            else:
                a[i] = trial_i
                a[j] = trial_j
                s_tmp = a.sum()
                f2 = 2.0 * n * fast_convolve(a / s_tmp).max()
                a[i] = trial_i + x2
                a[j] = trial_j - x2

            for _ in range(18):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    trial_i = a[i] - x1
                    trial_j = a[j] + x1
                    if trial_i < -1e-14 or trial_j < -1e-14:
                        f1 = 1e10
                    else:
                        a[i] = trial_i
                        a[j] = trial_j
                        s_tmp = a.sum()
                        f1 = 2.0 * n * fast_convolve(a / s_tmp).max()
                        a[i] = trial_i + x1
                        a[j] = trial_j - x1
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    trial_i = a[i] - x2
                    trial_j = a[j] + x2
                    if trial_i < -1e-14 or trial_j < -1e-14:
                        f2 = 1e10
                    else:
                        a[i] = trial_i
                        a[j] = trial_j
                        s_tmp = a.sum()
                        f2 = 2.0 * n * fast_convolve(a / s_tmp).max()
                        a[i] = trial_i + x2
                        a[j] = trial_j - x2

            t_opt = 0.5 * (t_lo + t_hi)
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and round_num % 10 == 0:
            print(f"  CD round={round_num}, best={best_score:.8f}")

    return best_a, best_score


def lp_greedy(f, M):
    """Greedy approximation to LP direction."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)

    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[0:min(k_end - k_start + 1, n)]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(len(f_slice)):
            if f_slice[idx] > 1e-15 and slack[idx] > 0:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice[:max_gj and (k_end - k_start + 1)] * max_gj

    return g


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def raised_cosine(n):
    """Raised cosine window normalized to sum=1."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window normalized to sum=1."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def check_update_candidate(a, best_seq, best_score, label):
    """Evaluate candidate and return updated (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP fictitious play, mirror descent, L-BFGS annealing, coordinate descent,
    and extensive length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1024
        parent = raised_cosine(n_init)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP Fictitious Play on parent ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_update_candidate(a_lp, best_seq, best_score, "lp_fp")

    # === Phase 2: L-BFGS Annealing on parent ===
    print(f"\n=== Phase 2: L-BFGS Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t_alloc, T_start=5000.0, T_end=30.0, n_temps=5)
        best_seq, best_score = check_update_candidate(a_lb, best_seq, best_score, "lbgfs_anneal")

    # === Phase 3: Mirror Descent on parent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, sc_md = mirror_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_update_candidate(a_md, best_seq, best_score, "mirror_descent")

    # === Phase 4: Length Exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.8f}) ===")
    candidate_lengths = sorted(set([
        n_parent - 4, n_parent - 3, n_parent - 2, n_parent - 1,
        n_parent + 1, n_parent + 2, n_parent + 3, n_parent + 4,
        n_parent + 8, n_parent - 8,
        512, 768, 1024, 1280, 1536, 2048, 3072, 4096
    ]))
    candidate_lengths = [n for n in candidate_lengths if 128 <= n <= 5000]

    for alt_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_total = min(rem * 0.06, 50.0)
        if t_total < 10:
            break

        # Try multiple initializations for this length
        inits = [
            interpolate_sequence(best_seq, alt_n),
            raised_cosine(alt_n),
        ]

        best_for_len = None
        best_for_len_score = float('inf')

        for idx, a_init in enumerate(inits):
            rem_init = deadline - time.monotonic()
            if rem_init < 10:
                break
            t_init = min(t_total / 2.0, 25.0)
            if t_init < 5:
                break

            # Quick mirror descent
            a_md, sc_md = mirror_descent(a_init, t_init * 0.6, verbose=False)
            # Quick L-BFGS
            rem2 = deadline - time.monotonic()
            if rem2 > 8:
                a_lb, sc_lb = lbgfs_smoothed(a_md, min(rem2 * 0.3, 15.0),
                                             T_start=3000.0, T_end=100.0, n_temps=3)
            else:
                a_lb = a_md
                sc_lb = sc_md

            if sc_lb < best_for_len_score:
                best_for_len_score = sc_lb
                best_for_len = a_lb

        if best_for_len is not None:
            best_seq, best_score = check_update_candidate(
                best_for_len, best_seq, best_score, f"n{alt_n}")

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coordinate Descent Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.25, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs=400, verbose=True)
        best_seq, best_score = check_update_candidate(a_cd, best_seq, best_score, "cd_polish")

    # === Phase 6: Final LP on best ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.20, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_update_candidate(a_lp2, best_seq, best_score, "lp_final")

    # === Phase 7: Final L-BFGS ===
    print(f"\n=== Phase 7: Final L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.40, 100.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin, sc_fin = lbgfs_smoothed(best_seq, t_alloc, T_start=2000.0, T_end=10.0, n_temps=5)
        best_seq, best_score = check_update_candidate(a_fin, best_seq, best_score, "lbgfs_final")

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.60, 60.0)
        a_fin2, sc_fin2 = coordinate_descent(best_seq, t_alloc, n_pairs=500, verbose=False)
        best_seq, best_score = check_update_candidate(a_fin2, best_seq, best_score, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
