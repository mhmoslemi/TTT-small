```
def softmax(z):
   z = np.asarray(z, dtype=np.float64)
   if z.size ==0: return z.copy()
   z = z - np.max(z)
   e = np.exp(z)
   s = e.sum()
   if s <=0 or not np.isfinite(s):
       return np.ones(z.size)/z.size
   return e/s
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Main new components:
  1. A fast constrained Gauss-Newton / active-peak optimizer.  For a chosen set
     of active convolution peaks it computes a sum-preserving direction that
     linearly flattens those peaks, then does an FFT-based exact line search.
  2. A score-based coordinate descent polish that chooses each coordinate change
     by the true objective, not just by reducing the raw convolution maximum.
  3. A smoothed LSE Adam optimizer in softmax variables for basin hopping.
  4. Multi-length exploration: the parent is optimized, perturbed, and lifted
     to twice and four times the length when time permits.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft


def normalize_nonneg(a):
    """Clip to non-negative values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = int(len(a))
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def quick_score(a):
    """Fast internal score using FFT convolution."""
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0:
        return float("inf")
    c = fast_convolve(a)
    return float(2.0 * n * np.max(c))


def softmax(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z.copy()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def make_z_from_a(a, rng, noise):
    """Log-domain initialization from a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-30))
    if rng is not None and noise > 0.0:
        z = z + rng.normal(0.0, float(noise), a.size)
    z = z - float(np.mean(z))
    return np.clip(z, -12.0, 12.0)


def perturb_sequence(a, rng, sigma=0.05):
    """Multiplicative log-normal perturbation on the simplex."""
    a = normalize_nonneg(a)
    if rng is None:
        rng = np.random.default_rng(0)
    z = make_z_from_a(a, rng, sigma)
    return softmax(z)


def interpolate_to_n(a, n):
    """Interpolate a normalized sequence to length n."""
    a = normalize_nonneg(a)
    old_n = len(a)
    if old_n == n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n)
    y = np.interp(new_x, old_x, a)
    return normalize_nonneg(y)


# ---------------------------------------------------------------------------
# Smooth LSE Adam optimizer
# ---------------------------------------------------------------------------

def weighted_corr(w, a):
    """
    Compute g[j] = sum_k w[k] * a[k-j].
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    nw = len(w)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = nw + n - 1
    fft_size = 1 << (size - 1).bit_length()
    q = w[::-1]
    W = rfft(q, fft_size)
    A = rfft(a, fft_size)
    h = irfft(W * A, fft_size)[:size]
    return h[n - 1:2 * n - 1][::-1].copy()


def lse_grad(z, beta):
    """
    Gradient of the scaled LSE smooth approximation of max(conv(a,a)),
    where a = softmax(z).  The beta-scaled objective has beta-independent
    gradient with respect to the simplex directions.
    """
    a = softmax(z)
    c = fast_convolve(a)
    cm = float(np.max(c))
    if beta <= 0.0:
        w = np.full_like(c, 1.0 / float(len(c)), dtype=np.float64)
    else:
        x = beta * (c - cm)
        e = np.exp(x)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    g = weighted_corr(w, a)
    der = 2.0 * g
    dot = float(np.dot(a, der))
    gz = a * (der - dot)
    gz[~np.isfinite(gz)] = 0.0
    return gz, a, c, cm


def adam_smooth(
    z0,
    n,
    time_limit,
    beta_start=1000.0,
    beta_end=20000.0,
    lr_start=0.015,
    lr_end=0.003,
    b1=0.90,
    b2=0.999,
    eps=1e-8,
    z_clip=12.0,
    stall_steps=3000,
    restart_noise=0.10,
    rng=None,
    verbose=False,
    label="",
):
    """Adam on softmax variables for the smooth LSE approximation."""
    if rng is None:
        rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    if time_limit <= 0.0:
        a = softmax(z0)
        return normalize_nonneg(a), quick_score(a)

    n = int(n)
    z = make_z_from_a(softmax(z0), rng, 0.0)
    z = np.clip(z, -z_clip, z_clip)

    a = softmax(z)
    c = fast_convolve(a)
    best_score = float(2.0 * n * np.max(c))
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    stall = 0
    restarts = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-12))
        beta = beta_start * (beta_end / max(beta_start, 1e-12)) ** frac
        lr = lr_start * (lr_end / max(lr_start, 1e-12)) ** frac

        gz, a, c, cm = lse_grad(z, beta)
        score = float(2.0 * n * cm)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * gz * gz
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = z - float(np.mean(z))
        z = np.clip(z, -z_clip, z_clip)

        if stall > stall_steps:
            if restarts < 40:
                z = make_z_from_a(best_a, rng, restart_noise + 0.02 * restarts)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t = 0
                stall = 0
                restarts += 1
            else:
                break

        if verbose and time.monotonic() - last_print > 20.0:
            print(f"    Adam {label}: it={t} elapsed={time.monotonic()-t0:.0f}s best={best_score:.6f}")
            last_print = time.monotonic()

    return normalize_nonneg(best_a), best_score


# ---------------------------------------------------------------------------
# Constrained active-peak Gauss-Newton style optimizer
# ---------------------------------------------------------------------------

def select_active_peaks(c, M, top_max, reduction):
    """Select peaks above M*(1-reduction), capped at top_max."""
    lc = int(len(c))
    if lc == 0:
        return np.array([], dtype=int)
    top_max = max(1, min(int(top_max), lc))
    if not np.isfinite(M) or M <= 0.0:
        if top_max >= lc:
            return np.arange(lc, dtype=int)
        return np.argpartition(c, -top_max)[-top_max:].astype(int)

    threshold = M * (1.0 - float(reduction))
    if threshold > M:
        threshold = M
    if threshold < 0.0:
        threshold = 0.0

    idx = np.nonzero(c >= threshold)[0]
    if idx.size == 0:
        if top_max >= lc:
            return np.arange(lc, dtype=int)
        return np.argpartition(c, -top_max)[-top_max:].astype(int)

    if idx.size > top_max:
        vals = c[idx]
        part = np.argpartition(vals, -top_max)[-top_max:]
        idx = idx[part]
    return idx.astype(int)


def compute_active_delta(a, c, K, target, mu=1e-8):
    """
    Sum-preserving least-squares direction that reduces the selected peaks.

    Let H[k,j] = a[k-j] when valid.  For a sum-zero direction of the form
    delta = P H^T u, the predicted change of selected peaks is 2 B u,
    where B = H P H^T.  This solves a small K-by-K system.
    """
    n = int(len(a))
    m = int(len(K))
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)

    H = np.zeros((m, n), dtype=np.float64)
    for ii, k in enumerate(K):
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j1 >= j0:
            H[ii, j0:j1 + 1] = a[k - j1:k - j0 + 1][::-1]

    b = np.asarray(c[K], dtype=np.float64) - float(target)

    HHt = H @ H.T
    onset = H.sum(axis=1)
    B = HHt - np.outer(onset, onset) / float(n)
    B = 0.5 * (B + B.T)

    Mmat = 4.0 * (B @ B) + float(mu) * B
    rhs = -2.0 * (B @ b)

    tr = float(np.trace(Mmat))
    ridge = 1e-12 * max(1.0, tr / max(m, 1))
    Mmat = Mmat + ridge * np.eye(m, dtype=np.float64)

    try:
        u = np.linalg.solve(Mmat, rhs)
    except Exception:
        try:
            u, _, _, _ = np.linalg.lstsq(Mmat, rhs, rcond=None)
        except Exception:
            return np.zeros(n, dtype=np.float64)

    delta = H.T @ u
    delta = delta - float(np.mean(delta))
    delta[~np.isfinite(delta)] = 0.0
    return delta


def line_search_delta(a, delta, t_vals, max_t):
    """Exact FFT line search along a sum-preserving direction."""
    a = normalize_nonneg(a)
    n = len(a)
    c0 = fast_convolve(a)
    M0 = float(np.max(c0))
    best_at = a.copy()
    best_ct = c0.copy()
    best_M = M0
    best_score = float(2.0 * n * M0)
    best_t = 0.0

    if max_t <= 1e-12:
        return best_at, best_ct, best_M, best_score

    vals = []
    for t in t_vals:
        t = float(t)
        if not np.isfinite(t) or t <= 1e-12:
            continue
        if t > max_t:
            t = max_t
        vals.append(t)

    if not vals:
        return best_at, best_ct, best_M, best_score

    vals = np.unique(np.round(np.asarray(vals, dtype=np.float64), 12))

    for t in vals:
        t = float(t)
        at = a + t * delta
        if np.any(at < -1e-12):
            continue
        at = np.maximum(at, 0.0)
        s = float(at.sum())
        if s <= 1e-300:
            continue
        at = at / s
        ct = fast_convolve(at)
        Mt = float(np.max(ct))
        sc = float(2.0 * n * Mt)
        if np.isfinite(Mt) and Mt < best_M - 1e-14:
            best_M = Mt
            best_score = sc
            best_at = at.copy()
            best_ct = ct.copy()
            best_t = t

    return best_at, best_ct, best_M, best_score


def active_peak_optimize(a0, time_limit, verbose=False, label=""):
    """
    Iteratively flatten the current most-active convolution peaks.
    """
    rng = np.random.default_rng(2024)
    t0 = time.monotonic()
    if time_limit <= 0.0:
        a = normalize_nonneg(a0)
        return a, quick_score(a)

    a = normalize_nonng(a0) if hasattr(normalize_nonneg, '__self__') else normalize_nonneg(a0)
    c = fast_convolve(a)
    M = float(np.max(c))
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * M)
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        c = fast_convolve(best_a)
        M = float(np.max(c))
        best_score = float(2.0 * n * M)
        a = best_a.copy()

    step_scale = 0.05
    stall = 0
    restarts = 0
    last_print = t0
    it = 0

    # (top_max, reduction_factor, regularization)
    param_cycle = [
        (12, 0.05, 1e-8),
        (24, 0.10, 1e-8),
        (48, 0.15, 1e-7),
        (32, 0.10, 1e-6),
        (64, 0.25, 1e-6),
        (16, 0.05, 1e-5),
    ]
    pidx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if not np.isfinite(M) or M <= 0.0:
            break

        top_max, reduction, mu = param_cycle[pidx]
        K = select_active_peaks(c, M, top_max, reduction)
        if K.size == 0:
            break

        target = M * (1.0 - float(reduction))
        delta = compute_active_delta(a, c, K, target, mu=mu)
        delta = delta - float(np.mean(delta))

        norm = float(np.linalg.norm(delta))
        if norm < 1e-14:
            stall += 1
            step_scale = max(1e-5, 0.85 * step_scale)
            pidx = (pidx + 1) % len(param_cycle)
            if stall > 300:
                break
            continue

        if norm > step_scale:
            delta = delta * (step_scale / norm)

        neg = delta < -1e-15
        if np.any(neg):
            ratios = -a[neg] / delta[neg]
            t_max = float(np.min(ratios))
            t_max = min(1.0, max(0.0, t_max * (1.0 - 1e-12)))
        else:
            t_max = 1.0

        if t_max <= 1e-7:
            stall += 1
            step_scale = max(1e-5, 0.8 * step_scale)
            pidx = (pidx + 1) % len(param_cycle)
            if stall > 300:
                break
            continue

        base_ts = [0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.00]
        t_vals = np.asarray(base_ts, dtype=np.float64)
        if t_max < 1.0:
            t_vals = np.concatenate((t_vals, np.asarray([0.5 * t_max, 0.9 * t_max], dtype=np.float64)))

        a_new, c_new, M_new, score_new = line_search_delta(a, delta, t_vals, t_max)

        if M_new < M - 1e-14:
            a = a_new
            c = c_new
            M = M_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
            stall = 0
            step_scale = min(0.12, 1.05 * step_scale)
        else:
            stall += 1
            step_scale = max(1e-5, 0.8 * step_scale)
            if stall > 20:
                pidx = (pidx + 1) % len(param_cycle)

            if stall > 120 and restarts < 30:
                restarts += 1
                a = perturb_sequence(best_a, rng, 0.04 + 0.02 * restarts)
                c = fast_convolve(a)
                M = float(np.max(c))
                stall = 0
                step_scale = 0.05
                pidx = (pidx + 2) % len(param_cycle)
            elif stall > 400:
                break

        if verbose and time.monotonic() - last_print > 20.0:
            print(f"    Active {label}: it={it} elapsed={time.monotonic()-t0:.0f}s best={best_score:.6f} M={M:.10e}")
            last_print = time.monotonic()

    return normalize_nonneg(best_a), best_score


# ---------------------------------------------------------------------------
# Score-based coordinate descent
# ---------------------------------------------------------------------------

def score_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent that minimizes the true normalized objective.
    """
    t0 = time.monotonic()
    a = normalize_nonneg(a0)
    n = len(a)
    if n == 0 or time_limit <= 0.0:
        return a, quick_score(a)

    c = fast_convolve(a)
    S = 1.0
    M = float(np.max(c))
    score = float(2.0 * n * M)

    best_a = a.copy()
    best_score = score

    rel_factors = np.asarray([0.0, 0.25, 0.5, 0.75, 0.90, 1.05, 1.20, 1.50, 2.00], dtype=np.float64)
    mean_factors = np.asarray([0.5, 1.0, 2.0, 4.0], dtype=np.float64)

    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            x = float(a[i])
            mean = S / float(n)

            ys = x * rel_factors
            if mean > 0.0:
                ys = np.concatenate((ys, mean * mean_factors))
            ys = ys[np.isfinite(ys) & (ys >= 0.0)]
            if ys.size == 0:
                continue
            ys = np.unique(ys)

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_left = np.max(c[:k_start]) if k_start > 0 else -1e300
            unaff_right = np.max(c[k_end + 1:]) if k_end < 2 * n - 2 else -1e300
            unaff_max = float(max(unaff_left, unaff_right))

            affected = c[k_start:k_end + 1]
            a_old = a
            best_y = x
            best_new_score = score
            best_new_M = M
            best_new_S = S

            for y in ys:
                y = float(y)
                if abs(y - x) <= 1e-15:
                    continue
                delta = y - x
                new_S = S + delta
                if new_S < 1e-6:
                    continue

                new_aff = affected + 2.0 * delta * a_old
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_aff[idx_2i] = c[2 * i] + 2.0 * delta * x + delta * delta

                aff_max = float(np.max(new_aff))
                new_M = max(unaff_max, aff_max)
                if new_M < 0.0:
                    new_M = 0.0
                new_score = float(2.0 * n * new_M / (new_S * new_S))

                if np.isfinite(new_score) and new_score < best_new_score:
                    best_new_score = new_score
                    best_new_M = new_M
                    best_new_S = new_S
                    best_y = y

            if best_y != x and best_new_score < score - 1e-14:
                delta = best_y - x
                changes = 2.0 * delta * a_old
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * delta * x + delta * delta

                c[k_start:k_end + 1] += changes
                a[i] = best_y
                S = best_new_S
                M = best_new_M
                score = best_new_score

                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 12 and sweeps > 10:
                break
        else:
            no_improve = 0

        # Keep numbers well scaled without hurting the invariant objective.
        if abs(S - 1.0) > 1e-4 or sweeps % 10 == 0:
            scale = 1.0 / S
            a = a * scale
            c = c * scale * scale
            S = 1.0
            M = float(np.max(c))
            score = float(2.0 * n * M)

        if verbose and sweeps % 20 == 0:
            print(f"    Coord {label}: sweep={sweeps} elapsed={time.monotonic()-t0:.0f}s best={best_score:.6f}")

    return normalize_nonneg(best_a), best_score


# ---------------------------------------------------------------------------
# Candidate update / exact evaluation wrapper
# ---------------------------------------------------------------------------

def update_best_candidate(a, best_a, best_score, label=""):
    """
    Updated the incumbent only if the exact evaluator confirms an improvement.
    If the evaluator is unavailable or too expensive, fall back to FFT score.
    """
    a2 = normalize_nonneg(a)
    if a2.size == 0:
        return best_a, best_score

    cs = quick_score(a2)
    if not np.isfinite(cs):
        return best_a, best_score

    if best_a is None:
        best_a = a2.copy()
        best_score = cs
        if a2.size <= 4096:
            try:
                ex = float(evaluate_sequence([float(x) for x in a2]))
                if np.isfinite(ex):
                    best_score = ex
            except Exception:
                pass
        print(f"  SET INITIAL [{label}]: exact={best_score:.6f} (n={len(best_a)})")
        return best_a, best_score

    # Use a small tolerance to avoid exact calls for numerically identical scores.
    if cs < best_score - 1e-12:
        used_exact = False
        if a2.size <= 4096:
            try:
                ex = float(evaluate_sequence([float(x) for x in a2]))
                used_exact = True
                if np.isfinite(ex) and ex < best_score:
                    best_a = a2.copy()
                    best_score = ex
                    print(f"  NEW BEST exact [{label}]: {best_score:.6f} (n={len(best_a)})")
                elif np.isfinite(ex) and ex <= cs + 1e-9:
                    # Exact evaluator says it is not better.  Trust exact.
                    pass
            except Exception:
                used_exact = False

        if not used_exact and cs < best_score:
            best_a = a2.copy()
            best_score = cs
            print(f"  NEW BEST quick [{label}]: {best_score:.6f} (n={len(best_a)})")

    return best_a, best_score


# ---------------------------------------------------------------------------
# Main search driver
# ---------------------------------------------------------------------------

def propose_candidate():
    """
    Search for a low peak convolution sequence within the allowed time budget.
    Returns a list of non-negative floats.
    """
    t_start = time.monotonic()
    DEADLINE = t_start + 1770.0
    SAFETY = 70.0
    rng = np.random.default_rng(20240524)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 0.01:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(1024, dtype=np.float64) / 1024.0

    if len(parent) > 4096:
        parent = interpolate_to_n(parent, 4096)

    best_a, best_score = update_best_candidate(parent, None, float("inf"), "parent")
    if best_a is None:
        best_a = parent.copy()
        best_score = quick_score(parent)

    n0 = int(len(best_a))
    print(f"Start: n={n0}, best_score={best_score:.6f}")

    # --- Phase 1: intensive active-peak optimization on parent length ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 80.0:
        tl = min(450.0, rem * 0.30, 550.0)
        a_cand, s_cand = active_peak_optimize(best_a, tl, verbose=True, label=f"A_n{n0}")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"A_n{n0}")

    # --- Phase 2: perturbed active-peak restarts on parent length ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 70.0:
        tl = min(260.0, rem * 0.25, 320.0)
        start = perturb_sequence(best_a, rng, 0.04)
        a_cand, s_cand = active_peak_optimize(start, tl, verbose=True, label=f"B1_n{n0}")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"B1_n{n0}")

    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 60.0:
        tl = min(170.0, rem * 0.20, 220.0)
        start = perturb_sequence(best_a, rng, 0.12)
        a_cand, s_cand = active_peak_optimize(start, tl, verbose=True, label=f"B2_n{n0}")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"B2_n{n0}")

    # --- Phase 3: coordinate descent polish on parent length ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 60.0 and len(best_a) <= 4096:
        tl = min(260.0, rem * 0.22, 320.0)
        a_cand, s_cand = score_coord_descent(best_a, tl, verbose=True, label=f"C_n{len(best_a)}")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"C_n{len(best_a)}")

    # --- Phase 4: elevated length 2*n (if reasonable) ---
    n2 = 2 * n0
    if n2 >= 256 and n2 <= 4096:
        rem = DEADLINE - time.monotonic() - SAFETY
        if rem > 100.0:
            start = interpolate_to_n(best_a, n2)
            tl = min(280.0, rem * 0.28, 360.0)
            a_cand, s_cand = active_peak_optimize(start, tl, verbose=True, label=f"E_n{n2}")
            best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"E_n{n2}")

        rem = DEADLINE - time.monotonic() - SAFETY
        if rem > 70.0:
            start = perturb_sequence(interpolate_to_n(best_a, n2), rng, 0.05)
            tl = min(120.0, rem * 0.18, 180.0)
            a_cand, s_cand = active_peak_optimize(start, tl, verbose=True, label=f"F_n{n2}")
            best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"F_n{n2}")

        rem = DEADLINE - time.monotonic() - SAFETY
        if rem > 50.0 and len(best_a) <= 2048:
            tl = min(150.0, rem * 0.20, 200.0)
            a_cand, s_cand = score_coord_descent(best_a, tl, verbose=True, label=f"G_n{len(best_a)}")
            best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"G_n{len(best_a)}")

    # --- Phase 5: elevated length 4*n (active only, coordinate skipped) ---
    n4 = 4 * n0
    if n4 >= 512 and n4 <= 4096:
        rem = DEADLINE - time.monotonic() - SAFETY
        if rem > 120.0:
            start = interpolate_to_n(best_a, n4)
            tl = min(220.0, rem * 0.30, 300.0)
            a_cand, s_cand = active_peak_optimize(start, tl, verbose=True, label=f"H_n{n4}")
            best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"H_n{n4}")

    # --- Phase 6: Adam basin hopping on the current best length ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 100.0 and len(best_a) <= 2048:
        z0 = make_z_from_a(best_a, rng, 0.02)
        tl = min(180.0, rem * 0.35, 260.0)
        a_cand, s_cand = adam_smooth(
            z0, len(best_a), tl,
            beta_start=800.0, beta_end=20000.0,
            lr_start=0.018, lr_end=0.003,
            rng=rng, verbose=True, label=f"AD_n{len(best_a)}"
        )
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, f"AD_n{len(best_a)}")

    # --- Phase 7: final short active polish ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 60.0:
        tl = min(140.0, rem * 0.45, 180.0)
        a_cand, s_cand = active_peak_optimize(best_a, tl, verbose=True, label="FINAL_A")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, "FINAL_A")

    # --- Phase 8: final short coordinate polish if length is manageable ---
    rem = DEADLINE - time.monotonic() - SAFETY
    if rem > 40.0 and len(best_a) <= 2048:
        tl = min(80.0, rem * 0.55, 100.0)
        a_cand, s_cand = score_coord_descent(best_a, tl, verbose=True, label="FINAL_C")
        best_a, best_score = update_best_candidate(a_cand, best_a, best_score, "FINAL_C")

    # --- Final verification and return ---
    best_a = normalize_nonneg(best_a)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s <= 1e-300:
        best_a = np.ones(1024, dtype=np.float64) / 1024.0
    else:
        best_a = best_a / s

    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(final_score):
            print(f"FINAL verified score: {final_score:.6f} (n={len(best_a)})")
        else:
            print(f"final evaluator returned non-finite; quick={quick_score(best_a):.6f}")
    except Exception as e:
        print(f"final evaluator exception: {e}; quick={quick_score(best_a):.6f}")

    out = [float(x) for x in best_a]
    return out
```
