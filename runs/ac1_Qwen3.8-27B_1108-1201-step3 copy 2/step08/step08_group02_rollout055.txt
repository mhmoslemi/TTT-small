Optimized approach: Hybrid LP + exponential weights optimization
with constraint generation, multi-start, and adaptive length search.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve_auto(a):
    """Linear auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def solve_lp_direction(f, n, peaks, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peaks.
    Uses scipy linprog with HiGHS solver.
    Returns g or None.
    """
    K = len(peaks)
    if K == 0:
        return np.ones(n) / n
    
    # Build constraint matrix: A_eq @ g <= M for each peak
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((K, n))
    for row in range(K):
        k = int(peaks[row])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For j in [lo, hi]: A[row, j] = f[k - j]
        A_ub[row, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=np.full(K, M), bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.maximum(res.x, 0)
    if g.sum() <= 0:
        return None
    return g

def exp_weights_direction(f, n, peaks, b, M, eta=1.0):
    """
    Fast exponential weights direction.
    g[j] proportional to exp(-eta * damage[j])
    where damage[j] = sum_k w_k * f[k-j] for valid k-j
    """
    damage = np.zeros(n)
    for k in peaks:
        w_k = b[k] / M
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi+1)
        damage[j_range] += w_k * f[k - j_range]
    
    # Softmax with temperature
    scaled = -eta * damage
    scaled -= scaled.max()
    g = np.exp(scaled)
    s = g.sum()
    if s < 1e-30:
        g = np.ones(n) / n
    else:
        g /= s
    return g

    """
    Find best t in [0, 1] to minimize max(conv((1-t)f + tg, (1-t)f + tg)).
    Uses the fact that conv((1-t)f+tg, (1-t)f+tg) = (1-t)^2 conv(f,f) + 2t(1-t) conv(f,g) + t^2 conv(g,g)
    """
    b_ff = fast_convolve_auto(f)
    b_fg = fast_convolve_cross(f, g)
    b_gg = fast_convolve_auto(g)
    
    # For each t, the convolution is (1-t)^2 * b_ff + 2t(1-t) * b_fg + t^2 * b_gg
    # We want to minimize max_k of this expression
    
    best_t = 0.0
    best_val = M_current
    
    # Try a grid
    t_vals = np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5])
    for t in t_vals:
        b_t = (1-t)**2 * b_ff + 2*t*(1-t) * b_fg + t**2 * b_gg
        val = b_t.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        lo = max(0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        t_fine = np.linspace(lo, hi, 20)
        for t in t_fine:
            b_t = (1-t)**2 * b_ff + 2*t*(1-t) * b_fg + t**2 * b_gg
            val = b_t.max()
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def main_optimize(f_init, time_limit, use_lp_every=15, label="main"):
    """
    Main optimization loop using hybrid LP + exponential weights.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    
    best_f = f.copy()
    best_b = fast_convolve_auto(f).max()
    it = 0
    stuck = 0
    eta = 1.0  # temperature for exp weights
    peak_count = 50
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve_auto(f)
        M = b.max()
        
        # Select peaks
        threshold = 0.9 * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > peak_count:
            peaks = np.argpartition(b, -peak_count)[-peak_count:]
        if len(peaks) < 10:
            peaks = np.argpartition(b, -max(10, len(b)//20))[-max(10, len(b)//20):]
        
        # Choose direction: LP every N iterations, else exp weights
        if it % use_lp_every == 0:
            g = solve_lp_direction(f, n, peaks, M)
        else:
            g = exp_weights_direction(f, n, peaks, b, M, eta=eta)
        
        if g is None or g.sum() <= 1.0 + 1e-6:
            # Try LP always if exp weights fail
            g = solve_lp_direction(f, n, peaks, M)
        
        if g is None:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        S_g = g.sum()
        h = g / S_g
        
        # Line search
        best_t, best_t_obj = line_search(f, h, M, n)

            stuck += 1
            if stuck > 50:
                break
            # Try smaller step
            f_trial = f + 0.01 * (h - f)
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            obj = fast_convolve_auto(f_trial).max()
            if obj < M:
                f = f_trial
                continue
            else:
                break
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve_auto(f).max()
        if cur_obj < best_b - 1e-12:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
            # Slowly cool down
            eta = max(0.3, eta * 0.995)
            peak_count = min(100, peak_count + 1)
        else:
            stuck += 1
            eta = min(3.0, eta * 1.05)

            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "

            f = f + rng.standard_normal(n) * 0.001
            f = np.maximum(f, 0)
            f /= f.sum()
            stuck = 0
            eta = 1.0
            peak_count = 30
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """
    Fast coordinate descent using incremental convolution update.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    
    b = fast_convolve_auto(f)
    best_b = b.max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        sweeps += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break
            if f[j] < 1e-15:
                continue
            
            old_val = f[j]
            best_j_val = old_val
            best_j_obj = best_b
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                # Compute new convolution efficiently
                # conv_new = conv_old - old_val * (a convolved with e_j) + new_val * (a convolved with e_j)
                # Actually: conv(f, f) changes only at positions involving j
                # (f*f)[k] includes terms f[j]*f[k-j] and f[k-j]*f[j] for valid indices
                # So delta at position k (for k involving j):
                # If we change f[j] from old to new:
                # delta_k = (new^2 - old^2) if k == 2j  (f[j]*f[j])
                # delta_k = (new - old) * f[k-j] if k-j != j and k-j valid
                # delta_k = (new - old) * f[j] if j != k-j (i.e., 2j != k)
                
                # Simpler: just compute the change
                # Positions affected: k where the term f[j]*f[k-j] or f[k-j]*f[j] appears
                # This is k in [j, j+n-1] in the full convolution
                # Specifically: for position k in full conv:
                #   contribution involving f[j]: f[j] * f[k-j] if 0 <= k-j < n
                #   plus f[k-j] * f[j] if 0 <= k-j < n (same thing for auto-conv)
                # Wait, for auto-conv: (f*f)[k] = sum_i f[i]*f[k-i]
                # The terms involving f[j]: f[j]*f[k-j] (when i=j) + f[k-j]*f[j] (when i=k-j)
                # If j != k-j: both terms are present, so contribution = 2*f[j]*f[k-j]
                # If j == k-j (i.e., k=2j): contribution = f[j]^2
                
                # Instead of this complex calculation, let me just recompute the affected positions
                # For changing f[j]: affected positions are k where f[k-j] is involved
                # That's k in [j, j+n-1] (since k-j ranges from 0 to n-1)
                
                lo_k = j
                hi_k = j + n - 1
                lo_k = max(0, lo_k)
                hi_k = min(2*n-2, hi_k)
                
                k_range = np.arange(lo_k, hi_k + 1)
                # For each k, the change in (f*f)[k]:
                # Old contribution of f[j]: f[j]*f[k-j] (and f[k-j]*f[j] if k-j != j)
                # New contribution: new_val * f[k-j] (and f[k-j]*new_val if k-j != j)
                
                # f[k-j] for k in k_range: k-j ranges from lo_k-j to hi_k-j
                idx = k_range - j  # indices into f
                idx = np.clip(idx, 0, n-1)
                f_vals = f[idx]
                
                # For auto-conv at position k:
                # (f*f)[k] = sum_i f[i]*f[k-i]
                # Terms involving f[j]:
                #   i=j: f[j]*f[k-j]
                #   i=k-j: f[k-j]*f[j] (same as above if j==k-j, else different term)
                # So total contribution of f[j] to (f*f)[k] is:
                #   2*f[j]*f[k-j] if j != k-j (i.e., k != 2j)
                #   f[j]^2 if k == 2j
                
                kg = k_range
                is_diag = (kg == 2*j)
                # Old contribution
                old_contrib = np.where(is_diag, old_val**2, 2*old_val*f_vals)
                # New contribution
                new_contrib = np.where(is_diag, new_val**2, 2*new_val*f_vals)
                
                delta = new_contrib - old_contrib
                new_b_affected = b[lo_k:hi_k+1] + delta
                
                # New max
                # The max could be at the affected positions or elsewhere
                # elsewhere max: need to track this
                # For speed, just check if any affected position exceeds current max
                # and if the current max position is affected
                
                # Simple approach: just check the affected positions and the old max
                new_max = max(new_b_affected.max(), best_b)  # upper bound
                # Actually we need the true max. Let's just recompute if needed.
                # For now, use an approximation: if max of affected < best_b and best_b position is not affected, then max stays
                max_pos = np.argmax(b)
                if lo_k <= max_pos <= hi_k:
                    new_max = max(new_b_affected.max(), np.delete(b, slice(lo_k, hi_k+1)).max() if hi_k < lo_k else b[:lo_k].max() if lo_k > hi_k else max(b[:lo_k].max() if lo_k > 0 else 0, b[hi_k+1:].max() if hi_k < 2*n-2 else 0))
                else:
                    new_max = max(new_b_affected.max(), best_b)
                
                if new_max < best_j_obj - 1e-14:
                    best_j_obj = new_max
                    best_j_val = new_val
            
            if best_j_val != old_val:
                # Update f and b
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                b = fast_convolve_auto(f)
                best_b = b.max()
                improved = True
        
        if not improved:
            break
    
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Hybrid LP + Exponential Weights optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast exp-weights iterations with periodic LP refinement via constraint generation,
adaptive peak selection, multi-length exploration, and coordinate descent polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve_auto(a):
    """Linear auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def solve_lp_direction(f, n, peaks, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peaks.
    Uses scipy linprog with HiGHS.
    Returns g array or None.
    """
    K = len(peaks)
    if K == 0:
        return np.ones(n) / n

    A_ub = np.zeros((K, n))
    for row in range(K):
        k = int(peaks[row])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[row, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]

    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(K, M)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    if g.sum() <= 1e-15:
        return None
    return g


def exp_weights_direction(f, n, peaks, b, M, eta=1.0):
    """
    Fast exponential weights direction: g[j] prop to exp(-eta * damage[j]).
    damage[j] = sum over active peaks k of (b[k]/M) * f[k-j].
    """
    damage = np.zeros(n)
    for k in peaks:
        w_k = b[k] / M
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        damage[j_range] += w_k * f[k - j_range]

    scaled = -eta * damage
    mx = scaled.max()
    if mx > 500:
        scaled = scaled - mx
    g = np.exp(scaled)
    s = g.sum()
    if s < 1e-30:
        g = np.ones(n) / n
    else:
        g /= s
    return g


def line_search_quad(f, g, b_ff, b_fg, b_gg, M_current):
    """
    Line search using quadratic decomposition:
    conv((1-t)f+tg, (1-t)f+tg) = (1-t)^2 b_ff + 2t(1-t) b_fg + t^2 b_gg
    Returns (best_t, best_val).
    """
    best_t = 0.0
    best_val = M_current

    t_coarse = np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5])
    for t in t_coarse:
        val = ((1 - t) ** 2 * b_ff + 2 * t * (1 - t) * b_fg + t ** 2 * b_gg).max()
        if val < best_val - 1e-15:
            best_val = val
            best_t = t

    if best_t > 0:
        lo = max(0.001, best_t - 0.06)
        hi = min(0.99, best_t + 0.06)
        t_fine = np.linspace(lo, hi, 30)
        for t in t_fine:
            val = ((1 - t) ** 2 * b_ff + 2 * t * (1 - t) * b_fg + t ** 2 * b_gg).max()
            if val < best_val - 1e-15:
                best_val = val
                best_t = t

    return best_t, best_val


def main_optimize(f_init, time_limit, lp_every=10, label="main"):
    """
    Main optimization: hybrid exp-weights + periodic LP.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s

    b_ff = fast_convolve_auto(f)
    best_b = b_ff.max()
    best_f = f.copy()
    it = 0
    stuck = 0
    eta = 1.0
    peak_count = 40
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = b_ff.max()

        # Peak selection
        threshold = 0.88 * M
        peaks_mask = b_ff >= threshold
        peaks = np.where(peaks_mask)[0]
        if len(peaks) > peak_count:
            peaks = np.argpartition(b_ff, -peak_count)[-peak_count:]
        if len(peaks) < 8:
            n_peaks = max(10, len(b_ff) // 25)
            peaks = np.argpartition(b_ff, -n_peaks)[-n_peaks:]

        # Direction
        used_lp = False
        if it % lp_every == 0:
            g = solve_lp_direction(f, n, peaks, M)
            used_lp = True

        if g is None:
            g = exp_weights_direction(f, n, peaks, b_ff, M, eta=eta)

        if g is None or g.sum() <= 1.0 + 1e-8:
            # Fallback: try LP with fewer peaks
            if len(peaks) > 15:
                g = solve_lp_direction(f, n, peaks[:15], M)
                used_lp = True
            if g is None:
                g = np.ones(n) / n

        S_g = g.sum()
        h = g / S_g

        # Line search via quadratic decomposition
        b_fg = fast_convolve_cross(f, h)
        b_gg = fast_convolve_auto(h)
        best_t, best_t_val = line_search_quad(f, h, b_ff, b_fg, b_gg, M)

        if best_t <= 0:
            stuck += 1
            if stuck > 60:
                # Perturb and reset
                f = f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                f /= f.sum()
                b_ff = fast_convolve_auto(f)
                stuck = 0
                eta = 1.0
                peak_count = 30
                continue
            if stuck % 10 == 0:
                # Try smaller step manually
                t_try = 0.01
                f_trial = (1 - t_try) * f + t_try * h
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                obj_trial = fast_convolve_auto(f_trial).max()
                if obj_trial < M * (1 - 1e-8):
                    f = f_trial
                    b_ff = fast_convolve_auto(f)
                    continue
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        b_ff = fast_convolve_auto(f)
        cur_obj = b_ff.max()

        if cur_obj < best_b - 1e-13:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
            eta = max(0.4, eta * 0.997)
            peak_count = min(120, peak_count + 1)
        else:
            stuck += 1
            eta = min(2.5, eta * 1.03)

        if it % 30 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"eta={eta:.3f} peaks={len(peaks)} stuck={stuck} "
                  f"lp={used_lp} t={time.monotonic()-t0:.0f}s")

        if stuck > 30 and stuck % 10 == 0:
            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0)
            f /= f.sum()
            b_ff = fast_convolve_auto(f)
            stuck = 5
            eta = 1.0
            peak_count = 30

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """
    Coordinate descent: try scaling each coordinate and keep improvement.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    b = fast_convolve_auto(f)
    best_b = b.max()
    best_f = f.copy()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.2])
    sweeps = 0
    max_sweeps_no_improve = 3

    while time.monotonic() - t0 < time_limit:
        improved = False
        sweeps += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break
            if f[j] < 1e-14:
                continue

            old_val = f[j]
            best_j_val = old_val
            best_j_obj = best_b

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve_auto(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                f[j] = best_j_val
                f /= f.sum()
                b = fast_convolve_auto(f)
                if best_j_obj < best_b:
                    best_b = best_j_obj
                    best_f = f.copy()
                improved = True
                best_j_obj = best_b  # recompute for next multiplier in same j

        if not improved:
            break

    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
