                no_improve = 0

                break
    
    return best_f

"""
Optimized step function coefficient search using exact coordinate descent,
mirror descent (multiplicative weights), and multi-length exploration.
Targets score < 1.5027 for the auto-convolution peak problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(f)
    return 2.0 * n * b.max() / (s * s)


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def trial_score(b_slice, f_slice, max_unaff, x, ai):
    delta = x - ai
    new_sum = 1.0 + delta
    if new_sum < 1e-10:
        return 1e10
    max_aff = np.max(b_slice + delta * 2.0 * f_slice)
    max_total = max(max_aff, max_unaff)
    return max_total / (new_sum * new_sum)


def coord_optimize(f, b, i, n):
    ai = f[i]
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    if lo > hi:
        return ai
    fk_lo = max(0, lo - i)
    fk_hi = min(n - 1, hi - i)
    f_slice = f[fk_lo:fk_hi + 1]
    b_slice = b[lo:hi + 1]
    max_unaff = 0.0
    if lo > 0:
        max_unaff = max(max_unaff, float(np.max(b[:lo])))
    if hi < 2 * n - 2:
        max_unaff = max(max_unaff, float(np.max(b[hi + 1:])))
    candidates = [0.0]
    if ai > 1e-15:
        candidates.extend([0.05 * ai, 0.2 * ai, 0.5 * ai, 0.8 * ai])
        candidates.extend([1.2 * ai, 1.5 * ai, 2.0 * ai, 3.0 * ai, 5.0 * ai])
    best_x = ai
    best_sc = trial_score(b_slice, f_slice, max_unaff, ai, ai)
    for x in candidates:
        sc = trial_score(b_slice, f_slice, max_unaff, x, ai)
        if sc < best_sc:
            best_sc = sc
            best_x = x
    if best_x > 0.0:
        span = max(0.3 * ai, 0.01 * best_x)
        lo_x = max(0.0, best_x - span)
        hi_x = best_x + span
    else:
        lo_x = 0.0
        hi_x = max(0.5 * ai, 0.05)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(18):
        c = hi_x - gr * (hi_x - lo_x)
        d = lo_x + gr * (hi_x - lo_x)
        sc_c = trial_score(b_slice, f_slice, max_unaff, c, ai)
        sc_d = trial_score(b_slice, f_slice, max_unaff, d, ai)
        if sc_c <= sc_d:
            hi_x = d
        else:
            lo_x = c
        if hi_x - lo_x < 1e-13 * max(1.0, ai):
            break
    mid = (lo_x + hi_x) / 2.0
    sc_mid = trial_score(b_slice, f_slice, max_unaff, mid, ai)
    if sc_mid < best_sc:
        best_x = mid
        best_sc = sc_mid
    return max(0.0, best_x)


def run_cd(f_init, time_limit, verbose=False):
    t0 = time.monotonic()
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64)
    f = normalize(f)
    best_f = f.copy()

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = normalize(np.ones(1024))
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    print(f"\n{'='*60}")
    print(f"Starting optimization. Initial score: {best_score:.8f}, n={n_parent}")
    print(f"Target: 1.5027")
    print(f"{'='*60}")

    # Phase 1: Coordinate descent on parent
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"\n--- Phase 1: Coordinate descent (n={n_parent}, budget={t1:.0f}s) ---")
        f1 = run_cd(parent, time_limit=t1, verbose=True)
        s1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(s1) and s1 < best_score:
            best_score = s1
            best_seq = f1.copy()
            print(f"  IMPROVED: {best_score:.8f}")
        else:
            print(f"  No improvement: CD gave {s1:.8f}")

    # Phase 2: Smooth descent polish
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 120.0)
    if t2 > 15:
        print(f"\n--- Phase 2: Smooth descent (n={len(best_seq)}, budget={t2:.0f}s) ---")
        f2 = run_smooth_descent(best_seq, time_limit=t2, tau_init=0.2, verbose=True)
        s2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(s2) and s2 < best_score:
            best_score = s2
            best_seq = f2.copy()
            print(f"  IMPROVED: {best_score:.8f}")
        else:
            print(f"  No improvement: SD gave {s2:.8f}")

    # Phase 3: CD again on improved
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 20:
        print(f"\n--- Phase 3: CD polish (n={len(best_seq)}, budget={t3:.0f}s) ---")
        f3 = run_cd(best_seq, time_limit=t3, verbose=True)
        s3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_seq = f3.copy()
            print(f"  IMPROVED: {best_score:.8f}")
        else:
            print(f"  No improvement: CD gave {s3:.8f}")

    # Phase 4: Multi-length exploration
    print(f"\n--- Phase 4: Multi-length exploration ---")
    n_current = len(best_seq)
    target_lengths = [896, 1024, 1152, 1280, 1408, 1536, 1792, 1920, 2048, 2304]
    target_lengths = [t for t in target_lengths if t != n_current and 640 <= t <= 2500]
    target_lengths.sort(key=lambda x: abs(x - n_current))

    for target_n in target_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 80:
            print("  Not enough time remaining, skipping further lengths.")
            break
        t4 = min(rem * 0.13, 130.0)
        if t4 < 25:
            break

        print(f"  Exploring n={target_n} (budget={t4:.0f}s)")

        f_interp = interpolate_sequence(best_seq, target_n)
        inits = [
            f_interp,
            hann_window(target_n),
            blackman_window(target_n),
            gaussian_window(target_n, 0.35),
        ]

        best_for_n = None
        best_sc_n = float('inf')
        t_per = t4 / len(inits)

        for init_idx, ini in enumerate(inits):
            if time.monotonic() > deadline - 20:
                break
            t_sub = min(t_per, deadline - time.monotonic() - 10)
            if t_sub < 10:
                break

            f_opt = run_cd(ini, time_limit=t_sub)
            sc = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(sc) and sc < best_sc_n:
                best_sc_n = sc
                best_for_n = f_opt
            if np.isfinite(sc) and sc < best_score:
                print(f"    init[{init_idx}] n={target_n}: score={sc:.8f} (NEW BEST!)")

        if best_for_n is not None and best_sc_n < best_score:
            best_score = best_sc_n
            best_seq = best_for_n.copy()
            print(f"  NEW BEST at n={target_n}: {best_score:.8f}")
        else:
            print(f"  n={target_n}: best={best_sc_n:.8f} (no global improvement)")

    # Phase 5: Final CD burst
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.35, 200.0)
    if t5 > 20:
        print(f"\n--- Phase 5: Final CD (n={len(best_seq)}, budget={t5:.0f}s) ---")
        f5 = run_cd(best_seq, time_limit=t5, verbose=True)
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  IMPROVED: {best_score:.8f}")

    # Phase 6: Final smooth descent
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.3, 60.0)
        if t6 > 8:
            f6 = run_smooth_descent(best_seq, time_limit=t6, tau_init=0.3)
            s6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(s6) and s6 < best_score:
                best_score = s6
                best_seq = f6.copy()
                print(f"  Final SD improved: {best_score:.8f}")

    # Phase 7: Tiny final CD
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem * 0.4, 30.0)
        if t7 > 5:
            f7 = run_cd(best_seq, time_limit=t7)
            s7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(s7) and s7 < best_score:
                best_score = s7
                best_seq = f7.copy()
                print(f"  Final CD improved: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    else:
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.8f}, n={len(best_seq)}")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:+.6f}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
