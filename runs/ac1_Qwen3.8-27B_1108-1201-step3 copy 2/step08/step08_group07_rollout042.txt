def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response_full(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0. (All constraints active.)"""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None),
                   verbose=False, tol_gap_abs=1e-10, tol_gap_rel=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=(prev_g is not None))
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

        A = rfft(a, fft_size)

        # grad_a(conv_k) = 2 * (shifted/reversed a) for the relevant lag

        # But we need to account for the linear convolution structure
        # Actually for linear conv: (a*a)[k] = sum_i a[i]*a[k-i]
        # d/d(a[j]) (a*a)[k] = a[k-j] + a[k-j] = 2*a[k-j] (for interior)
        # So grad_a[j] = 2 * sum_k w_k * a[k-j] = 2 * (w * a_reversed)[j]
        # In FFT: this is 2 * IFFT(FFT(w) .* FFT(a)) but with the right shift
        # Let's use the correlation interpretation:
        # sum_k w_k * a[k-j] = cross-correlation of w and a evaluated at j
        # = IFFT(conj(FFT(a)) * FFT(w)) ... hmm this is getting complex
        # Simpler: just compute directly for small n, or use the FFT trick

        # d/d(a[j]) = sum_k w_k * d(a[k-j])/d(a[j]) = sum_k w_k * [k-j in range] * a[k-j]
        # Let l = k-j, sum over valid l: w[j+l] * a[l] (where j+l in range)
        # This is the cross-correlation of w and a at offset j
        # Using FFT: IFFT(conj(FFT(a_pad)) * FFT(w_pad)) gives circular cross-correlation
        # For linear: we need zero-padding to fft_size
        
        A_full = rfft(a, fft_size)
        W_full = rfft(w, fft_size)
        # Cross-correlation: corr[j] = sum_l a[l] * w[l+j] (linear)
        # = IFFT(conj(FFT(a)) * FFT(w)) (circular, but with proper padding it's linear)

        # Let me just compute: grad[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n else 0 + a[k-j] if 0<=k-j<n else 0)
        # = 2 * sum_k w[k] * a[k-j] * [0<=k-j<n] (the factor 2 is because both terms contribute)
        # Wait no. (a*a)[k] = sum_i a[i]a[k-i]. d/d(a[j]) = a[k-j] (if valid) + a[k-j] (if valid, from j appearing as the second index)
        # = 2*a[k-j] if 0<=k-j<n, else just a[k-j] if k-j=j (i.e., one term)
        # For simplicity, when n is large and j is not at the boundary, it's 2*a[k-j]
        # The FFT approach: if we define b = a reversed, then (a*a)[k] = sum_i a[i]*b[n-1-k+i]... 
        # This is getting too complicated. Let me just use the direct computation via the correlation.
        
        # Direct approach: grad_a[j] = sum_k w[k] * d(a*a)[k]/da[j]
        # d(a*a)[k]/da[j] = a[k-j] + a[k-j] (if both indices valid)

        # d/d(a[j]) = a[k-j] (if 0<=k-j<=n-1, from the a[i] term with i=j)
        #           + a[k-j] (if 0<=k-j<=n-1, from the a[k-i] term with k-i=j, i.e., i=k-j)
        # If j != k-j (i.e., 2j != k), both terms are present: 2*a[k-j]
        # If 2j == k, only one term: a[j]
        # So grad_a[j] = 2*sum_{k: 0<=k-j<n, 2j!=k} w[k]*a[k-j] + sum_{k: k=2j} w[k]*a[j]
        # = 2*sum_{l: 0<=l<n, l!=j} w[j+l]*a[l] + w[2j]*a[j] (if 2j valid)

        # Using FFT for the main term:
        # sum_l w[j+l]*a[l] = cross-correlation
        # Let's compute: C[j] = sum_{l=0}^{n-1} a[l] * w[l+j] (where w is zero-padded)
        # This is a "sliding dot product" = IFFT(FFT(w_padded) * conj(FFT(a_padded)))

        # grad_a = 2 * np.convolve(w, a[::-1], mode='same') ... no that's not right either.

        # grad_a[j] = sum_k w[k] * (a[k-j] if 0<=k-j<n else 0)
        # This is: for each j, sum over k in range, weight w[k]*a[k-j]
        # = cross-correlation of w and a, evaluated at offset j

        # corr[j] = sum_l a[l] * w[j+l] (mod fft_size), but we want linear
        # With zero padding to fft_size >= 2n-1, circular = linear for valid indices
        
        # The cross-correlation R[j] = sum_l a[l] * conj(w[l-j]) = sum_l a[l+j] * conj(w[l])
        # Hmm, let me just use numpy correlate for correctness
        # Actually for n up to 2048, direct O(n^2) is too slow. Let me use FFT properly.
        
        # (a*a)[k] = sum_i a[i]*a[k-i]
        # Define the "gradient row" for the k-th output: it's a vector v_k where v_k[j] = a[k-j] (if valid)
        # grad_a[j] = sum_k w[k] * v_k[j] = sum_k w[k] * a[k-j] * [0<=k-j<n]
        # = sum_{l=0}^{n-1} a[l] * w[l+j] * [0<=l+j<2n-1]  (substituting l = k-j)
        # = sum_l a[l] * w_ext[l+j] where w_ext is w zero-padded to length 2n-1-n+1...

        # Let W = rfft(w, fft_size) and A = rfft(a, fft_size)
        # Then IFFT(W * A) gives the circular convolution (w * a)_circ
        # IFFT(conj(W) * A) gives the circular cross-correlation
        # We want: sum_{l} a[l] * w[l+j] = (a * w_reversed)[n-1+j] ... 
        # This is getting too complicated. Let me just compute it as:
        # grad_a = irfft(rfft(a, fft_size) * rfft(w_reversed, fft_size), fft_size)
        # where w_reversed = w[::-1] zero-padded
        
        # Actually the cleanest way:
        # sum_k w[k] * a[k-j] = (w * a_flip)[n-1-j] where a_flip = a[::-1]
        # Because (w * a_flip)[m] = sum_k w[k] * a_flip[m-k] = sum_k w[k] * a[n-1-m+k]
        # Let l = n-1-m+k, so k = l - n + 1 + m
        # = sum_l w[l-n+1+m] * a[l]
        # For m = n-1-j: sum_l w[l-j] * a[l] ... that's sum_l a[l]*w[l-j]
        # We want sum_k w[k]*a[k-j]. Let k' = k-j: sum_k w[k]*a[k-j] = sum_{k'} w[k'+j]*a[k']
        # = sum_l a[l]*w[l+j] (renaming k' to l)
        
        # So we need sum_l a[l]*w[l+j]. This is the "anti-correlation" or "cross-covariance".
        # = (a * w_flip)[n-1+j] where w_flip = w[::-1] (zero-padded to appropriate length)
        # (a * w_flip)[m] = sum_l a[l] * w_flip[m-l] = sum_l a[l] * w[n-1-(m-l)] = sum_l a[l]*w[n-1-m+l]
        # For m = n-1+j: sum_l a[l]*w[n-1-(n-1+j)+l] = sum_l a[l]*w[l-j]
        # That gives w[l-j], not w[l+j]. 
        # For m = n-1-j: sum_l a[l]*w[n-1-(n-1-j)+l] = sum_l a[l]*w[j+l]. YES!
        
        # So: grad_a[j] / 2 = (a * w_flip)[n-1-j] where w_flip = w[::-1] zero-padded
        # Using FFT: IFFT(A * W_flip) where W_flip = FFT(w_flip)
        
        w_flip = np.zeros(fft_size)
        w_flip[:len(w)] = w[::-1]
        W_flip = rfft(w_flip, fft_size)
        A_pad = rfft(a, fft_size)
        cross = irfft(A_pad * W_flip, fft_size)
        # cross[m] = (a * w_flip)[m] (linear, due to zero padding)
        # We want cross[n-1-j] for j = 0,...,n-1
        grad_a = 2.0 * cross[n - 1 : n - 1 + n][::-1].copy()
        # Wait, let me double-check the indexing.
        # We want grad_a[j] = 2 * sum_l a[l] * w[l+j] = 2 * cross[n-1-j]
        # So grad_a = 2 * cross[n-1 : n-1-(n-1) : -1] = 2 * cross[n-1 :: -1][:n]
        # In numpy: grad_a = 2.0 * cross[n-1:n-1-n-1:-1][:n] ... simpler:
        grad_a = 2.0 * np.array([cross[n - 1 - j] for j in range(n)])
        # This is O(n) indexing, fine.
        
        # But wait, there's the subtlety that when l+j >= len(w) (i.e., l+j >= 2n-1), w is 0.
        # And when l+j < 0, it's also 0 (but l>=0, j>=0 so l+j>=0 always).
        # Also, the "diagonal" correction (when k=2j) should give a[j] not 2*a[j].
        # For large n this is negligible.
        
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        # The objective is 2n * LSE(T*conv)/T, so gradient w.r.t. a is 2n * grad_a / T * T = 2n * grad_a
        # Wait: LSE(T*conv) = (1/T)log(sum exp(T*conv_k)), d/d(a) = sum w_k * grad_a(conv_k) where w = softmax(T*conv)
        # So d/d(a) [2n * LSE(T*conv)] = 2n * sum_k w_k * grad(conv_k)/da = 2n * grad_a (as computed, with w already included)
        # Then d/dz = diag(a) * (d/da - (d/da)^T a)
        return 2.0 * n * grad_z

def lbgfs_simple(a0, time_limit, T=500, maxiter=3000, verbose=False):
    """Simpler L-BFGS that directly optimizes in a-space with projection."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    def func_and_grad(a):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(a)
        conv = fast_convolve(a)
        shifted = conv / T
        m = shifted.max()
        w = np.exp(shifted - m)
        w = w / w.sum()
        val = 2.0 * n * T * (m + np.log(w.sum()))
        # Gradient w.r.t. a (before projection)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        w_flip = np.zeros(fft_size)
        w_flip[:len(w)] = w[::-1]
        W_flip = rfft(w_flip, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(A * W_flip, fft_size)
        grad = 2.0 * np.array([cross[n - 1 - j] for j in range(n)])
        val = 2.0 * n * T * (m + np.log(np.exp(shifted - m).sum()))
        return val, 2.0 * n * grad

    # Manual L-BFGS with simplex projection
    # Use scipy with bounds (all >= 0) and equality constraint sum=1
    from scipy.optimize import minimize as sp_min
    
    def f(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e15
        x = x / sx
        conv = fast_convolve(x)
        shifted = conv / T
        m = shifted.max()
        lse = m + np.log(np.exp(shifted - m).sum())
        return 2.0 * n * T * lse

    def g(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(x)
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            return np.zeros_like(x)
        x = x / sx
        conv = fast_convolve(x)
        shifted = conv / T
        m = shifted.max()
        w = np.exp(shifted - m)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        w_flip = np.zeros(fft_size)
        w_flip[:len(w)] = w[::-1]
        W_flip = rfft(w_flip, fft_size)
        A = rfft(x, fft_size)
        cross = irfft(A * W_flip, fft_size)
        grad = 2.0 * np.array([cross[n - 1 - j] for j in range(n)])
        # Project gradient to simplex tangent (remove component along 1-vector)
        grad = grad - grad.mean()
        return 2.0 * n * grad

    try:
        res = sp_min(fun=f, x0=a, jac=g, method='L-BFGS-B',
                     bounds=[(0, None)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                              'maxfun': maxiter * 200, 'disp': False})
        a_out = np.maximum(res.x, 0)
        so = a_out.sum()
        if so > 1e-15:
            a_out = a_out / so
            score = compute_score(a_out)
            if score < best_score:
                best_a = a_out
                best_score = score
    except Exception:
        pass

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, verbose=True):
    """Iteratively target and reduce the top peaks of the convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        peak_val = conv.max()
        peak_idx = conv.argmax()
        
        # Compute gradient of the peak w.r.t. a
        # d(conv[peak_idx])/da[j] = a[peak_idx - j] (if valid) + a[peak_idx - j] (if valid, other side)
        # = 2*a[peak_idx - j] for interior, a[j] for diagonal
        k = peak_idx
        grad = np.zeros(n)
        for j in range(n):
            l = k - j
            if 0 <= l < n:
                if 2 * j == k:
                    grad[j] = a[l]  # diagonal: only one term
                else:
                    grad[j] = 2.0 * a[l]
        
        # Subgradient step: move a in direction -grad, projected to simplex
        step_size = 0.01 / (np.linalg.norm(grad) + 1e-10)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
        
        # Occasionally add noise to escape
        if iters % 100 == 0:
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PeakRed iter={iters}, best={best_score:.8f}, peak={peak_val:.6f}, time={elapsed:.0f}s")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, verbose=True, label="MW"):
    """Multiplicative weights / exponential update for peak reduction."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 2.0  # learning rate

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        peak_idx = conv.argmax()
        peak_val = conv[peak_idx]
        
        # Compute "loss gradient" for the peak
        k = peak_idx
        loss_grad = np.zeros(n)
        for j in range(n):
            l = k - j
            if 0 <= l < n:
                if 2 * j == k:
                    loss_grad[j] = a[l]
                else:
                    loss_grad[j] = 2.0 * a[l]
        
        # Exponential update
        eta_curr = eta / (1.0 + 0.01 * iters)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta_curr * loss_grad
        log_a_new -= log_a_new.max()  # normalization
        a_new = np.exp(log_a_new)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=500, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j] + a[i]) if a[j] > 1e-15 else a[i]

            for frac in [0.03, 0.07, 0.15, 0.3, 0.5]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.002
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Check a candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t1, step_init=0.5,
                                       use_topk=False, label="LP-full")
    best_seq, best_score = check_and_update(a_lp1, "lp_full", best_seq, best_score)

    # === Phase 2: LP Fictitious Play (top-K constraints) ===
    print(f"\n=== Phase 2: LP-FP TopK (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t2, step_init=0.4,
                                           use_topk=True, topk=8, label="LP-top8")
        best_seq, best_score = check_and_update(a_lp2, "lp_top8", best_seq, best_score)
        a_lp2b, sc_lp2b = lp_fictitious_play(best_seq, 60.0, step_init=0.3,
                                             use_topk=True, topk=15, label="LP-top15")
        best_seq, best_score = check_and_update(a_lp2b, "lp_top15", best_seq, best_score)

    # === Phase 3: Multiplicative Weights ===
    print(f"\n=== Phase 3: Multiplicative Weights (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if rem > 40:
        print(f"  Budget: {t3:.0f}s")
        a_mw, sc_mw = multiplicative_weights(best_seq, t3, label="MW")
        best_seq, best_score = check_and_update(a_mw, "mw", best_seq, best_score)

    # === Phase 4: L-BFGS Smoothing (multi-temperature) ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 500, 1000, 2000]:
        rem = deadline - time.monotonic()
        t4 = min(rem * 0.10, 80.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t4:.0f}s")
        a_lb, sc_lb = lbgfs_simple(best_seq, t4, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 5: Targeted Peak Reduction ===
    print(f"\n=== Phase 5: Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if rem > 30:
        print(f"  Budget: {t5:.0f}s")
        a_pr, sc_pr = targeted_peak_reduction(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_pr, "peak_red", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if rem > 20:
        print(f"  Budget: {t6:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t6, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord", best_seq, best_score)

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.8f}) ===")
    alt_lengths = []
    if n_parent % 2 == 0:
        alt_lengths = [n_parent + 2, n_parent - 2, n_parent + 4, n_parent + 6]
    else:
        alt_lengths = [n_parent + 1, n_parent - 1, n_parent + 3, n_parent - 3]
    alt_lengths = [l for l in alt_lengths if 64 <= l <= 2048]
    
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t7 = min(rem * 0.10, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t7:.0f}s")
        a_alt, sc_alt = lbgfs_simple(init_alt, t7, T=500, maxiter=2000)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.30, 120.0)
        print(f"  Budget: {t8:.0f}s")
        a_final, sc_final = lp_fictitious_play(best_seq, t8, step_init=0.2,
                                               use_topk=False, label="LP-final")
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # === Phase 9: Final L-BFGS + CD polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.30, 60.0)
        a_final2, sc_final2 = lbgfs_simple(best_seq, t9, T=300, maxiter=2000)
        best_seq, best_score = check_and_update(a_final2, "final_lbfgs", best_seq, best_score)
        
        rem = deadline - time.monotonic()
        if rem > 10:
            t10 = min(rem * 0.5, 30.0)
            a_final3, sc_final3 = coordinate_descent(best_seq, t10, n_pairs=200, verbose=False)
            best_seq, best_score = check_and_update(a_final3, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
