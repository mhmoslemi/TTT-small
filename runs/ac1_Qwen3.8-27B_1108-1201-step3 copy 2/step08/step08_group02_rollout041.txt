from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute subgradient of max(conv(a,a)) at positions K."""
    n = len(a)
    sg = np.zeros(n)
    for k in K:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # sg[i] += a[k-i] for i in [lo, hi]
        for i in range(lo, hi + 1):
            sg[i] += a[k - i]
    return sg

    """
    t0 = time.monotonic()
    n = len(a_init)

        M = C.max()

            sg[idx] += a[k - idx]

    n = len(a)
    size = 2 * n - 1

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]

    b_ub = np.full(Kf, M)
    bounds = [(0, None)] * n
    try:

                      method='highs', options={'time_limit': time_limit})

            if g.sum() > 0:

def subgradient_direction(a, K, n):
    sg = np.zeros(n)
    for k in K:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        sg[lo:hi+1] += a[k-lo:k-hi+1]
    return sg


def do_line_search(a, h, M_cur):
    best_t = 0.0
    best_obj = M_cur
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
        trial = (1-t)*a + t*h
        trial = np.maximum(trial, 0)
        s = trial.sum()
        if s < 1e-10:
            continue
        trial /= s
        obj = fast_convolve(trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if best_t > 0:
        for t in [best_t*0.7, best_t*1.3, best_t*1.6]:
            if t <= 0 or t > 0.95:
                continue
            trial = (1-t)*a + t*h
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            obj = fast_convolve(trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def optimize(a_init, t_limit, label="", max_peaks=120, init_frac=0.995,
             final_frac=0.92, patience=25):
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init.astype(float), 0)
    a /= a.sum()
    C = fast_convolve(a)
    best_M = C.max()
    best_a = a.copy()
    it = 0
    stuck = 0
    frac = init_frac
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < t_limit:
        it += 1
        C = fast_convolve(a)
        M = C.max()
        
        thr = frac * M
        K = np.where(C >= thr)[0]
        if len(K) > max_peaks:
            K = np.argpartition(C, -max_peaks)[-max_peaks:]
        elif len(K) < 5:
            K = np.argpartition(C, -max(10, len(C)//30))[-max(10, len(C)//30):]
        
        # LP direction
        g = solve_lp(a, K, M, n)
        
        if g is not None and g.sum() > 1 + 1e-5:
            h = g / g.sum()
        else:
            # Subgradient fallback
            sg = subgradient_direction(a, K, n)
            sn = np.linalg.norm(sg)
            if sn < 1e-15:
                stuck += 1
                if stuck > patience:
                    a = best_a + rng.standard_normal(n) * 0.003 * best_a
                    a = np.maximum(a, 1e-10)
                    a /= a.sum()
                    stuck = 0
                continue
            h = project_simplex(a - 0.1 * (sg / sn), 1.0)
        
        t_best, obj_best = do_line_search(a, h, M)
        
        if t_best > 0 and obj_best < M:
            a = (1 - t_best) * a + t_best * h
            a = np.maximum(a, 0)
            a /= a.sum()
            stuck = 0
        else:
            stuck += 1
        
        # Track best
        C = fast_convolve(a)
        M = C.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
        
        # Adapt
        if it % 30 == 0:
            frac = max(final_frac, frac - 0.004)
        
        if stuck > patience:
            a = best_a + rng.standard_normal(n) * 0.003 * best_a
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            stuck = 0
            frac = init_frac
        
        if it % 100 == 0:
            print(f"  {label} it={it} score={2*n*best_M:.6f} peaks={len(K)} "
                  f"stuck={stuck} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_M


def sg_polish(a_init, t_limit, label="sg"):
    """Subgradient with momentum."""
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init.astype(float), 0)
    a /= a.sum()
    C = fast_convolve(a)
    best_M = C.max()
    best_a = a.copy()
    eta = 0.015
    v = np.zeros(n)
    beta = 0.85
    no_imp = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < t_limit:
        it += 1
        C = fast_convolve(a)
        M = C.max()
        n_p = min(20, len(C))
        K = np.argpartition(C, -n_p)[-n_p:]
        thr = C.max() * 0.99
        K = K[C[K] >= thr]
        if len(K) == 0:
            K = np.array([np.argmax(C)])
        
        sg = subgradient_direction(a, K, n)
        sn = np.linalg.norm(sg)
        if sn < 1e-15:
            break
        sg /= sn
        
        v = beta * v - eta * sg
        a_new = project_simplex(a + v, 1.0)
        
        # Line search
        best_t = 0
        best_obj = M
        for t in [0.1, 0.3, 0.5, 1.0]:
            tr = (1-t)*a + t*a_new
            tr = np.maximum(tr, 0)
            s = tr.sum()
            if s < 1e-10:
                continue
            tr /= s
            o = fast_convolve(tr).max()
            if o < best_obj:
                best_obj = o
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*a_new
            a = np.maximum(a, 0)
            a /= a.sum()
        
        C = fast_convolve(a)
        M = C.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 15 == 0:
                eta *= 0.9
            if no_imp > 50:
                a = best_a + rng.standard_normal(n) * 0.002 * best_a
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                v = np.zeros(n)
                eta = 0.015
                no_imp = 0
        
        if it % 200 == 0:
            print(f"  {label} it={it} score={2*n*best_M:.6f} eta={eta:.5f} "
                  f"t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_M


def interp_length(a, new_n):
    n = len(a)
    x = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

**Alternative: Direct quadratic programming**

The problem is: minimize `max_k conv(a,a)[k]` subject to `a >= 0, sum(a) = 1`.

This can be written as: minimize M subject to `a >= 0, sum(a) = 1, conv(a,a)[k] <= M for all k`.

The constraints `conv(a,a)[k] <= M` are quadratic (bilinear) in a. So this is a non-convex optimization problem (the feasible set is not convex because of the bilinear constraints).

Wait, is it actually non-convex? Let's see: `conv(a,a)[k] = sum_i a[i] * a[k-i]`. This is a quadratic form in a. The set `{a : a^T Q a <= M}` is convex if Q is positive semidefinite.

For the auto-convolution, the quadratic form is `sum_i a[i] * a[k-i]`. This is NOT a standard quadratic form (it's not `a^T Q a` for a fixed Q). It's more like a "shifted" quadratic form.

Actually, `conv(a,a)[k] = (S_k a)^T a` where S_k is a shift matrix. So it's `a^T S_k a`. The matrix S_k is symmetric (since `a^T S_k a = a^T S_k^T a` for the relevant range), but it's not necessarily positive semidefinite.

Hmm, for non-negative a, the bilinear form `sum_i a[i] * a[k-i]` is always non-negative. But the constraint set `{a >= 0 : sum_i a[i] * a[k-i] <= M}` is not necessarily convex.

Example: n=2, k=0. `conv(a,a)[0] = a[0]^2`. The set `{a >= 0, a[0]^2 <= M}` is convex (it's just `a[0] <= sqrt(M)`).

n=2, k=1. `conv(a,a)[1] = 2*a[0]*a[1]`. The set `{a >= 0, 2*a[0]*a[1] <= M}` is... the superlevel set of a product. This is NOT convex! (The hypograph of a product of two non-negative variables is not convex.)

So the problem is indeed non-convex. The LP approach works around this by linearizing (using the current a as the "first factor" in the bilinear form). This is a valid approach (similar to DC programming or successive linearization).

Given this, the LP approach is reasonable. The key is to make it converge fast.
