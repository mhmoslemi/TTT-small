                prob.solve(solver=cp.MOSEK, msdpdp={'intpnt_co_tol_rel_gap': 1e-9})
            elif sname == 'SCS':
                prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
            elif sname == 'ECOS':
                if n > 3000:
                    continue
                prob.solve(solver=cp.ECOS, max_iters=100000)
            elif sname == 'CLARABEL':
                prob.solve(solver=cp.CLARABEL)
            else:
                continue
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(g_val)) and g_val.sum() > 1e-10:
                    break
                else:
                    g_val = None
        except Exception:
            g_val = None
            continue
    return g_val


def cp_lp_optimize(f_init, time_limit, initial_k=50, max_k=400, label="cp"):
    """Cutting-plane LP optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    # Initialize active set with top peaks
    k_init = min(initial_k, len(b))
    active_idx = np.argpartition(b, -k_init)[-k_init:]
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Solve LP
        g = solve_lp(f, n, M, active_idx)
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-7:
            stuck += 1
            if stuck > 25:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_k = min(max_k, len(b_cur))
                new_top = np.argpartition(b_cur, -new_k)[-new_k:]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane: check violated constraints
        fg = convolve_direction(f, g)
        violated = fg > M * (1.0 + 1e-8)
        if violated.any():
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, set(active_idx.tolist()))]
            if len(not_active) > 0 and len(active_idx) < max_k:
                n_add = min(40, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                # Re-solve
                g = solve_lp(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-7:
            stuck += 1
            if stuck > 25:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search: combined scan + refinement
        best_t = 0.1
        best_t_obj = M
        scan_ts = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        for t in scan_ts:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-12:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        lo = max(0.0, best_t - 0.08)
        hi = min(1.0, best_t + 0.15)
        if hi > lo + 1e-4:
            for _ in range(8):
                m1 = lo + (hi - lo) * 0.382
                m2 = hi - (hi - lo) * 0.382
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                fa1 = fa1 / sa1 if sa1 > 1e-12 else f
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                fa2 = fa2 / sa2 if sa2 > 1e-12 else f
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) * 0.5
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-12:
                f_mid = f_mid / s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        # Update
        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        f = f / s if s > 1e-12 else np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically refresh active set
        if it % 12 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = np.argsort(b[active_idx])[-max_k:]
            else:
                new_top = np.argpartition(b, -min(max_k, len(b)))[-min(max_k, len(b)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
                active_idx = np.argsort(b[active_idx])[-max_k:]

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            score_est = 2.0 * n * best_obj
            print(f"  {label} it={it} obj={best_obj:.10f} est_score={score_est:.6f} "
                  f"k={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def smooth_max_gradient(f_init, time_limit, alpha=50.0, n_top=60, lr_init=0.003,
                        momentum=0.9, label="sg"):
    """Smooth-max gradient descent with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    alpha_cur = alpha
    it = 0
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        b_max = b.max()

        # Compute softmax weights over top peaks
        scaled = b / (b_max * 0.1)
        scaled = scaled - scaled.max()
        w = np.exp(scaled)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w = np.zeros_like(w)
            w[top_idx] = np.exp(scaled[top_idx] - scaled[top_idx].max())

        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Subgradient: for each peak index k with weight w[k],
        # gradient contribution to f[j] is w[k] * (f[k-j] + f[j-k]) (with bounds)
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            # (a*a)_k = sum_i a_i * a_{k-i}
            # d/d a_j of (a*a)_k = a_{k-j} (if 0<=k-j<n) + a_{j-k} (if 0<=j-k<n, j!=k-j)
            # Simplified: contribution from index k to gradient at j:
            #   f[k-j] if 0 <= k-j < n
            #   f[j-k] if 0 <= j-k < n and j != k-j
            k_minus_j = k - np.arange(n)  # k - j for j=0..n-1
            valid1 = (k_minus_j >= 0) & (k_minus_j < n)
            grad[valid1] += w[k] * f[k_minus_j[valid1]]
            j_minus_k = np.arange(n) - k  # j - k for j=0..n-1
            valid2 = (j_minus_k >= 0) & (j_minus_k < n)
            # Avoid double counting when j = k-j, i.e., 2j = k
            if k % 2 == 0:
                j_half = k // 2
                if 0 <= j_half < n:
                    valid2[j_half] = False
            grad[valid2] += w[k] * f[j_minus_k[valid2]]

        # Update with momentum
        v = momentum * v - lr * grad
        f_new = f + v
        f = project_simplex(f_new)

        # Periodic evaluation
        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.85
                    alpha_cur *= 1.15
                    no_improve = 0
                if no_improve > 30:
                    # Restart with perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

            if it % 200 == 0:
                print(f"  {label} it={it} obj={best_obj:.10f} est_score={2.0*n*best_obj:.6f} "
                      f"lr={lr:.6f} alpha={alpha_cur:.1f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def diversify(f_init, rng, strength=0.1):
    """Create a diversified version of f_init."""
    n = len(f_init)
    # Random perturbation
    pert = rng.standard_normal(n) * strength
    f_new = f_init + pert
    return project_simplex(f_new)


def upsample(f_old, n_new):
    """Upsample a sequence to a new length via interpolation."""
    old_x = np.linspace(0.0, 1.0, len(f_old))
    new_x = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(new_x, old_x, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    return f_new / s if s > 1e-12 else np.ones(n_new) / n_new


def make_structured_init(n, kind="uniform"):
    """Create a structured initial sequence."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        x = np.linspace(-3, 3, n)
        f = np.exp(-0.5 * x ** 2)
        return f / f.sum()
    elif kind == "exponential":
        f = np.exp(-2.0 * np.linspace(0, 1, n))
        return f / f.sum()
    elif kind == "bimodal":
        f = np.exp(-10.0 * np.sin(np.pi * np.linspace(0, 1, n)) ** 2)
        return f / f.sum()
    elif kind == "cosine":
        x = np.linspace(0, 1, n)
        f = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        f = np.maximum(f, 0.0)
        return f / f.sum()
    else:
        return np.ones(n) / n
