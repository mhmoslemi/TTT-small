"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using diverse non-symmetric initializations,
tournament-based L-BFGS LogSumExp optimization with multi-stage annealing,
LP-based alternating refinement, subgradient mirror descent, and multi-length exploration.
Key: exploits non-symmetry to reduce autocorrelation peak below the Cauchy-Schwarz bound.
"""

import numpy asnp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_a(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b>=0, sum=1, minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_a(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.3, maxiter=8000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            Tb = T_val * b_in
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            Tb = T_val * b_in
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for idx_stage, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters_here = max(20, int(iters_per_stage * (remaining / time_limit + 0.5)))
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_here, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_here * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 5:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def mirror_descent_opt(a0, time_limit, lr_init=0.1, n_steps=100000):
    """Mirror descent with subgradient oracle on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.01 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-16:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            lr *= 0.8
            no_improve = 0
        lr = max(lr_init / (1.0 + step * 0.001), 1e-9)

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=20000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.03 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:10]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:3]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.96
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_diverse_inits(n, rng, parent=None):
    """Generate many diverse non-symmetric initializations."""
    inits = []

    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramps (various slopes and directions)
    x = np.linspace(0, 1, n)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = x ** alpha
        a = a / a.sum()
        inits.append(a.copy())
        a_rev = a[::-1].copy()
        inits.append(a_rev)

    # Exponential decays (various rates, from both ends)
    for tau in [0.2, 0.3, 0.5, 0.7, 1.0]:
        a = np.exp(-x / tau)
        a = a / a.sum()
        inits.append(a.copy())
        a_rev = a[::-1].copy()
        inits.append(a_rev)

    # Offset Gaussians
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for sigma in [0.05, 0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            a = a / a.sum()
            inits.append(a.copy())

    # Sine/cosine windows
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.sin(np.pi * x) ** p
        a = a / a.sum()
        inits.append(a.copy())
    for p in [0.5, 1.0, 2.0]:
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** p
        a = a / a.sum()
        inits.append(a.copy())

    # Comb patterns (non-symmetric)
    for period in [3, 4, 5, 7]:
        a = np.ones(n)
        a[n // 3:] *= 0.5
        a = a / a.sum()
        inits.append(a.copy())

    # Quadratic with offset
    for c in [0.3, 0.5, 0.7]:
        a = 1.0 - ((x - c) ** 2)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a = a / a.sum()
            inits.append(a.copy())

    # Random
    for seed in [1, 7, 42, 123, 999]:
        r = np.random.default_rng(seed)
        a = r.random(n)
        a = a / a.sum()
        inits.append(a.copy())

    # Beta-distribution samples (various shapes)
    for alpha_b, beta_b in [(0.5, 2.0), (2.0, 0.5), (0.5, 5.0), (5.0, 0.5), (0.3, 1.0), (1.0, 0.3)]:
        r = np.random.default_rng(alpha_b * 10 + beta_b)
        a = r.beta(alpha_b, beta_b, size=n)
        a = a / a.sum()
        inits.append(a.copy())

    return inits


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def quick_lbgfs(a0, time_limit=5.0, T_start=50.0, T_end=2.0, maxiter=2000):
    """Quick L-BFGS for tournament."""
    return lbgfs_refine(a0, time_limit, T_start=T_start, T_end=T_end, maxiter=maxiter)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with diverse initialization and tournament selection.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
            return True
        return False

    # === Phase 1: Diverse tournament at current length ===
    print(f"\n=== Phase 1: Tournament n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_tournament = min(rem * 0.15, 150.0)
    if t_tournament > 20:
        inits = generate_diverse_inits(n_best, rng, parent)
        # Deduplicate (remove very similar ones)
        unique_inits = [inits[0]]
        for init in inits[1:]:
            if len(unique_inits) >= 20:
                break
            is_unique = True
            for u in unique_inits:
                if np.linalg.norm(init - u) / np.linalg.norm(init) < 0.1:
                    is_unique = False
                    break
            if is_unique:
                unique_inits.append(init)

        candidates = []
        per_init_time = max(3.0, t_tournament / len(unique_inits))
        for i, init in enumerate(unique_inits):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_here = min(per_init_time, rem - 5)
            if t_here < 2:
                break
            a_opt = quick_lbgfs(init, time_limit=t_here, T_start=30.0, T_end=3.0, maxiter=1500)
            sc = 2.0 * n_best * fast_convolve(a_opt).max()
            candidates.append((sc, a_opt))
            if i % 5 == 0:
                print(f"  Tournament {i}/{len(unique_inits)}: best_so_far={min(c[0] for c in candidates):.6f}")

        candidates.sort(key=lambda x: x[0])
        print(f"  Tournament done: top score={candidates[0][0]:.6f}")

        # Keep top 3 for deep optimization
        top_three = [c[1] for c in candidates[:3]]
    else:
        top_three = [best_seq.copy()]

    # === Phase 2: Deep L-BFGS on top candidates ===
    print(f"\n=== Phase 2: Deep L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_deep = min(rem * 0.30, 400.0)
    if t_deep > 30:
        per_deep = t_deep / len(top_three)
        for i, a_start in enumerate(top_three):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_here = min(per_deep, rem - 10)
            a_opt = lbgfs_refine(a_start, time_limit=t_here, T_start=80.0, T_end=0.2, maxiter=10000)
            try_update(a_opt, f"deep_lbgfs_{i}")

    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.20, 300.0)
    if t_lp > 20:
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=80000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 4: Second LP pass + mirror descent ===
    print(f"\n=== Phase 4: LP + Mirror (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.12, 180.0)
    if t_lp2 > 15:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver=solver, n_max_iters=40000)
        try_update(a_lp2, "nesterov_lp2")

    rem = deadline - time.monotonic()
    t_md = min(rem * 0.06, 90.0)
    if t_md > 10:
        a_md, _ = mirror_descent_opt(best_seq, t_md, lr_init=0.08, n_steps=100000)
        try_update(a_md, "mirror_desc")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.06, 80.0)
    if t_ps > 8:
        a_ps, _ = peak_shave_local(best_seq, t_ps, n_steps=25000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        alt_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
            alt_n = int(n_best * mult)
            alt_n = max(128, min(4096, alt_n))
            if alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:3]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_total = min(rem * 0.25, 200.0)
            if t_total < 30:
                break
            print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

            # Try both interpolated and fresh structured init
            init_interp = interpolate_sequence(best_seq, alt_n)
            init_struct = generate_diverse_inits(alt_n, rng, None)
            # Pick a non-symmetric one
            a_start = init_struct[2] if len(init_struct) > 2 else init_interp

            t_half = t_total * 0.55
            a_opt = lbgfs_refine(a_start, time_limit=t_half, T_start=60.0, T_end=0.5, maxiter=6000)
            try_update(a_opt, f"lbgfs_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
                t_lp_alt = min(rem * 0.3, 60.0)
                if t_lp_alt > 10:
                    a_lp_alt, _ = nesterov_lp_refine(best_seq, t_lp_alt, solver=solver, n_max_iters=15000)
                    try_update(a_lp_alt, f"lp_n{alt_n}")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.15, 120.0)
        n_cur = len(best_seq)
        for noise_scale in [0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_here = min(t_pert / 2, rem - 10)
            if t_here < 10:
                break
            a_pert = lbgfs_refine(perturbed, time_limit=t_here, T_start=10.0, T_end=0.5, maxiter=5000)
            try_update(a_pert, f"perturbed_{noise_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.1, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps2 = min(rem * 0.2, 30.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps2, n_steps=10000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.3, 25.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
