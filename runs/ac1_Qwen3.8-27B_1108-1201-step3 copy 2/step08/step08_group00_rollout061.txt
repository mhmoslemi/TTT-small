```python
from scipy.optimize import linprog

def solve_lp_best_response(a_fixed, m, T_matrix):
    """Solve the best response LP.
    a_fixed: length n, symmetric
    m: n/2
    T_matrix: n x m matrix where T[k,j] = a[k-j] + a[k-n+1+j]
    """
    n = len(a_fixed)
    # Variables: y_0..y_{m-1}, M (all >= 0), total m+1
    c = np.zeros(m + 1)
    c[m] = 1.0  # minimize M
    
    # Equality: sum(y) = 1/2
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([0.5])
    
    # Inequality: T @ y - M <= 0 for each k
    A_ineq = np.zeros((n, m + 1))
    A_ineq[:, :m] = T_matrix
    A_ineq[:, m] = -1.0
    b_ineq = np.zeros(n)
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * (m + 1), method='highs')
    if res.status != 0:
        return None, None
    y_val = res.x[:m]
    M_val = res.x[m]
    return y_val, M_val
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using symmetric LP alternation,
FISTA with adaptive step, and coordinate descent polish.
Enforces symmetry (a[i] = a[n-1-i]) to halve the search space.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def make_symmetric_from_half(x):
    """Create full symmetric sequence from its first half."""
    n = 2 * len(x)
    return np.concatenate([x, x[::-1]])


def half_of_symmetric(a):
    """Get first half of a symmetric sequence."""
    return a[:len(a) // 2].copy()


def normalize_symmetric(a):
    """Enforce symmetry and normalize to sum 1."""
    a = symmetrize(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_T_symmetric(a):
    """Build n x m matrix where (T @ x) = first n entries of conv(a, b)
    when b is symmetric with first half x."""
    n = len(a)
    m = n // 2
    k = np.arange(n)
    j = np.arange(m)
    idx1 = k[:, None] - j[None, :]
    idx2 = k[:, None] - n + 1 + j[None, :]
    T = np.zeros((n, m))
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    T[valid1] = a[idx1[valid1]]
    T[valid2] += a[idx2[valid2]]
    return T


def solve_lp_best_response(T_matrix, m):
    """Solve: min M s.t. T@x <= M, sum(x) = 1/2, x >= 0.
    Returns (x, M) or (None, None)."""
    # Variables: x_0..x_{m-1}, M (all >= 0)
    c = np.zeros(m + 1)
    c[m] = 1.0
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([0.5])
    n_rows = T_matrix.shape[0]
    A_ub = np.zeros((n_rows, m + 1))
    A_ub[:, :m] = T_matrix
    A_ub[:, m] = -1.0
    b_ub = np.zeros(n_rows)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * (m + 1), method='highs')
    if res.status != 0:
        return None, None
    return res.x[:m].copy(), float(res.x[m])


def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)


def nesterov_lp_symmetric(a0, time_limit, verbose=True):
    """Nesterov dual averaging with symmetric LP best responses."""
    n = len(a0)
    m = n // 2
    a = normalize_symmetric(a0.copy())
    x = half_of_symmetric(a)  # m-dimensional, sum = 1/2
    y = x.copy()  # Dual variable (half-vector)
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Build T for initial a
    T = build_T_symmetric(a)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        # Solve LP for best response to y
        # We need T_y: the convolution matrix with y as the "fixed" sequence
        # conv(y_full, b) where y_full = make_symmetric_from_half(y)
        y_full = make_symmetric_from_half(y)
        T_y = build_T_symmetric(y_full)
        
        x_best, M_val = solve_lp_best_response(T_y, m)
        if x_best is None:
            break
        
        # Nesterov update
        alpha = 2.0 / (iteration + 2)
        y = (1 - alpha) * y + alpha * x_best
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy * 0.5  # Normalize to sum 1/2
        
        # Primal: weighted average
        a_new = make_symmetric_from_half(y)
        a_full = fast_convolve(a_new)
        score_new = 2.0 * n * a_full.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Also try the LP solution directly
        a_lp = make_symmetric_from_half(x_best)
        a_lp = a_lp / a_lp.sum()  # Should already be sum=1
        score_lp = score_of(a_lp)
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_lp.copy()
        
        iteration += 1
        if verbose and iteration % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
