def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    # T1[j] = conv(a, w[::-1])[2n-2-j] for j = 0..n-1
    w_rev = softmax_w[::-1]
    cw_rev = convolve_ab(a, w_rev)  # length 3n-2
    # We need cw_rev[2n-2-j] for j=0..n-1, i.e., indices from 2n-2 down to n-1
    T1 = cw_rev[n-1:2*n-1][::-1]
    
    grad = T1 + T2
    return smooth_val, grad

def smooth_max_optimize(f_init, time_limit, tau_start=50, tau_end=500, 
                        lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8,
                        n_iter_check=50):
    """

    """
    t0 = time.monotonic()
    n = len(f_init)

    t_step = 0
    tau = tau_start
    
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_step += 1

        progress = (time.monotonic() - t0) / time_limit

        smooth_val, grad = smoothed_max_gradient(f, tau)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)

def lp_optimize(f_init, time_limit, top_k=80, max_constraints=500):
    """
    LP-based optimization with aggressive constraint generation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Start with top-k peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        active_list = top_idx.tolist()
        active_set = set(active_list)
        
        g = None
        for cg_iter in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            
            K = len(active_list)
            if K > max_constraints:
                b_active = b[active_list]
                keep = np.argsort(b_active)[-max_constraints:]
                active_list = [active_list[i] for i in keep]
                active_set = set(active_list)
                K = len(active_list)
            
            # Build constraint matrix
            A = np.zeros((K, n))
            for row_i in range(K):
                k = active_list[row_i]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                A[row_i, lo:hi+1] = f[lo:hi+1]
            
            g_var = cp.Variable(n)
            prob = cp.Problem(
                cp.Maximize(cp.sum(g_var)),
                [g_var >= 0, A @ g_var <= M]
            )
            
            solved = False
            for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, max_iters=200000)
                    solved = True
                    break
                except Exception:
                    continue
            
            if not solved or g_var.value is None:
                break
            
            g = np.maximum(g_var.value, 0.0)
            if np.any(np.isnan(g)) or np.any(np.isinf(g)):
                g = None
                break
            
            # Check for violated constraints (check ALL)
            fg = fast_cross_convolve(f, g)
            mask = np.zeros(len(b), dtype=bool)
            mask[active_list] = True
            violations = np.where((fg > M + 1e-9) & (~mask))[0]
            
            if len(violations) == 0:
                break
            
            # Add ALL violated constraints (more aggressive)
            for v_obj in violations:
                vi = int(v_obj)
                if vi not in active_set:
                    active_set.add(vi)
                    active_list.append(vi)
        
        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        h = g / g.sum()
        
        # Line search
        best_t = 0.05
        best_trial = best_obj
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t
        
        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 20 == 0:
            print(f"  LP iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def proposed_max_gradient(a, tau, top_k=None):
    """
    Compute gradient of smoothed max of auto-convolution, optionally restricted to top-k.
    Returns (value, gradient).
    """
    b = fast_convolve(a)
    if top_k is not None and top_k < len(b):
        # Restrict to top-k peaks
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = np.zeros_like(b)
        b_top[top_idx] = b[top_idx]
        b = b_top
    
    b_max = b.max()
    if b_max < 1e-15:
        return 0.0, np.zeros(len(a))
    
    shifted = b - b_max
    lb = tau * shifted
    lb = np.clip(lb, -500, 500)
    lb_max = lb.max()
    exp_lb = np.exp(lb - lb_max)
    total = exp_lb.sum()
    if total < 1e-30:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w = exp_lb / total
    
    smooth_val = b_max + (1.0 / tau) * np.log(max(total, 1e-300))
    
    # Gradient via FFT
    n = len(a)
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # T1[j] = conv(a, w[::-1])[2n-2-j]
    # T2[j] = conv(a, w)[j]
    
    cw = convolve_ab(a, w)  # length n + (2n-1) - 1 = 3n-2
    T2 = cw[:n]
    
    w_rev = w[::-1]
    cw_rev = convolve_ab(a, w_rev)
    T1 = cw_rev[n-1:2*n-1][::-1]
    
    grad = T1 + T2
    return smooth_val, grad

def adam_smooth_optimize(f_init, time_limit, tau_start=30.0, tau_end=200.0,
                         lr_init=0.005, top_k=None):
    """
    Adam optimization of smoothed max auto-convolution peak.
    Uses adaptive tau annealing and restarts.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    rng = np.random.default_rng(12345)
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    lr = lr_init
    no_improve = 0
    tau = tau_start
    
    while time.monotonic() - t0 < time_limit:
        t_step += 1
        progress = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        smooth_val, grad = proposed_max_gradient(f, tau, top_k=top_k)
        
        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t_step)
        v_hat = v / (1 - 0.999 ** t_step)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        f = f_new
        
        # Check every 20 steps
        if t_step % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.02)
            else:
                no_improve += 1
                if no_improve % 10 == 0:
                    lr *= 0.9
                if no_improve > 50:
                    # Restart with perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    no_improve = 0
                    lr = lr_init * 0.5
            
            if t_step % 200 == 0:
                print(f"  adam: step={t_step} tau={tau:.1f} obj={best_obj:.10f} "
                      f"score={2*n*best_obj:.6f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def flat_top_window(n):
    w = np.ones(n)
    a_coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a_coeffs):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def tuslar_window(n):
    """Tuslar (minimum sidelobe) window - optimal for our purpose."""
    # Use a combination approach: start with flat-top and apply iterative sidelobe reduction
    w = flat_top_window(n)
    return w

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_max_grad(a, tau, top_k=None):
    """
    Compute value and gradient of smoothed max of auto-convolution.
    Uses FFT-based correlation for O(n log n) gradient computation.
    Returns (smooth_value, gradient_vector).
    """
    b = fast_convolve(a)
    if top_k is not None and top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_masked = np.zeros_like(b)
        b_masked[top_idx] = b[top_idx]
        b = b_masked

    b_max = b.max()
    if b_max < 1e-15:
        return 0.0, np.zeros(len(a))

    shifted = b - b_max
    lb = tau * shifted
    lb = np.clip(lb, -500.0, 500.0)
    lb_max = lb.max()
    exp_lb = np.exp(lb - lb_max)
    total = exp_lb.sum()
    if total < 1e-30:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w = exp_lb / total

    smooth_val = b_max + (1.0 / tau) * np.log(max(total, 1e-300))

    # Gradient: grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i]
    # Computed via: corr = np.correlate(w, a, 'full'), grad = 2*corr[:n][::-1]
    n = len(a)
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[:n][::-1]

    return smooth_val, grad


def adam_smooth_optimize(f_init, time_limit, tau_start=30.0, tau_end=300.0,
                         lr_init=0.01, top_k=None, beta1=0.9, beta2=0.999):
    """
    Adam optimization of smoothed max auto-convolution peak on the probability simplex.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    rng = np.random.default_rng(42)
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    lr = lr_init
    no_improve = 0
    tau = tau_start
    tau_anneal_power = 0.6

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (progress ** tau_anneal_power)

        smooth_val, grad = compute_smooth_max_grad(f, tau, top_k=top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)
        f = f_new

        # Check improvement every 15 steps
        if t_step % 15 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.005, 0.05)
            else:
                no_improve += 1
                if no_improve % 8 == 0:
                    lr *= 0.93
                if no_improve > 40:
                    # Restart with perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    no_improve = 0
                    lr = lr_init * 0.5
                    tau = max(10.0, tau * 0.5)

            if t_step % 150 == 0:
                print(f"    adam: step={t_step} tau={tau:.1f} obj={best_obj:.10f} "
                      f"score={2*n*best_obj:.6f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def peak_shave(f_init, time_limit, n_peaks=3):
    """
    Direct peak shaving: iteratively reduce the top peaks using their gradients.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        b = fast_convolve(f)

        # Get top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Weight peaks by their excess over the median
        b_sorted = np.sort(b)
        median_b = b_sorted[len(b_sorted) // 2]
        weights = np.maximum(b - median_b, 0.0)
        w_sum = weights.sum()
        if w_sum < 1e-20:
            weights = np.ones(len(b)) / len(b)
        else:
            weights /= w_sum

        # Only use top peaks
        if n_peaks < len(b):
            w_sel = np.zeros(len(b))
            w_sel[top_idx] = weights[top_idx]
            ws = w_sel.sum()
            if ws > 1e-20:
                w_sel /= ws
            weights = w_sel

        # Compute combined gradient using correlation
        corr_w = np.correlate(weights, f, mode='full')
        grad = 2.0 * corr_w[:n][::-1]

        # Try multiple step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            f_trial = project_simplex(f - alpha * grad / max(grad.max(), 1e-15))
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            no_improve += 1
            continue

        f = project_simplex(f - best_alpha * grad / max(grad.max(), 1e-15))
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_obj < 1e-10:
            break

    return best_f, best_obj
