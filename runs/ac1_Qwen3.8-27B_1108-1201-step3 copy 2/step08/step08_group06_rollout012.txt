```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Mirror Descent (entropy-based multiplicative weights) with temperature annealing
2. LP Fictitious Play with geometric mean updates
3. Projected subgradient method (L2 projection onto simplex)
4. Coordinate descent with exact line search
5. Length exploration and final polish

Key novelty: Fast O(n log n) mirror descent with adaptive temperature scheduling,
combined with LP-based refinement and spectral initialization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    Computed via FFT as: 2 * (w * reverse(a))[n-1 : 2n-2]
    """
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)
    conv_result = irfft(W * A_rev, fft_size)[:size]
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def softmax_safe(x, T=1.0):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x_max = x.max()
    if x_max > 700 * T:
        x = x - x_max
        x_max = 0.0
    exp_x = np.exp(x / T - x_max / T)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return exp_x / s


def mirror_descent(a0, time_limit, T_start=500.0, T_end=5.0,
                   eta_start=5.0, eta_end=0.1, verbose=True):
    """
    Mirror descent with entropy regularizer (multiplicative weights update).
    Adaptive temperature annealing and step size schedule.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Log-space representation for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    
    n_iters = 0
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature and step size
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        # Compute convolution and softmax weights
        conv = fast_convolve(a)
        w = softmax_safe(conv, T)
        
        # Compute gradient
        grad = compute_gradient(a, w)
        
        # Mirror descent update in log-space
        log_a = log_a - eta * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        n_iters += 1
        
        if patience > max_patience:
            # Perturb to escape local minimum
            rng = np.random.default_rng(n_iters)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            log_a = np.log(np.maximum(a, 1e-300))
            patience = 0
        
        if verbose and n_iters % 1000 == 0:
            print(f"    MD iter={n_iters}, T={T:.1f}, eta={eta:.4f}, "
                  f"score={score_new:.8f}, best={best_score:.8f}, "
                  f"elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def projected_subgradient(a0, time_limit, verbose=True):
    """
    Projected subgradient method with projection onto the simplex.
    Uses the exact subgradient of max_k (a*a)(k).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    step_size = n / (n + 10) * 0.5
    n_iters = 0
    patience = 0
    max_patience = 2000
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Adaptive step size
        step = step_size * (1.0 / (1.0 + 0.01 * n_iters))
        
        # Compute subgradient at active peaks
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Use top peaks with softmax weighting
        T_sub = max_val * 0.1
        w = softmax_safe(conv, T_sub)
        grad = compute_gradient(a, w)
        
        # Gradient step
        v = a - step * grad
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.clip(v, 0, None)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-15:
            a_new = a_new / a_new_sum
        else:
            a_new = a.copy()
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        n_iters += 1
        
        if patience > max_patience:
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            patience = 0
        
        if verbose and n_iters % 2000 == 0:
            print(f"    PS iter={n_iters}, score={score_new:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def build_C(f):
    """Build constraint matrix for LP: C[k,j] = f[k-j] if valid."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def greedy_lp(f, M):
    """Greedy approximation to LP: max sum(g) s.t. (f*g)(k) <= M, g >= 0."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]
        valid = f_slice > 1e-15
        max_gj = np.inf
        if valid.any():
            for idx in range(n):
                if f_slice[idx] > 1e-15:
                    cand = slack[idx] / f_slice[idx]
                    if cand < max_gj:
                        max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def lp_fictitious_play(f_init, time_limit, verbose=True):
    """LP-based fictitious play with geometric mean updates."""
    f = np.maximum(f_init.copy(), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        g = greedy_lp(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Geometric mean update
        t_step = 2.0 / (k + 3)
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_sum = a_new.sum()
        if a_sum < 1e-30:
            break
        a_new = a_new / a_sum
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, score={score_new:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        
        # Generate candidate pairs
        n_pairs = min(n, 300)
        pairs = []
        
        # Peak-focused
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(n_pairs // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Random
        for _ in range(n_pairs // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_at_t(t):
                a_t = a.copy()
                a_t[i] -= t
                a_t[j] += t
                if a_t[i] < -1e-14 or a_t[j] < -1e-14:
                    return 1e10
                a_t = np.maximum(a_t, 0)
                ss = a_t.sum()
                if ss < 1e-15:
                    return 1e10
                a_t = a_t / ss
                return compute_score(a_t)
            
            f1 = eval_at_t(x1)
            f2 = eval_at_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_at_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_at_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 5 == 0:
                print(f"    CD score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def spectral_init(n, verbose=True):
    """
    Create an initial sequence with a relatively flat power spectrum
    using a chirp-like phase modulation.
    """
    # Start with a broad spectrum shape
    a = np.ones(n) / n
    # Add structured perturbation
    t = np.linspace(0, 1, n)
    for freq in [3, 7, 13]:
        a += 0.02 * np.sin(2 * np.pi * freq * t + freq * t * t)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    if verbose:
        sc = compute_score(a)
        print(f"    Spectral init: n={n}, score={sc:.8f}")
    return a


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization combining mirror descent, LP fictitious play,
    projected subgradient, and coordinate descent.

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = spectral_init(n_init)
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    print(f"Starting best: {best_score:.8f}")
    
    # === Phase 1: Mirror Descent with annealing ===
    print(f"\n=== Phase 1: Mirror Descent [annealing] (best={best_score:.8f}) ===")
```
