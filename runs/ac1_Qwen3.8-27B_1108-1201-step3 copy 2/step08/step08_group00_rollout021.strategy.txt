        base_seq = np.exp(-0.5 * np.linspace(0, 1, len(height_sequence_1))**2)  # Smoother than step function
        base_n = len(base_seq)
        if base_n not in [2,4,8,16,32,64,128,256,512,1024]:
            base_n = 512
        candidate_lengths.insert(0, base_n)

    # Sort candidates from smallest to largest
    candidate_lengths = sorted(set(candidate_lengths))
    
    # Initialize
    best_seq = None
    best_score = float('inf')

    # Time budget allocation
    total_time = budget
    for idx, n in enumerate(candidate_lengths):
        # Exponential budget decay - smaller sequences get more time
        budget_per_n = total_time * (0.6) ** idx
        current_time = time.time()
        
        # Optimization
        seq = structured_init(n)
        optimized_seq = budget_active_optimization(seq, budget_per_n)
        score = evaluate_sequence(optimized_seq)
        
        # Update best candidate
        if score < best_score:
            best_score = score
            best_seq = optimized_seq

    return best_seq
