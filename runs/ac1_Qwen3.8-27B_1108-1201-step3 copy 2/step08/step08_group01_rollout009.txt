    # c[i] = sum_m a[m] * w_rev[2n-2-i-m]
    # = (a * w_rev)[2n-2-i] (convolution at index 2n-2-i)
    # So c = conv(a, w_rev)[n-1 : 2n-1][::-1]...

    # For n up to 4096, O(n^2) is 16M which takes ~10ms in numpy

    # grad_a[i] = 2 * sum_{k} w[k] * a[k-i]  (a out of range = 0)
    #           = 2 * (w ⊛ a)[i]  where ⊛ is cross-correlation

    a_pad[:n] = a
    W = rfft(w, 2*n-1)
    Ap = rfft(a_pad, 2*n-1)
    # Cross-correlation: IFFT(FFT(w) * conj(FFT(a_pad)))
    # This gives sum_k w[k] * conj(a_pad[k-i]) for circular... 
    # Actually in numpy/scipy:
    # np.correlate(w, a_pad, mode='full') gives the cross-correlation
    # But that's O(n^2)

    # This gives the circular cross-correlation, which for zero-padded signals gives linear
    corr = irfft(W * np.conj(Ap), 2*n-1)

    # For i in 0..n-1, we want sum_k w[k] * a_pad[k-i] where k-i in [0, n-1]
    # With zero-padding, a_pad[m] = 0 for m >= n
    # So corr[i] = sum_{k=0}^{2n-2} w[k] * a_pad[(k-i) mod (2n-1)]
    # For i >= 0: (k-i) mod (2n-1) = k-i if k>=i, else k-i+(2n-1)
    # a_pad[k-i] = a[k-i] if 0<=k-i<n, else 0
    # a_pad[k-i+2n-1] = 0 (since k-i+2n-1 >= 2n-1)
    # So corr[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i]
    # That's exactly what we want!
    
    grad_a = 2.0 * corr[:n]

    # a = softmax(x), so da_i/dx_j = a_i * (delta_{ij} - a_j)
    # dx_j = sum_i grad_a[i] * da_i/dx_j = a_j * (grad_a[j] - sum_i a_i * grad_a[i])
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using convex optimization:
1. Log-sum-exp (LSE) smoothing with adaptive beta + L-BFGS
2. FISTA (accelerated projected gradient) for robustness
3. Active-set peak equalization for final refinement
4. Coordinate descent polishing
Since the problem is convex (min of max of quadratic forms over simplex),
any local optimum is global. Multiple runs with different parameters
ensure we find the best numerical solution.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_val_and_grad(a, beta):
    """Compute LSE value and gradient w.r.t. a for beta * conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    
    # LSE value
    max_c = conv.max()
    bc = beta * (conv - max_c)
    lse = max_c + np.log(np.sum(np.exp(bc))) / beta
    
    # Softmax weights
    w = np.exp(bc)
    w /= w.sum()
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # Using FFT: corr = ifft(fft(w) * conj(fft(a_pad)))
    L = 2 * n - 1
    a_pad = np.zeros(L)
    a_pad[:n] = a
    W = rfft(w, L)
    Ap = rfft(a_pad, L)
    corr = irfft(W * np.conj(Ap), L)
    grad = 2.0 * corr[:n]
    
    return lse, grad, conv, w

def lbfgs_lse_optimize(a_init, beta_schedule, time_limit, label=""):
    """
    L-BFGS optimization of LSE-approximated objective.
    Uses log parameterization: a = softmax(x).
    """
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-15)
    x0 = np.log(a_safe)
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_convolve(a_init).max()
    t0 = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj(x):
            xm = x.max()
            ax = np.exp(x - xm)
            a = ax / ax.sum()
            val, _, _, _ = lse_val_and_grad(a, beta)
            return val
        
        def grad(x):
            xm = x.max()
            ax = np.exp(x - xm)
            a = ax / ax.sum()
            _, grad_a, _, _ = lse_val_and_grad(a, beta)
            dot = np.dot(a, grad_a)
            return a * (grad_a - dot)
        
        try:
            result = minimize(
                obj, x0.copy(), jac=grad, method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            x0 = result.x
        except Exception as e:
            print(f"    L-BFGS error beta={beta}: {e}")
            break
        
        # Evaluate actual score
        xm = x0.max()
        a = np.exp(x0 - xm)
        a /= a.sum()
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
        
    return best_a, best_score

def fista_lse(a_init, beta, n_iters=5000, time_limit=60.0, label=""):
    """
    FISTA (accelerated proximal gradient) for the LSE-approximated problem.
    Projected onto the simplex at each step.
    """
    n = len(a_init)
    a = a_init / a_init.sum()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Compute Lipschitz constant (upper bound)
    # The Hessian of LSE(beta * conv) is bounded by... let's just estimate it
    # For now, use a conservative step size
    conv0 = fast_convolve(a)
    max_conv = conv0.max()
    # Lipschitz constant of gradient is at most 2 * beta * maxConv * 2 (rough)
    L = 4.0 * beta * max(n, 1) * max_conv
    step = 1.0 / L
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        _, grad_y, _, _ = lse_val_and_grad(y, beta)
        # Gradient step
        a_new = y - step * grad_y
        # Project to simplex
        a_new = simplex_project(a_new)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y = a_new + ((t_k - 1) / t_new) * (a_new - a)
        a = a_new
        t_k = t_new
        
        # Evaluate
        if it % 10 == 0:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    # Final evaluation
    score = 2.0 * n * fast_convolve(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def coord_descent_polish(a_init, time_limit, n_trials=8, label=""):
    """
    Fast coordinate descent with O(n) sensitivity per coordinate.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]  # a[k-i] for k in [k_start, k_end]
            
            best_delta = 0.0
            best_new_max = M
            
            for trial in range(n_trials):
                if trial < n_trials // 2:
                    delta = -0.5 * a[i] * (trial + 1) / (n_trials // 2)
                else:
                    delta = a[i] * (trial - n_trials // 2 + 1) / (n_trials - n_trials // 2)
                
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    if not isinstance(new_affected, np.ndarray) or new_affected is affected:
                        new_affected = new_affected.copy()
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    if changes is a_shifted or not isinstance(changes, np.ndarray):
                        changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def peak_equalize(a_init, time_limit, K=10, label=""):
    """
    Active-set peak equalization: iteratively identify top-K peaks
    and solve a small QP to equalize them.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active set: peaks within 1% of max
        threshold = M * 0.999
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # For each active peak k, the constraint is:
        # sum_j a[j]*a[k-j] <= t
        # We want to minimize t subject to these constraints + simplex
        
        # Use L-BFGS with penalty
        def penalty_obj(a_flat):
            a_p = simplex_project(a_flat)
            c = fast_convolve(a_p)
            pen = 0.0
            for k in active:

                pen += 100.0 * violation * violation
            return c.max() + pen
        
        # Simpler approach: just do gradient step on the active peaks
        # Compute gradient of c[k] for active k

        bc = 40.0 * (conv - max_c)
        w = np.exp(bc)
        w /= w.sum()
        
        L = 2 * n - 1
        a_pad = np.zeros(L)
        a_pad[:n] = a
        W = rfft(w, L)
        Ap = rfft(a_pad, L)

        grad = 2.0 * corr[:n]
        
        # Gradient step + project
        step = 1.0 / (4.0 * 40.0 * M * n)  # conservative
        a_new = a - step * grad

        if score < best_score:
