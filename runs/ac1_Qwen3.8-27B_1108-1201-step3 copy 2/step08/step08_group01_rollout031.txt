    A = rfft(a, fft_size)

def simplex_project(v):

    n = len(v)

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def precise_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Precise coordinate descent tracking sum.
    For each coordinate, tries multiple candidate deltas and picks the best
    based on the actual score change (2n*M/s^2).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (s * s)
    best_a = a.copy()
    best_M = M
    best_s = s
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved_this_sweep = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            k_len = k_end - k_start + 1

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Coefficient vector: a[k-i] for k = k_start,...,k_end
            # k-i ranges from 0 to k_len-1
            a_coeff = a[0:k_len]

            # Candidate deltas (relative to ai)
            best_d = 0.0
            best_new_score = best_score  # we want to improve on current best

            # Generate candidates
            fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                     -0.5, -0.3, -0.2, -0.1, -0.05, -0.02, -0.01, -0.005, -0.002, -0.001]

            for fr in fracs:
                d = fr * ai
                new_ai = ai + d
                if new_ai < -1e-15:
                    continue

                # New affected conv
                new_aff = conv[k_start:k_end + 1].copy() + 2.0 * d * a_coeff
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * d * ai + d * d

                new_M = max(unaff_max, float(new_aff.max()))
                new_s = s + d
                if new_s < 0.01:
                    continue
                new_score = 2.0 * n * new_M / (new_s * new_s)

                if new_score < best_new_score:
                    best_new_score = new_score
                    best_d = d

            if best_d != 0.0 and best_new_score < best_score - 1e-14:
                # Apply the update
                a[i] = max(a[i] + best_d, 0.0)
                s_new = s + best_d

                # Update conv incrementally
                a_coeff_old = a.copy()
                a_coeff_old[i] = a_coeff_old[i] - best_d  # recover old value
                changes = 2.0 * best_d * a_coeff_old[:k_len]
                conv[k_start:k_end + 1] += changes
                if 2 * i >= k_start and 2 * i <= k_end:
                    conv[2 * i] += best_d * best_d

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                s = s_new

                if best_new_score < best_score:
                    best_score = best_new_score
                    best_a = a.copy()
                    best_M = M
                    best_s = s
                improved_this_sweep = True
                stall = 0
            else:
                stall += 1

        if not improved_this_sweep:
            if stall > 2 * n:
                break
        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {label} sweep={sweeps} ({elapsed:.1f}s): best={best_score:.7f}")

    return best_a, best_score


def pairwise_coord_descent(a0, time_limit, n_pairs_per_sweep=200, verbose=False, label=""):
    """
    Pairwise mass-transfer coordinate descent.
    For pairs (i,j), try transferring mass: a[i]+d, a[j]-d (sum unchanged).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M / (s * s)
    best_a = a.copy()

    sweeps = 0
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Select pairs: mix of random and "promising" pairs
        # Promising: high a[i] paired with low a[j]
        idx_high = np.argsort(a)[-n // 4:]
        idx_low = np.argsort(a)[:n // 4]

        pairs = []
        n_rand = n_pairs_per_sweep // 2
        for _ in range(n_rand):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((i, j))
        # Add structured pairs
        for _ in range(n_pairs_per_sweep // 2):
            i = idx_high[rng.integers(0, len(idx_high))]
            j = idx_low[rng.integers(0, len(idx_low))]
            if i != j:
                pairs.append((i, j))

        for (ii, jj) in pairs:
            if a[ii] < 1e-15 or a[jj] < 1e-15:
                continue

            # For pair (i, j): a[i] -> a[i]+d, a[j] -> a[j]-d
            # sum unchanged, so score = 2n*M_new/s^2
            # Delta conv[k] = 2d*(a[k-i] - a[k-j]) + d^2*(delta(k,2i) - 2*delta(k,i+j) + delta(k,2j))

            # Affected k range: union of [i, i+n-1] and [j, j+n-1] (clipped)
            k_start = min(ii, jj)
            k_end = max(ii + n - 1, jj + n - 1)

            k_start = max(0, k_start)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:

            if k_end < 2 * n - 2:

            diff_coeff = coeff_i - coeff_j

            scale = min(a[ii], a[jj])
            best_d = 0.0
            best_new_M = M

            for fr in [0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,

                if (ii + jj) >= k_start and (ii + jj) <= k_end:
                    idx = (ii + jj) - k_start
                    new_aff[idx] += -2.0 * d * d
                if 2 * jj >= k_start and 2 * jj <= k_end:

                    new_aff[idx] += d * d

                new_M = max(unaff_max, float(new_aff.max()))

            if best_d != 0.0 and best_new_M < M - 1e-16:
                a[ii] += best_d
                a[jj] -= best_d
                a = np.maximum(a, 0.0)

                # Update conv

                if 2 * ii >= k_start and 2 * ii <= k_end:
                    idx = 2 * ii - k_start
                    changes[idx] += best_d * best_d
                if (ii + jj) >= k_start and (ii + jj) <= k_end:
                    idx = (ii + jj) - k_start
                    changes[idx] += -2.0 * best_d * best_d
                if 2 * jj >= k_start and 2 * jj <= k_end:
                    idx = 2 * jj - k_start
                    changes[idx] += best_d * best_d

                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

def subgradient_method(a0, time_limit, verbose=False, label=""):
    """
    Subgradient method on the simplex for minimizing max_k conv(a,a)[k].
    Uses smoothed subgradient (average over top-k active lags with softmax weights).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    no_improve = 0
    smooth_window = 0
    smoothed_grad = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            if lr > 1e-5:
                lr *= 0.8
            else:
                break
            no_improve = 0

        # Compute smoothed subgradient using softmax over top lags
        k_top = min(50, 2 * n - 1)
        top_indices = np.argpartition(conv, -k_top)[-k_top:]
        top_vals = conv[top_indices]
        temp = 0.05 * (top_vals.max() - top_vals.min() + 1e-15)
        w = np.exp((top_vals - top_vals.max()) / temp)
        w = w / w.sum()

        # Subgradient: grad[i] = sum_k w_k * (a[k-i] + a[k-i]) = 2 * sum_k w_k * a[k-i]
        grad = np.zeros(n)
        for idx in range(len(top_indices)):
            if w[idx] < 1e-12:
                continue
            k = int(top_indices[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Smooth
        smooth_window = min(smooth_window + 1, 20)
        smoothed_grad = (smoothed_grad * (smooth_window - 1) + grad) / smooth_window

        # Mirror descent update (multiplicative)
        scaled = a * np.exp(-lr * smoothed_grad)
        a = simplex_project(scaled)

        lr *= 0.9995

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG {label} it={it} ({elapsed:.1f}s): best={best_score:.7f} lr={lr:.6f}")

    return best_a, best_score

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    damp = 1.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = (1.0 - damp) * a + damp * a_new
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        damp = max(0.3, damp * 0.98)

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP {label} it={iters} ({elapsed:.1f}s): best={best_score:.7f} damp={damp:.3f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng, base=None):
    """Initialize using random Fourier superposition."""
    if base is not None:
        a = base.copy()
    else:
        a = np.ones(n) / n
    n_harmonics = int(rng.integers(3, 15))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def polish_sequence(a, remaining_time, verbose=False):
    """
    Multi-stage polishing: precise CD -> pairwise CD -> subgradient -> precise CD.
    """
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)

    # Stage 1: Precise coordinate descent
    t1 = min(remaining_time * 0.35, 400.0)
    if t1 > 10:
        a1, s1 = precise_coord_descent(best_a, t1, verbose=verbose, label="polish1")
        if s1 < best_score:
            best_score = s1
            best_a = a1
        remaining_time -= (time.monotonic() - t0)

    # Stage 2: Pairwise coordinate descent
    elapsed = time.monotonic() - t0
    t2 = min(remaining_time - elapsed, 300.0)
    if t2 > 10:
        a2, s2 = pairwise_coord_descent(best_a, t2, verbose=verbose, label="polish2")
        if s2 < best_score:
            best_score = s2
            best_a = a2

    # Stage 3: Subgradient for escape
    elapsed = time.monotonic() - t0
    t3 = min(remaining_time - elapsed, 200.0)
    if t3 > 10:
        a3, s3 = subgradient_method(best_a, t3, verbose=verbose, label="polish3")
        if s3 < best_score:
            best_score = s3
            best_a = a3

    # Stage 4: Final precise CD
    elapsed = time.monotonic() - t0
    t4 = min(remaining_time - elapsed, 200.0)
    if t4 > 5:
        a4, s4 = precise_coord_descent(best_a, t4, verbose=verbose, label="polish4")
        if s4 < best_score:
            best_score = s4
            best_a = a4

    # Stage 5: Very fine pairwise
    elapsed = time.monotonic() - t0
    t5 = min(remaining_time - elapsed, 100.0)
    if t5 > 5:
        a5, s5 = pairwise_coord_descent(best_a, t5, n_pairs_per_sweep=100, verbose=verbose, label="polish5")
        if s5 < best_score:
            best_score = s5
            best_a = a5

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Load parent, evaluate
    2. Precise coordinate descent on parent
    3. LP alternating for structural change
    4. Subgradient for smooth exploration
    5. Pairwise moves for mass transfer
    6. Multi-length exploration
    7. Final intensive polish
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.7f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Default initialization
        n0 = 1024
        best_seq = np.ones(n0) / n0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n0}, score={best_score:.7f}")

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # --- Phase 1: Precise coordinate descent on parent ---
    print(f"\n=== Phase 1: Precise CD (score={best_score:.7f}) ===")
    t1_budget = 300.0
    t1_start = time.monotonic()
    a1, s1 = precise_coord_descent(best_seq, t1_budget, verbose=True, label="p1_cd")
    if s1 < best_score:
        best_score = s1
        best_seq = a1
        print(f"  Phase 1 done: {best_score:.7f}")

    # --- Phase 2: LP alternating ---
    print(f"\n=== Phase 2: LP Alternating (score={best_score:.7f}) ===")
    if solver is not None:
        rem = deadline - time.monotonic()
        t2_budget = min(rem * 0.2, 250.0)
        if t2_budget > 30:
            a2, s2 = lp_alternating(best_seq, t2_budget, solver=solver, verbose=True, label="p2_lp")
            if s2 < best_score:
                best_score = s2
                best_seq = a2
                print(f"  Phase 2 done: {best_score:.7f}")

            # Also try LP from a perturbed start
            rem = deadline - time.monotonic()
            if rem > 100 and best_score > 1.503:
                perturbed = best_seq + rng.standard_normal(len(best_seq)) * 0.02
                perturbed = np.maximum(perturbed, 0)
                perturbed = perturbed / perturbed.sum()
                t2b = min(rem * 0.15, 100.0)
                a2b, s2b = lp_alternating(perturbed, t2b, solver=solver, verbose=False, label="p2b_lp")
                if s2b < best_score:
                    best_score = s2b
                    best_seq = a2b
                    print(f"  Phase 2b done: {best_score:.7f}")

    # --- Phase 3: Subgradient method ---
    print(f"\n=== Phase 3: Subgradient (score={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t3_budget = min(rem * 0.15, 150.0)
    if t3_budget > 20:
        a3, s3 = subgradient_method(best_seq, t3_budget, verbose=True, label="p3_sg")
        if s3 < best_score:
            best_score = s3
            best_seq = a3
            print(f"  Phase 3 done: {best_score:.7f}")
        else:
            # Try from perturbed
            pert = init_fourier(len(best_seq), rng, base=best_seq)
            a3b, s3b = subgradient_method(pert, min(t3_budget, 80.0), verbose=False, label="p3b_sg")
            if s3b < best_score:
                best_score = s3b
                best_seq = a3b
                print(f"  Phase 3b done: {best_score:.7f}")

    # --- Phase 4: Pairwise coordinate descent ---
    print(f"\n=== Phase 4: Pairwise CD (score={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t4_budget = min(rem * 0.2, 250.0)
    if t4_budget > 20:
        a4, s4 = pairwise_coord_descent(best_seq, t4_budget, n_pairs_per_sweep=300, verbose=True, label="p4_pw")
        if s4 < best_score:
            best_score = s4
            best_seq = a4
            print(f"  Phase 4 done: {best_score:.7f}")

    # --- Phase 5: Multi-length exploration ---
    print(f"\n=== Phase 5: Multi-length (score={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    alt_lengths = [n_best]
    for alt in [n_best - 128, n_best + 128, n_best - 64, n_best + 64, 768, 1280, 1536, 2048]:
        if 256 <= alt <= 3000 and alt not in alt_lengths:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:5]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.2, 150.0)
        if n_alt == n_best:
            cand = best_seq.copy()
        else:
            cand = interpolate_seq(best_seq, n_alt)
            # Add some Fourier perturbation
            cand = init_fourier(n_alt, rng, base=cand)

        a_alt, s_alt = precise_coord_descent(cand, min(t_alloc, 80.0), verbose=False, label=f"p5_n{n_alt}")
        if s_alt < best_score:
            best_score = s_alt
            best_seq = a_alt
            print(f"  New best at n={n_alt}: {best_score:.7f}")

    # --- Phase 6: Final intensive polish ---
    print(f"\n=== Phase 6: Final Polish (score={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 15
    if rem > 30:
        a_fin, s_fin = polish_sequence(best_seq, rem, verbose=True)
        if s_fin < best_score:
            best_score = s_fin
            best_seq = a_fin

    # --- Verify and return ---
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
