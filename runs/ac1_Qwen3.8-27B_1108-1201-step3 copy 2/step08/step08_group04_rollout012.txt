**Step size selection:**

After solving the LP and getting g (normalized to sum=1), we want to find t that minimizes:
obj(t) = max_k ((1-t)*f + t*g) * ((1-t)*f + t*g)_k

This is a 1D optimization problem. We can use golden section search or just try a grid of t values.

Actually, a simpler approach: since the LP ensures that conv(f, g) <= M, we have:
conv((1-t)f + tg, (1-t)f + tg) = (1-t)^2 conv(f,f) + 2t(1-t) conv(f,g) + t^2 conv(g,g)
<= (1-t)^2 M + 2t(1-t) M + t^2 max(conv(g,g))
= M((1-t)^2 + 2t(1-t)) + t^2 * max(conv(g,g))
= M(1-t^2) + t^2 * max(conv(g,g))

So the new max is at most M(1-t^2) + t^2 * G where G = max(conv(g,g)).
If G < M, then the new max is less than M for any t > 0.
If G > M, we need t to be small enough.

A good step size: t = (M - M_new_target) / (G - M) ... this is getting complicated. Let me just use a bisection search.

Actually, the simplest approach that works well in practice: use t = 0.3 to 0.5 and see if it improves. If not, halve it.

"""
Optimized LP-based minimization of 2*n*max(conv(a,a))/(sum(a))^2.
Uses dense LP (all convolution constraints), KKT-based multiplicative weight polishing,
and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build the 2n-1 x n matrix A where (A@g)_k = (f*g)_k."""
    full_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(full_len):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(full_len, n))
    return A

def solve_dense_lp(f, M, solver=None):
    """Solve max sum(g) s.t. g >= 0, A@g <= M. Returns g or None."""
    n = len(f)
    full_len = 2 * n - 1
    A = build_conv_matrix(f, n)
    
    g = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g))
    cons = [g >= 0, A @ g <= M]
    prob = cp.Problem(obj, cons)
    
    if solver is None:
        # Try solvers in order of preference
        for s in ['CBC', 'GLPK', 'ECOS', 'SCS']:
            try:
                if s == 'CBC':
                    prob.solve(solver=cp.CBC, gap_rel=1e-10)
                elif s == 'GLPK':
                    prob.solve(solver=cp.GLPK, gap_rel=1e-10)
                elif s == 'ECOS':
                    prob.solve(solver=cp.ECOS, max_iters=50000, abs_tol=1e-10)
                elif s == 'SCS':
                    prob.solve(solver=cp.SCS, max_iters=500000, eps_abs=1e-10, eps_rel=1e-10)
                if g.value is not None:
                    return np.maximum(g.value, 0.0)
            except Exception:
                continue
    else:
        try:
            prob.solve(solver=solver)
        except Exception:
            return None
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

def compute_pressure(w, f):
    """Compute p_i = sum_k w_k * f[k-i] (cross-correlation)."""
    return np.correlate(w, f, mode='valid')

def kkt_polish(f_init, time_limit, alpha_init=0.05, n_iter_max=5000):
    """KKT-based multiplicative weight polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    best_score = 2.0 * n * best_obj
    alpha = alpha_init
    no_improve = 0
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Soft active set
        temp = max(1e-6 * M, M * 0.005)
        w = np.exp((b - M) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2*n-1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Compute pressure
        p = compute_pressure(w, f)
        
        # KKT: p should be constant on support
        support = f > 1e-10
        if not support.any():
            break
        target = p[support].mean()
        
        if target < 1e-15:
            break
        
        # Multiplicative update
        p_safe = np.maximum(p, 1e-15)
        ratio = target / p_safe
        f_new = f * (ratio ** alpha)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            best_score = 2.0 * n * obj
            no_improve = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            f = f_new  # still accept for exploration
            no_improve += 1
            if no_improve > 50:
                alpha *= 0.9
                if alpha < 0.001:
                    break
                no_improve = 0
    
    return best_f, best_score

def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=100):
    """Subgradient optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    v = np.zeros(n)
    lr = lr_init
    temp = 0.001
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Subgradient of max_k (f*f)_k
        # Use softmax weighting
        scaled = (b - M) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Restrict to top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new
        
        # Compute subgradient: g_i = sum_k w_k * 2 * f[k-i]
        p = compute_pressure(w, f)
        g = 2.0 * p
        
        # Update with momentum
        v = 0.85 * v - lr * g
        f = project_simplex(f + v)
        
        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                best_score = 2.0 * n * obj
                no_improve = 0
                lr = min(lr * 1.03, 0.02)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0
            
            if it % 150 == 0:
                temp = min(temp * 1.15, 0.5)
            
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0
    
    return best_f, best_score

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.astype(np.float64).copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

At the optimum, the KKT conditions imply:

This is a very strong condition. It means that a is (approximately) the inverse filter of mu (in some sense).

1. Guess the active set (which peaks are at the maximum)
2. Compute the corresponding mu (dual variables)
3. Solve for a such that conv(a, mu) is constant
4. Check if conv(a,a) has the right peaks active
5. Iterate

"""
Dense LP + KKT multiplicative-weight polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key ideas:
1. Use ALL convolution constraints (dense LP) via scipy HiGHS solver for exact directions
2. KKT-based multiplicative weight update as a complement to LP direction finding
3. Careful 1D line search for step size
4. Multi-length exploration around parent
5. Symmetry exploitation option for speed

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build 2n-1 x n matrix A where (A@g)_k = (f*g)_k."""
    full_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(full_len):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(full_len, n))


def solve_lp_dense(f, M):
    """Solve max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k. Uses HiGHS."""
    n = len(f)
    full_len = 2 * n - 1
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(full_len, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    # Fallback: top-k LP
    K = min(200, full_len)
    b = fast_convolve(f)
    top_idx = np.argpartition(b, -K)[-K:]
    A_sub = A[top_idx, :]
    b_sub = np.full(K, M)
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20.0})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_pressure(w, f):
    """Compute p_i = sum_k w[k] * f[k-i] via cross-correlation."""
    return np.correlate(w, f, mode='valid')


def kkt_polish(f_init, time_limit, alpha_init=0.04):
    """KKT-based multiplicative weight polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    alpha = alpha_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Soft active set with adaptive temperature
        temp = max(1e-7, 0.003 * M)
        w = np.exp((b - M) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Compute pressure p_i = sum_k w[k] * f[k-i]
        p = compute_pressure(w, f)

        # KKT condition: p should be ~constant on support
        support = f > 1e-10
        if not support.any():
            break
        target = p[support].mean()
        if target < 1e-15:
            break

        # Multiplicative update
        p_safe = np.maximum(p, 1e-15)
        ratio = target / p_safe
        f_new = f * (ratio ** alpha)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            best_score = 2.0 * n * obj
            no_improve = 0
            alpha = min(alpha * 1.03, 0.3)
        else:
            no_improve += 1
            if no_improve > 30:
                alpha *= 0.85
                if alpha < 0.002:
                    # Try perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    best_obj = fast_convolve(f).max()
                    alpha = alpha_init * 0.5
                no_improve = 0

        f = f_new

        if it % 200 == 0:
            print(f"    KKT it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"alpha={alpha:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def subgradient_polish(f_init, time_limit, lr_init=0.002):
    """Subgradient method with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    lr = lr_init
    temp = 0.002
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    no_improve = 0
    it = 0
    rng = np.random.default_rng(555)
    n_top = max(50, n // 10)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weighting
        scaled = (b - M) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        # Subgradient: grad of max_k (f*f)_k approximated by weighted sum
        p = compute_pressure(w, f)
        g = 2.0 * p

        v = 0.85 * v - lr * g
        f = project_simplex(f + v)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                best_score = 2.0 * n * obj
                no_improve = 0
                lr = min(lr * 1.02, 0.015)
            else:
                no_improve += 1
                if no_improve > 15:
                    lr *= 0.88
                    no_improve = 0

            if it % 100 == 0:
                temp = min(temp * 1.1, 0.3)
                print(f"    SG it={it} obj={best_obj:.10f} score={best_score:.6f} "

            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.astype(np.float64).copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

            # Try very small step

def coordinate_descent(f_init, time_limit):
    """Fast coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5, 4.0]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_score

        f_res, _ = coordinate_descent(best_seq, time_limit=t6)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [CD]: {real_score:.6f}")

    rem = deadline - time.monotonic()
    if rem > 8:
        f_res, _ = kkt_polish(best_seq, time_limit=min(rem * 0.5, 40.0), alpha_init=0.01)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final KKT]: {real_score:.6f}")

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
