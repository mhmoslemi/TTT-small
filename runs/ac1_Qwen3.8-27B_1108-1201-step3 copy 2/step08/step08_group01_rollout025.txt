```python
import numpy as np
from scipy.fft import rfft, irfft
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_max(conv, tau):
    """Smoothed max using softmax."""
    shifted = conv / tau
    shifted -= shifted.max()  # for numerical stability
    exp_shifted = np.exp(shifted)
    return tau * (np.log(exp_shifted.sum()) + shifted.max())

def softmax_max_grad(conv, tau):
    """Gradient of softmax max with respect to conv."""
    shifted = conv / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w = w / w.sum()
    return w  # d(softmax_max)/d(conv_k) = w_k

    """Compute smoothed objective and gradient for sum(a)=1, a>=0."""
```

```python
def solve_lp_best_response(a, method='highs'):
    """Solve min_b max_k (a*b)_k s.t. sum(b)=1, b>=0."""
    n = len(a)
    # Build A_ub: (2n-1) x (n+1)
    # Row k: [a_{min(k,n-1)}, a_{min(k-1,n-1)}, ..., -1]
    # Actually: row k of A_ub has (A_ub)_{k,i} = a_{k-i} if 0<=k-i<n else 0, for i=0..n-1
    # And (A_ub)_{k,n} = -1
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = a[j]
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]  # b_i >= 0, M free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method=method)
    
    if res.status != 0:
        return None, None
    
    b_opt = res.x[:n]
    M_opt = res.x[n]
    return b_opt, M_opt
```

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def solve_lp_br(a):
    """Find best response b to a: min_b max_k (a*b)_k."""
    n = len(a)
    T = build_conv_matrix(a)
    A_ub = np.hstack([T, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.hstack([np.ones(n), [[0]]])
    b_eq = np.array([1.0])
    c = np.zeros(n+1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1]
