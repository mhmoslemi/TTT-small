import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        return None, None
    return b_val / s, float(M_var.value)


def mirror_descent(a0, time_limit, T=50.0, lr_init=0.15, verbose=True, label=""):
    """
    Mirror descent (multiplicative weights) with log-sum-exp soft subgradient.
    Very fast per iteration: O(n log n).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive: reduce lr if stuck
        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-6:
                # Try a "kick": perturb and restart with higher lr
                a = a + 0.01 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.1

        # Compute soft subgradient using log-sum-exp
        shifted = T * (conv - M)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        grad = 2.0 * cross_correlate_fft(w, a)

        # Mirror descent: a_new = a * exp(-lr * grad), normalize
        log_update = -lr * grad
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a = np.ones(n) / n
        else:
            a = a_new / s_new

        # Slowly anneal T upward (sharper peak focusing)
        if it % 1000 == 0:
            T = min(T * 1.05, 300.0)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {label} it={it} ({elapsed:.0f}s): score={score:.6f} best={best_score:.6f} T={T:.0f} lr={lr:.4f}")

    return best_a, best_score, it


def lp_fictitious_play(n, time_limit, initial_a=None, solver=None, verbose=True):
    """LP fictitious play with Polyak averaging and best-iterate tracking."""
    t0 = time.monotonic()
    if initial_a is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial_a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    a_avg = a.copy().astype(np.float64)
    b = a.copy()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Best response to a
        b_new, _ = solve_lp_br(a, solver)
        if b_new is not None:
            b = b_new

        # Best response to b
        a_new, _ = solve_lp_br(b, solver)
        if a_new is not None:
            a = a_new

        # Polyak averaging
        a_avg = (iters * a_avg + a) / (iters + 1.0)

        # Check candidates
        for cand in [a, b, a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

        # Symmetrized version
        a_sym = 0.5 * (a + a[::-1])
        a_sym = np.maximum(a_sym, 0)
        s_sym = a_sym.sum()
        if s_sym > 1e-10:
            a_sym = a_sym / s_sym
            score_sym = 2.0 * n * fast_convolve(a_sym).max()
            if score_sym < best_score:
                best_score = score_sym
                best_a = a_sym.copy()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):
    """Fast coordinate descent with O(n) sensitivity per update."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for trial in range(n_trials):
                if trial < 4:
                    delta = -0.5 ** trial * 0.3 * a[i]
                else:
                    delta = 0.5 ** (trial - 4) * 0.3 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-18:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)
                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    old_ai = a[i] - best_delta
                    changes[idx_2i] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            # Very fine steps
            for factor in [0.995, 0.99, 0.9975, 0.9925, 1.0025, 1.005, 1.0075, 1.01]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-19:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-18:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    old_ai = a[i] - best_delta
                    changes[idx_2i] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5 and sweeps > 20:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetrize(a):
    """Average with reverse to enforce symmetry."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s


def check_and_update(a, best_score, label=""):
    """Evaluate candidate and return score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        return a_norm.copy(), score
    return None, best_score


def compute_score_internal(a):
    """Compute the objective without calling evaluate_sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    return 2.0 * n * fast_convolve(a_norm).max()
