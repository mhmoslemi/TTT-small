def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lse_gradient(a, T):

def nesterov_subgradient(a, T, time_limit, eta0=0.05, verbose=True, label="NSGD"):

    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov lookahead

    best_a = a.copy()
    best_score = compute_score(a)

    eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Nesterov momentum
        t = iters + 1
        mu = (t - 1) / (t + 2)
        y_new = a + mu * (a - a_prev)
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = a
            y_sum = 1.0
        y_new = y_new / y_sum
        
        # Compute gradient at y_new
        grad, w = lse_gradient(y_new, T)
        
        # Update
        a_new = y_new - eta / np.sqrt(iters) * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new

def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):

    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def softmax(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        grad_a, _ = lse_gradient(softmax(z), T)
        a = softmax(z)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        a_out = softmax(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def sidon_init(n, seed=42):
    """Initialize using a Sidon-like set (quadratic residues mod p)."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    # Use quadratic residues as a starting point
    # Find a prime near n
    p = n
    while not all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
        p += 1
    # Quadratic residues mod p
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    residues.discard(0)
    
    # Map to our length
    scale = n / len(residues)
    for r in residues:
        idx = int(r * n / p)
        if idx < n:
            a[idx] = 1.0
    
    # Smooth and normalize
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 0:
        a = a / ss
    else:
        a = np.ones(n) / n
    return a


def exponential_decay_init(n, rate=0.5, seed=42):
    """Initialize with exponential decay plus noise."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    a = np.exp(-rate * t / n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def sinusoidal_init(n, freq=1.5, seed=42):
    """Initialize with sinusoidal modulation."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t / n) + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def away_step(P, Q, R, v):
    """Away-step projected gradient (Juditsky-Nesterov)."""
    Pi = P - Q
    Piv = np.dot(Pi, v)
    if Piv < 0:
        t = np.dot(Pi * Pi, v)
        if t == 0:
            return P
        Rv = np.dot(R, v)
        if Rv == 0:
            return P
        tau = Piv / (t * Rv)
        return P + tau * Pi
    else:
        return P


def juditsky_nesterov(a0, T, time_limit, eta0=0.1, verbose=True, label="JN"):
    """Juditsky-Nesterov accelerated away-step method."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    Q = a.copy()  # Previous "reference" point
    P = a.copy()  # Reference point for away step
    R_list = [a.copy()]  # List of vertices
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Nesterov momentum point
        t = iters + 1
        mu = (t - 1) / (t + 2)
        y = Q + mu * (a - Q)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a
            ys = 1.0
        else:
            y = y / ys
        
        # Gradient at y
        grad, w = lse_gradient(y, T)
        
        # Away-step projection
        v = eta / np.sqrt(iters) * grad
        P_new = away_step(P, Q, R_list[0], v)
        a_new = P_new - v
        
        # Project to simplex (basic projection)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Update vertices (keep at most 20)
        Q = P_new
        P = away_step(P, Q, R_list[0], v)
        R_list.append(a_new.copy())
        if len(R_list) > 20:
            R_list = R_list[-10:]
        
        a = a_new
    
    return best_a, best_score

Multi-method optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key methods: LSE-smoothed FFT gradient, mirror descent, LP fictitious play,

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lse_gradient(a, T):
    """Gradient of (2n/T)*log(sum(exp(T*c))) where c=conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    w = np.exp(T * (conv - max_c))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = w / w_sum
    # grad[j] = (2n/T) * 2 * sum_m w[j+m] * a[m]
    size = 3*n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv_full = irfft(W * A_rev, fft_size)[:size]
    grad = 2.0 * conv_full[n-1 : 2*n-1]
    return (2.0 * n / T) * grad, w


def softmax_func(z):

def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_func(z)
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        lse = mx + np.log(np.exp(shifted - mx).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_func(z)
        grad_a, _ = lse_gradient(a, T)
        return softmax_gradient(z, grad_a)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 20, 'disp': False})
        a_out = softmax_func(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
