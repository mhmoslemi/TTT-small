Hybrid LP + gradient optimization for auto-convolution peak minimization.
Uses full-constraint LP, adaptive line search, perturbative restarts,
and multi-length exploration to find near-optimal step functions.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def score_normalized(f):
    """Compute score for a normalized sequence (sum=1)."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()

def lp_solve_full(f, M, n, time_limit=60.0):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Uses all 2n-1 constraints.
    """
    K = 2 * n - 1
    
    # Build constraint matrix (banded structure)
    # A[k, j] = f[j] if j is in range for lag k, else 0
    # For lag k: j ranges from max(0, k-n+1) to min(n-1, k)
    A = np.zeros((K, n))
    for k in range(K):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    t0 = time.monotonic()
    g_val = None
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - t0 > time_limit:
                continue
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g_val = np.maximum(g_var.value, 0.0)
                break
        except Exception:
            continue
    
    return g_val

def lp_solve_topk(f, M, n, top_k=100, time_limit=60.0):

def line_search(f, direction, f_sum=1.0):
    """
    Find optimal step size t in [0,1] to minimize max(convolve((1-t)f + t*direction, (1-t)f + t*direction)).
    Uses golden section search.
    """
    n = len(f)
    
    def obj_at(t):
        ft = (1.0 - t) * f + t * direction
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()
    
    # Coarse search
    t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]
    best_t = t_candidates[0]
    best_val = obj_at(best_t)
    
    for t in t_candidates:
        val = obj_at(t)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Golden section refinement around best
    idx = t_candidates.index(best_t)
    lo = t_candidates[max(0, idx - 1)]
    hi = t_candidates[min(len(t_candidates) - 1, idx + 1)]
    
    if hi - lo < 1e-6:
        return best_t
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(15):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-6:
            break
    
    return (lo + hi) / 2

def lp_iteration(f, n, top_k=100, time_per_lp=30.0):
    """
    One LP iteration: solve LP, line search, return new f and improvement.
    """
    b = fast_convolve(f)
    M = b.max()
    
    if top_k >= 2 * n - 1:
        g = lp_solve_full(f, M, n, time_limit=time_per_lp)
    else:
        g = lp_solve_topk(f, M, n, top_k=top_k, time_limit=time_per_lp)
    
    if g is None:
        return f, M, 0.0
    
    S_g = g.sum()
    if S_g <= 1.0 + 1e-6:
        return f, M, 0.0
    
    h = g / S_g  # normalize to sum 1
    
    # Line search
    t_best = line_search(f, h)
    
    f_new = (1.0 - t_best) * f + t_best * h
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    
    b_new = fast_convolve(f_new)
    M_new = b_new.max()
    
    return f_new, M_new, M - M_new

def run_lp_optimization(f_init, time_limit, label="lp", top_k_start=80, top_k_end=200):
    """

    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Gradually increase top_k
        if it % 30 == 0:
            top_k = min(top_k + 10, top_k_end)
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        time_per_lp = min(30.0, remaining * 0.2)
        if time_per_lp < 3:
            break
        
        f_new, M_new, improvement = lp_iteration(f, n, top_k=top_k, time_per_lp=time_per_lp)
        
        if improvement > 1e-14:
            f = f_new
            if M_new < best_obj:
                best_obj = M_new
                best_f = f_new.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        
        if it % 20 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")
        
        if stuck > 40:
            break
    
    return best_f, best_obj

def perturb(f, rng, method="gaussian", scale=0.01):
    """Apply a perturbation to f and project back to simplex."""
    n = len(f)
    if method == "gaussian":
        f_pert = f + rng.standard_normal(n) * scale
    elif method == "swap":
        f_pert = f.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        delta = f_pert[i] * scale * 5
        f_pert[i] -= delta
        f_pert[j] += delta
    elif method == "bump":
        f_pert = f.copy()
        center = rng.integers(0, n)
        width = max(1, n // 50)
        lo = max(0, center - width)
        hi = min(n, center + width)
        f_pert[lo:hi] += scale * 0.5
    elif method == "modulation":
        freq = rng.uniform(0, 3)
        phase = rng.uniform(0, 2 * np.pi)
        f_pert = f * (1 + scale * np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    else:
        f_pert = f + rng.standard_normal(n) * scale
    
    f_pert = np.maximum(f_pert, 0.0)
    s = f_pert.sum()
    if s < 1e-15:
        return f.copy()
    return f_pert / s

def subgradient_descent(f_init, time_limit, n_top=30):
    """

def flat_top_window(n):
    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def resample_sequence(f, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n, indices):
    """Build constraint matrix rows for given lag indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_solve(f, n, constraint_indices, M, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in constraint_indices.
    Returns g (unnormalized) or None on failure.
    """
    A = build_constraint_matrix(f, n, constraint_indices)
    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                      [g_var >= 0, A @ g_var <= M])
    t0 = time.monotonic()
    for solver in [cp.PDLP, cp.SCS]:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g_val = np.maximum(g_var.value, 0.0)
                if g_val.sum() > 1e-15:
                    return g_val
        except Exception:
            continue
    return None


def line_search_peak(f, direction):
    """
    Find optimal t in [0,1] minimizing max(convolve((1-t)f+t*dir, itself)).
    Uses adaptive coarse+fine search.
    """
    def obj_at(t):
        ft = (1.0 - t) * f + t * direction
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    # Coarse scan
    t_coarse = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8]
    best_t = 0.05
    best_val = obj_at(0.05)
    for t in t_coarse:
        val = obj_at(t)
        if val < best_val:
            best_val = val
            best_t = t

    # Fine search via golden section around best
    idx = t_coarse.index(best_t)
    lo = t_coarse[max(0, idx - 1)]
    hi = t_coarse[min(len(t_coarse) - 1, idx + 1)]
    if hi - lo < 1e-7:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(14):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-7:
            break
    return (lo + hi) / 2


def multi_direction_lp(f, n, M, time_per_lp=20.0):

    # Top-k by peak value

        # Random subset

        constraint_sets.append(np.sort(idx_rand))

def subgradient_descent_bb(f_init, time_limit, n_top=40):
    """
    Subgradient descent with Barzilai-Borwein step size and momentum.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.005
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(7777)

    prev_s = None
    prev_y = None
    prev_f = f.copy()

    while time.monotonic() - t0 < time_limit:
        batch = 80
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            b = fast_convolve(f)
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
            else:
                top_idx = np.arange(len(b))

            # Compute subgradient (weighted combination of active peak gradients)
            g = np.zeros(n)
            weights = b[top_idx]
            w_sum = weights.sum()
            if w_sum < 1e-15:
                break
            weights /= w_sum

            for i, k in enumerate(top_idx):
                w = weights[i]
                if w < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w * f[k - idx]

            # Small regularization to prevent degradation
            g += 0.0005 * (f - 1.0 / n)

            # Barzilai-Borwein step size
            s_step = f - prev_f
            y_step = g - v  # approximate y
            s_norm = np.dot(s_step, s_step)
            y_norm = np.dot(y_step, y_step)
            if s_norm > 1e-15 and y_norm > 1e-15:
                lr_bb1 = s_norm / max(np.dot(s_step, y_step), 1e-15)
                lr_bb2 = max(np.dot(s_step, y_step), 1e-15) / y_norm
                # Alternate between the two
                if it % 2 == 0:
                    lr_new = np.clip(lr_bb1, 0.0001, 0.05)
                else:
                    lr_new = np.clip(lr_bb2, 0.0001, 0.05)
                lr = 0.7 * lr + 0.3 * lr_new

            prev_f = f.copy()

            # Momentum update
            momentum = 0.85
            v = momentum * v - lr * g
            f_new = project_simplex(f + v)
            f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 3 == 0:
                lr *= 0.92
            if no_improve > 15:
                # Restart with perturbation
                scale = 0.002 + 0.003 * min(no_improve / 10.0, 1.0)
                f = best_f + rng.standard_normal(n) * scale
                f = np.maximum(f, 0)
                s = f.sum()
                if s > 1e-15:
                    f /= s
                v = np.zeros(n)
                no_improve = 0

        if it % 300 == 0:
            print(f"    sg_bb it={it} obj={best_obj:.10f} "
                  f"score={2*n*best_obj:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def flat_top_window(n):
    """Normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def resample_sequence(f, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s


def perturb_sequence(f, rng, method="mixed", scale=0.008):
    """Apply diverse perturbation to escape local optima."""
    n = len(f)
    if method == "gaussian":
        f_p = f + rng.standard_normal(n) * scale
    elif method == "sparse":
        f_p = f.copy()
        n_sparse = max(1, n // 20)
        idx = rng.choice(n, size=n_sparse, replace=False)
        f_p[idx] += rng.standard_normal(n_sparse) * scale * 3
    elif method == "block":
        f_p = f.copy()
        block_size = max(2, n // 30)
        start = rng.integers(0, n - block_size)
        f_p[start:start + block_size] += rng.standard_normal(block_size) * scale
    elif method == "modulation":
        freq = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + scale * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        f_p = f * mod
    elif method == "swap":
        f_p = f.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        delta = f_p[i] * rng.uniform(0.1, 0.5)
        f_p[i] -= delta
        f_p[j] += delta
    else:  # mixed
        methods = ["gaussian", "sparse", "block", "modulation", "swap"]
        m = methods[rng.integers(0, len(methods))]
        return perturb_sequence(f, rng, m, scale)

    f_p = np.maximum(f_p, 0.0)
    s = f_p.sum()
    if s < 1e-15:
        return f.copy()
    return f_p / s

    Main optimization: Hybrid LP + subgradient with multi-direction exploration,
    adaptive constraint sets, perturbative restarts, and multi-length search.
    Targets score < 1.5027 (lower is better).

    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    print(f"Starting best: n={len(best_seq)}, score={best_score:.8f}, target=1.5027")

    # === Phase 1: LP on parent with full constraints (main phase) ===
    print(f"\n{'='*50}")
    print(f"Phase 1: LP full-constraints on parent (n={len(parent)})")
    print(f"{'='*50}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.32, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimization(parent, time_limit=t1, label="lp_main",
                                        top_k_start=0, top_k_end=400)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_main]: {best_score:.8f}")
        else:
            print(f"  lp_main: {score1:.8f} (no improvement over {best_score:.8f})")

    # === Phase 2: Aggressive perturbative restarts ===
    print(f"\n{'='*50}")
    print(f"Phase 2: Perturbative restarts (best={best_score:.8f})")
    print(f"{'='*50}")
    rem = deadline - time.monotonic()
    t2_total = min(rem * 0.22, 320.0)
    if t2_total > 40:
        n_pert = 6
        per_pert = t2_total / n_pert
        methods = ["gaussian", "sparse", "modulation", "block", "swap", "mixed"]
        scales = [0.005, 0.01, 0.007, 0.012, 0.02, 0.008]

        for i in range(n_pert):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_pert = min(per_pert, remaining - 5)
            if t_pert < 15:
                break

            f_pert = perturb_sequence(best_seq, rng, methods[i], scales[i])
            t_opt = t_pert * 0.8
            f_opt, obj_opt = run_lp_optimization(f_pert, time_limit=t_opt,
                                                  label=f"pert{i}", top_k_start=0,
                                                  top_k_end=200)
            score_pert = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score_pert) and score_pert < best_score:
                best_score = score_pert
                best_seq = f_opt.copy()
                print(f"  NEW BEST [pert_{i}/{methods[i]}]: {best_score:.8f}")
            else:
                print(f"  pert_{i} ({methods[i]}): {score_pert:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*50}")
    print(f"Phase 3: Multi-length exploration (best={best_score:.8f})")
    print(f"{'='*50}")
    n_base = len(best_seq)
    all_candidates = [n_base - 128, n_base - 64, n_base + 64, n_base + 128,
                      n_base + 256, 1024, 1280, 1536, 1920, 2048]
    candidates = sorted(set(c for c in all_candidates if 512 <= c <= 2500 and c != n_base))
    candidates.sort(key=lambda x: abs(x - n_base))

    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.22, 350.0)
    n_try = min(4, len(candidates))

    for target_n in candidates[:n_try]:
        remaining = deadline - time.monotonic()
        if remaining < per_len * 0.4:
            break
        t_len = min(per_len, remaining - 5)
        if t_len < 25:
            break

        # Two initializations: interpolated and window

                best_score_len = score_reg
                best_for_len = f_reg.copy()

        if best_for_len is not None and best_score_len < best_score:
            best_score = best_score_len
            best_seq = best_for_len.copy()

    # === Phase 4: Subgradient BB fine-tuning ===
    print(f"\n{'='*50}")
    print(f"Phase 4: Subgradient BB (best={best_score:.8f})")

        print(f"  Budget: {t4:.0f}s, n={n_b}")

                                           n_top=max(30, n_b // 8))
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg_bb]: {best_score:.8f}")
        else:
            print(f"  sg_bb: {score4:.8f}")

    # === Phase 5: Final LP burst on best ===
    print(f"\n{'='*50}")

        if np.isfinite(score5) and score5 < best_score:

    # === Phase 6: Final polish ===
