def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def build_C(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C

def solve_lp(f, M):
    n = len(f)
    C = build_C(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)

def geometric_mean_update(f, g, t):
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

def lp_fictitious_play(f_init, time_limit, verbose=True, use_momentum=True):
    """LP fictitious play with geometric mean, momentum, and line search."""
    f = normalize(f_init.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    if use_momentum:
        log_f = np.log(np.maximum(f, 1e-30))
        momentum = np.zeros(n)
        beta = 0.7
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        g = solve_lp(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        if use_momentum:
            # Momentum in log space
            log_g = np.log(np.maximum(g, 1e-30))
            direction = log_g - log_f
            momentum = beta * momentum + direction
            log_new = log_f + momentum / (k + 1)
            log_new -= log_new.max()
            a_new = np.exp(log_new)
            a_new = normalize(a_new)
        else:
            t_step = min(1.0, 3.0 / (k + 3))
            a_new = geometric_mean_update(f, g, t_step)
        
        # Line search: try several step sizes
        if not use_momentum:
            best_t = t_step
            best_sc = compute_score(a_new)
            for t_try in [t_step * 0.5, t_step * 1.5, t_step * 2.0]:
                a_try = geometric_mean_update(f, g, t_try)
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_t = t_try
                    a_new = a_try
        
        score = compute_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_f = a_new.copy()
        
        f = a_new
        if use_momentum:
            log_f = np.log(np.maximum(f, 1e-30))
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k}, M={M:.6e}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_f, best_score

def variance_minimize(f_init, time_limit, verbose=True, lr=0.01, beta1=0.9, beta2=0.999):
    """Minimize ||conv(a)||^2 using Adam in log-space."""
    a = normalize(f_init.astype(np.float64).copy())
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))  # log-space parameter
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    eps = 1e-8
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Convert z to a
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        
        # Compute gradient of ||conv||^2 w.r.t. a
        conv = fast_convolve(a)
        # grad_a[i] = 4 * sum_j a[j] * conv[i+j]
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 3).bit_length()
        A = rfft(a_rev, fft_size)
        C = rfft(conv, fft_size)
        conv_result = irfft(A * C, fft_size)
        grad_a = 4.0 * conv_result[n-1 : 2*n-1]
        
        # Project gradient to log-space: grad_z = a * (grad_a - dot(grad_a, a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Adam update
        t = k + 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Check score
        if k % 10 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and k % 500 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  VarAdam iter={k}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score

def spectral_flatten(f_init, time_limit, verbose=True, n_iter=1000, mix=0.3):
    """Iterative spectral flattening with non-negativity projection."""
    a = normalize(f_init.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for k in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute DFT
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        
        # Target: geometric mean of magnitudes (or just a constant)
        # Use the current mean as target
        mean_mag = np.mean(mag[mag > 1e-10])
        
        # Blend: new_mag = mag^(1-mix) * mean_mag^mix
        new_mag = np.power(mag + 1e-30, 1 - mix) * (mean_mag ** mix)
        
        # New spectrum
        A_new = new_mag * phase
        
        # Inverse DFT
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        # Blend with current (to maintain stability)
        a = (1 - mix) * a + mix * a_new
        a = normalize(a)
        
        if k % 10 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and k % 200 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  Spec iter={k}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=True, n_pairs=200):
    """Coordinate descent with golden-section search."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        
        # Generate pairs
        pairs = set()
        # Peak-focused
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(n_pairs * 2 // 3):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        # Random
        for _ in range(n_pairs // 3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            f1 = _eval_transfer(a, i, j, x1)
            f2 = _eval_transfer(a, i, j, x2)
            
            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = _eval_transfer(a, i, j, x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = _eval_transfer(a, i, j, x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = normalize(a)
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                a = best_a + rng.standard_normal(n) * 0.001
                a = normalize(a)
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose:
            print(f"  CD score={best_score:.8f}")
    
    return best_a, best_score

def _eval_transfer(a, i, j, t):
    """Evaluate score after transferring t from i to j."""
    a_trial = a.copy()
    a_trial[i] -= t
    a_trial[j] += t
    if a_trial[i] < 0 or a_trial[j] < 0:
        return 1e10
    s = a_trial.sum()
    if s < 1e-15:
        return 1e10
    a_trial = a_trial / s
    return compute_score(a_trial)

def subgradient_method(a0, time_limit, verbose=True):
    """Projected subgradient method on the max."""
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for k in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        score = compute_score(a)
        
        # Subgradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]
        
        # Step size: Polyak-like
        eta = 1.0 / (k + 1) ** 0.7
        
        # Update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        if compute_score(a_new) < best_score:
            best_score = compute_score(a_new)
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    n = len(f)
    C = build_C(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0)
    except Exception:
        return None


def greedy_lp(f, M):
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def geometric_mean_update(f, g, t):
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1.0 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, verbose=True, use_momentum=True, beta=0.7):
    f = normalize(f_init.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    log_f = np.log(np.maximum(f, 1e-30))
    momentum = np.zeros(n)

    for k in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        g = solve_lp(f, M)
        if g is None or g.sum() < 1e-15:
            g = greedy_lp(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        if use_momentum:
            log_g = np.log(np.maximum(g, 1e-30))
            direction = log_g - log_f
            momentum = beta * momentum + (1.0 - beta) * direction
            decay = 1.0 / (k + 1) ** 0.5
            log_new = log_f + decay * momentum
            log_new -= log_new.max()
            a_new = np.exp(log_new)
            a_new = normalize(a_new)
            log_f = np.log(np.maximum(a_new, 1e-30))
        else:
            t_step = min(1.0, 2.0 / (k + 2))
            a_new = geometric_mean_update(f, g, t_step)
            # Quick line search
            best_sc = compute_score(a_new)
            best_a = a_new
            for t_try in [t_step * 0.5, t_step * 1.5, t_step * 2.0, t_step * 0.3]:
                if t_try <= 0 or t_try > 1:
                    continue
                a_try = geometric_mean_update(f, g, t_try)
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try
            a_new = best_a

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_f = a_new.copy()

        f = a_new

        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP[{use_momentum}] iter={k}, M={M:.6e}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_f, best_score


def variance_minimize(f_init, time_limit, verbose=True, lr=0.05, beta1=0.9, beta2=0.999):
    a = normalize(f_init.astype(np.float64).copy())
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    fft_size = 1 << (3 * n - 3).bit_length()

    for k in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()

        conv = fast_convolve(a)
        a_rev = a[::-1]
        A = rfft(a_rev, fft_size)
        C = rfft(conv, fft_size)
        conv_result = irfft(A * C, fft_size)
        grad_a = 4.0 * conv_result[n - 1:2 * n - 1]

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        t = k + 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if k % 20 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose and k % 500 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  VarAdam iter={k}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def spectral_flatten(f_init, time_limit, verbose=True, n_iter=2000, mix=0.15):
    a = normalize(f_init.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for k in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)

        valid = mag > 1e-10
        if not valid.any():
            break
        mean_mag = np.mean(mag[valid])

        new_mag = np.power(mag + 1e-30, 1.0 - mix) * (mean_mag ** mix)
        A_new = new_mag * phase

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        a = (1.0 - mix) * a + mix * a_new
        a = normalize(a)

        if k % 10 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose and k % 500 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  Spec iter={k}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def eval_transfer(a, i, j, t):
    a_t = a.copy()
    a_t[i] -= t
    a_t[j] += t
    if a_t[i] < 0 or a_t[j] < 0:
        return 1e10
    s = a_t.sum()
    if s < 1e-15:
        return 1e10
    a_t = a_t / s
    return compute_score(a_t)


def coordinate_descent(a0, time_limit, verbose=True, n_pairs=300):
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))

        pairs = set()
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(n_pairs * 2 // 3):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        for _ in range(n_pairs // 3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        pair_list = list(pairs)
        rng.shuffle(pair_list)

        for (i, j) in pair_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            f1 = eval_transfer(a, i, j, x1)
            f2 = eval_transfer(a, i, j, x2)

            for _ in range(18):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(a, i, j, x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(a, i, j, x2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = normalize(a)
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                noise_scale = 0.002 / (1.0 + no_improve * 0.5)
                a = best_a + rng.standard_normal(n) * noise_scale
                a = normalize(a)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD sweep={sweep}, score={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def subgradient_method(a0, time_limit, verbose=True):
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    avg_a = a.copy()

    for k in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        score = compute_score(a)

        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        eta = 1.0 / (k + 1) ** 0.6
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        avg_a = ((k * avg_a) + a_new) / (k + 1)

        sc_new = compute_score(a_new)
        sc_avg = compute_score(avg_a)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        a = a_new

        if verbose and k % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Subgrad iter={k}, score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def mix_sequences(a1, a2, alpha):
    return normalize(alpha * a1 + (1.0 - alpha) * a2)


def try_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp
