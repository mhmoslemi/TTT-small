import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_perturb(f, rng, strength=0.05):
    """Add a spectral perturbation: modify the spectrum, project back to simplex."""
    n = len(f)
    fft_size = 1 << (n - 1).bit_length()
    F = rfft(f, fft_size)
    # Add random phase perturbation
    phase = rng.uniform(-strength, strength, size=len(F))
    F_pert = F * (1.0 + 1j * phase)
    f_pert = irfft(F_pert, fft_size)[:n]
    return project_simplex(f_pert)


def lp_solve(f, active_indices, M, time_limit=10.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_indices.
    Returns (g, dual_values) where dual_values are the optimal dual variables.
    """
    n = len(f)
    K = len(active_indices)
    
    if K == 0:
        g = np.ones(n) / n
        return g, np.array([])
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = active_indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    y_var = cp.Variable(K)  # dual variables
    
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M * np.ones(K)]
    )
    
    g = None
    dual = None
    t0 = time.monotonic()
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver, max_iters=200000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                # Get dual values
                if y_var.value is not None:
                    dual = y_var.value
                break
        except Exception:
            continue
    
    if g is None or np.any(np.isnan(g)):
        return None, None
    
    return g, dual


def line_search(f, direction, f_min=None, max_iter=20):
    """Golden section search for best step size t in f - t*(f-direction)."""
    n = len(f)
    
    def obj(t):
        f_new = f - t * (f - direction)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            return 1e10
        f_new = f_new / s
        return fast_convolve(f_new).max()
    
    # Coarse search
    best_t = 0.1
    best_val = obj(0.1)
    for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
        v = obj(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Fine search
    if best_t < 0.05:
        lo, hi = 0.0, best_t * 2
    elif best_t > 0.9:
        lo, hi = best_t * 0.5, 1.0
    else:
        lo, hi = max(0, best_t - 0.1), min(1, best_t + 0.1)
    
    gr = (math.sqrt(5) - 1) / 2
    a = lo
    b = hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(15):
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
        if b - a < 1e-5:
            break
    
    t_star = (a + b) / 2
    return t_star


def compute_pressure(f, n_active=50):
    """
    Compute the pressure map: which lags are most "active" and their weights.
    Returns (active_indices, weights) where weights sum to 1.
    """
    c = fast_convolve(f)
    M = c.max()
    # Use softmax-like weighting near the peak
    threshold = M * 0.95  # lags within 5% of peak
    active = np.where(c >= threshold)[0]
    
    if len(active) > n_active:
        # Keep only the top n_active
        c_active = c[active]
        top = np.argsort(c_active)[-n_active:]
        active = active[top]
    
    # Weights proportional to how close to peak
    c_active = c[active]
    weights = c_active / c_active.sum()
    
    return active, weights

def focused_descent(f_init, time_limit, n_peaks=5):
    """Direct descent on top peaks with adaptive step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(f)
        if n_peaks < len(c):
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(c))
        
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]
        
        if grad.max() < 1e-15:
            break
        
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
            f_trial = project_simplex(f - alpha * grad)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha
        
        if best_alpha is None:
            break
        
        f = project_simplex(f - best_alpha * grad)
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj

We want min_t max_k q_k(t) for t in [0, 1].

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP main (40% budget) ===
    print(f"\n=== Phase 1: LP main (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 680.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimization(parent, time_limit=t1,
                                         max_constraints=200, label="lp_main")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_main]: {score1:.8f}")
        else:
            print(f"  lp_main: {score1:.8f} (no improvement)")

    # === Phase 2: Spectral-LP hybrid (20% budget) ===
    print(f"\n=== Phase 2: Spectral-LP hybrid (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = spectral_lp_hybrid(best_seq, time_limit=t2, label="spec_lp")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [spec_lp]: {score2:.8f}")
        else:
            print(f"  spec_lp: {score2:.8f} (no improvement)")

    # === Phase 3: LP deep (15% budget) ===
    print(f"\n=== Phase 3: LP deep (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = run_lp_optimization(best_seq, time_limit=t3,
                                         max_constraints=300, label="lp_deep")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp_deep]: {score3:.8f}")
        else:
            print(f"  lp_deep: {score3:.8f} (no improvement)")

    # === Phase 4: Focused descent (8% budget) ===
    print(f"\n=== Phase 4: Focused descent (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 80.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        f4, obj4 = focused_descent(best_seq, time_limit=t4, n_peaks=8)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [focused]: {score4:.8f}")
        else:
            print(f"  focused: {score4:.8f} (no improvement)")

    # === Phase 5: Multi-length exploration (10% budget) ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    candidates = [n_base - 64, n_base + 64, n_base - 32, n_base + 32, n_base + 128, n_base - 128]
    candidates = [c for c in candidates if 512 <= c <= 4000 and c != n_base]
    # Deduplicate
    seen = set()
    deduped = []
    for c in candidates:
        if c not in seen:
            seen.add(c)
            deduped.append(c)
    candidates = deduped
    # Sort by distance to n_base using a helper
    def dist_from_base(x):
        return abs(x - n_base)
    candidates.sort(key=dist_from_base)

    for target_n in candidates[:3]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t5 = min(rem * 0.15, 100.0)
        if t5 < 20:
            break

        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s

        print(f"  n={target_n}, time={t5:.0f}s")
        f5, obj5 = run_lp_optimization(f_interp, time_limit=t5,
                                         max_constraints=200, label=f"n{target_n}")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.8f}")
        else:
            print(f"  n{target_n}: {score5:.8f} (no improvement)")

    # === Phase 6: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(rem * 0.5, 180.0)
        print(f"\n=== Phase 6: Final LP (n={len(best_seq)}, time={t6:.0f}s) ===")
        f6, obj6 = run_lp_optimization(best_seq, time_limit=t6,
                                         max_constraints=250, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.8f}")
        else:
            print(f"  lp_final: {score6:.8f} (no improvement)")

    # === Phase 7: Final focused descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem * 0.4, 50.0)
        if t7 > 5:
            print(f"\n=== Phase 7: Final focused (time={t7:.0f}s) ===")
            f7, obj7 = focused_descent(best_seq, time_limit=t7, n_peaks=10)
            score7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(score7) and score7 < best_score:
                best_score = score7
                best_seq = f7.copy()
                print(f"  NEW BEST [final_focused]: {score7:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
