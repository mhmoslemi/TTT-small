import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_A(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A

def solve_lp(f, n, M, active_idx, eps=1e-8):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        return None
    if g.value is None:
        if n <= 2000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None
        else:
            return None
    if g.value is None or np.any(np.isnan(g.value)):
        return None
    return np.maximum(g.value, 0.0)

def iaa_optimize(f_init, n_iters=200, beta=0.1, time_limit=None, t0=None):
    """IAA with soft non-negativity handling."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for it in range(n_iters):
        if t0 is not None and time.monotonic() - t0 > time_limit:
            break
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        
        # Water-filling: reduce peaks while maintaining energy
        peak = R.max()
        floor = beta * peak
        R_target = np.where(R > floor, floor + (R - floor) * 0.5, R)
        
        # Preserve total energy
        if R_target.sum() > 1e-15:
            R_target *= R.sum() / R_target.sum()
        
        # Recover new amplitude
        R_t = rfft(R_target, fft_size)
        real_part = np.maximum(R_t.real, 0.0)
        new_amp = np.sqrt(real_part)
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]
        
        # Soft non-negativity: replace small negatives with 0
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_obj

2. IAA (Iterative Amplitude Adjustment) with water-filling spectral shaping

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix for LP: row i encodes (f*g)[k_i]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx, eps=1e-8):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        pass
    if g.value is None:
        if n <= 2500:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None
        else:
            return None
    if g.value is None or np.any(np.isnan(g.value)):
        return None
    return np.maximum(g.value, 0.0)


def iaa_waterfill(f_init, n_iters=300, beta=0.15, time_limit=60.0):
    """IAA with water-filling: flatten the auto-convolution spectrum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        # Water-filling: compress dynamics
        peak = R.max()
        if peak < 1e-15:
            break
        floor = beta * peak
        # Smooth compression: values above floor get pulled toward floor
        excess = np.maximum(R - floor, 0)
        R_target = R - 0.4 * excess

        # Preserve total energy
        if R_target.sum() > 1e-15:
            R_target *= R.sum() / R_target.sum()

        # Recover from modified spectrum
        R_t = rfft(R_target, fft_size)
        real_part = np.maximum(R_t.real, 0.0)
        new_amp = np.sqrt(real_part)
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_amp * phase
        f_new = irfft(F_new, fft_size)[:n]

        # Soft non-negativity
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
            break

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new

    return best_f, best_obj

    t0 = time.monotonic()

        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contribs = f[idx] * f[peak_k - idx]
        
        # Get top n_pairs contributors
        if len(contribs) > n_pairs:
            top_local = np.argpartition(contribs, -n_pairs)[-n_pairs:]
        else:
            top_local = np.arange(len(contribs))

        improved = False
        for li in top_local:
            m = idx[li]
            m2 = peak_k - idx[li]
            if m == m2:

            # Try reducing the larger of f[m], f[m2] and adding to a small element

                j_reduce = m
            else:
                j_reduce = m2
            
            # Find a small element to compensate

def subgradient_opt(f_init, time_limit, lr_init=0.004, temp_init=0.0008, n_top=60):

    no_improve = 0

        it += 1
        b = fast_convolve(f)

        # Softmax weights on peaks
        scaled = b / temp

        w = np.exp(scaled)
        w /= w.sum()

def generate_initializations(n, parent=None):

    a /= a.sum()
    inits.append(("sin1", a))

    a /= a.sum()
    inits.append(("bell", a))

    for score, name, a in init_candidates:
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST from {name}: {score:.6f}")

    # === Phase 2: LP on best candidates ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP optimization (best={best_score:.6f})")
    print(f"{'='*60}")

    t_p2_start = time.monotonic()
    p2_budget = min(500.0, (deadline - t_p2_start) * 0.28)

    # Run LP on top-2 initializations
    for idx, (score, name, a) in enumerate(init_candidates[:2]):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = p2_budget / max(1, 2 - idx)
        if t_alloc < 30:
            t_alloc = 30
        print(f"  LP on '{name}' (n={len(a)}, budget={t_alloc:.0f}s)")
        f_lp, obj_lp = lp_cutting_plane(a, time_limit=t_alloc,
                                         initial_k=50, max_k=350, label=f"lp_{name}")
        s_lp = float(evaluate_sequence(f_lp.tolist()))
        print(f"    Result: {s_lp:.6f}")
        if np.isfinite(s_lp) and s_lp < best_score:
            best_score = s_lp
            best_seq = f_lp.copy()
            print(f"    NEW BEST: {s_lp:.6f}")

    # === Phase 3: Continued LP on best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Deep LP (best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    t_p3_start = time.monotonic()
    p3_budget = min(350.0, (deadline - t_p3_start) * 0.2)

    if p3_budget > 30:
        n_b = len(best_seq)
        # Use more constraints for deeper optimization
        f_lp3, obj3 = lp_cutting_plane(best_seq, time_limit=p3_budget,
                                        initial_k=80, max_k=500, label="deep")
        s3 = float(evaluate_sequence(f_lp3.tolist()))
        print(f"  Deep LP: {s3:.6f}")
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_seq = f_lp3.copy()
            print(f"  NEW BEST: {s3:.6f}")

    # === Phase 4: Peak-directed coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak-directed CD (best={best_score:.6f})")
    print(f"{'='*60}")

    t_p4_start = time.monotonic()
    p4_budget = min(200.0, (deadline - t_p4_start) * 0.12)

    if p4_budget > 20:
        f_cd, obj_cd = peak_directed_cd(best_seq, time_limit=p4_budget, n_pairs=10)
        s_cd = float(evaluate_sequence(f_cd.tolist()))
        print(f"  CD: {s_cd:.6f}")
        if np.isfinite(s_cd) and s_cd < best_score:
            best_score = s_cd
            best_seq = f_cd.copy()
            print(f"  NEW BEST: {s_cd:.6f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Subgradient (best={best_score:.6f})")
    print(f"{'='*60}")

    t_p5_start = time.monotonic()
    p5_budget = min(200.0, (deadline - t_p5_start) * 0.1)

    if p5_budget > 20:
        n_b = len(best_seq)
        f_sg, obj_sg = subgradient_opt(best_seq, time_limit=p5_budget,
                                        lr_init=0.003, temp_init=0.0006,
                                        n_top=max(40, n_b // 12))
        s_sg = float(evaluate_sequence(f_sg.tolist()))
        print(f"  SG: {s_sg:.6f}")
        if np.isfinite(s_sg) and s_sg < best_score:
            best_score = s_sg
            best_seq = f_sg.copy()
            print(f"  NEW BEST: {s_sg:.6f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Multi-length (best={best_score:.6f})")
    print(f"{'='*60}")

    t_p6_start = time.monotonic()
    p6_budget = min(200.0, (deadline - t_p6_start) * 0.15)

    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 0.95, 1.05, 1.15, 1.3]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(p6_budget / 3.0, rem * 0.7)
        if t_alloc < 20:
            break

        old_x = np.linspace(0, 1, n_base)
        new_x = np.linspace(0, 1, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s

        f_ml, _ = lp_cutting_plane(f_interp, time_limit=t_alloc,
                                    initial_k=40, max_k=250, label=f"n{target_n}")
        s_ml = float(evaluate_sequence(f_ml.tolist()))
        print(f"  n={target_n}: {s_ml:.6f}")
        if np.isfinite(s_ml) and s_ml < best_score:
            best_score = s_ml
            best_seq = f_ml.copy()
            print(f"  NEW BEST: {s_ml:.6f}")

    # === Phase 7: Stochastic perturbation + short LP ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Stochastic restarts (best={best_score:.6f})")
    print(f"{'='*60}")

    t_p7_start = time.monotonic()
    p7_budget = min(150.0, (deadline - t_p7_start) * 0.15)

    rng = np.random.default_rng(2024)
    n_b = len(best_seq)

    for round_i in range(4):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = p7_budget / max(1, 4 - round_i)

        # Perturb best sequence
        f_pert = best_seq.copy()
        mask = rng.random(n_b) < 0.08
        f_pert[mask] *= rng.uniform(0.6, 1.4, mask.sum())
        f_pert = np.maximum(f_pert, 0)
        s = f_pert.sum()
        if s > 1e-10:
            f_pert /= s

        # Quick LP
        t_lp = t_alloc * 0.6
        f_new, _ = lp_cutting_plane(f_pert, time_limit=t_lp,
                                     initial_k=40, max_k=250, label=f"stoch{round_i}")
        s_new = float(evaluate_sequence(f_new.tolist()))
        print(f"  Round {round_i}: {s_new:.6f}")
        if np.isfinite(s_new) and s_new < best_score:
            best_score = s_new
            best_seq = f_new.copy()
            print(f"  NEW BEST: {s_new:.6f}")

    # === Phase 8: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n{'='*60}")
        print(f"Phase 8: Final polish (best={best_score:.6f}, rem={rem:.0f}s)")
        print(f"{'='*60}")
        t_final = min(rem * 0.8, 100.0)
        f_fin, _ = lp_cutting_plane(best_seq, time_limit=t_final,
                                     initial_k=60, max_k=450, label="final")
        s_fin = float(evaluate_sequence(f_fin.tolist()))
        if np.isfinite(s_fin) and s_fin < best_score:
            best_score = s_fin
            best_seq = f_fin.copy()
            print(f"  Final: NEW BEST: {s_fin:.6f}")

    # === Final validation ===
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        if parent is not None:
            best_seq = parent.copy()
            final_score = float(evaluate_sequence(best_seq.tolist()))
        else:
            best_seq = np.ones(1024) / 1024
            final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
