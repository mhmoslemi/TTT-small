    a1_safe = np.maximum(a1, 1e-30)
    a2_safe = np.maximum(a2, 1e-30)

    log_new -= log_new.max()
    result = np.exp(log_new)

    return result / s

    q = a[::-1]
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(p, fft_size)
    Q = rfft(q, fft_size)
    conv = irfft(P * Q, fft_size)
    return conv[n - 1: n]


def solve_qp(a_init, p, n_iters=80, eta_scale=0.5):
    """Projected gradient for QP: min a^T P a s.t. simplex."""
    n = len(a_init)
    a = a_init.copy()
    max_p = max(p.max(), 1e-15)
    L = 2.0 * np.sqrt(n * max_p) + 1e-10
    eta = eta_scale / L
    
    # Precompute FFT of p
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P_fft = rfft(p, fft_size)
    
    for it in range(n_iters):
        # grad = 2 * P @ a
        Q = rfft(a[::-1], fft_size)
        conv = irfft(P_fft * Q, fft_size)
        grad = 2.0 * conv[n - 1: n]
        
        a = project_simplex(a - eta * grad)
    
    return a


def geometric_blend(a1, a2, t):
    a1s = np.maximum(a1, 1e-30)
    a2s = np.maximum(a2, 1e-30)
    log_new = (1.0 - t) * np.log(a1s) + t * np.log(a2s)
    log_new -= log_new.max()
    res = np.exp(log_new)
    s = res.sum()
    if s < 1e-30:
        return a1.copy()
    return res / s


def alternating_optimize(a0, time_limit, T_start=50.0, T_end=1.0, 
                         n_inner=30, verbose=True):
    """Main optimization: alternating minimax with annealing."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    outer_iter = 0
    total_outers = 5000
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-10))
        
        # Anneal temperature
        T = T_start * (T_end / T_start) ** progress
        
        C = fast_convolve(a)
        C_shifted = C - C.max()
        w = np.exp(C_shifted / T)
        p = w / w.sum()
        
        # Solve QP (inexact)
        a_qp = solve_qp(a, p, n_iters=n_inner)
        
        # Blend

        a_new = geometric_blend(a, a_qp, t_step)
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        outer_iter += 1
        
        if verbose and outer_iter % 100 == 0:
            print(f"    Alt: iter={outer_iter}, T={T:.3f}, score={best_score:.8f}, "
                  f"elapsed={elapsed:.0f}s")

def coordinate_descent(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        conv = fast_convolve(a)
        
        # Top peaks
        n_peaks = min(3, len(conv))
        peak_idxs = np.argsort(conv)[::-1][:n_peaks]
        
        for pk in peak_idxs:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            if js >= je:
                continue
            
            for _ in range(30):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(0, n))
                if i == j or a[i] < 1e-14:
                    continue

                    delta = a[i] * frac

                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

            no_improve += 1
            if no_improve > 15:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 100 == 0:
            print(f"    CD: iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    x = np.arange(n) / n
    rc = np.sin(np.pi * x) ** 2
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())
    
    # Gaussian (wide)
    center = (n - 1) / 2.0
    gauss = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    inits.append(gauss / gauss.sum())
    
    # Gaussian (narrow)
    gauss2 = np.exp(-((x - 0.5) ** 2) / (2 * 0.08 ** 2))
    inits.append(gauss2 / gauss2.sum())
    
    # Exponential bell
    exp_bell = np.exp(-3.0 * np.abs(x - 0.5))
    inits.append(exp_bell / exp_bell.sum())
    
    # Random smoothed
    r = rng.standard_normal(n)
    # Smooth via convolution with a small kernel
    kernel = np.ones(5) / 5.0
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    ss = r_smooth.sum()
    if ss > 1e-10:
        inits.append(r_smooth / ss)
    
    # Triangular (inverted - peaks in middle)
    tri = 1.0 - np.abs(2.0 * x - 1.0)
    tri = np.maximum(tri, 1e-10)
    inits.append(tri / tri.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an alternating minimax approach:
- Outer loop: compute soft-max weighted peaks, solve QP subproblem via projected gradient
- QP matvec accelerated by FFT (Hankel structure): O(n log n) per iteration
- Geometric mean blending in log-space for stable outer updates
- Temperature annealing from smooth to peak-focused weighting
- Coordinate descent polish with peak-targeted transfers
- Multi-length and multi-initialization exploration
Key innovation: FFT-based Hankel matvec enables 10x+ more QP iterations than LP approach.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def hankel_matvec(a, p):
    """(P@a)[j] = sum_m p[j+m]*a[m] via FFT. p length 2n-1, a length n."""
    n = len(a)
    q = a[::-1]
    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = rfft(p, fft_size)
    Q = rfft(q, fft_size)
    conv = irfft(P * Q, fft_size)
    return conv[n - 1: n]


def solve_qp(a_init, p, n_iters=60, eta_scale=0.8, momentum=0.0):
    """Projected gradient for min a^T P a s.t. simplex. FFT-accelerated."""
    n = len(a_init)
    a = a_init.copy()
    max_p = max(p.max(), 1e-15)
    L = 2.0 * np.sqrt(n * max_p) + 1e-10
    eta = eta_scale / L

    size = len(p) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    P_fft = rfft(p, fft_size)

    v = np.zeros(n)
    for it in range(n_iters):
        Q = rfft(a[::-1], fft_size)
        conv = irfft(P_fft * Q, fft_size)
        grad = 2.0 * conv[n - 1: n]

        if momentum > 0.0:
            v = momentum * v - eta * grad
            a = project_simplex(a + v)
        else:
            a = project_simplex(a - eta * grad)

    return a


def geometric_blend(a1, a2, t):
    a1s = np.maximum(a1, 1e-30)
    a2s = np.maximum(a2, 1e-30)
    log_new = (1.0 - t) * np.log(a1s) + t * np.log(a2s)
    log_new -= log_new.max()
    res = np.exp(log_new)
    s = res.sum()
    if s < 1e-30:
        return a1.copy()
    return res / s


def alternating_optimize(a0, time_limit, T_start=100.0, T_end=0.5,
                         n_inner=40, momentum=0.0, verbose=True):
    """Alternating minimax: soft-max peaks -> QP -> geometric blend."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    outer_iter = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-10))

        T = T_start * (T_end / T_start) ** progress
        T = max(T, 0.01)

        C = fast_convolve(a)
        C_shifted = C - C.max()
        w = np.exp(C_shifted / T)
        w = np.maximum(w, 1e-300)
        p = w / w.sum()

        a_qp = solve_qp(a, p, n_iters=n_inner, momentum=momentum)

        t_step = max(0.03, 2.0 / (outer_iter + 2.0))
        a_new = geometric_blend(a, a_qp, t_step)

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        outer_iter += 1

        if verbose and outer_iter % 200 == 0:
            print(f"    Alt: iter={outer_iter}, T={T:.4f}, score={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def alternating_hard_peak(a0, time_limit, top_k=5, n_inner=50, verbose=True):
    """Variant: use hard top-k peaks instead of soft-max."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    outer_iter = 0

    while time.monotonic() - t0 < time_limit:
        C = fast_convolve(a)
        # Top-k peaks with equal weight
        k = min(top_k, len(C))
        peak_idx = np.argsort(C)[::-1][:k]
        p = np.zeros(len(C))
        p[peak_idx] = 1.0 / k

        a_qp = solve_qp(a, p, n_iters=n_inner, momentum=0.5)

        t_step = max(0.05, 2.0 / (outer_iter + 2.0))
        a_new = geometric_blend(a, a_qp, t_step)

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        outer_iter += 1

        if verbose and outer_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AltHard: iter={outer_iter}, score={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score
