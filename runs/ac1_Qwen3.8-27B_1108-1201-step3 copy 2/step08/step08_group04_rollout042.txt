If `k` is odd, `A_k` has 1/2 at positions `(i, k-i)` for all valid i. This is also PSD (it's a correlation matrix structure).

Actually, I'm not sure these are all PSD. Let me check a small example.

For n=3, k=1: valid pairs are (0,1) and (1,0). So `A_1 = [[0, 1/2, 0], [1/2, 0, 0], [0, 0, 0]]`. Eigenvalues: 1/2, -1/2, 0. NOT PSD!

So the quadratic constraints are non-convex. That means we can't directly solve the QP.

However, the LP relaxation (replacing the quadratic constraint with linear one using the current f) is convex, which is what the previous approach does.

OK so let me stick with the LP + gradient approach.

"""
Hybrid optimization combining softmax-approximated gradient descent,
dense LP refinement, and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_objective(b, tau):
    """Compute tau * log(sum(exp(b/tau))) and the softmax weights."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / tau)
    log_sum = b.max() + tau * np.log(exp_b.sum())
    p = exp_b / exp_b.sum()
    return log_sum, p


def compute_gradient(p, f):
    """Compute gradient of softmax objective wrt f.
    grad[j] = 2 * sum_i f[i] * p[j+i] for valid j+i
    """
    # p has length 2n-1, f has length n
    # grad[j] = 2 * sum_{i=0}^{n-1} f[i] * p[j+i]
    # This is 2 * np.correlate(p, f, mode='valid')
    return 2.0 * np.correlate(p, f, mode='valid')


def softmax_gd(f_init, time_limit, tau_init=0.02, tau_min=0.0005, 
               momentum=0.7, n_restarts=5, label="sgd"):
    """Softmax-approximated gradient descent on the simplex."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    rng = np.random.default_rng(123)
    restart_count = 0
    
    while time.monotonic() - t0 < time_limit:
        tau = max(tau_min, tau_init * (tau_min / tau_init) ** (min(restart_count, 20) / 20.0))
        
        # Phase: decreasing tau within a restart
        v = np.zeros(n)
        lr = 0.001
        no_improve = 0
        max_no_improve = 50
        
        for it in range(2000):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_convolve(f)
            obj = b.max()
            
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > max_no_improve:
                break
            
            # Compute softmax and gradient
            _, p = softmax_objective(b, tau)
            grad = compute_gradient(p, f)
            
            # Momentum update
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            
            # Backtracking line search
            obj_new = fast_convolve(f_new).max()
            if obj_new > obj + 1e-12:
                # Need to reduce step
                lr *= 0.5
                v = momentum * v - lr * grad
                f_new = project_simplex(f + v)
                obj_new = fast_convolve(f_new).max()
            
            if obj_new <= obj + 1e-12:
                f = f_new
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
            
            # Adapt lr
            if obj_new < obj - 1e-10:
                lr = min(lr * 1.05, 0.05)
            
            # Anneal tau
            tau = max(tau_min, tau * 0.999)
        
        restart_count += 1
        
        # Restart from best with perturbation
        if restart_count < n_restarts:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002 * (1 + restart_count))
        else:
            break
        
        if (restart_count) % 2 == 0:
            print(f"    {label} restart={restart_count} obj={best_obj:.12f} score={best_score:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def dense_lp_step(f, M, solver='SCS'):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)_k <= M for ALL k.
    Returns g or None.
    """
    n = len(f)
    full_len = 2 * n - 1
    
    # Build constraint matrix
    g = cp.Variable(n)
    constraints = [g >= 0]
    
    # For each k in 0..2n-2, the constraint is:
    # sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i] <= M
    # (f*g)_k = sum_i f[i] * g[k-i] where 0<=i<=n-1 and 0<=k-i<=n-1
    
    # Build the matrix: A[k, i] = f[i] if 0 <= k-i <= n-1, else 0
    # But this is a (2n-1) x n matrix, which for n=1152 is 2303 x 1152 ≈ 2.6M entries
    # That's manageable but might be slow.
    
    # Alternative: use the structure. A is a "convolution matrix" (Toeplitz-like).
    # For now, let's build it explicitly.
    
    A = np.zeros((full_len, n))
    for k in range(full_len):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    
    constraints.append(A @ g <= M)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None
    
    if g.value is None:
        return None
    
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    
    return g_val


def continuous_line_search(f, direction, M, max_evals=20):
    """
    Find the best t in (0,1] such that max(conv((1-t)f + t*direction)) is minimized.
    Uses golden section search.
    """
    def objective(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            return M + 1.0
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # First, check if t=1 is better than t=0
    obj1 = objective(1.0)
    if obj1 >= M:
        # Try smaller t
        best_t = 0.0
        best_obj = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            obj_t = objective(t)
            if obj_t < best_obj:
                best_obj = obj_t
                best_t = t
        return best_t, best_obj
    
    # Golden section on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(15):
        if b - a < 1e-4:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = objective(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = objective(d)
    
    best_t = (a + b) / 2
    best_obj = objective(best_t)
    return best_t, best_obj

def coordinate_descent(f_init, time_limit):
    """Coordinate descent."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
    
    no_pass_improve = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            f[j] = max(f[j], 0.0)
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

"""
Hybrid optimizer combining softmax-approximated gradient descent, dense LP refinement,
constraint-generation LP, and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Targets score < 1.5027 starting from parent n=1152 (score 1.504855).
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(p, f):
    """Compute gradient of softmax objective wrt f. grad[j] = 2*sum_i f[i]*p[j+i]."""
    return 2.0 * np.correlate(p, f, mode='valid')


def softmax_gd(f_init, time_limit, tau_init=0.015, tau_min=0.0003,
               momentum=0.75, n_restarts=10, label="sgd"):
    """Softmax-approximated gradient descent on the simplex with annealing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    rng = np.random.default_rng(42)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        tau = tau_init * (tau_min / tau_init) ** (restart / max(n_restarts - 1, 1))
        tau = max(tau, tau_min)
        v = np.zeros(n)
        lr = 0.002 / (1 + 0.1 * restart)
        no_improve = 0
        max_no_improve = 80 + 20 * restart
        it = 0

        while no_improve < max_no_improve and time.monotonic() - t0 < time_limit:
            it += 1
            b = fast_convolve(f)
            obj = b.max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0

            # Softmax weights
            b_shift = b - b.max()
            exp_b = np.exp(b_shift / tau)
            p = exp_b / exp_b.sum()

            grad = compute_gradient(p, f)
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Momentum
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)

            # Check improvement
            b_new = fast_convolve(f_new)
            obj_new = b_new.max()

            if obj_new <= obj + 1e-14:
                f = f_new
                if obj_new < obj - 1e-12:
                    lr = min(lr * 1.08, 0.08)
                else:
                    lr *= 0.98
            else:
                # Backtrack
                lr *= 0.5
                v = momentum * v - lr * grad
                f_new = project_simplex(f + v)
                obj_new = fast_convolve(f_new).max()
                if obj_new <= obj + 1e-14:
                    f = f_new
                else:
                    lr *= 0.5
                    if lr < 1e-9:
                        break
            no_improve += 1
            tau = max(tau_min, tau * 0.998)

        # Restart
        if restart < n_restarts - 1:
            perturb = 0.001 * (1 + 0.3 * restart)
            f = project_simplex(best_f + rng.standard_normal(n) * perturb)

        if restart % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} r={restart} obj={best_obj:.12f} score={best_score:.6f} t={elapsed:.0f}s")

    return best_f, best_score


def build_conv_matrix(f, n, peak_indices=None):
    """Build constraint matrix for selected peak indices."""
    if peak_indices is None:
        peak_indices = range(2 * n - 1)
    peak_indices = list(peak_indices)
    K = len(peak_indices)
    A = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A, peak_indices


def solve_lp(f, M, peak_indices=None, solver='SCS'):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for selected k."""
    n = len(f)
    if peak_indices is None:
        peak_indices = range(2 * n - 1)
    A, _ = build_conv_matrix(f, n, peak_indices)

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)

    try:
        if len(peak_indices) <= 500 and n <= 1200:
            prob.solve(solver=cp.ECOS, max_iters=30000, abs_acceleration=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=300000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def continuous_line_search(f, direction, M, n_evals=12):
    """Find best step size t for combining f and direction."""
    best_t = 0.0
    best_obj = M

    # Try a coarse grid first
    t_grid = np.concatenate([np.arange(0.02, 1.0, 0.08), [1.0]])
    t_grid = t_grid[t_grid <= 1.0]

    for t in t_grid:
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine around best_t
    if best_t > 0:
        t_lo = max(0, best_t - 0.08)
        t_hi = min(1.0, best_t + 0.08)
        for t in np.linspace(t_lo, t_hi, 8):
            f_trial = (1.0 - t) * f + t * direction
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-15:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj


def lp_optimization(f_init, time_limit, top_k=200, use_all=False, label="lp"):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    full_len = 2 * n - 1
    f = f_init.copy()
    f = f / f.sum()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    stuck = 0
    k_use = min(top_k, full_len)
    momentum_dir = np.zeros(n)
    mom_beta = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Select peaks
        if use_all or full_len <= 600:
            indices = range(full_len)
        else:
            # Adaptive: top peaks + boundary + center
            top_idx = np.argpartition(b, -k_use)[-k_use:]
            n_pts = min(n // 10 + 5, 20)
            boundary = list(range(0, min(n_pts, full_len)))
            boundary += list(range(max(0, full_len - n_pts), full_len))
            center = list(range(max(0, (full_len - 1) // 2 - n_pts // 2),
                                min(full_len, (full_len - 1) // 2 + n_pts // 2 + 1)))
            candidates = list(set(list(top_idx) + boundary + center))
            candidates.sort(key=lambda x: b[int(x)], reverse=True)
            indices = candidates[:k_use]

        # Solve LP
        g = solve_lp(f, M, peak_indices=indices)

        if g is None:
            stuck += 2
            if stuck > 40:
                break
            k_use = max(20, k_use // 2)
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 40:
                break
            k_use = min(k_use * 2, full_len)
            continue

        h = g / S_g
        # Momentum
        momentum_dir = mom_beta * momentum_dir + (1.0 - mom_beta) * h
        ms = momentum_dir.sum()
        if ms > 1e-10:
            h_mom = momentum_dir / ms
        else:
            h_mom = h

        # Try both directions
        best_t, best_new_obj = continuous_line_search(f, h, M)
        t2, obj2 = continuous_line_search(f, h_mom, M)
        if obj2 < best_new_obj:
            best_t, best_new_obj, h = t2, obj2, h_mom

        if best_t == 0.0:
            stuck += 1
            if stuck > 40:
                break
            k_use = min(k_use * 2, full_len)
            continue

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_score = 2.0 * n * best_obj
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        # Grow k_use slowly
        if it % 20 == 0 and not use_all:
            k_use = min(k_use + 20, min(600, full_len))

        if it % 10 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={k_use} t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.5, 0.65, 0.75, 0.85, 0.9, 1.1, 1.15, 1.25, 1.4, 1.6]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                f = f / f.sum()
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_imp = True
        if not any_imp:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a normalized sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimizer: softmax GD -> LP -> multi-n -> alternating -> CD.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print("No parent, using uniform n=512")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # Phase 1: Softmax GD on parent
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 350.0)
    print(f"\nPhase 1: Softmax GD n={n_parent}, budget={t1:.0f}s")
    if t1 > 30:
        f_res, _ = softmax_gd(best_seq, time_limit=t1, tau_init=0.015, tau_min=0.0003,
                              momentum=0.75, n_restarts=10, label="sgd1")
        rs = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(rs) and rs < best_score:
            best_score = rs
            best_seq = f_res.copy()
            print(f"  NEW BEST [sgd]: {rs:.6f}")

    # Phase 2: LP on current best (with moderate constraints)
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 350.0)
    print(f"\nPhase 2: LP n={len(best_seq)}, budget={t2:.0f}s")
    if t2 > 30:
        f_res, _ = lp_optimization(best_seq, time_limit=t2, top_k=150, label="lp1")
        rs = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(rs) and rs < best_score:
            best_score = rs
            best_seq = f_res.copy()
            print(f"  NEW BEST [lp]: {rs:.6f}")

    # Phase 3: Multi-n exploration
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 330.0)
    print(f"\nPhase 3: Multi-n, budget={t3:.0f}s")
    if t3 > 40:
        n_base = len(best_seq)
        n_list = []
        for frac in [0.65, 0.75, 0.85, 1.0, 1.1, 1.25, 1.4]:
            nt = int(n_base * frac)
            nt = max(256, min(1800, nt))
            if nt not in n_list:
                n_list.append(nt)
        n_list.sort()
        per_n = t3 / len(n_list)

        for nt in n_list:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            alloc = min(per_n, rem * 0.9)
            if alloc < 20:
                break
            f_init = interp_sequence(best_seq, nt)
            t_gd = min(alloc * 0.5, 80.0)
            t_lp = alloc - t_gd - 5
            if t_gd > 10:
                f_res, _ = softmax_gd(f_init, time_limit=t_gd, tau_init=0.01,
                                      tau_min=0.0003, momentum=0.7, n_restarts=4,
                                      label=f"sgd{nt}")
            else:
                f_res = f_init
            if t_lp > 10:
                f_res, _ = lp_optimization(f_res, time_limit=t_lp,
                                           top_k=min(100, 2 * nt - 1), label=f"lp{nt}")
            rs = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(rs) and rs < best_score:
                best_score = rs
                best_seq = f_res.copy()
                print(f"  NEW BEST [n={nt}]: {rs:.6f}")

    # Phase 4: Deep LP with more constraints on best
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    print(f"\nPhase 4: Deep LP n={len(best_seq)}, budget={t4:.0f}s")
    if t4 > 30:
        f_res, _ = lp_optimization(best_seq, time_limit=t4, top_k=400, label="lp_deep")
        rs = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(rs) and rs < best_score:
            best_score = rs
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {rs:.6f}")

    # Phase 5: Alternating softmax + LP on best
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 150.0)
    print(f"\nPhase 5: Alternate n={len(best_seq)}, budget={t5:.0f}s")
    if t5 > 30:
        rounds = 3
        per_round = t5 / rounds
        for rnd in range(rounds):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            half = min(per_round / 2, rem / 2 - 5)
            if half < 8:
                break
            f_res, _ = softmax_gd(best_seq, time_limit=half, tau_init=0.005,
                                  tau_min=0.0002, momentum=0.8, n_restarts=3, label="alt_s")
            rs = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(rs) and rs < best_score:
                best_score = rs
                best_seq = f_res.copy()
            half2 = min(per_round / 2, (deadline - time.monotonic()) / 2 - 5)
            if half2 < 8:
                break
            f_res, _ = lp_optimization(best_seq, time_limit=half2, top_k=300, label="alt_l")
            rs = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(rs) and rs < best_score:
                best_score = rs
                best_seq = f_res.copy()

    # Phase 6: Coordinate descent
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.07, 80.0)
    print(f"\nPhase 6: CD, budget={t6:.0f}s")
    if t6 > 10:
        f_res, _ = coordinate_descent(best_seq, time_limit=t6)
        rs = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(rs) and rs < best_score:
            best_score = rs
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd]: {rs:.6f}")

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
