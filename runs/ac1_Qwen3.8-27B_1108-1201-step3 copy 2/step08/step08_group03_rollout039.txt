**Mirror Descent for this problem:**

Objective: `F(a) = max_k conv(a,a)_k` (with `sum(a)=1, a>=0`)

Subgradient at `a`: 
`g = (1/|S|) * sum_{k in S} grad conv(a,a)_k`
where `S = {k : conv(a,a)_k = max}` (or use soft weights).

Mirror descent update (with KL divergence as the Bregman divergence):
`a_new_j = a_j * exp(-eta * g_j) / Z`
where `Z = sum_j a_j * exp(-eta * g_j)`

This is equivalent to:
`log(a_new_j) = log(a_j) - eta * g_j - log(Z)`

With Nesterov acceleration:
`y_k = a_k + (t_{k-1} - 1)/(t_k) * (a_k - a_{k-1})`
`a_{k+1} = argmin_{a in simplex} <g_k, a> + (1/(2*eta_k)) * D(a, y_k)`

where `D` is the KL divergence.

Let me code this up:

```python
def mirror_descent_step(a, g, eta):
    """One step of mirror descent with KL divergence."""
    # a_new_j = a_j * exp(-eta * g_j) / Z
    log_a = np.log(a + 1e-30)
    log_a_new = log_a - eta * g
    log_a_new -= log_a_new.max()  # for numerical stability
    a_new = np.exp(log_a_new)
    a_new /= a_new.sum()
    return a_new
```

Optimized sequence search combining accelerated mirror descent (fast O(n log n)
per-iteration subgradient via FFT), cutting-plane LP refinement, and coordinate
descent polish. Designed to improve upon the LP-only approach by using the
much faster mirror descent for the bulk of the optimization, then switching
to LP for fine-tuning.

from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, w):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    w_rev = w[::-1]
    A = rfft(a, N)
    W = rfft(w_rev, N)
    full_conv = irfft(A * W, N)[:3 * n - 2]
    return 2.0 * full_conv[n - 1 : 2 * n - 1][::-1]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

Optimized sequence search using accelerated mirror descent (fast O(n log n)
per-iteration subgradient via FFT cross-correlation), cutting-plane LP
refinement, and coordinate descent polish. Key innovations over previous:
(1) FFT-based subgradient computed in O(n log n) per step, enabling millions
of iterations; (2) Nesterov acceleration in mirror (KL) geometry for faster
convergence; (3) adaptive cooling of peak weights; (4) smart restarts with
diversification to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, w):
    """
    Compute subgradient of max_k conv(a,a)_k w.r.t. a using FFT.
    g_i = 2 * sum_j a_j * w_{i+j}
    """
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    w_rev = w[::-1]
    A = rfft(a, N)
    W = rfft(w_rev, N)
    full_conv = irfft(A * W, N)[:3 * n - 2]
    return 2.0 * full_conv[n - 1 : 2 * n - 1][::-1]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Nesterov-accelerated mirror descent with adaptive temperature.
    Uses KL geometry (multiplicative updates) for efficient optimization.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()

    b = fast_convolve(a)
    cur_obj = b.max()
    best_obj = cur_obj

    t_nest = 1.0
    eta = 0.08
    no_improve = 0
    restart_count = 0
    rng = np.random.default_rng(777)

    for it in range(20000000):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.05:
            break

        # Periodic logging and parameter adjustment
        if it > 0 and it % 50000 == 0:
            if rem < 0.5:
                break
            print(f"  {name} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"eta={eta:.5f} restarts={restart_count} t={time.monotonic()-t0:.0f}s")

            # Adapt eta based on progress
            if no_improve > 20000:
                # Try different restart strategies
                restart_count += 1
                strategy = restart_count % 3
                if strategy == 0:
                    # Small noise perturbation
                    a = a + rng.standard_normal(n) * 0.002
                elif strategy == 1:
                    # Smoothing (blur)
                    kernel = np.array([0.25, 0.5, 0.25])
                    a_blur = np.convolve(a, kernel, mode='same')
                    a = a_blur
                else:
                    # Blend with uniform
                    a = 0.9 * a + 0.1 * (1.0 / n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                a_prev = a.copy()
                t_nest = 1.0
                eta = max(eta * 0.6, 0.005)
                no_improve = 0
                if restart_count > 20:
                    break
            continue

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        if t_nest > 1.0 + 1e-10:
            beta = (t_nest - 1.0) / t_new
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-30:
                y = a.copy()
            else:
                y /= sy
        else:
            y = a.copy()
            sy = y.sum()
            if sy > 1e-30:
                y /= sy

        # Compute convolution and soft-max weights
        b = fast_convolve(y)
        M = b.max()

        # Cooling schedule: hot early (explore), cold late (exploit)
        frac = min(1.0, (it + 1) / 1000000.0)
        tau_cur = 0.15 * (0.005) ** frac
        scaled = (b - M) / max(tau_cur, 1e-8)
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient via FFT
        g = compute_subgradient(y, w)

        # Mirror descent step: a_new_j = y_j * exp(-eta * g_j) / Z
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * g
        log_new = np.clip(log_new, -80.0, 80.0)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        sa = a_new.sum()
        if sa < 1e-30:
            a_new = y.copy()
        else:
            a_new /= sa

        # Evaluate new point
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < cur_obj - 1e-16:
            a_prev = a
            a = a_new
            cur_obj = obj_new
            no_improve = 0
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_a = a.copy()
                # Slowly increase step size when making progress
                eta = min(eta * 1.0005, 0.5)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                eta *= 0.8
                if eta < 1e-7:
                    # Stuck: accept and reset
                    a_prev = a
                    a = a_new
                    cur_obj = max(obj_new, cur_obj * 0.999)
                    eta = 0.02
                    no_improve = 0

        t_nest = t_new

    return best_a, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for LP direction finding."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    """Solve max sum(g) s.t. g>=0, conv(f,g) at active indices <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9)
    except Exception:
        if n <= 2000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    if g_val.sum() < 1e-10:
        return None
    return g_val


def lp_refine(f_init, time_limit, initial_k=50, max_k=400, name="lp"):
    """LP-based cutting plane refinement for fine-tuning."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    k_init = min(initial_k, len(b) - 1)
    active_idx = np.sort(np.argpartition(b, -k_init)[-k_init:])

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_lp_direction(f, n, M, active_idx)
        if g is None:
            stuck += 1
            if stuck > 8:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                # Expand active set
                if len(active_idx) < max_k:
                    b_cur = fast_convolve(f)
                    new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                    active_idx = np.unique(np.concatenate([active_idx, new_top]))
                    active_idx = active_idx[np.argsort(b_cur[active_idx])[-max_k:]]
                else:
                    break
            continue

        stuck = 0
        h = g / S_g

        # Check for violated constraints (cutting plane)
        fg = np.convolve(f, h)[:2 * n - 1]
        violated = fg > M * (1 + 1e-7)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            new_ones = v_idx[~np.isin(v_idx, active_idx)]
            if len(new_ones) > 0:
                n_add = min(20, len(new_ones))
                to_add = new_ones[np.argsort(fg[new_ones])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 10:
                        break
                    continue
                h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.03, 0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # Refresh active set periodically
        if it % 8 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 5 == 0:
            print(f"  {name} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 600)

    for sweep in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-16:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3, 1.5]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-15:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-15:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def proposed_projected_gradient(a_init, time_limit, name="pg"):
    """
    Projected gradient with Nesterov momentum (FISTA variant).
    Alternative to mirror descent - operates in Euclidean geometry.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    z = a.copy()  # extrapolated point
    t_k = 1.0
    lr = 0.003
    best_a = a.copy()
    b = fast_convolve(a)
    best_obj = b.max()
    momentum = 0.7
    rng = np.random.default_rng(42)
    no_improve = 0

    for it in range(5000000):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.05:
            break
        if it > 0 and it % 30000 == 0:
            if no_improve > 15000:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a /= a.sum()
                z = a.copy()
                t_k = 1.0
                lr = max(lr * 0.7, 1e-6)
                no_improve = 0
            if it % 100000 == 0:
                print(f"  {name} it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")
            continue

        b = fast_convolve(z)
        M = b.max()

        # Soft-max weights
        tau = 0.05
        scaled = (b - M) / tau
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient
        g = compute_subgradient(z, w)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        a_new = project_simplex(z - lr * g + beta * (z - a))
        z = a_new
        a = a_new
        t_k = t_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_obj
