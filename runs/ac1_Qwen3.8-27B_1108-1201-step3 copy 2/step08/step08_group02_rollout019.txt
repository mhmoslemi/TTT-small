def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):

    """

    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = hi - lo + 1
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]

    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, n, peaks, M, eps=1e-9):
    K = len(peaks)
    # Build A matrix: A[k, j] = f[j] if j is in the valid range for peak k
    A = np.zeros((K, n))
    for i, k in enumerate(peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=100000, eps=eps)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val
        except Exception:
            continue
    return None


def golden_line_search(f, h):
    """Golden section search for best t in (0,1]."""
    def obj(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-10:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()
    
    gr = (1.0 - np.sqrt(5.0)) / 2.0  # golden ratio ~0.382
    a, b = 0.001, 1.0
    c = a + (1.0 - gr) * (b - a)
    d = a + gr * (b - a)
    fc, fd = obj(c), obj(d)
    for _ in range(25):
        if fc < fd:
            b, d, fd = d, c, fc
            c = a + (1.0 - gr) * (b - a)
            fc = obj(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = obj(d)
    t = (a + b) / 2.0
    return t, obj(t)


def lp_optimize(f0, time_limit, label, peak_frac_init=0.95, peak_frac_final=0.80,

        elif len(peaks) < 5:

                f /= f.sum()
                stuck = 0

        t_best, obj_best = golden_line_search(f, h)
