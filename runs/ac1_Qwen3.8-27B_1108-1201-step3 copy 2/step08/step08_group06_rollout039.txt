"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space with log-sum-exp smoothing (fast, new)
2. Mirror descent / multiplicative weights (new)
3. LP fictitious play with geometric mean updates (proven)
4. Coordinate descent with golden-section search (polish)
5. Multiple initializations and length exploration
6. Random restarts for diversity
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def cross_correlate_weights(w, a):
    """Compute S1[j] = sum_k w[k] * a[k-j] for j in [0, n-1]."""
    n = len(a)
    b = a[::-1]
    size = 2 * n - 1 + n - 1  # len(w) + len(b) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    B = rfft(b, fft_size)
    conv = irfft(W * B, fft_size)[:size]
    return conv[n-1:2*n-1]

def build_C(f):
    """Build constraint matrix C where C[k,j] = f[|k-j|] if |k-j|<n else 0.
    Actually C[k,j] = f[k-j] if 0<=k-j<n else 0 (for the convolution constraint)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)

def adam_logspace(a_init, time_limit, T0=500.0, T_min=1.0, lr=0.5, beta1=0.9, beta2=0.999, verbose=True):
    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    a = normalize(a_init)
    
    # Convert to log-space (z such that a = softmax(z))
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe) - np.log(a_safe).mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # T schedule: geometric decay
    total_iters_est = 5000
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t += 1
        # T schedule
        progress = min(1.0, t / total_iters_est)
        T = max(T_min, T0 * (T_min / T0) ** progress)
        
        # Compute current a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        # Compute convolution and weights
        conv = fast_convolve(a)
        shifted = conv / T
        shifted -= shifted.max()
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        
        # Compute S1[j] = sum_k w[k] * a[k-j]
        S1 = cross_correlate_weights(w, a)
        S2 = np.dot(w, conv)
        
        # Gradient in log-space: grad_z[j] = 4n * a[j] * (S1[j] - S2)
        grad_z = 4.0 * n * a * (S1 - S2)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Check improvement
        if it % 10 == 0:
            score = compute_score(a)
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()
        
        if verbose and it % 200 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  Adam iter={it}, T={T:.2f}, score={score:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    # Final check
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a_final = exp_z / exp_z.sum()
    score = compute_score(a_final)
    if score < best_score:
        best_score = score
        best_a = a_final.copy()
    
    return best_a, best_score

def mirror_descent(a_init, time_limit, T0=200.0, T_min=0.5, eta_scale=1.0, verbose=True):
    """Mirror descent / multiplicative weights with adaptive step size."""
    n = len(a_init)
    a = normalize(a_init)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    total_iters_est = 3000
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / total_iters_est)
        T = max(T_min, T0 * (T_min / T0) ** progress)
        
        conv = fast_convolve(a)
        shifted = conv / T
        shifted -= shifted.max()
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        
        # Gradient: grad[j] = 4n * sum_k w[k] * a[k-j]
        S1 = cross_correlate_weights(w, a)
        grad = 4.0 * n * S1
        
        # Adaptive eta: scale so max update is reasonable
        eta = eta_scale * T / (n * grad.max() + 1e-10)
        eta = min(eta, 1.0)  # cap
        
        # Multiplicative update
        log_factor = -eta * grad
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = normalize(a)
        
        if it % 20 == 0:
            score = compute_score(a)
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()
        
        if verbose and it % 200 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"  MD iter={it}, T={T:.2f}, eta={eta:.4f}, score={score:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    score = compute_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def lp_fictitious_play(a_init, time_limit, verbose=True):
    """LP fictitious play with geometric mean updates."""
    n = len(a_init)
    a = normalize(a_init)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    a_avg = a.copy()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(a).max()
        g = solve_lp_direction(a, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Geometric mean update with decreasing step
        t_step = 2.0 / (it + 2)
        t_step = min(t_step, 0.3)
        
        a_safe = np.maximum(a, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(a_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = normalize(a_new)
        
        # Track average
        a_avg = ((it * a_avg) + a_new) / (it + 1)
        
        score_cur = compute_score(a_new)
        score_avg = compute_score(a_avg)
        
        if score_cur < best_score - 1e-15:
            best_score = score_cur
            best_a = a_new.copy()
        if score_avg < best_score - 1e-15:
            best_score = score_avg
            best_a = a_avg.copy()
        
        a = a_new
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it}, M={M:.6e}, cur={score_cur:.8f}, avg={score_avg:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score

def coordinate_descent_polish(a_init, time_limit, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a_init)
    a = normalize(a_init)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Get current peaks
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        
        # Generate candidate pairs
        pairs = set()
        # Peak-focused
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(100):
            i = rng.integers(js, je + 1)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        # Random
        for _ in range(100):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Golden section search
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t_val):
                a[i] -= t_val
                a[j] += t_val
                if a[i] < 0 or a[j] < 0:
                    a[i] += t_val
                    a[j] -= t_val
                    return 1e10
                a = normalize(a)
                sc = compute_score(a)
                # Restore (we'll apply at the end)
                a[j] -= t_val
                a[i] += t_val
                return sc
          
            # Actually, let's do it more carefully with a copy
            a_orig = a.copy()
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            a = a_orig  # restore
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = normalize(a)
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Adam optimizer in log-space with log-sum-exp smoothing (fast gradient method)
2. Minimax LP direction-finding combined with geometric mean updates
3. Mirror descent / multiplicative weights
4. Coordinate descent with golden-section search for final polish
5. Multiple diverse initializations and length exploration
The key novel contributions are: (a) Adam in log-space with proper gradient
derivation for the softmax-parameterized problem, (b) a minimax LP formulation
that directly targets reducing the peak cross-correlation, and (c) a combined
annealing schedule that smoothly transitions from exploration to exploitation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def cross_correlate_weights(w, a):
    """Compute S1[j] = sum_k w[k] * a[k-j] for j in [0, n-1].
    w has length 2n-1, a has length n."""
    n = len(a)
    b = a[::-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    B = rfft(b, fft_size)
    conv = irfft(W * B, fft_size)[:size]
    return conv[n - 1:2 * n - 1]


def build_C(f):
    """Build constraint matrix: C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_widest(f, M):
    """LP: max sum(g) s.t. C*g <= M*1, g >= 0. Returns g or None."""
    n = len(f)
    C = build_C(f)
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 10})
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0)


def solve_lp_minimax(f):
    """Minimax LP: min t s.t. C*g <= t*1, sum(g)=1, g>=0. Returns (g, t) or None."""
    n = len(f)
    C = build_C(f)
    # Variables: [g_0, ..., g_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    # Constraints: C*g - t*I <= 0
    A_ub = np.hstack([C, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 10})
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x[:n], 0)
    g = g / max(g.sum(), 1e-30)
    return g, res.x[-1]


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation: f^(1-t) * g^t, normalized."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    return normalize(a)


def adam_logspace(a_init, time_limit, T0=1000.0, T_min=0.5, lr=0.3,
                  beta1=0.9, beta2=0.999, verbose=True):
    """Adam optimizer in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    a = normalize(a_init)
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe) - np.log(a_safe).mean()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        step += 1
        progress = min(1.0, step / 3000)
        T = max(T_min, T0 * (T_min / T0) ** progress)

        # Get current a from z (softmax)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()

        # Compute convolution and weights
        conv = fast_convolve(a)
        shifted = conv / T
        shifted -= shifted.max()
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient in log-space: grad_z[j] = 4n * a[j] * (S1[j] - S2)
        S1 = cross_correlate_weights(w, a)
        S2 = np.dot(w, conv)
        grad_z = 4.0 * n * a * (S1 - S2)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Adaptive learning rate: scale down as T decreases
        lr_eff = lr * max(0.1, T / T0) ** 0.5
        z -= lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Periodic check
        if it % 5 == 0:
            score = compute_score(a)
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()

        if verbose and it % 200 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"    Adam iter={it}, T={T:.1f}, lr={lr_eff:.4f}, "
                  f"score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a_final = exp_z / exp_z.sum()
    score = compute_score(a_final)
    if score < best_score:
        best_score = score
        best_a = a_final.copy()

    return best_a, best_score


def mirror_descent(a_init, time_limit, T0=500.0, T_min=0.3, eta_base=0.8, verbose=True):
    """Mirror descent / multiplicative weights with adaptive step."""
    n = len(a_init)
    a = normalize(a_init)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / 2000)
        T = max(T_min, T0 * (T_min / T0) ** progress)

        conv = fast_convolve(a)
        shifted = conv / T
        shifted -= shifted.max()
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient magnitude for step sizing
        S1 = cross_correlate_weights(w, a)
        grad_mag = np.max(S1) + 1e-15

        # Adaptive eta
        eta = eta_base * T / (n * grad_mag + 1.0)
        eta = min(eta, 1.5)

        # Multiplicative update: a_j *= exp(-eta * S1[j])
        log_factor = -eta * S1
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = normalize(a)

        if it % 10 == 0:
            score = compute_score(a)
            if score < best_score - 1e-15:
                best_score = score
                best_a = a.copy()

        if verbose and it % 300 == 0:
            score = compute_score(a)
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it}, T={T:.1f}, eta={eta:.5f}, "
                  f"score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    score = compute_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

        t_alloc = min(rem * 0.85, 100.0)
        a_last, _ = lp_fictitious_play(best_seq, t_alloc, use_minimax=False, verbose=False)
        best_seq, best_score = check_update(a_last, best_seq, best_score, "last_lp")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(64) / 64
    best_seq = normalize(best_seq)
    final_sc = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
