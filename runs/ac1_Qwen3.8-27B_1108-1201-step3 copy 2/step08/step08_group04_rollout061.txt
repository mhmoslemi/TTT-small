Uses FFT-based subgradient method with spectral shaping, Nesterov momentum,
and multi-length exploration.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_fft(a, w, fft_size):
    """
    Compute subgradient of max_k (a*a)_k using FFT.
    w: weights on the peaks (length 2n-1), summing to 1.
    Returns gradient vector g of length n.
    """
    n = len(a)
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g

def spectral_shape(a, M_target, alpha=0.5):
    """
    Spectral shaping: reduce spectral peaks above M_target.
    Returns modified a (not yet projected to simplex).
    """
    n = len(a)
    fft_size = 1 << (3*n - 3).bit_length()  # enough for linear operations
    A = fft(a, fft_size)
    P = np.abs(A)**2
    
    # Reduce peaks
    excess = np.maximum(0, P - M_target)
    P_new = P - alpha * excess
    P_new = np.maximum(P_new, 0)
    
    # Reconstruct spectrum
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    # Keep DC component fixed
    A_new[0] = A[0]
    
    a_new = np.real(ifft(A_new))[:n]
    return a_new

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def coordinate_descent_polish(a_init, time_limit):
    """
    Coordinate descent: try to improve by adjusting individual or paired elements.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        any_improved = False
        
        for j in range(0, n, max(1, n // 200)):  # Skip some for speed
            if time.monotonic() - t0 > time_limit:
                break
            
            for mult in multipliers:
                a_trial = a.copy()
                a_trial[j] *= mult
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial = a_trial / s
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    any_improved = True
                    break
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0
    
    return best_a, 2.0 * n * best_obj

We have a = (a_0, ..., a_{n-1}), a_j ≥ 0, sum(a) = 1.
The DFT (with zero-padding to N ≥ 2n-1) is F_k = sum_j a_j * exp(-2πijk/N).
The "power spectrum" is |F_k|^2.
The autocorrelation (a*a)_k is proportional to the inverse DFT of |F|^2.

Specifically, (a*a)_k = (1/N) * sum_{m=0}^{N-1} |F_m|^2 * exp(2πimk/N).

We want to minimize max_k (a*a)_k. By Parseval, sum_k (a*a)_k = sum_j a_j^2 (the energy).

So we want to minimize the peak of the inverse DFT of |F|^2, subject to |F_0| = 1 (since F_0 = sum a = 1) and the constraint that a = IDFT(F) is non-negative.

This is essentially a PAPR minimization problem with the constraint that the time-domain signal is non-negative.

A classic approach to PAPR reduction is the "successive over-sampling" or "selective mapping" method, but those are for complex signals. For our problem, we need a different approach.

**Key insight**: The constraint a ≥ 0 means that F is the characteristic function of a probability distribution on {0, 1, ..., n-1}. The question is: what is the minimum possible PAPR of the characteristic function of a probability distribution on n points?

For the uniform distribution on n points, the characteristic function is F_k = (1/n) * sum_{j=0}^{n-1} exp(-2πijk/n) (for N=n, circular case). This has a peak at k=0 and zeros elsewhere (for N=n). But we're using N ≥ 2n-1 (linear case), so the spectrum is different.

F(k) = sum_{j=0}^{n-1} a_j * exp(-2πik/N) for k = 0, ..., N-1, where N ≥ 2n-1.

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(a, w, fft_size):
    """
    Compute gradient of sum_k w_k * (a*a)_k using FFT cross-correlation.
    w: weights (length 2n-1), need not sum to 1.
    Returns gradient vector of length n.
    """
    n = len(a)
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g


def spectral_shape_step(a, M_target, alpha, fft_size):
    """
    One step of spectral shaping: reduce |F(k)|^2 above M_target.
    Returns modified a (not projected).
    """
    n = len(a)
    A = fft(a, fft_size)
    P = np.abs(A) ** 2

    # Soft clipping: reduce excess above target
    excess = np.maximum(0.0, P - M_target)
    P_new = P - alpha * excess
    P_new = np.maximum(P_new, 1e-30)

    # Reconstruct with modified amplitude, same phase
    magnitude = np.sqrt(P_new)
    phase = np.angle(A)
    A_new = magnitude * np.exp(1j * phase)
    # Preserve DC
    A_new[0] = A[0]

    a_new = np.real(ifft(A_new))[:n]
    return a_new

def spectral_optimize(a_init, time_limit, label="spec"):
    """
    Spectral shaping optimization: iteratively flatten the power spectrum
    while maintaining non-negativity via simplex projection.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    fft_size = 1 << (3 * n - 3).bit_length()
    conv_fft_size = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    alpha = 0.2
    blend = 0.4
    no_improve = 0
    it = 0
    rng = np.random.default_rng(4242 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = fast_convolve(a)
        M = c.max()

        # Shape the spectrum
        a_shaped = spectral_shape_step(a, M * 1.005, alpha, fft_size)

        # Blend and project
        a_new = (1.0 - blend) * a + blend * a_shaped
        a_new = project_simplex(a_new)

        c_new = fast_convolve(a_new)
        M_new = c_new.max()

        if M_new < best_obj * (1.0 - 1e-13):
            best_obj = M_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.003, 0.8)
        else:
            a = a_new.copy()
            no_improve += 1
            if no_improve > 100:
                alpha = max(alpha * 0.92, 0.03)
                blend = max(blend * 0.97, 0.05)
            if no_improve > 400:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                alpha = 0.15
                blend = 0.3
                no_improve = 0

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} n={n} it={it} score={score:.6f} "
                  f"alpha={alpha:.3f} blend={blend:.3f} t={elapsed:.0f}s")

    return best_a, 2.0 * n * best_obj


def combined_optimize(a_init, time_limit, label="combo"):
    """
    Alternates between subgradient steps and spectral shaping steps.
    Spectral shaping every 50 iterations for direct peak reduction.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    fft_size = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lr = 0.003
    momentum = 0.88
    temp = 0.008
    prev_a = a.copy()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(31337 + n)
    spec_alpha = 0.15
    spec_blend = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Every 40th iteration: spectral shaping step
        if it % 40 == 0:
            c = fast_convolve(a)
            M = c.max()
            a_shaped = spectral_shape_step(a, M * 1.003, spec_alpha, fft_size)
            a_new = (1.0 - spec_blend) * a + spec_blend * a_shaped
            a_new = project_simplex(a_new)
            c_new = fast_convolve(a_new)
            M_new = c_new.max()
            prev_a = a
            a = a_new
            if M_new < best_obj * (1.0 - 1e-13):
                best_obj = M_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Standard subgradient step
            a_y = a + momentum * (a - prev_a)
            a_y = np.maximum(a_y, 0.0)
            s_y = a_y.sum()
            if s_y > 1e-15:
                a_y = a_y / s_y

            c = fast_convolve(a_y)
            M = c.max()

            scaled = (c - M) / (temp * M + 1e-30)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(c))
                w[np.argmax(c)] = 1.0
            else:
                w = w / w_sum

            g = compute_subgradient_fft(a_y, w, fft_size)
            a_new = a_y - lr * g
            a_new = project_simplex(a_new)

            c_new = fast_convolve(a_new)
            M_new = c_new.max()

            prev_a = a
            a = a_new

            if M_new < best_obj * (1.0 - 1e-13):
                best_obj = M_new
                best_a = a_new.copy()
                no_improve = 0
                if lr < 0.06:
                    lr *= 1.003
            else:
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.94
                    temp = min(temp * 1.05, 0.5)
                if no_improve > 200:
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                    prev_a = a.copy()
                    lr = 0.003
                    temp = 0.008
                    no_improve = 0

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} n={n} it={it} score={score:.6f} "
                  f"lr={lr:.5f} t={elapsed:.0f}s")

    return best_a, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def coordinate_descent_polish(a_init, time_limit):
    """
    Coordinate descent: try multipliers on individual coordinates.
    Uses stride to limit computation per pass.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    multipliers = np.array([0.6, 0.8, 0.92, 0.97, 1.03, 1.08, 1.2, 1.5])
    no_pass = 0
    stride = max(1, n // 300)

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(0, n, stride):
            if time.monotonic() - t0 > time_limit:
                return best_a, 2.0 * n * best_obj
            if a[j] < 1e-14:
                continue
            for mult in multipliers:
                a[j] *= mult
                s = a.sum()
                if s < 1e-12:
                    a[j] /= mult
                    continue
                a_norm = a / s
                obj = fast_convolve(a_norm).max()
                if obj < best_obj - 1e-14:
                    a = a_norm.copy()
                    best_obj = obj
                    best_a = a.copy()
                    any_improved = True
                    break
                else:
                    a[j] /= mult  # revert

        if not any_improved:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0
            stride = max(1, stride // 2)  # Refine: check more coordinates

    return best_a, 2.0 * n * best_obj
