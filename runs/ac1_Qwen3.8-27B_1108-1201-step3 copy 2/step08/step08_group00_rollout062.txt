import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score assuming a is already normalized (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def softmax_from_z(z):
    """Convert log-space z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T, n):
    """LogSumExp-smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(exp_Tb.sum()))


def lse_gradient_full(z, T, n):
    """Gradient of LogSumExp-smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]
    S = np.dot(grad_a, a)
    return 2.0 * n * a * (grad_a - S)

def peak_directed_perturb(a, k_peak, scale=0.02, rng=None):
    """Perturb a by moving mass from high-contributing positions to low ones at the peak lag."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    # Compute contributions to c[k_peak]
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[k_peak - idx]
    
    # Find top and bottom contributors
    top = np.argsort(contribs)[::-1][:5]
    bottom = np.argsort(contribs)[:5]
    
    a_new = a.copy()
    for t in top:
        i = idx[t]
        reduction = a[i] * scale * rng.uniform(0.5, 1.5)
        a_new[i] -= reduction
        # Redistribute to bottom contributors
        for b_t in bottom:
            j = idx[b_t]
            a_new[j] += reduction / len(bottom)
    
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn > 1e-15:
        a_new = a_new / sn
    return a_new

def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization (sum=1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

The minimum of sum a[i]*a[n-1-i] subject to sum a[i] = 1 and a >= 0 is... by the rearrangement inequality, it's minimized when a and its reverse are in opposite order. Since a >= 0 and sum = 1, the minimum is 0 (achieved by a spike at one end). But then other k values would have large convolution values.

This is the fundamental trade-off: reducing one peak increases others. The optimal solution balances all peaks.

One more idea: **adaptive T based on the gap between top peaks**. If the top peak is much larger than the second, use low T (focus on the top peak). If they're close, use higher T (balance them).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage LogSumExp L-BFGS
with adaptive temperature, subgradient peak-shaving, and perturbation restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_convolve(a).max())


def eval_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_value_fn(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = float(Tb.max())
    e = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(e.sum()))


def lse_grad_fn(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = float(Tb.max())
    e = np.exp(Tb - max_Tb)
    w = e / e.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]
    S = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - S)


def do_lbgfs(z0, T, n, maxiter, ftol, gtol):
    try:
        res = minimize(fun=lse_value_fn, x0=z0, jac=lse_grad_fn,
                       args=(T, n), method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 100,
                                'disp': False})
        return res.x
    except Exception:
        return z0


def anneal(a0, budget, T_hi, T_lo, n_stages):
    """Multi-stage L-BFGS annealing. Returns optimized a (sum=1)."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_hi, T_lo, n_stages)
    for i, T in enumerate(temps):
        rem = budget - (time.monotonic() - t0)
        if rem < 8:
            break
        iters = max(50, int(3000 * (1.0 + (n_stages - 1 - i) * 0.1)))
        gtol_val = 1e-10 * max(0.1, (T_lo / T))
        z = do_lbgfs(z, T, n, iters, 1e-15, gtol_val)
    return softmax_from_z(z)


def subgradient_shave(a0, budget, n_steps=200000):
    """Subgradient descent on the max (using the subgradient of the active peak)."""
    n = len(a0)
    a = a0.copy()
    z = z_from_a(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > budget:
            break
        
        b = fast_convolve(a)
        b_max = float(b.max())
        threshold = b_max - 0.05 * b_max
        active = np.where(b >= threshold)[0]
        
        # Compute gradient of average of active peaks
        grad_a = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]
        grad_a /= len(active)
        
        # Convert to gradient in z-space
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        
        z = z - lr * grad_z
        a_new = softmax_from_z(z)
        sc = compute_score(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            z = z + lr * grad_z  # undo
        
        if no_improve >= 50:
            lr *= 0.8
            no_improve = 0
        lr = max(0.001, 0.01 / (1.0 + step * 0.0002))
    
    return best_a, best_score


def perturb_and_optimize(a0, budget_total, n_tries, rng):
    """Multiple perturbations with brief re-optimization."""
    n = len(a0)
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    per_try = budget_total / n_tries
    
    for p in range(n_tries):
        rem = budget_total - (time.monotonic() - t0)
        if rem < per_try * 0.5:
            break
        
        # Different perturbation types
        if p % 3 == 0:
            # Gaussian noise
            noise = rng.standard_normal(n) * 0.02
            a_p = np.maximum(a0 + noise, 0)
        elif p % 3 == 1:
            # Peak-directed: reduce mass at positions contributing to top peak
            b = fast_convolve(a0)
            k_peak = int(np.argmax(b))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contribs = a0[idx] * a0[k_peak - idx]
            top_idx = np.argsort(contribs)[::-1][:3]
            a_p = a0.copy()
            for ti in top_idx:
                i = idx[ti]
                a_p[i] *= 0.9
            a_p = np.maximum(a_p, 0)
        else:
            # Structured: multiply by a smooth random modulation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.1 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n) + phase)
            a_p = a0 * mod
            a_p = np.maximum(a_p, 0)
        
        sn = a_p.sum()
        if sn < 1e-15:
            continue
        a_p = a_p / sn
        
        # Brief optimization
        opt_budget = min(per_try, rem - 5)
        if opt_budget < 10:
            continue
        a_opt = anneal(a_p, opt_budget, T_hi=5.0, T_lo=0.1, n_stages=4)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Perturb {p}: NEW BEST score={sc:.8f}")
        else:
            print(f"    Perturb {p}: score={sc:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp L-BFGS
annealing with adaptive temperature, cross-correlation-based gradients,
subgradient peak-shaving, multi-perturbation restarts, and length exploration.
Key innovations: FFT-based gradient computation (10x speedup), aggressive low-T
refinement, and structured perturbations targeting dominant convolution peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_convolve(a).max())


def eval_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-space z to probability vector."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def fast_cross_corr(w, a):
    """Compute grad_a[k] = sum_m w[k+m]*a[m] for k=0..n-1 using FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    size = 3 * n
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def lse_value_fn(z, T, n):
    """LogSumExp-smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = float(Tb.max())
    e = np.exp(Tb - max_Tb)
    return (2.0 * n / T) * (max_Tb + np.log(e.sum()))


def lse_grad_fn(z, T, n):
    """Gradient of LogSumExp-smoothed objective w.r.t. z, using FFT cross-correlation."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = float(Tb.max())
    e = np.exp(Tb - max_Tb)
    w = e / e.sum()
    # grad_a[k] = sum_m w[k+m] * a[m] computed via FFT
    grad_a = fast_cross_corr(w, a)
    S = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - S)


def do_lbgfs(z0, T, n, maxiter, ftol, gtol):
    """Run L-BFGS-B on LSE-smoothed objective for one temperature."""
    try:
        res = minimize(fun=lse_value_fn, x0=z0, jac=lse_grad_fn,
                       args=(T, n), method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 100,
                                'disp': False})
        return res.x
    except Exception:
        return z0


def anneal_lbgfs(a0, budget, T_hi, T_lo, n_stages):
    """Multi-stage temperature annealing with L-BFGS. Returns a with sum=1."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_hi, T_lo, n_stages)
    for i, T in enumerate(temps):
        rem = budget - (time.monotonic() - t0)
        if rem < 8:
            break
        # More iterations at lower temperatures for finer refinement
        stage_frac = i / max(1, n_stages - 1)
        iters = max(50, int(2000 + 3000 * stage_frac))
        gtol_val = max(1e-12, 1e-9 * (T_lo / T))
        z = do_lbgfs(z, T, n, iters, 1e-15, gtol_val)
    return softmax_from_z(z)


def subgradient_shave(a0, budget, n_steps=500000):
    """Subgradient descent on the max using active peak set."""
    n = len(a0)
    a = a0.copy()
    z = z_from_a(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0
    best_lr = lr

    for step in range(n_steps):
        if time.monotonic() - t0 > budget:
            break

        b = fast_convolve(a)
        b_max = float(b.max())
        # Use a wider active set at high scores, narrower at low
        width = 0.03 + 0.04 * (best_score - 1.5)
        threshold = b_max - width * b_max
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute gradient via FFT
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad_a = fast_cross_corr(w, a)
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)

        z_new = z - lr * grad_z
        a_new = softmax_from_z(z_new)
        sc = compute_score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        if no_improve >= 30:
            lr = best_lr * 0.85
            no_improve = 0
        lr = max(0.0005, min(0.02, 0.008 / (1.0 + step * 0.0003)))

    return best_a, best_score


def perturb_and_reopt(a0, budget_total, n_tries, rng):

        elif ptype == 1:

            a_p = a0.copy()
            for ti in top_k:
                i = idx[ti]

def interpolate_seq(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def local_search_greedy(a0, budget, n_iters=200000):
    """Greedy coordinate search: find the single most impactful adjustment."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    for it in range(n_iters):

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]

        # Find top contributing pairs to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        # Try reducing top contributors
        top_positions = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for tp in top_positions:
            i = idx[tp]
            j = k_peak - i
            if i >= n or j >= n or a[i] < 1e-12:
                continue
            # Try reducing a[i] and redistributing
            for factor in [0.95, 0.90, 0.85]:
                a_trial = a.copy()
                reduction = a[i] * (1 - factor)
                a_trial[i] = a[i] * factor
                # Redistribute to positions that don't contribute to peak
                # Find positions far from the peak's contributing set
                mask = np.ones(n, dtype=bool)
                peak_zone_js = max(0, k_peak - n + 1)
                peak_zone_je = min(n - 1, k_peak)
                mask[peak_zone_js:peak_zone_je + 1] = False
                mask[i] = False
                if mask.sum() < 1:
                    mask[:] = True
                    mask[i] = False
                targets = np.where(mask)[0]
                if len(targets) == 0:
                    continue
                # Distribute proportionally to existing values
                target_vals = a_trial[targets]
                total_tv = target_vals.sum()
                if total_tv < 1e-15:
                    a_trial[targets] += reduction / len(targets)
                else:
                    a_trial[targets] += reduction * target_vals / total_tv
                a_trial = np.maximum(a_trial, 0)
                sn = a_trial.sum()
                if sn < 1e-15:
                    continue
                a_trial = a_trial / sn
                sc = compute_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Add small random perturbation to escape
                a = a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            # Try random coordinate
            j_rand = int(rng.integers(0, n))
            if a[j_rand] > 1e-10:
                for factor in [0.97, 1.03]:
                    a_trial = a.copy()
                    a_trial[j_rand] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    sn = a_trial.sum()
                    if sn < 1e-15:
                        continue
                    a_trial = a_trial / sn
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a_trial.copy()
                        improved = True
                        stall = 0
                        break

    return best_a, best_score
