    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""

2. L-BFGS with LogSumExp multi-temperature annealing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n Toeplitz matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    T_rows = []
    for k in range(size):
        # Row k of T: (T @ b)[k] = sum_j a[k-j] * b[j]
        # We want: (T @ b)[k] = conv(a, b)[k] = sum_j a[k-j] * b[j]
        # So T[k, j] = a[k-j] if 0 <= k-j < n else 0
        row = np.zeros(n)
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        T_rows.append(row)
    return np.array(T_rows)


_solvers_tried = False
_solvers_working = []


def get_working_solvers():
    """Find working LP solvers, cached."""
    global _solvers_tried, _solvers_working
    if _solvers_tried:
        return _solvers_working
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solvers_working.append(solver_name)
        except Exception:
            continue
    if not _solvers_working:
        _solvers_working = [cp.ECOS]
    _solvers_tried = True
    return _solvers_working


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] subject to sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers = [solver] if solver else get_working_solvers()
    status = None
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                status = prob.status
                break
        except Exception:
            continue
    
    if status is None:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0)


def compute_score(a):
    """Compute the objective value (assuming sum(a) = 1, a >= 0)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with adaptive perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall_count += 1
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 100 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                rng = np.random.default_rng(42 + it)
                noise = rng.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_total=10000):
    """L-BFGS with LogSumExp multi-temperature annealing."""
    from scipy.optimize import minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [100.0, 40.0, 15.0, 5.0, 2.0, 0.7, 0.3]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_total // n_stages)

    def func_factory(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def jac_factory(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        stage_iters = min(iters_per_stage, max(50, int(remaining * 200)))
        try:
            res = minimize(
                fun=func_factory(stage_T),
                x0=z,
                jac=jac_factory(stage_T),
                method='L-BFGS-B',
                options={'maxiter': stage_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': stage_iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = compute_score(a_check)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={stage_T:.1f} ({elapsed:.0f}s): score={sc:.6f}")

    return softmax_from_z(z)


def exact_subgradient_optimize(a0, time_limit, n_max_iters=500000):
    """
    Exact subgradient method: at each step, identify the peak of conv(a,a),
    compute the exact subgradient (2*a_{k*-j}), and take a projected step.
    Uses adaptive step size with backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    eta = 0.01
    min_eta = 1e-8
    no_improve = 0
    patience = 200

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient: g[j] = 2 * a[k_star - j] for valid indices
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2.0 * a[idx]
        g = g / max(np.linalg.norm(g), 1e-15)

        # Try a few step sizes
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            eta_trial = eta * factor
            a_trial = project_simplex(a - eta_trial * g)
            score_trial = compute_score(a_trial)
            if score_trial < best_score - 1e-14:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                improved = True
                eta = eta_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= patience:
                eta *= 0.7
                no_improve = 0
                if eta < min_eta:
                    break

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.6f}")

    return best_a, best_score


def subgradient_with_near_peaks(a0, time_limit, n_max_iters=200000, n_peaks=3):
    """
    Subgradient method that considers multiple near-peak positions simultaneously.
    The "subgradient" is a weighted average of gradients from the top k_peaks positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    eta = 0.005
    min_eta = 1e-9
    no_improve = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Get top n_peaks positions
        top_k = np.argsort(b)[::-1][:n_peaks]
        peak_val = b[top_k[0]]
        # Weight by proximity to peak
        weights = np.exp(-(b[top_k] - peak_val) * 50)
        weights = weights / weights.sum()

        # Combined gradient
        g = np.zeros(n)
        for w, k in zip(weights, top_k):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    g[j] += w * 2.0 * a[idx]

        # Try step
        improved = False
        for factor in [1.0, 0.5, 0.25]:
            eta_trial = eta * factor
            a_trial = project_simplex(a - eta_trial * g)
            score_trial = compute_score(a_trial)
            if score_trial < best_score - 1e-14:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                improved = True
                eta = eta_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= 100:
                eta *= 0.8
                no_improve = 0
                if eta < min_eta:
                    break

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMulti iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_steps=100000):
    """
    Coordinate descent: try reducing individual coordinates that contribute
    to the peak, and redistribute the mass.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    rng = np.random.default_rng(123)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Find top contributors to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        top_n = min(8, len(idx))
        top_contrib = idx[np.argsort(contribs)[::-1][:top_n]]

        # Also pick a random coordinate for exploration
        random_idx = rng.integers(0, n)
        candidates = list(set(list(top_contrib) + [random_idx]))

        improved = False
        for i in candidates:
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                # Redistribute to all other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve >= 500:
                # Try pair reduction
                for i in top_contrib[:4]:
                    j = k_star - i
                    if j == i or j < 0 or j >= n or a[i] < 1e-14 or a[j] < 1e-14:
                        continue
                    for f1, f2 in [(0.97, 0.97), (0.95, 0.97), (0.97, 0.95)]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        diff = a[i] * (1 - f1) + a[j] * (1 - f2)
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if no_improve >= 3000:
                break
        else:
            no_improve = 0

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        # Asymmetric: mass concentrated on one side
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x)
    elif choice == 4:
        # Double peak
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.5 * np.exp(-20 * (x - 0.7) ** 2)
    else:
        # Linear ramp (completely asymmetric)
        x = np.linspace(0, 1, n)
        a = 1 - x + 0.1
        
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP alternating, L-BFGS, exact subgradient,
    and coordinate descent methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solvers = get_working_solvers()
    solver = solvers[0]
    print(f"Using LP solver: {solver}")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Try multiple initializations and pick the best
        n_init = 960
        best_init = None
        best_init_score = float('inf')
        for trial in range(5):
            a_try = structured_init(n_init, rng)
            sc = compute_score(a_try)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_try
        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: LP-based alternating optimization ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "lp_phase1")

    # === Phase 2: L-BFGS with multi-temperature LogSumExp ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Time budget: {t_alloc:.0f}s, T: 80->0.2")
        T_stages = [80.0, 30.0, 10.0, 4.0, 1.5, 0.7, 0.3, 0.15]
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_total=15000)
        try_update(a_lb, "lbgfs_phase2")

    # === Phase 3: Exact subgradient with near-peak awareness ===
    print(f"\n=== Phase 3: Exact Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_with_near_peaks(best_seq, t_alloc, n_max_iters=500000, n_peaks=3)
        try_update(a_sg, "subgrad_phase3")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 10:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t_alloc, n_steps=200000)
        try_update(a_cd, "coord_phase4")

    # === Phase 5: Second L-BFGS pass (refined) ===
    print(f"\n=== Phase 5: L-BFGS Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"  Time budget: {t_alloc:.0f}s, T: 5->0.1")
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.25, 0.12]
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages, maxiter_total=8000)
        try_update(a_lb2, "lbgfs_phase5")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        alt_lengths = []
        for mult in [1.25, 1.5, 0.75]:
            alt_n = int(n_best * mult)
            if 384 <= alt_n <= 3072 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:2]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_total = min(rem * 0.4, 150.0)
            if t_total < 30:
                break
            print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_seq, alt_n)
            
            t_lp = t_total * 0.5
            a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=20000)
            try_update(a_alt, f"lp_n{alt_n}")

            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
                t_lb = min(rem * 0.3, 60.0)
                if t_lb > 10:
                    a_alt2 = lbgfs_refine(best_seq, t_lb, T_stages=[10.0, 3.0, 1.0, 0.4], maxiter_total=4000)
                    try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Final subgradient polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sg = min(rem * 0.4, 80.0)
        a_fin, _ = exact_subgradient_optimize(best_seq, t_sg, n_max_iters=100000)
        try_update(a_fin, "final_subgrad")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.3, 50.0)
        a_cd2, _ = coordinate_descent(best_seq, t_cd, n_steps=50000)
        try_update(a_cd2, "final_coord")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lb = min(rem * 0.4, 40.0)
        a_lb3 = lbgfs_refine(best_seq, t_lb, T_stages=[2.0, 0.8, 0.3], maxiter_total=3000)
        try_update(a_lb3, "final_lbgfs")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nVerified score: {best_score:.6f}")

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]
