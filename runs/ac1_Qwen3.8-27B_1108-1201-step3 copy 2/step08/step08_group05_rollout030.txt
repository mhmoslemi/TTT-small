```python
LP-based optimization for minimum peak auto-convolution.
Uses full constraint set LP with exact line search and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    M = np.sort(v)[::-1]
    cumsum = np.cumsum(M)
    rho = np.arange(1, n + 1)
    cond = M * rho > (cumsum - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cumsum[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_direction_lp(f, M, top_k=None):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k (or top-k).
    Returns g (normalized to sum 1) or None.
    """
    n = len(f)
    b_size = 2 * n - 1
    
    if top_k is not None and top_k < b_size:
        # Only use top-k constraints
        b = fast_convolve(f)  # Wait, this is f*f not f*g... 
        # Actually we need to use the current f*f to pick which constraints to include
        # But the constraint is on f*g, not f*f
        # For constraint selection, we use the current peaks of f*f as a proxy
        # For full constraints, we include all
        pass
    
    # Build constraint matrix A where A[k, :] is the row for constraint (f*g)[k] <= M
    # (f*g)[k] = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((b_size, n))
    for k in range(b_size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A[k, lo:hi+1] = f[lo:hi+1] reversed? No...
        # (f*g)[k] = sum_i f[i] * g[k-i]
        # For g[j] to appear, we need k-i = j, so i = k-j
        # So coefficient of g[j] in (f*g)[k] is f[k-j] if 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(k, n-1)
        # A[k, j] = f[k-j] for valid j
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    
    g_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(obj, constraints)
    
    solved = False
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g_var.value is None:
        return None
    
    g = np.maximum(g_var.value, 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def line_search(f, h, n_evals=80):
    """
    Find t in [0, 1] that minimizes max(( (1-t)f + t*h ) * ((1-t)f + t*h )).
    Uses grid + parabolic refinement.
    """
    n = len(f)
    
    def peak(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()
    
    # Grid search
    ts = np.linspace(0, 1, n_evals)
    vals = np.array([peak(t) for t in ts])
    
    # Find best grid point
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Parabolic refinement
    if 0 < best_idx < n_evals - 1:
        y0, y1, y2 = vals[best_idx-1], vals[best_idx], vals[best_idx+1]
        denom = (y0 - 2*y1 + y2)
        if abs(denom) > 1e-15:
            delta = 0.5 * (y0 - y2) / denom
            t_refined = best_t + delta * (ts[1] - ts[0])
            if 0 <= t_refined <= 1:
                v_refined = peak(t_refined)
                if v_refined < best_val:
                    best_t = t_refined
                    best_val = v_refined
    
    return best_t

def run_lp_optimization(f_init, time_limit, label=""):
    """Main LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

        elapsed = time.monotonic() - t0

        if remaining < 10:

        b = fast_convolve(f)
        M = b.max()
        
        # Solve direction LP
        h = solve_direction_lp(f, M)
        
        if h is None:
            # Fallback: steepest descent
            peak_idx = b.argmax()
            h = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for j in range(lo, hi + 1):
                h[j] = f[peak_idx - j]
            s = h.sum()
            if s < 1e-10:
                break
            h = h / s
        else:
            if np.allclose(h, f, atol=1e-8):
                stuck += 1
                if stuck > 10:
                    break

        t = line_search(f, h, n_evals=60)
        
        if t < 1e-6:
            stuck += 1
            if stuck > 15:

        # Update
        f_new = (1.0 - t) * f + t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
        
        obj_new = fast_convolve(f_new).max()

            best_obj = obj_new
            best_f = f_new.copy()

            stuck = 0
        else:
            stuck += 1
            if stuck > 20:

            print(f"  {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.8f} t={elapsed:.0f}s")
    
    return best_f, best_obj
```

Wait, let me be more careful. For a given k, the valid j values are from max(0, k-n+1) to min(k, n-1). For each such j, A[k,j] = f[k-j].

So for j = lo, k-j = k-lo. For j = hi, k-j = k-hi.

k-lo = k - max(0, k-n+1)
k-hi = k - min(k, n-1)

If k < n: lo = 0, hi = k. So A[k, 0:k+1] = f[k:k:-1] = [f[k], f[k-1], ..., f[0]].
If k >= n: lo = k-n+1, hi = n-1. So A[k, k-n+1:n] = f[k-(n-1):k-(k-n+1)+1] = f[k-n+1:k-n+1+...] 

Let me just do it element by element in a vectorized way:

```python
def build_conv_matrix(f, n):
```

Now, one concern: the LP might be infeasible if M is too tight. But since M = max(f*f) and f itself is a feasible point for the constraint (f*g)[k] <= M when g = f (wait, no, the constraint is on (f*g)[k], not (f*f)[k]).

Actually, the LP is:
max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k

g = 0 is always feasible, so the LP is always feasible. The question is whether the optimal g has sum > 0, which it does as long as M > 0.

But we need sum(g) > 1 (or rather, we need the direction to be different from f) for progress. If the optimal g is proportional to f, then we're at a local optimum (for this LP relaxation).

OK let me now think about the **key improvement** I want to make over the previous code:

The previous code uses top-k constraints (starting with 40, growing to 200). I'll use ALL constraints (2n-1 = 2303 for n=1152). This should give a better direction because it accounts for all potential peaks, not just the current ones.

The tradeoff is computational: the LP is larger. But for n=1152, it should still be fast enough.

Let me also add a **constraint generation** approach as a fallback: if the full LP is too slow, fall back to top-k.

Another improvement: **multi-start**. Try the optimization from several different starting points (parent, uniform, window functions) and keep the best.

```python
"""
LP-based optimization for minimizing the peak of auto-convolution.
Key improvements over previous:
1. Full constraint set in LP (not just top-k)
2. Exact line search via grid + parabolic refinement
3. Multi-length exploration
4. Adaptive fallback to top-k if full LP is too slow
5. Local polishing phase

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    M = np.sort(v)[::-1]
    cumsum = np.cumsum(M)
    rho = np.arange(1, n + 1)
    cond = M * rho > (cumsum - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cumsum[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix for cross-convolution with f."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        A[k, js] = f[k - js]
    return A

def solve_direction_lp(f, M, use_full=True, top_k=200):
    """Solve direction LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M."""
    n = len(f)
    b_size = 2 * n - 1
    
    if use_full:
        A = build_conv_matrix(f, n)
    else:
        # Only top-k constraints
        b = fast_convolve(f)
        if top_k < b_size:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(b_size)
        A = np.zeros((len(top_idx), n))
        for row_i, k in enumerate(top_idx):
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            js = np.arange(lo, hi + 1)
            A[row_i, js] = f[int(k) - js]
    
    g_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(obj, constraints)
    
    solved = False
    for solver in [cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g_var.value is None:
        return None
    
    g = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g)):
        return None
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def line_search(f, h, n_evals=80):
    """Find best t in [0,1] for (1-t)f + t*h."""
    n = len(f)
    
    def peak(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()
    
    ts = np.linspace(0, 1, n_evals)
    vals = np.array([peak(t) for t in ts])
    
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    
    if 0 < best_idx < n_evals - 1:
        y0, y1, y2 = vals[best_idx-1], vals[best_idx], vals[best_idx+1]
        denom = (y0 - 2*y1 + y2)
        if abs(denom) > 1e-15:
            delta = 0.5 * (y0 - y2) / denom
            t_ref = best_t + delta * (ts[1] - ts[0])
            if 0 <= t_ref <= 1:
                v_ref = peak(t_ref)
                if v_ref < vals[best_idx]:
                    best_t = t_ref
    
    return best_t

def run_lp_optimization(f_init, time_limit, label="", use_full=True, top_k=200):
    """Main LP optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    use_full_current = use_full
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 15:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        h = solve_direction_lp(f, M, use_full=use_full_current, top_k=top_k)
        
        if h is None:
            # Fallback: gradient direction
            peak_idx = b.argmax()
            h = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for j in range(lo, hi + 1):
                h[j] = f[peak_idx - j]
            s = h.sum()
            if s < 1e-10:
                break
            h = h / s
        else:
            if np.max(np.abs(h - f)) < 1e-10:
                stuck += 1
                if stuck > 10:
                    break
                continue
        
        t = line_search(f, h, n_evals=60)
        
        if t < 1e-6:
            stuck += 1
            if stuck > 15:
                if use_full_current:
                    use_full_current = False
                    stuck = 0
                else:
                    break
            continue
        
        f_new = (1.0 - t) * f + t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 25:
                break
        
        if it % 10 == 0:
            print(f"  {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.8f} t={elapsed:.0f}s")
    
    return best_f, best_obj

def local_polish(f_init, time_limit, label=""):
    """Local polishing: try small perturbations to improve the peak."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Try random coordinate perturbations
        improved = False
        n_trials = min(200, int((time_limit - (time.monotonic() - t0)) * 2))
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick 2-5 random coordinates to perturb
            k = rng.integers(2, 6)
            idx = rng.integers(0, n, size=k)
            
            for di in range(k):
                j = idx[di]
                # Try increasing or decreasing f[j]
                for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                    f_trial = f.copy()
                    f_trial[j] = max(0, f_trial[j] + delta)
                    # Renormalize
                    s = f_trial.sum()
                    if s < 1e-10:
                        continue
                    f_trial = f_trial / s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_f, best_obj
```

```python
"""
LP-based and Adam-softmax optimization for minimum peak auto-convolution.
Combines fast subgradient/Adam iterations with high-precision LP direction finding,
exact line search, multi-length exploration, and local polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    M = np.sort(v)[::-1]
    cumsum = np.cumsum(M)
    rho = np.arange(1, n + 1)
    cond = M * rho > (cumsum - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cumsum[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n, indices=None):
    A = np.zeros((len(indices) if indices is not None else 2 * n - 1, n))
    if indices is not None:
        ranges = indices
    else:
        ranges = range(2 * n - 1)
    for row_i, k in enumerate(ranges):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        A[row_i, js] = f[k - js]
    return A


def solve_direction_lp(f, M, constraint_indices=None, n_constraints=None):
    n = len(f)
    b_size = 2 * n - 1

    if constraint_indices is not None:
        idx = constraint_indices
    elif n_constraints is not None and n_constraints < b_size:
        b = fast_convolve(f)
        idx = np.argpartition(b, -n_constraints)[-n_constraints:]
    else:
        idx = None

    A = build_conv_matrix(f, n, idx)

    g_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(obj, constraints)

    solved = False
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            solved = True
            break
        except Exception:
            continue

    if not solved or g_var.value is None:
        return None

    g = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g)) or np.any(np.isinf(g)):
        return None
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def exact_line_search(f, h, ff, fh, hh, n_evals=200):
    ts = np.linspace(0, 1, n_evals)
    best_t = 0.0
    best_val = np.inf
    for t in ts:
        c = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fh + t ** 2 * hh
        v = c.max()
        if v < best_val:
            best_val = v
            best_t = t
    # Golden section refinement
    idx = int(round((best_t / 1.0) * (n_evals - 1)))
    lo = ts[max(0, idx - 1)]
    hi = ts[min(n_evals - 1, idx + 1)]
    if hi - lo < 1e-8:
        return best_t
    gr = (np.sqrt(5) - 1) / 2
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    def peak_at(t):
        c = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fh + t ** 2 * hh
        return c.max()

    fc = peak_at(c_pt)
    fd = peak_at(d_pt)
    for _ in range(30):
        if fc <= fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = peak_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = peak_at(d_pt)
        if hi - lo < 1e-8:
            break
    return (lo + hi) / 2

        m = beta1 * m + (1.0 - beta1) * grad_g
        v = beta2 * v + (1.0 - beta2) * grad_g * grad_g
        m_hat = m / (1.0 - beta1 ** adam_t)
        v_hat = v / (1.0 - beta2 ** adam_t)
        g = g - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 500:
                cur_lr *= 0.97

        if it % 10000 == 0:
            score = 2.0 * n * best_obj
            print(f"  {label} iter={it} obj={best_obj:.10f} score={score:.8f} lr={cur_lr:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def run_lp_optimization(f_init, time_limit, label="", full_constraints=True, max_con

straints=None):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    use_full = full_constraints
    n_cons = max_constraints

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-15:
            f = f / s

        ff = fast_convolve(f)
        M = ff.max()

        h = solve_direction_lp(f, M, n_constraints=None if use_full else n_cons)

        if h is None:
            h = np.zeros(n)
            pk = int(ff.argmax())
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            js = np.arange(lo,

hi + 1)
            h[js] = f[pk - js]
            hs = h.sum()
            if hs < 1e-12:
                break
            h = h / hs
        else:
            if np.max(np.abs(h - f)) < 1e-10:
                stuck += 1
                if stuck > 15:
                    break
                continue

        fh = fast_cross_convolve(f, h)
        hh = fast_convolve(h)
        t = exact_line_search(f, h, ff, fh, hh, n_evals=150)

        if t < 1e-7:
            stuck += 1
            if stuck > 20:
                if use_full and n_cons is None:
                    use_full = False
                    n_cons = 300
                    stuck = 0
                elif stuck > 40:
                    break
            continue

        f_new = (1.0 - t) * f + t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-12:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 5 == 0:
            print(f"  {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.8f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def local_polish(f_init, time_limit, label=""):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(999)
    no_improve = 0
    batch = 0

    while time.monotonic() - t0 < time_limit:
        batch += 1
        improved = False
        n_trials = min(300, max(20, int((time_limit - (time.monotonic() - t0)) * 3)))

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(rng.integers(2, 6))
            idx = rng.integers(0, n, size=k)
            for di in range(k):
                j = int(idx[di])
                for delta in [0.002, 0.005, 0.01, 0.02, 0.04, -0.002, -0.005, -0.01, -0.02, -0.04]:
                    f_trial = f.copy()
                    f_trial[j] = max(0.0, f_trial[j] + delta)
                    ss = f_trial.sum()
                    if ss < 1e-10:
                        continue
                    f_trial = f_trial / ss
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        if batch % 50 == 0:
            print(f"  {label} batch={batch} obj={best_obj:.10f} score={2*n*best_obj:.8f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def uniform_init(n):
    return np.ones(n) / n


def flat_top_window(n):
    x = np.arange(n)
    w = np.ones(n)
    coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(coeffs):
        w -= ai * np.cos(2 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s if s > 1e-10 else uniform_init(n)


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s if s > 1e-10 else uniform_init(n)
```
