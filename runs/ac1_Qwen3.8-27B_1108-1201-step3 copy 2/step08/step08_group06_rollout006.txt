"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with HiGHS solver (primary, long-running)
2. Subgradient mirror descent on the probability simplex
3. Coordinate descent with golden-section line search (polish)
4. L-BFGS in log-space with log-sum-exp smoothing (secondary)
5. Length exploration near the parent

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C_matrix(f):
    """Build constraint matrix C where (f*g)(k) = sum_j C[k,j]*g[j]."""
    n = len(f)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    C = build_C_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, verbose=True, use_avg=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for k in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        g = solve_lp_direction(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Optimal step size for convex optimization: 2/(k+2) or 1/sqrt(k+1)
        t_step = 2.0 / (k + 2)
        
        a_new = geometric_mean_update(f, g, t_step)
        
        if use_avg:
            f_avg = ((k * f_avg) + a_new) / (k + 1)
            score_avg = compute_score(f_avg)
            if score_avg < best_score - 1e-14:
                best_score = score_avg
                best_f = f_avg.copy()
        
        score_cur = compute_score(a_new)
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and (k + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k+1}, M={M:.6e}, score_avg={compute_score(f_avg):.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(a0, time_limit, verbose=True, rng=None):
    """Coordinate descent with golden-section search for optimal transfer."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Select pairs: mix of peak-focused and random
        conv = fast_convolve(a)
        top_idx = int(np.argmax(conv))
        
        n_pairs = min(n, 200)
        pairs = set()
        
        # Peak-focused
        js = max(0, top_idx - n + 1)
        je = min(n - 1, top_idx)
        for _ in range(n_pairs // 2):
            i = js + rng.integers(0, je - js + 1)
            j = js + rng.integers(0, je - js + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Random
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Golden section search
            t_lo = -a[j] + 1e-16
            t_hi = a[i] - 1e-16
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_at(t_val):
                new_val = a[j] + t_val
                if new_val < 0 or a[i] + t_val < 0:  # wait, t is transfer from i to j
                    # Actually: a[i] new = a[i] - t_val, a[j] new = a[j] + t_val
                    # Wait, I defined t as: a[i] -= t, a[j] += t
                    # So a[i]_new = a[i] - t_val, need >= 0 => t_val <= a[i]
                    # a[j]_new = a[j] + t_val, need >= 0 => t_val >= -a[j]
                    return 1e10
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                ss = a_trial.sum()
                a_trial = a_trial / ss
                return compute_score(a_trial)
            
            f1 = eval_at(x1)
            f2 = eval_at(x2)
            
            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_at(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_at(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and iters % 20 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def subgradient_mirror_descent(f_init, time_limit, verbose=True, rng=None):
    """Subgradient mirror descent on the simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = f_init.copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    z = np.log(np.maximum(a, 1e-30))
    eta0 = 0.1
    
    for k in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        eta = eta0 / np.sqrt(k + 1)
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of max_k (f*f)(k) at k_star
        # d/df[i] (f*f)(k_star) = 2*f[k_star - i] if 0 <= k_star - i < n, else 0
        grad = np.zeros(n)
        # For position i, the derivative is 2*f[k_star - i] if valid
        # i ranges over [max(0, k_star-n+1), min(k_star, n-1)]
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if j != i:
                    grad[i] = 2.0 * a[j]
                else:
                    grad[i] = 2.0 * a[i]
        
        # Project gradient onto the tangent space of the simplex
        S = np.dot(grad, a)
        grad_proj = a * (grad - S)
        
        # Mirror descent update in log-space
        z -= eta * grad_proj
        z -= z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        z = np.log(np.maximum(a, 1e-30))
        
        if verbose and (k + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={k+1}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score

        c_max = conv.max()

        # Actually, let me think more carefully.
        # We want h[i] = sum_j a[j] * w[i+j] for i in [0, n-1]
        # Let w_ext = [0, ..., 0, w[0], w[1], ..., w[2n-2], 0, ..., 0] (padded)
        # Then h[i] = (a * w_ext)(n-1+i) ... hmm this is getting complicated.
        # Let me just compute it directly for now (O(n^2) but n is small)
        
        # Actually, let's use a different approach. The gradient of the smoothed objective
        # w.r.t. a is:
        # grad_a[i] = 2n * sum_k w[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = 2*a[k-i] if 0<=k-i<n

        # = 4n * sum_{j=0}^{n-1} w[i+j] * a[j]
        # (where i+j must be in [0, 2n-2])

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with HiGHS solver (primary, long-running)
2. Subgradient mirror descent on the probability simplex
3. Coordinate descent with golden-section line search (polish)
4. Length exploration near the parent
Key insight: The LP direction at each step is the "best response" in a
zero-sum game, and geometric mean interpolation in log-space provides
stable, monotone improvement of the max-convolution peak.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C_matrix(f):
    """Build constraint matrix where (f*g)(k) = sum_j C[k,j]*g[j]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    C = build_C_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0)
    except Exception:
        return None


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, verbose=True, use_avg=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    f = f / f.sum()
    n = len(f)
    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    for k in range(20000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_f = fast_convolve(f)
        M = conv_f.max()

        g = solve_lp_direction(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Step size: optimal 2/(k+2) for convex problems
        t_step = 2.0 / (k + 2)
        if k > 10000:
            t_step = max(0.005, 1.0 / np.sqrt(k + 1))

        a_new = geometric_mean_update(f, g, t_step)

        if use_avg:
            f_avg = ((k * f_avg) + a_new) / (k + 1)
            score_avg = compute_score(f_avg)
            if score_avg < best_score - 1e-14:
                best_score = score_avg
                best_f = f_avg.copy()

        score_cur = compute_score(a_new)
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()

        f = a_new

        if verbose and (k + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k+1}, M={M:.6e}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_f, best_score


def cd_eval(a, i, j, t_val, n):
    """Evaluate score after transferring t_val from position i to position j."""
    new_i = a[i] - t_val
    new_j = a[j] + t_val
    if new_i < -1e-15 or new_j < -1e-15:
        return 1e10
    a_trial = a.copy()
    a_trial[i] = max(0.0, new_i)
    a_trial[j] = max(0.0, new_j)
    ss = a_trial.sum()
    if ss < 1e-15:
        return 1e10
    a_trial = a_trial / ss
    return compute_score(a_trial)


def coordinate_descent(a0, time_limit, verbose=True, rng=None):
    """Coordinate descent with golden-section search."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        conv = fast_convolve(a)
        top_idx = int(np.argmax(conv))
        sec_idx = int(np.argmax(np.where(conv == conv.max(), 0, conv)))

        n_pairs = min(n, 300)
        pairs = []

        # Peak-focused pairs for top peak
        js = max(0, top_idx - n + 1)
        je = min(n - 1, top_idx)
        for _ in range(n_pairs // 3):
            i = js + rng.integers(0, je - js + 1)
            j = js + rng.integers(0, je - js + 1)
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        # Peak-focused pairs for second peak
        js2 = max(0, sec_idx - n + 1)
        je2 = min(n - 1, sec_idx)
        for _ in range(n_pairs // 3):
            i = js2 + rng.integers(0, je2 - js2 + 1)
            j = js2 + rng.integers(0, je2 - js2 + 1)
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        # Random pairs
        for _ in range(n_pairs // 3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue

            t_lo = -a[j] + 1e-16
            t_hi = a[i] - 1e-16
            if t_hi <= t_lo:
                continue

            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            f1 = cd_eval(a, i, j, x1, n)
            f2 = cd_eval(a, i, j, x2, n)

            for _ in range(22):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = cd_eval(a, i, j, x1, n)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = cd_eval(a, i, j, x2, n)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 8:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and iters % 10 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")

    return best_a, best_score


def subgradient_mirror_descent(f_init, time_limit, verbose=True, rng=None):
    """Subgradient mirror descent on the probability simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = f_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    z = np.log(np.maximum(a, 1e-30))

    for k in range(20000000):
        if time.monotonic() - t0 > time_limit:
            break

        eta = 0.1 / np.sqrt(k + 1)
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Subgradient: d/da[i] (a*a)(k_star) = 2*a[k_star-i] if 0<=k_star-i<n
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Project to tangent space of simplex
        S = np.dot(grad, a)
        grad_proj = a * (grad - S)

        z -= eta * grad_proj
        z -= z.max()
        a_new = np.exp(z)
        ss = a_new.sum()
        if ss < 1e-30:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        z = np.log(np.maximum(a, 1e-30))

        if verbose and (k + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={k+1}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score):
