import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def prefix_suffix_max(arr):
    """Compute prefix max and suffix max arrays."""
    n = len(arr)
    pre = np.zeros(n)
    suf = np.zeros(n)
    pre[0] = arr[0]
    for i in range(1, n):
        pre[i] = max(pre[i-1], arr[i])
    suf[n-1] = arr[n-1]
    for i in range(n-2, -1, -1):
        suf[i] = max(suf[i+1], arr[i])
    return pre, suf


def unaffected_max(pre, suf, l, r, size):
    """Max of arr outside range [l, r] inclusive, using pre/suf arrays."""
    if l > 0:
        left = pre[l-1]
    else:
        left = -1e30
    if r < size - 1:
        right = suf[r+1]
    else:
        right = -1e30
    return max(left, right)


def exact_minimax_step(c, affected_start, affected_end, a, i, U, n, n_iters=50):
    """
    Find the optimal delta for coordinate i using binary search on subgradient.
    c: current convolution
    affected_start, affected_end: range of affected k indices
    a: current sequence (normalized, sum=1)
    i: coordinate to update
    U: unaffected max
    n: length of sequence
    Returns optimal delta.
    """
    # Affected indices and their coefficients
    k_vals = np.arange(affected_start, affected_end + 1)
    coeffs = a[k_vals - i]  # a_{k-i} for each affected k

    # The quadratic term at k=2i
    has_quad = (2 * i >= affected_start) and (2 * i <= affected_end)
    quad_idx = 2 * i - affected_start if has_quad else -1

    # Domain: delta in [-a[i], a[i] * 10] (allow some increase)
    delta_lo = -a[i]
    delta_hi = a[i] * 5.0

    # We want to minimize F(delta) = max(U, max_k (c_k + 2*delta*coeff_k + [k==2i]*delta^2))
    # The subgradient contains 0 at the optimum (if interior).

    def compute_F_and_subgrad(delta):
        vals = c[affected_start:affected_end+1].astype(np.float64) + 2.0 * delta * coeffs
        if has_quad:
            vals[quad_idx] = c[2*i] + 2.0 * delta * a[i] + delta * delta
        m = vals.max()
        F_val = max(U, m)
        # Subgradient: if F_val == U and U > m, subgrad = 0
        # Otherwise subgrad is in {coeff_{k*} + [k*==2i]*delta : k* is argmax of vals}
        if U > m + 1e-18:
            return F_val, 0.0, None
        # Find argmax positions
        close_to_max = np.where(vals >= m - 1e-14)[0]
        if len(close_to_max) == 1:
            idx = close_to_max[0]
            sg = coeffs[idx]
            if idx == quad_idx:
                sg += delta
            return F_val, sg, int(k_vals[idx])
        else:
            # Multiple argmax: subgradient is interval [min, max] of possible values
            sgs = coeffs[close_to_max].copy()
            mask = (close_to_max == quad_idx)
            sgs[mask] += delta
            return F_val, sgs, None  # Return array of subgradients

    # Binary search for where subgradient contains 0
    # F is convex, so subgradient is monotonically non-decreasing
    best_delta = 0.0
    best_F = compute_F_and_subgrad(0.0)[0]

    lo = delta_lo
    hi = delta_hi

    for _ in range(n_iters):
        mid = (lo + hi) / 2.0
        F_mid, sg, _ = compute_F_and_subgrad(mid)

        if isinstance(sg, np.ndarray):
            # Subgradient is an interval [sg.min(), sg.max()]
            if sg.min() > 0:
                hi = mid
            elif sg.max() < 0:
                lo = mid
            else:
                # 0 is in the subgradient -> optimum found
                best_delta = mid
                best_F = F_mid
                break
        else:
            if sg > 0:
                hi = mid
            elif sg < 0:
                lo = mid
            else:
                best_delta = mid
                best_F = F_mid
                break

        if F_mid < best_F:
            best_F = F_mid
            best_delta = mid

    return best_delta

            # Actually it's easier to just recompute:
            a_shifted_old = np.zeros(ke - ks + 1)
            for idx in range(ke - ks + 1):
                k = ks + idx
                j = k - i
                if 0 <= j < n and j != i:
                    a_shifted_old[idx] = 2.0 * delta * a[j]
                elif 0 <= j < n and j == i:
                    a_shifted_old[idx] = 2.0 * delta * old_ai + delta * delta

            conv[ks:ke+1] += a_shifted_old
            M = max(U, float(conv[ks:ke+1].max()))

            cur_score = 2.0 * n * M / (sum_a * sum_a)
            if cur_score < best_score - 1e-15:

            print(f"    MCD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.8f}")

    # Final recompute to ensure correctness

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_global
    if '_lp_solver_global' in globals():
        try:
            if _lp_solver_global is not None:
                return _lp_solver_global
        except Exception:
            pass
    _lp_solver_global = None
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP, cp.CBC, cp.CVXOPT]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_global = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_global = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, timeout=30)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, timeout=30)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_descent(a0, time_limit, verbose=False, label=""):
    """
    Subgradient descent with adaptive step size for the non-convex problem.
    Uses the max of top-k peaks for a smoothed subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.02
    no_improve = 0
    top_k = 10

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.95
            no_improve = 0
            if lr < 1e-7:
                break

        # Compute weighted subgradient from top-k peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.02 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            # grad at coordinate j: a[k-j] + [k==2j]*a_j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += w[idx] * a[k - j]
                if k == 2 * j:
                    grad[j] += w[idx] * a[j]

        # Step
        a_new = a - lr * grad
        a = simplex_project(a_new)

        lr *= 0.9995

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_tapered(n, rng):
    """Initialize with a tapered (Bell-shaped) sequence."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_linear(n, rng):
    """Initialize with a linearly varying sequence."""
    t = np.arange(n, dtype=np.float64)
    if rng.random() < 0.5:
        a = t + 1
    else:
        a = (n - 1 - t) + 1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_noise(n, base, rng, sigma=0.15):
    """Add Gaussian noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def perturb_escape(a0, rng, magnitude=0.05):
    """Perturb to escape local optimum."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s
