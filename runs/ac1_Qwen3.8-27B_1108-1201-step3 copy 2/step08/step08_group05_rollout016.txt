"""
Hybrid Nesterov-accelerated gradient (smooth max) + LP refinement for
minimizing the auto-convolution peak score. Uses FFT-based gradient
computation, adaptive temperature, multi-length exploration, and
constraint-generation LP for final polish.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_gradient_fft(p, a):
    """Compute grad[j] = sum_k p[k] * a[k-j] via FFT cross-correlation."""
    n = len(a)
    L = len(p) + n - 1
    fft_size = 1 << (L - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    R = irfft(P * np.conj(A), fft_size)
    return R[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_objective(a, tau):
    """Compute smooth max of auto-convolution and its gradient."""
    b = fast_convolve(a)
    tau_b = tau * b
    tau_b -= tau_b.max()
    exp_vals = np.exp(tau_b)
    p = exp_vals / exp_vals.sum()
    # Gradient: sum_k p[k] * a[k-j] (factor of 2 absorbed into lr)
    grad = compute_gradient_fft(p, a)
    obj_val = b.max()
    return obj_val, p, grad, b


def nesterov_gradient_opt(f_init, time_limit, tau_init=5.0, tau_max=2000.0,
                           lr_init=0.02, mu=0.85, batch_check=200,
                           lp_interval=800, top_k_lp=60, label="nest"):
    """
    Nesterov-accelerated gradient descent on smooth max of auto-convolution.
    Periodically runs LP refinement.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = project_simplex(f)
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    lp_count = 0

    while time.monotonic() - t0 < time_limit:
        # Inner batch of gradient steps
        batch = min(batch_check, max(50, int((time_limit - (time.monotonic() - t0)) * 1.5)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            # Compute gradient
            obj_val, p, grad, b = smooth_max_objective(f, tau)

            # Adaptive: if tau is large, the gradient is concentrated on top peaks
            # Scale gradient so its max is around 1 for stable steps
            g_max = np.abs(grad).max()
            if g_max > 1e-15:
                grad = grad / g_max

            # Nesterov momentum
            v = mu * v - lr * grad
            f_new = project_simplex(f + v)

            # Evaluate
            b_new = fast_convolve(f_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Backtracking if stuck
            if no_improve > 30:
                lr *= 0.8
                v *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    # Restart with perturbation
                    rng = np.random.default_rng(it)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    v = np.zeros(n)
                    lr = lr_init * 0.5
                    no_improve = 0

            f = f_new

        # Periodic parameter updates
        if it % 100 == 0:
            tau = min(tau * 1.08, tau_max)
            lr = max(lr * 0.995, 1e-6)

        # Periodic LP refinement
        lp_count += 1
        if lp_count >= lp_interval or (no_improve > 50 and time.monotonic() - t0 < time_limit * 0.7):
            rem = time_limit - (time.monotonic() - t0)
            if rem > 20:
                t_lp = min(rem * 0.15, 60.0)
                f_lp, obj_lp = lp_refine(best_f, time_limit=t_lp, top_k=top_k_lp)
                if obj_lp < best_obj - 1e-15:
                    best_obj = obj_lp
                    best_f = f_lp.copy()
                    f = f_lp.copy()
                    v = np.zeros(n)
                    lr = lr_init * 0.3
                    no_improve = 0
                    print(f"    {label} iter={it} LP improved: obj={best_obj:.12f} "
                          f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
            lp_count = 0
            top_k_lp = min(top_k_lp + 20, 200)

        if it % 500 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} obj={best_obj:.12f} score={score:.6f} "
                  f"tau={tau:.1f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def lp_refine(f, time_limit, top_k=60):
    """Quick LP refinement with constraint generation."""
    t0 = time.monotonic()
    n = len(f)
    b = fast_convolve(f)
    M = b.max()

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    active_list = top_idx.tolist()
    active_set = set(active_list)

    g = None
    for cg_iter in range(10):
        if time.monotonic() - t0 > time_limit:
            break

        K = len(active_list)
        if K > 250:
            b_active = b[active_list]
            keep = np.argsort(b_active)[-250:]
            active_list = [active_list[i] for i in keep]
            active_set = set(active_list)
            K = len(active_list)

        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = active_list[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )

        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=200000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            if g is not None:
                break
            return f, M

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)) or np.any(np.isinf(g)):
            break

        # Check violations
        if time.monotonic() - t0 > time_limit * 0.7:
            break

        fg = fast_cross_convolve(f, g)
        mask = np.zeros(len(b), dtype=bool)
        mask[active_list] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]

        if len(violations) == 0:
            break

        worst = np.argsort(fg[violations])[-min(3, len(violations)):]
        for vi in worst:
            v = int(vi)
            if v not in active_set:
                active_set.add(v)
                active_list.append(v)

    if g is None or g.sum() <= 1.0 + 1e-5:
        return f, M

    h = g / g.sum()

    # Line search
    best_t = 0.1
    best_trial = M
    for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_trial:
            best_trial = obj
            best_t = t

    f_new = (1.0 - best_t) * f + best_t * h
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    obj_new = fast_convolve(f_new).max()

    if obj_new < M:
        return f_new, obj_new
    return f, M


def coordinate_descent(f, time_limit, max_rounds=10000):
    """
    Coordinate descent: for each coordinate, optimize its value
    given others fixed (1D minimization of the peak).
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    rng = np.random.default_rng(123)

    order = list(range(n))

    rounds = 0
    while time.monotonic() - t0 < time_limit and rounds < max_rounds:
        rounds += 1
        improved = False
        rng.shuffle(order)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try perturbing coordinate j
            base = f[j]
            # Try a few values
            candidates = [base * 0.5, base * 0.8, base * 1.2, base * 1.5, 0.0]
            # Also try shifting from coordinate j to neighbor
            if j > 0:
                candidates.append(base + f[j-1] * 0.3)
            if j < n - 1:
                candidates.append(base + f[j+1] * 0.3)

            best_val = base
            best_trial = best_obj
            for c in candidates:
                c = max(0.0, c)
                f[j] = c
                s = f.sum()
                if s < 1e-10:
                    f[j] = base
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_trial - 1e-15:
                    best_trial = obj
                    best_val = c
            if best_val != base:
                f[j] = best_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_trial < best_obj - 1e-15:
                    best_obj = best_trial
                    best_f = f.copy()
                    improved = True
            else:
                f[j] = base

        if not improved and rounds > 10:
            break

    return best_f, best_obj


def flat_top_window(n):
    """Normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def blackman_window(n):
    """Normalized Blackman window."""
    w = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)) \
        + 0.08 * np.cos(4 * np.pi * np.arange(n) / max(1, n - 1))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def kaiser_window(n, beta=6.0):
    """Normalized Kaiser window."""
    from scipy.special import i0
    r = np.sqrt(1.0 - (2.0 * np.arange(n) / max(1, n - 1) - 1.0) ** 2)
    w = i0(beta * r) / i0(beta)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length and normalize."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s


def peak_shaving(f, time_limit, n_peaks=3):
    """
    Iteratively reduce the top peaks by finding the best
    1D direction that reduces them.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        b = fast_convolve(f)
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Compute gradient from top peaks
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max

        # Try step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.01, 0.02, 0.05, 0.1]:
            f_trial = project_simplex(f - alpha * grad)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            no_improve += 1
            # Try adding noise to escape
            if no_improve > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue

        f = project_simplex(f - best_alpha * grad)
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0

    return best_f, best_obj

            print(f"  NEW BEST [n{target_n}]: {best_score:.8f}")
        else:
            print(f"  n{target_n}: best={best_score_len:.8f} (no improvement over {best_score:.8f})")

    # === Phase 5: Peak shaving on best ===
    print(f"\n=== Phase 5: Peak shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 120.0)
    if t5 > 15:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = peak_shaving(best_seq, time_limit=t5, n_peaks=5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [peak_shave]: {score5:.8f}")
        else:
            print(f"  peak_shave: {score5:.8f} (no improvement)")

    # === Phase 6: LP deep refinement ===
    print(f"\n=== Phase 6: LP deep (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 180.0)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s")
        # Run LP multiple times with increasing constraints
        f6 = best_seq.copy()
        obj6 = fast_convolve(f6).max()
        for lp_round in range(3):
            rem_inner = t6 / (3 - lp_round)
            f6, obj6 = lp_refine(f6, time_limit=min(rem_inner, t6 - (time.monotonic() - t0 + t6)),
                                  top_k=80 + lp_round * 40)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_deep]: {score6:.8f}")
        else:
            print(f"  lp_deep: {score6:.8f} (no improvement)")

    # === Phase 7: Final Nesterov polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 200.0)
    if t7 > 15:
        print(f"  Budget: {t7:.0f}s")
        f7, obj7 = nesterov_gradient_opt(best_seq, time_limit=t7, tau_init=100.0,
                                          tau_max=5000.0, lr_init=0.005, mu=0.95,
                                          batch_check=100, lp_interval=5000,
                                          top_k_lp=150, label="polish")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [polish]: {score7:.8f}")

    # === Phase 8: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.4, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final LP (t={t8:.0f}s) ===")
            f8, obj8 = lp_refine(best_seq, time_limit=t8, top_k=200)
            score8 = float(evaluate_sequence(f8.tolist()))
            if np.isfinite(score8) and score8 < best_score:
                best_score = score8
                best_seq = f8.copy()
                print(f"  NEW BEST [final_lp]: {score8:.8f}")

    # === Phase 9: Final peak shave ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t9 = min(rem * 0.5, 50.0)
        if t9 > 5:
            f9, obj9 = peak_shaving(best_seq, time_limit=t9, n_peaks=10)
            score9 = float(evaluate_sequence(f9.tolist()))
            if np.isfinite(score9) and score9 < best_score:
                best_score = score9
                best_seq = f9.copy()
                print(f"  NEW BEST [final_shave]: {score9:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Improvement: {1.5027 - best_score:.6f}" if best_score < 1.5027
          else f"Gap to target: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
