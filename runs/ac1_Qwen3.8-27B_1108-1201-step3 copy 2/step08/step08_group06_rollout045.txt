1. Adaptive-temperature smoothed gradient descent in log-space with Nesterov momentum
2. Multiple diverse initializations and random restarts
3. Active-set refinement focusing on near-peak lags
4. Fine coordinate descent with golden-section line search
5. Length exploration via interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_from_conv(a, conv):
    """Compute score given precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def compute_score(a):
    """Compute the optimization score."""
    conv = fast_convolve(a)
    return compute_score_from_conv(a, conv)


def logsumexp(x, T):
    """Compute T * log(sum(exp(x/T))) stably."""
    x_max = x.max()
    if x_max > 1e100:
        return x_max
    shifted = x / T - x_max / T
    return T * (x_max / T + np.log(np.sum(np.exp(shifted))))


def smooth_max_and_weights(conv, T):
    """Compute smoothed max and softmax weights of convolution values."""
    conv_max = conv.max()
    shifted = (conv - conv_max) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    smooth_max = conv_max + T * np.log(exp_shifted.sum())
    return smooth_max, w


def grad_smooth_max(conv, w, a, n):
    """Compute gradient of T*log(sum(exp(c(k)/T))) w.r.t. a[i]."""
    # d/dc(k) * dc(k)/da[i]
    # sum_k w_k * dc(k)/da[i]
    # dc(k)/da[i] = a[k-i] + a[i] if 0 <= k-i < n (with special case k=2i: 2*a[i])
    # Actually: c(k) = sum_j a[j]*a[k-j]
    # dc(k)/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i term)
    # Both valid iff 0 <= k-i < n
    # If k-i == i (k=2i), it's 2*a[i]
    
    # Efficient computation via cross-correlation:
    # sum_k w_k * a[k-i] * [0<=k-i<n] = (w *' a)[i] where *' is cross-correlation
    # Actually: for fixed i, sum_k w[k] * a[k-i] over valid k
    # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
    # sum_{k=i}^{i+n-1} w[k] * a[k-i] = sum_j w[i+j] * a[j] for j=0..n-1
    # This is the cross-correlation of w and a at lag i
    
    # Using FFT: cross-correlation of w and a
    # corr[i] = sum_j w[i+j] * a[j] = sum_j w[j+i] * a[j]
    # This is (w reversed) * a at position ... let me think differently
    
    # Actually, let's compute it directly or via FFT
    # Define: for each i, grad_component[i] = sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # = sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
    
    # This is the cross-correlation of w and a:
    # xcorr[i] = sum_j w[i+j] * a[j]
    # = (reverse(w) * reverse(a))[?] ... let me just use a direct FFT approach
    
    # Alternative: note that sum_k w[k] * a[k-i] [0<=k-i<n] 
    # Let m = k-i, then k = m+i, m ranges from 0 to n-1, and we need m+i < 2n-1
    # = sum_{m=0}^{n-1} w[m+i] * a[m] (with w indexed up to 2n-2)
    
    # This is the correlation of w and a: corr[i] = sum_m a[m] * w[m+i]
    # In terms of convolution: if we define w_pad as w zero-padded to length 2n-1,
    # then corr = ifft(fft(a) * conj(fft(w_pad))) ... 
    
    # Simpler: use the fact that this is a "sliding dot product"
    # corr[i] = dot(a, w[i:i+n])
    
    # FFT approach: 
    # corr[i] = sum_j a[j] * w[i+j]
    # Let's define v = w (length 2n-1), pad to fft_size
    # corr = ifft(conj(fft(a_padded)) * fft(v_padded))[0:n] ... 
    
    # Actually the cleanest way:
    # corr[i] = sum_j a[j] * w[i+j]
    # = (a * w_pad_reversed)[n-1-i] ... this is getting confusing
    
    # Let me just use a straightforward approach with the structure of the problem.
    # 
    # Total gradient: grad[i] = 2 * (sum_k w[k] * (a[k-i] + a[i]) * [0<=k-2i+... ])
    # 
    # Wait, let me redo this more carefully.
    # 
    # c(k) = sum_j a[j] * a[k-j] where j ranges over valid indices (0<=j<n, 0<=k-j<n)
    # 
    # dc(k)/da[i]:
    # - Term where j=i: a[k-i] (valid if 0<=k-i<n)
    # - Term where k-j=i, i.e., j=k-i: a[k-i] (valid if 0<=k-i<n)
    # 
    # Wait, the second term: j=k-i, so a[j]*a[k-j] = a[k-i]*a[i]. The derivative w.r.t. a[i] is a[k-i].
    # 
    # But if i = k-i (i.e., k=2i), then both terms give a[i], so total is 2*a[i].
    # 
    # So dc(k)/da[i] = a[k-i] * [0<=k-i<n] + a[i] * [0<=k-i<n]
    #                  = (a[k-i] + a[i]) * [0<=k-i<n]
    # 
    # (Note: when k=2i, this gives 2*a[i], which is correct since the term a[i]*a[i] appears once in the sum but has derivative 2*a[i].)
    
    # Hmm wait, that's not right either. Let me be very careful.
    # c(k) = sum_j a[j] * a[k-j]
    # The terms involving a[i] are:
    # - j = i: the term is a[i] * a[k-i], derivative w.r.t. a[i] is a[k-i]
    # - j = k-i: the term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i]
    # 
    # But these are the SAME term when j = i and j = k-i, which happens when i = k-i, i.e., k = 2i.
    # In that case, the term in the sum is a[i]*a[i] = a[i]^2, and its derivative is 2*a[i].
    # 
    # So: dc(k)/da[i] = 2*a[k-i] if k=2i (and 0<=k-i<n, which is 0<=i<n, always true)
    #                   = a[k-i] + a[i] if k≠2i and 0<=k-i<n  ... 
    
    # Wait no. Let me just index carefully.
    # c(k) = sum_j a[j] * a[k-j]
    # 
    # ∂c(k)/∂a[i] = sum_j (∂a[j]/∂a[i]) * a[k-j] + a[j] * (∂a[k-j]/∂a[i])
    # = a[k-i] (from j=i) + sum_j a[j] * (∂a[k-j]/∂a[i])
    # = a[k-i] + a[k-i] (from j such that k-j=i, i.e., j=k-i)
    # = 2*a[k-i] (if 0<=k-i<n)
    # 
    # Wait that gives 2*a[k-i] always? That doesn't seem right.
    
    # Let me try a specific example. a = [a0, a1, a2, a3], k=2.
    # c(2) = a0*a2 + a1*a1 + a2*a0 = 2*a0*a2 + a1^2
    # ∂c(2)/∂a0 = 2*a2
    # ∂c(2)/∂a1 = 2*a1
    # ∂c(2)/∂a2 = 2*a0
    # ∂c(2)/∂a3 = 0
    # 
    # Using my formula: for i=0, k-i=2, 0<=2<4, so ∂c(2)/∂a[0] = 2*a[2] ✓
    # For i=1, k-i=1, 0<=1<4, so ∂c(2)/∂a[1] = 2*a[1] ✓
    # For i=2, k-i=0, 0<=0<4, so ∂c(2)/∂a[2] = 2*a[0] ✓
    # For i=3, k-i=-1, not valid, so ∂c(2)/∂a[3] = 0 ✓
    
    # So the formula is: ∂c(k)/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    
    # Great! So the gradient of the smoothed max is:
    # grad[i] = 2 * sum_k w[k] * a[k-i] * [0<=k-i<n]
    
    # This is 2 * (w cross-correlate a)[i] where the cross-correlation is:
    # (w ⋆ a)[i] = sum_k w[k] * a[k-i] * [0<=k-i<n]
    #             = sum_{j=0}^{n-1} w[i+j] * a[j]
    
    # This is a cross-correlation that can be computed via FFT.
    
    # Let's compute it via FFT:
    # Define: pad a to length fft_size, pad w to length fft_size
    # cross[i] = sum_j w[i+j] * a[j] for j=0..n-1 (with w[i+j] = 0 if i+j >= 2n-1)
    
    # This is equivalent to: cross = ifft(fft(w_padded) * conj(fft(a_padded))) ... 
    # Hmm, let me think about this differently.
    
    # cross[i] = sum_j a[j] * w[i+j]
    # Let's reverse w: w_rev[m] = w[2n-2-m]
    # Then cross[i] = sum_j a[j] * w[i+j] = sum_j a[j] * w_rev[2n-2-i-j]
    # = (a * w_rev)[2n-2-i] (convolution)
    
    # So cross = convolve(a, w_rev)[::-1][0:n] or something like that.
    
    # Actually, let me just use a direct approach for simplicity and correctness:
    # For each i, cross[i] = dot(a, w[i:i+n])
    # This is O(n^2) total, which might be slow for large n.
    
    # FFT approach:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Pad w to length 2n-1 (it already is), then:
    # Define u = a (length n), v = w (length 2n-1)
    # cross[i] = sum_j u[j] * v[i+j]
    # This is the "valid" cross-correlation of u and v... 
    
    # Actually, the cleanest FFT-based computation:
    # cross[i] = sum_j u[j] * v[i+j]
    # = (u * v_flipped)[?] where v_flipped is v reversed
    
    # Let me just use the matrix approach for the gradient, since we need it to be fast.
    # Actually, for n up to a few thousand, O(n^2) might be too slow. Let's use FFT.
    
    # The key insight: cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the same as: (flip(a) * w)[2n-2-i] ... let me verify.
    # 
    # Let A = flip(a) = [a[n-1], a[n-2], ..., a[0]] (length n)
    # (A * w)[k] = sum_j A[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let m = n-1-j, j = n-1-m:
    # = sum_m a[m] * w[k-(n-1-m)] = sum_m a[m] * w[k-n+1+m]
    # We want this to equal cross[i] = sum_m a[m] * w[i+m]
    # So k-n+1 = i, i.e., k = i + n - 1
    # (A * w)[i+n-1] = cross[i] ✓
    
    # So: cross = convolve(flip(a), w)[n-1 : 2n-2] (indices n-1 to 2n-2)
    # Which has length n.
    
    n_fft = 1 << (3 * n - 2).bit_length()  # enough for conv of length n and 2n-1
    A_flip = rfft(a[::-1], n_fft)
    W = rfft(w, n_fft)
    conv_result = irfft(A_flip * W, n_fft)
    cross = conv_result[n-1:2*n-1]  # length n
    
    return 2.0 * cross


def log_softmax(z):
    """Stable softmax from logits."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def gradient_descent_smoothed(a0, time_limit, T_start=10.0, T_end=0.1, n_iters=None, verbose=True):
    """
    Gradient descent on the smoothed objective in log-space.
    Uses Nesterov momentum and adaptive step size.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    z = z - z.sum() / n  # centering for numerical stability
    
    T = T_start
    T_ratio = (T_end / T_start) ** 0.01  # geometric decay per 0.01 * total iters
    
    # Actually, let's use a simpler schedule
    if n_iters is None:
        n_iters = 5000
    
    best_a = a0.copy()
    s0 = a0.sum()
    best_score = compute_score(a0)
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    lr = 0.1  # learning rate
    
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = it / n_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Current a from z
        a = log_softmax(z)
        
        # Compute gradient of smoothed objective w.r.t. a
        conv = fast_convolve(a)
        smooth_max, w = smooth_max_and_weights(conv, T)
        grad_a = grad_smooth_max(conv, w, a, n)
        
        # Convert to log-space gradient: grad_z = a * (grad_a - a @ grad_a)
        S = np.dot(a, grad_a)
        grad_z = a * (grad_a - S)
        
        # Nesterov update
        v = mu * v + grad_z
        z = z - lr * v
        z = z - z.mean()  # keep centered
        
        # Check score periodically
        if it % 10 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and it % 500 == 0:
            print(f"    GD iter={it}, T={T:.4f}, score={best_score:.8f}")
    
    # Final check
    a = log_softmax(z)
    score = compute_score(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def multiplicative_update(a, conv, T, n_iter=100, verbose=False):
    """
    Multiplicative (Weiszfeld-type) update for minimizing max conv.
    Update: a_new[i] = a[i] / (1 + eta * pressure[i])
    where pressure[i] = sum_k w[k] * a[k-i] * [0<=k-i<n]
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        smooth_max, w = smooth_max_and_weights(conv, T)
        
        # Compute pressure: pressure[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
        # Same as cross-correlation computed in grad_smooth_max
        pressure = grad_smooth_max(conv, w, a, n) / 2.0  # divide by 2 since grad = 2*pressure
        
        # Normalize pressure
        p_mean = pressure.mean()
        if p_mean < 1e-15:
            break
        
        # Multiplicative update
        eta = 0.1
        a = a / (1.0 + eta * pressure / p_mean)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def active_set_refinement(a0, time_limit, top_k=20, verbose=True):
    """
    Refine by focusing on the top-k peak lags.
    Uses coordinate descent restricted to pairs that most affect the top peaks.
    """
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top k peaks
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Contributing pairs for this peak at lag pk:
            # c(pk) = sum_j a[j] * a[pk-j]
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            
            # For each contributing position, try transferring weight to a position
            # that does NOT contribute to this peak (or contributes less)
            idx_range = np.arange(js, je + 1)
            contribs = a[idx_range] * a[int(pk) - idx_range]
            
            # Get the top contributing positions
            for _ in range(3):
                if time.monotonic() - t0 > time_limit:
                    break
                ci = np.argmax(contribs)
                i_idx = idx_range[ci]
                
                if a[i_idx] < 1e-14:
                    contribs[ci] = -1
                    continue
                
                # Find a position to transfer to: one that doesn't contribute to any top peak
                # Simple heuristic: transfer to a random position
                for attempt in range(5):
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        continue
                    
                    # Golden section search for transfer amount
                    delta_max = a[i_idx] * 0.5
                    if delta_max < 1e-15:
                        break
                    
                    gr = (np.sqrt(5) - 1) / 2
                    lo = 1e-15
                    hi = delta_max
                    x1 = hi - gr * (hi - lo)
                    x2 = lo + gr * (hi - lo)
                    
                    def eval_delta(d):
                        a_trial = a.copy()
                        a_trial[i_idx] -= d
                        a_trial[j_idx] += d
                        if a_trial[i_idx] < 0:
                            return 1e10
                        a_trial = a_trial / a_trial.sum()
                        return compute_score(a_trial)
                    
                    f1 = eval_delta(x1)
                    f2 = eval_delta(x2)
                    
                    for _ in range(15):
                        if f1 < f2:
                            hi = x2
                            x2 = x1
                            f2 = f1
                            x1 = hi - gr * (hi - lo)
                            f1 = eval_delta(x1)
                        else:
                            lo = x1
                            x1 = x2
                            f1 = f2
                            x2 = lo + gr * (hi - lo)
                            f2 = eval_delta(x2)
                    
                    t_opt = (lo + hi) / 2
                    f_opt = min(f1, f2)
                    
                    if f_opt < best_score - 1e-12:
                        a[i_idx] -= t_opt
                        a[j_idx] += t_opt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_score = compute_score(a)
                        best_a = a.copy()
                        improved = True
                        break
                else:
                    # No improvement found, zero out this contribution
                    contribs[ci] = -1
        
        if not improved and iters % 20 == 0:
            # Perturb
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 50 == 0:
            print(f"    AS iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, n_pairs_per_iter=200, verbose=True):
    """Fast coordinate descent with random pairs and golden section search."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Generate pairs
        idx_i = rng.integers(0, n, n_pairs_per_iter)
        idx_j = rng.integers(0, n, n_pairs_per_iter)
        
        for ii in range(n_pairs_per_iter):
            i = idx_i[ii]
            j = idx_j[ii]
            if i == j:
                continue
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Bidirectional golden section
            # t > 0: move from i to j
            # t < 0: move from j to i
            lo = -a[j] + 1e-15
            hi = a[i] - 1e-15
            if hi <= lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            
            def eval_t(t):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                if a_trial[i] < 0 or a_trial[j] < 0:
                    return 1e10
                return compute_score(a_trial)
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(12):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = eval_t(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = eval_t(x2)
            
            t_opt = (lo + hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if verbose and iters % 100 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_random_init(n, rng):
    """Create a random non-negative initialization."""
    # Mix of uniform and random
    a = np.ones(n) / n
    noise = rng.standard_normal(n) * 0.1
    a = a + noise
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def make_structured_init(n, rng):
    """Create a structured initialization using cosine modulation."""
    a = np.ones(n) / n
    # Add cosine structure
    for freq in rng.integers(1, 5, size=3):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Diverse initialization + quick gradient descent
    Phase 2: Adaptive-temperature smoothed gradient descent (main)
    Phase 3: Multiplicative updates
    Phase 4: Active-set refinement
    Phase 5: Coordinate descent polish
    Phase 6: Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    n_base = len(parent) if parent is not None else 768
    
    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    candidates = []
    
    if parent is not None:
        candidates.append(("parent", parent.copy()))
    
    # Add random and structured inits at the base length
    for i in range(4):
        candidates.append((f"rand_{i}", make_random_init(n_base, rng)))

    t1_budget = 30.0

        if time.monotonic() > deadline - 600:
            break

            refined_candidates.append((name, cand))

        a_ref, sc_ref = gradient_descent_smoothed(cand, budget, T_start=5.0, T_end=1.0, n_iters=1000, verbose=False)

        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"  {name}: score={sc_ref:.8f} (NEW BEST)")
        else:
            print(f"  {name}: score={sc_ref:.8f}")
