"""
Optimized auto-convolution peak minimization using full-constraint LP,
complementary sequence mixing, and multi-strategy gradient descent.
Targets score < 1.5027 for the auto-convolution peak problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_full_constraint_matrix(n, f):
    """Build the (2n-1) x n constraint matrix for (f*g)[k] <= M."""
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # Equivalent: A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    return A

def solve_lp_full(f, n, M, solver_list=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses full constraint matrix.
    Returns g (unnormalized) or None.
    """
    if solver_list is None:
        solver_list = [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS]
    
    A = build_full_constraint_matrix(n, f)
    
    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M, g_var >= 0])
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None:
                g = np.asarray(g_var.value, dtype=np.float64)
                g = np.maximum(g, 0.0)
                if not np.any(np.isnan(g)) and not np.any(np.isinf(g)):
                    return g
        except Exception:
            continue
    
    return None

def line_search_peak(f, direction, max_evals=15):
    """Find best step size t in [0, 1] to minimize peak of (1-t)f + t*direction."""
    n = len(f)
    
    def peak_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # Coarse search
    t_vals = np.linspace(0.01, 0.8, 12)
    best_t = 0.05
    best_obj = peak_at(0.05)
    for t in t_vals:
        obj = peak_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Golden section refinement
    if best_t < t_vals[-1]:
        idx = int(np.argmin(np.abs(t_vals - best_t)))
        lo = t_vals[max(0, idx-1)]
        hi = t_vals[min(len(t_vals)-1, idx+1)]
    else:
        lo = 0.4
        hi = 0.8
    
    gr = (math.sqrt(5) - 1) / 2

            d = c
            fd = fc

            fc = peak_at(c)

            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = peak_at(d)

                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    # Can't improve, try gradient
                    break
            continue
        
        h = g / S_g  # Normalize to sum=1
        
        # Try complementary mixing: (f + h) / 2
        f_mix = (f + h) / 2.0
        f_mix = np.maximum(f_mix, 0.0)
        s_mix = f_mix.sum()
        if s_mix > 1e-10:
            f_mix = f_mix / s_mix
        obj_mix = fast_convolve(f_mix).max()
        
        # Try standard line search
        best_t = line_search_peak(f, h)
        f_ls = (1.0 - best_t) * f + best_t * h
        f_ls = np.maximum(f_ls, 0.0)
        s_ls = f_ls.sum()
        if s_ls > 1e-10:
            f_ls = f_ls / s_ls
        obj_ls = fast_convolve(f_ls).max()
        
        # Pick the better
        if obj_mix < obj_ls:
            f_new = f_mix
            obj_new = obj_mix
        else:
            f_new = f_ls
            obj_new = obj_ls
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                break
        
        if it % log_interval == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"      LP iter={it} obj={best_obj:.10f} score={score:.8f} t={elapsed:.0f}s")
    
    return best_f, best_obj

def gradient_descent(f_init, time_limit, n_top=10, lr_init=0.01):
    """
    Subgradient descent on max of top peaks.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)  # Momentum
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    momentum = 0.8
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        b = fast_convolve(f)
        
        # Top-k peaks
        k = min(n_top, len(b))
        top_idx = np.argpartition(b, -k)[-k:]
        
        # Weighted subgradient
        # Weight peaks by their value (or by exponential scaling)
        peak_vals = b[top_idx]
        w = peak_vals / peak_vals.sum()
        
        grad = np.zeros(n)
        for i, k_val in enumerate(top_idx):
            k_int = int(k_val)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            for j in range(lo, hi + 1):
                grad[j] += w[i] * 2.0 * f[k_int - j]
        
        # Update with momentum
        v = momentum * v - lr * grad
        v = v - v.mean()  # Project to tangent space
        f_new = project_simplex(f + v)
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.8
            if no_improve > 50:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = lr_init
    
    return best_f, best_obj

def coordinate_descent(f_init, time_limit, n_sweeps=100):
    """
    Coordinate descent: optimize one coordinate at a time.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    def obj_at_coord(f, j, val):
        """Compute peak when coordinate j is set to val."""
        f_trial = f.copy()
        f_trial[j] = val
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # Precompute: for each j, the effect on b
    # b[k] = sum_i f[i]*f[k-i], the terms involving f[j] are:
    #   f[j]*f[k-j] for valid k-j
    #   f[k-j]*f[j] for valid k-j (same term)
    # So b[k] = base_k + f[j] * (f[k-j] + f[k-j]) = base_k + 2*f[j]*f[k-j] (if k != 2j)
    # Or b[k] = base_k + f[j]^2 (if k = 2j)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            # For coordinate j, b[k] as function of f[j]:
            # b[k] = (terms not involving f[j]) + f[j] * f[k-j] + f[k-j] * f[j]
            #      = base_k + 2*f[j]*f[k-j] (for k != 2j, both valid)
            #      = base_k + f[j]^2 (for k = 2j)
            
            # The peak is max_k b[k]. We want to choose f[j] >= 0 to minimize this.
            # This is a 1D optimization: min_x max_k (base_k + c_k * x + d_k * x^2)
            # where c_k = 2*f[k-j] (if valid) and d_k = 1 if k=2j else 0.
            
            # Simple approach: try a few values
            current = f[j]
            # The optimal x* minimizes the max, which occurs when
            # the "active" constraints are equal. For simplicity, use golden section.
            
            # Bounds: x in [0, 1] (since sum=1 and f[j] <= 1)
            lo = 0.0
            hi = max(1.0, current * 3)
            
            # Actually, let's just try a few multiplicative factors
            best_val = current
            best_obj_j = best_obj
            for factor in [0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
                val = current * factor
                if val < 0 or val > 1:
                    continue
                obj = obj_at_coord(f, j, val)
                if obj < best_obj_j:
                    best_obj_j = obj
                    best_val = val
            
            if best_val != current:
                f[j] = best_val
                # Renormalize
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                improved = True
                if fast_convolve(f).max() < best_obj - 1e-15:
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
        
        if not improved and it > 2:
            break
    
    return best_f, best_obj

def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # Exponential decay (left-heavy)
    x = np.exp(-np.arange(n) / (n * 0.3))
    inits.append(x / x.sum())
    
    # Exponential decay (right-heavy)
    x = np.exp(-np.arange(n)[::-1] / (n * 0.3))
    inits.append(x / x.sum())
    
    # Double exponential
    x = np.exp(-np.abs(np.arange(n) - n*0.3) / (n * 0.2))
    inits.append(x / x.sum())
    
    # Cosine window (shifted)
    x = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n) - n*0.2) / n))
    x = np.maximum(x, 0)
    if x.sum() > 1e-10:
        inits.append(x / x.sum())
    
    # Chirp-inspired (non-uniform spacing)
    x = np.abs(np.sin(np.pi * np.arange(n) / n * 1.5)) ** 0.5
    if x.sum() > 1e-10:
        inits.append(x / x.sum())
    
    # Random smooth (low-pass filtered random)
    r = rng.standard_normal(n)
    kernel = np.exp(-np.arange(-10, 11)**2 / 20.0)
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 0)
    if r.sum() > 1e-10:
        inits.append(r / r.sum())
    
    return inits
