from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_cross(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(f, temp, lr):
    """One step of tempered subgradient descent."""
    b = fast_convolve(f)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    n = len(f)
    g = np.zeros(n)
    # Efficient subgradient computation
    # g[j] = sum_k w[k] * 2*f[k-j] for valid k-j
    # This is a weighted sum of shifted copies of f
    for k in range(len(b)):
        if w[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w[k] * 2.0 * f[k - (np.arange(lo, hi+1))]
    
    f_new = project_simplex(f - lr * g)
    return f_new

def run_tempered_subgradient(f_init, time_limit, temp_init=0.01, lr_init=0.001):
    """Tempered subgradient with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temp = temp_init

    it = 0
    
    while time.monotonic() - t0 < time_limit:

        f = subgradient_step(f, temp, lr)

    return best_f, best_obj

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """Compute subgradient of max_k C_k given weights w on the C_k values."""
    n = len(f)
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += wk * 2.0 * f[k - hi:k - lo + 1]
    return g


def solve_lp(f, n, M, active_idx, eps_val):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx."""
    K = len(active_idx)
    if K == 0:
        return None

    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    g_val = None
    # Try SCS first (faster for large problems)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps_val)
        if g_var.value is not None:
            g_val = np.maximum(g_var.value, 0.0)
    except Exception:
        pass

    if g_val is None or np.any(np.isnan(g_val)):
        # Try ECOS for smaller problems
        if n <= 1500:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
            except Exception:
                pass
        else:
            # Try PDLP for medium problems
            try:
                prob.solve(solver=cp.PDLP, max_iter=200000)
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
            except Exception:
                pass

    if g_val is None or np.any(np.isnan(g_val)):
        return None
    return g_val

def tempered_subgradient(f_init, time_limit, temp_init=0.005, lr_init=0.003):
    """Tempered subgradient method with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temp = temp_init
    lr = lr_init
    momentum = 0.8
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Softmax weights (focus on peaks)
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        g = compute_subgradient(f, w)

        # Momentum update
        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 6:
                    lr *= 0.85
                    temp *= 0.9
                    if no_improve > 30:
                        # Restart from best with perturbation
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                        v = np.zeros(n)
                        lr = lr_init
                        temp = temp_init
                        no_improve = 0

        # Adaptive step size decay
        lr = max(1e-5, lr_init / (1.0 + 0.005 * it))
        temp = max(1e-4, temp_init * (0.998 ** it))

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    sg it={it} obj={best_obj:.12f} score={score:.6f} "
                  f"lr={lr:.6f} temp={temp:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates and normalization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    best_f = f.copy()

    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-15:
                continue
            orig = f[j]
            best_val = orig
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = orig * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                best_f = f.copy()
                improved = True
        if not improved:
            # Try finer steps
            improved = False
            for j in range(0, n, step):
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj
                if f[j] < 1e-15:
                    continue
                orig = f[j]
                best_val = orig
                best_obj_j = best_obj
                for mult in [0.97, 0.985, 1.015, 1.03]:
                    f[j] = orig * mult
                    s = f.sum()
                    if s < 1e-10:
                        f[j] = orig
                        continue
                    f_norm = f / s
                    obj = fast_convolve(f_norm).max()
                    if obj < best_obj_j - 1e-14:
                        best_obj_j = obj
                        best_val = orig * mult
                f[j] = best_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
            if not improved:
                break

    return best_f, best_obj


def parametric_init(n, power=0.5, shift=0.0, rng=None):
    """Create a parametric initial sequence with a power-law tilt."""
    x = np.linspace(0.0, 1.0, n)

    a = (x + shift) ** power
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    else:
        a /= s
    return a

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Primary CP-LP on parent ===
    print(f"\n=== Phase 1: CP-LP primary (n={len(parent)}) ===")
    n_p = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 600.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, initial_k=60, max_k=700")
        f1, _ = cp_lp_optimize(parent, time_limit=t1, initial_k=60, max_k=700, label="cp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [cp1]: {score1:.6f}")

    # === Phase 2: Second CP-LP pass with different parameters ===
    print(f"\n=== Phase 2: CP-LP second pass (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        # Use larger initial active set
        f2, _ = cp_lp_optimize(best_seq, time_limit=t2, initial_k=100, max_k=800, label="cp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [cp2]: {score2:.6f}")

    # === Phase 3: Tempered subgradient ===
    print(f"\n=== Phase 3: Tempered subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 20:
        f3, _ = tempered_subgradient(best_seq, time_limit=t3, temp_init=0.003, lr_init=0.002)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg]: {score3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    # Try a few lengths around the base
    target_lengths = []
    for delta in [-100, -50, 50, 100, 200, 300]:
        tl = n_base + delta
        if 512 <= tl <= 3000 and tl != n_base:
            target_lengths.append(tl)
    # Also try a couple of round numbers
    for tl in [1024, 1200, 1500, 2048]:
        if 512 <= tl <= 3000 and tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)

    # Sort by proximity to n_base
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.10, 100.0)
        if t4 < 15:
            break

        # Interpolate to new length
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  CP-LP n={target_n}, budget={t4:.0f}s")
        f4, _ = cp_lp_optimize(f_interp, time_limit=t4, initial_k=40, max_k=400,
                                label=f"n{target_n}")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {s4:.6f}")

    # === Phase 5: Diversification via parametric inits + quick LP ===
    print(f"\n=== Phase 5: Diversification (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5_total = min(rem * 0.12, 150.0)
    if t5_total > 30:
        n_b = len(best_seq)
        params = [
            (0.3, 0.0), (0.7, 0.1), (1.0, 0.05), (0.5, 0.2), (1.5, 0.0)
        ]
        t_each = t5_total / len(params)
        for power, shift in params:
            if time.monotonic() > deadline:
                break
            # Create perturbed version: mix of best_seq and parametric
            f_param = parametric_init(n_b, power=power, shift=shift, rng=rng)
            for mix in [0.3, 0.5, 0.7]:
                f_mix = (1.0 - mix) * best_seq + mix * f_param
                f_mix = np.maximum(f_mix, 0.0)
                s_mix = f_mix.sum()
                if s_mix > 1e-10:
                    f_mix /= s_mix

                t_mix = min(t_each * 0.6, 40.0)
                if t_mix < 5:
                    continue
                f5, _ = cp_lp_optimize(f_mix, time_limit=t_mix, initial_k=30, max_k=300,
                                        label=f"mix{mix}p{power}")
                s5 = float(evaluate_sequence(f5.tolist()))
                if np.isfinite(s5) and s5 < best_score:
                    best_score = s5
                    best_seq = f5.copy()
                    print(f"  NEW BEST [mix]: {s5:.6f}")
                    break

    # === Phase 6: Final CP-LP burst on best ===
    print(f"\n=== Phase 6: Final CP-LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 180.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}, k=150->800")
        f6, _ = cp_lp_optimize(best_seq, time_limit=t6, initial_k=150, max_k=800, label="final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {score6:.6f}")

    # === Phase 7: Coordinate polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 7: Coord polish (best={best_score:.6f}) ===")
        t7 = min(rem * 0.8, 60.0)
        f7, _ = coordinate_polish(best_seq, time_limit=t7)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {s7:.6f}")

    # === Final validation ===
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = best_seq.sum()
    if s_final < 0.01:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    else:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
