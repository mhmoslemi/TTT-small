```python
"""
Search for non-negative step-function heights minimizing

    2*n*max(np.convolve(a, a)) / (sum(a))^2

The algorithm uses:
  - scale-invariant weighted-quadratic coordinate descent with exact 1D minimizers,
  - active/softmax constraint weights (peaks of a*a),
  - symmetry projection and line search to keep true maximum decreasing,
  - stochastic restarts,
  - multi-length search (parent length, 1536, 2048, and best length),
  - a final direct coordinate polish for moderate sizes.
"""

import time
import numpy as np

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft


def _fft_size(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    size = _fft_size(out_len)
    A = rfft(a, size)
    c = irfft(A * A, size)[:out_len]
    # Remove tiny negative FFT roundoff.
    c = np.maximum(c, 0.0)
    return c


def _linweight(a, v):
    # Returns (Q_v a)_i = sum_j v_{i+j} a_j, i=0..n-1.
    a = np.asarray(a, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = _fft_size(3 * n - 2)
    A = rfft(v, size)
    B = rfft(a[::-1], size)
    out = irfft(A * B, size)[n - 1:n - 1 + n]
    out = np.maximum(out, 0.0)
    return out


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_raw(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    c = _convolve(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * np.max(c) / (s * s))


def _score(a):
    return _score_raw(_normalize(a))


def _interp(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    a = _normalize(a)
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    return _normalize(b)


def _make_weights(c, contrast, rho):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    mx = float(np.max(c))
    mn = float(np.min(c))
    if mx - mn <= 1e-18:
        w = np.ones(L, dtype=np.float64) / float(L)
    else:
        x = (c - mn) / (mx - mn)
        w = np.exp(contrast * x - contrast)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.ones(L, dtype=np.float64) / float(L)
        else:
            w = w / s
    w = (1.0 - rho) * w + rho * (1.0 / float(L))
    w = 0.5 * (w + w[::-1])
    w = np.maximum(w, 0.0)
    s = float(w.sum())
    if s <= 0.0:
        w = np.ones(L, dtype=np.float64) / float(L)
    else:
        w = w / s
    return w


def _active_weights(c, m, rho):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.ones(1, dtype=np.float64)
    m = int(max(1, min(m, L)))
    idx = np.argsort(c)[-m:]
    idx_sym = np.unique(np.concatenate((idx, (L - 1 - idx))))
    v = np.zeros(L, dtype=np.float64)
    v[idx_sym] = 1.0 / float(len(idx_sym))
    v = (1.0 - rho) * v + rho * (1.0 / float(L))
    v = 0.5 * (v + v[::-1])
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 0.0:
        v = np.ones(L, dtype=np.float64) / float(L)
    else:
        v = v / s
    return v


def _perturb(a, rng, scale):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    a2 = a + rng.standard_normal(n) * float(scale)
    a2 = np.maximum(a2, 0.0)
    a2 = _normalize(a2)
    a2 = 0.5 * (a2 + a2[::-1])
    return _normalize(a2)


def _init_edge(n, rng):
    t = np.linspace(0.0, 1.0, int(n))
    alpha = float(rng.uniform(0.25, 0.85))
    a = 1.0 + alpha * np.cos(np.pi * t)
    return _normalize(a)


def _init_fourier(n, rng):
    n = int(n)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.65, dtype=np.float64)
    m = int(rng.integers(2, 12))
    for _ in range(m):
        f = int(rng.integers(1, max(2, n // 4)))
        amp = float(rng.uniform(0.25, 0.75))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.cos(2.0 * np.pi * f * t / float(n) + ph)
    return _normalize(a)


def _init_cosine(n, rng):
    n = int(n)
    t = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    width = float(n) * float(rng.uniform(0.22, 0.48))
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return _normalize(a)


def _init_mixed(n, rng):
    a = _init_edge(n, rng)
    b = _init_fourier(n, rng)
    lam = float(rng.uniform(0.2, 0.8))
    return _normalize((1.0 - lam) * a + lam * b)


def _weighted_cd_steps(a, v, max_sweeps, alpha):
    """
    Coordinate descent for scale-invariant objective
    (v @ conv(a,a) + alpha ||a||^2) / sum(a)^2.
    Returns states after each sweep.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return [a]
    vs = float(v.sum())
    if vs <= 0.0 or not np.isfinite(vs):
        v = np.ones(v.size, dtype=np.float64) / float(v.size)
    else:
        v = v / vs

    n = a.size
    states = []
    if max_sweeps <= 0:
        states.append(a.copy())
        return states

    temp = np.empty(n, dtype=np.float64)
    alpha = float(alpha)

    for _sweep in range(int(max_sweeps)):
        c = _convolve(a)
        E = float(np.dot(v, c))
        sq = float(np.dot(a, a))
        lin = _linweight(a, v)
        lin = np.maximum(lin, 0.0)
        qvec = v[0::2]

        S = float(a.sum())
        if S <= 1e-12:
            a = np.ones(n, dtype=np.float64) / float(n)
            S = 1.0
        base_S = max(1.0, S)
        max_x = 20.0 * base_S

        for i in range(n):
            ai = float(a[i])
            R = S - ai
            if R <= 1e-13 * base_S:
                continue

            qv = float(qvec[i])
            L = float(lin[i]) - qv * ai
            if L < 0.0:
                L = 0.0

            q = qv + alpha
            C = E + alpha * sq - q * ai * ai - 2.0 * L * ai
            if C < 0.0:
                C = 0.0

            current_val = (E + alpha * sq) / (S * S) if S > 0.0 else 0.0
            best_x = ai
            best_val = current_val

            if R > 1e-13:
                val0 = C / (R * R)
                if np.isfinite(val0) and val0 < best_val:
                    best_val = val0
                    best_x = 0.0

            D = q * R - L
            if D > 1e-15:
                root = (C - L * R) / D
                if root > 0.0 and np.isfinite(root):
                    if root > max_x:
                        root = max_x
                    denom = R + root
                    if denom > 1e-15:
                        val = (q * root * root + 2.0 * L * root + C) / (denom * denom)
                        if np.isfinite(val) and val < best_val:
                            best_val = val
                            best_x = root
            else:
                if q < best_val:
                    cand = max(10.0 * R, 10.0 * ai, 1e-6)
                    if cand > max_x:
                        cand = max_x
                    denom = R + cand
                    if denom > 1e-15:
                        val = (q * cand * cand + 2.0 * L * cand + C) / (denom * denom)
                        if np.isfinite(val) and val < best_val:
                            best_val = val
                            best_x = cand

            if best_x > max_x:
                best_x = max_x
            if best_x < 0.0:
                best_x = 0.0

            delta = best_x - ai
            if abs(delta) <= 1e-15 * base_S:
                continue

            np.multiply(v[i:i + n], delta, out=temp)
            lin += temp

            E += 2.0 * delta * L + qv * delta * delta
            sq += 2.0 * delta * ai + delta * delta
            S += delta
            a[i] = best_x

            if (i & 0xFF) == 0:
                np.maximum(lin, 0.0, out=lin)
                if E < 0.0:
                    E = 0.0
                if S <= 1e-12:
                    S = 1e-12

        if S > 1e-12:
            a *= (1.0 / S)
        else:
            a = np.ones(n, dtype=np.float64) / float(n)

        states.append(a.copy())

    return states


def _optimize_track(a_init, n, time_limit, seed, tag=""):
    a_init = np.asarray(a_init, dtype=np.float64)
    if a_init.size != int(n):
        a_init = _interp(a_init, int(n))
    a_init = _normalize(a_init)

    n = int(n)
    best_a = a_init.copy()
    best_score = _score_raw(best_a)

    if time_limit <= 0:
        return best_a, best_score

    cur_a = a_init.copy()
    cur_score = best_score

    start = time.monotonic()
    deadline = start + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed))
    no_improve = 0
    it = 0
    temp = 1e-5

    while time.monotonic() < deadline - 1.0:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-12, time_limit))

        c = _convolve(cur_a)
        s = float(cur_a.sum())
        if s <= 1e-12:
            cur_a = best_a.copy()
            cur_score = best_score
            continue
        cur_score = float(2.0 * n * np.max(c) / (s * s))

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if cur_score <= best_score * 1.01:
            c_ref = c
        else:
            c_ref = _convolve(best_a)

        option = it % 4
        if no_improve > 5:
            option = 2 + (it % 2)

        if option == 0:
            v = _make_weights(c_ref, 3.0, 0.05)
        elif option == 1:
            v = _make_weights(c_ref, 8.0, 0.03)
        elif option == 2:
            m = max(8, int(0.02 * (2 * n - 1)))
            v = _active_weights(c_ref, m, 0.05)
        else:
            m = max(5, int(0.005 * (2 * n - 1)))
            v = _active_weights(c_ref, m, 0.02)

        reg = max(0.08, 2.0 * (1.0 - frac))
        alpha = reg * (2.0 / (2.0 * n - 1.0))

        if n <= 1200:
            max_sw = 3
        elif n <= 2500:
            max_sw = 2
        else:
            max_sw = 1

        states = _weighted_cd_steps(cur_a, v, max_sw, alpha)

        best_cand = None
        best_cscore = cur_score + temp

        for st in states:
            sc = _score_raw(st)
            if np.isfinite(sc) and sc < best_cscore:
                best_cscore = sc
                best_cand = st

        last = states[-1] if states else cur_a
        for lam in (0.25, 0.5):
            mix = cur_a.copy()
            mix *= float(1.0 - lam)
            mix += float(lam) * last
            mix = np.maximum(mix, 0.0)
            sc = _score_raw(mix)
            if np.isfinite(sc) and sc < best_cscore:
                best_cscore = sc
                best_cand = mix

        if best_cand is not None:
            sym = 0.5 * (best_cand + best_cand[::-1])
            sym = np.maximum(sym, 0.0)
            ss = float(sym.sum())
            if ss > 1e-15:
                sym /= ss
                scs = _score_raw(sym)
                if np.isfinite(scs) and scs < best_cscore:
                    best_cscore = scs
                    best_cand = sym

        if best_cand is not None and np.isfinite(best_cscore) and best_cscore < cur_score + temp:
            cur_a = best_cand
            cur_score = best_cscore
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"  [{tag}] it={it} n={n} score={best_score:.6f}")

            if best_score < cur_score and cur_score > best_score * 1.01:
                cur_a = best_a.copy()
                cur_score = best_score
        else:
            no_improve += 1

            if no_improve == 3:
                m = max(5, int(0.003 * (2 * n - 1)))
                v2 = _active_weights(c, m, 0.02)
                states2 = _weighted_cd_steps(cur_a, v2, 1, 0.2 * alpha)
                for st in states2:
                    sc = _score_raw(st)
                    if np.isfinite(sc) and sc < cur_score + temp:
                        cur_a = st
                        cur_score = sc
                        no_improve = 1
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = cur_a.copy()
                            print(f"  [{tag}] it={it} n={n} quick={best_score:.6f}")
                        break

            if no_improve > 8:
                scale = min(0.08, 0.01 + 0.005 * no_improve)
                cur_a = _perturb(best_a, rng, scale)
                cur_score = _score_raw(cur_a)
                no_improve = 3

        temp = max(1e-7, temp * 0.995)

    sym = 0.5 * (best_a + best_a[::-1])
    sym = np.maximum(sym, 0.0)
    ss = float(sym.sum())
    if ss > 1e-15:
        sym /= ss
        sc = _score_raw(sym)
        if np.isfinite(sc) and sc < best_score:
            best_a = sym
            best_score = sc

    return best_a, best_score


def _direct_polish(a0, time_limit):
    a = _normalize(a0)
    n = int(a.size)
    if n == 0:
        return a
    if time_limit <= 0:
        return a

    S = 1.0
    c = _convolve(a)
    M = float(np.max(c))
    best_score = float(2.0 * n * M)
    best_a = a.copy()

    t0 = time.monotonic()
    tmp = np.empty(n, dtype=np.float64)
    factor_vals = (0.0, 0.3, 0.6, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0)
    base_vals = (0.001, 0.004, 0.01, 0.03)

    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        time_out = False

        for i in range(n):
            if (i & 0xFF) == 0 and time.monotonic() - t0 > time_limit:
                time_out = True
                break

            ai = float(a[i])
            ks = i
            ke = i + n - 1

            unaff = 0.0
            if ks > 0:
                unaff = float(np.max(c[:ks]))
            if ke + 1 < 2 * n - 1:
                rr = float(np.max(c[ke + 1:]))
                if rr > unaff:
                    unaff = rr

            aff = c[ks:ke + 1]
            best_x = ai
            best_sc = float(2.0 * n * M)

            for f in factor_vals:
                x = float(f) * ai
                if x < 0.0:
                    continue
                if x > 10.0:
                    x = 10.0
                if abs(x - ai) <= 1e-14:
                    continue
                delta = x - ai
                newS = S + delta
                if newS <= 1e-12:
                    continue

                np.multiply(a, 2.0 * delta, out=tmp)
                tmp += aff
                tmp[i] += delta * delta

                amax = float(np.max(tmp))
                newM = unaff if unaff > amax else amax
                sc = float(2.0 * n * newM / (newS * newS))
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_x = x

            for bf in base_vals:
                x = float(bf) * S
                if x > 10.0:
                    x = 10.0
                if abs(x - ai) <= 1e-14:
                    continue
                delta = x - ai
                newS = S + delta
                if newS <= 1e-12:
                    continue

                np.multiply(a, 2.0 * delta, out=tmp)
                tmp += aff
                tmp[i] += delta * delta

                amax = float(np.max(tmp))
                newM = unaff if unaff > amax else amax
                sc = float(2.0 * n * newM / (newS * newS))
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_x = x

            if abs(best_x - ai) > 1e-14:
                delta = best_x - ai
                np.multiply(a, 2.0 * delta, out=tmp)
                tmp += aff
                tmp[i] += delta * delta

                c[ks:ke + 1] = tmp
                a[i] = best_x
                S += delta

                if S <= 1e-12 or abs(S - 1.0) > 1e-6:
                    if S > 1e-12:
                        scale = 1.0 / S
                        a *= scale
                        c *= scale * scale
                        S = 1.0
                    else:
                        a = np.ones(n, dtype=np.float64) / float(n)
                        c = _convolve(a)
                        S = 1.0

                M = float(np.max(c))
                cur_score = float(2.0 * n * M)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = _normalize(a.copy())
                improved = True

        if time_out:
            break
        if not improved and sweeps > 2:
            break

    return best_a


def propose_candidate():
    gstart = time.monotonic()
    gdeadline = gstart + 1770.0
    rng_global = np.random.default_rng(20240517)

    best_seq = None
    best_score = float("inf")

    # Use parent construction if available.
    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64)
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 0.01:
            p = _normalize(p)
            best_seq = p.copy()
            best_score = _score_raw(p)
            print(f"Parent: n={len(p)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = _init_edge(1024, rng_global)
        best_score = _score_raw(best_seq)
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.6f}")

    sym0 = _normalize(0.5 * (best_seq + best_seq[::-1]))
    sc0 = _score_raw(sym0)
    if np.isfinite(sc0) and sc0 < best_score:
        best_seq = sym0
        best_score = sc0
        print(f"Symmetrized initial: score={best_score:.6f}")

    # Phase 1: improve parent length.
    future_reserve = 1350.0
    rem = gdeadline - time.monotonic() - future_reserve
    t1 = min(320.0, rem)
    if t1 > 30.0:
        n1 = int(len(best_seq))
        a, b = _optimize_track(best_seq, n1, t1, seed=11, tag="P1-parent")
        if np.isfinite(b) and b < best_score:
            best_seq = a.copy()
            best_score = b
            print(f"P1 best: score={best_score:.6f}, n={len(best_seq)}")

    # Phase 2: try 1536.
    future_reserve = 1010.0
    rem = gdeadline - time.monotonic() - future_reserve
    t2 = min(380.0, rem)
    if t2 > 30.0:
        n2 = 1536
        init2 = _interp(best_seq, n2)
        a, b = _optimize_track(init2, n2, t2, seed=22, tag="P2-1536")
        if np.isfinite(b) and b < best_score:
            best_seq = a.copy()
            best_score = b
            print(f"P2 best: score={best_score:.6f}, n={len(best_seq)}")

    # Phase 3: try 2048.
    future_reserve = 560.0
    rem = gdeadline - time.monotonic() - future_reserve
    t3 = min(450.0, rem)
    if t3 > 30.0:
        n3 = 2048
        init3 = _interp(best_seq, n3)
        a, b = _optimize_track(init3, n3, t3, seed=33, tag="P3-2048")
        if np.isfinite(b) and b < best_score:
            best_seq = a.copy()
            best_score = b
            print(f"P3 best: score={best_score:.6f}, n={len(best_seq)}")

    # Phase 4: deep optimization at the current best length.
    future_reserve = 200.0
    rem = gdeadline - time.monotonic() - future_reserve
    t4 = min(300.0, rem)
    if t4 > 30.0 and best_seq is not None:
        n4 = int(len(best_seq))
        init4 = best_seq.copy() if best_seq.size == n4 else _interp(best_seq, n4)
        a, b = _optimize_track(init4, n4, t4, seed=44, tag="P4-deep")
        if np.isfinite(b) and b < best_score:
            best_seq = a.copy()
            best_score = b
            print(f"P4 best: score={best_score:.6f}, n={len(best_seq)}")

    # Final polishing.
    rem = gdeadline - time.monotonic() - 10.0
    if best_seq is not None and rem > 40.0:
        nf = int(len(best_seq))

        if nf <= 1600:
            td = min(180.0, rem - 10.0)
            if td > 20.0:
                a = _direct_polish(best_seq, td)
                sc = _score_raw(a)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = a.copy()
                    best_score = sc
                    print(f"Direct polish best: score={best_score:.6f}, n={nf}")

        rem = gdeadline - time.monotonic() - 10.0
        if rem > 60.0 and best_seq is not None:
            tw = min(120.0, rem - 10.0)
            if tw > 20.0:
                a, b = _optimize_track(best_seq, int(len(best_seq)), tw, seed=99, tag="FINAL")
                if np.isfinite(b) and b < best_score:
                    best_seq = a.copy()
                    best_score = b
                    print(f"Final weighted best: score={best_score:.6f}, n={len(best_seq)}")

    # Final candidate pool: current best and its symmetrization.
    final_seq = _normalize(best_seq) if best_seq is not None else _init_edge(1024, rng_global)
    final_score = _score_raw(final_seq)
    candidates = [final_seq]

    fsym = _normalize(0.5 * (final_seq + final_seq[::-1]))
    candidates.append(fsym)

    # If we started with a much shorter parent, include it too.
    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = _normalize(p)
            candidates.append(p)
    except Exception:
        pass

    best_final = final_seq
    best_final_score = final_score
    for cand in candidates:
        cand = _normalize(cand)
        sc = _score_raw(cand)
        if np.isfinite(sc) and sc < best_final_score:
            best_final_score = sc
            best_final = cand

    # Verify with the official evaluator when available, but do not let it fail.
    try:
        eval_candidates = [best_final, _normalize(0.5 * (best_final + best_final[::-1]))]
        for cand in eval_candidates:
            cand_list = [float(x) for x in cand.tolist()]
            val = float(evaluate_sequence(cand_list))
            if np.isfinite(val) and val < best_final_score:
                best_final_score = val
                best_final = cand
    except Exception:
        pass

    arr = np.asarray(best_final, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    arr = _normalize(arr)

    return [float(x) for x in arr.tolist()]
```
