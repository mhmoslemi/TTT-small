"""
Optimized step function search using soft-gradient Nesterov momentum with
adaptive learning rate, spectral perturbation, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(w, f):
    """Compute grad[j] = sum_i w[j+i] * f[i] for j=0..n-1."""
    n = len(f)
    lw = len(w)
    grad = np.empty(n)
    for j in range(n):
        hi = min(n, lw - j)
        if hi > 0:
            grad[j] = np.dot(w[j:j+hi], f[:hi])
        else:
            grad[j] = 0.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])  # The only point on the 1-d simplex with x>=0, sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_normalized(f):
    """Compute the score assuming sum(f) = 1."""
    n = len(f)
    b = fast_auto_conv(f)
    return 2.0 * n * b.max()


def soft_gradient(f, b, beta):
    """Compute soft gradient of max(b) w.r.t. f."""
    b_max = b.max()
    # Exponential weighting
    shifted = b - b_max
    # Avoid underflow
    shifted = np.maximum(shifted, -50.0 / max(beta, 1e-10))
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    return cross_correlate(w, f), w


def nesterov_descent(f_init, time_limit, lr_init=0.001, momentum=0.9, beta_init=10.0,
                     beta_growth=1.02, max_beta=500.0, label="nest"):
    """
    Nesterov-style accelerated gradient descent with soft-max gradient.
    Minimizes max(auto_conv(f)) subject to f >= 0, sum(f) = 1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    v = np.zeros(n)  # momentum
    lr = lr_init
    beta = beta_init
    momentum = momentum
    
    best_f = f.copy()
    best_obj = fast_auto_conv(f).max()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        b = fast_auto_conv(f)
        grad, w = soft_gradient(f, b, beta)
        
        # Nesterov update
        v_new = momentum * v - lr * grad
        f_trial = project_simplex(f + v_new)
        
        # Check objective
        obj = fast_auto_conv(f_trial).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_trial.copy()
            no_improve = 0
            # Increase learning rate slightly
            lr = min(lr * 1.01, lr_init * 10)
            # Increase beta (sharper gradient)
            beta = min(beta * beta_growth, max_beta)
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce learning rate
                lr *= 0.85
                v = v * 0.5
                no_improve = 0
            elif no_improve > 20:
                lr *= 0.95
        
        f = f_trial
        v = v_new
        
        # Periodic reporting
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                  f"lr={lr:.6f} beta={beta:.1f} t={elapsed:.0f}s")
        
        # Periodic perturbation to escape local minima

def lp_refine(f_init, time_limit, max_constraints=200):
    """
    Brief LP-based refinement: find the best direction g that reduces peak,
    then take a step towards it.
    """

    n = len(f)
    best_f = f.copy()
    best_obj = fast_auto_conv(f).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_auto_conv(f)
        M = b.max()
        
        # Get top constraints
        k = min(max_constraints, len(b))
        top_idx = np.argpartition(b, -k)[-k:]
        top_idx = np.sort(top_idx)
        
        # Build LP: maximize sum(g) s.t. g >= 0, (f*g)[lag] <= M for lag in top_idx
        A = np.zeros((k, n))
        for row_i in range(k):
            lag = int(top_idx[row_i])
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            A[row_i, lo:hi+1] = f[lo:hi+1]
        
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
        
        try:
            prob.solve(solver=cp.PDLP, max_iters=100000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=100000)
            except Exception:
                break
        
        if g_var.value is None:
            break
        
        g = np.maximum(g_var.value, 0.0)
        S_g = g.sum()
        
        if S_g <= 1.0 + 1e-5:
            break
        
        h = g / S_g
        
        # Try several step sizes

        best_trial = best_obj

def interpolate_sequence(f_old, n_new):
    """Interpolate a normalized sequence to a new length."""
    n_old = len(f_old)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(x_new, x_old, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return f_new / s

"""
Optimized step function search using soft-gradient Nesterov momentum with
adaptive learning rate, LP refinement bursts, and multi-length exploration.
Targets score < 1.5027 for the auto-convolution peak minimization problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(f, g):
    """Linear cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_gradient(w, f):
    """Compute grad[j] = 2 * sum_k w[k] * f[k-j] for valid indices."""
    n = len(f)
    grad = np.empty(n)
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        grad[j] = 2.0 * np.dot(w[lo:hi], f[:hi - lo])
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, beta):
    """Compute softmax weights over the convolution peaks."""
    b_max = b.max()
    shifted = b - b_max
    shifted = np.maximum(shifted, -50.0 / max(beta, 1e-10))
    w = np.exp(beta * shifted)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= s
    return w


def nesterov_soft(f_init, time_limit, lr_init=0.0005, mu_init=0.9,
                  beta_init=5.0, beta_growth=1.015, max_beta=500.0,

    f = project_simplex(f_init.copy())

    mu = mu_init

            best_f = f_trial.copy()
            no_improve = 0

        else:

def lp_refine(f_init, time_limit, max_constraints=200, label="lp"):
    """
    LP-based refinement with constraint generation.
    Maximizes sum(g) s.t. g >= 0, (f*g)[k] <= M for active constraints.
    Takes steps toward the LP optimum direction.
    """
    t0 = time.monotonic()
    f = project_simplex(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_auto_conv(f).max()
    outer_it = 0

    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        time_per_lp = min(rem * 0.3, 30.0)

        b = fast_auto_conv(f)
        M = b.max()

        k = min(max_constraints, len(b))
        top_idx = np.argpartition(b, -k)[-k:]
        top_idx = np.sort(top_idx)

        A = np.zeros((k, n))
        for row_i in range(k):
            lag = int(top_idx[row_i])
            lo = max(0, lag - n + 1)
            hi = min(n - 1, lag)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [g_var >= 0, A @ g_var <= M])

        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=200000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            break

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            break
        S_g = g.sum()

        if S_g <= 1.0 + 1e-5:
            break

        h = g / S_g

        # Line search
        best_t = 0.1
        best_trial = best_obj
        for t in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = project_simplex((1 - t) * f + t * h)
            obj = fast_auto_conv(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t

        f = project_simplex((1 - best_t) * f + best_t * h)

        if best_trial < best_obj - 1e-15:
            best_obj = best_trial
            best_f = f.copy()
            if outer_it % 5 == 0:
                print(f"    {label} outer={outer_it} obj={best_obj:.12f} "
                      f"score={2.0*n*best_obj:.8f} t={time.monotonic()-t0:.0f}s")
        else:
            break

    return best_f, best_obj


def interpolate_sequence(f_old, n_new):
    """Interpolate normalized sequence to new length."""
    n_old = len(f_old)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(x_new, x_old, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return f_new / s


def multi_peak_descent(f_init, time_limit, n_peaks=10, label="mpd"):
    """
    Focused descent on top peaks with adaptive step size.
    Uses combined gradient from multiple near-peak lags.
    """
    t0 = time.monotonic()
    f = project_simplex(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_auto_conv(f).max()
    lr = 0.001
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_auto_conv(f)
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Combined gradient from top peaks (equal weight)
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * f[k - lo:k - lo + (hi - lo + 1)][::-1]

        # Actually, gradient of b[k] w.r.t. f[j] = 2*f[k-j] for valid j
        # So for a fixed k: grad[j] += 2*f[k-j] for j in [k-n+1, k] intersect [0, n-1]
        # Let me redo this properly:
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                diff = k - j
                if 0 <= diff < n:
                    grad[j] += 2.0 * f[diff]

        if grad.max() < 1e-15:
            break

        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_trial = project_simplex(f - alpha * grad)
            obj = fast_auto_conv(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            if no_improve > 100:
                break
            no_improve += 1
            lr *= 0.9
            continue

        f = project_simplex(f - best_alpha * grad)
        if best_trial < best_obj - 1e-15:
            best_obj = best_trial
            best_f = f.copy()
            no_improve = 0
            lr = min(best_alpha * 1.1, 0.2)
        else:
            no_improve += 1
            lr *= 0.9

        if (outer_it if False else 0) or True:
            pass

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: Nesterov soft-gradient descent, LP refinement,
    multi-peak focused descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152.0
        p_score = 2.0
        print(f"No parent, using uniform n=1152, score=2.0")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Main Nesterov descent (long, careful) ===
    print(f"\n=== Phase 1: Nesterov descent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = rem * 0.40
    f1, obj1 = nesterov_soft(parent, time_limit=t1, lr_init=0.0004,
                              mu_init=0.9, beta_init=3.0, beta_growth=1.012,
                              max_beta=200.0, label="nest1")
    score1 = float(evaluate_sequence(f1.tolist()))
    if np.isfinite(score1) and score1 < best_score:
        best_score = score1
        best_seq = f1.copy()
        print(f"  NEW BEST [nest1]: {best_score:.8f}")
    else:
        print(f"  nest1: {score1:.8f} (no improvement over {best_score:.8f})")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP refinement (n={len(best_seq)}) ===")
