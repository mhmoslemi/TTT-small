import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(g, f):
    """Cross-convolution g*f via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    G = rfft(g, fft_size)
    F = rfft(f, fft_size)
    return irfft(G * F, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_full_constraint_rows(f, n):

def golden_line_search(f, direction, current_obj, golden_ratio=0.6180339887498949):
    """
    Golden section search for optimal step size t in [0, t_max].
    Minimizes max(conv((1-t)f + t*direction, (1-t)f + t*direction)).
    """
    t_max = 1.0
    a = 0.0
    b = t_max
    
    def obj_at_t(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # Initial points
    c = b - golden_ratio * (b - a)
    d = a + golden_ratio * (b - a)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    # Also check t=0
    f0 = obj_at_t(0.0)
    
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - golden_ratio * (b - a)
            fc = obj_at_t(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + golden_ratio * (b - a)
            fd = obj_at_t(d)
    
    best_t = (a + b) / 2
    best_obj = min(obj_at_t(best_t), f0)
    
    # Also try some discrete points to be safe
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
        o = obj_at_t(t)
        if o < best_obj:

            best_t = t

def spectral_init_asymmetric(n, seed=0):
    """
    Create an asymmetric initial sequence using spectral methods.
    The idea: design a frequency response with controlled asymmetry,
    then take IFFT and clip to non-negative.
    """
    rng = np.random.default_rng(seed)
    L = 1 << (2 * n - 1).bit_length()
    
    # Start with a "one-sided" spectrum
    # A(k) = exp(-alpha * |k|^beta) * exp(i * phi(k))
    # The phase creates asymmetry in time domain
    
    k = np.arange(L)
    k_centered = k - L // 2
    k_centered[k_centered < -L // 2] += L
    
    # Magnitude: Gaussian-like decay
    alpha = 2.0 + rng.random() * 3.0
    beta = 1.0 + rng.random() * 1.0
    mag = np.exp(-alpha * np.abs(k_centered) ** beta / n)
    
    # Phase: linear + random for asymmetry
    phase_shift = rng.uniform(-0.3, 0.3) * n
    phase = 2.0 * np.pi * phase_shift * k / L + rng.standard_normal(L) * 0.1
    
    A = mag * np.exp(1j * phase)
    # Ensure real output: A[-k] = conj(A[k])
    for i in range(1, L // 2 + 1):
        A[L - i] = np.conj(A[i])
    A[0] = np.abs(A[0])
    if L % 2 == 0:
        A[L // 2] = np.abs(A[L // 2])
    
    a = np.real(np.fft.ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        # Fallback: triangular-like asymmetric
        a = np.concatenate([np.linspace(1, 0.3, n // 2), np.linspace(0.1, 0.01, n - n // 2)])
        s = a.sum()
    a = a / s
    return a

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=100, momentum=0.85):

    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    no_pass_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining min-max LP, spectral initialization,
    and diverse asymmetry patterns to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512")
    
    best_seq = parent.copy()
    best_score = p_score
    
    def try_update(f_res, tag):
        nonlocal best_seq, best_score
        if f_res is not None and len(f_res) > 0:
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [{tag}]: {real_score:.6f} (n={len(f_res)})")
            return real_score
        return np.inf
    
    # ============================================================
    # Phase 1: Min-max LP on parent with full constraints
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Min-max LP on parent (score: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    
    if t1 > 30:
        n_p = len(parent)
        print(f"  n={n_p}, budget={t1:.0f}s")
        f_res, _ = run_minmax_lp_optimization(
            parent, time_limit=t1, use_full=True,
            max_stuck=60, label="P1")
        try_update(f_res, "P1 full")
    
    # ============================================================
    # Phase 2: Min-max LP at various lengths near parent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Length exploration (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    
    if t2 > 30:
        n_base = len(best_seq)
        lengths = []
        for factor in [0.6, 0.75, 0.9, 1.1, 1.3, 1.5]:
            n_l = int(n_base * factor)
            n_l = max(128, min(3000, n_l))
            if n_l not in lengths:
                lengths.append(n_l)
        lengths.sort()
        
        per_n = t2 / len(lengths)
        for n_trial in lengths:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_n, rem * 0.85)
            if alloc < 20:
                break
            
            f_init = interp_sequence(best_seq, n_trial)
            use_full = n_trial <= 800
            print(f"  n={n_trial}, full={use_full}, budget={alloc:.0f}s")
            
            f_res, _ = run_minmax_lp_optimization(
                f_init, time_limit=alloc, use_full=use_full,
                top_k=min(300, 2 * n_trial - 1),
                max_stuck=40, label=f"P2-{n_trial}")
            try_update(f_res, f"P2 n={n_trial}")
    
    # ============================================================
    # Phase 3: Diversified asymmetric initializations
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse initialization (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    
    if t3 > 30:
        n_target = len(best_seq)
        n_styles = min(4, int(t3 // 60))
        per_style = t3 / max(1, n_styles)
        
        for style in range(n_styles):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(per_style, rem * 0.85)
            if alloc < 20:
                break
            
            # Create diverse initialization
            if style == 0:
                f_init = make_asymmetric_template(n_target, style=0, seed=seed + 100)
            elif style == 1:
                f_init = make_asymmetric_template(n_target, style=1, seed=seed + 200)
            elif style == 2:
                f_init = make_asymmetric_template(n_target, style=3, seed=seed + 300)
            else:
                f_init = spectral_init_asymmetric(n_target, seed=seed + style * 77)
            
            # Quick check: is this init better than uniform?
            init_score = 2.0 * n_target * fast_convolve(f_init).max()
            print(f"  Style {style}: init_score={init_score:.6f}, budget={alloc:.0f}s")
            
            f_res, _ = run_minmax_lp_optimization(
                f_init, time_limit=alloc, use_full=(n_target <= 800),
                top_k=min(300, 2 * n_target - 1),
                max_stuck=30, label=f"P3-s{style}")
            try_update(f_res, f"P3 style={style}")
    
    # ============================================================
    # Phase 4: Deep continuation on current best
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep continuation (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t4:.0f}s")
        
        # Use full constraints if feasible, else top-K
        use_full = n_b <= 1200
        f_res, _ = run_minmax_lp_optimization(
            best_seq, time_limit=t4, use_full=use_full,
            top_k=min(500, 2 * n_b - 1),
            max_stuck=80, label="P4")
        try_update(f_res, "P4 deep")
    
    # ============================================================
    # Phase 5: Subgradient polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: SG polish (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    
    if t5 > 15:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t5:.0f}s")
        f_res, _ = subgradient_polish(
            best_seq, time_limit=t5, lr_init=0.002,
            n_top=max(50, n_b // 8))
        try_update(f_res, "P5 sg")
    
    # ============================================================
    # Phase 6: Coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: CD (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.05, 60.0)
    
    if t6 > 5:
        print(f"  n={len(best_seq)}, budget={t6:.0f}s")
        f_res, _ = coordinate_descent(best_seq, time_limit=t6)
        try_update(f_res, "P6 cd")
    
    # ============================================================
    # Phase 7: Final LP burst
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final burst (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t7 = rem * 0.50
    
    if t7 > 15:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t7:.0f}s")
        use_full = n_b <= 1000
        f_res, _ = run_minmax_lp_optimization(
            best_seq, time_limit=t7, use_full=use_full,
            top_k=min(600, 2 * n_b - 1),
            max_stuck=100, label="P7")
        try_update(f_res, "P7 final")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
