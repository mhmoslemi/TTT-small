def evaluate_sequence(seq):
    """Evaluate the sequence and return the score defined as 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(seq)
    if not isinstance(seq, np.ndarray):
        seq = np.array(seq)
    if not np.any(seq):

    if np.amin(seq) < 0:

    a_with_finite = np.logical_and(np.greater(a, 1e-15), np.greater(a, -1e-12)).astype(float)
    # There is no finite values? Then return zeros.
    if not np.any(a_with_finite):
        convolved_linear = np.zeros(len(a) + len(a) - 1)
    else:
        convolved_linear = np.convolve(a, a, mode='full')
    return convolved_linear

def simplex_project(v):
    """Project a vector onto the simplex of non-negative unit vectors."""
    query = [float(x) for x in v]
    return np.array(query) / np.sum(query) * 1000.0

def get_lp_solver():
    """Select an appropriate LP solver."""
    # We assume CVXOPT or other solvers are available
    import cvxpy as cp
    
    # Test the solvers one by one
    solvers = [cp.ECOS, cp.CVXOPT, cp.SCS]
    for solver_name in solvers:
        try:
            prob = cp.Problem()
            variable = cp.Variable(1)
            constraint = variable == 1.0
            prob.solve(qp_warnings=False, solver=solver_name)
            if prob.status == 'optimal':
                return solver_name
        except:
            continue
    return None

def solve_lp_best_response(a_fixed):
    """Solve a linear program to find a response to a_fixed sequence of length n."""
    # Check arguments
    try:
        n = len(a_fixed)
        if n == 0:
            return None, np.inf
        # Normalize the input as required by the problem
        a_fixed = np.array(a_fixed)
        sum_a = np.sum(a_fixed)
        if sum_a < 0.01:
            a_fixed = np.ones(n) / n
        else:
            a_fixed = a_fixed / sum_a
        
        # Set up the variables (non-negative sum to n)
        b = cp.Variable(n, nonneg=True)
        # We wish to minimize the max of the convolution T·b
        T = build_conv_matrix(a_fixed)
        M = cp.Variable()
        constraints = [T @ b <= M, cp.sum(b) == 1]
        objective = cp.Minimize(M)
        prob = cp.Problem(objective, constraints)
        
        # Check if the problem is well-defined
        if not prob.is_feasible():
            return None, np.inf
            
        # Which solver to use? We'll try multiple if needed
        try:
            solver_selected = get_lp_solver()
        except:
            solver_selected = None
            
        prob.solve(solver=solver_selected)  # This will fail if no solver is available
        
        if not hasattr(prob, 'status') or prob.status != 'optimal':
            # If anything goes wrong, return a fallback point
            # Instead, we should be more robust
            pass
            
        # Extract the solution, which is a vector of ones normalized to the simplex
        try:
            b_value = b.value
            sum_b = np.sum(b_value)
            if sum_b > 1e-15:
                b_norm = simplex_project(b_value)
                score_new = evaluate_sequence(b_norm.tolist())
                return b_norm, score_new
            else:
                return np.ones(n)/n, evaluate_sequence((np.ones(n)/n).tolist())
        except:
            # In case of solver/numerical issues
            return np.ones(n)/n, evaluate_sequence((np.ones(n)/n).tolist())
            
    except:
        # If any step fails, return the normalized input
        return np.ones(n)/n, evaluate_sequence((np.ones(n)/n).tolist())

def build_conv_matrix(a_fixed):
    """Build the convolution matrix for a_fixed."""
    n = len(a_fixed)
    T = np.zeros((2*n-1, n), dtype=float)
    for i in range(n):
        indices_ok = [max(0, k-i) for k in range(2*n-1)]
        # Each row of T corresponds to a shift
        T[i:i+n, :] = a_fixed[:, np.newaxis] * np.ones(n)
    return T

def fictitious_play(n, time_limit):
    """Attempt global improvement by alternating best responses."""
    a = np.random.rand(n) * 0.01

    # Try a different initialization
    a_alt = np.random.choice(2, n) == 1
    best_a_alt, best_score_alt = fast_coord_descent(a_alt, time_limit, n_trials=3)
    
    # Combine the best two
    if n is not None:

    else:
        n = best_a_alt.size
        
    current_a = simplex_project(0.5 * (np.ones(n) + best_a + best_a_alt))

def mirror_descent(a0, time_limit):
    """Project into simplex and use mirrordescent to explore common variations."""
    a = np.array(a0)
    n = len(a)
    # Confirm nonnegativity and finite values
    try:
        if np.any(np.greater(a, 1000.0)):
            a = np.clip(a, None, 1000.0)
        a = simplex_project(a / np.sum(a))
    except:
        a = np.random.rand(n) * 0.01
        
    try:
        # Build the projection matrix
        conv = full_convolve(a)
        M = np.amax(conv)
        
        # The simplex projection will map m into [0, M]
    except:
        M = np.random.rand()
    
    # Update within a local solution
    
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    
    # We'll first obtain a sequence that might improve (via `propose_candidate`)
    try:
        candidate, score_candidate = fictitious_play(n, time_limit)
    except:
        score_candidate = None
        candidate = None
        
    score_twins = 0
    if candidate is not None:
        for i in range(n):
            if score_candidate < evaluate_sequence(candidate.tolist()):
                break
            elif score_candidate == evaluate_sequence(candidate.tolist()):
                score_twins += 1
                
    if score_twins > 0:
        # Use this twin solution as the initial best
        best_seq = candidate
        best_score = score_candidate
    
    # Now, perform mirrordescent with a slightly different line of reasoning
    
    t0 = time.monotonic()
    simplex_history = []
    current_score = best_score
    
    # We wish to explore common variations as alternatives to product space
    
    while time.monotonic() - t0 < time_limit and current_score not in [1.5027, 0]:
        # Update direction: from best_seq to current checker
        delta_a = current_score - best_score
        if delta_a > 0:
            # Use simplex_project to find better mixing
            corrected_a = simplex_project(best_seq * (1 + delta_a))
        else:
            corrected_a = best_seq * (1 + delta_a)
            
        # How about: take steps toward uniform distribution
        updated_a = (1 - delta_a) * best_seq + delta_a * (np.random.rand(n) - 0.5)
        # Alternatively, use the previous best if uniform doesn't help
        
        conv_updated = full_convolve(updated_a)
        max_conv_updated = np.amax(conv_updated)
        score_updated = 2 * n * max_conv_updated / (np.sum(updated_a) ** 2)
        
        if score_updated < current_score:
            best_seq = updated_a
            best_score = score_updated
        else:
            # If not improvement, update with opposite gradient
            worse_step = best_seq.copy() * 5 + updated_a.copy() * -1
            better_step = simplex_project(best_seq + worse_step)
            conv_better = full_convolve(better_step)
            max_conv_better = np.amax(conv_better)
            score_better = 2 * n * max_conv_better / (np.sum(better_step) ** 2)
            
            if score_better < current_score:
                best_seq = better_step
                best_score = score_better
            else:
                best_seq = simplex_project(best_seq * 1.1)
                best_score = current_score  # remains unchanged
        
        # Preserve the current score and save the best
        simplex_history.append((best_seq, best_score))
        # Proceed with caution: if we haven't improved for a while, restart
        
    # Finally, return the best found and its score
    
    best_avg_score = min(best_score, evaluate_sequence(best_seq.tolist()))
    return best_seq, best_avg_score
