import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, weights, K):
    """Compute subgradient of max_k b_k weighted by softmax weights."""
    n = len(f)
    g = np.zeros(n)
    for k in range(K):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # db_k/da_j = a_{k-j} for j in [lo, hi]
        # where k-j is in [0, n-1]
        idx = np.arange(lo, hi + 1)
        g[idx] += w * f[k - idx]
    return g

def nesterov_subgradient(f_init, time_limit, lr_init=0.01, temp_init=0.001, 
                          temp_max=0.5, n_top=200, restart_every=100,
                          label="nag"):
    """Nesterov accelerated subgradient with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    x = f_init.copy()
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
        s = 1.0
    x = x / s
    y = x.copy()  # Nesterov lookahead point
    v = np.zeros(n)  # velocity
    t = 1.0
    
    best_x = x.copy()
    best_obj = fast_convolve(x).max()
    no_improve = 0
    temp = temp_init
    lr = lr_init
    rng = np.random.default_rng(123)
    
    K = 2 * n - 1
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1.0
        
        # Evaluate at lookahead point
        b = fast_convolve(y)
        M = b.max()
        
        # Compute softmax weights
        scaled = (b - M) / temp
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(K)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum
        
        # Restrict to top n_top for efficiency
        if n_top < K:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(K)
            w_new[top_idx] = w_sel
            w = w_new
        
        # Compute subgradient
        g = compute_subgradient(y, w, K)
        
        # Nesterov update

        v = beta * v - lr * g
        x_new = project_simplex(x + v)

        y = project_simplex(y)
        x = x_new

def multiplicative_update(f_init, time_limit, lr_init=0.05, n_top=100, label="mu"):
    """Multiplicative update method for non-negative optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    rng = np.random.default_rng(456)
    K = 2 * n - 1
    lr = lr_init
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Find peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        
        # Compute "pressure" on each coefficient
        pressure = np.zeros(n)
        for k in top_idx:
            if b[k] < M * 0.5:
                continue
            weight = (b[k] / M) ** 2
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            pressure[idx] += weight * f[k - idx]
        
        # Multiplicative update
        # f_i *= (1 - lr * pressure_i / (M + eps))
        ratio = 1.0 - lr * pressure / (M + 1e-10)
        ratio = np.clip(ratio, 0.01, 1.0)
        f_new = f * ratio
        s = f_new.sum()
        if s < 1e-15:
            f_new = f
        else:
            f_new = f_new / s
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            lr = min(lr * 1.01, 0.5)
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.9
                no_improve = 0
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                lr = lr_init
                no_improve = 0
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} best={best_obj:.12f} score={2*n*best_obj:.6f} "
                  f"lr={lr:.5f} t={elapsed:.0f}s")
    
    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def coordinate_descent(f_init, time_limit):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 1.5, 1.8, 2.5])
    
    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val > 10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, best_score

                  f"lr={lr:.5f} temp={temp:.5f} t={el:.0f}s")
    
    return best_x, 2.0 * n * best_obj

def lp_optimize(f_init, time_limit, top_k=80, label="lp"):
    """LP-based direction finding with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    K_total = 2 * n - 1
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    stuck = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        
        # Select peaks: top-K plus structural
        if top_k < K_total:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            # Add center and edges
            center = (K_total - 1) // 2
            structural = list(range(min(5, K_total)))
            structural += list(range(max(0, K_total - 5), K_total))
            structural += [center - d for d in range(3) if 0 <= center - d < K_total]
            structural += [center + d for d in range(3) if 0 <= center + d < K_total]
            all_idx = list(set(list(top_idx) + structural))
            all_idx.sort(key=lambda i: b[i], reverse=True)
            indices = np.array(all_idx[:top_k + 10])
        else:
            indices = np.arange(K_total)
        
        indices = np.sort(indices)
        K = len(indices)
        
        # Build A matrix
        A = np.zeros((K, n))
        for i in range(K):
            k = indices[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        # Solve LP
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
        
        solved = False
        try:
            if K <= 300 and n <= 2000:
                prob.solve(solver=cp.ECOS, max_iters=100000)
            else:
                prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-10, eps_rel=1e-10)
            solved = prob.status in ['optimal', 'optimal_inaccurate']
        except Exception:
            pass
        
        if not solved:
            try:
                prob.solve(solver=cp.CLARABEL, max_iters=100000)
                solved = prob.status in ['optimal', 'optimal_inaccurate']
            except Exception:
                pass
        
        if not solved or g_var.value is None:
            stuck += 2
            if stuck > 40:
                break
            continue
        
        g_val = np.maximum(g_var.value, 0.0)
        S = g_val.sum()
        
        if S <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 40:
                break
            continue
        
        h = g_val / S
        
        # Line search
        best_t = 0.0
        best_obj_t = M
        for t_val in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_t = (1 - t_val) * f + t_val * h
            f_t = np.maximum(f_t, 0.0)
            st = f_t.sum()
            if st < 1e-15:
                continue
            f_t = f_t / st
            obj_t = fast_convolve(f_t).max()
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t_val
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 40:
                break
            continue
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        
        cur = fast_convolve(f).max()
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 15:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stuck = 0
        
        if (int((time.monotonic() - t0) * 10) // 10) % 10 == 0:
            pass  # periodic print
    
    return best_f, 2.0 * n * best_obj

def cd_polish(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    mults = np.array([0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0])
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_o = best_obj
            for m in mults:
                nv = f[j] * m
                f[j] = nv
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_val
                    continue
                f_n = f / s
                o = fast_convolve(f_n).max()
                if o < best_o - 1e-14:
                    best_o = o
                    best_val = nv
                    f = f_n.copy()
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                f = f / s
                if best_o < best_obj:
                    best_obj = best_o
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, 2.0 * n * best_obj
