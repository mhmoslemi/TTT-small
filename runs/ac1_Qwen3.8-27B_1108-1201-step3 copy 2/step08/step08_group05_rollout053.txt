def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Build constraint matrix A where A[i,:] gives row for constraint i."""

        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_indices, time_budget=10.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_indices.
    Returns (g, success).
    """
    t0 = time.monotonic()
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n, True

    A = build_constraint_matrix(n, f, active_indices)

    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )

    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - t0 > time_budget:
            break
        try:
            prob.solve(solver=solver, max_iters=200000)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g = np.maximum(g_var.value, 0.0)
                return g, True
        except Exception:
            continue

    return None, False

                grad[j] += 2.0 * f[j]
            else:
                # Two contributions: from term f[j]*f[comp] (i=j) and term f[comp]*f[j] (i=comp)
                # d/dx_j of f[j]*f[comp] = f[comp]
                # d/dx_j of f[comp]*f[j] = f[comp]
                # Total: 2*f[comp]
                # But wait, we're iterating over j, and for each j we add to grad[j]
                # The term f[i]*f[k-i] with i=j gives f[j]*f[k-j], derivative w.r.t. f[j] is f[k-j]
                # The term f[i]*f[k-i] with i=k-j gives f[k-j]*f[j], derivative w.r.t. f[j] is f[k-j]
                # So grad[j] += 2*f[k-j]
                # BUT: the second term (i=k-j) is only in the sum if k-j is in [lo, hi]
                # Since j is in [lo, hi], and comp = k-j, we need comp in [lo, hi]
                # comp = k-j, and j is in [lo, hi] => comp is in [k-hi, k-lo]
                # k-lo = k - max(0, k-n+1) and k-hi = k - min(n-1, k)
                # If k < n: lo=0, hi=k, comp ranges from k to 0, all in [0,k]=[lo,hi]. OK.
                # If k >= n: lo=k-n+1, hi=n-1, comp = k-j ranges from k-(n-1) to k-(k-n+1) = n-1
                #   comp in [k-n+1, n-1] = [lo, hi]. Also OK!
                # So comp is always in [lo, hi] when j is in [lo, hi].
                grad[j] += 2.0 * f[comp]

    return grad


def run_lp_optimization(f_init, time_limit, top_k_init=50, epsilon_init=0.05,
                        max_constraints=300, label="lp"):
    """
    LP-based peak shaving optimization with adaptive constraint selection.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = project_simplex(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    epsilon = epsilon_init
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        c = fast_convolve(f)
        M = c.max()

        # Adaptive constraint selection
        active = get_active_constraints(c, M, epsilon, max_constraints=max_constraints)
        if len(active) < 5:
            # Fallback: top-k
            k_sel = min(top_k, len(c))
            active = np.argpartition(c, -k_sel)[-k_sel:]

        # Solve LP
        time_budget = min(rem * 0.3, 15.0)
        g, success = solve_lp_direction(f, n, M, active, time_budget=time_budget)

        if not success or g is None:
            # Fallback: subgradient direction
            grad = compute_subgradient_v2(f, n, active)
            if grad.max() < 1e-15:
                break
            g = np.maximum(grad, 0)
            if g.sum() < 1e-10:
                g = np.ones(n) / n

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 30:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                stuck = 0
                continue
            # Increase epsilon to get more constraints
            epsilon = min(epsilon * 1.3, 0.5)
            continue

        # Normalize
        h = g / S_g

        # Line search with backtracking
        cur_obj = M
        alpha = 0.5
        max_alpha = 1.0
        min_alpha = 1e-5
        accepted = False
        for _ in range(30):
            f_trial = (1.0 - alpha) * f + alpha * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                alpha *= 0.5
                continue
            f_trial = f_trial / s
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj - 1e-14:
                f = f_trial
                stuck = 0
                accepted = True
                break
            alpha *= 0.7

        if not accepted:
            stuck += 1
            if stuck > 50:
                break
            # Try smaller epsilon (fewer constraints)
            epsilon = max(epsilon * 0.7, 0.01)
            continue

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
            # Decrease epsilon (tighter constraints) if making progress
            epsilon = max(epsilon * 0.95, 0.01)
        else:
            stuck += 1
            epsilon = min(epsilon * 1.1, 0.5)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * cur_obj
            print(f"    {label} it={it} peak={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"eps={epsilon:.4f} |A|={len(active)} t={elapsed:.0f}s")

    return best_f.copy(), best_obj


def run_subgradient(f_init, time_limit, n_top=40, beta_init=5.0, label="sg"):
    """
    Subgradient method with momentum and adaptive temperature.
    Uses soft-max for smoothing.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    beta = beta_init
    temperature = 0.01
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 3)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            c = fast_convolve(f)
            M = c.max()

            # Soft-max weighting
            shifted = c - M
            w = np.exp(beta * shifted / max(temperature, 0.001))
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(c))
                w[c.argmax()] = 1.0
            else:
                w /= w_sum

            # Restrict to top indices
            k_sel = min(n_top, len(c))
            top_idx = np.argpartition(c, -k_sel)[-k_sel:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w_active = np.zeros(len(c))
            w_active[top_idx] = w_sel

            # Compute weighted gradient
            grad = np.zeros(n)
            for idx in range(len(c)):
                if w_active[idx] < 1e-12:
                    continue
                k = int(idx)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    comp = k - j
                    if comp == j:
                        grad[j] += w_active[idx] * 2.0 * f[j]
                    else:
                        grad[j] += w_active[idx] * 2.0 * f[comp]

            # Momentum update
            lr = 0.001 / (1.0 + 0.001 * it)
            v = 0.9 * v - lr * grad
            f = project_simplex(f + v)

        # Check improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                # Anneal
                temperature = temperature * 0.9
                beta = max(beta * 0.95, 1.0)
                no_improve = 0
            if no_improve > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} peak={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def run_smooth_optimization(f_init, time_limit, beta=10.0, label="smooth"):
    """
    Optimize smooth surrogate: soft-max peak, using projected gradient.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        it += 1

        c = fast_convolve(f)
        M = c.max()

        # Soft-max weight
        shifted = c - M
        w = np.exp(beta * shifted)
        w /= w.sum()

        # Compute gradient of soft-max peak
        grad = np.zeros(n)
        for idx in range(len(c)):
            if w[idx] < 1e-12:
                continue
            k = int(idx)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                comp = k - j
                if comp == j:
                    grad[j] += w[idx] * 2.0 * f[j]
                else:
                    grad[j] += w[idx] * 2.0 * f[comp]

        # Projected gradient descent
        best_alpha = None
        best_trial = M
        for alpha in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            f_trial = project_simplex(f - alpha * grad)
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_trial - 1e-14:
                best_trial = obj_trial
                best_alpha = alpha

        if best_alpha is None:
            stuck += 1
            if stuck > 30:
                beta *= 0.9
                if beta < 1.0:
                    break
                stuck = 0
            continue

        f = project_simplex(f - best_alpha * grad)
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} peak={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} beta={beta:.1f} t={elapsed:.0f}s")

    return best_f, best_obj


def flat_top_window(n):
    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolated_sequence(source, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(source)
    if src_n == target_n:
        return source.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, source)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return result / s


def run_combined(f_init, time_limit, label="combined"):
    """
    Run a combined optimization: LP -> smooth -> subgradient, with budget splitting.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Phase 1: LP (50% of budget)
    t1 = time_limit * 0.45
    rem = time_limit - (time.monotonic() - t0)
    t1 = min(t1, rem)
    if t1 > 5:
        f1, obj1 = run_lp_optimization(f, time_limit=t1, label=f"{label}_lp")
        if obj1 < best_obj:
            best_obj = obj1
            best_f = f1.copy()

    # Phase 2: Smooth (25% of budget)
    t2 = time_limit * 0.20
    rem = time_limit - (time.monotonic() - t0)
    t2 = min(t2, rem)
    if t2 > 5:
        f2, obj2 = run_smooth_optimization(best_f, time_limit=t2, beta=8.0,
                                            label=f"{label}_smooth")
        if obj2 < best_obj:
            best_obj = obj2
            best_f = f2.copy()

    # Phase 3: Subgradient (remaining)
    t3 = time_limit - (time.monotonic() - t0) - 2
    t3 = max(t3, 0)
    if t3 > 5:
        f3, obj3 = run_subgradient(best_f, time_limit=t3, n_top=max(20, n // 15),
                                    label=f"{label}_sg")
        if obj3 < best_obj:
            best_obj = obj3
            best_f = f3.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-strategy ensemble with LP-based peak shaving,
    smooth surrogate optimization, and subgradient methods.
    Explores multiple lengths and initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng_master = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: LP optimization on parent ===
    print(f"\n=== Phase 1: LP on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_combined(parent, time_limit=t1, label="parent")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [parent]: {score1:.8f}")
        else:
            print(f"  parent: {score1:.8f} (no improvement over {best_score:.8f})")

    # === Phase 2: Try different initializations at same length ===
    print(f"\n=== Phase 2: Alternative initializations (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    inits = {
        "hann": hann_window(n_parent),
        "flat": flat_top_window(n_parent),
        "perturbed": project_simplex(parent + rng_master.standard_normal(n_parent) * 0.05),
    }
    # Remove if too close to parent
    for name in list(inits.keys()):
        if np.abs(inits[name] - parent).max() < 0.01:
            del inits[name]

    t_per_init = min(rem * 0.20 / max(1, len(inits)), 120.0)
    for name, init_f in inits.items():
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_init = min(t_per_init, rem * 0.4)
        if t_init < 15:
            continue
        print(f"  Init={name}, budget={t_init:.0f}s")
        fi, obji = run_combined(init_f, time_limit=t_init, label=f"init_{name}")
        scorei = float(evaluate_sequence(fi.tolist()))
        if np.isfinite(scorei) and scorei < best_score:
            best_score = scorei
            best_seq = fi.copy()
            print(f"  NEW BEST [init_{name}]: {scorei:.8f}")
        else:
            print(f"  init_{name}: {scorei:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.8f}) ===")
    n_base = len(best_seq)
    candidate_lengths = []
    # Try lengths around the current best
    for mult in [0.75, 0.85, 1.0, 1.1, 1.25, 1.5]:
        nl = int(n_base * mult)
        nl = max(512, min(3000, nl))
        if nl not in candidate_lengths and nl != n_base:
            candidate_lengths.append(nl)
    # Also try some specific lengths
    for nl in [1024, 1280, 1536, 1920, 2048]:
        if nl not in candidate_lengths and 512 <= nl <= 3000:
            candidate_lengths.append(nl)

    candidate_lengths.sort(key=lambda x: abs(x - n_base))
    n_to_try = min(3, len(candidate_lengths))

    for idx, target_n in enumerate(candidate_lengths[:n_to_try]):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.25, 250.0)
        if t5 < 25:
            break

        # Interpolate best sequence to new length
        f_interp = interpolated_sequence(best_seq, target_n)

        print(f"  Trying n={target_n}, budget={t5:.0f}s")
        fi, obji = run_combined(f_interp, time_limit=t5, label=f"n{target_n}")
        scorei = float(evaluate_sequence(fi.tolist()))
        if np.isfinite(scorei) and scorei < best_score:
            best_score = scorei
            best_seq = fi.copy()
            print(f"  NEW BEST [n{target_n}]: {scorei:.8f}")
        else:
            print(f"  n{target_n}: {scorei:.8f} (best still {best_score:.8f})")

    # === Phase 4: Final polish on best ===
    print(f"\n=== Phase 4: Final polish (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.30, 200.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4, obj4 = run_combined(best_seq, time_limit=t4, label="polish")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [polish]: {score4:.8f}")
        else:
            print(f"  polish: {score4:.8f}")

    # === Phase 5: Aggressive subgradient on best ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t5 = min(rem * 0.5, 100.0)
        if t5 > 10:
            print(f"\n=== Phase 5: Subgradient polish ===")
            f5, obj5 = run_subgradient(best_seq, time_limit=t5,
                                        n_top=max(30, len(best_seq) // 10),
                                        beta_init=3.0, label="final_sg")
            score5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  NEW BEST [final_sg]: {score5:.8f}")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.6f}")
    print(f"Total time: {time.monotonic() + 30.0 - (deadline - budget_s + 30.0):.0f}s")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
