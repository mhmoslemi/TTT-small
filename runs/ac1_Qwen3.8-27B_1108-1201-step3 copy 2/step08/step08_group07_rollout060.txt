import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_active(a, active_set):
    """Compute subgradient of max-conv at active lags.
    g[j] = sum_{k in active_set} a[k-j] (with a[m]=0 for m out of range)
    """
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        # g[j] = sum_{k} s[k] * a[k-j] where only valid indices contribute
        lo = max(0, 0)  # k-j >= 0 => k >= j
        hi = min(2 * n - 2, n - 1 + j)  # k-j <= n-1 => k <= j+n-1
        if lo > hi:
            continue
        k_range = np.arange(lo, hi + 1)
        a_idx = k_range - j  # k-j, which is in [0, n-1]
        vals = a[active_set] if not isinstance(active_set, slice) else None
        # Direct computation
        s_vals = np.zeros(2 * n - 1)
        s_vals[active_set] = 1.0
        g[j] = np.dot(s_vals[lo:hi+1], a[a_idx])
    return g


def subgradient_fast(a, active_lags):
    """Fast subgradient using the formula g[j] = sum_{k in S} a[k-j]."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    s = np.zeros(fft_size)
    s[active_lags] = 1.0
    S_hat = rfft(s, fft_size)
    A_hat = rfft(a, fft_size)
    # Cross-correlation: sum_k s[k]*a[k-j] for j=0,...,n-1
    # This is ifft(S * conj(A)) but we need the right shift
    corr = irfft(S_hat * np.conj(A_hat), fft_size)
    # corr[m] = sum_k s[k] * a[(k-m) mod fft_size]
    # We want g[j] = sum_k s[k] * a[k-j] = corr with appropriate indexing
    # Actually: ifft(S * conj(A))[m] = sum_k s[k+m] * a[k] (circular)
    # Let me just use the direct formula for correctness
    g = np.zeros(n)
    for j in range(n):
        # g[j] = sum_{k=j}^{j+n-1} s[k] * a[k-j]  (k-j in [0,n-1])
        if j + n - 1 >= size:
            hi = size - 1
        else:
            hi = j + n - 1
        if j >= size:
            g[j] = 0.0
            continue
        s_slice = s[j:min(hi, size - 1) + 1]
        a_slice = a[:len(s_slice)]
        g[j] = np.dot(s_slice, a_slice)
    return g


def error_reduction(a0, time_limit, verbose=True, label="ER"):
    """Gerchberg-Saxton / error reduction: alternate between
    enforcing flat |A|^2 in frequency domain and a >= 0 in time domain."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    iters = 0
    tau = 0.3  # softness parameter for amplitude modification

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size

        # Modify amplitude to be flatter (geometric mean with target)
        ratio = np.where(power > 1e-30, (target / power) ** tau, 1.0)
        ratio = np.clip(ratio, 0.5, 2.0)
        A_new = A * ratio
        # Enforce Hermitian symmetry (A is from rfft so it's already real-ish)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Gradually increase tau for more aggressive flattening
        if iters % 100 == 0:
            tau = min(0.8, tau + 0.02)

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, tau={tau:.2f}, time={elapsed:.0f}s")

    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, verbose=True, label="MS"):
    """Multiplicative update based on subgradient of max-conv."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 0.1
    tol = 0.01  # tolerance for active set

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        active = np.where(conv >= M - tol * M)[0]
        
        # Compute subgradient: g[j] = sum_{k in active} a[k-j]
        # Use FFT for speed
        fft_size = 1 << (2 * n - 2).bit_length()
        s_vec = np.zeros(fft_size)
        s_vec[active] = 1.0
        S_hat = rfft(s_vec, fft_size)
        A_hat = rfft(a, fft_size)
        # g[j] = sum_k s[k] * a[k-j] for j in [0, n-1]
        # = cross-correlation
        g = irfft(S_hat * np.conj(A_hat), fft_size)
        # g[m] = sum_k s[k] * a[(k-m) mod N] -- need to fix indexing
        # Actually let's just compute directly for n <= 2000
        if n <= 2000:
            g_direct = np.zeros(n)
            s_full = np.zeros(2 * n - 1)
            s_full[active] = 1.0
            for j in range(n):
                lo = j
                hi = min(j + n, 2 * n - 1)
                g_direct[j] = np.dot(s_full[lo:hi], a[0:hi-lo])
            g = g_direct

        # Normalize subgradient
        g_norm = g / (g.mean() + 1e-15)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g_norm)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            # Increase eta slightly
            eta = min(0.3, eta * 1.02)
        else:
            eta = max(0.01, eta * 0.95)
        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.4f}, "
                  f"|S|={len(active)}, time={elapsed:.0f}s")

    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # C[k, j] = f[k-j] if 0 <= k-j < n
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        C[lo:hi, j] = f[hi - j - 1:lo - j - 1:-1] if hi > lo else 0
    # Simpler: C[k,j] = f[k-j] if 0<=k-j<n else 0
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            C[k, j] = f[k - j]
    return C


def build_conv_matrix_fast(f):
    """Build convolution matrix efficiently using strided tricks."""
    n = len(f)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols  # idx[k, j] = k - j
    # C[k, j] = f[k-j] if 0 <= k-j < n else 0
    valid = (idx >= 0) & (idx < n)
    C = np.zeros((2 * n - 1, n))
    C[valid] = f[idx[valid]]
    return C


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix_fast(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    solver_list = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, warm_start=(prev_g is not None))
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return prev_g if prev_g is not None else f.copy()


def lp_fictitious_play(a0, time_limit, step=0.5, verbose=True, label="LP-FP",
                        adapt_step=True):
    """LP-based fictitious play with adaptive step sizing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    best_step = step

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Adaptive step: try multiple steps, pick best
        if adapt_step:
            best_local_score = best_score
            best_local_a = a
            for trial_step in [step, step * 0.5, step * 1.5, 0.1, 0.8]:
                trial_step = max(0.01, min(1.0, trial_step))
                a_trial = (1 - trial_step) * a + trial_step * g_norm
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_a = a_trial
                    best_step = trial_step

            a = best_local_a
            score = best_local_score
        else:
            a_new = (1 - step) * a + step * g_norm
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                break
            a_new = a_new / ss
            a = a_new
            score = compute_score(a)
            best_local_score = score
            best_local_a = a

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Perturbation if stalled
        if stall_count > 200:
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
            step = min(0.8, step * 1.2)

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, verbose=True):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def smoothed_grad(z):
        a = softmax_from_z(z)
        a_safe = np.maximum(a, 1e-30)
        conv = fast_convolve(a_safe)
        # Log-sum-exp gradient
        c_max = conv.max()
        w = np.exp((conv - c_max) / T)
        w = w / w.sum()
        # Gradient of log-sum-exp w.r.t. a
        # d/d(a[j]) log(sum_k exp(c[k]/T)) = (1/T) sum_k w[k] * dc[k]/da[j]
        # dc[k]/da[j] = 2*a[k-j]
        fft_size = 1 << (2 * n - 2).bit_length()
        W_hat = rfft(w, fft_size)
        A_hat = rfft(a_safe, fft_size)
        grad_a = 2.0 * irfft(W_hat * np.conj(A_hat), fft_size)[:n]
        # Project to tangent space of simplex: grad_a -= dot(grad_a, a) * a
        S = float(np.dot(grad_a, a_safe))
        grad_z = a_safe * (grad_a - S)
        return (2.0 * n / T) * grad_z

    def func(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        c_max = conv.max()
        lse = c_max + T * np.log(np.exp((conv - c_max) / T).sum())
        return 2.0 * n * lse

    try:
        res = minimize(
            fun=func, x0=z0, jac=smoothed_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
            if verbose:
                print(f"  L-BFGS T={T}: improved to {best_score:.8f}")
    except Exception as e:
        if verbose:
            print(f"  L-BFGS T={T}: error {e}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=500, verbose=True):
    """Coordinate descent with pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue

            base = min(a[i], a[j] + a[i] * 0.5)
            best_local = best_score
            best_t = 0.0

            for frac in [0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                t = min(t, a[i] * 0.9)
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def projection_subgradient(a0, time_limit, verbose=True, label="PS"):
    """Projection subgradient method with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 1.0 / (4 * n)  # initial step
    tol_frac = 0.05  # active set tolerance (fraction of max)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        threshold = M * (1 - tol_frac)
        active = np.where(conv >= threshold)[0]

        # Compute subgradient
        fft_size = 1 << (2 * n - 2).bit_length()
        s_vec = np.zeros(fft_size)
        s_vec[active] = 1.0
        S_hat = rfft(s_vec, fft_size)
        A_hat = rfft(a, fft_size)
        # g[j] = sum_k s[k] * a[k-j]
        # Using correlation: ifft(conj(S_hat) * A_hat) gives sum_k s[k]*a[k+m]... 
        # Let me compute directly for correctness
        g = np.zeros(n)
        s_full = s_vec[:2*n-1]
        for j in range(n):
            lo = j
            hi = min(j + n, 2 * n - 1)
            g[j] = np.dot(s_full[lo:hi], a[0:hi-lo])

        # Normalize
        g_mean = g.mean()
        if g_mean < 1e-15:
            break
        g = (g - g_mean) / (np.linalg.norm(g - g_mean) + 1e-15)

        # Take step and project
        a_new = a - eta * g
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            eta = min(eta * 1.1, 5.0 / n)
        else:
            eta = max(eta * 0.9, 0.1 / n)
        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.2e}, "
                  f"|S|={len(active)}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Validate and update best sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}) ***")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization:
    1. LP fictitious play (adaptive step) - main driver
    2. Error reduction (Gerchberg-Saxton) - frequency domain
    3. Projection subgradient - time domain
    4. Multiplicative subgradient
    5. L-BFGS with temperature annealing
    6. Coordinate descent polish
    7. Length exploration
    8. Final LP polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting best: {best_score:.8f} (n={n_parent})")

    # === Phase 1: LP Fictitious Play - Large Step ===
    print(f"\n=== Phase 1: LP-FP (adapt, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if rem > 60:
        print(f"  Budget: {t1:.0f}s")
        a1, _ = lp_fictitious_play(best_seq, t1, step=0.5, adapt_step=True,
                                    label="LP-adapt", verbose=True)
        best_seq, best_score = check_and_update(a1, "lp_adapt", best_seq, best_score)

    # === Phase 2: Error Reduction (Gerchberg-Saxton) ===
    print(f"\n=== Phase 2: Error Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if rem > 40:
        print(f"  Budget: {t2:.0f}s")
        a2, _ = error_reduction(best_seq, t2, verbose=True, label="ER")
        best_seq, best_score = check_and_update(a2, "error_red", best_seq, best_score)

    # === Phase 3: Projection Subgradient ===
    print(f"\n=== Phase 3: Projection Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 150.0)
    if rem > 30:
        print(f"  Budget: {t3:.0f}s")
        a3, _ = projection_subgradient(best_seq, t3, verbose=True, label="PS")
        best_seq, best_score = check_and_update(a3, "proj_subgrad", best_seq, best_score)

    # === Phase 4: Multiplicative Subgradient ===
    print(f"\n=== Phase 4: Mult Subgrad (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a4, _ = multiplicative_subgradient(best_seq, t4, verbose=True, label="MS")
        best_seq, best_score = check_and_update(a4, "mult_subgrad", best_seq, best_score)

    # === Phase 5: L-BFGS with temperature annealing ===
    print(f"\n=== Phase 5: L-BFGS Annealing (best={best_score:.8f}) ===")
    for T_val in [100, 300, 1000, 3000]:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.08, 60.0)
        if rem < 20:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a5, _ = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=3000, verbose=True)
        best_seq, best_score = check_and_update(a5, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if rem > 20:
        print(f"  Budget: {t6:.0f}s")
        a6, _ = coordinate_descent(best_seq, t6, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a6, "coord_desc", best_seq, best_score)

    # === Phase 7: More LP with smaller step ===
    print(f"\n=== Phase 7: LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.10, 80.0)
    if rem > 20:
        print(f"  Budget: {t7:.0f}s")
        a7, _ = lp_fictitious_play(best_seq, t7, step=0.15, adapt_step=True,
                                    label="LP-polish", verbose=True)
        best_seq, best_score = check_and_update(a7, "lp_polish", best_seq, best_score)

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 4, n_parent + 8, n_parent - 4, n_parent + 16,
                  n_parent + 32, n_parent - 8]:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t8 = min(rem * 0.06, 40.0)
        print(f"  L-BFGS n={alt_n}, budget={t8:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a8, _ = lbgfs_optimize(init_alt, t8, T=500, maxiter=2000, verbose=True)
        best_seq, best_score = check_and_update(a8, f"n{alt_n}", best_seq, best_score)

        # Also try ER on the new length
        rem = deadline - time.monotonic()
        if rem > 15:
            t8b = min(rem * 0.04, 25.0)
            a8b, _ = error_reduction(a8, t8b, verbose=False, label="ER-alt")
            best_seq, best_score = check_and_update(a8b, f"ER_n{alt_n}", best_seq, best_score)

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.15, 50.0)
        print(f"  CD budget={t9:.0f}s")
        a9, _ = coordinate_descent(best_seq, t9, n_pairs=600, verbose=True)
        best_seq, best_score = check_and_update(a9, "final_cd", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 15:
        t9b = min(rem * 0.15, 40.0)
        print(f"  LP budget={t9b:.0f}s")
        a9b, _ = lp_fictitious_play(best_seq, t9b, step=0.1, adapt_step=True,
                                     label="LP-final", verbose=True)
        best_seq, best_score = check_and_update(a9b, "lp_final", best_seq, best_score)

    # === Phase 10: Another ER pass on best ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t10 = min(rem * 0.15, 40.0)
        print(f"  ER budget={t10:.0f}s")
        a10, _ = error_reduction(best_seq, t10, verbose=True, label="ER-final")
        best_seq, best_score = check_and_update(a10, "ER_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
            print(f"  Final eval gives better: {fs:.8f}")

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
