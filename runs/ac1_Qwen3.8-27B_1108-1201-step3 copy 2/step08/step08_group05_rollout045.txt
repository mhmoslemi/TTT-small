import cvxpy as cp
import time

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(n, f, indices):
    """Build constraint matrix for LP: row k has f values where (f*g)[k] is computed."""
    K = len(indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, M, active_indices, n, max_constraints=600):
    """Solve the LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in active_indices."""
    K = len(active_indices)
    if K == 0:
        return None
    if K > max_constraints:
        # Keep only the most important (highest f*f) constraints
        b = fast_convolve(f)
        vals = b[active_indices]
        keep = np.argsort(vals)[-max_constraints:]
        active_indices = [active_indices[i] for i in keep]
        K = len(active_indices)

    A = build_constraint_matrix(n, f, active_indices)

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    g_val = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g_val)) and not np.any(np.isinf(g_val)):
                    break
                g_val = None
        except Exception:
            continue

    return g_val


def lp_with_constraint_generation(f, n, time_budget, top_k_init=50, max_constraints=500):
    """LP with iterative constraint generation."""
    t0 = time.monotonic()
    b = fast_convolve(f)
    M = float(b.max())
    if M < 1e-15:
        return None

    # Initialize with top peaks
    if top_k_init < len(b):
        top_idx = np.argpartition(b, -top_k_init)[-top_k_init:]
    else:
        top_idx = np.arange(len(b))
    active_list = sorted(set(int(x) for x in top_idx))
    active_set = set(active_list)

    best_g = None
    for cg_iter in range(25):
        if time.monotonic() - t0 > time_budget:
            break

        g = solve_lp(f, M, active_list, n, max_constraints)
        if g is None:
            if best_g is not None:
                break
            return None

        best_g = g.copy()

        # Check for violated constraints
        if time.monotonic() - t0 > time_budget * 0.8:
            break

        fg = fast_cross_convolve(f, g)
        mask = np.zeros(len(b), dtype=bool)
        for idx in active_list:
            if idx < len(b):
                mask[idx] = True
        violations = np.where((fg > M + 1e-10) & (~mask))[0]

        if len(violations) == 0:
            break

        # Add most violated
        worst = np.argsort(fg[violations])[-min(10, len(violations)):]
        added = 0
        for v in worst:
            vi = int(v)
            if vi not in active_set:
                active_set.add(vi)
                active_list.append(vi)
                added += 1
        if added == 0:
            break

    return best_g


def trust_region_lp(f, n, active_indices, delta_max=0.05):
    """
    Trust region LP: find perturbation delta that minimizes max_k (f*delta)[k]
    subject to ||delta|| <= delta_max, f + delta >= 0.
    """
    b = fast_convolve(f)
    M = float(b.max())

    # Right-hand side for constraints
    rhs = np.zeros(len(active_indices))
    for i, k in enumerate(active_indices):
        rhs[i] = (M - b[k]) / 2.0  # (f*delta)[k] <= (M - (f*f)[k])/2

    A = build_constraint_matrix(n, f, active_indices)

    delta_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        A @ delta_var <= rhs,
        -f <= delta_var,  # f + delta >= 0
        cp.norm(delta_var, 2) <= delta_max
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints + [t_var >= cp.max(A @ delta_var)])

    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=200000)
            if delta_var.value is not None:
                d = delta_var.value
                if np.isfinite(d).all():
                    return d
        except Exception:
            continue
    return None


def peak_equalization(f, n, max_iter=50, tol=1e-10):
    """
    Try to equalize the top peaks by small adjustments.
    This is a heuristic: find the top K peaks and try to make them equal
    by redistributing mass.
    """
    f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    K = 10

    for it in range(max_iter):
        b = fast_convolve(f)
        top_idx = np.argpartition(b, -K)[-K:]
        top_vals = b[top_idx]
        max_val = top_vals.max()
        min_val = top_vals.min()
        spread = max_val - min_val

        if spread < tol:
            break

        # For each peak that is above average, try to reduce it
        avg_val = top_vals.mean()
        target = max_val - spread * 0.3  # Move 30% of the way

        improved = False
        for k in top_idx:
            if b[k] < target:
                continue
            # Find the pair (j, k-j) that contributes most to (f*f)[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            products = f[j_range] * f[k - j_range]
            j_max = j_range[np.argmax(products)]

            # Reduce f[j_max] and f[k-j_max] slightly
            j1 = j_max
            j2 = k - j_max
            if j1 == j2:
                reduce = f[j1] * 0.001
                f[j1] -= reduce
            else:
                reduce = 0.0005 * max(f[j1], f[j2])
                f[j1] -= reduce
                f[j2] -= reduce
            improved = True

        if not improved:
            break

        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:
            break
        f /= s

        cur_score = 2.0 * n * fast_convolve(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    return best_f, best_score / (2.0 * n)


def golden_line_search(f, direction, max_iters=25):
    """
    Golden section search for optimal t in f_new = (1-t)*f + t*direction.
    Objective: peak of auto-convolution.
    """
    direction = np.maximum(direction, 0)
    s_dir = direction.sum()
    if s_dir < 1e-15:
        return 0.0, f

    direction = direction / s_dir

    def obj_at(t):
        f_t = (1.0 - t) * f + t * direction
        f_t = np.maximum(f_t, 0)
        s = f_t.sum()
        if s < 1e-10:
            return 1e10
        f_t = f_t / s
        return fast_convolve(f_t).max()

    lo, hi = 0.01, 0.5
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(max_iters):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-5:
            break

    t_best = (lo + hi) / 2
    return t_best, None


def subgradient_step(f, n, lr=0.001, top_k=30, momentum=0.8):
    """Single subgradient step on the peak objective."""
    b = fast_convolve(f)
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))

    # Soft weights based on proximity to max
    b_top = b[top_idx]
    scale = max(b_top.max() - b_top.min(), 1e-10)
    w = np.exp((b_top - b_top.max()) / (scale * 0.1))
    w /= w.sum()

    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        grad[j_range] += w[i] * f[k - j_range]

    return grad


def run_optimization_phase(f_init, n, time_budget, label="phase"):
    """
    Main optimization loop: LP with constraint generation + line search.
    Returns best f found and its peak value.
    """
    t0 = time.monotonic()
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = float(b.max())
    stuck = 0
    it = 0
    top_k = 40

    while time.monotonic() - t0 < time_budget:
        it += 1
        rem = time_budget - (time.monotonic() - t0)
        if rem < 5:
            break

        # Adaptively increase top_k
        if it % 30 == 0 and top_k < 200:
            top_k = min(top_k + 10, 200)

        # Solve LP with constraint generation
        lp_time = min(rem * 0.6, 60.0)
        g = lp_with_constraint_generation(f, n, time_budget=lp_time,
                                          top_k_init=top_k, max_constraints=500)

        if g is None:
            # Fallback: subgradient direction
            b_cur = fast_convolve(f)
            n_pk = min(5, len(b_cur))
            top_idx = np.argpartition(b_cur, -n_pk)[-n_pk:]
            g = np.zeros(n)
            for k in top_idx:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                j_range = np.arange(lo, hi + 1)
                g[j_range] += f[k - j_range]

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 30:
                break
            top_k = min(top_k * 2, 300)
            continue

        h = g / S_g  # Normalize direction

        # Line search
        t_best, _ = golden_line_search(f, h, max_iters=20)

        if t_best < 0.001:
            # Try smaller steps
            for t_try in [0.01, 0.02, 0.05]:
                f_trial = (1.0 - t_try) * f + t_try * h
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    t_best = t_try
                    break

        if t_best < 0.001:
            stuck += 1
            if stuck > 40:
                break
            continue

        # Update
        f = (1.0 - t_best) * f + t_best * h
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label}: it={it} obj={best_obj:.12f} score={2*n*best_obj:.8f} t={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_polish(f_init, time_budget, top_k=30, momentum_beta=0.85):
    """Subgradient optimization with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = 0.001
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_budget:
        batch = min(200, max(50, int((time_budget - (time.monotonic() - t0)) * 3)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_budget:
                break
            it += 1
            grad = subgradient_step(f, n, lr=1.0, top_k=top_k)
            v = momentum_beta * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.005)
        else:
            no_improve += 1
            if no_improve > 3:
                lr *= 0.9
                no_improve = 0
            if no_improve > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            print(f"    sg: it={it} obj={best_obj:.12f} score={2*n*best_obj:.8f}")

    return best_f, best_obj


def flat_top_window(n):
    """Flat top window, normalized."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Hann window, normalized."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s
