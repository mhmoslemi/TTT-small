        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = f - step
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.85
                no_improve = 0

        f = f_new

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                  f"tau={tau:.6e} lr={lr:.6e} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, n_mults=4):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step_size = max(1, n // 300)
    multipliers = [0.7, 0.85, 1.15, 1.3]

    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step_size):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            orig = f[j]
            best_val = orig
            best_obj_j = best_obj
            for mult in multipliers:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = orig * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def try_length(f_best, target_n, time_limit, label="len"):
    n_cur = len(f_best)
    old_x = np.linspace(0.0, 1.0, n_cur)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f_best)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-10:
        f_interp /= s
    else:
        f_interp = np.ones(target_n) / target_n

    # Run a short LP optimization
    f_opt, obj = cp_lp_optimize(f_interp, time_limit=min(time_limit, 200),
                                initial_k=30, max_k=200, label=label)
    return f_opt, obj
