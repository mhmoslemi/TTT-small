"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k] s.t. b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1, a>=0."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lbgfs_stage(a, T_val, maxiter, time_limit, t_start):
    """Single L-BFGS stage at fixed temperature T_val."""
    n = len(a)
    z = z_from_a(a).copy()

    def func_and_jac(z):
        if time.monotonic() - t_start > time_limit:
            a_s = softmax_from_z(z)
            return 1e15, np.zeros(n)
        a_s = softmax_from_z(z)
        b = fast_convolve(a_s)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        f_val = (2.0 * n / T_val) * log_sum_exp

        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_s[k - idx]
        S = np.dot(grad_a, a_s)
        grad_z = a_s * (grad_a - S)
        return f_val, grad_z

    try:
        fun = lambda z: func_and_jac(z)[0]
        jac = lambda z: func_and_jac(z)[1]
        res = minimize(fun, z, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 200, 'disp': False})
        return softmax_from_z(res.x)
    except Exception:
        return a

def nesterov_lp(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov dual averaging LP alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    prev_score = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t_start > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = compute_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            elapsed = time.monotonic() - t_start
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_rounds=500):
    """Coordinate descent: adjust one element at a time to reduce max convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()

        if time.monotonic() - t_start > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributing indices to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing top contributors
        improved = False
        order = np.argsort(contribs)[::-1][:20]
        for ci in order:
            j = idx[ci]
            if a[j] < 1e-14:
                continue
            # Try different reduction factors
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
                a_trial = a.copy()
                a_trial[j] *= factor
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # If no single coordinate helped, try pair adjustments on near-peaks
        if not improved:
            near_thresh = max_b - 0.05 * max_b
            near_peaks = np.where(b >= near_thresh)[0][:5]
            pair_improved = False
            for k in near_peaks:
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                idx2 = np.arange(js2, je2 + 1)
                for ci in np.argsort(idx2)[:5]:
                    j = idx2[ci]
                    if a[j] < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[j] *= 0.97
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        pair_improved = True
                        break
                if pair_improved:
                    break

        if not improved:
            # Try increasing low-value positions (to spread mass)
            low_positions = np.where(a < np.median(a))[0]
            if len(low_positions) > 0:
                j = low_positions[rnd % len(low_positions)]
                a_trial = a.copy()
                a_trial[j] *= 1.05
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:

                    best_score = sc
                    best_a = a.copy()
                    improved = True

                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_score * 1.01:
                    best_a = a.copy()
                    if sc < best_score:
                        best_score = sc

        if rnd % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CoordDesc round={rnd} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def spectral_smoothing(a0, time_limit, n_steps=2000):
    """Spectral domain smoothing to reduce convolution peaks."""
    n = len(a0)
    z = z_from_a(np.maximum(a0, 0)).copy()
    t_start = time.monotonic()
    lr = 0.02
    best_a = softmax_from_z(z).copy()
    best_score = compute_score(best_a)

    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break

        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()

        # Focus on peaks: use top 20% of convolution values
        thresh = np.percentile(b, 80)
        active = b >= thresh

        # Compute subgradient from active peaks
        w = active.astype(np.float64) / active.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        z = z - lr * grad_z
        a_new = softmax_from_z(z)
        sc = compute_score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        else:
            z = z_from_a(best_a)

        # Adaptive learning rate

            elapsed = time.monotonic() - t_start
            print(f"    Spectral step={step} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.5f}")

    return best_a, best_score
