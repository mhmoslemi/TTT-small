"""
Adam-based optimization of step function auto-convolution peak.
Uses log-sum-exp smoothing with temperature annealing, FFT-based gradients,
and multiple sequence lengths.
"""

import numpy as np
import time

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def _fft_conv_size(n):
    """FFT size for linear auto-convolution of length-n signal."""
    return _next_pow2(2 * n - 1)

def _fft_cross_size(n):
    """FFT size for cross-convolution of length (2n-1) and length n signals."""
    total = 3 * n - 2  # length of full cross-conv
    return _next_pow2(total)

def _fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    sz = _fft_conv_size(n)
    A = np.fft.rfft(a, sz)
    return np.fft.irfft(A * A, sz)[:2*n-1]

def _compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _fast_conv(a)
    return 2.0 * n * conv.max()

def _compute_grad(a, T):
    """
    Compute gradient of 2n * (1/T) * log(sum(exp(T * conv(a,a))))
    assuming a >= 0 and sum(a) = 1.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = _fast_conv(a)
    # Numerically stable log-sum-exp
    max_c = conv.max()
    exp_c = np.exp(T * (conv - max_c))
    s = exp_c.sum()
    w = exp_c / s  # softmax weights, length 2n-1
    
    # Gradient: grad[m] = 2n * sum_k w[k] * (a[k-m] + a[m-k])
    # = 2n * (part1[m] + part2[m])
    # where part2 = first n of convolve(w, a)
    # and part1 = elements [n-1 : 2n-1] of convolve(w, a[::-1])
    sz = _fft_cross_size(n)
    W = np.fft.rfft(w, sz)
    A = np.fft.rfft(a, sz)
    conv_wa = np.fft.irfft(W * A, sz)[:n]  # part2
    Af = np.fft.rfft(a[::-1], sz)
    conv_waf = np.fft.irfft(W * Af, sz)[n-1:2*n-1]  # part1
    return 2.0 * n * (conv_waf + conv_wa)

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} (approximate)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

def _adam_optimize(a0, time_limit, T_start=100.0, T_end=1000.0,
                   lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                   noise_interval=300, noise_std=0.001, label="Adam"):
    """
    Adam optimization on the log-sum-exp smoothed objective.
    Temperature anneals from T_start to T_end.
    """
    n = len(a0)
    a = _project_simplex(a0.astype(np.float64))
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        progress = min(1.0, t / max(1, int(time_limit * 3000)))  # rough estimate
        # Actually, let's use a fixed number of "full anneals"
        # Or just linearly interpolate based on time
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T_cur = T_start + (T_end - T_start) * frac
        
        # Compute gradient
        grad = _compute_grad(a, T_cur)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)
        
        # Check score
        score = _compute_score(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Periodic noise to escape local minima
        if t % noise_interval == 0 and stall > noise_interval // 2:
            rng = np.random.default_rng(t)
            a = best_a + rng.standard_normal(n) * noise_std
            a = _project_simplex(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0
        
        # Verbosity
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} t={t}, best={best_score:.8f}, T={T_cur:.1f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score

"""
Adam-based optimization with log-sum-exp smoothing, FFT gradients,
temperature annealing, and multi-length search.
Combines subgradient methods with smoothed gradient descent for
minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_conv_size(n):
    return _next_pow2(2 * n - 1)

def _fft_cross_size(n):
    return _next_pow2(3 * n - 2)

def _fast_conv(a):
    n = len(a)
    sz = _fft_conv_size(n)
    A = np.fft.rfft(a, sz)
    return np.fft.irfft(A * A, sz)[:2 * n - 1]

def _compute_score(a):
    n = len(a)
    conv = _fast_conv(a)
    return 2.0 * n * conv.max()

def _compute_grad(a, T):
    n = len(a)
    conv = _fast_conv(a)
    max_c = conv.max()
    exp_c = np.exp(T * (conv - max_c))
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_c / s
    sz = _fft_cross_size(n)
    W = np.fft.rfft(w, sz)
    A = np.fft.rfft(a, sz)
    conv_wa = np.fft.irfft(W * A, sz)[:n]
    Af = np.fft.rfft(a[::-1], sz)
    conv_waf = np.fft.irfft(W * Af, sz)[n - 1:2 * n - 1]
    return 2.0 * n * (conv_waf + conv_wa)

def _subgrad_dir(a):
    """Return subgradient direction at the peak of conv(a,a)."""
    n = len(a)
    conv = _fast_conv(a)
    peaks = np.where(conv >= conv.max() - 1e-12)[0]
    peak = peaks[0]
    # d conv[peak] / da[m] = a[peak-m] + a[m-peak]
    # For m in 0..n-1:
    # a[peak-m] is valid when 0 <= peak-m <= n-1
    # a[m-peak] is valid when 0 <= m-peak <= n-1
    s = np.zeros(n)
    # First term: a[peak - m] for m where 0 <= peak-m <= n-1
    lo1 = max(0, peak - (n - 1))
    hi1 = min(n - 1, peak)
    if lo1 <= hi1:
        s[lo1:hi1 + 1] += a[peak - hi1:peak - lo1 + 1][::-1]
    # Second term: a[m - peak] for m where 0 <= m-peak <= n-1
    lo2 = max(0, peak)
    hi2 = min(n - 1, peak + n - 1)
    if lo2 <= hi2:
        s[lo2:hi2 + 1] += a[lo2 - peak:hi2 - peak + 1]
    return s

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

def _interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _adam_optimize(a0, time_limit, T_list, lr_list,
                   noise_interval=500, noise_std=0.002, label="Adam"):
    """
    Adam with phased temperature. T_list and lr_list define phases.
    """

    a = _project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()

    for phase_idx, (T_cur, lr) in enumerate(zip(T_list, lr_list)):
        phase_time = time_limit / len(T_list) if len(T_list) > 0 else time_limit
        phase_end = t0 + (phase_idx + 1) * phase_time
        if phase_idx > 0:
            a = best_a.copy()

        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0

        while time.monotonic() < phase_end:
            grad = _compute_grad(a, T_cur)
            t += 1
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = a - step
            a_new = _project_simplex(a_new)

            score = _compute_score(a_new)
            if score < best_score - 1e-12:

                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

            a = a_new

            if t % noise_interval == 0 and stall > noise_interval // 3:
                rng = np.random.default_rng(1000 + t)
                a = best_a + rng.standard_normal(n) * noise_std

                m = np.zeros(n)

    return best_a, best_score

def _subgrad_optimize(a0, time_limit, label="Subgrad"):
    """Subgradient descent on the true max with diminishing step size."""
    n = len(a0)
    a = _project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = _compute_score(a)
    t0 = time.monotonic()
    t = 0
    c = 0.5

    while time.monotonic() - t0 < time_limit:
        t += 1
        s = _subgrad_dir(a)
        s_norm = np.linalg.norm(s)
        if s_norm < 1e-15:
            break
        lr = c / np.sqrt(t)
        a_new = a - lr * s / s_norm
        a_new = _project_simplex(a_new)
        score = _compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            c = min(c * 1.01, 2.0)
        else:
            c = max(c * 0.999, 0.01)
        a = a_new

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} t={t} best={best_score:.8f} c={c:.4f} "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score

def _lp_best_response_fast(f, M, n_iter=20):
    """
    Approximate LP best response using multiplicative weights (proximal).
    Solve: max sum(g) s.t. C(f)@g <= M, g >= 0

    """

    g = np.ones(n) / n
    # y is the dual variable (weights on constraints)
    y = np.ones(2 * n - 1) / (2 * n - 1)
    eta = 0.5

        # C[k,j] = f contribution to conv[k] from g[j]

        # Actually let's just compute it directly
        # (C^T y)[j] = sum_k y[k] * f[max(0, min(n-1, k-j))]  -- no, f[k-j] with f=0 outside
        # = cross-correlation of y and f at lag j
        ct_y = np.fft.irfft(Y * np.fft.rfft(f, sz), sz)[:n]
        # Wait, that's convolve not cross-correlation. Let me think...
        # (C^T y)[j] = sum_k C[k,j] * y[k]
        # C[k,j]: the contribution of g[j] to conv[k] is f[k-j] (if 0<=k-j<n)
        # So (C^T y)[j] = sum_{k=j}^{j+n-1} y[k] * f[k-j] = sum_{i=0}^{n-1} y[j+i] * f[i]
        # This is the cross-correlation of y and f at lag j (with y being the longer signal)
        # In FFT: irfft(fft(y) * conj(fft(f))) gives cross-correlation
        Yf = np.fft.rfft(y, sz)

        ct_y = np.fft.irfft(Yf * np.conj(Ff), sz)[:n]

        ratio = ct_y  # should be <= 1 at optimum

        g = g / g.sum()

        # Update y: proportional to constraint violation
        conv_fg = _fast_conv(f)  # placeholder, need conv(f, g)

        sz2 = _next_pow2(2 * n - 1)

        violation = np.maximum(conv_fg / M - 1.0, 0)
        y = y * (1 + eta * violation)
        y = y / y.sum()

    return g
