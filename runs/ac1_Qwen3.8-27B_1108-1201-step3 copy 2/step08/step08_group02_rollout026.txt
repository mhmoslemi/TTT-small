Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP direction finding with golden-section line search,
multiple restarts, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_score(a):
    """Return (max_conv, score) for a normalized sequence."""
    b = fast_convolve(a)
    M = b.max()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 0.0, np.inf
    score = 2.0 * n * M / (s * s)
    return M, score


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    A[mask] = f[idx[mask]]
    return A

                if not (np.any(np.isnan(g_val)) or np.any(np.isinf(g_val))):

    return None


def golden_section_search(f, g, n_evals_max=50):
    """
    Golden section search for min_t F((1-t)*f + t*g) where F(a) = max(conv(a,a)).
    t is in [0, 1].
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    a_lo = 0.0
    a_hi = 1.0
    
    # Initial evaluations
    x1 = a_lo + (1 - phi) * (a_hi - a_lo)
    x2 = a_lo + phi * (a_hi - a_lo)
    f1 = objective_at_point(f, g, x1)
    f2 = objective_at_point(f, g, x2)
    
    n_evals = 2

            x2 = x1
            f2 = f1
            x1 = a_lo + (1 - phi) * (a_hi - a_lo)

            f1 = f2

            f2 = objective_at_point(f, g, x2)
        n_evals += 1
        if a_hi - a_lo < 1e-10:
            break

    return t_opt

                  f"S_g={S_g:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def subgradient_optimize(f_init, time_limit, label="sg", lr_init=0.005, n_top=100):
    """Projected subgradient descent with adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = 0.01
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Softmax weights on top peaks
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Compute gradient: grad[j] = sum_k w[k] * f[k-j] for valid k
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[j] += w[k] * f[k-j] for j in [lo, hi]
            # Let i = k - j, so i ranges from k-hi to k-lo
            j_start = lo
            j_end = hi + 1
            i_start = k - j_end + 1
            i_end = k - j_start
            grad[j_start:j_end] += w[k] * f[i_start:i_end]
        
        # Momentum update
        v = 0.85 * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        if obj < best_b - 1e-12:
            best_b = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new
            no_improve += 1
            if no_improve > 20:
                lr *= 0.9
                no_improve = 0
                if lr < 1e-5:
                    # Reset with perturbation
                    f = best_f + rng.standard_normal(n) * 0.002
                    f = project_simplex(f)
                    v = np.zeros(n)
                    lr = lr_init
    
        if it % 200 == 0:
            temp *= 0.95
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_structured_init(n, seed=0):
    """Create a structured initial sequence."""
    rng = np.random.default_rng(seed)
    # Start with a smooth bump + noise
    x = np.linspace(0, 1, n)
    base = np.exp(-8 * (x - 0.5)**2)  # Gaussian bump

    seq = base + noise
    seq = np.maximum(seq, 0)
    s = seq.sum()
    if s > 1e-10:
        seq /= s
    return seq

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP direction finding with golden-section line search,
subgradient descent with momentum, multi-length exploration, and coordinate descent polishing.
Key improvements over baseline: (1) full LP with all 2n-1 convolution constraints,
(2) convex 1D golden-section line search, (3) adaptive multi-phase schedule.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    mask = (idx >= 0) & (idx < n)
    A[mask] = f[idx[mask]]
    return A


def solve_full_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k=0..2n-2."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=2000000, verbose=False)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not (np.any(np.isnan(g_val)) or np.any(np.isinf(g_val))):
                    return g_val
        except Exception:
            continue
    return None


def objective_at_point(f, g, t):
    """Compute max(conv(h,h)) where h = (1-t)*f + t*g, normalized to sum=1."""
    h = (1.0 - t) * f + t * g
    h = np.maximum(h, 0.0)
    s = h.sum()
    if s < 1e-10:
        return np.inf
    h = h / s
    b = fast_convolve(h)
    return b.max()


def golden_section_search(f, g, n_evals_max=45):
    """
    Golden section search for min_t max(conv(((1-t)*f+t*g), ...)) over t in [0,1].
    The function is convex along this line (max of quadratics in t).
    """
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0

    x1 = a_lo + (1.0 - phi) * (a_hi - a_lo)
    x2 = a_lo + phi * (a_hi - a_lo)
    f1 = objective_at_point(f, g, x1)
    f2 = objective_at_point(f, g, x2)
    n_evals = 2

    for _ in range(n_evals_max):
        if f1 < f2:
            a_hi = x2
            x2 = x1
            f2 = f1
            x1 = a_lo + (1.0 - phi) * (a_hi - a_lo)
            f1 = objective_at_point(f, g, x1)
        else:
            a_lo = x1
            x1 = x2
            f1 = f2
            x2 = a_lo + phi * (a_hi - a_lo)
            f2 = objective_at_point(f, g, x2)
        n_evals += 1
        if a_hi - a_lo < 1e-10:
            break

    t_opt = (a_lo + a_hi) / 2.0
    return t_opt

                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_bump_init(n, center=0.5, width=0.15, seed=0):
    """Create a Gaussian bump initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    base = np.exp(-((x - center) / width) ** 2)
    noise = rng.standard_normal(n) * 0.05
    seq = base + noise
    seq = np.maximum(seq, 0)
    s = seq.sum()
    if s > 1e-10:
        seq /= s
    return seq
