    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:

    return 2.0 * n * conv.max() / (s * s)

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Scale to sum=1 first, then handle non-negativity
    # Actually, project onto {x >= 0, sum(x) = s}
    # This is the same as s * project_1(v/s)
    v_normalized = v / s if s > 0 else v
    n = len(v_normalized)
    mu = np.sort(v_normalized)[::-1]
    rho = np.where(1 + mu * n > 0)
    rho = rho[0].size if len(rho[0]) > 0 else 0
    # Find threshold
    # Standard algorithm:
    cumsum = np.cumsum(mu)
    condition = mu - (cumsum - 1) / np.arange(1, n + 1) > 0
    if not condition.any():

    rho = np.where(condition)[0][-1] + 1
    theta = (cumsum[rho - 1] - 1) / rho
    w = np.maximum(v_normalized - theta, 0)
    return w * s

    return grad

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP fictitious play, projected subgradient with backtracking,
mirror descent, and intensive coordinate descent with line search.
Explores multiple sequence lengths for better global optimization.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    scaled = v / target_sum if target_sum > 0 else v
    mu = np.sort(scaled)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - 1.0) / rho > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    idx = np.where(cond)[0][-1]
    theta = (cv[idx] - 1.0) / (idx + 1)
    w = np.maximum(scaled - theta, 0.0)
    return w * target_sum


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    col = np.zeros(2 * n - 1)
    col[:n] = f
    row = np.zeros(n)
    row[0] = f[0]
    return toeplitz(col, row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def solve_best_response_min(f, prev_g=None):
    """Solve: min max_k (f*g)_k s.t. g >= 0, sum(g) = 1."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    t = cp.Variable()
    constraints = [C @ g <= t, cp.sum(g) == 1]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    g_val = np.maximum(g.value, 0)
    gs = g_val.sum()
    if gs < 1e-15:
        return f.copy()
    return g_val / gs


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP"):
    """LP-based fictitious play."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        if stall_count > 200:
            step = max(0.05, step * 0.8)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0003
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, t={elapsed:.0f}s/{time_limit:.0f}s")
    return best_a, best_score


def projected_subgradient(a0, time_limit, verbose=True, label="PGD"):
    """Projected subgradient with smoothed peaks and backtracking."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.01
    T_peaks = 2.0  # temperature for peak smoothing
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        # Smoothed gradient: weighted average of subgradients at top peaks
        shifted_conv = conv - M
        weights = np.exp(shifted_conv / T_peaks)
        weights /= weights.sum()
        # Compute smoothed gradient
        grad = np.zeros(n)
        # Use FFT-based approach for efficiency
        # grad_i = 2 * sum_k weights[k] * a[k-i] (if 0<=k-i<n)
        # This is 2 * (weights reversed) convolved with a, truncated
        w_rev = weights[::-1]
        # (w_rev * a) gives sum_j w_rev[j] * a[j] at each position
        # We need: for each i, sum_k w[k] * a[k-i]
        # = sum_k w[k] * a[k-i] = (w * a_flipped)[n-1-i] ... 
        # Actually: sum_k w[k] * a[k-i] where 0<=k-i<n means i<=k<=i+n-1
        # This is a cross-correlation of w and a
        # Let's use FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w_rev, fft_size)
        A = rfft(a, fft_size)
        # conv(w_rev, a)[i] = sum_j w_rev[j] * a[i-j] = sum_j w[2n-2-j] * a[i-j]
        # We want sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = cross-correlation
        # Let's compute directly: grad[i] = 2 * sum_{k=max(i,0)}^{min(i+2n-2, 2n-2)} w[k]*a[k-i] (with a[j]=0 outside [0,n-1])
        # = 2 * sum_{j=0}^{n-1} w[i+j]*a[j] (where w[i+j] valid)
        # This is 2 * (w padded) correlated with a
        # FFT approach: if we compute ifft(conj(FFT(w)) * FFT(a)), we get cross-correlation
        Wfull = rfft(np.concatenate([weights, np.zeros(fft_size - len(weights))]), fft_size)
        Afull = rfft(np.concatenate([a, np.zeros(fft_size - n)]), fft_size)
        cross = irfft(np.conj(Wfull) * Afull, fft_size)
        # cross[i] = sum_j conj(w[j-i]) * a[j] = sum_j w[j-i] * a[j] (w real)
        # We want sum_j w[i+j] * a[j] = cross at position -i = cross[fft_size - i] or circular
        # Actually for real sequences: cross_corr[i] = sum_j w[j] * a[j-i] 
        # Let me just do it with a different approach
        # grad[i] = 2 * sum_{k} w[k] * a[k-i] for 0 <= k-i < n
        # Let's define b[k] = w[k] for k in [0, 2n-2], 0 otherwise
        # Then grad[i] = 2 * sum_j b[i+j] * a[j] = 2 * (cross-correlation of b and a at lag i)
        # xcorr[i] = sum_j b[i+j]*a[j]
        # This equals conv(b, a_flipped)[n-1+i]
        a_flip = a[::-1]
        b_full = np.zeros(fft_size)
        b_full[:2*n-1] = weights
        B = rfft(b_full, fft_size)
        Af = rfft(a_flip, fft_size)
        conv_result = irfft(B * Af, fft_size)
        grad = 2.0 * conv_result[n-1:n-1+n]

        # Project gradient descent
        for trial_eta in [eta, eta*0.5, eta*0.25, eta*1.5, eta*2]:
            a_trial = a - trial_eta * grad
            a_trial = project_simplex(a_trial, 1.0)
            sc = compute_score(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
            if sc < compute_score(a):
                a = a_trial
                eta = min(eta * 1.1, 0.1)
                break
        else:
            eta = max(eta * 0.7, 0.001)

        if verbose and iters % 100 == 0:
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.5f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_score


def mirror_descent(a0, time_limit, verbose=True, label="MD"):
    """Mirror descent (exponentiated gradient) on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.1
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        # Subgradient at the peak
        k_star = int(np.argmax(conv))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        # Smoothed: use top peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:max(1, n // 4)]
        w_vals = conv[top_k]
        w_max = w_vals.max()
        w_nrm = np.exp((w_vals - w_max) / 0.5)
        w_nrm /= w_nrm.sum()
        grad_smooth = np.zeros(n)
        for ww, kk in zip(w_nrm, top_k):
            for i in range(n):
                j = int(kk) - i
                if 0 <= j < n:
                    grad_smooth[i] += ww * 2.0 * a[j]

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad_smooth
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            eta = min(eta * 1.05, 0.5)
        else:
            eta = max(eta * 0.9, 0.01)
        a = a_new

        if verbose and iters % 100 == 0:
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.4f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_score


def coordinate_descent_intensive(a0, time_limit, n_pairs=500, verbose=True, label="CD"):
    """Intensive coordinate descent with line search on pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Try transferring from i to j and vice versa
            base = min(a[i], a[j]) if a[i] > 1e-15 and a[j] > 1e-15 else min(a[i], a[j]) + 0.01
            best_local = best_score
            best_t = 0.0
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                t = frac * base
                if t >= a[i] * 0.95:
                    continue
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 10:
            # Perturb
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            iters = 0

        if verbose and iters % 50 == 0:
            print(f"  {label} iter={iters}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_score


def subgradient_refined(a0, time_limit, verbose=True, label="SG-refined"):
    """Refined subgradient: use top-k peaks with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = 0.005
    T = 0.1
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        # Find peaks and compute smoothed gradient
        shifted = conv - M
        # Adaptive temperature: smaller when converged
        T_current = max(0.01, T * (1.0 / (1.0 + iters / 100)))
        w = np.exp(shifted / T_current)
        w /= w.sum()

        # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (if valid)
        # Use FFT for efficiency
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        w_pad = np.zeros(fft_size)
        w_pad[:2*n-1] = w
        w_pad_flipped = w_pad[::-1].copy()
        w_pad_flipped[0] = w_pad[0]  # keep DC
        W = rfft(w_pad, fft_size)
        A = rfft(a, fft_size)
        # We want: grad[i] = 2 * sum_k w[k] * a[k-i] for k-i in [0,n-1]
        # = 2 * (w * a_rev) circular conv, then extract
        # Actually: let's compute xcorr = ifft(conj(FFT(w)) * FFT(a))
        # xcorr[i] = sum_j conj(w[j-i]) * a[j] = sum_j w[j-i] * a[j] (w real, symmetric-ish)
        # Hmm, let me just do: grad[i] = 2 * sum_j a[j] * w[i+j] (where w defined on [0, 2n-2])
        # This is the cross-correlation of a with w, evaluated at lag i
        # xcorr_a_w[i] = sum_j a[j] * w[i+j]
        # = conv(a, w_flipped)[n-1+i] where w_flipped = w[::-1]
        w_flip = np.zeros(fft_size)
        n_w = 2 * n - 1
        w_flip[n - 1 : n - 1 + n_w] = w[::-1]
        Wf = rfft(w_flip, fft_size)
        xcorr = irfft(A * Wf, fft_size)
        grad = 2.0 * xcorr[n - 1 : n - 1 + n]

        # Project and take step
        a_trial = a - eta * grad
        a_trial = project_simplex(a_trial, 1.0)
        sc = compute_score(a_trial)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            eta = min(eta * 1.02, 0.05)
            stall = 0
        else:
            eta = max(eta * 0.85, 0.0005)
            stall += 1
            if stall > 100:
                # Reseed from best with small perturbation
                rng = np.random.default_rng(iters)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 0.005
                stall = 0

        if verbose and iters % 200 == 0:
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.6f}, T={T_current:.4f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:

    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    print(f"\n=== Phase 4: Mirror Descent ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = mirror_descent(best_seq, t4, label="MD")
        best_seq, best_score = check_and_update(a4, "mirror", best_seq, best_score)

    # === Phase 5: Intensive Coordinate Descent ===
    print(f"\n=== Phase 5: Intensive CD ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if rem > 30:
        print(f"  Budget: {t5:.0f}s")
        a5, s5 = coordinate_descent_intensive(best_seq, t5, n_pairs=400, label="CD")
        best_seq, best_score = check_and_update(a5, "coord_desc", best_seq, best_score)

    # === Phase 6: LP with small step (polish) ===
    print(f"\n=== Phase 6: LP Polish (small step) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 200.0)
    if rem > 30:
        print(f"  Budget: {t6:.0f}s")
        a6, s6 = lp_fictitious_play(best_seq, t6, step=0.1, label="LP-polish")
        best_seq, best_score = check_and_update(a6, "lp_polish", best_seq, best_score)

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    alt_lengths = [n_parent + 32, n_parent + 64, n_parent + 128, n_parent - 32]
    alt_lengths = [l for l in alt_lengths if 64 <= l <= 4096]
    # Also try powers of 2
    for pn in [1024, 1536, 2048]:
        if pn not in alt_lengths and 64 <= pn <= 4096:
            alt_lengths.append(pn)

    for alt_n in alt_lengths[:3]:  # Limit to 3 alternative lengths
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t7 = min(rem * 0.20, 200.0)
        if t7 < 30:
            break
        print(f"  Trying n={alt_n}, budget={t7:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick LP
        t_lp = t7 * 0.5
        t_sg = t7 * 0.3
        t_cd = t7 * 0.2
        a_alt, sc_alt = lp_fictitious_play(init_alt, t_lp, step=0.4, label=f"LP-n{alt_n}")
        a_alt, sc_alt = subgradient_refined(a_alt, t_sg, label=f"SG-n{alt_n}")
        a_alt, sc_alt = coordinate_descent_intensive(a_alt, t_cd, n_pairs=200, label=f"CD-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = rem * 0.5
        print(f"  Final subgradient, budget={t8:.0f}s")
        a8, s8 = subgradient_refined(best_seq, t8 * 0.4, label="SG-final")
        best_seq, best_score = check_and_update(a8, "sg_final", best_seq, best_score)

        rem = deadline - time.monotonic()
        if rem > 30:
            t8b = rem * 0.5
            print(f"  Final CD, budget={t8b:.0f}s")
            a8b, s8b = coordinate_descent_intensive(best_seq, t8b, n_pairs=500, label="CD-final")
            best_seq, best_score = check_and_update(a8b, "cd_final", best_seq, best_score)

        rem = deadline - time.monotonic()
        if rem > 20:
            t8c = rem * 0.6
            print(f"  Final LP, budget={t8c:.0f}s")
            a8c, s8c = lp_fictitious_play(best_seq, t8c, step=0.05, label="LP-final")
            best_seq, best_score = check_and_update(a8c, "lp_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs) and fs < best_score:
                best_score = fs
        except Exception:
            pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
