    # R_aw(i) = sum_j a_j * w_{i+j} for i in [0, 2n-2]
    # = cross-correlation

    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad_a = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]

    """Adam update. params and grads are dicts of arrays."""
    m = state['m']
    v = state['v']
    t = state['t']
    t += 1
    
    m_new = {}
    v_new = {}
    for key in params:
        m_new[key] = beta1 * m[key] + (1 - beta1) * grads[key]
        v_new[key] = beta2 * v[key] + (1 - beta2) * grads[key] ** 2
    
    params_new = {}
    for key in params:
        m_hat = m_new[key] / (1 - beta1 ** t)
        v_hat = v_new[key] / (1 - beta2 ** t)
        params_new[key] = params[key] - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    state['m'] = m_new
    state['v'] = v_new
    state['t'] = t
    return params_new

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):

    n = len(a)
    s = a.sum()
    if s < 1e-15:

    a = a / s
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_obj_and_grad(a, T):
    """
    Compute smoothed objective f_T(a) = 2n*T*log(sum exp(conv_k/T))
    and gradient grad_a_i = 2 * sum_j a_j * w_{i+j}
    where w_k = exp(conv_k/T) / sum exp(conv_j/T)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    logits = (conv - max_c) / T
    exp_logits = np.exp(logits)
    w = exp_logits / exp_logits.sum()

    W = rfft(w, fft_size)

def log_space_gradient(grad_a, a):
    """Project gradient from a-space to log-space (tangent space of simplex)."""
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def adam_step(z, grad_z, state, lr):

    v_hat = state['v'] / (1 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new

        # Adaptive learning rate: decrease as T decreases
        lr = lr_base * (0.3 + 0.7 * (1 - frac))
        z = adam_step(z, grad_z, state, lr)

        # Convert back to simplex
        z_max = z.max()
        a = np.exp(z - z_max)
        a = a / a.sum()

        # Check score periodically (expensive)
        if it % 10 == 0 or it == n_steps - 1:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = compute_score(a)
            print(f"    GF[{it}/{n_steps}] T={T:.2f} score={score:.8f} best={best_score:.8f} t={elapsed:.0f}s")

    return best_a, best_score


def annealing_schedule(T0, T1, n_phases, steps_per_phase):
    """Create a multi-phase annealing schedule."""
    schedule = []
    for p in range(n_phases):
        T_start = T0 * (T1 / T0) ** (p / n_phases)
        T_end = T0 * (T1 / T0) ** ((p + 1) / n_phases)
        for s in range(steps_per_phase):
            frac = s / max(1, steps_per_phase - 1)
            schedule.append(T_start + (T_end - T_start) * frac)
        # Restart: jump back up slightly
        if p < n_phases - 1:
            schedule.append(T_start * 1.5)  # Small bump
    return schedule


def full_optimization(a_init, time_budget, verbose=True):
    """Run full multi-phase optimization on a single initialization."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if verbose:
        print(f"  Starting: n={n}, score={best_score:.8f}")

    # Max convolution value for setting T
    conv = fast_convolve(a)
    max_conv = conv.max()
    mean_conv = conv.mean()

    # Phase 1: High temperature (explore flatness)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        T0 = max_conv * 3
        T1 = max_conv * 0.5
        a, sc = gradient_flow(a, T0, T1, 2000, lr_base=0.02,
                              time_budget=rem * 0.3, verbose=verbose)
        if sc < best_score:
            best_score, best_a = sc, a
        if verbose:
            print(f"  Phase1 done: score={best_score:.8f}")

    # Phase 2: Medium temperature (refine)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        conv = fast_convolve(best_a)
        max_conv = conv.max()
        T0 = max_conv * 0.5
        T1 = max_conv * 0.05
        a, sc = gradient_flow(best_a, T0, T1, 3000, lr_base=0.01,
                              time_budget=rem * 0.35, verbose=verbose)
        if sc < best_score:
            best_score, best_a = sc, a
        if verbose:
            print(f"  Phase2 done: score={best_score:.8f}")

    # Phase 3: Low temperature (focus on peaks)
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        conv = fast_convolve(best_a)
        max_conv = conv.max()
        T0 = max_conv * 0.05
        T1 = max_conv * 0.001
        a, sc = gradient_flow(best_a, T0, T1, 3000, lr_base=0.005,
                              time_budget=rem * 0.35, verbose=verbose)
        if sc < best_score:
            best_score, best_a = sc, a
        if verbose:
            print(f"  Phase3 done: score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_budget, verbose=True):
    """
    Coordinate descent: transfer mass between pairs to reduce the max peak.
    Uses the KKT structure: at optimum, pairs (i, k*-i) should have balanced products.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        if no_improve > 30:
            break

        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        current_max = conv[peak_idx]

        # Find contributing pairs for the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        contribs = {}
        for j in range(js, je + 1):
            contribs[j] = a[j] * a[peak_idx - j]

        # Sort by contribution (descending)
        sorted_pairs = sorted(contribs.items(), key=lambda x: -x[1])

        improved = False
        for j, contrib in sorted_pairs[:20]:
            if time.monotonic() - t0 > time_budget:
                break
            if a[j] < 1e-14:
                continue

            # Try transferring mass from j to a "safe" position
            # Safe = position that doesn't contribute to the top peak
            k_reflect = peak_idx - j  # The paired position

            # Find positions far from the peak's contribution range
            safe_positions = []
            for p in range(n):
                # p contributes to peak if p in [js, je] AND (peak_idx - p) in [js, je]
                if js <= p <= je and js <= (peak_idx - p) <= je:
                    continue
                if p == j:
                    continue
                safe_positions.append(p)

            if not safe_positions:
                continue

            # Try moving some mass from j to a safe position
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = a[j] * frac
                for target in safe_positions[:5]:
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[target] += delta
                    if a_trial[j] < 0 or a_trial[target] < 0:
                        continue
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            # Try random pair transfers
            for _ in range(50):
                if time.monotonic() - t0 > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-14:
                    continue
                for frac in [0.005, 0.01, 0.02]:
                    delta = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                # Perturb
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            if int(no_improve) % 30 == 0:
                print(f"    CD: score={best_score:.8f} no_improve={no_improve} t={elapsed:.0f}s")

    return best_a, best_score


def build_C(f):
    """Build constraint matrix for LP: C[k,j] = f[k-j] if valid."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def lp_best_response(f, M, use_scipy=True):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0)
    except Exception:
        return None


def lp_fictitious_play(a0, time_budget, verbose=True):
    """LP fictitious play with geometric mean updates."""
    n = len(a0)
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    a_avg = a.copy()

    for k in range(100000):
        if time.monotonic() - t0 > time_budget:
            break

        conv = fast_convolve(a)
        M = conv.max()
        g = lp_best_response(a, M)
        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Geometric mean update
        t_step = 2.0 / (k + 2)
        a_safe = np.maximum(a, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(a_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()

        a_avg = ((k * a_avg) + a_new) / (k + 1)

        sc_new = compute_score(a_new)
        sc_avg = compute_score(a_avg)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new

        if verbose and k % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{k}] M={M:.6e} best={best_score:.8f} t={elapsed:.0f}s")

    return best_a, best_score


def make_init_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) / n
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_raised_cosine(n, beta=1.0):
    """Raised cosine (Kaiser-like) initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    t = np.abs(x - center) / center
    a = np.cos(np.pi * beta * t / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_chirp(n, alpha=0.5, eps=0.3):
    """Chirp-like initialization."""
    x = np.arange(n)
    a = 1.0 + eps * np.cos(2 * np.pi * alpha * x ** 2 / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_init_uniform(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_score(a):
    """Evaluate using the official function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(evaluate_sequence((a / s).tolist()))

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
