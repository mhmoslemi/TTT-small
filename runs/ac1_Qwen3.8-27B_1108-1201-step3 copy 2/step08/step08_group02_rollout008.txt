from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_gradient(a, peak_indices):
    """Compute the 'convolution gradient' for given peak indices.
    grad[j] = sum over k in peak_indices of a[k-j] (valid indices only).
    """
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        # grad[j] += a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # j ranges from lo to hi
        # a[k-j] for j in [lo, hi] gives a[k-hi] to a[k-lo]
        # So grad[lo:hi+1] += a[k-hi : k-lo+1][::-1]
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return grad

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast multiplicative updates, LP polishing, simulated annealing,
and multi-start exploration with diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_pressure(a, peak_frac=0.9, n_peaks=200):
    """Compute the convolution pressure: for each coordinate j,
    how much it contributes to the top peaks of the autocorrelation.
    Returns (pressure_vector, M) where M = max(autocorr).
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    if M <= 0:
        return np.zeros(n), 0.0, b
    
    # Select peaks
    threshold = peak_frac * M
    peaks = np.where(b >= threshold)[0]
    if len(peaks) > n_peaks:
        peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
    if len(peaks) < 5:
        peaks = np.argpartition(b, -min(20, len(b)))[:min(20, len(b))]
        peaks = peaks[::-1]
    
    # Compute pressure
    pressure = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # pressure[j] += a[k-j] for j in [lo, hi]
            # a[k-j] for j=lo..hi => a[k-hi .. k-lo] reversed
            pressure[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    return pressure, M, b


def multiplicative_optimize(f_init, time_limit, lr_init=0.1, peak_frac_init=0.9,
                            peak_frac_final=0.7, n_peaks=200, label="mul"):
    """Fast multiplicative/subgradient optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

                                        (peak_frac_final - peak_frac_init) * min(1, it/500),
                                        n_peaks=n_peaks)
        
        # Multiplicative update: reduce coordinates with high pressure
        p_mean = pressure.mean()
        if p_mean > 1e-15:
            # a_new[j] = a[j] * exp(-lr * pressure[j] / p_mean)
            # Use a bounded version to avoid underflow
            ratio = np.exp(-lr * pressure / (p_mean + 1e-15))
            f_new = f * ratio
            s = f_new.sum()
            if s < 1e-10:
                break
            f_new /= s
            # Ensure positivity
            f_new = np.maximum(f_new, 1e-15)
            f_new /= f_new.sum()
            f = f_new
        else:
            break
        
        if it % 20 == 0:
            cur_b = fast_convolve(f).max()
            if cur_b < best_b - 1e-15:
                best_b = cur_b
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve == 3:
                    lr *= 0.9
                elif no_improve == 6:
                    lr *= 0.9
                    # Perturbation
                    f = f + rng.standard_normal(n) * 0.001 * f
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    no_improve = 0
                elif no_improve > 20:
                    f = best_f + rng.standard_normal(n) * 0.002 * best_f
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    lr = lr_init
                    no_improve = 0
            
            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} lr={lr:.4f} t={time.monotonic()-t0:.0f}s")
        
        # Gradually adapt learning rate
        if it % 100 == 0:
            lr = max(lr * 0.98, lr_init * 0.1)
    
    return best_f, best_b


def subgradient_momentum(f_init, time_limit, lr_init=0.005, n_peaks=100, label="sgm"):
    """Subgradient method with momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    beta = 0.7  # momentum
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Select top peaks
        n_sel = min(n_peaks, len(b))
        peaks = np.argpartition(b, -n_sel)[-n_sel:]
        
        # Compute subgradient (average over selected peaks)
        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += f[k-hi:k-lo+1][::-1]
        grad /= n_sel
        
        # Momentum update
        v = beta * v - lr * grad
        f_new = f + v
        
        # Project to simplex
        f_new = project_simplex_vec(f_new)
        f = f_new
        
        if it % 30 == 0:
            cur_b = fast_convolve(f).max()
            if cur_b < best_b - 1e-15:
                best_b = cur_b
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve == 5:
                    lr *= 0.9
                elif no_improve > 30:
                    f = best_f + rng.standard_normal(n) * 0.003 * best_f
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    v = np.zeros(n)
                    lr = lr_init
                    no_improve = 0
            
            if it % 300 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
        
        lr = max(lr * 0.999, lr_init * 0.05)
    
    return best_f, best_b


def project_simplex_vec(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, n, peaks, M, eps=1e-8):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in peaks."""
    K = len(peaks)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            solved = (prob.status in ['optimal', 'optimal_inaccurate'])
            if solved:
                break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.95,
                final_peak_frac=0.80, max_peaks=300):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(2024)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Adaptive peak selection
        frac = initial_peak_frac + (final_peak_frac - initial_peak_frac) * min(1.0, it / 200)
        threshold = frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -min(20, len(b)))[:min(20, len(b))]
            peaks = np.sort(peaks)[::-1]
        
        g = lp_direction(f, n, peaks, M)
        
        if g is None:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        # Line search
        h = g / S_g
        best_t = 0.0
        best_t_obj = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        if best_t <= 0:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 15:
            f = best_f + rng.standard_normal(n) * 0.001 * best_f
            f = np.maximum(f, 0.0)
            f /= f.sum()
            stuck = 0
        
        if it % 5 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def simulated_annealing(f_init, time_limit, label="sa", T_init=0.01, T_final=1e-5):
    """Simulated annealing on the simplex."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    rng = np.random.default_rng(777)
    
    total_time = time_limit
    n_iter = 1000  # approximate
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / total_time
        T = T_init * (T_final / T_init) ** progress
        
        # Propose perturbation
        k = rng.integers(0, 3)  # perturb 1, 2, or 3 coordinates
        idx = rng.integers(0, n, size=k)
        f_new = f.copy()
        for j in idx:
            f_new[j] *= np.exp(rng.normal(0, 0.3))
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s
        
        # Evaluate
        obj_new = fast_convolve(f_new).max()
        obj_cur = fast_convolve(f).max()
        delta = obj_new - obj_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            f = f_new
            if obj_new < best_b - 1e-15:
                best_b = obj_new
                best_f = f.copy()
        
        if it % 500 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} T={T:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b


def make_initial(length, seed=0, method="raised_cosine"):
    """Create various initial sequences."""
    n = length
    rng = np.random.default_rng(seed)
    
    if method == "uniform":
        f = np.ones(n) / n
    elif method == "raised_cosine":
        x = np.linspace(0, 2*np.pi, n)
        f = 0.5 * (1 + np.cos(x))
        f /= f.sum()
    elif method == "tent":
        center = n // 2
        f = np.abs(1 - 2 * np.arange(n) / (n - 1))
        f /= f.sum()
    elif method == "gaussian":
        x = np.linspace(-3, 3, n)
        f = np.exp(-x**2 / 2)
        f /= f.sum()
    elif method == "bump":
        x = np.linspace(0, 1, n)
        f = np.where((x > 0.2) & (x < 0.8), 1.0, 0.0)
        f += 0.1 * np.random.default_rng(seed).standard_normal(n)
        f = np.maximum(f, 0)
        f /= f.sum()
    elif method == "random":
        f = rng.random(n)
        f /= f.sum()
    elif method == "two_bump":
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.3)**2) / 0.02) + np.exp(-((x - 0.7)**2) / 0.02)
        f /= f.sum()
    elif method == "symmetric_random":
        half = n // 2
        f_half = rng.random(half)
        f = np.concatenate([f_half, f_half[::-1]])
        if n % 2 == 1:
            f = f[:n]
        f /= f.sum()
    else:
        f = np.ones(n) / n
    
    return f


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
