import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.signal import correlate as sp_correlate


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""

    if _lp_solver_cached is not None:

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

                _lp_solver_cached = solver_name
                return solver_name
        except Exception:

    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):

    if solver is None:

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:

                continue
            try:

                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def lbgfs_func(z, T_val, n, t0, time_limit):
    """L-BFGS objective: smoothed log-sum-exp of convolution."""
    if time.monotonic() - t0 > time_limit:
        return 1e15, None
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    val = (2.0 * n / T_val) * log_sum_exp
    return val, (a, b, exp_Tb, log_sum_exp)

    g = (4.0 * n / T_val) * g_raw
    S = np.dot(g, a)
    grad_z = a * (g - S)

    val = (2.0 * n / T_val) * (max_Tb + np.log(exp_Tb.sum()))
    return grad_z, val


def lbgfs_refine(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=10):
    """L-BFGS with multi-stage geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, 15000 // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_local):
            val, _ = lbgfs_func(z_local, stage_T, n, t0, time_limit)
            return val

        def jac(z_local):
            grad, _ = lbgfs_grad(z_local, stage_T, n, t0, time_limit)
            return grad

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False,
                         'maxls': 50}
            )
            z = res.x
            a = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a).max()
            print(f"    L-BFGS stage {stage_idx+1}/{n_stages} T={stage_T:.2f}: score={sc:.6f} ({time.monotonic()-t0:.0f}s)")
        except Exception as e:
            print(f"    L-BFGS stage {stage_idx+1} error: {e}")

    return softmax_from_z(z)


def peak_shave_ultra(a0, time_limit, n_steps=50000):
    """Ultra-fine peak shaving with pair and adaptive coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (excluding edges sometimes)
        b_work = b.copy()
        if len(b_work) > 4:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        sorted_peaks = np.argsort(b_work)[::-1]

        improved = False
        for peak_k in sorted_peaks[:3]:
            if b[peak_k] < best_score / (2.0 * n) * 0.95:
                continue

            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:8]

            for ci in top_contrib:
                i = idx[ci]
                if a[i] < 1e-14:
                    continue
                # Single coordinate reduction
                for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

                # Pair reduction
                j = peak_k - i
                if j != i and 0 <= j < n and a[j] > 1e-14:
                    for f1 in [0.99, 0.98, 0.97]:
                        for f2 in [0.99, 0.98, 0.97]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-15:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break

            if improved:
                break

        if not improved:
            # Try random perturbation
            n_pert = rng.integers(1, 4)
            perturb_idx = rng.integers(0, n, size=n_pert)
            a_trial = a.copy()
            for pi in perturb_idx:
                a_trial[pi] *= (1.0 + rng.uniform(-0.02, 0.02))
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, variant=0, rng=None):
    """Create diverse structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)

    x = np.linspace(0, 1, n)

    if variant == 0:
        a = np.cos(np.pi * x) ** 2
    elif variant == 1:
        a = np.exp(-4 * x * (1 - x))
    elif variant == 2:
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif variant == 3:
        a = np.sin(np.pi * x) ** 0.5
    elif variant == 4:
        # Super-Gaussian (flat top)
        center = 0.5
        sigma = 0.3
        a = np.exp(-((x - center) / sigma) ** 6)
    elif variant == 5:
        # Plateau with smooth edges
        a = np.ones(n)
        edge = max(3, n // 10)
        for i in range(edge):
            a[i] *= ((i + 1) / (edge + 1)) ** 0.8
            a[n - 1 - i] *= ((i + 1) / (edge + 1)) ** 0.8
    elif variant == 6:
        # Cosine raised to power (adjustable shape)
        p = rng.uniform(1.0, 3.0)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** p
    elif variant == 7:
        # Bimodal
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    elif variant == 8:
        # Watermelon shape (high at edges)
        a = 1.0 - 0.7 * np.sin(np.pi * x) ** 0.5
    else:
        # Random smooth
        coeffs = rng.normal(0, 1, size=min(n, 20))
        freqs = np.linspace(0, 5, len(coeffs))
        a = 1.0
        for c, f in zip(coeffs, freqs):
            a += c * np.cos(2 * np.pi * f * x)
        a = np.abs(a) ** 2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP best-response, L-BFGS with log-sum-exp, and ultra-fine peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = 960
    if parent is not None:
        n_base = len(parent)
    else:
        best_seq = structured_init(n_base, variant=0, rng=rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Candidate pool: (sequence, score, label)
    candidates = [(best_seq.copy(), best_score, "parent")]

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01 or len(a) == 0:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Multi-start LP at base length ===
    print(f"\n=== Phase 1: Multi-start LP n={n_base} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_starts = 6
    t_per_start = min(rem * 0.30 / n_starts, 90.0)

    for vi in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(t_per_start, rem * 0.5)
        if t_alloc < 15:
            break

        if vi == 0:
            init = best_seq.copy()
        else:
            init = structured_init(n_base, variant=vi % 9, rng=rng)

        a_lp, sc_lp = nesterov_lp_refine(init, t_alloc, solver=solver, n_max_iters=20000)
        candidates.append((a_lp.copy(), 2.0 * n_base * fast_convolve(a_lp).max(), f"lp_v{vi}"))
        try_update(a_lp, f"lp_v{vi}")

    # Sort candidates by score, keep top 3
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:3]
    print(f"  Top candidates after Phase 1: {[f'{c[2]}:{c[1]:.6f}' for c in top_candidates]}")

    # === Phase 2: L-BFGS on top candidates ===
    print(f"\n=== Phase 2: L-BFGS refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_lbgfs = min(3, len(top_candidates))
    t_per_lbgfs = min(rem * 0.35 / n_lbgfs, 280.0)

    for ci in range(n_lbgfs):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(t_per_lbgfs, rem * 0.6)
        if t_alloc < 30:
            break

        start_a = top_candidates[ci][0].copy()
        label = top_candidates[ci][2]
        print(f"  L-BFGS on {label} (n={len(start_a)}), budget={t_alloc:.0f}s")

        a_lb = lbgfs_refine(start_a, t_alloc, T_start=100.0, T_end=0.1, n_stages=10)
        try_update(a_lb, f"lbgfs_{label}")
        candidates.append((a_lb.copy(), 2.0 * len(a_lb) * fast_convolve(a_lb).max(), f"lbgfs_{label}"))

    # === Phase 3: Alternative lengths ===
    print(f"\n=== Phase 3: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = []
    for mult in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_base * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_base:
            alt_lengths.append(alt_n)
    # Also try rounded values
    for extra in [512, 896, 1024, 1152]:
        if extra not in alt_lengths and abs(extra - n_base) > 50:
            alt_lengths.append(extra)
    alt_lengths = sorted(set(alt_lengths))[:4]

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 50:
            break

        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)

        # LP phase
        t_lp = t_total * 0.55
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=15000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lb = min(rem * 0.3, 80.0)
            if t_lb > 20:
                a_alt2 = lbgfs_refine(a_alt, t_lb, T_start=50.0, T_end=0.5, n_stages=6)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 4: Ultra-fine peak shaving ===
    print(f"\n=== Phase 4: Ultra Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_ps = min(rem * 0.25, 150.0)
        if t_ps > 20:
            a_ps, _ = peak_shave_ultra(best_seq, t_ps, n_steps=100000)
            try_update(a_ps, "peak_shave_ultra")

    # === Phase 5: Low-temperature L-BFGS polish ===
    print(f"\n=== Phase 5: Low-T L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.3, 120.0)
        if t_fin > 15:
            a_fin = lbgfs_refine(best_seq, t_fin, T_start=20.0, T_end=0.05, n_stages=8)
            try_update(a_fin, "final_lbgfs_lowT")

    # === Phase 6: Final micro-shave ===
    print(f"\n=== Phase 6: Final Micro-Shave (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_micro = min(rem * 0.4, 60.0)
        if t_micro > 10:
            a_micro, _ = peak_shave_ultra(best_seq, t_micro, n_steps=50000)
            try_update(a_micro, "final_micro_shave")

    # === Phase 7: One more LP round on best ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp_fin = min(rem * 0.5, 60.0)
        if t_lp_fin > 15:
            a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
            try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
