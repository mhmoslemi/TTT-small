"""
Minimizing 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with simplex projection, combined with LP cutting-plane polish and multi-length
exploration. Uses curriculum on the softmax parameter alpha for adaptive
sharpness, and multiple restarts for robustness.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(w, a, L):
    """Compute gradient of smooth max: g[m] = 2 * sum_k w[k] * a[k-m]."""
    W = fft(w, L)
    A = fft(a, L)
    grad = ifft(W * np.conj(A)).real[:len(a)]
    return 2.0 * grad


def smooth_max_descent(a_init, time_limit, alpha_init=50.0, alpha_max=5000.0,
                       lr_init=0.5, verbose=True, label=""):
    """
    Smooth-max gradient descent with simplex projection.
    Minimize max_k C_k(a) subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    L = 1 << (3 * n - 2).bit_length()
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    alpha = alpha_init
    lr = lr_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute autocorrelation
        C = fast_convolve(a)
        
        # Compute softmax weights
        alpha_C = alpha * C
        alpha_C -= alpha_C.max()  # numerical stability
        w = np.exp(alpha_C)
        w_sum = w.sum()
        if w_sum < 1e-100:
            w = np.zeros(len(C))
            w[int(C.argmax())] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        g = compute_gradient(w, a, L)
        
        # Projected gradient step
        a_new = project_simplex(a - lr * g)
        
        # Check for improvement
        if it % 5 == 0:
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adapt alpha (curriculum)
            if it % 50 == 0:
                alpha = min(alpha * 1.03, alpha_max)
            
            # Adapt learning rate
            if no_improve > 20:
                lr *= 0.9
                no_improve = 0
            
            # Periodic restart to escape local minima
            if no_improve > 200 and it % 500 == 0:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
                lr = lr_init * 0.5
                no_improve = 0
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"alpha={alpha:.1f} lr={lr:.4f} t={elapsed:.0f}s")
    
    return best_a, best_obj

"""
Minimizing 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with Nesterov momentum and simplex projection, combined with cutting-plane LP
polish and multi-length exploration. Key ideas: (1) smooth approximation of max
via softmax with curriculum on sharpness, (2) FFT-based O(n log n) gradient
computation, (3) fast convergence via FISTA-style momentum, (4) multi-restart
strategy with structured initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(w, a, L):
    """Compute g[m] = 2*sum_k w[k]*a[k-m] using FFT cross-correlation."""
    W = np.fft.fft(w, L)
    A = np.fft.fft(a, L)
    grad = np.fft.ifft(W * np.conj(A)).real[:len(a)]
    return 2.0 * grad


def smooth_max_fista(a_init, time_limit, alpha_init=None, alpha_max_mult=20.0,
                     lr_init=None, momentum_init=0.8, verbose=True, label=""):
    """
    FISTA (Nesterov accelerated) projected gradient descent on smooth max
    approximation of max_k C_k(a) over the probability simplex.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    if alpha_init is None:
        C0 = fast_convolve(a)
        alpha_init = max(10.0, n * 0.5)
    alpha_max = alpha_init * alpha_max_mult

    if lr_init is None:
        lr_init = min(0.5, 2.0 / max(1.0, alpha_init * 0.1))

    L_fft = 1 << (3 * n - 1).bit_length()

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    alpha = alpha_init
    lr = lr_init
    momentum = momentum_init
    it = 0
    no_improve = 0
    tol = 1e-12

    # Nesterov momentum state
    x = a.copy()
    y = a.copy()
    t_accel = 1.0

    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute autocorrelation at y (Nesterov point)
        C = fast_convolve(y)

        # Softmax weights with numerical stability
        alpha_C = alpha * C
        alpha_C -= alpha_C.max()
        w = np.exp(alpha_C)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(C))
            w[int(C.argmax())] = 1.0
        else:
            w /= w_sum

        # Gradient at y
        g = compute_gradient_fft(w, y, L_fft)

        # Gradient step
        x_new = project_simplex(y - lr * g)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_accel * t_accel)) / 2.0
        beta = (t_accel - 1.0) / t_new
        y_new = x_new + beta * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new /= s_y
        else:
            y_new = x_new.copy()

        x = x_new
        y = y_new
        t_accel = t_new

        # Evaluate periodically
        if it % 3 == 0:
            obj = fast_convolve(x).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = x.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adapt alpha (curriculum: increase sharpness)
        if it % 60 == 0:
            alpha = min(alpha * 1.04, alpha_max)
            lr = min(lr_init * (alpha_init / alpha) ** 0.3, 0.5)

        # Adapt learning rate on stagnation
        if no_improve > 30 and no_improve % 15 == 0:
            lr *= 0.85
            no_improve = 0

        # Periodic restart to escape local minima
        if no_improve > 300 and it % 500 == 0:
            noise_scale = 0.005 * (1.0 + 0.5 * rng.random())
            x = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = x.copy()
            t_accel = 1.0
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"alpha={alpha:.0f} lr={lr:.5f} no_imp={no_improve} t={elapsed:.0f}s")

    return best_a, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9)
    except Exception:
        g = None
    if g is None or g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_polish(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """Cutting-plane LP for fine polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f /= s
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    active_idx = np.sort(active_idx)
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, n, M, active_idx)
        if g is None:
            stuck += 1
            if stuck > 6:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane check
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        F_fft = rfft(f, fft_size)
        G_fft = rfft(g, fft_size)
        fg = irfft(F_fft * G_fft, fft_size)[:size]
        violated = fg > M * (1 + 1e-7)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                to_add = not_active[np.argsort(fg[not_active])[-20:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 4:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 10:
                        break
                    continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Refine
        lo, hi = max(0.0, best_t - 0.05), min(1.0, best_t + 0.1)
        for _ in range(5):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
            f1 /= max(f1.sum(), 1e-15)
            f2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
            f2 /= max(f2.sum(), 1e-15)
            if fast_convolve(f1).max() < fast_convolve(f2).max():
                hi = m2
            else:
                lo = m1

        t_mid = (lo + hi) / 2.0
        f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
        f_mid /= max(f_mid.sum(), 1e-15)
        o_mid = fast_convolve(f_mid).max()
        if o_mid < best_t_obj:
            best_t_obj = o_mid
            best_t = t_mid

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        f /= max(f.sum(), 1e-15)
        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        if it % 20 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k:]]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def make_structured_init(n, rng):
    """Create a structured initialization with oscillating pattern to break symmetry."""
    a = np.ones(n)
    # Add a low-frequency oscillation to break the triangular autocorrelation
    freq = 1.0 + rng.random() * 2.0
    phase = rng.random() * 2 * np.pi
    osc = 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = 1.0 + osc
    a = np.maximum(a, 0.1)
    a /= a.sum()
    return a


def make_picket_init(n, n_peaks, rng):
    """Create initialization with n_peaks Gaussian bumps at random positions."""
    a = np.zeros(n)
    positions = rng.choice(n, size=n_peaks, replace=False)
    widths = rng.uniform(n * 0.01, n * 0.05, size=n_peaks)
    for pos, width in zip(positions, widths):
        dist = np.minimum(np.abs(np.arange(n) - pos), n - np.abs(np.arange(n) - pos))
        a += np.exp(-0.5 * (dist / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a

            s7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(s7) and s7 < best_score:
                best_score = s7
                best_seq = f7.copy()
                print(f"  NEW BEST [fista_final]: {s7:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
