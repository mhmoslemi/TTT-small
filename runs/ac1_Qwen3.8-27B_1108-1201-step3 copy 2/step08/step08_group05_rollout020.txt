"""
Multi-phase optimization for minimizing the auto-convolution peak of non-negative sequences.
Combines LP-based direction finding, peak shaving, spectral shaping, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, M, active_indices, n, max_constraints=400):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_indices.
    Returns the optimal g or None.
    """
    active_indices = list(active_indices)
    if len(active_indices) > max_constraints:
        # Keep most violated ones
        b_f = fast_convolve(f)
        # Sort by b_f value (most likely to be violated)
        active_indices.sort(key=lambda k: -b_f[k])
        active_indices = active_indices[:max_constraints]
    
    K = len(active_indices)
    if K == 0:
        return None
    
    # Build constraint matrix A where row i corresponds to lag k_i
    # (f*g)[k_i] = sum_j f[j] * g[k_i - j] <= M
    A = np.zeros((K, n))
    for row_i in range(K):
        k = active_indices[row_i]
        for j in range(n):
            gi = k - j
            if 0 <= gi < n:
                A[row_i, gi] = f[j]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g)) and not np.any(np.isinf(g)):
                    return g
        except Exception:
            continue
    
    return None

def subgradient_method(f, time_limit, n_top=30, lr_init=0.002):
    """Subgradient optimization with adaptive learning rate."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()

    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        batch = 50
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_convolve(f)
            
            # Weight peaks by their excess over the mean
            b_mean = b.mean()
            excess = np.maximum(b - b_mean, 0)
            w = excess / (excess.sum() + 1e-30)
            
            # Focus on top-k
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_top = w[top_idx].copy()
                ws = w_top.sum()
                if ws > 1e-30:
                    w_top /= ws
                else:
                    w_top = np.ones(len(top_idx)) / len(top_idx)
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_top
                w = w_new
            
            # Compute gradient: d/dx_j [sum_k w_k (a*a)[k]]
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # d/dx_j (a*a)[k] = a[k-j] + a[j-k] (with appropriate bounds)
            grad = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-13:
                    continue
                for j in range(n):
                    gi = k - j
                    if 0 <= gi < n:
                        grad[j] += w[k] * f[gi]
                    gi2 = j - k
                    if 0 <= gi2 < n:
                        grad[j] += w[k] * f[gi2]
            
            # Momentum update
            v = 0.9 * v - lr * grad
            f_new = f + v
            f_new = project_simplex(f_new)
            f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve_count = 0
            lr = min(lr * 1.02, 0.01)

            if no_improve_count > 5:
                lr *= 0.85
                no_improve_count = 0

                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, k_list, n):
    """Build constraint matrix for LP: row i corresponds to lag k_list[i]."""
    K = len(k_list)
    A = np.zeros((K, n))
    for i in range(K):
        k = k_list[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            A[i, m] = f[k - m]
    return A


def solve_lp(f, M, k_list, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in k_list."""
    K = len(k_list)
    if K == 0 or n == 0:
        return None
    
    A = build_constraint_matrix(f, k_list, n)
    
    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
    
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if np.all(np.isfinite(g)) and g.sum() > 1e-10:
                    return g
        except Exception:
            continue
    return None


def golden_search_peak(f, h, lo=0.0, hi=1.0, max_iter=20):
    """Golden section search for best t in [lo,hi] minimizing peak of (1-t)*f + t*h."""
    n = len(f)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def obj_at(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-10:
            return 1e10
        ft = ft / s
        return fast_convolve(ft).max()

    best_t = 0.05
    best_obj = obj_at(0.05)
    for t in candidates:
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t
    
    # Refine around best
    idx = candidates.index(best_t)
    lo_g = candidates[max(0, idx - 1)]
    hi_g = candidates[min(len(candidates) - 1, idx + 1)]
    if lo_g == hi_g:
        return best_t
    
    c = hi_g - gr * (hi_g - lo_g)
    d = lo_g + gr * (hi_g - lo_g)

            break
        
        fft_size = 1 << (2 * n).bit_length()
        F = rfft(f, fft_size)
        power = np.abs(F) ** 2
        
        # Try to equalize power (flatten spectrum)
        for alpha in [0.005, 0.01, 0.02, 0.05]:
            target = np.full_like(power, power.mean())
            power_new = (1.0 - alpha) * power + alpha * target
            # Preserve DC
            power_new[0] = power[0]
            
            F_new = F * np.sqrt(np.maximum(power_new, 1e-30) / (power + 1e-30))
            f_new = irfft(F_new, fft_size)[:n]
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                break
    
    if it > 0:
        print(f"    {label} iters={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f}")
    
    return best_f, best_obj


def perturbation_restart(f_best, f_current, n, rng, strength=0.01):
    """Create a perturbed version biased toward the best solution."""
    f = f_current.copy()
    # Blend toward best
    f = 0.7 * f + 0.3 * f_best
    # Add perturbation
    f += rng.standard_normal(n) * strength
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f = f / s
    return f


def subgradient_refine(f, time_limit, n_top=40, lr=0.001, label="sg"):
    """Subgradient optimization with momentum for final polishing."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_imp = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break
        it += 1
        batch = 30
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            b = fast_convolve(f)
            # Weights: emphasize peaks
            b_shifted = b - b.min()
            w = b_shifted ** 2 / (b_shifted ** 2).sum()
            
            # Top-k focus
            if n_top < len(b):
                top = np.argpartition(b, -n_top)[-n_top:]
                wt = w[top].copy()
                ws = wt.sum()
                if ws > 1e-30:
                    wt /= ws
                w_new = np.zeros(len(b))
                w_new[top] = wt
                w = w_new
            
            # Gradient
            grad = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                wv = w[k]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += wv * f[k - j]
            
            v = 0.85 * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5:
                lr *= 0.8
                no_imp = 0
            if no_imp > 20:
                break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new
