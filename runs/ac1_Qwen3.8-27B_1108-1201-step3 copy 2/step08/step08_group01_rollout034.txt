So `T_a^T @ k_w = ifft(conj(rfft(a, N)) * rfft(k_w, N))[:n]` where N >= 2n-1.

Great, so the Hedge update is:
```python
N = 1 << (2*n-1).bit_length()
A = rfft(a, N)  # compute once
for t in range(hedge_iters):
    c = irfft(A * rfft(b, N), N)[:2*n-1]  # conv(a, b)
    k_w = softmax(beta * c)  # soft-max weights over convolution positions
    grad = irfft(np.conj(A) * rfft(k_w, N), N)[:n]  # T_a^T @ k_w
    b = np.exp(-eta * grad)
    b = b / b.sum()
    b_avg = (t * b_avg + b) / (t + 1)
```

This is O(n log n) per iteration (dominated by FFT). Very fast!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Hedge-based fictitious play (fast approximate LP best responses via multiplicative weights)
2. Smooth surrogate (log-sum-exp) gradient descent with adaptive temperature
3. Sensitivity-based coordinate descent for fine polishing
4. Multi-length exploration with diverse initialization
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, b):
    """Compute sum_m a[m]*b[m+j] for j=0..n-1 using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(np.conj(A) * B, fft_size)[:n]

def softmax(x, beta=1.0):
    """Stable softmax."""
    x = beta * x - beta * x.max()
    e = np.exp(x)
    return e / e.sum()

def hedge_best_response(a, init_b=None, n_iters=200, beta=5.0, eta=None, return_avg=True):
    """
    Approximate best response using multiplicative weights (Hedge algorithm).
    Given fixed a (non-negative, normalized), find b minimizing max_k conv(a,b)[k].
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    
    if init_b is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(init_b, 0).copy()
        s = b.sum()
        b = b / s if s > 1e-15 else np.ones(n) / n
    
    b_avg = b.copy().astype(np.float64)
    
    if eta is None:
        eta = 1.0 / max(a.max(), 1e-10) * 0.5
    
    for t in range(n_iters):
        # Compute conv(a, b)
        c = irfft(A * rfft(b, fft_size), fft_size)[:size]
        
        # Soft-max weights
        k_w = softmax(c, beta=beta)
        
        # Gradient: T_a^T @ k_w = cross-correlation
        grad = irfft(np.conj(A) * rfft(k_w, fft_size), fft_size)[:n]
        grad = np.maximum(grad, 0)  # ensure non-negative
        
        # Multiplicative update
        b = np.exp(-eta * grad)
        s = b.sum()
        if s < 1e-30:
            b = np.ones(n) / n
        else:
            b = b / s
        
        # Running average
        b_avg = ((t + 1) * b_avg + b) / (t + 2)
    
    if return_avg:
        b_avg = np.maximum(b_avg, 0)
        s = b_avg.sum()
        if s > 1e-15:
            b_avg = b_avg / s
        else:
            b_avg = np.ones(n) / n
        return b_avg
    
    return b

def smooth_surrogate_grad(a, beta=50.0):
    """
    Compute gradient of log(sum_k exp(beta * c_k)) w.r.t. a.
    Returns (gradient, surrogate_value).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    
    c = irfft(A * A, fft_size)[:size]  # conv(a, a)
    
    # Softmax weights
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w_sum = w.sum()
    w = w / w_sum
    
    # Gradient: df/da_i = beta * sum_k w_k * 2 * a[k-i] / ... 
    # Actually: df/da_i = beta * (sum_k w_k * dc_k/da_i) = beta * 2 * (T_a^T w)[i]
    # where (T_a^T w)[i] = sum_k w_k * a[k-i] = cross_corr(a, w)[i]
    
    grad_conv = irfft(np.conj(A) * rfft(w, fft_size), fft_size)[:n]
    grad = 2.0 * beta * grad_conv
    
    surrogate = np.log(w_sum) + beta * c.max()  # log(sum exp(beta*c))
    
    return grad, float(surrogate)

def smooth_gradient_descent(a0, time_limit, beta_init=10.0, beta_max=200.0, lr_init=0.1, verbose=False, label=""):
    """
    Gradient descent on the smooth log-sum-exp surrogate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad, surr = smooth_surrogate_grad(a, beta)
        
        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = a_new - a_new.min()  # project to non-negative
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        conv = fast_convolve(a_new)
        score = 2.0 * n * conv.max()
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 50:
                # Try increasing beta (sharper max approximation)
                beta = min(beta * 1.2, beta_max)
                lr = lr * 0.95
                no_improve = 0
                if lr < 1e-8:
                    break
        
        # Adaptive learning rate
        if it % 10 == 0:
            lr *= 0.999
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth GD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.1f} lr={lr:.6f}")
    
    return best_a, best_score

def fast_coord_descent(a0, time_limit, n_trials=10, verbose=False, label=""):
    """Fast coordinate descent with sensitivity analysis."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a[0:n].copy()

            best_delta = 0.0
            best_new_max = M

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.5 * a[i]
                elif trial == 1:
                    delta = -0.25 * a[i]
                elif trial == 2:
                    delta = -0.1 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = -0.02 * a[i]
                elif trial == 5:
                    delta = 0.02 * a[i]
                elif trial == 6:
                    delta = 0.05 * a[i]
                elif trial == 7:
                    delta = 0.1 * a[i]
                elif trial == 8:
                    delta = 0.25 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 100:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def hedge_fictitious_play(n, time_limit, initial=None, verbose=False, label=""):
    """Fictitious play using Hedge (multiplicative weights) for approximate best responses."""
    t0 = time.monotonic()
    
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        a = a.reshape(-1)
        if len(a) != n:
            # Interpolate
            old = a
            a_new = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(old)), old)
            a = a_new
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    b = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    outer = 0
    no_improve = 0
    
    # Parameters
    hedge_iters = 300
    beta_hedge = 3.0
    eta_base = 0.5
    
    while time.monotonic() - t0 < time_limit:
        outer += 1
        
        # Adaptive parameters
        alpha = min(1.0, 5.0 / (outer + 1))
        # Increase hedge precision over time
        h_iters = min(hedge_iters + outer * 5, 800)
        b_hedge = min(beta_hedge + outer * 0.1, 20.0)
        
        # Best response for b given a
        b_new = hedge_best_response(a, init_b=b, n_iters=h_iters, beta=b_hedge, eta=eta_base)
        # Best response for a given b_new
        a_new = hedge_best_response(b_new, init_b=a, n_iters=h_iters, beta=b_hedge, eta=eta_base)
        
        # Damped update
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
        
        b = (1 - alpha) * b + alpha * b_new
        b = np.maximum(b, 0)
        s = b.sum()
        b = b / s if s > 1e-15 else np.ones(n) / n
        
        # Evaluate candidates
        for cand in [a, b, 0.5*(a+b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc > 1e-10:
                cand_n = cand / sc
                score = 2.0 * n * fast_convolve(cand_n).max()
                if score < best_score:
                    best_score = score
                    best_a = cand_n.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        # Check convergence
        if no_improve > 200:
            # Perturb
            rng = np.random.default_rng(outer)
            a = a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and outer % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HedgeFP {label} iter={outer} ({elapsed:.1f}s): best={best_score:.6f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_cosine(n, rng):
    """Initialize with cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_u_shape(n, rng):
    """Initialize with U-shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_random(n, rng):
    """Random initialization."""
    a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Hedge (multiplicative weights) fictitious play for fast global optimization
2. Adaptive log-sum-exp smooth surrogate gradient descent

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Compute cross-correlation: result[j] = sum_m a[m]*b[m+j] for j=0..n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(np.conj(A) * B, fft_size)[:n]


def softmax_stable(x, beta=1.0):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    scaled = beta * x
    scaled -= scaled.max()
    e = np.exp(scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def hedge_best_response(a, init_b=None, n_iters=200, beta=5.0, eta=None, use_avg=True):
    """
    Approximate LP best response via multiplicative weights (Hedge algorithm).
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    if init_b is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(np.asarray(init_b, dtype=np.float64), 0).copy()
        s = b.sum()
        b = b / s if s > 1e-15 else np.ones(n) / n

    b_avg = b.copy().astype(np.float64)

    if eta is None:
        a_max = a.max()
        eta = 2.0 / max(a_max, 1e-10)

    for t in range(n_iters):
        # conv(a, b)
        c = irfft(A * rfft(b, fft_size), fft_size)[:size]

        # Soft-max weights over convolution positions
        k_w = softmax_stable(c, beta=beta)

        # Gradient: T_a^T @ k_w
        grad = irfft(np.conj(A) * rfft(k_w, fft_size), fft_size)[:n]
        grad = np.maximum(grad, 0.0)

        # Multiplicative update
        log_b = -eta * grad
        log_b -= log_b.max()
        b = np.exp(log_b)
        s = b.sum()
        if s < 1e-30:
            b = np.ones(n) / n
        else:
            b = b / s

        if use_avg:
            b_avg = ((t + 1) * b_avg + b) / (t + 2)

    result = b_avg if use_avg else b
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n) / n
    return result

    Gradient descent on log-sum-exp smooth surrogate of max(conv(a,a)).
    Adaptive beta (increases over time for sharper max approximation).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    beta = beta_init
    lr = lr_init
    no_improve = 0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = rfft(a, fft_size)
        c = irfft(A * A, fft_size)[:size]

        # Softmax weights
        bc = beta * c
        bc -= bc.max()
        e = np.exp(bc)
        w_sum = e.sum()
        if w_sum < 1e-300:
            w = np.ones(size) / size
        else:
            w = e / w_sum

        # Gradient: 2*beta*(T_a^T w)
        grad = 2.0 * beta * fast_cross_corr(a, w)

        # Gradient step with projection onto simplex
        a_new = a - lr * grad
        # Soft projection to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 30:
                # Anneal: increase beta, decrease lr
                beta = min(beta * 1.3, 300.0)
                lr = lr * 0.9
                no_improve = 0
                if lr < 1e-10:
                    break

        if it % 20 == 0:
            lr = lr * 0.998

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} beta={beta:.1f} lr={lr:.6e}")

    return best_a, best_score


def coord_descent(a0, time_limit, n_trials=10, use_annealing=False, T_init=0.01,
                  verbose=False, label=""):
    """
    Sensitivity-based coordinate descent. Optionally with simulated annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(777)
    temperature = T_init

    # Precompute delta fractions
    if n_trials <= 8:
        deltas = np.array([-0.5, -0.25, -0.1, -0.05, 0.05, 0.1, 0.25, 0.5])[:n_trials]
        if n_trials < 8:
            deltas = deltas[:n_trials]
    elif n_trials <= 12:
        deltas = np.array([-0.5, -0.3, -0.15, -0.07, -0.03, -0.01,
                           0.01, 0.03, 0.07, 0.15, 0.3, 0.5])[:n_trials]
    else:
        deltas = np.array([-0.6, -0.4, -0.25, -0.15, -0.08, -0.04, -0.02, -0.01,
                           0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.4, 0.6])[:n_trials]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            k_len = k_end - k_start + 1

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = conv[:k_start].max()
            if k_end < 2 * n - 2:
                right_max = conv[k_end + 1:].max()
                if right_max > unaff_max:
                    unaff_max = right_max

            affected = conv[k_start:k_end + 1]
            a_shift = a  # a[k-i] for k=i..i+n-1 gives a[0]..a[n-1]

            best_delta = 0.0
            best_new_max = M + 1e-30  # require strict improvement

            for d_frac in deltas:
                delta = d_frac * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a_shift
                # Handle k=2i diagonal
                if 2 * i >= k_start and 2 * i <= k_end:
                    new_aff = new_aff.copy()
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_aff.max()))

                if use_annealing and temperature > 1e-10:
                    # Accept with probability based on energy difference
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = delta
                    else:
                        # Simulated annealing: accept worse with prob exp(-diff/T)
                        diff = (new_max - best_new_max) * 2.0 * n  # approximate score diff
                        if rng.random() < np.exp(-diff / temperature):
                            best_new_max = new_max
                            best_delta = delta
                else:
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                # Apply update
                old_ai = a[i]
                a[i] = old_ai + best_delta
                a[i] = max(a[i], 0.0)

                # Update convolution
                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    changes = changes.copy()
                    idx = 2 * i - k_start
                    changes[idx] = 2.0 * best_delta * (old_ai + best_delta) + best_delta * best_delta - 2.0 * best_delta * old_ai
                    # Actually: new c[2i] = old c[2i] + 2*delta*old_ai + delta^2
                    # The linear part already added 2*delta*a[i] = 2*delta*old_ai
                    # So the correction is just delta^2
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if use_annealing:
            temperature *= 0.995

        if not improved:
            no_improve += 1
            if no_improve > (200 if use_annealing else 50):
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Very fine steps
    factors = [0.995, 0.99, 0.985, 0.9925, 0.9975, 1.0025, 1.0075, 1.0125, 1.0175]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = conv[:k_start].max()
            if k_end < 2 * n - 2:
                right_max = conv[k_end + 1:].max()
                if right_max > unaff_max:
                    unaff_max = right_max

            affected = conv[k_start:k_end + 1]
            a_shift = a

            best_delta = 0.0
            best_new_max = M

            for f in factors:
                delta = (f - 1.0) * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    new_aff = new_aff.copy()
                    idx = 2 * i - k_start
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_aff = new_aff.copy()

                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-18:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0:
                old_ai = a[i]
                a[i] = old_ai + best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_shift
                if 2 * i >= k_start and 2 * i <= k_end:
                    changes = changes.copy()
                    idx = 2 * i - k_start
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.65)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_w_shape(n, rng):
    """Initialize with W-shape (mass at edges and center)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(1.0, 3.0)
    beta = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(2 * np.pi * t) ** 2 + beta * np.cos(np.pi * t) ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
