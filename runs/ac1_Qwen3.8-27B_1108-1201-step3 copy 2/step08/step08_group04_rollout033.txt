KKT-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses the Karush-Kuhn-Tucker conditions to derive an efficient iterative scheme:
- Dual variables (softmax weights over peaks) determine the "load" at each position
- Primal update reduces high-load positions (multiplicative weights)
- FFT-based computation makes each iteration O(n log n)
- Adaptive temperature controls global vs local focus
- LP-based escape for local minima
- Multi-length exploration with transfer
- Coordinate descent final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def kkt_gradient(a, p):
    """
    Compute g[j] = sum_k p[k] * d(a*a)[k]/da[j]
    = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    using FFT.
    """
    n = len(a)
    q = a[::-1]  # reversed a
    # (p * q)[j + n - 1] = sum_m p[j+m] * a[m]
    conv_len = len(p) + n - 1

    P = rfft(p, fft_len)
    Q = rfft(q, fft_len)
    conv_pq = irfft(P * Q, fft_len)
    # g[j] = 2 * conv_pq[j + n - 1] for j = 0..n-1
    g = 2.0 * conv_pq[n - 1: 2 * n - 1]
    return g

            p = np.zeros(len(b))
            p_vals = b[top_idx]
            p_scaled = (p_vals - p_vals.max()) / beta
            p_w = np.exp(p_scaled)
            p_w /= p_w.sum()
            p[top_idx] = p_w
        else:
            p_scaled = (b - b.max()) / beta
            p = np.exp(p_scaled)
            p /= p.sum()

        # Compute gradient (load) at each position
        g = kkt_gradient(a, p)

        # Primal update: multiplicative weights
        active = g > 1e-12
        if not active.any():
            no_improve += 1
            if no_improve > patience:
                break
            continue

        lam = g[active].mean()
        ratio = np.ones(n)
        ratio[active] = (lam / g[active]) ** alpha
        a_new = a * ratio
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_score = 2.0 * n * best_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            # Backtrack with smaller step
            improved = False
            for try_alpha in [alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                ratio2 = np.ones(n)
                ratio2[active] = (lam / g[active]) ** try_alpha
                a_try = a * ratio2
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_score = 2.0 * n * obj_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                # Periodic perturbation to escape
                if no_improve > patience // 2 and it % 20 == 0:
                    a = project_simplex(a + rng.standard_normal(n) * 0.003)
                    no_improve = 0
                if no_improve > patience:
                    # Reset to best and perturb more
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                    no_improve = 0
                    beta = max(0.01, beta * 0.5)

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} score={best_score:.6f} beta={beta:.4f} "
                  f"npeaks={n_peaks} t={elapsed:.0f}s")

    return best_a, best_score


def lp_escape(a_init, top_k, M_target):
    """
    Quick LP-based direction finding for escape from local minima.
    Solves: max sum(g) s.t. g>=0, (a*g)[k] <= M_target for selected k.
    Returns the direction g or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a_init)
    b = fast_convolve(a_init)
    # Select top peaks
    if top_k < len(b):
        indices = np.argpartition(b, -top_k)[-top_k:]
    else:
        indices = np.arange(len(b))
    indices = np.sort(indices)

    # Build constraint matrix
    K = len(indices)
    A_mat = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_mat[i, lo:hi + 1] = a_init[lo:hi + 1]

    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A_mat @ g_var <= M_target]
    )

    try:
        if K <= 500 and n <= 1000:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        else:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-5)
        except Exception:
            return None

    if g_var.value is None:
        return None
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    s_g = g_val.sum()
    if s_g <= 1.0 + 1e-4:
        return None
    return g_val / s_g


def subgradient_polish(a_init, time_limit, lr_init=0.003, n_top=100, momentum=0.8):
    """Subgradient optimization with adaptive learning rate."""
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    v = np.zeros(n)
    lr = lr_init
    temp = 0.001
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()

        # Softmax weights on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        p_vals = b[top_idx]
        p_scaled = (p_vals - p_vals.max()) / temp
        p_w = np.exp(p_scaled)
        p_w /= p_w.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            k = top_idx[idx]
            pk = p_w[idx]
            if pk < 1e-14:
                continue
            # d(a*a)[k]/da[j] = 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # a[k-j] for j in [lo, hi] means indices k-hi to k-lo in a
            a_idx_lo = k - hi
            a_idx_hi = k - lo
            grad[lo:hi + 1] += pk * 2.0 * a[a_idx_lo:a_idx_hi + 1]

        # Update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        if it % 30 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.05, 0.01)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0
            if it % 150 == 0:
                temp = min(temp * 1.15, 0.3)
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_a, 2.0 * n * best_obj


def coordinate_descent(a_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    best_score = 2.0 * n * best_obj

    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-12:
                continue
            best_j_val = a[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = a[j] * mult
                if new_val < 0:
                    continue
                a[j] = new_val
                s_cur = a.sum()
                if s_cur < 1e-12:
                    a[j] = best_j_val
                    continue
                a_norm = a / s_cur
                obj = fast_convolve(a_norm).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                a[j] = best_j_val
            if abs(best_j_val - a[j]) > 1e-15:
                a[j] = best_j_val
                s_cur = a.sum()
                if s_cur > 1e-12:
                    a = a / s_cur
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_a = a.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_a, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def make_symmetric(a):
    """Make a sequence palindromic (symmetric)."""
    n = len(a)
    sym = np.zeros(n)
    for i in range(n):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
    s = sym.sum()
    if s > 1e-15:
        sym = sym / s
    return sym
