```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm: find threshold tau such that sum(max(v - tau, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the largest rho such that 1 + u[rho] * rho > 0
    cssv = np.cumsum(u)
    rho = np.max(np.where(1 + u * np.arange(1, n+1) > 0)[0])
    # tau = (cssv[rho] - 1) / (rho + 1)
    tau = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Minimax-QP dual iteration (novel: alternates between QP in primal and softmax dual update)
2. Log-space gradient descent with FFT-accelerated gradient (cross-correlation)
3. Coordinate descent with exact golden-section line search
4. Multiple temperatures and length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def cross_corr(p, a):
    """(M(p)@a)_j = sum_m a[m]*p[m+j], computed via FFT."""
    n = len(a)
    L = 3 * n - 2
    fft_size = 1 << (L - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(P) * A, fft_size)[:n]
    return result

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(1 + u * np.arange(1, n + 1) > 0)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    tau = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)

def minimax_qp(a_init, time_budget, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    # Initialize p: delta at center
    c0 = fast_convolve(a)
    p = np.zeros(2 * n - 1)
    p[np.argmax(c0)] = 1.0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T = 50.0  # temperature for softmax
    alpha = 0.1  # initial step size
    
    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Inner: projected gradient on simplex for min a^T M(p) a
        f_curr = a @ cross_corr(p, a)
        for inner in range(30):
            grad = 2.0 * cross_corr(p, a)
            # Simple step with damping
            a_trial = project_simplex(a - alpha * grad)
            f_trial = a_trial @ cross_corr(p, a_trial)
            if f_trial < f_curr:
                a = a_trial
                f_curr = f_trial
                alpha = min(alpha * 1.1, 1.0)
            else:
                alpha *= 0.5
                if alpha < 1e-6:
                    break
        
        # Outer: update p
        c = fast_convolve(a)
        T = max(1.0, 50.0 * 0.99**it)
        p_new = np.exp(c / T - c.max())
        p_new = p_new / p_new.sum()
        p = 0.7 * p + 0.3 * p_new
        
        score = compute_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 100 == 0:
            print(f"  MM iter={it}, T={T:.2f}, alpha={alpha:.4f}, "
                  f"score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def log_space_descent(a_init, time_budget, T=100.0, verbose=True):
    """Gradient descent in log-space with softmax-smoothed max."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    alpha = 0.1
    
    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break
        
        c = fast_convolve(a)
        # Softmax weights
        logits = c / T
        logits -= logits.max()
        w = np.exp(logits)
        w = w / w.sum()
        
        # Gradient in a-space: 2n * 2 * cross_corr(w, a) = 4n * cross_corr(w, a)
        # But we're minimizing 2n*max(c), so gradient of 2n*smoothed_max is:
        # 2n * sum_k w_k * grad(c_k) = 2n * sum_k w_k * 2*cross_corr_at_k
        # Actually: grad_j of smoothed_max = sum_k w_k * 2*a_{k-j} = 2 * cross_corr(w, a)_j
        # So grad of 2n * smoothed_max = 2n * 2 * cross_corr(w, a) = 4n * cross_corr(w, a)
        
        grad_a = 4.0 * n * cross_corr(w, a)
        
        # Gradient in log-space:
        # a = exp(z) / sum(exp(z)), z = log(a) (up to additive constant)
        # d/dz_j [f(a)] = a_j * (df/da_j - sum_i a_i df/da_i)
        correction = np.dot(a, grad_a)
        grad_z = a * (grad_a - correction)
        
        # Step
        a_trial = a * np.exp(-alpha * grad_z)
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        
        score_trial = compute_score(a_trial)
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
            a = a_trial
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha *= 0.7
        
        # Adaptive temperature
        T = max(1.0, 100.0 * 0.999**it)
        
        if verbose and it % 200 == 0:
            print(f"  LDS iter={it}, T={T:.1f}, alpha={alpha:.4f}, "
                  f"score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a_init, time_budget, verbose=True):
    """Pairwise coordinate descent with exact line search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        
        # Get top peaks
        c = fast_convolve(a)
        peak_idx = int(np.argmax(c))
        
        # Generate candidate pairs
        n_pairs = min(n, 300)
        pairs = set()
        
        # Pairs in the peak's contributing range
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for _ in range(n_pairs // 2):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add(tuple(sorted((i, j))))
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add(tuple(sorted((i, j))))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            # Golden section search for optimal transfer
            # t > 0: move mass from i to j
            t_lo = -a[j]
            t_hi = a[i]
            
            if t_hi - t_lo < 1e-15:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_transfer(t):
                # t > 0: a[i] -= t, a[j] += t
                ai = a[i] - t
                aj = a[j] + t
                if ai < 0 or aj < 0:
                    return 1e10
                # Quick score estimate: only recompute affected convolution values
                # For speed, use full recompute
                a_trial = a.copy()
                a_trial[i] = ai
                a_trial[j] = aj
                a_trial = a_trial / a_trial.sum()
                return compute_score(a_trial)
            
            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            
            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Perturb
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and (it := int((time.monotonic() - t0) * 2)) % 5 == 0:
            pass  # skip frequent prints
    
    return best_a, best_score
