```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with Nesterov momentum (log-sum-exp smoothed max)
2. Fast coordinate descent with golden-section line search  
3. Simulated annealing with peak-targeted perturbations
4. Multi-length exploration (512, 768, 1024, 1536, 2048)
5. Local restarts with structured perturbations
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def compute_gradient(a, weights, top_k=100):
    """Compute gradient of smoothed max using precomputed peak weights."""
    n = len(a)
    grad = np.zeros(n)
    
    # Find active lags
    active = np.where(weights > 1e-10)[0]
    
    for k in active:
        k = int(k)
        wk = weights[k]
        if k == 0:
            grad += 2.0 * wk * a
        else:
            grad[:n-k] += wk * a[k:]
            grad[k:] += wk * a[:n-k]
    
    return grad

def get_smoothed_weights(conv, T, top_k=100):
    """Compute normalized weights for log-sum-exp smoothing."""
    size = len(conv)
    max_c = conv.max()
    
    if top_k < size:
        # Use top-k peaks only
        part_idx = np.argpartition(conv, -top_k)[-top_k:]
        thresh = conv[part_idx].min()
        active = np.where(conv >= thresh)[0]
    else:
        active = np.arange(size)
    
    w_vals = (conv[active] - max_c) / T
    w_vals = np.clip(w_vals, -50, 0)
    w_exp = np.exp(w_vals)
    w_sum = w_exp.sum()
    
    weights = np.zeros(size)
    if w_sum > 1e-300:
        weights[active] = w_exp / w_sum
    
    return weights

def grad_descent_momentum(a_init, time_limit, T_init=3.0, top_k=80, 
                          lr_init=0.1, momentum=0.9, verbose=True):
    """
    Gradient descent with Nesterov momentum on the smoothed max-conv objective.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    patience = 0
    max_patience = 200
    T_min = 0.5
    
    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        weights = get_smoothed_weights(conv, T, top_k)
        grad = compute_gradient(a, weights)
        
        # Nesterov: v = momentum * v - lr * grad
        v = momentum * v - lr * grad
        
        # Trial step
        a_trial = a + v
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            v *= 0.5
            continue
        a_trial = a_trial / s
        
        score = compute_score(a_trial)
        
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            # Slowly decrease temperature
            if T > T_min:
                T = max(T_min, T * 0.9995)
        else:
            patience += 1
            if patience > max_patience:
                # Stuck: reduce temperature, reset momentum, restart from best
                T = max(T_min, T * 0.8)
                lr *= 0.7
                v = np.zeros(n)
                a = best_a.copy()
                patience = 0
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  GD iter={it}, T={T:.3f}, lr={lr:.5f}, "
                  f"score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score
```

Now let me think about coordinate descent more carefully. The key insight is that for a pair transfer (i,j), we want to find the optimal t that minimizes the max convolution.

For a pair (i,j) and transfer t:
- a[i] -> a[i] - t
- a[j] -> a[j] + t

The change in convolution at lag k is:
delta_k = -t * (a[_{i+k}] + a[_{i-k}]) + t * (a[_{j+k}] + a[_{j-k}])

where a[_{x}] = 0 if x is out of bounds.

This is a piecewise linear function of t (since we're looking at the max of linear functions). The optimal t is where the max is minimized, which occurs at a point where two or more of the linear functions are equal (or at a boundary).

However, finding the exact optimal t requires checking all pairs of lags, which is O(n^2) per pair. The golden section search is simpler and works well in practice.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with Nesterov momentum on log-sum-exp smoothed max-conv
2. Peak-targeted coordinate descent with golden-section search
3. Multi-length exploration with adaptive interpolation
4. Structured restarts to escape local minima
Key insight: asymmetric sequences can achieve scores below 2 because the
convolution center (a*a)[n-1] = sum_j a[j]*a[n-1-j] need not equal ||a||_2^2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def get_smoothed_weights(conv, T, top_k=120):
    size = len(conv)
    max_c = conv.max()
    if top_k < size and top_k > 0:
        part_idx = np.argpartition(conv, -top_k)[-top_k:]
        thresh = conv[part_idx].min()
        active = np.where(conv >= thresh)[0]
    else:
        active = np.arange(size)
    w_vals = (conv[active] - max_c) / T
    w_vals = np.clip(w_vals, -60, 0)
    w_exp = np.exp(w_vals)
    w_sum = w_exp.sum()
    weights = np.zeros(size)
    if w_sum > 1e-300:
        weights[active] = w_exp / w_sum
    else:
        weights[np.argmax(conv)] = 1.0
    return weights


def compute_grad(a, weights):
    n = len(a)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        k = int(k)
        wk = weights[k]
        if k == 0:
            grad += 2.0 * wk * a
        else:
            grad[:n - k] += wk * a[k:]
            grad[k:] += wk * a[:n - k]
    return grad


def gd_momentum(a_init, time_limit, T_init=None, top_k=100,
                lr_init=0.08, momentum=0.85, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    conv0 = fast_convolve(a)
    max0 = conv0.max()
    if T_init is None:
        T_init = max0 * 0.3
    T = min(T_init, 1.0)
    T_floor = max0 * 0.02
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    patience = 0
    max_patience = 300

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        weights = get_smoothed_weights(conv, T, top_k)
        grad = compute_grad(a, weights)

        v = momentum * v - lr * grad
        a_trial = a + v
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            v *= 0.3
            continue
        a_trial = a_trial / s

        score = compute_score(a_trial)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            if T > T_floor:
                T *= 0.998
        else:
            patience += 1
            if patience > max_patience:
                T = max(T_floor, T * 0.7)
                lr = max(0.005, lr * 0.6)
                v = np.zeros(n)
                a = best_a.copy()
                patience = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD it={it}, T={T:.5f}, lr={lr:.5f}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coord_descent(a_init, time_limit, verbose=True):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)

        # Peak-targeted pairs
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:top_k]

        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            # Pick random contributing positions
            if je < js:
                continue
            n_try = min(8, je - js + 1)
            candidates = rng.integers(js, je + 1, size=n_try)
            for j in candidates:
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(j)
                if a[j] < 1e-10:
                    continue
                # Find non-contributing targets
                contrib_mask = np.zeros(n, dtype=bool)
                contrib_mask[js:je + 1] = True
                non_contr = np.where(~contrib_mask)[0]
                if len(non_contr) == 0:
                    continue
                target = non_contr[rng.integers(0, len(non_contr))]

                t_lo = 1e-12
                t_hi = a[j] * 0.4
                if t_hi <= t_lo:
                    continue

                x1 = t_hi - gr * (t_hi - t_lo)
                x2 = t_lo + gr * (t_hi - t_lo)

                a_test = a.copy()
                f1 = _eval_pair(a_test, j, int(target), x1)
                f2 = _eval_pair(a_test, j, int(target), x2)

                a_buf = a.copy()
                for _ in range(18):
                    if time.monotonic() - t0 > time_limit:
                        break
                    if f1 < f2:
                        t_hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = t_hi - gr * (t_hi - t_lo)
                        f1 = _eval_pair(a_buf, j, int(target), x1)
                    else:
                        t_lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = t_lo + gr * (t_hi - t_lo)
                        f2 = _eval_pair(a_buf, j, int(target), x2)

                t_opt = (t_lo + t_hi) / 2.0
                f_opt = min(f1, f2)

                if f_opt < best_score - 1e-13:
                    a[j] -= t_opt
                    a[int(target)] += t_opt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_score = compute_score(a)
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                a = best_a + rng.standard_normal(n) * 0.004
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it := 0:
            pass
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 60 == 0 and elapsed > 0:
                print(f"    CD t={elapsed:.0f}s, score={best_score:.8f}")

    return best_a, best_score


def _eval_pair(a_base, i, j, t):
    a_buf = a_base.copy()
    a_buf[i] -= t
    a_buf[j] += t
    if a_buf[i] < 0 or a_buf[j] < 0:
        return 1e10
    s = a_buf.sum()
    if s < 1e-15:
        return 1e10
    a_buf = a_buf / s
    n = len(a_buf)
    conv = fast_convolve(a_buf)
    return 2.0 * n * conv.max()


def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n + 1)
    f_ext = np.append(f, 0.0)
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_ext)
    f_interp = f_interp[:-1]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def random_init(n, rng):
    a = rng.random(n)
    a += 0.1
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def structured_perturb(a, rng, scale=0.01):
    n = len(a)
    noise = rng.standard_normal(n) * scale
    # Add structured component: low-frequency
    freq = rng.integers(1, max(2, n // 50))
    t = np.linspace(0, 2 * np.pi * freq, n)
    noise += 0.5 * scale * np.cos(t)
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def evaluate_safe(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    try:
        sc = float(evaluate_sequence(a.tolist()))
```
