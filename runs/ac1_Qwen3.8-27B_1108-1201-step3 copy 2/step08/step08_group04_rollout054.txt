"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Uses a combination of:
1. Fast projected subgradient with Nesterov momentum (FFT-based gradient)
2. LP-based direction finding for refinement
3. Multi-start with diverse initializations
4. Coordinate descent polish
5. Adaptive n exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, tau):
    """
    Compute the soft-max subgradient of max(c) with respect to a.
    
    g[j] = sum_k softmax(c/tau)[k] * a[k-j]
    
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # Soft-max weights
    c_scaled = c / tau
    c_max = c_scaled.max()
    w = np.exp(c_scaled - c_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    
    # Compute g[j] = sum_k w[k] * a[k-j] via FFT
    # g = conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()  # enough for linear conv
    W = rfft(w, L)
    AR = rfft(a_rev, L)
    conv_result = irfft(W * AR, L)
    g = conv_result[n-1 : 2*n-1]
    
    return g


def nesterov_subgradient(a_init, time_limit, tau_init=0.01, tau_min=0.001,
                           nesterov_momentum=0.9, label="ng"):
    """
    Nesterov accelerated projected subgradient method.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    y = a.copy()  # Nesterov lookahead
    beta = nesterov_momentum  # momentum parameter
    tau = tau_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    patience = 0
    eta = 1.0 / (np.linalg.norm(a, 2) * 10)  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        if it > 1:
            beta_new = (1 + np.sqrt(1 + 4 * beta * beta)) / 2
            y = a + (beta / beta_new - 1) * (a - a_prev)
            beta = beta_new
        else:
            a_prev = a.copy()
        
        # Compute gradient at y
        c_y = fast_convolve(y)
        g = compute_subgradient(y, c_y, tau)
        
        # Gradient step
        a_new = y - eta * g
        a_new = project_simplex(a_new)
        
        # Gradient descent step size: eta_t = 1/(L*t) where L is Lipschitz constant
        # L ≈ max |g| * n (rough bound)
        # For now, use adaptive step size
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive step size
        if patience > 5:
            eta *= 0.8
            patience = 0
        elif it % 100 == 0 and best_obj < fast_convolve(a).max() * 0.999:
            eta *= 1.1
        
        # Adaptive temperature
        if it % 50 == 0:
            tau = max(tau * 0.95, tau_min)
        
        a = a_new
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"tau={tau:.6f} eta={eta:.2e} t={elapsed:.0f}s")
        
        if patience > 200:
            # Restart
            a = project_simplex(best_a + np.random.default_rng(it).standard_normal(n) * 0.01)
            y = a.copy()
            beta = nesterov_momentum
            eta = 1.0 / (n * 10)
            patience = 0
    
    return best_a, best_score


def lp_refine(a_init, time_limit, top_k_init=50, top_k_max=500, label="lp"):
    """LP-based refinement (similar to previous but optimized)."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    top_k = top_k_init
    momentum_dir = np.zeros(n)
    full_len = 2 * n - 1
    momentum_beta = 0.3
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 20, top_k_max)
        
        b = fast_convolve(a)
        M = b.max()
        
        # Peak selection
        if top_k < full_len:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            # Add boundary and center peaks
            n_pts = min(n, 5)
            extra = list(range(0, min(n_pts, full_len)))
            extra += list(range(max(0, full_len - n_pts), full_len))
            center = (full_len - 1) // 2
            extra += list(range(max(0, center - n_pts//2), min(full_len, center + n_pts//2 + 1)))
            all_idx = set(list(top_idx) + extra)
            all_idx = sorted([i for i in all_idx if 0 <= i < full_len], key=lambda i: b[i], reverse=True)
            indices = np.array(all_idx[:top_k])
        else:
            indices = np.arange(full_len)
        
        # Build constraint matrix
        K = len(indices)
        A_mat = np.zeros((K, n))
        for i in range(K):
            k = indices[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[i, lo:hi+1] = a[lo:hi+1]
        
        # Solve LP: max sum(g) s.t. g >= 0, A_mat @ g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A_mat @ g_var <= M])
        
        try:
            if K <= 500 and n <= 1500:
                prob.solve(solver=cp.ECOS, max_iters=50000)
            else:
                prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-7, eps_rel=1e-7)
            except Exception:
                stuck += 2
                if stuck > 50:
                    break
                continue
        
        if g_var.value is None:
            stuck += 2
            if stuck > 50:
                break
            continue
        
        g_val = np.maximum(g_var.value, 0.0)
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 40:
                break
            continue
        
        h = g_val / S_g
        
        # Momentum
        momentum_dir = momentum_beta * momentum_dir + (1 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h
        
        # Line search
        best_t = 0.0
        best_t_obj = M
        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                trial = (1 - t) * a + t * direction
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj_t = fast_convolve(trial).max()
                if obj_t < best_t_obj:
                    best_t_obj = obj_t
                    best_t = t
                    best_trial = trial.copy()
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        a = best_trial
        cur_obj = fast_convolve(a).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={K} t={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    best_a = a.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit and no_pass < 3:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-18:
                continue
            
            best_val = a[j]
            best_obj_j = best_obj
            for mult in multipliers:
                new_val = a[j] * mult
                a[j] = new_val
                s = a.sum()
                if s < 1e-12:
                    a[j] = best_val
                    continue
                a_norm = a / s
                obj_j = fast_convolve(a_norm).max()
                if obj_j < best_obj_j - 1e-14:
                    best_obj_j = obj_j
                    best_val = new_val
            a[j] = best_val
        
        s = a.sum()
        if s > 1e-12:
            a = a / s
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_pass += 1
    
    return best_a, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def generate_initializations(n, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine (Hann)
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(hann / hann.sum())
    
    # Gaussian (centered)
    sigma = n / 4
    gauss = np.exp(-0.5 * ((np.arange(n) - n/2) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Random
    r = np.abs(rng.standard_normal(n))
    inits.append(r / r.sum())
    
    # Power law
    p = np.arange(1, n+1) ** (-0.3)
    inits.append(p / p.sum())
    
    # Bimodal
    bimodal = np.exp(-0.5 * ((np.arange(n) - n*0.25) / (n*0.1))**2) + \
              np.exp(-0.5 * ((np.arange(n) - n*0.75) / (n*0.1))**2)
    inits.append(bimodal / bimodal.sum())
    
    # Sinc-like (from low-pass filter)
    L = 1 << (2*n - 1 - 1).bit_length()
    F = np.zeros(L)
    F[:n//4] = 1.0
    F[-n//4:] = 1.0
    F[0] = 1.0
    a_sinc = np.real(np.fft.ifft(F))[:n]
    a_sinc = np.maximum(a_sinc, 0)
    if a_sinc.sum() > 1e-10:
        inits.append(a_sinc / a_sinc.sum())
    else:
        inits.append(np.ones(n) / n)
    
    return inits
