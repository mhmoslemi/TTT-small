                    continue
                new_score = 2.0 * n * new_max / (new_s * new_s)

                if new_score < best_new_M:  # using best_new_M as best_score tracker
                    best_new_M = new_score
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                s += best_delta

                # Update conv
                changes = 2.0 * best_delta * a_contrib
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                M = max(U, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f} M={M:.8f}")

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps


def block_coord_descent(a0, time_limit, block_size=2, verbose=False, label=""):
    """
    Block coordinate descent: update pairs of coordinates simultaneously.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    s = 1.0
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(0, n - 1, block_size):
            j = i + 1
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue

            # Combined affected region
            k_start = max(0, min(i, j))
            k_end = min(max(i, j) + n - 1, 2 * n - 2)

            if k_start > 0:
                U = conv[:k_start].max()
            else:
                U = -1e30
            if k_end < 2 * n - 2:
                U2 = conv[k_end + 1:].max()
                U = max(U, U2)
            else:
                U = max(U, -1e30)

            affected = conv[k_start:k_end + 1].copy()
            len_aff = k_end - k_start + 1

            # Contribution vectors for i and j
            # For position k, change from delta_i: 2*delta_i*a[k-i] (if 0<=k-i<n)
            # For position k, change from delta_j: 2*delta_j*a[k-j] (if 0<=k-j<n)
            k_vals = np.arange(k_start, k_end + 1)
            contrib_i = np.zeros(len_aff)
            contrib_j = np.zeros(len_aff)
            mask_i = (k_vals >= i) & (k_vals - i < n)
            mask_j = (k_vals >= j) & (k_vals - j < n)
            contrib_i[mask_i] = a[k_vals[mask_i] - i]
            contrib_j[mask_j] = a[k_vals[mask_j] - j]

            # Quadratic terms at k=2i and k=2j
            quad_i = (2 * i >= k_start) & (2 * i <= k_end)
            quad_j = (2 * j >= k_start) & (2 * j <= k_end)
            quad_both = quad_i and quad_j and (2 * i == 2 * j)
            # Cross term at k = i+j: a[i]*a[j] appears twice, change involves delta_i*delta_j
            # Actually: c[i+j] includes a[i]*a[j] and a[j]*a[i] = 2*a[i]*a[j]
            # When we change both: (a[i]+di)(a[j]+dj) + (a[j]+dj)(a[i]+di)
            # = 2a[i]a[j] + 2a[i]dj + 2a[j]di + 2di*dj
            # So there's a cross term 2*di*dj at k=i+j
            cross_k = i + j
            has_cross = (cross_k >= k_start) and (cross_k <= k_end)

            # Try a few (delta_i, delta_j) combinations
            best_di, best_dj = 0.0, 0.0
            best_new_score = 2.0 * n * M / (s * s)

            steps = [-0.3, -0.1, 0.1, 0.3]
            for di_f in steps:
                di = di_f * a[i]
                if a[i] + di < -1e-15:
                    continue
                for dj_f in steps:
                    dj = dj_f * a[j]
                    if a[j] + dj < -1e-15:
                        continue

                    # Compute new affected
                    new_affected = affected + 2.0 * di * contrib_i + 2.0 * dj * contrib_j
                    # Quadratic at k=2i
                    if quad_i:
                        idx = 2 * i - k_start
                        new_affected[idx] += di * di
                    if quad_j:
                        idx = 2 * j - k_start
                        new_affected[idx] += dj * dj
                    # Cross term at k=i+j
                    if has_cross:
                        idx = cross_k - k_start
                        new_affected[idx] += 2.0 * di * dj

                    new_max = max(U, float(new_affected.max()))
                    new_s = s + di + dj
                    if new_s < 0.01:
                        continue
                    new_score = 2.0 * n * new_max / (new_s * new_s)
                    if new_score < best_new_score:
                        best_new_score = new_score
                        best_di, best_dj = di, dj

            if best_di != 0.0 or best_dj != 0.0:
                a[i] = max(a[i] + best_di, 0.0)
                a[j] = max(a[j] + best_dj, 0.0)
                s += best_di + best_dj

                # Update conv
                changes = 2.0 * best_di * contrib_i + 2.0 * best_dj * contrib_j
                if quad_i:
                    idx = 2 * i - k_start
                    changes[idx] += best_di * best_di
                if quad_j:
                    idx = 2 * j - k_start
                    changes[idx] += best_dj * best_dj
                if has_cross:
                    idx = cross_k - k_start
                    changes[idx] += 2.0 * best_di * best_dj
                conv[k_start:k_end + 1] += changes

                M = max(U, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a, best_score


def spectral_refine(a0, time_limit, verbose=False):
    """
    Refine using spectral domain operations.
    The auto-convolution c = IFFT(FFT(a)^2) (with zero-padding).
    We try to flatten c by adjusting the spectrum.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    fft_size = 1 << (2 * n - 1).bit_length() - 1
    # Actually let's use the same as fast_convolve
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute spectrum
        A = rfft(a, fft_size)
        # A^2 is the spectrum of the auto-convolution
        C_spec = A * A  # This is complex!
        # c = irfft(C_spec, fft_size)[:2*n-1]

        # Strategy: reduce the peaks in c by adjusting |A|
        # Find the peaks in c
        peaks = np.where(conv > 0.9 * M)[0]
        if len(peaks) == 0:
            peaks = [np.argmax(conv)]

        # For each peak, compute the "pressure" it exerts
        # and adjust the spectrum to reduce it
        # Simple approach: reduce |A[m]| for m that contribute to peaks
        # c[k] = (1/N) sum_m |A[m]|^2 exp(2i*theta_m) exp(2πikm/N)
        # where A[m] = |A[m]| exp(i*theta_m)
        # Actually c = irfft(A^2), so c[k] = (1/N) sum_m A[m]^2 exp(2πikm/N)
        # = (1/N) sum_m |A[m]|^2 exp(2i*arg(A[m])) exp(2πikm/N)

        # Simple heuristic: slightly reduce |A[m]| where the "energy" contributes to peaks
        # and increase elsewhere
        target = conv.mean()
        # Compute gradient with respect to |A[m]|
        # dc[k]/d|A[m]| = 2|A[m]| * cos(arg(A[m])) * cos(2π(k-m)N_neg/N) ...
        # This is getting complex. Let's use a simpler approach:
        # Just do a small multiplicative update on a based on the peak structure

        # Find positions contributing to the worst peaks
        worst_k = np.argmax(conv)
        # Perturb a at positions that contribute to worst_k
        i_start = max(0, worst_k - n + 1)
        i_end = min(n - 1, worst_k)
        contributors = np.arange(i_start, i_end + 1)

        # Reduce contributors and increase non-contributors
        alpha = 0.01 / (1 + 0.01 * it)
        a_new = a.copy()
        a_new[contributors] *= (1 - alpha)
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) > 0:
            a_new[non_contrib] *= (1 + alpha * len(contributors) / max(len(non_contrib), 1))
        a_new = normalize(a_new)

        new_conv = fast_convolve(a_new)
        new_M = float(new_conv.max())
        new_score = 2.0 * n * new_M
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            conv = new_conv
            M = new_M
        else:
            # Revert and try different peak
            if it % 100 == 0:
                best_k = np.argmax(conv)
                # Try different perturbation
                a_new = a + rng.standard_normal(n) * 0.01
                a_new = normalize(a_new)
                new_conv = fast_convolve(a_new)
                new_score = 2.0 * n * new_conv.max()
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    a = a_new

        if it % 500 == 0:
            print(f"    Spectral it={it}: score={best_score:.6f}")

    return best_a, best_score


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def mirror_descent(a0, time_limit, lr_init=0.005, top_k=20, verbose=False):
    """Mirror descent (multiplicative weight) for the problem."""
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.95
            no_improve = 0
            if lr < 1e-7:
                break

        # Subgradient: weight the top peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.02 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        # Gradient of max(conv) w.r.t. a
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Mirror descent update
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)

        lr *= 0.9998

        if verbose and it % 200 == 0:
            print(f"    Mirror it={it}: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except Exception:
            pass
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS
