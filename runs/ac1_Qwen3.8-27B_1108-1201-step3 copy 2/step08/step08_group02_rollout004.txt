"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp gradient descent, LP-based refinement,
spectral initialization, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, w):
    """Compute corr[j] = sum_m a[m]*w[j+m] for j=0..n-1 using FFT.
    w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    prod = irfft(A * W, fft_size)
    return prod[n - 1:n - 1 + n]


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - target) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient_step(a, beta, lr):
    """One projected gradient step on LSE(beta * conv(a,a))/beta.
    Returns (new_a, current_M, current_lse)."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    
    # LSE value
    lc = c * beta
    lc -= lc.max()
    lse_val = (lc.max() + np.log(np.exp(lc).sum())) / beta
    
    # Softmax weights
    w = np.exp(lc)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return a, M, lse_val
    w /= w_sum
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * cross_corr(a, w)[j]
    corr = fast_cross_corr(a, w)
    grad = 2.0 * corr
    
    # Gradient step with projection
    a_new = project_simplex(a - lr * grad, target=1.0)
    
    return a_new, M, lse_val


def lse_optimize(f_init, time_limit, beta_start=5.0, beta_end=500.0,
                 lr_init=0.005, label="lse"):
    """Adaptive log-sum-exp gradient descent with increasing temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-10:
        a /= s
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_score = 2.0 * n * best_M
    lr = lr_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)
    
    total_iters_target = max(100, int(time_limit * 80))
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive beta: increase over time
        progress = min(1.0, it / total_iters_target)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        a_new, M_cur, lse_val = lse_gradient_step(a, beta, lr)
        
        # Check improvement
        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase lr when improving
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.85
                no_improve = 0
            if no_improve > 100:
                # Perturb and restart from best
                a = best_a + rng.standard_normal(n) * 0.002
                a = project_simplex(a, 1.0)
                lr = lr_init
                no_improve = 0
                beta = max(beta_start, beta * 0.5)
        
        a = a_new
        s = a.sum()
        if s > 1e-10:
            a /= s
        
        if it % 500 == 0:
            cur_M = fast_conv(a).max()
            score = 2.0 * n * best_M
            print(f"  {label} it={it} M={best_M:.10f} score={score:.6f} "
                  f"beta={beta:.1f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
            if time.monotonic() - t0 > time_limit * 0.95:
                break
    
    return best_a, best_M


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def line_search(f, h, M_current, t_grid=None):
    """Line search for best step size t in f + t*(h - f)."""
    if t_grid is None:
        t_grid = np.array([0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7])
    best_t = 0.0
    best_obj = M_current
    
    for t in t_grid:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_conv(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t
    
    # Refine
    if best_t > 0:
        for dt in [best_t * 0.7, best_t * 1.3, best_t * 1.6]:
            if dt <= 0 or dt > 0.95:
                continue
            f_trial = (1.0 - dt) * f + dt * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_conv(f_trial).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = dt
    
    return best_t, best_obj


def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.96,
                    final_peak_frac=0.80, max_peaks=400, restart_patience=30,
                    lp_eps=1e-9):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_b = fast_conv(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_conv(f)
        M = b.max()
        
        # Adaptive peak selection
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 3:
            peaks = np.argpartition(b, -max(10, n))[-max(10, n):]
        
        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)
        
        if g is None:
            stuck += 1
            if stuck > restart_patience:
                # Try with fewer peaks
                peaks_small = np.argpartition(b, -min(20, len(b)))[-min(20, len(b)):]
                g = solve_lp_direction(f, n, peaks_small, M, eps=1e-7)
                if g is None:
                    if stuck > restart_patience * 2:
                        break
                    continue
            else:
                continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > restart_patience:
                break
            # Try with more peaks
            n_peaks_try = min(len(b), max_peaks * 2)
            more_peaks = np.argpartition(b, -n_peaks_try)[-n_peaks_try:]
            g2 = solve_lp_direction(f, n, more_peaks, M, eps=lp_eps)
            if g2 is not None and g2.sum() > 1.0 + 1e-5:
                g = g2
                S_g = g2.sum()
            else:
                continue
        
        h = g / S_g
        best_t, best_t_obj = line_search(f, h, M)
        
        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                # Perturb
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 0.0)
                f /= f.sum()
                peak_frac = max(final_peak_frac, peak_frac - 0.02)
                stuck = 0
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
        
        cur_obj = fast_conv(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 5 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.003)
        
        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} it={it} M={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} S_g={S_g:.4f} pf={peak_frac:.3f} t={time.monotonic()-t0:.0f}s")
        
        if stuck > restart_patience // 2 and stuck % 4 == 0:
            f = f + rng.standard_normal(n) * 0.0003 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_conv(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.2, 3.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                # Try adding a small value
                if f.sum() > 1e-10:
                    f_trial = f.copy()
                    f_trial[j] = 0.01 / n
                    f_trial /= f_trial.sum()
                    obj = fast_conv(f_trial).max()
                    if obj < best_b - 1e-14:
                        best_b = obj
                        f = f_trial
                        improved = True
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_conv(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            # Try random restarts
            if time.monotonic() - t0 < time_limit * 0.8:
                f = project_simplex(f + np.random.default_rng(hash(j) % 10000).standard_normal(n) * 0.005, 1.0)
                new_b = fast_conv(f).max()
                if new_b < best_b:
                    best_b = new_b
                else:
                    f = project_simplex(f, 1.0)
                    continue
            else:
                break
    
    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def cosine_window_init(n, power=0.5):
    """Create a cosine-windowed initialization."""
    x = np.linspace(0, 1, n)
    # Half-cosine raised: higher in middle
    w = np.cos((x - 0.5) * np.pi * power) ** 2
    w = np.maximum(w, 0.01)
    w /= w.sum()
    return w

    rng = np.random.default_rng(777)
    # Create a random complex sequence with flat magnitude
    freq = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    freq /= np.abs(freq)
    freq *= np.sqrt(n)  # normalize energy
    a = np.real(ifft(freq))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: combines LSE gradient descent, LP refinement,
    spectral initialization, multi-length search, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    best_seq = None
    best_score = np.inf
    
    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    else:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, uniform n=1024, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 0: Quick initialization choice ===
    # Try a few initializations and pick the best after quick LSE
    rem = deadline - time.monotonic()
    inits = [best_seq.copy()]
    
    # Cosine window init
    cw = cosine_window_init(n0, power=0.7)
    inits.append(cw)
    
    # Slightly peaked init
    x = np.linspace(0, 1, n0)
    peaked = 1.0 + 0.3 * np.cos(2 * np.pi * x)
    peaked = np.maximum(peaked, 0.1)
    peaked /= peaked.sum()
    inits.append(peaked)
    
    # Valley init (higher at edges)
    valley = 1.0 - 0.2 * np.cos(2 * np.pi * x)
    valley = np.maximum(valley, 0.1)
    valley /= valley.sum()
    inits.append(valley)
    
    # Quick LSE on each to find best start
    best_start = best_seq
    best_start_score = best_score
    t_quick = min(rem * 0.03, 30.0)
    for i, init in enumerate(inits):
        if i == 0:
            continue
        t_per = t_quick / (len(inits) - 1)
        if t_per < 3:
            continue
        try:
            fa, Ma = lse_optimize(init, time_limit=t_per, beta_start=5, beta_end=100,
                                   lr_init=0.003, label=f"init{i}")
            sa = 2.0 * n0 * Ma
            if sa < best_start_score:
                best_start_score = sa
                best_start = fa
                print(f"  Better init {i}: score={sa:.6f}")
        except Exception as e:
            print(f"  Init {i} failed: {e}")
    
    if best_start is not best_seq:
        best_seq = best_start
        best_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(best_score):
            print(f"  Updated best start: score={best_score:.6f}")
    
    # === Phase 1: LSE gradient descent (main) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LSE gradient (n={n0}, budget={t1:.0f}s) ===")
        try:
            f1, obj1 = lse_optimize(best_seq, time_limit=t1,
                                     beta_start=8.0, beta_end=800.0,
                                     lr_init=0.004, label="lse_main")
            score1 = 2.0 * n0 * obj1
            vs1 = float(evaluate_sequence(f1.tolist()))
            if np.isfinite(vs1) and vs1 < best_score:
                best_score = vs1
                best_seq = f1.copy()
                print(f"  NEW BEST [lse_main]: {vs1:.6f}")
        except Exception as e:
            print(f"  LSE main failed: {e}")
    
    # === Phase 2: LP refinement (main) ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: LP main (n={n_b}, budget={t2:.0f}s) ===")
        try:
            f2, obj2 = run_lp_optimize(best_seq, time_limit=t2, label="lp_main",
                                        initial_peak_frac=0.97, final_peak_frac=0.82,
                                        max_peaks=350, restart_patience=25, lp_eps=1e-9)
            score2 = 2.0 * n_b * obj2
            vs2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(vs2) and vs2 < best_score:
                best_score = vs2
                best_seq = f2.copy()
                print(f"  NEW BEST [lp_main]: {vs2:.6f}")
        except Exception as e:
            print(f"  LP main failed: {e}")
    
    # === Phase 3: Deeper LP ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 25:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: LP deep (n={n_b}, budget={t3:.0f}s) ===")
        try:
            f3, obj3 = run_lp_optimize(best_seq, time_limit=t3, label="lp_deep",
                                        initial_peak_frac=0.94, final_peak_frac=0.78,
                                        max_peaks=500, restart_patience=30, lp_eps=1e-10)
            score3 = 2.0 * n_b * obj3
            vs3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(vs3) and vs3 < best_score:
                best_score = vs3
                best_seq = f3.copy()
                print(f"  NEW BEST [lp_deep]: {vs3:.6f}")
        except Exception as e:
            print(f"  LP deep failed: {e}")
    
    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.05, 1.1, 1.15, 0.85]:
        tl = int(n_base * factor)
        tl = max(256, min(4096, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))
    
    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.10, 100.0)
        if t4 < 20:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 4: n={target_n} (budget={t4:.0f}s) ===")
        try:
            # Quick LSE + LP
            t4a = t4 * 0.4
            t4b = t4 * 0.6
            if t4a > 5:
                f4a, _ = lse_optimize(f_interp, time_limit=t4a,
                                       beta_start=10, beta_end=200,
                                       lr_init=0.003, label=f"lse_n{target_n}")
                f_interp = f4a
            f4, obj4 = run_lp_optimize(f_interp, time_limit=t4b, label=f"lp_n{target_n}",
                                        initial_peak_frac=0.96, final_peak_frac=0.82,
                                        max_peaks=300, restart_patience=15, lp_eps=1e-9)
            vs4 = float(evaluate_sequence(f4.tolist()))
            if np.isfinite(vs4) and vs4 < best_score:
                best_score = vs4
                best_seq = f4.copy()
                print(f"  NEW BEST [n{target_n}]: {vs4:.6f}")
        except Exception as e:
            print(f"  n={target_n} failed: {e}")
    
    # === Phase 5: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 150.0)
    if t5 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 5: CD polish (n={n_b}, budget={t5:.0f}s) ===")
        try:
            f5, obj5 = coordinate_descent(best_seq, time_limit=t5, label="cd")
            vs5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(vs5) and vs5 < best_score:
                best_score = vs5
                best_seq = f5.copy()
                print(f"  NEW BEST [cd]: {vs5:.6f}")
        except Exception as e:
            print(f"  CD failed: {e}")
    
    # === Phase 6: Final LP burst ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.30, 200.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Final LP (n={n_b}, budget={t6:.0f}s) ===")
        try:
            f6, obj6 = run_lp_optimize(best_seq, time_limit=t6, label="lp_final",
                                        initial_peak_frac=0.95, final_peak_frac=0.75,
                                        max_peaks=500, restart_patience=25, lp_eps=1e-10)
            vs6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(vs6) and vs6 < best_score:
                best_score = vs6
                best_seq = f6.copy()
                print(f"  NEW BEST [lp_final]: {vs6:.6f}")
        except Exception as e:
            print(f"  Final LP failed: {e}")
    
    # === Phase 7: Final CD + LSE micro-polish ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.20, 100.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final polish (budget={t7:.0f}s) ===")
        try:
            t7a = t7 * 0.4
            t7b = t7 * 0.6
            if t7a > 5:
                f7a, _ = lse_optimize(best_seq, time_limit=t7a,
                                       beta_start=50, beta_end=1000,
                                       lr_init=0.002, label="lse_final")
                vs7a = float(evaluate_sequence(f7a.tolist()))
                if np.isfinite(vs7a) and vs7a < best_score:
                    best_score = vs7a
                    best_seq = f7a.copy()
                    print(f"  NEW BEST [lse_final]: {vs7a:.6f}")
            if t7b > 5:
                f7b, _ = coordinate_descent(best_seq, time_limit=t7b, label="cd_final")
                vs7b = float(evaluate_sequence(f7b.tolist()))
                if np.isfinite(vs7b) and vs7b < best_score:
                    best_score = vs7b
                    best_seq = f7b.copy()
                    print(f"  NEW BEST [cd_final]: {vs7b:.6f}")
        except Exception as e:
            print(f"  Final polish failed: {e}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback to parent
        if parent is not None:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
