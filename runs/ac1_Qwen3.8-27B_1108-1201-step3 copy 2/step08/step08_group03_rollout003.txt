"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_full_lp(f, n, M, solver_preference=None):

def line_search(f, h, n, max_evals=300):

            best_t_grid = t

    # Golden section refinement
    lo = max(0.0, best_t_grid - 0.05)
    hi = min(1.0, best_t_grid + 0.05)
    for _ in range(20):
        if hi - lo < 1e-6:
            break
        m1 = lo + 0.382 * (hi - lo)
        m2 = hi - 0.382 * (hi - lo)
        ft1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
        s1 = ft1.sum()
        o1 = fast_convolve(ft1 / s1).max() if s1 > 1e-10 else best_obj
        ft2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
        s2 = ft2.sum()
        o2 = fast_convolve(ft2 / s2).max() if s2 > 1e-10 else best_obj
        if o1 < o2:
            hi = m2
        else:
            lo = m1

    t_best = (lo + hi) / 2.0
    ft = np.maximum((1 - t_best) * f + t_best * h, 0.0)
    s = ft.sum()
    if s > 1e-10:
        ft = ft / s
        obj = fast_convolve(ft).max()
        if obj < best_obj:
            best_t = t_best
            best_obj = obj

    return best_t, best_obj


def lp_optimize(f_init, time_limit, label="lp", solver_pref=None):
    """Iterative LP-based optimization: solve LP, line search, update."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
        s = 1.0
    f = f / s

    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Solve LP (full for small n, cutting plane for large)
        if n <= 1500:
            g = solve_full_lp(f, n, M, solver_preference=solver_pref)
        else:
            g = solve_lp_cutting_plane(f, n, M, max_k=400, initial_k=60,
                                       time_limit=min(rem * 0.3, 30.0),
                                       solver_pref=solver_pref)
            if g is None:
                g = solve_full_lp(f, n, M, solver_preference=solver_pref)

        if g is None:
            no_improve += 2
            if no_improve > 10:
                break
            # Perturb and retry
            f = project_simplex(f + 0.001 * np.random.default_rng(it).standard_normal(n))
            continue

        g = np.maximum(g, 0.0)
        S_g = g.sum()
        if S_g <= 1.0 + 1e-8:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        h = g / S_g
        no_improve = 0

        # Line search
        best_t, new_obj = line_search(f, h, n)

        if best_t < 1e-6:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            break
        f = f / s

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def ia_diversify(f_init, n_iters=100, peak_frac=0.3, rng=None):
    """IAA-style diversification in frequency domain."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        peak_val = R.max()
        floor = peak_frac * peak_val
        R_target = np.clip(R, floor, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()
        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = np.ones_like(F, dtype=complex)
        mask = abs_F > 1e-15
        phase[mask] = F[mask] / abs_F[mask]
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
    return f


def simulated_annealing(f_init, time_limit, rng=None):
    """SA polish: random perturbations on simplex, accept improvements."""
    if rng is None:
        rng = np.random.default_rng()
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temperature = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Random direction: pick 2 coordinates, move mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = temperature * f[i] * rng.uniform(0.5, 1.5)
        f_trial = f.copy()
        f_trial[i] = max(0.0, f[i] - delta)
        f_trial[j] = f[j] + delta
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()

        if obj < best_obj:
            best_obj = obj
            best_f = f_trial.copy()
            temperature = max(0.0005, temperature * 0.99)
        elif obj < f_obj if 'f_obj' in dir() else False:
            pass
        # Accept with probability based on improvement
        f_obj = fast_convolve(f).max() if it % 50 == 0 else M_cache
        if obj <= f_obj or (rng.random() < np.exp(-(obj - f_obj) / temperature)):
            f = f_trial
        if it % 100 == 0:
            temperature = max(0.0005, temperature * 0.97)
            if it % 500 == 0:
                print(f"    SA it={it} best_score={2.0*n*best_obj:.6f} temp={temperature:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coord_descent_polish(f_init, time_limit, n_sweeps=5):
    """Coordinate descent with multiplicative updates on blocks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    block_size = max(1, n // 200)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            end = min(start + block_size, n)
            block = f[start:end]
            if block.max() < 1e-12:
                continue
            # Try scaling the entire block
            best_scale = 1.0
            best_block_obj = best_obj
            for scale in [0.7, 0.85, 1.0, 1.15, 1.3]:
                f_trial = f.copy()
                f_trial[start:end] = block * scale
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_block_obj:
                    best_block_obj = obj
                    best_scale = scale
            f[start:end] = block * best_scale
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj:
                best_obj = cur_obj

    return f, best_obj


def make_initial(n, style="asym", rng=None):
    """Create diverse initial sequences."""
    if rng is None:
        rng = np.random.default_rng()
    if style == "uniform":
        f = np.ones(n)
    elif style == "asym":
        # Asymmetric: increase then decrease (shifted peak)
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.3) ** 2) / 0.1) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.1)
    elif style == "oscillatory":
        x = np.linspace(0, 1, n)
        f = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * x) + 0.2 * np.sin(2 * np.pi * 11 * x)
        f = np.maximum(f, 0.01)
    elif style == "random":
        f = rng.standard_normal(n) ** 2
    elif style == "two_bump":
        x = np.linspace(0, 1, n)
        f = np.exp(-((x - 0.25) ** 2) / 0.03) + np.exp(-((x - 0.75) ** 2) / 0.03)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: full-LP optimization with PDLP, multi-length exploration,
    IAA diversification, coordinate descent, and SA polish.
    """
    global M_cache
    M_cache = 0.0
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = make_initial(1152, style="asym", rng=rng)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n=1152, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Full LP on parent (main optimization) ===
    n_parent = len(parent)
    print(f"\n=== Phase 1: Full LP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = lp_optimize(parent, time_limit=t1, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {score1:.6f}")
        else:
            print(f"  [lp1] score={score1:.6f} (no improvement)")

    # === Phase 2: Deeper LP on current best ===
    print(f"\n=== Phase 2: Deep LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 20:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t2:.0f}s")
        f2, obj2 = lp_optimize(best_seq, time_limit=t2, label="lp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {score2:.6f}")
        else:
            print(f"  [lp2] score={score2:.6f}")

    # === Phase 3: IAA diversification + LP ===
    print(f"\n=== Phase 3: Diversification (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.2, 250.0)
    if t3_total > 40:
        n_b = len(best_seq)
        div_variants = [
            ("iaa_f02", 0.2, 150),
            ("iaa_f04", 0.4, 120),
            ("iaa_f06", 0.6, 100),
            ("osc", None, 0),
            ("two_bump", None, 0),
        ]
        per_div = t3_total / len(div_variants)
        for name, pkf, n_iter in div_variants:
            rem3 = deadline - time.monotonic()
            if rem3 < 20:
                break
            if pkf is not None:
                f_div = ia_diversify(best_seq, n_iters=n_iter, peak_frac=pkf, rng=rng)
            else:
                f_div = make_initial(n_b, style=name, rng=rng)
            t3a = min(per_div * 0.8, rem3 * 0.4)
            if t3a < 10:
                continue
            f3, _ = lp_optimize(f_div, time_limit=t3a, label=f"div_{name}")
            s3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(s3) and s3 < best_score:
                best_score = s3
                best_seq = f3.copy()
                print(f"  NEW BEST [{name}]: {s3:.6f}")
            else:
                print(f"  [{name}] score={s3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.8, 0.9, 1.0, 1.1, 1.25, 1.5]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base:
            target_lengths.append(tl)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem4 = deadline - time.monotonic()
        if rem4 < 50:
            break
        t4 = min(rem4 * 0.15, 150.0)
        if t4 < 20:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  n={target_n}, budget={t4:.0f}s")
        f4, _ = lp_optimize(f_interp, time_limit=t4, label=f"len{target_n}")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [n={target_n}]: {s4:.6f}")
        else:
            print(f"  [n={target_n}] score={s4:.6f}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coord descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.1, 80.0)
    if t5 > 10:
        f5, _ = coord_descent_polish(best_seq, time_limit=t5)
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {s5:.6f}")
        else:
            print(f"  [cd] score={s5:.6f}")

    # === Phase 6: SA polish ===
    print(f"\n=== Phase 6: SA polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.2, 150.0)
    if t6 > 15:
        global M_cache
        M_cache = fast_convolve(best_seq).max()
        f6, obj6 = simulated_annealing(best_seq, time_limit=t6, rng=rng)
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [sa]: {s6:.6f}")
        else:
            print(f"  [sa] score={s6:.6f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.3, 200.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"  n={n_b}, budget={t7:.0f}s")
        f7, _ = lp_optimize(best_seq, time_limit=t7, label="final")
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [final]: {s7:.6f}")
        else:
            print(f"  [final] score={s7:.6f}")

    # === Phase 8: Final coord descent ===
    rem = deadline - time.monotonic()
    if rem > 10:

        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [cd2]: {s8:.6f}")

    if not np.isfinite(fs):
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
