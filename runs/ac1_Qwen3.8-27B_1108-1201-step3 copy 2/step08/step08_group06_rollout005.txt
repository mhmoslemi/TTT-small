def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e15
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_C(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M, use_cvxpy=True):
    n = len(f)
    if use_cvxpy:
        g = _solve_lp_cvxpy(f, M)
        if g is not None:
            return g
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def _solve_lp_cvxpy(f, M):

    except ImportError:
        return None

def geometric_mean_update(f, g, t):
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def linear_update(f, g, t):
    a = (1 - t) * f + t * g
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

        conv = fast_convolve(a)
        num_pairs = min(n * 2, 2000)
        
        pairs = set()
        
        # Peak-focused pairs
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[::-1][:top_k]
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            rng_idx = np.arange(js, je + 1)
            for _ in range(3):
                i = rng_idx[rng.integers(0, len(rng_idx))]
                j = rng_idx[rng.integers(0, len(rng_idx))]
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(num_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Anti-peak pairs (edge to center)
        for _ in range(num_pairs // 4):
            i = rng.integers(0, max(1, n // 8))
            j = rng.integers(n - n // 8, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)

                a_trial[j] = max(0, a_trial[j])

                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)

                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)

    n = len(a0)

                break
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            
            # Try reducing top contributors
            top_ci = np.argsort(contribs)[::-1][:4]
            for ci in top_ci:
                i_idx = int(idx[ci])
                if a[i_idx] < 1e-14:

                    if os_sum < 1e-15:

                    a_trial[mask] = others + delta * others / os_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial

                        best_a = a.copy()

def lbgfs_logspace(a0, time_limit, T=500, maxiter=2000):
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe / a_safe.sum())
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def softmax_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s
    
    def func_and_grad(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z)
        a_tmp = softmax_z(z)
        conv = fast_convolve(a_tmp)
        shifted = conv / T
        max_sh = shifted.max()
        w = np.exp(shifted - max_sh)
        w = w / w.sum()
        log_sum_exp = max_sh + np.log(w.sum())
        obj = 2.0 * n * T * log_sum_exp
        
        fft_size = 1 << (conv.size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_tmp, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a_tmp)
        grad_z = a_tmp * (grad_a - S)
        
        sc = compute_score(a_tmp)
        if sc < best_score:
            best_score = sc
            best_a = a_tmp.copy()
        
        return obj, 2.0 * n * grad_z
    
    try:
        res = minimize(
            fun=None, x0=z0,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
    except Exception:
        pass
    
    # Manually do L-BFGS iterations
    z = z0.copy()
    for it in range(maxiter):
        if time.monotonic() - t0 > time_limit:
            break
        obj, grad = func_and_grad(z)
        if np.max(np.abs(grad)) < 1e-12:
            break
        # Simple gradient step with backtracking
        step = 0.1 / (1 + 0.01 * it)
        for _ in range(10):
            z_new = z - step * grad
            obj_new, _ = func_and_grad(z_new)
            if obj_new < obj:
                z = z_new
                break
            step *= 0.5
        else:
            break
    
    a_out = softmax_z(z)
    score = compute_score(a_out)
    if score < best_score:
        best_a = a_out
        best_score = score
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_weights_play(a0, time_limit, verbose=True):
    """Pure multiplicative weights / mirror descent on the max-convolution."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    k = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        peak_idx = np.argmax(conv)
        
        # Subgradient: gradient of (a*a)(peak_idx) w.r.t. a
        # (a*a)(k) = sum_j a(j) * a(k-j)
        # d/d a(j) = a(k-j) + a(j-k) (with appropriate bounds)
        grad = np.zeros(n)
        pk = int(peak_idx)
        for j in range(n):
            v1 = pk - j
            if 0 <= v1 < n:
                grad[j] += a[v1]
            v2 = j - pk
            if 0 <= v2 < n:
                grad[j] += a[v2]
        
        # Full subgradient using all peaks with weight 1/(num_peaks)
        # For simplicity, use just the top peak
        # Objective gradient (on simplex): grad - mean(grad)*a (tangential component)
        # Multiplicative weights: a_new = a * exp(-eta * grad) / norm
        
        eta = 0.5 / (k + 10)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        k += 1
        
        if verbose and k % 50 == 0:
            print(f"    MW iter={k}, eta={eta:.4f}, score={sc:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

            if sc_u < best_sc - 1e-15:
                best_sc = sc_u
                best_a_candidate = a_u
        
        # Also try the average of f and g
        a_avg = 0.5 * a + 0.5 * g
        a_avg = np.maximum(a_avg, 0)
        a_avg = a_avg / a_avg.sum()
        sc_avg = compute_score(a_avg)
        
        if best_a_candidate is not None and best_sc < compute_score(a):
            a_new = best_a_candidate
        elif sc_avg < compute_score(a):
            a_new = a_avg
            best_sc = sc_avg
            best_a_candidate = a_avg
        else:
            a_new = geometric_mean_update(a, g, 0.1)
            best_sc = compute_score(a_new)
            best_a_candidate = a_new
        
        if best_sc < best_score - 1e-14:
            best_score = best_sc
            best_a = a_new.copy()
        
        a = a_new
        k += 1
        
        if verbose and k % 20 == 0:
            print(f"    DG iter={k}, M={M:.6e}, score={best_sc:.8f}, best={best_score:.8f}, time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def try_length(base_seq, target_n, time_budget, verbose=False):
    """Try a specific length: interpolate, LP, CD, return best."""
    if target_n < 32 or target_n > 4000:
        return base_seq, compute_score(base_seq)
    
    a = interpolate_sequence(base_seq, target_n)
    t0 = time.monotonic()
    
    # Quick LP
    lp_budget = min(time_budget * 0.5, 60.0)
    if lp_budget > 5:
        a_lp, sc_lp = lp_fictitious_play(a, lp_budget, verbose=False)
        a = a_lp
    
    # Quick CD
    cd_budget = min(time_budget * 0.4, 40.0)
    if cd_budget > 3:
        a_cd, sc_cd = coordinate_descent(a, cd_budget, verbose=False)
        a = a_cd
    
    return a, compute_score(a)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining LP fictitious play, dual-guided updates,
    coordinate descent, peak surgery, and length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t_arr = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.05 * np.cos(t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    def _check(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
    
    # === Phase 1: LP Fictitious Play with Line Search ===
    print(f"\n=== Phase 1: LP Play + Line Search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(500.0, rem * 0.3)
    print(f"  Budget: {t1:.0f}s")
    a1, sc1 = lp_fictitious_play(best_seq, t1, verbose=True, line_search=True)
    _check(a1, "lp_play_ls")
    
    # === Phase 2: Dual-Guided Update ===
    print(f"\n=== Phase 2: Dual-Guided (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(300.0, rem * 0.2)
    if rem > t2 + 500:
        print(f"  Budget: {t2:.0f}s")
        a2, sc2 = dual_guided_update(best_seq, t2, verbose=True)
        _check(a2, "dual_guided")
    
    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(300.0, rem * 0.2)
    if rem > 30:
        print(f"  Budget: {t3:.0f}s")
        a3, sc3 = coordinate_descent(best_seq, t3, verbose=True, rng_seed=42)
        _check(a3, "cd")
    
    # === Phase 4: Peak Surgery ===
    print(f"\n=== Phase 4: Peak Surgery (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(200.0, rem * 0.15)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a4, sc4 = peak_surgery(best_seq, t4, verbose=True)
        _check(a4, "peak_surgery")
    
    # === Phase 5: More LP from improved start ===
    print(f"\n=== Phase 5: LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(200.0, rem * 0.2)
    if rem > 30:
        print(f"  Budget: {t5:.0f}s")
        a5, sc5 = lp_fictitious_play(best_seq, t5, verbose=True, line_search=True)
        _check(a5, "lp_polish")
    
    # === Phase 6: L-BFGS ===
    print(f"\n=== Phase 6: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for T_val in [300, 1000]:
        rem = deadline - time.monotonic()
        t6 = min(100.0, rem * 0.15)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t6:.0f}s")
        a6, sc6 = lbgfs_logspace(best_seq, t6, T=T_val, maxiter=1500)

    # === Phase 7: Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = []
    for dn in [-4, -2, -1, 1, 2, 4, 8, -8, 16]:
        alt_n = n_parent + dn
        if alt_n >= 64 and alt_n <= 3000:
            nearby_ns.append(alt_n)

    for alt_n in [512, 640, 896, 1024, 1152, 1280]:
        if alt_n not in nearby_ns:
            nearby_ns.append(alt_n)

        rem = deadline - time.monotonic()

            break
        t_alt = min(80.0, rem * 0.15)
        if t_alt < 15:

        a_alt, sc_alt = try_length(best_seq, alt_n, t_alt)

    # === Phase 8: Final Coordinate Descent ===
    print(f"\n=== Phase 8: Final CD Polish (best={best_score:.8f}) ===")

    t9 = min(100.0, rem * 0.3)

        print(f"  Budget: {t9:.0f}s")
        a9, sc9 = peak_surgery(best_seq, t9, verbose=False)
        _check(a9, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    return [float(x) for x in best_seq]
