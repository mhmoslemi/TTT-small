def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def objective(f):
    """Compute 2*n*max(conv(f,f)) / (sum(f))^2"""
    n = len(f)
    b = fast_convolve(f)
    max_b = b.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def objective_smooth(f, temp=0.01):
    """Smooth approximation of the max using log-sum-exp."""
    n = len(f)
    b = fast_convolve(f)
    # Log-sum-exp approximation of max
    b_max = b.max()
    lse = b_max + temp * np.log(np.sum(np.exp((b - b_max) / temp)))
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * lse / (sum_a ** 2)

def gradient_smooth(f, temp=0.01):
    """Gradient of the smooth objective."""
    n = len(f)
    b = fast_convolve(f)
    b_max = b.max()
    # Softmax weights
    w = np.exp((b - b_max) / temp)
    w /= w.sum()
    
    # Gradient of max_b w.r.t. f: for each k, d b[k]/d f[j] = f[k-j] (if valid)
    # So d max_b / d f[j] = sum_k w[k] * f[k-j]
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w[k] * f[k - (np.arange(lo, hi+1))]
    
    sum_a = f.sum()

    # = 2*n*(g[j]*sum_a - 2*max_b) / sum_a^3
    obj_grad = 2.0 * n * (g * sum_a - 2.0 * b_max) / (sum_a ** 3)
    return obj_grad, objective_smooth(f, temp)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_dual(f, n, M, active_idx):
    """Solve dual: min M*sum(y) s.t. A^T y >= 1, y >= 0.
    Dual optimal value = primal optimal value = max sum(g)."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    # Dual: min M * sum(y) s.t. A.T @ y >= ones(n), y >= 0
    y = cp.Variable(K)
    constraints = [y >= 0, A.T @ y >= np.ones(n)]
    prob = cp.Problem(cp.Minimize(M * cp.sum(y)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9)
    except Exception:
        return None
    if y.value is None:
        return None
    y_val = np.maximum(y.value, 0.0)
    if np.any(np.isnan(y_val)):
        return None
    # Recover primal solution from complementary slackness
    # g[j] > 0 only if (A^T y)[j] == 1 (binding in dual)
    # Use the dual to get the primal via: g = argmax{sum(g) : Ag <= M, g >= 0}
    # Just solve primal as fallback
    return None


def solve_lp_primal(f, n, M, active_idx, eps=1e-8):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g_val = None
    if g.value is None and n <= 3000:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def simulated_annealing_diversify(f_init, time_limit=60.0, n_restarts=5):
    """
    Simulated annealing to find diverse local optima.
    Returns a list of diverse normalized sequences.
    """
    t0 = time.monotonic()
    n = len(f_init)
    rng = np.random.default_rng(777)
    results = [f_init.copy()]
    f = f_init.copy()
    cur_obj = fast_convolve(f).max()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        f_pert = f + rng.standard_normal(n) * 0.02
        f_pert = project_simplex(f_pert)

        # Short SA run
        temp = 0.01
        f_sa = f_pert.copy()
        sa_obj = fast_convolve(f_sa).max()
        best_sa = f_sa.copy()
        best_sa_obj = sa_obj
        it = 0
        max_it = 200
        t_re = time_limit - (time.monotonic() - t0)
        time_per_iter = t_re / (max_it * (n_restarts - restart))

        while it < max_it and time.monotonic() - t0 < time_limit:
            it += 1
            # Random coordinate perturbation
            idx = rng.integers(0, n, size=max(1, n // 50))
            f_try = f_sa.copy()
            for j in idx:
                mult = 1.0 + rng.uniform(-0.3, 0.3)
                f_try[j] = max(0.0, f_try[j] * mult)
            f_try = project_simplex(f_try)
            try_obj = fast_convolve(f_try).max()

            if try_obj < sa_obj or rng.random() < math.exp((sa_obj - try_obj) / temp):
                f_sa = f_try
                sa_obj = try_obj
                if sa_obj < best_sa_obj:
                    best_sa_obj = sa_obj
                    best_sa = f_try.copy()

            temp *= 0.995

        results.append(best_sa)

    return results


def coordinate_descent(f_init, time_limit, step_factor=500):
    """Adaptive coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // step_factor)
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            # Try multiple multipliers
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True

        if not improved:
            # Try with smaller step
            step = max(1, step // 2)
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
        else:
            no_improve_sweeps = 0

    return f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=40, temp_init=0.005):
    """Subgradient method with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    rng = np.random.default_rng(123)
    momentum = 0.8
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Softmax weighting
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 6:
                    lr *= 0.85
                    temp *= 1.15
                    no_improve = 0
                if no_improve > 30:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

    return best_f, best_obj

    print(f"\n=== Phase 5: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 100.0)
    if t5 > 15:
        f5, _ = subgradient_polish(best_seq, time_limit=t5, lr_init=0.003,
                                   n_top=max(30, len(best_seq) // 15))
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {s5:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 180.0)
    if t6 > 15:
        print(f"  Budget: {t6:.0f}s, n={len(best_seq)}, k=80->500")
        f6, _ = lp_cutting_planes(best_seq, time_limit=t6, initial_k=80, max_k=500, label="final")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {s6:.6f}")

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 7: Coord descent (best={best_score:.6f}) ===")
        t7 = min(rem * 0.6, 80.0)
        f7, _ = coordinate_descent(best_seq, time_limit=t7)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {s7:.6f}")

    # === Phase 8: Final micro-polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 8: Micro-polish (best={best_score:.6f}) ===")
        t8 = min(rem * 0.8, 40.0)
        f8, _ = coordinate_descent(best_seq, time_limit=t8, step_factor=2000)
        s8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [micro]: {s8:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
