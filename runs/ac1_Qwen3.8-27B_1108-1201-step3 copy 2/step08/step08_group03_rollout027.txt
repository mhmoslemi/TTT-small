multi-start exploration, and coordinate descent polish.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_pdlp(f, n, M, active_idx, eps=1e-10):
    K = len(active_idx)
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.PDLP, eps_abs=eps, eps_rel=eps, max_iter=100000)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    if g_val.sum() <= 0:
        return None
    return g_val


def peak_shave(f, M, n_peaks=10, iterations=50):
    """
    Directly reduce peak by moving mass away from peak-contributing positions.
    For each peak at lag k, the contributing pairs are (j, k-j).
    We reduce the larger of each pair and add to a "sink" position.
    """
    n = len(f)
    f = f.copy()
    size = 2*n - 1
    
    for _ in range(iterations):
        b = fast_convolve(f)
        peak = b.max()
        if peak <= M * (1 + 1e-10):
            break
        
        # Find top peaks
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        top_idx = sorted(top_idx.tolist(), key=lambda k: -b[k])
        
        for k in top_idx:
            if b[k] <= M * (1 + 1e-8):
                break
            # Find contributing pairs
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Pairs: (j, k-j) for j in [lo, hi]
            js = np.arange(lo, hi+1)
            partners = k - js
            products = f[js] * f[partners]
            
            if products.sum() <= 0:
                continue
            
            # Find the pair with the largest product
            max_idx = np.argmax(products)
            j_star = js[max_idx]
            p_star = partners[max_idx]
            
            # Move a small amount from the larger of the two
            if f[j_star] >= f[p_star]:
                src = j_star
                dst = p_star  # actually we want to move to somewhere else
            else:
                src = p_star
                dst = j_star
            
            # Find a good destination: a position with low contribution to all peaks
            # Simple heuristic: move to a position with small value
            candidate_positions = np.argsort(f)[:n//4]  # bottom quarter
            # Choose the one farthest from all peak-contributing positions
            best_dst = src  # default: no move
            best_score = -np.inf
            for pos in candidate_positions[:20]:
                # Score: how much would moving here reduce the peak?
                # Moving from src to pos changes conv at lags involving src and pos
                # This is complex, so use a simple heuristic
                if pos == src:
                    continue
                # Distance from peak-contributing positions
                min_dist = min(abs(pos - j) for j in js)
                score = min_dist * f[pos]  # prefer far positions with small value
                if score > best_score:
                    best_score = score
                    best_dst = pos
            
            if best_dst == src:
                continue
            
            # Move a small amount
            amt = f[src] * 0.01
            if amt > 1e-15:
                f[src] -= amt
                f[best_dst] += amt
                f = np.maximum(f, 0)
                s = f.sum()
                f /= s
    
    return f

def coordinate_descent_polish(f_init, time_limit, n_trials_per_coord=5):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    # Process coordinates in groups for efficiency
    step = max(1, n // 200)  # process every 'step'th coordinate
    
    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Randomize order
        indices = np.arange(0, n, step)
        np.random.shuffle(indices)
        
        for j in indices:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            
            best_val = f[j]
            best_local_obj = best_obj
            
            try mults = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
            for mult in try_mults:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_local_obj - 1e-14:
                    best_local_obj = obj
                    best_val = best_val * mult
            
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            
            if best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                improved = True
        
        if not improved and sweep > 3:
            break
    
    return f, best_obj

"""
Optimization via PDLP-based cutting-plane LP with adaptive active sets,
peak shaving, multi-start diversification, length exploration, and
coordinate descent polish. Uses faster PDLP solver and more aggressive
constraint management to push below the 1.5027 target.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def build_A(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx, tol=1e-9):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    g_val = None
    try:
        prob.solve(solver=cp.PDLP, eps_abs=tol, eps_rel=tol, max_iter=200000)
        g_val = g.value
    except Exception:
        pass
    if g_val is None:
        try:
            prob.solve(solver=cp.SCS, eps=tol, max_iters=200000)
            g_val = g.value
        except Exception:
            pass
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() <= 1e-15:
        return None
    return g_val


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def line_search(f, h, M_init, n_scan=12):
    best_t = 0.05
    best_obj = M_init
    scan_t = np.linspace(0.01, 1.0, n_scan)
    for t in scan_t:
        f_t = (1.0 - t) * f + t * h
        f_t = np.maximum(f_t, 0.0)
        s = f_t.sum()
        if s < 1e-10:
            continue
        f_t = f_t / s
        obj = fast_convolve(f_t).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    if 0.005 < best_t < 0.98:
        lo = max(0.0, best_t - 0.08)
        hi = min(1.0, best_t + 0.08)
        for _ in range(10):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            f1 = np.maximum((1.0 - m1) * f + m1 * h, 0.0)
            s1 = f1.sum()
            if s1 > 1e-10:
                f1 /= s1
            else:
                f1 = f
            f2 = np.maximum((1.0 - m2) * f + m2 * h, 0.0)
            s2 = f2.sum()
            if s2 > 1e-10:
                f2 /= s2
            else:
                f2 = f
            o1 = fast_convolve(f1).max()
            o2 = fast_convolve(f2).max()
            if o1 < o2:
                hi = m2
            else:
                lo = m1
        t_mid = (lo + hi) * 0.5
        f_mid = np.maximum((1.0 - t_mid) * f + t_mid * h, 0.0)
        s_mid = f_mid.sum()
        if s_mid > 1e-10:
            f_mid /= s_mid
            o_mid = fast_convolve(f_mid).max()
            if o_mid < best_obj:
                best_obj = o_mid
                best_t = t_mid
    return best_t, best_obj


def run_lp_opt(f_init, time_limit, init_k=80, max_k=600, label="lp"):

    it = 0
    stuck = 0
    no_improve = 0
    best_obj_stagnant = M

        stuck = 0

        best_t, best_t_obj = line_search(f, h, M)

def peak_shave(f_init, n_shaves=200, n_peaks=8):
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    for _ in range(n_shaves):
        b = fast_convolve(f)
        peak = b.max()
        top = np.argpartition(b, -n_peaks)[-n_peaks:]
        for k in sorted(top.tolist(), key=lambda x: -b[x]):
            if b[k] <= peak * (1.0 - 1e-6):
                break
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            partners = k - js
            prods = f[js] * f[partners]
            if prods.sum() < 1e-20:
                continue
            max_i = np.argmax(prods)
            j1 = js[max_i]
            j2 = partners[max_i]
            if f[j1] >= f[j2]:
                src = j1
            else:
                src = j2
            candidates = np.argsort(f)[:max(10, n // 8)]
            best_dst = -1
            best_d = -1.0
            for pos in candidates:
                if pos == src:
                    continue
                d = min(abs(int(pos) - int(j)) for j in js) if len(js) > 0 else n
                sc = float(d) - 2.0 * float(f[pos]) * n
                if sc > best_d:
                    best_d = sc
                    best_dst = int(pos)
            if best_dst < 0:
                continue
            amt = f[src] * 0.005
            if amt > 1e-16:
                f[src] -= amt
                f[best_dst] += amt
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f /= s
    return f


def coord_polish(f_init, time_limit):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    mults = [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]
    rng = np.random.default_rng(7)
    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        idx = np.arange(0, n, step)
        rng.shuffle(idx)
        for j in idx:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-16:
                continue
            bv = f[j]
            bo = best_obj
            for m in mults:
                f[j] = bv * m
                s = f.sum()
                if s < 1e-10:
                    f[j] = bv
                    continue
                fn = f / s
                obj = fast_convolve(fn).max()
                if obj < bo - 1e-15:
                    bo = obj
                    bv = bv * m
            f[j] = bv
            s = f.sum()
            if s > 1e-10:
                f /= s
            if bo < best_obj - 1e-15:
                best_obj = bo
                improved = True
        if not improved and sweep > 5:
            break
    return f, best_obj
