"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Chebyshev-inspired and parametric initializations targeting U-shaped profiles
2. Log-sum-exp smoothed gradient method (new optimization paradigm)

4. Multi-start diversity with adaptive n search
5. Low-dimensional polynomial parameterization for global search

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_to_sum1(f):
    s = f.sum()
    if s > 1e-15:
        return f / s
    return np.ones(len(f)) / len(f)


def compute_score(f):
    """Compute the objective (assuming sum=1)."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()


def generate_chebyshev_init(n, order=3, alpha=0.7):
    """Generate a Chebyshev-inspired window (high at edges, low center)."""
    t = 2.0 * np.arange(n) / (n - 1) - 1.0  # [-1, 1]
    # T_order(t) = cos(order * arccos(t))
    T = np.cos(order * np.arccos(np.clip(t, -1, 1)))
    # Mix: high where T is high (edges for odd order, center for even)
    # For odd order, T is high at edges and low at center
    a = 1.0 + alpha * T
    a = np.maximum(a, 0.01)
    return normalize_to_sum1(a)


def generate_powerlaw_init(n, p=0.5, side='both'):
    """Generate a power-law edge-concentrated sequence."""
    t = np.arange(n) / (n - 1)  # [0, 1]
    if side == 'both':
        # U-shaped
        a = (abs(t - 0.5) * 2.0) ** p + 0.1
    elif side == 'right':
        a = t ** p + 0.1
    elif side == 'left':
        a = (1.0 - t) ** p + 0.1
    else:
        a = np.ones(n)
    return normalize_to_sum1(a)


def generate_exponential_init(n, beta=1.0, side='both'):
    """Generate an exponential edge-concentrated sequence."""
    t = np.arange(n) / (n - 1)
    if side == 'both':
        a = np.exp(beta * (2.0 * t - 1.0) ** 2 - beta) + 0.05
    elif side == 'right':
        a = np.exp(beta * t) + 0.05
    else:
        a = np.exp(-beta * t) + 0.05
    return normalize_to_sum1(a)


def generate_polynomial_init(n, coeffs=None):
    """Generate sequence from polynomial in t = 2*i/(n-1) - 1."""
    t = 2.0 * np.arange(n) / (n - 1) - 1.0
    if coeffs is None:
        # Default: 1 + 0.5*t^2 (U-shaped)
        a = 1.0 + 0.5 * t**2
    else:
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * t**i
    a = np.maximum(a, 0.01)
    return normalize_to_sum1(a)


def generate_gaussian_init(n, k=2, rng=None):
    """Generate sequence as sum of Gaussian bumps."""
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n) / (n - 1)
    a = np.zeros(n)
    for _ in range(k):
        mu = rng.uniform(0, 1)
        sig = rng.uniform(0.02, 0.15)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((t - mu) / sig) ** 2)
    a = np.maximum(a, 0.001)
    return normalize_to_sum1(a)

        labels.append(f"exp_b{beta}")

        labels.append(f"gauss_k{k}")

    labels.append("uniform")

    return inits, labels


def solve_lp_direction(f, indices, M):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f, g)[k] <= M for k in indices.
    Returns g or None.
    """
    n = len(f)
    K = len(indices)
    if K == 0 or n == 0:
        return None

    # Build constraint matrix
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]

    prob = cp.Problem(objective, constraints)

    try:
        if K <= 600 and n <= 1200:
            prob.solve(solver=cp.ECOS, max_iters=30000, abs_acceleration=1e-10)
        elif K <= 2000 and n <= 3000:
            prob.solve(solver=cp.PDLP, max_iter=50000, time_limit=min(30, 30))
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

                extra_idx = remaining[np.argsort(b[remaining])[-extra_k:]]
                indices = np.sort(np.concatenate([idx_above, extra_idx]))
            else:
                indices = idx_above
        else:
            indices = idx_above

            tol_frac = max(tol_frac * 0.95, 0.1)

            # Perturb and continue

def run_coordinate_descent(f_init, time_limit, label=""):
    """Adaptive coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = normalize_to_sum1(f_init.copy())
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    multipliers = [0.3, 0.5, 0.7, 0.85, 0.9, 1.0, 1.1, 1.15, 1.2, 1.3, 1.5, 1.8, 2.5]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = f[j]
            f_backup = f.copy()

            for mult in multipliers:
                f_trial = f_backup.copy()
                f_trial[j] = f_backup[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f_backup[j] * mult

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
