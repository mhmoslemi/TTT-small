def fast_self_conv(a):
    """Compute linear self-convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def compute_grad_weights(w, a):
    """
    Compute grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for i in [0, n-1].
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    # conv(w, a) has length (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 3).bit_length()
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)
    
    # term2[i] = (w * a)[i] for i in [0, n-1]
    conv_wa = np.fft.irfft(W * A, N)
    term2 = conv_wa[:n]
    
    # term1[i] = (w_rev * a)[(2n-2)-i] for i in [0, n-1]
    conv_wra = np.fft.irfft(Wr * A, N)
    term1 = conv_wra[n-1:2*n-1][::-1]
    
    return term1 + term2
