    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def subgradient_at_k(a, k, n):
    """Subgradient of conv(a,a)[k] w.r.t. a: grad[m] = 2*a[k-m] if 0<=k-m<n."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for m in range(lo, hi):
        j = k - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed*b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov LP best-response with stochastic active sets and ergodic averaging."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    ergodic = a.copy()
    ergodic_count = 0
    t0 = time.monotonic()
    stall_count = 0
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: mix of top peaks and random near-max
        progress = min(1.0, t / 500.0)
        # Start with more constraints, tighten over time
        n_active = max(8, int(n * (0.25 - 0.15 * progress)))
        n_active = min(n_active, int(0.5 * n))

        tol = 0.10 * (1.0 - progress) + 0.01 * progress
        base_threshold = M_curr * (1.0 - tol)
        candidates = np.where(conv_y >= base_threshold)[0]

        # Always include top peaks
        n_top = max(5, min(20, n // 80))
        top_k = np.argsort(conv_y)[-n_top:]
        candidates = np.unique(np.concatenate([candidates, top_k]))

        # Random subset for stochasticity
        if len(candidates) > n_active:
            # Keep top n_active//2, random rest
            top_idx = np.argsort(conv_y[candidates])[-(n_active // 2):]
            top_idx = candidates[top_idx]
            remaining = np.setdiff1d(candidates, top_idx)
            if len(remaining) > 0:
                n_rand = min(len(remaining), n_active - len(top_idx))
                rand_idx = rng.choice(remaining, size=n_rand, replace=False)
                candidates = np.concatenate([top_idx, rand_idx])
            else:
                candidates = top_idx
        active_set = candidates[:n_active]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Retry with full active set (small)
            active_set = np.argsort(conv_y)[-max(10, n // 40):]
            b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Ergodic average
        ergodic_count += 1
        ergodic = ergodic * (ergodic_count - 1) / ergodic_count + a * (1.0 / ergodic_count)
        s_e = ergodic.sum()
        if s_e > 1e-15:
            ergodic = ergodic * norm / s_e

        # Track best among y, a, ergodic
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if n_iters % 20 == 0:
            score_e = fast_autocorr(ergodic).max()
            if score_e < best_score:
                best_score = score_e
                best_a = ergodic.copy()
                stall_count = 0

        # Stall restart
        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            ergodic = best_a.copy()
            ergodic_count = 0
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, verbose=False, seed=42):
    """Multi-peak subgradient with adaptive step and Nesterov momentum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 2.0 / n ** 0.4
    patience = 0
    max_patience = 800
    momentum = 0.0
    v = np.zeros(n)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Select active peaks with diversity
        n_peaks = max(5, min(15, n // 60))
        sorted_idx = np.argsort(conv)[::-1]
        top_indices = sorted_idx[:n_peaks]

        # Add random near-max for exploration
        if t % 10 == 0:
            threshold = M_curr * (0.8 + 0.15 * rng.random())
            candidates = np.where((conv >= threshold) & (conv < M_curr))[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices, k_rand)

        # Exponential weights
        diffs = M_curr - conv[top_indices]
        scale = max(M_curr * 0.03, 1e-10)
        weights = np.exp(-diffs / scale)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for m in range(lo, hi):
                j = k - m
                if 0 <= j < n:
                    grad[m] += 2.0 * w * a[j]

        # Project gradient to sum-zero subspace
        support_thresh = 0.001 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            support = np.arange(n)
        grad_mean = np.mean(grad[support])
        grad_proj = grad - grad_mean

        # Nesterov momentum
        momentum = 0.9 * (1.0 - 50.0 / (t + 50.0))
        v = momentum * v + (1.0 - momentum) * grad_proj

        # Backtracking line search
        best_trial = None
        best_trial_score = best_score
        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            eta = step * lr_mult
            a_trial = a - eta * v
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial * norm / s_trial
            score_trial = fast_autocorr(a_trial).max()
            if score_trial < best_trial_score:
                best_trial = a_trial
                best_trial_score = score_trial
            if score_trial < M_curr * 0.999:
                break

        if best_trial is not None and best_trial_score < best_score - 1e-14:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            patience = 0
            step = min(step * 1.02, 3.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-12:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with efficient autocorrelation tracking."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Candidates
            typical = norm / n
            candidates = sorted(set([
                0.0, aj * 0.2, aj * 0.5, aj * 0.8, aj,
                aj * 1.3, aj * 1.7, aj * 2.5,
                typical * 0.3, typical, typical * 2.0, typical * 4.0
            ]))
            candidates = [max(0.0, min(c, norm * 0.4)) for c in candidates]

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-13:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-13:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

    remaining = time_budget - (time.monotonic() - t_global)

    if best_seq is not None:

    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
