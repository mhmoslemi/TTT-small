$g_i = 2 \sum_j u(i+j) a'(n-1-j)$

Let $m = n-1-j$: $j = n-1-m$
$g_i = 2 \sum_m u(i + n-1 - m) a'(m)$

import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_grad_fft(a, p):
    """
    Compute gradient of sum_k p_k * (a*a)_k w.r.t. a.
    p has length 2n-1 (weights for each convolution lag).
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    size = 2 * n - 1
    # g_i = 2 * sum_j p[i+j] * a[j]
    # = 2 * (p * a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    fft_size = 1 << (size + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(P * A, fft_size)[:size + n - 1]
    return 2.0 * conv[n - 1:2 * n - 1]

def hard_subgrad(a, k):
    """Exact subgradient of (a*a)_k w.r.t. a."""
    n = len(a)
    k = int(k)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    grad = np.zeros(n)
    idx = np.arange(i_start, i_end)
    j = k - idx
    valid = (j >= 0) & (j < n)
    grad[idx[valid]] += 2.0 * a[j[valid]]
    return grad

def norm_sum1(a):

    if s < 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_mirror_descent(a0, time_limit, c0=4.0, top_k_list=(1, 2, 4),
                            verbose=False, seed=42):
    """
    Nesterov accelerated mirror descent with cycling hard subgradients.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    ti = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution and find top peaks
        ac = fast_conv(y)

        # Cycle through different top_k values
        top_k = top_k_list[ti % len(top_k_list)]
        ti += 1

        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / w_sum

        # Weighted subgradient
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            g += wi * hard_subgrad(y, int(k))

        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic perturbation
        if stall > max_stall:
            sigma = 0.005 + 0.01 * (stall // (2 * max_stall))
            x_pert = best_a + rng.normal(0, sigma, n) * best_a
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            M_pert = fast_conv(x_pert).max()
            if M_pert < best_m:
                best_m = M_pert
                best_a = x_pert.copy()
            x = x_pert
            x_prev = x_pert.copy()
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m

                    e2 = min(n, k2_int + 1)
                    exclude[s2:e2] = True

                candidates = np.where(~exclude)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)

                # Sample candidates
                n_sample = min(len(candidates), 100)
                chosen = candidates[rng.choice(len(candidates), size=n_sample, replace=False)]

                best_dst = -1
                best_score = float('inf')
                for d in chosen:
                    if d == src or a[d] > 0.5 * a[src]:
                        continue
                    sc = scores[d]
                    if sc < best_score:
                        best_score = sc
                        best_dst = d

                if best_dst < 0:
                    continue

                # Try multiple transfer fractions
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                n_moves = rng.integers(1, 5)
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0.0) / a.sum()

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD final ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def local_2opt(a0, time_limit, verbose=False, seed=42):
    """
    2-opt style: swap mass between pairs of positions.
    More sophisticated than single-coordinate moves.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_peaks = np.argsort(ac)[::-1][:5]
        improved = False

        n_tries = min(200, n // 10)
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick two random distinct positions
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue

            # Try partial swap
            for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                amt = alpha * min(a[i], a[j])
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] -= amt
                a_trial[i] += alpha * a[j]
                a_trial[j] += alpha * a[i]
                # Fix: the above isn't quite right. Let's do a proper partial swap.
                a_trial = a.copy()
                ai, aj = a[i], a[j]
                a_trial[i] = (1 - alpha) * ai + alpha * aj
                a_trial[j] = (1 - alpha) * aj + alpha * ai
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
        else:
            stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    2-opt ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1750.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smooth mirror descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth MD n={n0} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t1,
                                         beta_start=2.0, beta_end=300.0,
                                         eta0=0.8, anneal_frac=0.5,
                                         verbose=True, seed=42)
        try_update(a_out, f"smd_n{n0}")

    # === Phase 2: Nesterov MD with hard subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Nesterov MD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t2,
                                           c0=5.0, top_k_list=(1, 2, 3),
                                           verbose=True, seed=7)
        try_update(a_out, f"nmd_n{len(best_seq)}")

    # === Phase 3: Smooth MD round 2 (from best so far) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Smooth MD round 2 n={len(best_seq)} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t3,
                                         beta_start=50.0, beta_end=500.0,
                                         eta0=0.3, anneal_frac=0.3,
                                         verbose=True, seed=99)
        try_update(a_out, f"smd2_n{len(best_seq)}")

    # === Phase 4: Nesterov MD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov MD round 2 n={len(best_seq)} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t4,
                                           c0=6.0, top_k_list=(1, 1, 2),
                                           verbose=True, seed=123)
        try_update(a_out, f"nmd2_n{len(best_seq)}")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate descent n={len(best_seq)} ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t5,
                                      verbose=True, seed=42, n_inner=4)
        try_update(a_out, f"cd_n{len(best_seq)}")

    # === Phase 6: 2-opt ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: 2-opt n={len(best_seq)} ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 10:
        a_out, _ = local_2opt(norm_sum1(best_seq), t6,
                              verbose=True, seed=77)
        try_update(a_out, f"2opt_n{len(best_seq)}")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final CD n={len(best_seq)} ===")
    t7 = min(remaining * 0.10, 150.0)
    if t7 > 10:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t7,
                                      verbose=True, seed=314, n_inner=6)
        try_update(a_out, f"cd_final_n{len(best_seq)}")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        for delta in [200, -200, 400, -400]:
            n_try = n_best + delta
            if n_try < 1000 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 8: Length n={n_try} ===")
            a_interp = interpolate_sequence(best_seq, n_try)
            # Quick smooth optimization
            t8 = min(remaining * 0.25, 120.0)
            if t8 > 20:
                a_out, _ = smooth_mirror_descent(norm_sum1(a_interp), t8,
                                                 beta_start=10.0, beta_end=300.0,
                                                 eta0=0.5, anneal_frac=0.4,
                                                 verbose=False, seed=55 + abs(delta))
                try_update(a_out, f"smd_n{n_try}")
                # Quick CD
                remaining = time_budget - (time.monotonic() - t_global)
                t8b = min(remaining * 0.10, 50.0)
                if t8b > 10:
                    a_out2, _ = coordinate_descent(norm_sum1(best_seq), t8b,
                                                   verbose=False, seed=88 + abs(delta), n_inner=3)
                    try_update(a_out2, f"cd_n{n_try}")

    # === Phase 9: Final intensive CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Intensive final CD n={n_best} ===")
    if remaining > 30:
        t9 = min(remaining * 0.7, 200.0)
        if t9 > 15:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t9,
                                          verbose=True, seed=271, n_inner=8)
            try_update(a_out, f"cd_intensive_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
