"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative spectral flattening (Fourier-domain amplitude equalization)
2. Projected subgradient with Anderson acceleration and adaptive step sizes
3. Pair-coordinate mass transfer optimization

5. Final coordinate polish with bisection

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_flatten(a, n_iter=50, damp=0.5):
    """Flatten power spectrum via iterative amplitude equalization."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        # Target: flat amplitude
        target = amp.mean()
        # Iteratively adjust: new_amp = (amp + target) * 0.5 (damped)
        new_amp = damp * amp + (1 - damp) * target
        # Preserve phase
        phase = A / (amp + 1e-30)
        A_new = new_amp * phase
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    return a

def spectral_flatten_v2(a, n_iter=50, power=0.5):
    """Alternative: raise amplitude to a power < 1 to flatten."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        # Compress dynamic range
        amp_max = amp.max()
        amp_min = amp[amp > 1e-10].min() if np.any(amp > 1e-10) else 1e-10
        # Normalize to [0,1] then apply power
        amp_norm = (amp - amp_min) / (amp_max - amp_min + 1e-15)
        amp_new = amp_min + (amp_max - amp_min) * (amp_norm ** power)
        # Or: geometric mean of amp and target
        target = np.geometric_mean(amp[amp > 1e-10]) if np.any(amp > 1e-10) else 1.0
        amp_new = np.sqrt(amp * target)  # geometric mean approach
        phase = A / (np.abs(A) + 1e-30)
        A_new = amp_new * phase
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a

def subgradient_topk(a, ac, top_k=30, beta=10.0):
    """Compute subgradient of max(ac) using top-k peaks with softmax weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Softmax weighting
    w = np.exp(beta * (top_vals - top_vals[0]))
    w = w / w.sum()
    
    # Gradient of c[k] wrt a[j] = 2*a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # a[k-j] for j in range(n): this is a shifted version of a
        # For j from max(0, k-(n-1)) to min(k, n-1): a[k-j] is valid
        j_start = max(0, k_int - (n - 1))
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * wi * a[k_int - js]
    return grad

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Algorithm from "Euclidean projection onto the simplex"
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def anderson_accelerate(x, g, history_X, history_G, m=5):
    """Simple Anderson acceleration. Returns new x."""
    if len(history_X) < m:
        # Just do a gradient step
        return x - g * (1.0 / (1 + len(history_X)))
    
    # Build the Anderson system
    # s_i = x_i - x_{i-1}, y_i = g_i - g_{i-1}
    X = np.array(history_X[-m:])  # (m, n)
    G = np.array(history_G[-m:])  # (m, n)
    
    s = np.diff(X, axis=0)  # (m-1, n)
    y = np.diff(G, axis=0)  # (m-1, n)
    
    # Solve for Beta coefficients: minimize ||g + sum(Beta_i * y_i)||^2
    # M[i,j] = y_i . y_j, rhs = -g . y_i
    M = s @ s.T  # (m-1, m-1) -- actually we want y . y
    M = y @ y.T
    rhs = -(G[-1] @ y.T)  # (m-1,)
    
    try:
        Beta = np.linalg.solve(M + 1e-10 * np.eye(len(M)), rhs)
    except:
        return x
    
    # x_new = x - sum(Beta_i * s_i)  ... wait, Anderson acceleration formula
    # x_{k+1} = x_k - sum_i Beta_i * (x_k - x_{k-i})  ... 
    # Actually: x_{k+1} = x_k + sum_i alpha_i * (x_{k-i} - x_k)  where alpha minimizes ||g_k + sum alpha_i (g_k - g_{k-i})||
    # Let me use the standard form:
    # x_{k+1} = (1 - sum alpha) * x_k + sum alpha_i * x_{k-i}
    # where alpha minimizes ||g_k - sum alpha_i * g_{k-i} + (1-sum alpha) * ... ||
    
    # Simpler: x_new = x + sum_i Beta_i * s_i  where s_i = x_{m-i} - x_{m-i-1}
    x_new = X[-1] + Beta @ s  # (n,)
    return x_new

def subgradient_method(a0, time_limit, top_k=50, beta=10.0, eta0=1.0, verbose=True):
    """
    Projected subgradient method with Anderson acceleration.
    Minimize max_k c[k] on the simplex.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    history_X = []
    history_G = []
    max_history = 8
    
    eta = eta0
    stall = 0
    max_stall = 200
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart with perturbation
            rng = np.random.default_rng(t)
            a = best_a * (1.0 + rng.normal(0, 0.01, n))
            a = project_simplex(a)
            history_X = []
            history_G = []
            eta = eta0
            stall = 0
            continue
        
        # Compute subgradient
        g = subgradient_topk(a, ac, top_k=top_k, beta=beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        g = g / gn  # normalize gradient
        
        # Step size
        step = eta / np.sqrt(t + 1)
        
        # Anderson acceleration
        if len(history_X) >= max_history:
            x_new = anderson_accelerate(a, g, history_X, history_G, m=max_history)
        else:
            x_new = a - step * g
        
        # Project to simplex
        x_new = project_simplex(x_new)
        
        # Update history
        history_X.append(a.copy())
        history_G.append(g.copy())
        if len(history_X) > max_history * 2:
            history_X = history_X[-max_history:]
            history_G = history_G[-max_history:]
        
        a = x_new
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def pair_swap_optimize(a0, time_limit, verbose=True, rng_seed=42):

            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                contrib[js] += a[js] * a[k_int - js]
        
        # Sort by contribution (high contributors are "bad")
        bad_coords = np.argsort(contrib)[::-1][:min(100, n)]
        good_coords = np.argsort(contrib)[:min(100, n)]

                break
            
            # Pick source from bad_coords, destination from good_coords
            src = bad_coords[rng.integers(len(bad_coords))]
            dst = good_coords[rng.integers(len(good_coords))]
            if src == dst or a[src] < 1e-10:
                continue
            
            n_pairs_tried += 1

                    a = a_trial
                    best_m = M_trial

                M_new = fast_autocorr(a).max()
                if M_new < best_m:

def coordinate_optimize(a0, time_limit, verbose=True):
    """
    For each coordinate, find optimal value via golden section search
    (with proportional redistribution of remaining mass).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    def score_with_a_j(a, j, t):
        """Compute max autocorr when a[j] = t and rest scaled proportionally."""
        if t < 0 or t >= 1.0:
            return float('inf')
        s_rest = 1.0 - t
        s_old_rest = 1.0 - a[j]
        if s_old_rest < 1e-15 or s_rest < 1e-15:
            return float('inf')
        scale = s_rest / s_old_rest
        a_new = a.copy() * scale
        a_new[j] = t
        # Quick check: only compute max of autocorr
        ac = fast_autocorr(a_new)
        return ac.max()
    
    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            if a[j] < 1e-12:
                continue
            
            # Golden section search for optimal a[j]
            # Current value: a[j]. Search in [0, min(1, a[j]*3)]
            lo = 0.0
            hi = min(1.0, a[j] * 3.0 + 0.01)
            
            if hi - lo < 1e-10:
                continue
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            f1 = score_with_a_j(a, j, c1)
            f2 = score_with_a_j(a, j, c2)
            f_curr = score_with_a_j(a, j, a[j])
            
            for _ in range(15):
                if hi - lo < 1e-8:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = score_with_a_j(a, j, c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = score_with_a_j(a, j, c2)
            
            t_opt = (lo + hi) / 2.0
            f_opt = score_with_a_j(a, j, t_opt)
            
            if f_opt < best_m - 1e-12:
                # Apply the update
                s_rest = 1.0 - t_opt
                s_old_rest = 1.0 - a[j]
                if s_old_rest > 1e-15 and s_rest > 1e-15:
                    scale = s_rest / s_old_rest
                    a = a * scale
                    a[j] = t_opt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
        
        if not improved and n_iter > 3:
            break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordOpt ({elapsed:.1f}s): J={J:.6f}, iters={n_iter}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def diverse_init(n, seed=0):
    """Create diverse random initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine
    x = np.linspace(0, 1, n)
    inits.append(np.sin(np.pi * x)**2 / np.sin(np.pi * x)**2.sum())
    
    # Cosine bump
    inits.append(np.cos(np.pi * (x - 0.5))**2 / (np.cos(np.pi * (x - 0.5))**2).sum())
    
    # Random
    a = rng.dirichlet(np.ones(n))
    inits.append(a)
    
    # Bimodal
    a = np.exp(-10*(x-0.25)**2) + np.exp(-10*(x-0.75)**2)
    inits.append(a / a.sum())
    
    # Triangular
    a = 1 - 2*np.abs(x - 0.5)
    inits.append(a / a.sum())
    
    # Random sparse-ish
    a = rng.random(n) ** 3
    inits.append(a / a.sum())
    
    return inits

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, n_iter=50, damp=0.3):
    """Flatten power spectrum via iterative amplitude equalization."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        target = amp.mean()
        new_amp = damp * amp + (1.0 - damp) * target
        phase = A / (amp + 1e-30)
        A_new = new_amp * phase
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def spectral_flatten_geomean(a, n_iter=50):
    """Flatten using geometric mean of amplitude and its mean."""
    a = normalize(a).copy()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        amp = np.abs(A)
        nonzero = amp > 1e-12
        if nonzero.sum() == 0:
            break
        target = np.mean(amp[nonzero])
        new_amp = np.sqrt(amp * target)
        phase = A / (amp + 1e-30)
        A_new = new_amp * phase
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def compute_grad_full(a, w):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1 using correlate."""
    cc = np.correlate(w, a, mode='full')
    return 2.0 * cc[:len(a)]


def full_spectrum_subgradient(a, ac):
    """
    Compute subgradient using all peaks above the average (flat target).
    Returns (gradient, max_val, f_star).
    """
    n = len(a)
    f_star = 1.0 / (2 * n - 1)
    excess = np.maximum(ac - f_star, 0.0)
    w_sum = excess.sum()
    if w_sum < 1e-18:
        return np.zeros(n), ac.max(), f_star
    w = excess / w_sum
    grad = compute_grad_full(a, w)
    return grad, ac.max(), f_star


def subgradient_polyak_anderson(a0, time_limit, verbose=True, max_hist=6, restart_period=500):
    """
    Subgradient method with Polyak step, Anderson acceleration, and periodic restarts.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    f_star = 1.0 / (2 * n - 1)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    hist_x = []
    hist_g = []
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > restart_period:
            a = best_a * (1.0 + rng.normal(0, 0.02, n))
            a = project_simplex(a)
            hist_x = []
            hist_g = []
            stall = 0
            continue
        
        # Compute full-spectrum subgradient
        grad, M_curr, _ = full_spectrum_subgradient(a, ac)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        g = grad / gn
        
        # Polyak step size
        gap = M_curr - f_star
        if gap < 1e-14:
            break
        eta = gap / (gn * gn)
        eta = min(eta, 2.0)
        
        # Anderson acceleration
        if len(hist_x) >= max_hist:
            X = np.array(hist_x[-max_hist:])
            G_arr = np.array(hist_g[-max_hist:])
            s_vecs = np.diff(X, axis=0)
            y_vecs = np.diff(G_arr, axis=0)
            
            M_mat = y_vecs @ y_vecs.T
            rhs = -(G_arr[-1] @ y_vecs.T)
            try:
                alpha = np.linalg.solve(M_mat + 1e-12 * np.eye(len(M_mat)), rhs)
                # Limit alpha to prevent divergence
                alpha = np.clip(alpha, -1.0, 1.0)
                x_new = a + alpha @ s_vecs
            except np.linalg.LinAlgError:
                x_new = a - eta * g
        else:
            x_new = a - eta * g
        
        # Project to simplex
        x_new = project_simplex(x_new)
        
        # Update history
        hist_x.append(a.copy())
        hist_g.append(g.copy())
        if len(hist_x) > max_hist + 2:
            hist_x = hist_x[-(max_hist + 2):]
            hist_g = hist_g[-(max_hist + 2):]
        
        a = x_new
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PG+AA t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.10f}")
    
    return best_a, best_m


def subgradient_topk(a0, time_limit, top_k=40, beta=10.0, verbose=True):
    """
    Subgradient method using top-K peaks with softmax weighting.
    Simpler but sometimes faster to converge near the optimum.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 800:
            a = best_a * (1.0 + rng.normal(0, 0.01, n))
            a = project_simplex(a)
            stall = 0
            continue
        
        # Top-K subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[top_idx]
        w = np.exp(beta * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - (n - 1))
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                grad[js] += 2.0 * wi * a[k_int - js]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        g = grad / gn
        
        # Step size: decreasing
        eta = 1.0 / np.sqrt(t + 1.0)
        x_new = a - eta * g
        x_new = project_simplex(x_new)
        a = x_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PG-topK t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, rng_seed=42):
    """
    Targeted peak shaving: identify the worst peak, find the coordinate
    contributing most to it, and move mass away.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 200:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))
        
        # Find coordinates contributing to this peak
        j_start = max(0, peak_lag - (n - 1))
        j_end = min(n - 1, peak_lag)
        
        if j_start > j_end:
            stall += 1
            continue
        
        js = np.arange(j_start, j_end + 1)
        contrib = a[js] * a[peak_lag - js]
        
        # Source: highest contributor
        src_local = np.argmax(contrib)
        src = js[src_local]
        
        if a[src] < 1e-10:
            stall += 1
            continue
        
        # Find best destination: minimize impact on top peaks
        top_lags = np.argsort(ac)[::-1][:min(8, len(ac))]
        
        # Candidate destinations: coordinates NOT in the peak's support
        in_peak = np.zeros(n, dtype=bool)
        for kl in top_lags[:5]:
            kl_i = int(kl)
            s2 = max(0, kl_i - (n - 1))
            e2 = min(n - 1, kl_i)
            in_peak[s2:e2 + 1] = True
        
        candidates = np.where(~in_peak)[0]
        if len(candidates) < 20:
            candidates = np.arange(n)
        
        # Sample a subset
        n_sample = min(300, len(candidates))
        chosen = candidates[rng.choice(len(candidates), n_sample, replace=False)]
        
        best_dst = -1
        best_score_delta = float('inf')
        
        # Sensitivity: how much would adding to dst increase top peaks?
        for dst in chosen:
            if dst == src:
                continue
            sens = 0.0
            for kl in top_lags[:4]:
                kl_i = int(kl)
                d1 = kl_i - int(dst)
                d2 = int(dst) - kl_i
                add = 0.0
                if 0 <= d1 < n:
                    add += a[d1]
                if 0 <= d2 < n and d2 != d1:
                    add += a[d2]
                sens += add
            if sens < best_score_delta:
                best_score_delta = sens
                best_dst = int(dst)
        
        if best_dst < 0:
            stall += 1
            continue
        
        # Try several transfer amounts
        improved = False
        for frac in [0.02, 0.05, 0.08, 0.12, 0.15, 0.2, 0.25, 0.35]:
            amt = a[src] * frac
            if amt < 1e-14:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.05, 0.15)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_init(n, seed=0):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine squared
    s = np.sin(np.pi * x) ** 2
    inits.append(s / s.sum())
    
    # Raised cosine
    r = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    r = np.maximum(r, 0)
    inits.append(r / r.sum())
    
    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 5.0))
    
    # Bimodal
    b = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    inits.append(b / b.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    inits.append(tri / tri.sum())
    
    # Power-law-ish
    p = (x + 0.01) ** 2
    inits.append(p / p.sum())
    p2 = (1 - x + 0.01) ** 2
    inits.append(p2 / p2.sum())
    
    # Random sparse
    a = rng.random(n) ** 5
    inits.append(a / a.sum())
    
    return inits
