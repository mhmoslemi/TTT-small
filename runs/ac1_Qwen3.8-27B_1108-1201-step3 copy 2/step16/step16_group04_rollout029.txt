Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative spectral equalization (DFT power flattening)
2. Smoothed gradient descent with log-sum-exp + Adam on softmax parameterization
3. Adaptive beta curriculum (smoothness increases over time)
4. Multi-length exploration via interpolation
5. Peak-targeted coordinate polish with exact evaluation

import numpy as np
import time


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fast_conv(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def logsumexp(x, beta=1.0):
    """Compute (1/beta)*log(sum(exp(beta*x))) stably."""
    bx = beta * x
    mx = bx.max()
    if mx > 500:
        return x.max() + np.log(x.size) / beta
    return (mx + np.log(np.sum(np.exp(bx - mx)))) / beta


def compute_gradient_fft(a, p):
    """
    Compute dJ_smooth/da where J_smooth = 2n * (1/beta)*log(sum(exp(beta*c)))
    and c = conv(a,a). Returns gradient w.r.t. a (length n).
    p = softmax(beta * c), length 2n-1.
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    t1 = np.fft.irfft(np.conj(P) * A, N)[:n]
    t2 = np.fft.irfft(P * A, N)[:n]
    return 2.0 * n * (t1 + t2)


def softmax(x):
    x = x - x.max()
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return ex / s


def spectral_equalize(a, n_iter=40, blend=0.5):
    """Iteratively flatten the power spectrum."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.rfft(a, 2 * n - 1 if False else 1 << (2 * n - 2).bit_length())
        P = np.abs(A) ** 2
        eps = 1e-12
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.irfft(A_new, len(A)))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
    return normalize(a)

    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def optimize_smoothed(a_init, time_budget, verbose=True, seed=42):
    """
    Main optimization: smoothed gradient descent with Adam on softmax params.
    a_init: normalized non-negative sequence (sum=1)
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a_init = normalize(a_init)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize w from a_init
    w = np.log(a_init + 1e-30)
    w_mean = w - w.mean()  # re-center

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_conv(a_init).max()
    best_ls = logsumexp(fast_conv(a_init), 10.0) * 2.0 * n

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta_ls = 5.0
    lr = 0.005
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    stall = 0
    max_stall = 2000
    last_best_t = 0

    while time.monotonic() - t0 < time_budget:
        t += 1
        a = softmax(w)

        # Adaptive beta schedule
        beta_ls = min(500.0, 5.0 * (1 + t / 500.0) ** 0.5)

        # Compute convolution and gradient
        c = fast_conv(a)
        L = logsumexp(c, beta_ls)
        J_smooth = 2.0 * n * L

        p = softmax(beta_ls * c)
        g_a = compute_gradient_fft(a, p)

        # Convert to w-gradient (softmax Jacobian)
        g_w = a * (g_a - np.dot(a, g_a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_w
        v = beta2 * v + (1.0 - beta2) * g_w ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_t = lr / (1.0 + 0.0001 * t)
        w = w - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Re-center w periodically
        if t % 500 == 0:
            w = w - w.mean()

        # Track best using exact objective periodically
        if t % 50 == 0 or t <= 10:
            J_exact = 2.0 * n * c.max()
            if J_exact < best_score:
                best_score = J_exact
                best_a = a.copy()
                stall = 0
                last_best_t = t
            else:
                stall += 50

        # Check time
        if time.monotonic() - t0 > time_budget:
            break

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb the best solution
            pert = rng.normal(0, 0.02, n)
            a_pert = best_a * (1.0 + pert)
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / a_pert.sum()
            w = np.log(a_pert)
            w = w - w.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            beta_ls = 5.0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Opt t={t} ({elapsed:.1f}s): best={best_score:.6f}, beta={beta_ls:.1f}")

    return best_a, best_score


def coordinate_polish(a_init, time_budget, verbose=True, seed=42):
    """Peak-targeted coordinate descent polish."""
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_budget and stall < 200:
        c = fast_conv(a)
        M = c.max()
        k_star = c.argmax()

        # Find the pair contributing most to c[k_star]
        k = int(k_star)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k - idx]
        src_offset = np.argmax(contrib)
        src = idx[src_offset]
        dst_donor = k - src  # the other index in the contributing pair

        # Try moving mass from src to a random "quiet" position
        in_peak = np.zeros(n, dtype=bool)
        top_lags = np.argsort(c)[::-1][:10]
        for lk in top_lags:
            lki = int(lk)
            s2 = max(0, lki - n + 1)
            e2 = min(n - 1, lki)
            in_peak[s2:e2 + 1] = True

        candidates = np.where(~in_peak)[0]
        if len(candidates) < 10:
            candidates = np.arange(n)

        improved = False
        n_try = min(200, len(candidates))
        chosen = candidates[rng.permutation(len(candidates))[:n_try]]

        for dst in chosen:
            if dst == src:
                continue
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                c_trial = fast_conv(a_trial)
                M_trial = c_trial.max()
                if M_trial < best_score / (2.0 * n) - 1e-16:
                    a = a_trial
                    best_score = 2.0 * n * M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Small random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if 2.0 * n * M_new < best_score:
                        best_score = 2.0 * n * M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / max(a.sum(), 1e-15)

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Coord ({elapsed:.1f}s): score={best_score:.6f}, stall={stall}")

    return best_a, best_score


def propose_candidate():
    """Main search orchestrator."""
    t_global = time.monotonic()
    BUDGET = 1740.0  # leave 60s margin

    print(f"Starting optimization, budget={BUDGET:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Spectral equalization
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    print(f"\n=== Phase 1: Spectral Equalization (budget={remaining:.0f}s) ===")
    a_spec = spectral_equalize(best_seq, n_iter=50, blend=0.6)
    try_update(a_spec, "spectral")

    # Phase 2: Smoothed gradient optimization at current length
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n0 = len(best_seq)
    t_phase2 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 2: Smoothed Grad Opt n={n0} (budget={t_phase2:.0f}s) ===")
    if t_phase2 > 30:
        a_out, _ = optimize_smoothed(best_seq, t_phase2, verbose=True, seed=42)
        try_update(a_out, f"smooth_n{n0}")

    # Phase 3: Try longer length
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_long = n0 + 800
    print(f"\n=== Phase 3: Longer length n={n_long} (budget={remaining:.0f}s) ===")
    if remaining > 200:
        t_phase3 = min(remaining * 0.30, 450.0)
        a_long = interpolate_sequence(best_seq, n_long)
        a_long = spectral_equalize(a_long, n_iter=30, blend=0.5)
        if t_phase3 > 30:
            a_out, _ = optimize_smoothed(a_long, t_phase3, verbose=True, seed=123)
            try_update(a_out, f"smooth_n{n_long}")

    # Phase 4: Try even longer
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_longer = n0 + 1600
    print(f"\n=== Phase 4: Longer n={n_longer} (budget={remaining:.0f}s) ===")
    if remaining > 200:
        t_phase4 = min(remaining * 0.25, 400.0)
        a_longer = interpolate_sequence(best_seq, n_longer)
        a_longer = spectral_equalize(a_longer, n_iter=25, blend=0.5)
        if t_phase4 > 30:
            a_out, _ = optimize_smoothed(a_longer, t_phase4, verbose=True, seed=456)
            try_update(a_out, f"smooth_n{n_longer}")

    # Phase 5: Re-optimize best at its length with different seed
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_best = len(best_seq)
    if remaining > 150:
        t_phase5 = min(remaining * 0.25, 350.0)
        print(f"\n=== Phase 5: Re-opt n={n_best} seed=789 (budget={t_phase5:.0f}s) ===")
        if t_phase5 > 30:
            a_out, _ = optimize_smoothed(best_seq, t_phase5, verbose=True, seed=789)
            try_update(a_out, f"smooth2_n{n_best}")

    # Phase 6: Coordinate polish
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_best = len(best_seq)
    if remaining > 60:
        t_phase6 = min(remaining * 0.20, 200.0)
        print(f"\n=== Phase 6: Coord Polish n={n_best} (budget={t_phase6:.0f}s) ===")
        if t_phase6 > 20:
            a_out, _ = coordinate_polish(best_seq, t_phase6, verbose=True, seed=42)
            try_update(a_out, f"coord_n{n_best}")

    # Phase 7: Another round of smoothed opt if time remains
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_best = len(best_seq)
    if remaining > 100:
        t_phase7 = min(remaining * 0.30, 250.0)
        print(f"\n=== Phase 7: Final Smooth Opt n={n_best} (budget={t_phase7:.0f}s) ===")
        if t_phase7 > 30:
            a_out, _ = optimize_smoothed(best_seq, t_phase7, verbose=True, seed=321)
            try_update(a_out, f"smooth3_n{n_best}")

    # Phase 8: Final coordinate polish
    elapsed = time.monotonic() - t_global
    remaining = BUDGET - elapsed
    n_best = len(best_seq)
    if remaining > 40:
        t_phase8 = min(remaining * 0.50, 150.0)
        print(f"\n=== Phase 8: Final Coord n={n_best} (budget={t_phase8:.0f}s) ===")
        if t_phase8 > 15:
            a_out, _ = coordinate_polish(best_seq, t_phase8, verbose=True, seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
