    n = len(a)

def spectral_equalize(a, n_iter=30, eps=1e-10, blend=0.3):
    """Flatten power spectrum via iterative DFT correction with non-negativity."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def solve_lp_direction(y, M, constraint_lags, max_iter_lp=2000):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in constraint_lags, b >= 0.
    Returns normalized direction or None.
    """
    n = len(y)
    m = len(constraint_lags)
    if m == 0:
        return np.ones(n) / n

    # Build constraint matrix A (m x n): A[i, j] = y[k_i - j] if valid
    A_mat = np.zeros((m, n), dtype=np.float64)
    for i, k in enumerate(constraint_lags):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        if js <= je:
            j_range = np.arange(js, je + 1)
            A_mat[i, js:je + 1] = y[k_int - j_range]

    # Solve: minimize -sum(b) subject to A_mat * b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'maxiter': max_iter_lp, 'presolve': True})

    if res.status != 0:
        return None

    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    constraint_lags = list(np.argsort(fast_autocorr(a))[::-1][:initial_k])
    x = a.copy()
    x_prev = a.copy()
    current_k = initial_k

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation (optional)
        if nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Refresh constraint set periodically
        if t % refresh_every == 0:
            top_peaks = np.argsort(ac)[::-1][:current_k]
            # Merge with existing (keep union, capped at max_k)
            existing = set(constraint_lags)
            for p in top_peaks:
                existing.add(int(p))
            # If too many, keep top max_k by ac value
            if len(existing) > max_k:
                sorted_lags = sorted(existing, key=lambda k: -ac[k])
                constraint_lags = sorted_lags[:max_k]
            else:
                constraint_lags = list(existing)

        # Solve LP
        direction = solve_lp_direction(y, M_curr, constraint_lags)
        if direction is None:
            # Try with relaxed M
            direction = solve_lp_direction(y, M_curr * 1.05, constraint_lags)
            if direction is None:
                continue

        # Check and add violated constraints
        violations = check_violations(y, direction, M_curr, constraint_lags, n)
        for v in violations:
            if len(constraint_lags) < max_k:
                constraint_lags.append(v)

        # Step size
        if line_search:
            # Try multiple step sizes
            best_t = None
            best_val = float('inf')
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                x_trial = (1.0 - alpha) * y + alpha * direction
                x_trial = np.maximum(x_trial, 0.0)
                st = x_trial.sum()
                if st < 1e-15:
                    continue
                x_trial = x_trial / st
                val = fast_autocorr(x_trial).max()
                if val < best_val:
                    best_val = val
                    best_t = x_trial.copy()
            if best_t is None:
                continue
            if best_val < M_curr:
                x_new = best_t
            else:
                # Use standard FW step
                alpha_fw = 2.0 / (t + 2.0)
                x_new = (1.0 - alpha_fw) * y + alpha_fw * direction
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y.copy()
                else:
                    x_new = x_new / sx
        else:
            alpha_fw = 2.0 / (t + 2.0)
            x_new = (1.0 - alpha_fw) * y + alpha_fw * direction
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new / sx

        x_prev = y
        x = x_new

        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Gradually increase constraint count
        if t % 30 == 0 and current_k < max_k:
            current_k = min(current_k + 10, max_k)

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, |C|={len(constraint_lags)}")

    return best_a, best_m


def peak_shave(a0, time_limit, verbose=True, rng_seed=42):
    """
    Greedy peak shaving: find dominant peak contributors and redistribute mass
    to locations with minimal sensitivity.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top peaks
        top_k = np.argsort(ac)[::-1][:15]

        improved_this_round = False
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            # Contribution of each pair
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the dominant source index
            src = idx[np.argmax(contrib)]

            # Compute sensitivity map: for each potential destination,
            # how much would it increase the top peaks?
            sens = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                # If we add mass at position d, the contribution to peak k2 is:
                # a[k2-d] (if valid) + a[d-(k2-n+1)] (if valid, from the other side)
                # More precisely: c[k2] gets a[d]*a[k2-d] added (plus symmetric)
                for d, delta in [(k2i - np.arange(n), a[np.arange(n)])]:
                    pass
                # Simplified: sensitivity at d for peak k2 is a[k2-d] if 0<=k2-d<n
                d_range = np.arange(n)
                val = k2i - d_range
                mask = (val >= 0) & (val < n)
                sens[mask] += a[val[mask]]
                # Also the "other side" contribution
                val2 = d_range - k2i
                mask2 = (val2 >= 0) & (val2 < n)
                # This would be a contribution from a[d2] where d2 = d + k2... 
                # Actually for convolution c[k2] = sum_j a[j]*a[k2-j], 
                # adding delta at position d adds delta*a[k2-d] to c[k2]
                # (only one term, not two, since we're adding to one position)
                # Wait, no: if we change a[d] by delta, then c[k] changes by:
                #   delta * a[k-d] (from a[d]*a[k-d]) + a[k-d]*delta (if k-d = d' != d)
                # Actually c[k] = sum_j a[j]*a[k-j]. Changing a[d] by delta:
                #   Delta c[k] = delta * a[k-d] + a[k-d] * delta = 2*delta*a[k-d]
                #   (if k-d is in [0,n-1])
                #   Plus: if d is the second index, i.e., k-j = d => j = k-d, same thing.
                # So total: Delta c[k] = 2*delta*a[k-d] if 0 <= k-d < n, else 0.
                # But we already counted this above. Let me redo.
            
            # Recompute sens properly
            sens = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                d_range = np.arange(n)
                val = k2i - d_range
                mask = (val >= 0) & (val < n)
                sens[mask] += 2.0 * a[val[mask]]

            # Find quiet destinations (low sensitivity, not the source)
            sens[src] = np.inf  # Can't move to itself
            excluded = set(int(x) for x in top_k[:5])
            for ex in excluded:
                if 0 <= ex < n:
                    sens[ex] = max(sens[ex], sens.max() * 0.5)

            # Pick top few quiet destinations
            quiet = np.argsort(sens)[:50]
            quiet = [q for q in quiet if q != src and sens[q] < np.inf]
            if not quiet:
                continue

            # Try moving small amounts to the quietest destinations
            best_move = None
            best_score = best_m
            for dst in quiet[:20]:
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_score:
                        best_score = M_trial
                        best_move = a_trial.copy()

            if best_move is not None:
                a = best_move
                best_m = best_score
                best_a = a.copy()
                improved_this_round = True
                stall = 0

        if not improved_this_round:
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m + 0.001:  # Allow slight increase for exploration
                        best_m = min(best_m, M_new)
                        if M_new < best_m:
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def nesterov_mirror_descent(a0, time_limit, top_k=60, beta=4.0, c0=4.0, verbose=True):
    """Nesterov-accelerated mirror descent (exponential family) for peak minimization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft gradient
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = y
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 5000:
            stall_count = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      NMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent for final polish."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_k = np.argsort(ac)[::-1][:10]
        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Sensitivity
            sens = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                d_range = np.arange(n)
                val = k2i - d_range
                mask = (val >= 0) & (val < n)
                sens[mask] += 2.0 * a[val[mask]]

            sens[src] = np.inf
            quiet = np.argsort(sens)[:100]
            quiet = [q for q in quiet if q != src and np.isfinite(sens[q])]

            if not quiet:
                continue

            found = False
            for dst in quiet[:15]:
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Coord ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        # Use an asymmetric initialization (exponential decay)
        decay = np.exp(-3.0 * x)
        best_seq = decay / decay.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init (expo): n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: Spectral equalization ===
    print(f"\n=== Phase 0: Spectral Equalization ===")
    a_spec = spectral_equalize(best_seq, n_iter=40, eps=1e-12, blend=0.25)
    try_update(a_spec, "spectral")

    # Also try a different spectral init
    a_spec2 = spectral_equalize(best_seq, n_iter=60, eps=1e-8, blend=0.5)
    try_update(a_spec2, "spectral2")

    # === Phase 1: Main LP FW (long run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-FW n={n_best}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_search(a_in, t1, verbose=True, initial_k=60, max_k=250,
                                line_search=True, refresh_every=12, nesterov=True)
        try_update(a_out, f"lpfw1_n{n_best}")

    # === Phase 2: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak Shave n={n_best}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = peak_shave(a_in, t2, verbose=True, rng_seed=42)
        try_update(a_out, f"shave_n{n_best}")

    # === Phase 3: LP FW round 2 (more constraints, no nesterov) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW r2 n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_search(a_in, t3, verbose=True, initial_k=80, max_k=350,
                                line_search=True, refresh_every=10, nesterov=False)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 4: Nesterov mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: NMD n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 30:
        a_in = normalize(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t4, top_k=80, beta=3.0, c0=5.0, verbose=True)
        try_update(a_out, f"nmd_n{n_best}")

    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_alt = n_best + 600 if n_best <= 2500 else n_best - 600
    n_alt = max(1200, min(n_alt, 4000))
    if n_alt != n_best and remaining > 200:
        print(f"\n=== Phase 5: Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = spectral_equalize(a_long, n_iter=20, blend=0.3)

        t5a = min(remaining * 0.5, 300.0)
        if t5a > 60:
            a_in = normalize(a_long)
            a_out, _ = lp_fw_search(a_in, t5a, verbose=True, initial_k=60, max_k=200,
                                    line_search=True, refresh_every=15, nesterov=True)
            try_update(a_out, f"lpfw_n{n_alt}")

    # === Phase 6: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord n={n_best}, budget={remaining:.0f}s ===")
    t6 = min(remaining * 0.25, 120.0)
    if t6 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 7: Final LP FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 7: Final LP-FW n={n_best}, budget={remaining:.0f}s ===")
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 40:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw_search(a_in, t7, verbose=True, initial_k=100, max_k=400,
                                    line_search=True, refresh_every=8, nesterov=False)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 8: Very final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 15:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
