    p = np.exp(z)

    # Objective
    obj = (z.max() + np.log(p.sum())) / beta  # = logsumexp(beta*c)/beta
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (with a[m]=0 outside [0,n-1])

    # S2[j] = sum_k p[k]*a[j-k] = convolution of p and a at index j

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on log-sum-exp smooth
approximation with beta-continuation, multiple initializations and lengths,
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import softmax

_global_t0 = 0.0
_GLOBAL_TIME_BUDGET = 1700.0

def time_remaining():
    return _GLOBAL_TIME_BUDGET - (time.monotonic() - _global_t0)

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def smooth_obj_and_grad(z, beta):
    """Compute smooth objective and gradient w.r.t. z (log-softmax parameter)."""
    a = softmax(z)
    n = len(a)
    c = fast_conv(a)
    zc = beta * c
    zc_max = zc.max()
    zc_shifted = zc - zc_max
    exp_zc = np.exp(zc_shifted)
    sum_exp = exp_zc.sum()
    log_sum_exp = zc_max + np.log(sum_exp)
    obj = log_sum_exp / beta
    p = exp_zc / sum_exp

    # FFT-based gradient computation
    size = 1
    while size < 3 * n:
        size *= 2
    A_fft = np.fft.rfft(a, size)
    P_fft = np.fft.rfft(p, size)
    conv_pa = np.fft.irfft(P_fft * A_fft, size)[:n]
    corr_pa = np.fft.irfft(np.conj(A_fft) * P_fft, size)[:n]
    grad_a = 2.0 * (corr_pa + conv_pa)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return obj, grad_z

def run_lbfgs(a_init, beta_schedule, time_limit, max_iter_per_beta=500):
    """Run L-BFGS with beta continuation. Returns best a found."""
    global _global_t0
    n = len(a_init)
    z = np.log(a_init + 1e-300)
    z = z - z.mean()  # center for numerical stability
    best_a = a_init.copy()
    best_true_obj = 2.0 * n * fast_conv(a_init).max()

    for beta in beta_schedule:
        if time_remaining() < 30:
            break
        t_start = time.monotonic()
        beta_limit = min(time_limit - (time.monotonic() - _global_t0), 300.0)

        def func_grad(z_in):
            return smooth_obj_and_grad(z_in, beta)

        try:
            res = minimize(
                lambda z: smooth_obj_and_grad(z, beta)[0],
                z,
                jac=lambda z: smooth_obj_and_grad(z, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_beta, 'gtol': 1e-10,
                         'ftol': 1e-14, 'maxfun': max_iter_per_beta * 3}
            )
            z = np.asarray(res.x, dtype=np.float64)
        except Exception:
            pass

        a_new = softmax(z)
        true_obj = 2.0 * n * fast_conv(a_new).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_new.copy()
        z = np.log(a_new + 1e-300)
        z = z - z.mean()

    return best_a, best_true_obj

            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = np.array([a[i] * a[k_int - i] if 0 <= k_int - i < n else 0.0 for i in idx])
            total = contrib.sum()
            if total < 1e-30:
                continue

            # Find hot and cold coordinates
            hot = idx[np.argmax(contrib)]
            # Cold: coordinates not in any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            cold_candidates = np.where(~in_peak)[0]
            if len(cold_candidates) < 10:
                cold_candidates = np.where(a < a.mean())[0]
            if len(cold_candidates) == 0:
                cold_candidates = np.arange(n)

            # Try transferring from hot to random cold
            n_try = min(len(cold_candidates), 50)
            chosen = cold_candidates[rng.permutation(len(cold_candidates))[:n_try]]

            for dst in chosen:
                if dst == hot or a[hot] < 1e-15:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    amt = a[hot] * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[hot] -= amt
                    a_trial[dst] += amt
                    if a_trial[hot] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_obj = 2.0 * n * best_m
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

        if not improved:

            if stall > 50:

def subgradient_descent(a_init, time_limit, c0=3.0, seed=0):

    max_stall = 2000

    x_prev = a.copy()
    t = 0

        t += 1
        if stall > max_stall:
            x = best_a + rng.normal(0, 0.003, n) * best_a
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
            max_stall = int(max_stall * 0.8)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_conv(y)
        k_star = int(np.argmax(ac))

        # Hard subgradient
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        for i in range(i_start, i_end):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * y[j]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y) - eta * grad
        log_y = log_y - log_y.max()
        a_new = np.exp(log_y)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

    return best_a, 2.0 * n * best_m

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss

        try_update(a_out, f"lbfgs2_n{n0}")

    # === Phase 3: L-BFGS with high-beta schedule ===
    print(f"\n=== Phase 3: L-BFGS parent n={n0} (high beta) ===")
    if time_remaining() > 60:
        a_in = best_seq / best_seq.sum()
        a_out, _ = run_lbfgs(a_in, beta_schedules[2], time_limit=time_remaining() * 0.3,
                             max_iter_per_beta=300)
        try_update(a_out, f"lbfgs3_n{n0}")

    # === Phase 4: Subgradient descent (different dynamics) ===
    print(f"\n=== Phase 4: Subgradient n={n0} ===")
    if time_remaining() > 60:
        a_in = best_seq / best_seq.sum()
        a_out, _ = subgradient_descent(a_in, time_limit=min(time_remaining() * 0.25, 250.0),
                                       c0=4.0, seed=42)
        try_update(a_out, f"sgd_n{n0}")

    # === Phase 5: L-BFGS after subgradient ===
    print(f"\n=== Phase 5: L-BFGS after SGD n={n0} ===")
    if time_remaining() > 60:
        a_in = best_seq / best_seq.sum()
        a_out, _ = run_lbfgs(a_in, [50, 100, 300, 700, 1500, 3000, 6000],
                             time_limit=min(time_remaining() * 0.25, 200.0),
                             max_iter_per_beta=300)
        try_update(a_out, f"lbfgs5_n{n0}")

    # === Phase 6: Different initializations ===
    print(f"\n=== Phase 6: Multiple initializations ===")
    init_names = []
    init_list = []

    # Gaussian bump
    n_g = n0
    xg = np.linspace(0, 1, n_g)
    gauss = np.exp(-20.0 * (xg - 0.5) ** 2)
    init_list.append(gauss / gauss.sum())
    init_names.append("gauss")

    # Cosine bump
    cos_bump = np.cos(np.pi * (xg - 0.5)) ** 2
    cos_bump = np.maximum(cos_bump, 0.001)
    init_list.append(cos_bump / cos_bump.sum())
    init_names.append("cosine")

    # Triangular
    tri = 1.0 - 2.0 * np.abs(xg - 0.5)
    init_list.append(tri / tri.sum())
    init_names.append("tri")

    # Two-bump (asymmetric)
    two_bump = np.exp(-30.0 * (xg - 0.3) ** 2) + 0.7 * np.exp(-30.0 * (xg - 0.7) ** 2)
    init_list.append(two_bump / two_bump.sum())
    init_names.append("twobump")

    # Flat top
    flat_top = np.where(np.abs(xg - 0.5) < 0.3, 1.0, 0.1)
    init_list.append(flat_top / flat_top.sum())
    init_names.append("flattop")

    for idx, (a_init, name) in enumerate(zip(init_list, init_names)):
        if time_remaining() < 60:
            break
        t_alloc = min(time_remaining() * 0.2, 150.0)
        if t_alloc < 30:
            break
        print(f"  Init: {name}")
        a_out, _ = run_lbfgs(a_init, beta_schedules[0], time_limit=t_alloc,
                             max_iter_per_beta=250)
        try_update(a_out, f"init_{name}_n{n0}")

    # === Phase 7: Different lengths ===
    print(f"\n=== Phase 7: Length exploration ===")
    n_best = len(best_seq)
    lengths_to_try = []
    for delta in [-400, -200, 200, 400, 600]:
        nt = n_best + delta
        if 800 <= nt <= 6000:
            lengths_to_try.append(nt)

    for nt in lengths_to_try:
        if time_remaining() < 80:
            break
        a_interp = interpolate_to_length(best_seq, nt)
        t_alloc = min(time_remaining() * 0.25, 200.0)
        if t_alloc < 30:
            break
        print(f"  Length n={nt}")
        a_out, _ = run_lbfgs(a_interp, beta_schedules[0], time_limit=t_alloc,
                             max_iter_per_beta=300)
        try_update(a_out, f"lbfgs_n{nt}")

    # === Phase 8: Coordinate descent polishing ===
    print(f"\n=== Phase 8: Coordinate polish n={len(best_seq)} ===")
    if time_remaining() > 30:
        a_in = best_seq / best_seq.sum()
        t_alloc = min(time_remaining() * 0.3, 250.0)
        a_out, _ = coordinate_polish(a_in, time_limit=t_alloc, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 9: More L-BFGS with very high beta ===
    print(f"\n=== Phase 9: High-beta L-BFGS n={len(best_seq)} ===")
    if time_remaining() > 40:
        a_in = best_seq / best_seq.sum()
        high_beta = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
        t_alloc = min(time_remaining() * 0.3, 200.0)
        a_out, _ = run_lbfgs(a_in, high_beta, time_limit=t_alloc, max_iter_per_beta=400)
        try_update(a_out, f"lbfgs_highbeta_n{len(best_seq)}")

    # === Phase 10: Final coordinate polish ===
    print(f"\n=== Phase 10: Final coord polish n={len(best_seq)} ===")
    if time_remaining() > 20:
        a_in = best_seq / best_seq.sum()
        t_alloc = min(time_remaining() * 0.4, 150.0)
        a_out, _ = coordinate_polish(a_in, time_limit=t_alloc, seed=314)
        try_update(a_out, f"coord_final_n{len(best_seq)}")

    # === Phase 11: One more L-BFGS pass ===
    print(f"\n=== Phase 11: Final L-BFGS n={len(best_seq)} ===")
    if time_remaining() > 30:
        a_in = best_seq / best_seq.sum()
        t_alloc = min(time_remaining() * 0.5, 150.0)
        a_out, _ = run_lbfgs(a_in, [100, 300, 700, 1500, 3500, 7000, 15000, 30000],
                             time_limit=t_alloc, max_iter_per_beta=400)
        try_update(a_out, f"lbfgs_final_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-_global_t0:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
