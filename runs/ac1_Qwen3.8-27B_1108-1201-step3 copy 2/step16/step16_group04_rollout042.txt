```python
"""
Algorithm summary
=================

The task is to minimize

    J(a) = 2 * n * max_k conv(a, a)_k / (sum a)^2

over non-negative sequences.  Since the objective is scale invariant, we work
with probability vectors sum(a)=1, so the goal is to make the linear
self-convolution of a as flat as possible.

This script uses a fast non-convex optimization pipeline built around the
following idea:

  Let q = conv(p,p) and let D_i = sum_k w_k p_{k-i}.  For a choice of non-
  negative lane weights w, D is the derivative of a smoothing objective
  (L2 flattening when w=q, peak-emphasized smoothing when w is a softmax of
  q) with respect to log p.  The update

      log p_i <- log p_i + eta * (log(mean D) - log(D_i))

  pushes p toward the first-order condition D_i = const.

Phases:
  1. L2 flattening (w=q).
  2. Smooth peak-emphasized balancing.
  3. Length extrapolation and re-balancing.
  4. Sharp peak-emphasized balancing.
  5. Peak-targeted coordinate polish.
  6. Short randomized restarts.

All heavy continuous statistics are computed by FFT.  Candidate solutions are
checked with the provided evaluate_sequence function only a small number of
times, and the returned vector is the empirically best one found.
"""

import numpy as np
import time

_FLOOR = 1e-25
_EPS = 1e-12


def next_pow2(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def fast_conv(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    m = 2 * n - 1
    size = next_pow2(m)
    A = np.fft.rfft(a, size)
    out = np.fft.irfft(A * A, size)[:m]
    return out


def hankel_product(w, p):
    """
    Returns D_i = sum_j w_{i+j} p_j for i=0..n-1, where w has length 2n-1.

    This is the derivative-with-respect-to-log-p of a quadratic lane objective.
    Computed in O(n log n) via convolution with a reversed vector.
    """
    n = len(p)
    w = np.asarray(w, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([w[0] * p[0]], dtype=np.float64)
    m = w.size
    expected = 2 * n - 1
    if m < expected:
        # Defensive fallback: pad.
        tmp = np.zeros(expected, dtype=np.float64)
        tmp[:m] = w[:m]
        w = tmp
    elif m > expected:
        w = w[:expected]
    size = next_pow2(w.size + n - 1)
    W = np.fft.rfft(w, size)
    Pr = np.fft.rfft(p[::-1], size)
    conv = np.fft.irfft(W * Pr, size)
    return conv[n - 1:2 * n - 1]


def pos_normalize(a):
    """Clip to non-negative finite values and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= _FLOOR:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def softmax_log(logp):
    """Stable softmax of a vector of log-probabilities."""
    logp = np.asarray(logp, dtype=np.float64).ravel()
    if logp.size == 0:
        return np.array([1.0], dtype=np.float64)
    m = float(np.max(logp))
    vals = np.exp(logp - m)
    s = float(vals.sum())
    if not np.isfinite(s) or s <= _FLOOR:
        return np.ones(logp.size, dtype=np.float64) / logp.size
    return vals / s


def internal_score(p):
    """FFT-based proxy for evaluate_sequence on normalized non-negative input."""
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return float("inf")
    if not np.all(np.isfinite(p)):
        return float("inf")
    n = p.size
    q = fast_conv(p)
    if q.size == 0:
        return float("inf")
    mx = float(np.max(q))
    if not np.isfinite(mx):
        return float("inf")
    if mx < 0.0 and mx > -1e-12:
        mx = 0.0
    return 2.0 * n * max(0.0, mx)


def q_normalized(q):
    """Normalize a convolution to sum 1 and remove numerical negatives."""
    q = np.asarray(q, dtype=np.float64).ravel()
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(s) or s <= _FLOOR:
        return np.ones_like(q, dtype=np.float64) / max(q.size, 1)
    return q / s


def objective_value(qn, max_weight):
    """
    Smooth objective for line search.

    max_weight = 0     -> minimize sum(q^2)
    max_weight in (0,1) -> convex blend of max(q) and sum(q^2)
    max_weight = 1     -> essentially max(q), with a tiny L2 tie-breaker
    """
    qn = np.asarray(qn, dtype=np.float64)
    mx = float(np.max(qn))
    l2 = float(np.dot(qn, qn))
    mw = float(max_weight)
    if not np.isfinite(mw):
        mw = 0.5
    mw = max(0.0, min(1.0, mw))
    if mw <= 1e-12:
        return l2
    if mw >= 1.0:
        return mx + 1e-6 * l2
    return mw * mx + (1.0 - mw) * l2


def make_weights(qn, peak_weight, beta, topk=0):
    """
    Build lane weights for gradient balancing.

    peak_weight = 0: uniform-in-l2 weights, i.e. qn.
    peak_weight > 0: blend of qn and a softmax of the largest lanes.
    """
    qn = np.asarray(qn, dtype=np.float64).ravel()
    m = qn.size
    if m == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(qn)):
        qn = np.where(np.isfinite(qn), qn, 0.0)
    s = float(qn.sum())
    if not np.isfinite(s) or s <= _FLOOR:
        return np.ones(m, dtype=np.float64) / m
    qn = qn / s

    pw = float(peak_weight)
    if not np.isfinite(pw):
        pw = 0.0
    pw = max(0.0, min(1.0, pw))

    if pw <= 1e-12:
        w = qn.copy()
    else:
        tk = int(topk)
        if tk <= 0 or tk >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(qn, m - tk)[-tk:]
        mask = np.zeros(m, dtype=bool)
        mask[idx] = True

        x = qn[mask]
        if x.size == 0:
            w = qn.copy()
        else:
            xm = float(np.max(x))
            b = float(beta)
            if not np.isfinite(b):
                b = 0.0
            if b <= 0.0:
                ex = np.ones_like(x)
            else:
                ex = np.exp(b * (x - xm))
                ex = np.where(np.isfinite(ex), ex, 0.0)
                se = float(ex.sum())
                if not np.isfinite(se) or se <= _FLOOR:
                    ex = np.ones_like(x)
                else:
                    ex = ex / se

            wmask = np.zeros(m, dtype=np.float64)
            wmask[mask] = ex
            w = (1.0 - pw) * qn + pw * wmask

    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= _FLOOR:
        w = np.ones(m, dtype=np.float64) / m
    else:
        w = w / sw

    # Add a tiny uniform background to keep all lanes weakly represented.
    tiny = 1e-6
    w = (1.0 - tiny) * w + tiny * (1.0 / m)
    sw = float(w.sum())
    if sw > _FLOOR:
        w = w / sw
    return w


def balance_by_weights(
    p_init,
    limit,
    start_time,
    peak_weight,
    beta,
    topk,
    max_weight,
    sigma0,
    stall_limit,
    seed,
    label,
    momentum=0.3,
    verbose=True,
):
    """
    Log-domain inverse-gradient balancing.

    For each current/extrapolated p:
      1. Compute q=p*p and lane weights w.
      2. Compute D_i = sum_k w_k p_{k-i}.
      3. Propose log p_i += sigma*(log(mean D)-log(D_i)).
      4. Accept by a small line search in {sigma, 0.5 sigma, occasionally 2 sigma}.
    """
    if limit <= 0:
        p = pos_normalize(p_init)
        return p, internal_score(p)

    end_time = start_time + float(limit)
    n = len(p_init)

    p = pos_normalize(p_init)
    p = np.maximum(p, _FLOOR)
    p = p / p.sum()

    best_p = p.copy()
    q0 = fast_conv(p)
    q0 = np.where(np.isfinite(q0), q0, 0.0)
    q0 = np.maximum(q0, 0.0)
    best_score = 2.0 * n * float(np.max(q0)) if q0.size else float("inf")
    qn0 = q_normalized(q0)
    best_obj = objective_value(qn0, max_weight)

    logp = np.log(np.maximum(p, _FLOOR))
    sigma = max(_FLOOR, float(sigma0))
    mom = float(momentum)
    if not np.isfinite(mom):
        mom = 0.0
    mom = max(0.0, min(0.8, mom))

    rng = np.random.default_rng(seed)
    last_improve = start_time
    p_prev = None
    it = 0

    while time.monotonic() < end_time:
        it += 1
        now = time.monotonic()

        p = pos_normalize(p)
        p = np.maximum(p, _FLOOR)
        p = p / p.sum()
        old_p = p.copy()

        if p_prev is not None and mom > 0.0:
            y = p + mom * (p - p_prev)
            y = np.maximum(y, _FLOOR)
            y = pos_normalize(y)
        else:
            y = p.copy()

        q = fast_conv(y)
        q = np.where(np.isfinite(q), q, 0.0)
        q = np.maximum(q, 0.0)
        qs = float(q.sum())
        if not np.isfinite(qs) or qs <= _FLOOR:
            break
        qn = q / qs

        # Track current p's true proxy score.
        q_p = fast_conv(p)
        q_p = np.where(np.isfinite(q_p), q_p, 0.0)
        q_p = np.maximum(q_p, 0.0)
        score = 2.0 * n * float(np.max(q_p)) if q_p.size else float("inf")
        qn_p = q_normalized(q_p)
        obj = objective_value(qn_p, max_weight)

        if score < best_score - _EPS:
            best_score = score
            best_p = p.copy()
            last_improve = now
        if obj < best_obj - 1e-12:
            best_obj = obj
            last_improve = now

        if now - last_improve > float(stall_limit):
            break

        w = make_weights(qn, peak_weight, beta, topk)
        D = hankel_product(w, y)
        D = np.where(np.isfinite(D), D, 0.0)
        D = np.maximum(D, _FLOOR)
        meanD = float(D.mean())
        if not np.isfinite(meanD) or meanD <= _FLOOR:
            break

        delta = np.log(meanD) - np.log(D)
        delta = np.clip(delta, -8.0, 8.0)

        cands = [sigma]
        if sigma > 0.02:
            cands.append(sigma * 0.5)
        elif sigma < 0.4:
            cands.append(sigma * 2.0)

        accepted = False
        best_trial = None
        best_trial_obj = float("inf")
        best_trial_score = float("inf")
        best_sgm = sigma

        for sgm in cands:
            new_logp = logp + sgm * delta
            p_try = softmax_log(new_logp)
            p_try = np.maximum(p_try, _FLOOR)
            s_try = float(p_try.sum())
            if not np.isfinite(s_try) or s_try <= _FLOOR:
                continue
            p_try = p_try / s_try

            q_try = fast_conv(p_try)
            q_try = np.where(np.isfinite(q_try), q_try, 0.0)
            q_try = np.maximum(q_try, 0.0)
            qts = float(q_try.sum())
            if not np.isfinite(qts) or qts <= _FLOOR:
                continue
            qn_try = q_try / qts
            score_try = 2.0 * n * float(np.max(q_try)) if q_try.size else float("inf")
            obj_try = objective_value(qn_try, max_weight)

            if obj_try < obj - 1e-12 or score_try < score - _EPS:
                if (
                    not accepted
                    or obj_try < best_trial_obj
                    or (
                        abs(obj_try - best_trial_obj) < 1e-12
                        and score_try < best_trial_score
                    )
                ):
                    accepted = True
                    best_trial = p_try
                    best_trial_obj = obj_try
                    best_trial_score = score_try
                    best_sgm = sgm

        if accepted:
            p = best_trial
            p_prev = old_p
            logp = np.log(np.maximum(p, _FLOOR))

            if best_trial_score < best_score - _EPS:
                best_score = best_trial_score
                best_p = p.copy()
                last_improve = now
            if best_trial_obj < best_obj - 1e-12:
                best_obj = best_trial_obj
                last_improve = now

            if best_sgm < sigma:
                sigma = max(_FLOOR, sigma * 0.85)
            else:
                sigma = min(0.9, sigma * 1.06)
        else:
            # Reset momentum and either shrink step or perturb from best.
            sigma = max(_FLOOR, sigma * 0.6)
            p_prev = old_p
            if sigma < 0.02 or it % 20 == 0:
                lp = np.log(np.maximum(best_p, _FLOOR)) + rng.normal(0.0, 0.03, n)
                p = softmax_log(lp)
                p = np.maximum(p, _FLOOR)
                p = p / p.sum()
                logp = np.log(np.maximum(p, _FLOOR))
                p_prev = None
                sigma = 0.08

        if verbose and it % 25 == 0:
            print(
                f"    {label} it={it} internal_best={best_score:.7f} "
                f"obj={best_obj:.7f} sigma={sigma:.4f}"
            )

    return best_p, best_score


def coordinate_polish(
    p_init,
    limit,
    start_time,
    beta_factor,
    topk_factor,
    seed,
    label,
    verbose=True,
):
    """
    Peak-targeted coordinate moves.

    Uses the same gradient vector D.  It moves mass from high-D coordinates to
    low-D coordinates and keeps any move that lowers the true max coefficient.
    """
    if limit <= 0:
        p = pos_normalize(p_init)
        return p, internal_score(p)

    end_time = start_time + float(limit)
    n = len(p_init)

    p = pos_normalize(p_init)
    p = np.maximum(p, _FLOOR)
    p = p / p.sum()

    best_p = p.copy()
    best_score = internal_score(p)
    last_improve = start_time
    it = 0

    rng = np.random.default_rng(seed)
    bfac = float(beta_factor)
    if not np.isfinite(bfac) or bfac <= 0.0:
        bfac = 50.0
    beta = bfac * n
    tfac = float(topk_factor)
    if not np.isfinite(tfac) or tfac <= 0.0:
        tfac = 0.02
    topk = int(max(5, tfac * n))
    topk = min(topk, 2 * n - 1)

    fracs = (0.25, 0.12, 0.05, 0.02)

    while time.monotonic() < end_time:
        it += 1
        now = time.monotonic()

        p = pos_normalize(p)
        p = np.maximum(p, _FLOOR)
        p = p / p.sum()

        q = fast_conv(p)
        q = np.where(np.isfinite(q), q, 0.0)
        q = np.maximum(q, 0.0)
        qs = float(q.sum())
        if not np.isfinite(qs) or qs <= _FLOOR:
            break
        qn = q / qs
        score = 2.0 * n * float(np.max(q)) if q.size else float("inf")

        if score < best_score - _EPS:
            best_score = score
            best_p = p.copy()
            last_improve = now

        if now - last_improve > 30.0:
            break

        w = make_weights(qn, 0.95, beta, topk)
        D = hankel_product(w, p)
        D = np.where(np.isfinite(D), D, 0.0)
        D = np.maximum(D, _FLOOR)

        src_order = np.argsort(D)[::-1]
        dst_order = np.argsort(D)

        improved = False
        nsrc = min(2, src_order.size)
        ndst = min(2, dst_order.size)

        for ii in range(nsrc):
            for jj in range(ndst):
                s_idx = int(src_order[ii])
                d_idx = int(dst_order[jj])
                if s_idx == d_idx or p[s_idx] <= _FLOOR:
                    continue
                for frac in fracs:
                    amt = float(p[s_idx]) * float(frac)
                    trial = p.copy()
                    trial[s_idx] -= amt
                    if trial[s_idx] < -1e-18:
                        continue
                    trial[d_idx] += amt
                    trial = np.maximum(trial, 0.0)
                    ss = float(trial.sum())
                    if not np.isfinite(ss) or ss <= _FLOOR:
                        continue
                    trial = trial / ss
                    qt = fast_conv(trial)
                    qt = np.where(np.isfinite(qt), qt, 0.0)
                    qt = np.maximum(qt, 0.0)
                    sc = 2.0 * n * float(np.max(qt)) if qt.size else float("inf")
                    if sc < best_score - _EPS:
                        p = trial
                        best_p = trial.copy()
                        best_score = sc
                        last_improve = now
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Small escape move from the strongest high-D to strongest low-D.
            if nsrc > 0 and ndst > 0:
                s_idx = int(src_order[0])
                d_idx = int(dst_order[0])
                if s_idx != d_idx and p[s_idx] > _FLOOR:
                    amt = float(p[s_idx]) * 0.002
                    trial = p.copy()
                    trial[s_idx] -= amt
                    trial[d_idx] += amt
                    trial = np.maximum(trial, 0.0)
                    ss = float(trial.sum())
                    if np.isfinite(ss) and ss > _FLOOR:
                        trial = trial / ss
                        qt = fast_conv(trial)
                        qt = np.where(np.isfinite(qt), qt, 0.0)
                        qt = np.maximum(qt, 0.0)
                        sc = 2.0 * n * float(np.max(qt)) if qt.size else float("inf")
                        if np.isfinite(best_score):
                            tol = best_score * 1e-5 + 1e-12
                        else:
                            tol = 0.0
                        if sc <= tol:
                            p = trial

            if it % 20 == 0:
                lp = np.log(np.maximum(best_p, _FLOOR)) + rng.normal(0.0, 0.02, n)
                p = softmax_log(lp)
                p = np.maximum(p, _FLOOR)
                p = p / p.sum()

        if verbose and it % 10 == 0:
            print(f"    {label} it={it} internal_best={best_score:.7f}")

    return best_p, best_score


def interpolate_to_length(p, new_n):
    """Interpolate a normalized sequence to a new length using cell centers."""
    p = pos_normalize(p)
    old_n = p.size
    new_n = int(new_n)
    if new_n <= 1:
        return np.array([1.0], dtype=np.float64)
    if old_n == new_n:
        return p.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n

    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    q = np.interp(x_new, x_old, p)
    return pos_normalize(q)


def evaluate_safe(a):
    """Safely call the externally provided evaluate_sequence."""
    try:
        if a is None:
            return float("inf")
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        lst = [float(x) for x in arr]
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def update_best(best_p, best_score, cand_p, label):
    """Update the globally best sequence using exact evaluate_sequence."""
    if cand_p is None:
        return best_p, best_score
    sc = evaluate_safe(cand_p)
    if np.isfinite(sc) and sc < best_score - _EPS:
        print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(cand_p)})")
        return cand_p, sc
    return best_p, best_score


def choose_work_candidate(current_work, best_p, best_score, cand_p):
    """
    Decide whether to continue optimization from cand_p or revert to the exact
    best sequence found so far.
    """
    if cand_p is None:
        return best_p
    iw = internal_score(cand_p)
    if np.isfinite(iw) and np.isfinite(best_score) and iw <= best_score * 1.005 + 1e-9:
        return pos_normalize(cand_p)
    return pos_normalize(best_p)


def propose_candidate(budget_s=1800.0):
    """Search for a low-score non-negative step-function height sequence."""
    total_start = time.monotonic()
    raw_budget = max(1.0, min(float(budget_s), 1800.0))
    if raw_budget > 120.0:
        total_budget = raw_budget - 60.0
    else:
        total_budget = max(5.0, raw_budget * 0.7)
    end_total = total_start + total_budget

    print("=== Starting candidate search ===")
    print(f"  usable budget: {total_budget:.1f}s")

    best_p = None
    best_score = float("inf")

    # Try to use the provided parent construction.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent.size > 0:
            p = pos_normalize(parent)
            sc = evaluate_safe(p)
            if np.isfinite(sc):
                best_p = p
                best_score = sc
            else:
                sc2 = internal_score(p)
                if np.isfinite(sc2):
                    best_p = p
                    best_score = sc2
    except Exception:
        pass

    # Fallback deterministic-ish random start.
    if best_p is None:
        n = 2400
        rng0 = np.random.default_rng(20251234)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        base = (
            1.0
            + 0.35 * np.sin(2.0 * np.pi * x)
            + 0.20 * np.cos(4.0 * np.pi * x)
            + 0.25 * rng0.normal(0.0, 1.0, n)
        )
        base = np.maximum(base, 1e-6)
        best_p = pos_normalize(base)
        best_score = evaluate_safe(best_p)
        if not np.isfinite(best_score):
            best_score = internal_score(best_p)

    if not np.isfinite(best_score):
        best_score = internal_score(best_p)

    work_p = pos_normalize(best_p)
    print(f"  initial: n={len(best_p)}, exact_score={best_score:.7f}")

    # Phase 1: L2 flattening on the current best length.
    rem = max(0.0, end_total - time.monotonic())
    if rem > 120.0:
        lim = min(rem * 0.30, 500.0)
        print(f"\n=== Phase 1: L2 flatten, n={len(work_p)}, budget={lim:.1f}s ===")
        cand_p, _ = balance_by_weights(
            work_p,
            lim,
            time.monotonic(),
            peak_weight=0.0,
            beta=0.0,
            topk=0,
            max_weight=0.0,
            sigma0=0.12,
            stall_limit=45.0,
            seed=11,
            label="L2",
            momentum=0.3,
            verbose=True,
        )
        best_p, best_score = update_best(best_p, best_score, cand_p, "L2")
        work_p = choose_work_candidate(work_p, best_p, best_score, cand_p)

    # Phase 2: Smooth peak-weighted balancing.
    rem = max(0.0, end_total - time.monotonic())
    if rem > 120.0:
        n = len(work_p)
        lim = min(rem * 0.28, 450.0)
        print(f"\n=== Phase 2: smooth peak balance, n={n}, budget={lim:.1f}s ===")
        cand_p, _ = balance_by_weights(
            work_p,
            lim,
            time.monotonic(),
            peak_weight=0.5,
            beta=30.0 * n,
            topk=0,
            max_weight=0.5,
            sigma0=0.08,
            stall_limit=35.0,
            seed=22,
            label="SMOOTH",
            momentum=0.3,
            verbose=True,
        )
        best_p, best_score = update_best(best_p, best_score, cand_p, "SMOOTH")
        work_p = choose_work_candidate(work_p, best_p, best_score, cand_p)

    # Phase 3: Try a longer length by interpolation, then re-balance.
    rem = max(0.0, end_total - time.monotonic())
    n_best = len(best_p)
    if rem > 450.0 and n_best < 5000:
        n2 = int(round(1.25 * n_best))
        if n2 - n_best < 300:
            n2 = n_best + 300
        n2 = min(max(800, n2), 5000)
        if n2 != n_best:
            p2 = interpolate_to_length(best_p, n2)
            print(f"\n=== Phase 3: length extrapolation {n_best} -> {n2} ===")

            rem = max(0.0, end_total - time.monotonic())
            if rem > 150.0:
                lim = min(rem * 0.12, 150.0)
                p2, _ = balance_by_weights(
                    p2,
                    lim,
                    time.monotonic(),
                    peak_weight=0.0,
                    beta=0.0,
                    topk=0,
                    max_weight=0.0,
                    sigma0=0.10,
                    stall_limit=30.0,
                    seed=33,
                    label="LEN_L2",
                    momentum=0.25,
                    verbose=True,
                )
                best_p, best_score = update_best(best_p, best_score, p2, "LEN_L2")

            rem = max(0.0, end_total - time.monotonic())
            if rem > 200.0:
                n2 = len(p2)
                lim = min(rem * 0.25, 300.0)
                p2, _ = balance_by_weights(
                    p2,
                    lim,
                    time.monotonic(),
                    peak_weight=0.5,
                    beta=30.0 * n2,
                    topk=0,
                    max_weight=0.5,
                    sigma0=0.08,
                    stall_limit=30.0,
                    seed=44,
                    label="LEN_SMOOTH",
                    momentum=0.3,
                    verbose=True,
                )
                best_p, best_score = update_best(best_p, best_score, p2, "LEN_SMOOTH")

            rem = max(0.0, end_total - time.monotonic())
            if rem > 80.0:
                n2 = len(p2)
                lim = min(rem * 0.08, 80.0)
                p2, _ = coordinate_polish(
                    p2,
                    lim,
                    time.monotonic(),
                    beta_factor=50.0,
                    topk_factor=0.02,
                    seed=55,
                    label="LEN_COORD",
                    verbose=True,
                )
                best_p, best_score = update_best(best_p, best_score, p2, "LEN_COORD")

            work_p = choose_work_candidate(work_p, best_p, best_score, p2)

    # Phase 4: Sharp peak balancing on the best current length.
    rem = max(0.0, end_total - time.monotonic())
    if rem > 120.0:
        n = len(work_p)
        lim = min(rem * 0.30, 350.0)
        print(f"\n=== Phase 4: sharp peak balance, n={n}, budget={lim:.1f}s ===")
        cand_p, _ = balance_by_weights(
            work_p,
            lim,
            time.monotonic(),
            peak_weight=0.95,
            beta=100.0 * n,
            topk=0,
            max_weight=0.95,
            sigma0=0.05,
            stall_limit=30.0,
            seed=66,
            label="SHARP",
            momentum=0.2,
            verbose=True,
        )
        best_p, best_score = update_best(best_p, best_score, cand_p, "SHARP")
        work_p = choose_work_candidate(work_p, best_p, best_score, cand_p)

    # Phase 5: Final coordinate polish.
    rem = max(0.0, end_total - time.monotonic())
    if rem > 60.0:
        n = len(work_p)
        lim = min(rem * 0.55, 260.0)
        print(f"\n=== Phase 5: coordinate polish, n={n}, budget={lim:.1f}s ===")
        cand_p, _ = coordinate_polish(
            work_p,
            lim,
            time.monotonic(),
            beta_factor=70.0,
            topk_factor=0.02,
            seed=77,
            label="COORD",
            verbose=True,
        )
        best_p, best_score = update_best(best_p, best_score, cand_p, "COORD")
        work_p = choose_work_candidate(work_p, best_p, best_score, cand_p)

    # Phase 6: A few short randomized restarts if there is still useful time.
    rem = max(0.0, end_total - time.monotonic())
    restarts = 0
    while rem > 160.0 and restarts < 3:
        n = len(best_p)
        rng_r = np.random.default_rng(1000 + 17 * restarts)
        lp_r = np.log(np.maximum(best_p, _FLOOR)) + 0.05 * rng_r.normal(0.0, 1.0, n)
        p_r = softmax_log(lp_r)
        p_r = np.maximum(p_r, _FLOOR)
        p_r = p_r / p_r.sum()

        lim = min(rem * 0.25, 150.0)
        print(
            f"\n=== Phase 6: random restart {restarts + 1}, n={n}, budget={lim:.1f}s ==="
        )
        cand_p, _ = balance_by_weights(
            p_r,
            lim,
            time.monotonic(),
            peak_weight=0.5,
            beta=40.0 * n,
            topk=0,
            max_weight=0.5,
            sigma0=0.08,
            stall_limit=20.0,
            seed=2000 + 17 * restarts,
            label=f"RESTART{restarts + 1}",
            momentum=0.2,
            verbose=True,
        )
        best_p, best_score = update_best(best_p, best_score, cand_p, f"RESTART{restarts + 1}")
        work_p = choose_work_candidate(work_p, best_p, best_score, cand_p)
        restarts += 1
        rem = max(0.0, end_total - time.monotonic())

    # Final exact check of the working candidate.
    best_p, best_score = update_best(best_p, best_score, work_p, "FINAL_WORK")

    # Guarantee a valid finite output.
    if best_p is None or not np.isfinite(best_score):
        # Last-resort simple valid sequence.
        n = 2400
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        best_p = pos_normalize(1.0 + 0.3 * np.sin(2.0 * np.pi * x))
        best_score = evaluate_safe(best_p)
        if not np.isfinite(best_score):
            best_score = internal_score(best_p)

    final_score = evaluate_safe(best_p)
    if np.isfinite(final_score):
        best_score = final_score

    best_p = pos_normalize(best_p)
    print("\n=== Final ===")
    print(f"  n = {len(best_p)}")
    print(f"  exact_score = {best_score:.7f}")
    print(f"  elapsed = {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_p]
```
