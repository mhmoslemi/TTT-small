def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f_norm)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return out / ss


def make_init(n, kind='uniform', seed=42):

def soft_subgrad(a, ac, top_k=40, beta=6.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        x_prev = x
        x = a_new

        M_new = fast_autoconv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 120)
            cur_beta = max(cur_beta - 0.3, 2.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, t_limit, solver, top_K=200, verbose=True):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autoconv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    b_var = cp.Variable(n, nonneg=True)

    for t in range(1, 50000):
        if time.monotonic() - t0 > t_limit:
            break

        ac = fast_autoconv(y)
        M_curr = ac.max()

        # Build Toeplitz constraint matrix for current y
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        # Select top-K binding constraints
        margins = M_curr - T_full @ y
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx]
        M_sel = M_curr * np.ones(K_sel)

        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autoconv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autoconv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coord_descent(a0, t_limit, rng_seed=77, verbose=True):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autoconv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < t_limit and stall < 500:
        ac = fast_autoconv(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        # Target the highest peaks
        for k in top_k[:8]:
            if time.monotonic() - t0 > t_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destination
            sens = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        sens[d] += a[d1]
            sens[src] = 1e10

            dst = np.argmin(sens)
            if dst == src:
                continue

            for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                amt = src_val * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-12:
                    continue
                a_t = a_t / s_t
                M_t = fast_autoconv(a_t).max()
                if M_t < best_m - 1e-14:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autoconv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            J_local = 2.0 * len(a) * fast_autoconv(a / a.sum()).max()
            print(f"  *** NEW BEST [{label}]: eval={sc:.6f}, local_J={J_local:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init (uniform n={n_base}): score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: AMD from multiple inits
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD multi-init (n={n0}) ===")
    t1 = min(remaining * 0.30, 500.0)

    if t1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(parent)
            labels.append("parent")
        inits.append(make_init(n0, 'uniform'))
        labels.append("uniform")
        inits.append(make_init(n0, 'ramp'))
        labels.append("ramp")
        inits.append(make_init(n0, 'bimodal'))
        labels.append("bimodal")
        inits.append(make_init(n0, 'smooth_rand', seed=77))
        labels.append("srand")

        t_per = max(15.0, t1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            if time.monotonic() - t_global > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)")
            a_out, _ = amd_optimize(init, t_avail, top_k=40, beta=5.0, c0=4.0,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 4000:
        a_in = norm1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, top_K=250, verbose=True)
        try_update(a_out, f"lp_n{n_best}")

    # Phase 3: AMD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 3: AMD polish (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, top_k=70, beta=4.0, c0=3.5,
                                 restart_every=8000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_in = norm1(best_seq)
        a_out, _ = coord_descent(a_in, t4, rng_seed=42, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 5: Length sweep - try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 5000]
    print(f"\n=== Phase 5: Length sweep ===")

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n}")
        a_long = interp_seq(best_seq, new_n)

        t5a = min(remaining * 0.30, 180.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=50, beta=5.0, c0=3.5,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4000:
            t5b = min(remaining * 0.25, 150.0)
            if t5b > 20:
                a_in2 = norm1(best_seq)
                a_out2, _ = lp_frank_wolfe(a_in2, t5b, solver, top_K=200, verbose=False)
                try_update(a_out2, f"lp_n{new_n}")

    # Phase 6: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 50 and n_best <= 4000:
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 20:
            a_in = norm1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, top_K=400, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # Phase 7: Final coord
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n0
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.5, 120.0)
        if t7 > 10:
            a_in = norm1(best_seq)
            a_out, _ = coord_descent(a_in, t7, rng_seed=999, verbose=True)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(2400) / 2400
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
