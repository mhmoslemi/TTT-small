def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_weights(ac, T):
    ac_clipped = np.maximum(ac, 1e-30)
    log_w = ac_clipped / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros_like(ac)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = w / sw
    return w


def compute_ssl_grad(w, a):
    n = len(a)
    a_flipped = a[::-1].copy()
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flipped, fft_size)
    conv = np.fft.irfft(W * AF, fft_size)
    return conv[n - 1:n - 1 + n]


def smooth_obj_grad(x, T, fft_size):
    n = len(x)
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    a = exp_x / Z
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
    ac = np.maximum(ac, 0.0)
    max_ac = ac.max()
    if max_ac < 1e-30:
        return 0.0, np.zeros_like(x)
    shifted = (ac - max_ac) / T
    shifted = np.minimum(shifted, 50.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    value = max_ac + T * log_sum
    w = np.exp(shifted)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros_like(ac)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = w / sw
    grad_a = compute_ssl_grad(w, a)
    mean_grad = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_grad)
    return value, grad_x

def smoothed_mirror_descent(a0, time_limit, T_start=0.05, T_end=0.003, c0=3.0,
                            verbose=True, rng_seed=42):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    current_c = c0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1.0))
        T = max(T_start * (T_end / T_start) ** progress, 0.001)

        beta = (total_iters - 1.0) / (total_iters + 1.5) if total_iters > 1 else 0.0
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        w = softmax_weights(ac, T)
        grad = compute_ssl_grad(w, y)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        else:
            grad = np.zeros(n)

        eta = current_c / (np.sqrt(total_iters + 1) + 1.0) * np.sqrt(n)
        eta = min(eta, 5.0)

        log_a = np.log(np.maximum(y, 1e-15)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        total_iters += 1

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            current_c = min(current_c * 1.3, 12.0)
            ac_best = fast_autocorr(best_a)

            x = x_restarted
            x_prev = x_restarted
            total_iters = max(total_iters, 50)

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M

def exact_amd(a0, time_limit, c0=3.0, threshold_frac=0.995, verbose=True, rng_seed=7):

    best_a = a.copy()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    current_c = c0
    current_thresh = threshold_frac

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()
        active = np.where(ac >= current_thresh * M)[0]
        if len(active) == 0:
            active = np.array([int(ac.argmax())])

        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * y[j_idx]

        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn

        eta = current_c / (np.sqrt(t) + 1.0) * np.sqrt(n)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0

        if stall > max_stall:
            stall = 0
            current_c = min(current_c * 1.25, 15.0)
            if current_thresh > 0.98:
                current_thresh = max(0.98, current_thresh - 0.005)
            noise = rng.normal(0, 0.008, n) * best_a
            x_restart = np.maximum(best_a + noise, 1e-10)
            x_restart = x_restart / x_restart.sum()
            x = x_restart
            x_prev = x_restart

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, thresh={current_thresh:.4f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    AMD done ({elapsed:.1f}s): J={J:.6f}")

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 60

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_peaks = min(12, len(ac))
        peak_idx = np.argsort(ac)[::-1][:n_peaks]

        sens = np.zeros(n)
        for k in peak_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            sens[idx] += ac[k_int] * a[j_idx]

        high_sens = np.argsort(sens)[::-1][:min(60, n)]
        low_sens = np.argsort(sens)[:min(60, n)]

        improved = False
        for src in high_sens[:40]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in low_sens[:30]:
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = max(5, n // 150)
            for _ in range(n_pert):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.06)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
            if not improved and stall > 30:
                break

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Coord done ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        do_try_update(a_out, f"lbfgsb_n{n_best}")

    # === Phase 2: Smoothed mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SMD annealing n={n_best} ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 30:
        a_out, _ = smoothed_mirror_descent(normalize(best_seq), t2,
                                           T_start=0.05, T_end=0.003,
                                           c0=3.0, verbose=True, rng_seed=42)
        do_try_update(a_out, f"smd_n{n_best}")

    # === Phase 3: Exact AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Exact AMD n={n_best} ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = exact_amd(normalize(best_seq), t3, c0=3.5,
                             threshold_frac=0.995, verbose=True, rng_seed=7)
        do_try_update(a_out, f"amd_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 20:
        a_out, _ = coordinate_descent(normalize(best_seq), t4, verbose=True, rng_seed=123)
        do_try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: SMD again from improved point (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SMD round 2 n={n_best} ===")
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 20:
        a_out, _ = smoothed_mirror_descent(normalize(best_seq), t5,
                                           T_start=0.02, T_end=0.002,
                                           c0=4.0, verbose=True, rng_seed=99)
        do_try_update(a_out, f"smd2_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length n_best={n_best} ===")
        for n_try in [n_best + 200, n_best - 200, n_best + 400, n_best - 400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            if n_try < 500:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(remaining * 0.30, 120.0)
            if t_sub < 20:
                continue
            print(f"  Trying n={n_try}, time={t_sub:.0f}s")
            a_out, _ = smoothed_mirror_descent(normalize(a_interp), t_sub,
                                               T_start=0.03, T_end=0.003,
                                               c0=3.0, verbose=True, rng_seed=n_try % 1000)
            do_try_update(a_out, f"smd_len{n_try}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} ===")
    if remaining > 20:
        t7 = min(remaining * 0.35, 120.0)
        if t7 > 10:
            a_out, _ = coordinate_descent(normalize(best_seq), t7, verbose=True, rng_seed=456)
            do_try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final SMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.50, 80.0)
        if t8 > 10:
            a_out, _ = smoothed_mirror_descent(normalize(best_seq), t8,
                                               T_start=0.005, T_end=0.001,
                                               c0=2.0, verbose=True, rng_seed=789)
            do_try_update(a_out, f"smd_subpolish_n{n_best}")

    # === Phase 9: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.50, 60.0)
        if t9 > 10:
            a_out, _ = exact_amd(normalize(best_seq), t9, c0=2.0,
                                 threshold_frac=0.999, verbose=True, rng_seed=321)
            do_try_update(a_out, f"amd_polish_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
