        return np.ones(n) / n
    rho = rho_candidates[-1]

    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected gradient on the simplex
2. Exact subgradient with top-K active lag tracking
3. Spectral perturbation for escaping local minima  
4. Peak-aware coordinate descent for final polish
5. Multi-scale exploration (adaptive length search)

import numpy as np
import time
import math

# --- Core functions (top-level, no closures) ---

def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * np.conj(A), fft_size)[:size]

def safe_eval(a):
    """Call the external evaluator safely."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return result / s  # Extra normalization for numerical stability

def compute_grad_topk(a, K=40, sharpness=15.0):
    """
    Compute smoothed subgradient using top-K lags.
    Returns (gradient, peak_value).
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Get top K lags
    top_k = np.argsort(ac)[::-1][:K]
    top_vals = ac[top_k]
    
    # Softmax weights (sharper = more focus on the max)
    shifted = top_vals - top_vals[0]
    weights = np.exp(sharpness * shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights = weights / w_sum
    
    # Gradient: for lag k, d(c[k])/d(a[p]) = 2*a[k-p] for p in valid range
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_k):
        k_int = int(k)
        p_lo = max(0, k_int - n + 1)
        p_hi = min(k_int, n - 1)
        if p_lo > p_hi:
            continue
        idx = np.arange(p_lo, p_hi + 1)
        grad[idx] += 2.0 * w * a[k_int - idx]
    
    return grad, M

def nesterov_gradient(a0, time_limit, K=40, sharpness=15.0, L=2.5, 
                       momentum_decay=0.999, verbose=True, seed=42):
    """
    Nesterov-accelerated projected gradient on the simplex.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = norm1(a0)
    x = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    step_mult = 1.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta = 0.5 * (1.0 + t) / (t + 2.5)  # or (t-1)/(t+1.5)
        y = x + beta * (x - x_prev)
        y = proj_simplex(y)
        
        # Compute gradient at y
        g, M_y = compute_grad_topk(y, K=K, sharpness=sharpness)
        
        # Adaptive step size
        eta = step_mult / L
        
        # Gradient step with projection
        x_new = proj_simplex(y - eta * g)
        
        # Evaluate
        M_new = fast_autocorr(x_new).max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Backtracking if no improvement for a while
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.2)
            step_mult = max(0.5, step_mult * 0.8)
            # Occasionally perturb
            if max_stall > 2000:
                noise = rng.normal(0, 0.01, n) * best_a
                x = proj_simplex(best_a + noise)
                x_prev = x.copy()
        
        # Periodic step size annealing
        if t % 10000 == 0:
            step_mult = max(0.3, step_mult * 0.95)
        
        x_prev = x
        x = x_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    NG t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.8f}, step={eta:.4f}")
    
    return best_a, best_M

def spectral_shake(a, rng, num_shakes=5):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    a = norm1(a)
    fft_size = 1 << (2*n - 2).bit_length()
    
    for _ in range(num_shakes):
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        mag = np.abs(A)
        
        # Random multiplicative perturbation
        pert = 1.0 + rng.normal(0, 0.05, len(mag))
        pert[0] = 1.0  # Don't perturb DC
        mag_new = np.maximum(mag * pert, 0)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm1(a_new)
        
        M_new = fast_autocorr(a_new).max()
        M_old = fast_autocorr(a).max()
        if M_new < M_old:
            a = a_new
    
    return a
