    Compute gradient of sum(c[k]^p) w.r.t. a, where c = conv(a,a).

def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True, label="LP-FW"):
    """LP Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_m = fast_convolution(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_acc = a.copy()
    x_acc_prev = a.copy()
    iter_count = 0
    stall = 0
    max_stall = 50

    # Pre-allocate T
    T = np.zeros((2 * n - 1, n))

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolution(y)
        M_curr = c.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix T[k,j] = y[k-j] for 0 <= k-j < n
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Accelerated sequence
        if t == 1:
            x_acc_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_acc_new = y_new + beta_t * (y_new - y)
            x_acc_new = np.maximum(x_acc_new, 0.0)
            sx = x_acc_new.sum()
            if sx < 1e-15:
                x_acc_new = y_new.copy()
            else:
                x_acc_new = x_acc_new / sx

        y = y_new
        x_acc = x_acc_new
        x_acc_prev = x_acc
        iter_count += 1

        my = fast_convolution(y).max()
        mx = fast_convolution(x_acc).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            stall = 0
        elif mx < best_m:
            best_m = mx
            best_a = x_acc.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            y = best_a + 0.01 * np.random.default_rng(iter_count).normal(0, 1, n) * best_a
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            x_acc = y.copy()
            x_acc_prev = y.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    {label} iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

    n = len(a0)

        peak_idx = int(np.argmax(c))

        je = min(n - 1, k_int)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_int - idx]
