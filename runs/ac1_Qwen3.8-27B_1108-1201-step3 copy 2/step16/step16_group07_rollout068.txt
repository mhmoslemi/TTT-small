            if prob.status in ('optimal', 'optimal_inaccurate'):
                _GLOBAL_SOLVER = s
                print("  LP solver: %s" % s)
                return _GLOBAL_SOLVER
        except Exception:
            continue
    _GLOBAL_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _GLOBAL_SOLVER


def _fft_size(n):
    r = 1
    while r < n:
        r <<= 1
    return r


def fast_autocorr(a):
    """Full linear convolution a*a (autocorrelation), length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    N = _fft_size(size)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def max_eff(a):
    return float(fast_autocorr(a).max())


def normalize_to(a, n):
    nrm = math.sqrt(2.0 * n)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * nrm / n
    return a * nrm / s


def evaluate_safe(a):
    try:
        a = np.asarray(a, dtype=np.float64)
        if a.size == 0 or not np.isfinite(a).all():
            return float('inf')
        lst = [float(x) for x in a]
        sc = evaluate_sequence(lst)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(z, S):
    """Project z onto {x>=0, sum x = S} via bisection on the threshold."""
    z = np.asarray(z, dtype=np.float64)
    lo = z.min() - S
    hi = z.max()
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = np.maximum(z - mid, 0.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(z - 0.5 * (lo + hi), 0.0)


def _smooth(c, tau):
    m = float(c.max())
    v = np.exp(tau * c - tau * m)
    return m + math.log(v.sum()) / tau


def _softmax_weights(c, tau):
    m = float(c.max())
    v = np.exp(tau * c - tau * m)
    z = v.sum()
    if z <= 0:
        return np.ones_like(c) / c.size
    return v / z


def gradient_fft(a, w):
    """Gradient g[j] = sum_k w[k]*(a[k-j]+a[j-k]) computed with 2 FFTs."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    L = w.size
    need = 3 * n
    N = _fft_size(need)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Q[j] = (w * a)[j]   (linear convolution, zero padded)
    Q = np.fft.irfft(W * A, N)[:n]
    # P[j] = sum_m w[m+j]*a[m]  = conv(flip(w), a)[L-1+j]
    wrev = np.zeros(N)
    wrev[:L] = w[::-1]
    WRev = np.fft.rfft(wrev, N)
    C = np.fft.irfft(WRev * A, N)
    P = C[L - 1:L - 1 + n]
    return P + Q


def gradient_direct(a, w):
    """O(n^2) reference gradient (used only for self test / tiny n)."""
    n = a.size
    g = np.zeros(n)
    for j in range(n):
        s = 0.0
        k0 = max(0, j)
        k1 = min(2 * n - 1, j + n)
        for k in range(k0, k1):
            s += w[k] * a[k - j]
        k0 = max(0, j - n + 1)
        k1 = min(2 * n - 1, j + 1)
        for k in range(k0, k1):
            s += w[k] * a[j - k]
        g[j] = s
    return g


def _self_test_gradient():

def peak_shave(a, S, base_lr):
    """One projected line-search step along the argmax subgradient."""
    c = fast_autocorr(a)
    k = int(np.argmax(c))
    one = np.zeros_like(c)
    one[k] = 1.0
    g = gradient_fft(a, one)               # == dc[k]/da
    best = a
    bestc = float(c.max())
    lr = base_lr
    for _ in range(25):
        if lr < 1e-7 * S:
            break
        z = a - lr * g
        an = project_simplex(z, S)
        cn = float(fast_autocorr(an).max())
        if cn < bestc:
            best = an
            bestc = cn
            a = an
            c = fast_autocorr(a)
            lr *= 1.35
        else:
            lr *= 0.5
    return best, bestc


def softmax_optimize(a0, S, time_limit, seed=0, verbose=False):
    """Smoothed-softmax FISTA (annealed tau) + periodic peak shaving."""
    rng = np.random.default_rng(seed)
    n = a0.size
    t0 = __import__('time').monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())

    y = a.copy()
    a_old = a.copy()
    t_mom = 1.0
    alpha0 = S / math.sqrt(n)

    tau = 1.0
    tau_target = max(4.0, 0.6 * n)
    it = 0
    stall = 0
    stall_lim = 120

    while True:
        if __import__('time').monotonic() - t0 > time_limit:
            break
        it += 1

        c = fast_autocorr(y)
        f = _smooth(c, tau)
        w = _softmax_weights(c, tau)
        g = gradient_fft(a, w)             # gradient direction (uncentered)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        dhat = g / gn

        # backtracking first-decreasing line search on smooth value
        alpha = alpha0
        accepted = None
        for k in range(14):
            if alpha < 1e-7 * S:
                break
            z = y - alpha * dhat
            an = project_simplex(z, S)
            cn = fast_autocorr(an)
            fn = _smooth(cn, tau)
            if fn <= f + 1e-15 or k == 0:
                accepted = (an, cn, fn, alpha)
                break
            alpha *= 0.5
        if accepted is None:
            alpha = max(alpha0 * (0.7 ** stall), 1e-7 * S)
            z = y - alpha * dhat
            an = project_simplex(z, S)
            cn = fast_autocorr(an)
            fn = _smooth(cn, tau)
            accepted = (an, cn, fn, alpha)

        an, cn, fn, used = accepted
        # Nesterov y update
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = an + (t_mom / t_new - 1.0) * (an - a_old)
        y_new = project_simplex(np.maximum(y_new, 0.0), S)
        y = y_new
        a_old = a
        a = an
        t_mom = t_new

        # anneal temperature
        tau = min(tau * 1.015, tau_target)

        # track best on true objective
        sc = float(cn.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # periodic guaranteed-mono peak shaving
        if it % 20 == 0:
            sp = S
            a, sc2 = peak_shave(a, S, alpha0 * 0.5)
            if sc2 < best_score:
                best_score = sc2
                best_a = a.copy()
                stall = 0
            y = a.copy()

        if stall > stall_lim:
            # small random nudge in feasible space then keep going
            a = project_simplex(np.maximum(best_a * (1.0 + 0.02 * rng.standard_normal(n)), 0.0), S)
            y = a.copy()
            a_old = a.copy()
            t_mom = 1.0
            stall = 0

        if verbose and it % 400 == 0:
            el = __import__('time').monotonic() - t0
            print("    smoothFISTA it=%d (%.0fs) best=%.10f tau=%.2f" % (it, el, best_score, tau))

    # final peak-shave sweep
    a = best_a.copy()
    for _ in range(40):
        if __import__('time').monotonic() - t0 > time_limit:
            break
        a, sc2 = peak_shave(a, S, S / math.sqrt(n) * 0.5)
        if sc2 < best_score:
            best_score = sc2
            best_a = a.copy()
    return best_a, best_score

def subgradient_at_k_fast(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraints[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, seed=0, verbose=False):
    rng = np.random.default_rng(seed)
    n = len(a0)
    nrm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * nrm / n
    else:
        a = a * nrm / s0
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = __import__('time').monotonic()
    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall = 0
    conv_size = 2 * n - 1
    for t in range(1, 10000000):
        if __import__('time').monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        M = float(conv_y.max())
        if M < 1e-12:
            break
        if active_set is None or t % 40 == 0:
            progress = min(1.0, t / 1000.0)
            tol = 0.08 * (1.0 - progress) + 0.004 * progress
            base = M * (1.0 - tol)
            active_set = np.where(conv_y >= base)[0]
            ntop = max(5, n // 80)
            top = np.argsort(conv_y)[-ntop:]
            active_set = np.unique(np.concatenate([active_set, top]))
            if len(active_set) > int(1.5 * n):
                keep = np.argsort(conv_y[active_set])[-int(1.5 * n):]
                active_set = active_set[keep]
        b, duals = lp_best_response(y, M, active_set, solver)
        if b is None and t % 6 == 0:
            b, duals = lp_best_response(y, M, np.arange(conv_size), solver)
        if b is None:
            continue
        bs = b.sum()
        if bs < 1e-15:
            continue
        g = b * nrm / bs
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * nrm / sy
        if t == 1:
            a_new = g.copy()
        else:
            mom = min((t - 1.0) / (t + 2.0), 0.93)
            a_new = y_new + mom * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * nrm / sa
        y_prev = y
        y = y_new
        a = a_new
        scy = float(fast_autocorr(y).max())
        if scy < best_score:
            best_score = scy
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        sca = float(fast_autocorr(a).max())
        if sca < best_score:
            best_score = sca
            best_a = a.copy()
            stall = 0
        if stall > 50:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0
        if verbose and (t % 500) == 0:
            el = __import__('time').monotonic() - t0
            print("    LP it=%d (%.0fs) best=%.10f" % (t, el, best_score))
    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    nrm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * nrm / s0
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = __import__('time').monotonic()
    size = 2 * n - 1
    no_imp = 0
    max_no = 4
    while __import__('time').monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        for j in rng.permutation(n):
            if __import__('time').monotonic() - t0 > time_limit:
                break
            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue
            typical = nrm / n
            cands = sorted(set([max(0.0, min(c, nrm * 0.5)) for c in
                                [0.0, aj * 0.3, aj * 0.6, aj, aj * 1.4, aj * 1.8,
                                 typical * 0.5, typical, typical * 1.5, typical * 3.0]]))
            best_x = aj
            best_obj = float(fast_autocorr(a).max())
            for x in cands:
                if abs(x - aj) < 1e-12:
                    continue
                nc = conv.copy()
                d = x - aj
                nc[lo:hi] += 2.0 * d * a[lo - j:hi - j]
                if 2 * j < size:
                    nc[2 * j] += d * d
                new_max = float(nc.max())
                scale = nrm / (a.sum() + d)
                nm = new_max * scale * scale
                if nm < best_obj:
                    best_obj = nm
                    best_x = x
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                sn = a.sum()
                if sn > 1e-15:
                    a = a * nrm / sn
                    ns = float(fast_autocorr(a).max())
                    if ns < best_score - 1e-13:
                        best_score = ns
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= max_no:
                break
    return best_a, best_score


def dual_guided_polish(a0, time_limit, solver):
    if solver is None:
        return a0, float('inf')
    n = len(a0)
    nrm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * nrm / s0
    best_a = a.copy()
    best_score = float(fast_autocorr(a).max())
    t0 = __import__('time').monotonic()
    patience = 0
    step = 0.5 / (n ** 0.3)
    for t in range(1, 5000000):
        if __import__('time').monotonic() - t0 > time_limit or patience > 400:
            break
        conv = fast_autocorr(a)
        M = float(conv.max())
        active_set = np.where(conv >= M * 0.95)[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 100):]
        b, duals = lp_best_response(a, M, active_set, solver)
        if b is None:
            patience += 1
            continue
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-5.0 * (M - conv[active_set]) / max(M * 0.02, 1e-10))
            weights = weights / weights.sum()
        grad = np.zeros(n)
        for ww, k in zip(weights, active_set):
            grad += ww * subgradient_at_k_fast(a, int(k), n)
        support = np.where(a > 0.005 * a.max())[0]
        if len(support) < 2:
            break
        gm = float(np.mean(grad[support]))
        direction = grad - gm
        direction[support] -= 0.1 * float(np.std(grad[support]))
        at = a - step * direction
        at = np.maximum(at, 0.0)
        st = at.sum()
        if st < 1e-15:
            at = a.copy()
        else:
            at = at * nrm / st
        sc = float(fast_autocorr(at).max())
        if sc < best_score - 1e-14:
            a = at
            best_score = sc
            best_a = at.copy()
            patience = 0
            step = min(step * 1.005, 1.5)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break
    return best_a, best_score

def _resample_to(src, n):
    src = np.asarray(src, dtype=np.float64)
    m = src.size
    if m == n:
        return src.copy()
    x0 = np.linspace(0.0, 1.0, m)
    x1 = np.linspace(0.0, 1.0, n)
    return np.interp(x1, x0, src)


def _diverse_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    out = []
    out.append(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x))
    f = 2.0 + 3.0 * rng.random()
    out.append(0.5 + 0.35 * np.cos(2 * np.pi * f * x + rng.random() * 6.28))
    out.append(1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 4) * x))
    out.append(1.0 + 0.3 * rng.standard_normal(n))
    return out


def _try_update(best_seq, best_score, cand, label, t0):
    if cand is None:
        return best_seq, best_score
    cand = np.maximum(np.asarray(cand, dtype=np.float64), 0.0)
    if cand.size == 0 or cand.sum() < 0.01 or not np.isfinite(cand).all():
        return best_seq, best_score
    sc = evaluate_safe(cand)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = cand.copy()
        print("  *** NEW BEST [%s]: %.8f (n=%d, t=%.0fs)" % (label, sc, cand.size,
              __import__('time').monotonic() - t0))
        return best_seq, best_score
    return best_seq, best_score

                best_seq, best_score = _try_update(best_seq, best_score, outA,
                                                   "smooth%d_n%d" % (si, n), t0)
                s_arr = normalize_to(best_seq if best_seq.size == n else outA, n)
            # Phase B: LP polish
            if solver is not None and remaining() > 40:
                tB = min(alloc * 0.30, 110.0)
                if tB > 20:
                    outB, _ = nesterov_lp_optimize(normalize_to(s_arr, n), tB, solver,
                                                   seed=500 + n + si)
                    best_seq, best_score = _try_update(best_seq, best_score, outB,
                                                       "lp_n%d_%d" % (n, si), t0)
                    s_arr = normalize_to(best_seq if best_seq.size == n else outB, n)
            # Phase C: coordinate descent
            if remaining() > 30:
                tC = min(alloc * 0.25, 80.0)
                if tC > 15:
                    outC, _ = coordinate_descent_fast(normalize_to(s_arr, n), tC, seed=7 + si)
                    best_seq, best_score = _try_update(best_seq, best_score, outC,
                                                       "cd_n%d_%d" % (n, si), t0)

    # ---- global final polish on the overall best ----
    if best_seq is not None:
        n = best_seq.size
        s_arr = normalize_to(best_seq, n)
        S = math.sqrt(2.0 * n)

        if remaining() > 120:
            print("\n=== Final polish n=%d (remaining %.0fs) ===" % (n, remaining()))
            t = min(remaining() * 0.35, 220.0)
            out, _ = softmax_optimize(s_arr, S, t, seed=9999, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, out, "finalSmooth", t0)

        if solver is not None and remaining() > 90:
            t = min(remaining() * 0.30, 200.0)
            out, _ = nesterov_lp_optimize(normalize_to(best_seq, n), t, solver, seed=12345)
            best_seq, best_score = _try_update(best_seq, best_score, out, "finalLP", t0)
            s_arr = normalize_to(best_seq, n)

        if remaining() > 50:
            t = min(remaining() * 0.30, 150.0)
            out, _ = coordinate_descent_fast(normalize_to(best_seq, n), t, seed=777)
            best_seq, best_score = _try_update(best_seq, best_score, out, "finalCD", t0)

        if remaining() > 30:
            t = min(remaining() * 0.30, 90.0)
            out, _ = dual_guided_polish(normalize_to(best_seq, n), t, solver)
            best_seq, best_score = _try_update(best_seq, best_score, out, "finalDual", t0)

    # guaranteed mono peak-shave sweep on best regardless of budget leftover
    if best_seq is not None and remaining() > 5:
        n = best_seq.size
        S = math.sqrt(2.0 * n)
        a = normalize_to(best_seq, n)
        sc = float(fast_autocorr(a).max())
        while remaining() > 2:
            a, sc2 = peak_shave(a, S, S / math.sqrt(n))
            if sc2 <= sc:
                sc = sc2
                a = a
            else:
                break
        best_seq, best_score = _try_update(best_seq, best_score, a, "peakShave", t0)

    if best_seq is None:
        print("ERROR no valid sequence")
        return [1.0]
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL score=%.8f n=%d time=%.0fs" % (best_score, best_seq.size,
          __import__('time').monotonic() - t0))
    return [float(x) for x in np.maximum(best_seq, 0.0)]
