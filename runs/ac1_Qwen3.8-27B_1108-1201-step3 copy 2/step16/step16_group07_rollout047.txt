    # But w has length 2n-1 and a has length n, so (w ⋆ a)[j] is defined for j = 0, ..., n-1
    # (since we need l from 0 to n-1 and l+j from 0 to 2n-2, so j up to 2n-2)
    #
    # For the second part: sum_k w[k]*a[j-k]
    # Let m = j-k: sum_m w[j-m]*a[m] = sum_m a[m]*w[j-m]
    # This is the convolution (a * w)[j] = sum_m a[m]*w[j-m] ✓
    # = np.convolve(a, w, 'full')[j]
    #
    # So: grad[j] = cross_corr(w, a)[j] + conv(a, w)[j] for j = 0, ..., n-1
    #
    # Using FFT:
    # cross_corr(w, a) = ifft(fft(w, N) * conj(fft(a, N)))
    # conv(a, w) = ifft(fft(a, N) * fft(w, N))
    #
    # So grad = ifft(fft(w)*conj(fft(a))) + ifft(fft(a)*fft(w))
    #          = ifft(fft(w) * (conj(fft(a)) + fft(a)))
    #          = ifft(fft(w) * 2*real(fft(a)))  [since a is real, fft(a) has conjugate symmetry]
    #          = 2 * ifft(fft(w) * real(fft(a)))
    
    N = 1 << (conv_size + n - 1 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    
    # Part 1: cross-correlation
    # (w ⋆ a)[j] = ifft(fft(w) * conj(fft(a)))[j]
    # For real signals: conj(fft(a)) at frequency k = fft(a) at frequency N-k
    # rfft gives frequencies 0 to N/2
    # This is tricky with rfft. Let me use full fft for clarity.
    
    W_full = np.fft.fft(w, N)
    A_full = np.fft.fft(a, N)
    
    # cross_corr = ifft(W_full * conj(A_full))
    # conv = ifft(A_full * W_full)
    # grad = cross_corr[:n] + conv[:n]
    
    cross_corr = np.fft.ifft(W_full * np.conj(A_full)).real[:n]
    conv_part = np.fft.ifft(A_full * W_full).real[:n]
    
    return cross_corr + conv_part

def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a, n):
    """Normalize to sum = sqrt(2n)."""
    target = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target / n
    return a * target / s

def compute_soft_subgradient(a, alpha):
    """
    Compute soft subgradient of max_k conv(a,a)[k].
    Uses log-sum-exp smoothing with temperature alpha.
    Returns (gradient, smoothed_max, conv).
    """
    n = len(a)
    conv = fast_conv(a)
    M = conv.max()
    
    # Exponential weights
    diff = M - conv
    # Avoid overflow
    weights = np.exp(-diff / alpha)
    total_w = weights.sum()
    if total_w < 1e-30:
        # All mass on the argmax
        idx = np.argmax(conv)
        weights = np.zeros_like(conv)
        weights[idx] = 1.0
    else:
        weights /= total_w
    
    # Compute gradient: grad[j] = sum_k weights[k] * (a[k-j] + a[j-k])_valid
    # Use top-K peaks for efficiency
    K = min(50, n // 10)
    top_idx = np.argsort(weights)[-K:]
    top_w = weights[top_idx]
    
    grad = np.zeros(n)
    for wk, k in zip(top_w, top_idx):
        k = int(k)
        # For each j, contribution is a[k-j] (if valid) + a[j-k] (if valid)
        # a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1,k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wk * a[k - np.arange(j_lo, j_hi+1)]
        
        # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += wk * a[np.arange(j_lo2, j_hi2+1) - k]
    
    smoothed_max = M - alpha * np.log(total_w)
    return grad, smoothed_max, conv, M

def mirror_descent_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Mirror descent (entropic) for minimizing max autocorrelation.
    Uses multiplicative updates which naturally maintain non-negativity.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    t0 = time.monotonic()
    eta = 0.5  # Initial step size
    eta_decay = 0.9995
    alpha = best_score * 0.05  # Initial temperature
    alpha_decay = 0.999
    patience = 0
    max_patience = 200
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        if patience > max_patience:
            break
        
        grad, smooth_max, conv, M = compute_soft_subgradient(a, alpha)
        
        # Project gradient: remove mean to maintain sum constraint (in log space)
        # Actually, for multiplicative updates, we just normalize after the step
        grad_centered = grad - grad.mean()
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break
        
        # Multiplicative update
        log_step = eta * grad_centered / gnorm
        # Clip to avoid extreme values
        log_step = np.clip(log_step, -3.0, 3.0)
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize(a_new, n)
        
        score = fast_conv(a_new).max()
        n_iter += 1
        
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.01, 5.0)
        else:
            patience += 1
            eta *= eta_decay
            alpha *= alpha_decay + 0.0005  # Slowly cool but don't go to 0
            alpha = max(alpha, best_score * 0.005)
        
        if verbose and n_iter % 100 == 0:
            print(f"    MD iter={n_iter}, score={best_score:.10f}, eta={eta:.4f}, alpha={alpha:.6f}")
    
    return best_a, best_score

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    # Build constraint matrix
    # (a_fixed * b)[k] = sum_i a_fixed[k-i]*b[i] (for valid indices)
    # For each k in active_set, this is a linear function of b
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """LP best-response iteration with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, n)
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    stall_count = 0
    n_iters = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv(y)
        M_curr = conv_y.max()
        
        # Active set: peaks above threshold
        progress = min(1.0, t / 500.0)
        tol = 0.05 * (1.0 - progress) + 0.003 * progress
        threshold = M_curr * (1.0 - tol)
        active_set = np.where(conv_y >= threshold)[0]
        n_top = max(5, n // 100)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        if len(active_set) > int(1.2 * n):
            conv_at = conv_y[active_set]
            keep = np.argsort(conv_at)[-int(1.2 * n):]
            active_set = active_set[keep]
        
        b = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        g = b * np.sqrt(2.0*n) / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, n)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, n)
        
        y_prev = y
        y = y_new
        a = a_new
        
        score_y = fast_conv(y).max()
        score_a = fast_conv(a).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        elif score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            stall_count = 0
        
        if verbose and n_iters % 100 == 0:
            print(f"    LP iter={n_iters}, score={best_score:.10f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):

        if no_improve >= max_no_improve:

        conv = fast_conv(a)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            s_others = target - aj
            
            # Try to find better value for a[j]
            # The change in conv when a[j] changes by delta:

            # Plus delta^2 at k=2j if in range
            lo = max(0, j)
            hi = min(2*n-1, j + n)
            if lo >= hi:
                continue
            
            # Current relevant conv values
            relevant_conv = conv[lo:hi]
            relevant_grad = a[lo-j:hi-j] + a[hi-j-n:lo-j-n] ... 
            # This is getting complicated. Let me just try a few values.
            
            # Candidate values
            typical = target / n
            candidates = np.concatenate([

            ])
