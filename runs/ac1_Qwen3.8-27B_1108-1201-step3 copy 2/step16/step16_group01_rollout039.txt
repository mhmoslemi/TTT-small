def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2008
    if v.sum() > 1:
        # Scale down
        u = v / v.sum()
        return np.maximum(u, 0)
    # Better: use the standard algorithm
    mu = np.zeros_like(v)
    sv = np.sort(v)[::-1]
    cumulative = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (cumulative - 1))[0][-1]
    theta = (cumulative[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure sum is exactly 1
    s = w.sum()
    if s < 1e-15:

    cv = np.cumsum(sv)

    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1.0)

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            # Escape: random perturbation
            n_pert = max(5, n // 50)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.02, 0.1)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.1f}s): obj={2*n*best_m:.6f}, stall={stall}")

    return best_a, best_m


def fw_direction(a, beta):
    """
    Frank-Wolfe direction: find the simplex vertex that most reduces
    the smooth objective. Returns the direction (a single basis vector index).
    """
    n = len(a)
    C = fast_conv(a)
    C_max = C.max()
    shifted = beta * (C - C_max)
    shifted = np.clip(shifted, -700, 700)
    exp_C = np.exp(shifted)
    w = exp_C / exp_C.sum()

    # v[i] = sum_k w[k] * a[k-i] = gradient of the smooth approx
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    v = corr[:n]

    # FW direction: move to vertex with minimum v
    j_star = int(np.argmin(v))
    return j_star, v, C


def fw_optimize(a0, time_limit, beta=10.0, verbose=True, rng_seed=55):
    """Frank-Wolfe optimization with smooth max."""
    n = len(a0)
    a = norm_nonneg(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        j_star, v, C = fw_direction(a, beta)
        e = np.zeros(n)
        e[j_star] = 1.0

        # FW step size: 2/(t+2)
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * e
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new
        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.1f}s): obj={2*n*best_m:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent sequence ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # If no parent, create a structured initialization
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        # Asymmetric initialization: slight ramp
        best_seq = np.sin(np.pi * x) ** 2 + 0.3 * x
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smooth GD with annealing (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smooth GD anneal n={n0} ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = sgd_anneal(norm_nonneg(best_seq), t1,
                              beta_start=1.0, beta_end=80.0,
                              eta=10.0/n0, mu=0.88,
                              verbose=True, rng_seed=1)
        try_update(a_out, f"sgd_n{n0}")

    # === Phase 2: Mirror descent (different geometry) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_best} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = mirror_descent(norm_nonneg(best_seq), t2,
                                  c0=4.0, thresh=0.99,
                                  verbose=True, rng_seed=2)
        try_update(a_out, f"md_n{n_best}")

    # === Phase 3: Coordinate descent (fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = coord_descent(norm_nonneg(best_seq), t3,
                                 verbose=True, rng_seed=3)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Smooth GD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SGD (sharper) n={n_best} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = sgd_anneal(norm_nonneg(best_seq), t4,
                              beta_start=5.0, beta_end=200.0,
                              eta=5.0/n_best, mu=0.90,
                              verbose=True, rng_seed=4)
        try_update(a_out, f"sgd_sharp_n{n_best}")

    # === Phase 5: FW optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Frank-Wolfe n={n_best} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_out, _ = fw_optimize(norm_nonneg(best_seq), t5,
                               beta=15.0, verbose=True, rng_seed=5)
        try_update(a_out, f"fw_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 150:
        print(f"\n=== Phase 6: Multi-length ===")
        # Try slightly longer
        n_long = min(n_best + 300, 3000)
        if n_long > n_best and remaining > 100:
            print(f"  Trying n={n_long}")
            a_long = interpolate_seq(best_seq, n_long)
            t6a = min(remaining * 0.25, 120.0)
            if t6a > 25:
                a_out, _ = sgd_anneal(a_long, t6a,
                                      beta_start=1.0, beta_end=60.0,
                                      eta=10.0/n_long, mu=0.88,
                                      verbose=True, rng_seed=6)
                try_update(a_out, f"sgd_long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        # Try slightly shorter
        n_best_now = len(best_seq)
        n_short = max(n_best_now - 300, 1000)
        if remaining > 80 and n_short < n_best_now:
            print(f"  Trying n={n_short}")
            a_short = interpolate_seq(best_seq, n_short)
            t6b = min(remaining * 0.20, 100.0)
            if t6b > 20:
                a_out, _ = sgd_anneal(a_short, t6b,
                                      beta_start=1.0, beta_end=60.0,
                                      eta=10.0/n_short, mu=0.88,
                                      verbose=True, rng_seed=7)
                try_update(a_out, f"sgd_short_n{n_short}")

    # === Phase 7: Final polish - SGD + Coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 15:
        a_out, _ = sgd_anneal(norm_nonneg(best_seq), t7,
                              beta_start=10.0, beta_end=300.0,
                              eta=3.0/n_best, mu=0.92,
                              verbose=True, rng_seed=8)
        try_update(a_out, f"sgd_polish_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7b = min(remaining * 0.40, 120.0)
    if t7b > 10:
        a_out, _ = coord_descent(norm_nonneg(best_seq), t7b,
                                 verbose=True, rng_seed=9)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: MD final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 15:
            a_out, _ = mirror_descent(norm_nonneg(best_seq), t8,
                                      c0=2.0, thresh=0.995,
                                      verbose=True, rng_seed=10)
            try_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
