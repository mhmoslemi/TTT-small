    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_flattening(a0, time_limit, relax=0.5, verbose=True):
    """Gerchberg-Saxton style spectral flattening with non-negativity."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n - 2).bit_length()
    
    iter_count = 0
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count += t
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        
        # Target: flat magnitude (except DC)
        mag = np.abs(A)
        # Average of non-DC components
        avg_mag = mag[1:].mean()
        
        # Relax the magnitude
        target_mag = np.zeros_like(mag)
        target_mag[0] = mag[0]  # Keep DC
        target_mag[1:] = avg_mag
        
        # Relaxation
        new_mag = (1 - relax) * mag + relax * target_mag
        
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Non-negativity projection
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        a_new = normalize(a_new)
        
        a = a_new
        M = fast_autocorr(a).max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        if verbose and (t+1) % 100 == 0:
            print(f"    Spectral iter={t+1}: max={M:.8f}")
    
    return best_a, best_m


def softmax_subgradient(a0, time_limit, top_k=100, beta=10.0, verbose=True, seed=42):
    """Nesterov-accelerated subgradient with softmax weighting on peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall_count = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Softmax weighting on top peaks
        k_max = min(top_k, len(ac))
        sorted_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[sorted_idx]
        # Softmax weights
        w = np.exp(beta * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, k in zip(w, sorted_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step (multiplicative weights)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stuck
        if stall_count > 5000:
            stall_count = 0
            perturbation = rng.normal(0, 0.03, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SoftSGD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    # Pre-build the constraint structure (it changes with y)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix: T[k, j] = y[j] if j <= k <= j+n-1
        # T @ b gives conv(y, b)
        # Row k: columns max(0, k-n+1) to min(n-1, k), values y[col]
        # Build as sparse
        rows = []
        cols = []
        vals = []
        for j in range(n):
            r_start = j
            r_end = j + n
            rows.extend(range(r_start, r_end))
            cols.extend([j] * n)
            vals.extend([y[j]] * n)
        
        T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_descent_peaks(a0, time_limit, verbose=True, seed=0):
    """Coordinate descent targeting the highest autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = np.argmax(ac)
        
        # Find contributing indices for the peak
        k_int = int(peak_idx)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_vals = k_int - idx
        contrib = a[idx] * a[j_vals]
        
        if contrib.max() < 1e-20:
            stall += 1
            break
        
        # Compute sensitivity for each position
        # A position p contributes to peak k if p + q = k for some q
        # sensitivity[p] = sum over top peaks of how much p contributes
        top_peaks = np.argsort(ac)[::-1][:8]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d = pk_int - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]
        
        improved = False
        # Try moving mass from high-contributing positions to low-sensitivity positions
        src_order = np.argsort(contrib)[::-1]
        dst_candidates = np.argsort(sensitivity)[:max(30, n // 20)]
        
        for src_rank in src_order[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
