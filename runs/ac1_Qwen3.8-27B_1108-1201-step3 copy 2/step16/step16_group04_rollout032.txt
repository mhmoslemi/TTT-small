def proj_simplex(z):
    """Project z onto {a >= 0, sum(a) = 1}."""
    n = len(z)
    z_sorted = np.sort(z)[::-1]
    css = np.cumsum(z_sorted)
    rho = np.arange(1, n + 1, dtype=np.float64)

    rho_val = idx[-1] + 1

def compute_lse_gradient(a, beta):

        momentum = (t - 1) / (t + 2)
        y = x + momentum * (x - x_prev)
        y = proj_simplex(y)
        
        # Compute gradient
        g, c, max_c = compute_lse_gradient(y, beta)
        
        # Adaptive step size: based on Lipschitz constant estimate
        w_max = np.exp(beta * (c - c.max())).max() / np.exp(beta * (c - c.max())).sum()
        L_est = 4.0 * beta * w_max * max(1.0, n * 0.5)
        step = 1.0 / L_est
        
        # Backtracking line search (at most 10 iterations)
        f_y = (1.0 / beta) * (np.log(np.exp(beta * (c - c.max())).sum()) + beta * c.max())
        for _ in range(10):
            z = y - step * g
            x_new = proj_simplex(z)
            g_new, c_new, _ = compute_lse_gradient(x_new, beta)
            f_new = (1.0 / beta) * (np.log(np.exp(beta * (c_new - c_new.max())).sum()) + beta * c_new.max())
            if f_new <= f_y + 0.01 * step * np.sum(g * (x_new - y)):
                break
            step *= 0.5
        
        x_prev = y
        x = x_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m


def active_set_polish(a0, time_limit, verbose=True):
    """
    Active-set based polishing: identify active lags, solve KKT-like system.
    Uses iterative peak reduction with exact local adjustments.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for outer in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        M = c.max()
        
        # Find active lags (within 0.1% of max)
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Build the KKT system: find weights w on active lags
        # such that sum_j w_j * a[active_j - i] is constant for i in support(a)
        support = np.where(a > 1e-10)[0]
        n_active = len(active)
        n_support = len(support)
        
        if n_support < 2 or n_active < 1:
            break
        
        # Build matrix: B[i, j] = a[active_j - support[i]] if 0 <= active_j - support[i] < n, else 0
        # We want B @ w = const * 1, sum(w) = 1, w >= 0
        # Use least squares approach
        B = np.zeros((min(n_support, 200), n_active))
        for j_idx in range(n_active):
            k = active[j_idx]
            for i_idx in range(min(n_support, 200)):
                i = support[i_idx]
                diff = k - i
                if 0 <= diff < n:
                    B[i_idx, j_idx] = a[diff]
        
        # Solve: minimize ||B@w - mean(B@w)||^2 subject to sum(w)=1, w>=0
        # Use simple iterative approach: initialize w uniformly, iterate
        w = np.ones(n_active) / n_active
        for inner_w in range(50):
            Bw = B @ w
            target = Bw[support[:min(n_support, 200)].size]
            target_mean = target.mean()
            # Gradient of ||Bw - target_mean * 1||^2 wrt w
            grad_w = 2.0 * B.T @ (Bw - target_mean)
            # Projection: w = proj_simplex(w - eta * grad_w) but with sum(w)=1
            eta = 0.1
            w_new = w - eta * grad_w
            w_new = proj_simplex(w_new)
            if np.max(np.abs(w_new - w)) < 1e-10:
                w = w_new
                break
            w = w_new
        
        # Compute the direction: d = -sum_j w_j * grad(c[active_j])
        # grad(c[k])[i] = 2*a[k-i] for valid k-i
        d = np.zeros(n)
        for j_idx in range(n_active):
            k = active[j_idx]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            d[i_start:i_end + 1] -= 2.0 * w[j_idx] * a[k - np.arange(i_start, i_end + 1)]
        
        # Normalize direction to have zero sum (maintain sum(a)=1)
        d -= d.mean()
        
        # Line search
        best_t = 0.0
        best_trial_m = M
        for t in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            a_trial = a + t * d
            a_trial = proj_simplex(a_trial)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_trial_m - 1e-15:
                best_trial_m = M_trial
                best_t = t
        
        if best_t > 0:
            a = a + best_t * d
            a = proj_simplex(a)
            if best_trial_m < best_m:
                best_m = best_trial_m
                best_a = a.copy()
        
        if verbose and outer % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ActiveSet iter={outer} ({elapsed:.1f}s): J={J:.6f}, n_active={n_active}")
    
    return best_a, best_m


def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def accelerated_lp_fw(a0, time_limit, solver, top_k_init=60, top_k_max=200, verbose=True):
    """Accelerated LP Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 2.5) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        
        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue
        
        # Strong step size (backtracking)
        alpha = 2.0 / (t + 2.0)
        for _ in range(3):
            x_new = (1.0 - alpha) * y + alpha * direction
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y.copy()
            else:
                x_new = x_new / sx
            mx_new = fast_autocorr(x_new).max()
            if mx_new <= M_curr:
                break
            alpha *= 0.5
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if t % 50 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:min(15, len(c))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            chosen = candidates[rng.permutation(len(candidates))[:min(100, len(candidates))]]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    if 0 <= d1 < n:
                        sens += a[d1]
                    d2 = dst - k2i
                    if 0 <= d2 < n and d2 != d1:
                        sens += a[d2]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and not improved:
                pass
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization strategies:
    1. LP-FW for fast initial convergence
    2. FISTA with full-gradient log-sum-exp for smooth polishing
    3. Active-set KKT for final micro-optimization
    4. Coordinate descent for local adjustments
    5. Multi-length exploration
    """
    rng = np.random.default_rng(123)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # === Spectral pre-conditioning ===
    a_spec = spectral_equalize(best_seq, n_iter=15)
    try_update(a_spec, "spectral")
    
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 1: LP-FW (main workhorse, 35% of time) ===
    print(f"\n=== Phase 1: LP-FW n={n0}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_in = normalize(best_seq)
        a_out, _ = accelerated_lp_fw(a_in, t1, solver, top_k_init=60, top_k_max=180, verbose=True)
        try_update(a_out, f"lpfw_n{n0}")
    
    # === Phase 2: FISTA with log-sum-exp (30% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FISTA-LSE n={n_best}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = fista_lse(a_in, t2, beta_start=20.0, beta_end=300.0, n_iter_max=100000, verbose=True)
        try_update(a_out, f"fista_n{n_best}")
    
    # === Phase 3: Active-set polishing (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: ActiveSet n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = normalize(best_seq)
        a_out, _ = active_set_polish(a_in, t3, verbose=True)
        try_update(a_out, f"active_n{n_best}")
    
    # === Phase 4: LP-FW refined (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW refined n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 20:
        a_in = normalize(best_seq)
        a_out, _ = accelerated_lp_fw(a_in, t4, solver, top_k_init=80, top_k_max=250, verbose=True)
        try_update(a_out, f"lpfw2_n{n_best}")
    
    # === Phase 5: FISTA second pass (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FISTA second n={n_best}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = fista_lse(a_in, t5, beta_start=100.0, beta_end=800.0, n_iter_max=80000, verbose=True)
        try_update(a_out, f"fista2_n{n_best}")
    
    # === Phase 6: Coordinate polish (5% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 6: Coord n={n_best} ===")
        t6 = min(remaining * 0.05, 100.0)
        if t6 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=42)
            try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 7: Try alternative length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for n_alt in [n_best + 300, n_best - 300, n_best + 600]:
        if n_alt < 500 or n_alt > 4000 or remaining < 120:
            continue
        if n_alt == n_best:
            continue
        
        print(f"\n=== Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        a_alt = spectral_equalize(a_alt, n_iter=10)
        
        t_alt1 = min(remaining * 0.35, 200.0)
        if t_alt1 > 30:
            a_out, _ = accelerated_lp_fw(a_alt, t_alt1, solver, top_k_init=50, top_k_max=150, verbose=True)
            try_update(a_out, f"lpfw_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_alt2 = min(remaining * 0.3, 150.0)
            if t_alt2 > 20:
                a_in = normalize(best_seq)
                a_out, _ = fista_lse(a_in, t_alt2, beta_start=50.0, beta_end=400.0, n_iter_max=60000, verbose=True)
                try_update(a_out, f"fista_n{n_best}_alt")
        
        remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final Coord n={n_best} ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Final: one more FISTA with very high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t_final = min(remaining * 0.7, 100.0)
        if t_final > 10:
            print(f"\n=== Final FISTA beta=1000+ n={n_best} ===")
            a_in = normalize(best_seq)
            a_out, _ = fista_lse(a_in, t_final, beta_start=500.0, beta_end=2000.0, n_iter_max=50000, verbose=True)
            try_update(a_out, f"fista_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
