def fast_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def soft_subgradient_exp(a, ac, beta=8.0, min_weight=1e-6):
    """
    Exponential soft subgradient: weights each lag by exp(beta*(ac[k]-M)/(M+eps)).
    This gives a smooth gradient that emphasizes near-max lags.
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Exponential weighting
    weights = np.exp(beta * (ac - M) / (M + 1e-30))
    # Zero out very small weights for efficiency
    weights[weights < min_weight] = 0.0
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum
    
    # Compute subgradient
    g = np.zeros(n)
    active_idx = np.where(weights > 0.0)[0]
    for k in active_idx:
        k_int = int(k)
        w = weights[k_int]
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * w * a[j_idx]
    
    return g

def hard_subgradient(a, ac, n_active=10):
    """Hard subgradient from top n_active lags."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    top_idx = np.argsort(ac)[::-1][:n_active]
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    # Normalize
    gn = np.linalg.norm(g)
    if gn > 1e-15:
        g = g / gn
    return g

def log_space_update(a, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_extrapolate(x, x_prev, t):
    """Nesterov momentum extrapolation."""
    beta_t = (t - 1.0) / (t + 1.5)
    y = x + beta_t * (x - x_prev)
    y = np.maximum(y, 1e-15)
    sy = y.sum()
    if sy < 1e-15:
        return x.copy()
    return y / sy

                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def subspace_refine(a0, time_limit, verbose=True, rng_seed=77):
    """
    Refine in a low-dimensional subspace defined by the top gradient directions.
    This helps navigate the landscape more effectively than full-dimensional steps.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Build a small basis from past gradients
    basis_list = []
    for _ in range(20):
        ac_curr = fast_conv(a)
        g = soft_subgradient_exp(a, ac_curr, beta=6.0)
        gn = np.linalg.norm(g)
        if gn > 1e-10:
            g = g / gn
            # Orthogonalize
            for b in basis_list:
                g = g - np.dot(g, b) * b
            gn = np.linalg.norm(g)
            if gn > 0.1:
                basis_list.append(g / gn)
            # Take a small step
            a_new = a - 0.01 * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
            m = fast_conv(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    if len(basis_list) < 3:
        return best_a, best_m
    
    # Now do local optimization in the subspace
    K = min(len(basis_list), 10)
    basis = np.array(basis_list[:K])
    
    # Project current onto subspace and optimize coefficients
    for _ in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random direction in subspace
        coeffs = rng.normal(0, 0.05, K)
        direction = basis.T @ coeffs
        direction = direction - np.mean(direction)  # Keep sum roughly constant
        
        for step in [0.5, 1.0, 2.0, 0.1]:
            a_trial = a + step * direction
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            m = fast_conv(a_trial).max()
            if m < best_m:
                best_m = m
                best_a = a_trial.copy()
                a = a_trial
                break
            elif m < fast_conv(a).max():
                a = a_trial
                break
    
    return best_a, best_m

def parametric_init(n, rng):
    """
    Generate diverse parametric initializations.
    """
    candidates = []
    
    # Beta distribution shapes
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
            x = np.linspace(0.01, 0.99, n)
            f = x ** (alpha - 1) * (1 - x) ** (beta_p - 1)
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 0:
                candidates.append(f / s)
    
    # Symmetric U-shapes
    x = np.linspace(0, 1, n)
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f = np.minimum(x, 1-x) ** p
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            candidates.append(f / s)
    
    # Cosine window variants
    for a_param in [0.5, 1.0, 1.5, 2.0]:
        f = np.cos(np.pi * a_param * (x - 0.5)) ** 2
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            candidates.append(f / s)
    
    # Two-peak shapes
    for center_ratio in [0.25, 0.35, 0.45]:
        for width in [0.05, 0.1, 0.15, 0.2]:
            peak1 = np.exp(-((x - center_ratio) ** 2) / (2 * width ** 2))
            peak2 = np.exp(-((x - (1 - center_ratio)) ** 2) / (2 * width ** 2))
            f = peak1 + peak2
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-10:
                candidates.append(f / s)
    
    return candidates

def coarse_to_fine(a_final_target, n_target, time_budget, verbose=True):
    """
    Multi-scale optimization: optimize at increasing resolutions.
    """
    t0 = time.monotonic()
    scales = [100, 200, 400, 800, n_target]
    scales = [s for s in scales if s <= n_target]
    if scales[-1] != n_target:
        scales.append(n_target)
    
    a_current = interpolate_seq(a_final_target, scales[0])
    
    for i, n_scale in enumerate(scales):
        if n_scale != scales[0]:
            a_current = interpolate_seq(a_current, n_scale)
        
        # Time allocation: more time for larger scales
        frac = (n_scale ** 2) / sum(s ** 2 for s in scales)
        t_alloc = time_budget * frac * 0.8
        t_alloc = max(t_alloc, 2.0)
        
        remaining = time_budget - (time.monotonic() - t0)
        t_alloc = min(t_alloc, remaining * 0.9)
        
        if t_alloc < 1.0:
            break
        
        if verbose:
            print(f"  Coarse-to-fine: n={n_scale}, t_alloc={t_alloc:.1f}s")
        
        a_current, _ = amd_optimize(a_current, t_alloc, c0=3.0, beta_soft=6.0,
                                     verbose=False, rng_seed=100 + i)
    
    return interpolate_seq(a_current, n_target)

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with improved AMD + coordinate descent + 
    informed restarts + multi-scale approach.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    rng = np.random.default_rng(42)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent or create initial ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        # Parametric initialization sweep
        n_base = 2400
        candidates = parametric_init(n_base, rng)
        best_cand = None
        best_cand_score = float('inf')
        for i, cand in enumerate(candidates):
            sc = evaluate_safe(cand)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand
            if i % 20 == 0:
                print(f"  Init sweep: {i}/{len(candidates)}, best={best_cand_score:.4f}")
        
        if best_cand is not None:
            best_seq = best_cand
            best_score = best_cand_score
        else:
            best_seq = np.ones(n_base) / n_base
            best_score = evaluate_safe(best_seq)
        print(f"Initial: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Quick AMD to establish baseline ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD baseline n={n0} ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 15:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t1, c0=3.0, beta_soft=8.0,
                                use_soft=True, verbose=True, rng_seed=1)
        try_update(a_out, "amd_baseline")
    
    # === Phase 2: Multi-scale refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 200:
        print(f"\n=== Phase 2: Coarse-to-fine n={n_best} ===")
        t2 = min(remaining * 0.15, 250.0)
        if t2 > 30:
            a_out = coarse_to_fine(norm_sum1(best_seq), n_best, t2, verbose=True)
            try_update(a_out, "coarse_fine")
    
    # === Phase 3: Extended AMD with soft subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD extended n={n_best} ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t3, c0=4.0, beta_soft=12.0,
                                use_soft=True, verbose=True, rng_seed=2,
                                metropolis=True, T0=0.002)
        try_update(a_out, "amd_extended")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent n={n_best} ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = coordinate_descent_peak(norm_sum1(best_seq), t4, verbose=True, rng_seed=3)
        try_update(a_out, "coord_peak")
    
    # === Phase 5: AMD with hard subgradient (different view) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD hard n={n_best} ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_out, _ = amd_optimize(norm_sum1(best_seq), t5, c0=5.0, beta_soft=20.0,
                                use_soft=False, verbose=True, rng_seed=4)
        try_update(a_out, "amd_hard")
    
    # === Phase 6: Subspace refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Subspace n={n_best} ===")
        t6 = min(remaining * 0.08, 120.0)
        if t6 > 15:
            a_out, _ = subspace_refine(norm_sum1(best_seq), t6, verbose=True, rng_seed=5)
            try_update(a_out, "subspace")
    
    # === Phase 7: Informed restarts (multiple) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 7: Informed restarts n={n_best} ===")
        n_restarts = 3
        t_per_restart = min(remaining / (n_restarts + 1), 150.0)
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            # Structured restart
            a_start = best_seq.copy()
            ac = fast_conv(a_start)
            peak_lags = np.argsort(ac)[::-1][:3]
            
            # Perturb coordinates at peak lags
            n_perturb = max(10, n_best // 100)
            hot = set()
            for k in peak_lags:
                k_int = int(k)
                i_start = max(0, k_int - n_best + 1)
                i_end = min(n_best, k_int + 1)
                for i in range(i_start, i_end):
                    hot.add(i)
                    if 0 <= k_int - i < n_best:
                        hot.add(k_int - i)
            
            hot_list = list(hot)
            if len(hot_list) > 0:
                perturb_idx = rng.choice(hot_list, size=min(n_perturb, len(hot_list)), replace=False)
                for i in perturb_idx:
                    a_start[i] *= (1.0 + rng.normal(0, 0.05))
            
            a_start = np.maximum(a_start, 1e-10)
            a_start = a_start / a_start.sum()
            
            # Brief AMD from restarted point
            a_out, _ = amd_optimize(a_start, t_per_restart, c0=3.0, beta_soft=10.0,
                                    use_soft=True, verbose=False, rng_seed=10 + r,
                                    metropolis=True, T0=0.003)
            try_update(a_out, f"restart_{r+1}")
            
            # Quick coord polish
            remaining_sub = min(30.0, time_budget - (time.monotonic() - t_global))
            if remaining_sub > 5:
                a_out2, _ = coordinate_descent_peak(a_out, remaining_sub, verbose=False, rng_seed=20 + r)
                try_update(a_out2, f"restart_{r+1}_coord")
    
    # === Phase 8: Final AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 10:
            a_out, _ = amd_optimize(norm_sum1(best_seq), t8, c0=2.0, beta_soft=15.0,
                                    use_soft=True, verbose=False, rng_seed=6)
            try_update(a_out, "final_amd")
    
    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_out, _ = coordinate_descent_peak(norm_sum1(best_seq), t9, verbose=False, rng_seed=7)
            try_update(a_out, "final_coord")
    
    # === Phase 10: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 10: Length exploration ===")
        for n_try in [n_best + 200, n_best - 200, int(n_best * 1.1), int(n_best * 0.9)]:
            if time.monotonic() - t_global > time_budget - 40:
                break
            n_try = max(500, min(n_try, 3500))
            if n_try == n_best:
                continue
            a_try = interpolate_seq(best_seq, n_try)
            t_try = min(40.0, remaining * 0.25)
            if t_try < 10:
                continue
            a_out, _ = amd_optimize(a_try, t_try, c0=3.0, beta_soft=8.0,
                                    use_soft=True, verbose=False, rng_seed=30)
            try_update(a_out, f"len_{n_try}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
