    n = len(a)

    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    rho = rho[-1]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with multi-peak gradient targeting
2. Spectral domain reshaping for diversity and global structure
3. Asymmetric polynomial parameterization for coarse shape control

5. Multi-restart with diverse initializations

import numpy as np
import time


def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def grad_multi_peak(a, lags, weights=None):
    """
    Compute weighted sum of gradients of autocorrelation at given lags.
    grad c_k / da[m] = 2*a[k-m] if 0<=k-m<n else 0
    """
    n = len(a)
    grad = np.zeros(n)
    if weights is None:
        weights = np.ones(len(lags))
    for w, k in zip(weights, lags):
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            grad[ms] += 2.0 * w * a[k - ms]
    return grad


def nesterov_fw_multipeak(a0, time_limit, K=8, verbose=True, seed=42):
    """
    Nesterov-accelerated Frank-Wolfe targeting top-K autocorrelation peaks.
    The FW direction is the simplex vertex (standard basis vector) at the
    coordinate with minimum gradient component.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_mom = a.copy()
    stall = 0
    max_stall = 5000
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        
        # Top K peaks
        top_idx = np.argsort(ac)[::-1][:K]
        top_vals = ac[top_idx]
        
        # Weighted gradient
        if top_vals[0] < 1e-15:
            break
        w = top_vals / top_vals.sum()
        g = grad_multi_peak(x, top_idx, w)
        
        # FW direction: vertex with min gradient
        min_idx = int(np.argmin(g))
        
        # Nesterov acceleration
        alpha_t = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha_t) * x + alpha_t * 0.0  # base is 0
        # Actually FW: move towards e_min
        # x_new = (1-alpha)*x + alpha*e_min
        # But we need to project to simplex
        
        e_min = np.zeros(n)
        e_min[min_idx] = 1.0
        x_new = (1.0 - alpha_t) * x + alpha_t * e_min
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s_new
        
        # Momentum
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom_new = x_new + beta_t * (x_new - x)
        x_mom_new = np.maximum(x_mom_new, 0.0)
        s_mom = x_mom_new.sum()
        if s_mom < 1e-15:
            x_mom_new = x_new.copy()
        else:
            x_mom_new = x_mom_new / s_mom
        
        x = x_new
        
        # Track best
        for cand in [x, x_mom_new]:
            mc = fast_autocorr(cand).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = cand.copy()
                stall = 0
            else:
                stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            # Perturb best
            noise = rng.normal(0, 0.01, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_mom = x_re.copy()
        
        if verbose and t % 10000 == 0:
            J = 2.0 * n * best_m
            print(f"    NFW t={t}: J={J:.8f}, stall={stall}")
    
    return best_a, best_m
