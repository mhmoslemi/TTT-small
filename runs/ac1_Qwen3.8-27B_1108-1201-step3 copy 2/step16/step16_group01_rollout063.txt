import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    if s < 1e-15:

    return a / s

    Compute gradient of logsumexp(beta * conv(a,a)) at a, given softmax weights w.

def smooth_fista(a0, time_limit, beta_start=5.0, beta_end=300.0, n_phases=4, verbose=True, rng_seed=42):
    """
    Smooth FISTA with geometric beta annealing.
    """
    n = len(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Geometric annealing
        frac = (phase + 1) / n_phases
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Estimate Lipschitz constant
        # grad of logsumexp(beta*c) has Lipschitz ~ O(beta * n)
        # Be conservative
        L_est = 8.0 * beta * n
        step = 1.0 / L_est
        
        # FISTA state
        x = a.copy()
        y = a.copy()
        y_prev = a.copy()
        t_fc = 1.0
        
        phase_iters = 0
        stall = 0
        max_stall = 500
        
        while time.monotonic() - t0 < time_limit:
            phase_iters += 1
            
            # Compute smooth gradient at y
            c_y = fast_autocorr(y)
            beta_c = beta * c_y
            beta_c_shifted = beta_c - beta_c.max()
            exp_vals = np.exp(beta_c_shifted)
            w = exp_vals / exp_vals.sum()
            
            g = compute_grad_smooth(y, w, n, fft_size)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fc ** 2)) / 2.0
            momentum = (t_fc - 1.0) / t_new
            
            x_trial = project_simplex(y - step * g)
            
            # Backtracking: check if smooth objective decreased
            # (simplified: just accept and adapt step if stuck)
            
            # Evaluate exact objective
            c_x = fast_autocorr(x_trial)
            M = c_x.max()
            obj = 2.0 * n * M
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = x_trial.copy()
                stall = 0
            else:
                stall += 1
            
            # Update FISTA state
            y_new = x_trial + momentum * (x_trial - x)
            y_new = project_simplex(y_new)
            
            y_prev = y
            y = y_new
            x = x_trial
            t_fc = t_new
            
            # Adaptive step
            if stall > max_stall:
                step *= 0.5
                stall = 0
                # Restart from best
                if rng.random() < 0.3:
                    noise = rng.normal(0, 0.005, n) * best_a
                    x = norm_sum1(best_a + noise)
                    y = x.copy()
                    y_prev = x.copy()
                    t_fc = 1.0
            
            # Increase step if improving consistently
            if stall == 0 and phase_iters % 100 == 0:
                step *= 1.05  # small increase
            
            if verbose and phase_iters % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Phase {phase+1} (beta={beta:.1f}) iter={phase_iters} ({elapsed:.1f}s): obj={best_obj:.6f}, step={step:.2e}")
        
        # End of phase: restart from best
        a = best_a.copy()
        if verbose:
            print(f"    Phase {phase+1} done: obj={best_obj:.6f}, iters={phase_iters}")
    
    return best_a, best_obj

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    stall = 0
    max_stall = 200
    beta = 50.0  # fixed beta for sensitivity computation
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M = c.max()

        beta_c = beta * c

        w = np.exp(beta_c_shifted)

        # Positions with low gradient should gain mass

        dst_idx = np.argmin(g)

            stall += 1
            break

            a_trial[dst_idx] += amt
            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-15:

            c_trial = fast_autocorr(a_trial)
            M_trial = c_trial.max()
            if M_trial < M - 1e-15:
                a = a_trial
                best_obj = 2.0 * n * M_trial
                best_a = a.copy()
                moved = True
                stall = 0
                M = M_trial
                break

                        a = a_trial
                        best_obj = 2.0 * n * M_trial

                        moved = True

                    M_new = fast_autocorr(a).max()

                        best_a = a.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord done ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_grad_smooth(a, w, n, fft_size):
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    R = np.fft.irfft(W * np.conj(A), fft_size)
    return 4.0 * n * R[:n]


def logsumexp_stable(x):
    x_max = x.max()
    if x_max > 700:
        return x_max + np.log(np.exp(x - x_max).sum())
    return np.log(np.exp(x).sum()) + x_max


def smooth_fista(a0, time_limit, beta_start=3.0, beta_end=200.0, n_phases=5,

def coord_descent(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    beta = 30.0
    c2n = 2 * n - 1
    fft_size = 1 << (c2n + n - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M = c.max()

        bc = beta * c
        bc -= bc.max()
        w = np.exp(bc)
        w /= w.sum()

        g = compute_grad_smooth(a, w, n, fft_size)

        top_src = np.argsort(g)[::-1][:50]
        top_dst = np.argsort(g)[:50]

        improved = False
        for src in top_src[:30]:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            if a[src] < 1e-14:
                continue
            for dst in top_dst[:30]:
                if dst == src or a[dst] > a[src]:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s = a_t.sum()
                    if s < 1e-15:
                        continue
                    a_t /= s
                    M_t = fast_autocorr(a_t).max()
                    if M_t < M - 1e-14:
                        a = a_t
                        best_obj = 2.0 * n * M_t
                        best_a = a.copy()
                        improved = True
                        M = M_t
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random escape
            n_pert = max(3, n // 300)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    fc = rng.uniform(0.01, 0.04)
                    amt = a[i] * fc
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a /= a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_obj / (2.0 * n):
                    best_obj = 2.0 * n * M_new
                    best_a = a.copy()
                    stall = max(0, stall - 50)

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s) obj={best_obj:.6f} stall={stall}")

    if verbose:
        print(f"    CD done: obj={best_obj:.6f}")
    return best_a, best_obj


def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def make_initialization(n, style="sin_cos", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if style == "sin_cos":
        a = np.sin(np.pi * x) ** 2 + 0.15 * np.cos(2 * np.pi * x) ** 2
    elif style == "gaussian":
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif style == "two_peak":
        a = np.sin(2 * np.pi * x) ** 2 + 0.1
    elif style == "power":
        center = (n - 1) / 2.0
        a = (1.0 + np.abs(np.arange(n) - center) / (n / 2.0)) ** (-1.5)
    elif style == "uniform":
        a = np.ones(n) + 0.01 * rng.normal(0, 1, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
