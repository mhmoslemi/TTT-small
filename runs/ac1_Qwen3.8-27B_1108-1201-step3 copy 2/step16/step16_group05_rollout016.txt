def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to simplex (sum=1, non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        beta_t = (t - 1.0) / (t + 1.5)

        if M_new < best_m - 1e-18:

            stall += 1
        
        # Adaptive epsilon: tighten over time

        # Build constraints: for each selected lag k, sum_i a_i * b_{k-i} <= M_curr
        # i.e., (a * b)_k <= M_curr
        constraints = []
        for k in top_lags:
            k_int = int(k)
            # (a*b)_k = sum_i a_i * b_{k-i}
            # Valid i: max(0, k-n+1) <= i <= min(k, n-1)
            i_start = max(0, k_int - n + 1)
            i_end = min(k_int, n - 1)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            # b_{k-i} for i in idx: indices are k_int - idx
            b_idx = k_int - idx
            expr = cp.sum(cp.multiply(a[i_start:i_end+1], b_var[b_idx]))
            constraints.append(expr <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        
        y = y_new
        
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-16:
                best_m = mc
                best_a = candidate.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.8f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      LP-FW done ({elapsed:.0f}s): J={J:.8f}")
    
    return best_a, best_m


def active_set_refine(a0, time_limit, solver, max_active=15, verbose=True):
    """
    Active-set method: iteratively identify critical lags and solve QP
    to minimize the max over those laged subject to simplex constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Identify active set: lags within 0.1% of max
        threshold = M_curr * 0.999
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_active:
            active_lags = np.argsort(ac)[::-1][:max_active]
        
        # Solve: minimize t s.t. b_k(a) <= t for k in active_lags, sum(a)=1, a>=0
        # This is a QCQP. For small active set, should be fast.
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        for k in active_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(k_int, n - 1)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            # b_k = sum_i a_i * a_{k-i}
            # Build quadratic form: this is (sum_i a[i]*a[k-i]) <= t
            # We can write this as a quadratic expression
            # b_k = sum_{i in valid} a_i * a_{k-i}
            # This is not a simple quadratic form in cvxpy because the indices are shifted
            # Let's build it as sum of products
            quad_expr = 0
            counted = set()
            for i in range(i_start, i_end + 1):
                j = k_int - i
                pair = (min(i, j), max(i, j))
                if pair in counted:
                    continue
                counted.add(pair)
                if i == j:
                    quad_expr += a_var[i] * a_var[i]
                else:
                    quad_expr += 2 * a_var[i] * a_var[j]
            
            constraints.append(quad_expr <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if verbose:
                print(f"      AS: solve error at outer={outer}")
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if verbose:
                print(f"      AS: status={prob.status} at outer={outer}")
            break
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Check convergence
            if abs(M_new - M_curr) < 1e-12:
                break
            a = a_new
        
        if verbose and outer % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      AS outer={outer} ({elapsed:.0f}s): J={J:.8f}, active={len(active_lags)}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      AS done ({elapsed:.0f}s): J={J:.8f}")
    
    return best_a, best_m


def peak_shave(a0, time_limit, verbose=True, seed=77):
    """
    Direct peak shaving: identify coordinates contributing most to peak lags
    and redistribute mass to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(30, len(ac))]
        
        # For each top lag, compute contribution of each coordinate
        # b_k = sum_i a_i * a_{k-i}, so coordinate i contributes a_i * a_{k-i} to lag k
        impact = np.zeros(n)
        for k in top_lags[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(k_int, n - 1)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = k_int - idx
            # Contribution of coordinate i to lag k is a_i * a_{k-i}
            # If we reduce a_i by delta, the change in b_k is approximately -delta * a_{k-i}
            # (ignoring second-order and effects on other lags)
            impact[idx] += a[partner]  # sensitivity: how much b_k changes per unit change in a_i
        
        # Find high-impact source and low-impact destination
        src_rank = np.argsort(impact)[::-1]
        dst_rank = np.argsort(impact)
        
        improved = False
        for src in src_rank[:50]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            # Try moving to low-impact destinations
            for dst in dst_rank[:50]:
                if dst == src:
                    continue
                # Try a small transfer
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if n > 10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PS ({elapsed:.0f}s): J={J:.8f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      PS done ({elapsed:.0f}s): J={J:.8f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_initializations(n, best_seq):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    inits = []
    
    # Best sequence (if provided)
    if best_seq is not None and len(best_seq) > 0:
        inits.append(norm_sum1(best_seq))
    
    # Raised cosine
    rc = (1 + np.cos(2 * np.pi * x)) / 2
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())
    
    # Hanning window
    hw = 0.5 * (1 - np.cos(2 * np.pi * x))
    hw = np.maximum(hw, 1e-10)
    inits.append(hw / hw.sum())
    
    # Sine squared
    ss_ = np.sin(np.pi * x) ** 2
    ss_ = np.maximum(ss_, 1e-10)
    inits.append(ss_ / ss_.sum())
    
    # Broadened Gaussian (wider than natural)
    sigma = 0.4
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())
    
    # Very broad Gaussian
    sigma2 = 0.45
    g2 = np.exp(-0.5 * ((x - 0.5) / sigma2) ** 2)
    inits.append(g2 / g2.sum())
    
    # Bimodal
    bim = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2)
    inits.append(bim / bim.sum())
    
    # Uniform (baseline)
    inits.append(np.ones(n) / n)
    
    # Blend with best
    if best_seq is not None and len(best_seq) > 0:
        b_norm = norm_sum1(interpolate_sequence(best_seq, n)) if len(best_seq) != n else norm_sum1(best_seq)
        blend = 0.5 * b_norm + 0.5 * inits[1]
        inits.append(norm_sum1(blend))
    
    return inits


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    rng = np.random.default_rng(2024)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    def remaining_time():
        return time_budget - (time.monotonic() - t_global)
    
    # === Phase 0: Setup initial best ===
    if parent is not None:
        n0 = len(parent)
        a_init = norm_sum1(parent)
        p_score = evaluate_safe(a_init)
        print(f"Parent: n={n0}, score={p_score:.8f}")
        best_seq = a_init.copy()
        best_score = p_score
    else:
        n0 = 2400
        x = np.linspace(0, 1, n0)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.8f}")
    
    # === Phase 1: Nesterov soft-max subgradient at parent length ===
    rem = remaining_time()
    print(f"\n=== Phase 1: NSM n={n0} ===")
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_soft_max(
            a_in, t1, epsilon_frac=0.01, top_k_cap=30,
            c0=2.5, verbose=True, seed=42)
        try_update(a_out, f"nsm_n{n0}")
    
    # === Phase 2: LP Frank-Wolfe ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW n={n_cur} ===")
    t2 = min(rem * 0.20, 300.0)
    if t2 > 60 and n_cur <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, top_constraints=80, verbose=True)
        try_update(a_out, f"lpfw_n{n_cur}")
    
    # === Phase 3: Active-set QP refine ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Active-set n={n_cur} ===")
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30 and n_cur <= 2500:
        a_in = norm_sum1(best_seq)
        a_out, _ = active_set_refine(a_in, t3, solver, max_active=12, verbose=True)
        try_update(a_out, f"active_n{n_cur}")
    
    # === Phase 4: NSM polish with tighter epsilon ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: NSM polish n={n_cur} ===")
    t4 = min(rem * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_soft_max(
            a_in, t4, epsilon_frac=0.003, top_k_cap=20,
            c0=3.0, verbose=True, seed=77)
        try_update(a_out, f"nsm_polish_n{n_cur}")
    
    # === Phase 5: Peak shaving ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Peak shave n={n_cur} ===")
    t5 = min(rem * 0.10, 120.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shave(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"ps_n{n_cur}")
    
    # === Phase 6: Try longer length ===
    rem = remaining_time()
    n_best = len(best_seq)
    n1 = min(n_best + 400, 2800)
    if n1 > n_best and rem > 150:
        print(f"\n=== Phase 6: Longer n={n1} ===")
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        t6 = min(rem * 0.20, 200.0)
        if t6 > 60:
            a_out, _ = nesterov_soft_max(
                a_in, t6, epsilon_frac=0.01, top_k_cap=30,
                c0=2.5, verbose=True, seed=99)
            try_update(a_out, f"nsm_n{n1}")
            
            # LP at longer length
            rem = remaining_time()
            if rem > 80 and n1 <= 2800:
                t6b = min(rem * 0.3, 120.0)
                a_out2, _ = lp_frank_wolfe(a_out, t6b, solver, top_constraints=80, verbose=True)
                try_update(a_out2, f"lpfw_n{n1}")
    
    # === Phase 7: Try different initializations at best length ===
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 100:
        print(f"\n=== Phase 7: Diverse inits n={n_best} ===")
        inits = get_initializations(n_best, best_seq)
        # Remove duplicate (the best seq is already first)
        unique_inits = [inits[0]]
        for init in inits[1:]:
            if np.abs(init - inits[0]).sum() > 0.1:
                unique_inits.append(init)
        
        t_per = min(rem * 0.25, 60.0)
        for idx, init in enumerate(unique_inits[1:6]):
            rem = remaining_time()
            if rem < 40:
                break
            t7 = min(t_per, rem - 20)
            if t7 > 20:
                a_out, _ = nesterov_soft_max(
                    init, t7, epsilon_frac=0.01, top_k_cap=25,
                    c0=2.5, verbose=False, seed=200 + idx)
                try_update(a_out, f"init{idx}_n{n_best}")
    
    # === Phase 8: Final LP polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP n={n_best} ===")
    if n_best <= 2500 and rem > 60:
        t8 = min(rem * 0.30, 150.0)
        if t8 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, top_constraints=60, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 9: Final peak shave ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final PS n={n_best} ===")
    if rem > 30:
        t9 = min(rem * 0.5, 80.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shave(a_in, t9, verbose=True, seed=314)
            try_update(a_out, f"ps_final_n{n_best}")
    
    # === Phase 10: One more NSM with very tight epsilon ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 10: Tight NSM n={n_best} ===")
    if rem > 40:
        t10 = min(rem * 0.6, 100.0)
        if t10 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_soft_max(
                a_in, t10, epsilon_frac=0.001, top_k_cap=15,
                c0=4.0, verbose=True, seed=555)
            try_update(a_out, f"tight_nsm_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
