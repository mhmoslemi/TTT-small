import numpy as np

from scipy import sparse

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_exp_gradient(a, ac, beta, top_k=80):
    """Compute gradient of smoothed max autocorrelation using exponential weights."""
    n = len(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    shifted = beta * (ac - ac_max)
    shifted = np.maximum(shifted, -60.0)
    weights = np.exp(shifted)
    
    # Only use top_k most significant lags
    k_total = len(weights)
    nk = min(top_k, k_total)
    top_idx = np.argsort(weights)[::-1][:nk]
    w_sum = weights[top_idx].sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for wi, k in zip(weights[top_idx], top_idx):
        w = wi / w_sum
        if w < 1e-12:
            continue
        k_int = int(k)
        # First part: a[k-m] for valid m
        m_start = max(0, k_int - n + 1)
        m_end = min(n - 1, k_int)
        if m_start <= m_end:
            grad[m_start:m_end + 1] += w * a[k_int - np.arange(m_start, m_end + 1)]
        # Second part: a[m-k] for valid m
        m_start2 = max(k_int, 0)
        m_end2 = min(n - 1, k_int + n - 1)
        if m_start2 <= m_end2:
            grad[m_start2:m_end2 + 1] += w * a[np.arange(m_start2, m_end2 + 1) - k_int]
    return grad


def nmd_optimize(a0, time_limit, beta_init=20.0, beta_min=3.0, c0=5.0,
                 top_k=60, restart_every=6000, verbose=True, rng_seed=42):
    """Nesterov mirror descent with exponential-weight gradient and adaptive beta."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    cur_top_k = top_k
    stall_count = 0
    best_beta = beta_init

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_exp_gradient(y, ac, beta=cur_beta, top_k=cur_top_k)

        # Adaptive step size
        eta = c0 / (np.sqrt(t) + 1.0)
        # Scale gradient by current max to make step size more meaningful
        ac_max = ac.max()
        if ac_max > 1e-15:
            g_scaled = g / ac_max
        else:
            g_scaled = g

        log_a = np.log(y) - eta * g_scaled
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            best_beta = cur_beta
        else:
            stall_count += 1

        # Adaptive beta: decrease if stalling (smoother gradient for exploration)
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adjust parameters
            cur_beta = max(cur_beta * 0.85, beta_min)
            cur_top_k = min(cur_top_k + 10, 120)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    NMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, tk={cur_top_k}")

        # Gradually decrease beta for smoother gradient
        if t % 2000 == 0:
            cur_beta = max(cur_beta * 0.98, beta_min)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, K_init=80, K_max=400,
                   K_inc_every=30, nesterov=True):
    """LP-based Frank-Wolfe with adaptive K and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y_mom = a.copy()
    x_mom = a.copy()
    K = K_init
    stall = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y_mom)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K most binding constraints
        K_sel = min(K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix: T_sel[k, i] = y_mom[k-i] if 0<=k-i<n
        T_sel = np.zeros((K_sel, n))
        for j in range(K_sel):
            k_int = int(top_idx[j])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                T_sel[j, i_start:i_end] = y_mom[k_int - np.arange(i_start, i_end)]

        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y_mom + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y_mom.copy()
        else:
            y_new = y_new / sy

        if nesterov:
            if t == 1:
                x_new = g_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y_mom)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
        else:
            x_new = y_new.copy()

        y_mom = y_new
        x_mom = x_new

        # Check improvement
        my = fast_autocorr(y_mom).max()
        if my < best_m:
            best_m = my
            best_a = y_mom.copy()
            stall = 0
        else:
            stall += 1

        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
            stall = 0

        # Adaptively increase K
        if t % K_inc_every == 0:
            K = min(K + 15, K_max)

        # Restart if stalled
        if stall > 100:
            stall = 0
            K = K_init
            y_mom = best_a + np.random.default_rng(t).normal(0, 0.01, n) * best_a
            y_mom = np.maximum(y_mom, 1e-10)
            y_mom = y_mom / y_mom.sum()
            x_mom = y_mom.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with sensitivity-based moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-sensitivity destination
            sens = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        sens[d] += a[d1]
                    d2 = d - k2i
                    if 0 <= d2 < n:
                        sens[d] += a[d2]

            candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Avoid positions that contribute to current peak
            in_peak = set()
            for k2 in top_k[:5]:
                k2i = int(k2)
                for d in range(max(0, k2i - n + 1), min(n, k2i + 1)):
                    in_peak.add(d)

            dest_candidates = [c for c in candidates if c not in in_peak]
            if not dest_candidates:
                dest_candidates = candidates[:20]

            if len(dest_candidates) == 0:
                continue

            src_sens = sens[src]
            best_dst = min(dest_candidates, key=lambda d: sens[d])
            if best_dst == src or sens[best_dst] >= src_sens:
                continue

            # Try multiple transfer fractions
            found = False
            for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.18]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m * (1.0 - 1e-8):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Try random destination
                d = int(rng.integers(n))
                if d != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[d] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_linear_ramp(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)


def make_bimodal(n, sep=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-80 * (x - 0.5 + sep)**2) + np.exp(-80 * (x - 0.5 - sep)**2) + 0.005
    return norm_sum1(a)


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-6)
    return norm_sum1(a)


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 6]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_tapered(n, alpha=2.0):
    """Tapered (raised cosine) window."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** alpha
    a = np.maximum(a, 1e-8)
    return norm_sum1(a)

        for idx in run_order:
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            lbl = unique_labels[idx]
            print(f"  NMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nmd_optimize(unique_inits[idx], t_avail,
                                     beta_init=20.0, beta_min=3.0, c0=5.0,
                                     top_k=60, restart_every=5000, verbose=False)
            try_update(a_out, f"nmd_{lbl}")

    # === Phase 2: LP-Frank-Wolfe on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 430.0)
    if t2 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   K_init=80, K_max=350, K_inc_every=25)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: NMD polish with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: NMD polish (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 130.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nmd_optimize(a_in, t3, beta_init=30.0, beta_min=5.0,
                                 c0=4.0, top_k=80, restart_every=8000, verbose=False)
        try_update(a_out, f"nmd_polish_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 200, n_best + 500, n_best + 1000]),
                            key=lambda x: x)
    longer_lengths = [l for l in longer_lengths if l <= 5000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        # Quick NMD
        t5a = min(remaining * 0.30, 150.0)
        if t5a > 20:
            a_out, _ = nmd_optimize(a_long, t5a, beta_init=15.0, beta_min=3.0,
                                     c0=4.0, top_k=50, restart_every=4000, verbose=False)
            try_update(a_out, f"nmd_n{new_n}")

        # LP on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4000:
            t5b = min(remaining * 0.25, 120.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False,
                                           K_init=60, K_max=250, K_inc_every=30)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP with large K ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t6 = min(remaining * 0.30, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True,
                                       K_init=150, K_max=500, K_inc_every=20)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.4, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more NMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: NMD alt (n={n_best}) ===")
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nmd_optimize(a_in, t8, beta_init=50.0, beta_min=10.0,
                                     c0=6.0, top_k=100, restart_every=3000,
                                     verbose=False, rng_seed=99)
            try_update(a_out, f"nmd_alt_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_tapered(n_fb, alpha=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
