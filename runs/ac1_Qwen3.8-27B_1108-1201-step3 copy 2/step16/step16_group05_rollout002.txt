Let me also think about the LP more carefully. The constraint matrix T has a special structure: it's a "convolution matrix" where each column j has the value y[j] in rows j through j+n-1 (with appropriate boundaries). Actually, let me be more precise.

`(f * g)_k = sum_j f_j * g_{k-j}` for k = 0, ..., 2n-2.

The valid j values for a given k are those where both j and k-j are in [0, n-1], i.e., j in [max(0, k-(n-1)), min(n-1, k)].

So the constraint is: `sum_{j in valid range} f_j * g_j <= M`... no wait, it's `sum_j f_j * g_{k-j}`. Let me use a different index.

`c_k = (f * g)_k = sum_{j=0}^{k} f_j * g_{k-j}` (with the convention that f_j = 0 for j >= n and g_j = 0 for j >= n).

So `c_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j}`.

If I want to write this as `T @ g <= M`, then `T[k, j] = f_{k-j}` when `k-j` is in [0, n-1], i.e., when j is in [k-(n-1), k]. And `T[k,j] = 0` otherwise.

So `T[k, j] = f[k-j]` if `0 <= k-j <= n-1`, else 0.

This means row k of T has non-zero entries for j in [max(0, k-n+1), min(n-1, k)].

For k=0: only j=0, T[0,0] = f[0]
For k=1: j=0,1, T[1,0]=f[1], T[1,1]=f[0]
...
For k=n-1: j=0,...,n-1, T[n-1,j] = f[n-1-j]
For k=n: j=1,...,n-1, T[n,j] = f[n-j]
...
For k=2n-2: only j=n-1, T[2n-2, n-1] = f[1]

So T is a "reversed Toeplitz" matrix (or more precisely, a convolution matrix). Each row is a shifted version of f.

This is a banded matrix with bandwidth n. For n=2400, each row has at most n non-zeros, and there are 2n-1=4799 rows. Total non-zeros ≈ n^2 = 5.76 million.

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_subgradient(a, num_peaks=30, epsilon_frac=0.1):
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n), M
    threshold = M * (1.0 - epsilon_frac)
    peak_lags = np.where(ac >= threshold)[0]
    top_idx = np.argsort(ac)[::-1][:num_peaks]
    peak_lags = np.unique(np.concatenate([peak_lags, top_idx]))
    weights = ac[peak_lags]
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for wi, k in zip(weights, peak_lags):
        k_int = int(k)
        if k_int == 0:
            grad += 2.0 * wi * a
        else:
            grad[:n - k_int] += wi * a[k_int:]
            grad[k_int:] += wi * a[:n - k_int]
    return grad, M


def subgradient_descent(a0, time_limit, verbose=True, seed=42, num_peaks=30, epsilon_frac=0.1):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    step_scale = 0.03
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(seed)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t > 1:
            momentum_frac = (t - 1) / (t + 1.5)
        else:
            momentum_frac = 0.0
        y = x + momentum_frac * (x - x_prev)
        y = project_simplex(y)
        g, M_curr = compute_peak_subgradient(y, num_peaks=num_peaks, epsilon_frac=epsilon_frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            g_norm = 1e-15
        eta = step_scale / (np.sqrt(t) * g_norm)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=50000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x_mom = a.copy()
    stall = 0
    max_stall = 200
    rng = np.random.default_rng(777)

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_convolution_matrix(y, n)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = np.ones(n) / n
            b_sum = 1.0
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        beta_t = (t - 1) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        y = y_new

        improved = False
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = candidate.copy()
                improved = True

        if not improved:
            stall += 1
        else:
            stall = 0

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.2)
            noise = rng.normal(0, 0.001, n) * best_a
            y_re = np.maximum(best_a + noise, 1e-10)
            y_re = y_re / y_re.sum()
            y = y_re
            x_mom = y_re.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_exact(a0, time_limit, verbose=True, seed=123, num_passes_max=500):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Fast subgradient warm-up ===
    rem = remaining_time()
    print(f"\n=== Phase 1: SubGradient n={n0} ===")
    t1 = min(rem * 0.08, 120.0)
    if t1 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_descent(
            a_in, t1, verbose=True, seed=42, num_peaks=40, epsilon_frac=0.15)
        try_update(a_out, f"subgrad_n{n0}")

    # === Phase 2: LP-Frank-Wolfe (main workhorse) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)} ===")
    t2 = min(rem * 0.35, 500.0)
    if t2 > 30 and len(best_seq) <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, max_iters=50000)
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 3: Coordinate descent polish ===
    rem = remaining_time()
    print(f"\n=== Phase 3: Coord n={len(best_seq)} ===")
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_exact(a_in, t3, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 4: Second LP-FW round ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round2 n={n_cur} ===")
    t4 = min(rem * 0.20, 300.0)
    if t4 > 30 and n_cur <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, max_iters=50000)
        try_update(a_out, f"lpfw2_n{n_cur}")

    # === Phase 5: Try longer length ===
    rem = remaining_time()
    n1 = min(n0 + 300, 3000)
    if n1 > n0 and rem > 150:
        print(f"\n=== Phase 5: Scale to n={n1} ===")
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        # Quick subgradient
        t5a = min(rem * 0.10, 60.0)
        if t5a > 15:
            a_out, _ = subgradient_descent(a_in, t5a, verbose=False, seed=99, num_peaks=30, epsilon_frac=0.1)
            try_update(a_out, f"subgrad_n{n1}")
        # LP-FW at longer length
        rem = remaining_time()
        t5b = min(rem * 0.25, 200.0)
        if t5b > 30 and n1 <= 3000:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=True, max_iters=50000)
            try_update(a_out, f"lpfw_n{n1}")
        # Coord at longer length
        rem = remaining_time()
        t5c = min(rem * 0.15, 100.0)
        if t5c > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_exact(a_in, t5c, verbose=False, seed=55)
            try_update(a_out, f"coord_n{n1}")

    # === Phase 6: Final LP-FW on best ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2800 and rem > 60:
        print(f"\n=== Phase 6: Final LP-FW n={n_best} ===")
        t6 = min(rem * 0.30, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, max_iters=50000)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 7: Final coordinate descent ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord n={n_best} ===")
    if rem > 30:
        t7 = min(rem * 0.60, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_exact(a_in, t7, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: One more subgradient burst on best ===
    rem = remaining_time()
    if rem > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: SubGrad burst n={n_best} ===")
        t8 = min(rem * 0.40, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_descent(a_in, t8, verbose=False, seed=2024, num_peaks=50, epsilon_frac=0.05)
            try_update(a_out, f"subgrad_final_n{n_best}")
        rem = remaining_time()
        if rem > 20:
            t8b = min(rem * 0.7, 60.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent_exact(a_in, t8b, verbose=False, seed=999)
                try_update(a_out, f"coord_ultra_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
