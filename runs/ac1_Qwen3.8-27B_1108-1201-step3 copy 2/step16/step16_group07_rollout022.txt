            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def ultra_fast_cd(a0, time_limit, verbose=False, seed=0):
    """
    Ultra-fast coordinate descent with O(n) incremental updates per step.
    No FFT calls in the inner loop.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s
    c = fast_autocorr(a).copy()
    best_a = a.copy()
    best_score = c.max()
    t0 = time.monotonic()
    n_passes = 0
    pts = 0

    # Precompute sorted indices for efficiency
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            aj = a[j]
            cmax = c.max()

            # Range of affected k values
            lo = j
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Generate candidates based on peak equalization logic
            typical = norm / n
            candidates = [0.0, aj * 0.4, aj * 0.7, aj * 1.0, aj * 1.4, aj * 2.0]
            candidates.extend([typical * 0.3, typical * 0.7, typical, typical * 1.5, typical * 2.5, typical * 4.0])
            # Add candidates that try to match the mean of affected peaks
            affected_vals = c[lo:hi]
            if len(affected_vals) > 0:
                mean_aff = np.mean(affected_vals)
                # Try values that would make affected peaks closer to mean
                candidates.append(aj * 0.1)
                candidates.append(aj + 0.5 * typical)
                candidates.append(aj - 0.5 * typical * (1 if aj > typical else -1))

            candidates = [max(0.0, min(val, norm * 0.9)) for val in candidates]
            candidates = sorted(set(candidates))

            best_x = aj
            best_obj = np.inf

            for x in candidates:
                if abs(x - aj) < 1e-13:
                    continue
                delta = x - aj
                new_sum = norm - aj + x
                if new_sum < 1e-10:
                    continue
                factor = norm / new_sum
                factor2 = factor * factor

                # Compute new max of affected region
                new_max_aff = 0.0
                for k in range(lo, hi):
                    if k == 2 * j:
                        val = (c[k] + 2.0 * delta * aj + delta * delta) * factor2
                    else:
                        val = (c[k] + 2.0 * delta * a[k - j]) * factor2
                    if val > new_max_aff:
                        new_max_aff = val

                # Conservative upper bound for unaffected region
                new_max_est = max(new_max_aff, cmax * factor2)

                if new_max_est < best_obj:
                    best_obj = new_max_est
                    best_x = x

            if abs(best_x - aj) > 1e-13:
                # Apply change with incremental update
                delta = best_x - aj
                new_sum = norm - aj + best_x
                factor = norm / new_sum
                factor2 = factor * factor

                # Update c incrementally (using old a values)
                for k in range(lo, hi):
                    if k == 2 * j:
                        c[k] += 2.0 * delta * aj + delta * delta
                    else:
                        c[k] += 2.0 * delta * a[k - j]

                # Rescale
                c *= factor2
                a[j] = best_x
                a *= factor

                pts += 1
                new_score = c.max()
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                else:
                    # Revert
                    a = best_a.copy()
                    c = fast_autocorr(a).copy()

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CDU pass={n_passes} pts={pts} ({elapsed:.0f}s): best={best_score:.10f}")

        if not improved:
            if n_passes > 20:
                break
            # Perturb to escape local minimum
            a = best_a + 0.002 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a * norm / s2
            c = fast_autocorr(a).copy()
            if c.max() > best_score * 1.05:
                break

    return best_a, best_score


def kkt_polish(a0, time_limit, solver, verbose=False):
    """
    Direct KKT-based polish: solve the active-set system and use it to guide updates.
    At KKT: sum_k w_k * 2*a[k-j] = lambda for all active j.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    iteration = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit and stagnation < 200:
        iteration += 1
        c = fast_autocorr(a)
        M_curr = c.max()

        # Identify active peaks
        tol = 0.02 + 0.03 * (1.0 - min(1.0, iteration / 50.0))
        active_peaks = np.where(c >= M_curr * (1.0 - tol))[0]
        if len(active_peaks) < 3:
            active_peaks = np.argsort(c)[-max(3, n // 50):]

        # Identify active coordinates
        support_thresh = 0.01 * a.max()
        active_coords = np.where(a > support_thresh)[0]
        if len(active_coords) < 2:
            break

        # Build the KKT system: A[j,k] * w = lambda/2
        # A[j,k] = 2*a[k-j] if 0 <= k-j < n, else 0
        # We solve: sum_k A[j,k]*w[k] = lambda for j in active_coords
        # Plus: sum_k w[k]*c[k] = M_curr (dual feasibility)
        # Plus: w[k] >= 0, w[k]*(c[k]-M) = 0 (complementary slackness)

        n_ac = len(active_coords)
        n_ap = len(active_peaks)

        # Build matrix B of shape (n_ac, n_ap)
        B = np.zeros((n_ac, n_ap))
        for i, j in enumerate(active_coords):
            for l, k in enumerate(active_peaks):
                diff = k - j
                if 0 <= diff < n:
                    B[i, l] = 2.0 * a[diff]

        # Solve least squares: B @ w ≈ lambda * ones(n_ac)
        # Normalize: divide by lambda, so B @ (w/lambda) ≈ ones
        # Let v = w / lambda, then B @ v ≈ ones(n_ac)
        # Also need: sum_k w[k] * c[k] = M_curr => lambda * sum_k v[k]*c[k] = M_curr

        try:
            # Add regularization for stability
            reg = 1e-6 * np.eye(n_ap)
            residual = B.T @ B + reg
            rhs = B.T @ np.ones(n_ac)
            v = np.linalg.solve(residual, rhs)
            v = np.maximum(v, 0.0)  # w >= 0
            if v.sum() < 1e-15:
                v = np.ones(n_ap) / n_ap
        except Exception:
            v = np.ones(n_ap) / n_ap

        # Compute lambda from the constraint sum w_k * c_k = M_curr
        # lambda * sum v_k * c_k = M_curr
        vwc = np.dot(v, c[active_peaks])
        if vwc < 1e-15:
            stagnation += 1
            continue
        lam = M_curr / vwc

        # Now compute the "ideal" gradient direction
        # grad[j] = sum_k w_k * 2*a[k-j] for j in active_coords
        # At KKT, this should equal lambda for all active j
        # The direction to move: reduce a[j] where grad[j] > lambda, increase where grad[j] < lambda

        grad_active = B @ (v * lam)  # = lam * B @ v ≈ lam * ones (if KKT satisfied)
        # Actually: grad[j] = sum_k w[k] * 2*a[k-j] = lam * sum_k v[k]*2*a[k-j] = lam * (B@v)[j]
        grad_active = lam * (B @ v)

        # Deviation from KKT
        deviation = grad_active - lam
        # We want to move a to reduce deviation
        # direction[j] = -deviation[j] (on active coords)
        # But we need to maintain sum = norm

        direction = np.zeros(n)
        direction[active_coords] = -deviation
        # Center to maintain sum constraint
        direction[active_coords] -= np.mean(direction[active_coords])

        dnorm = np.linalg.norm(direction[active_coords])
        if dnorm < 1e-12:
            stagnation += 1
            continue

        # Step size: try several
        for step_mult in [0.1, 0.05, 0.01, 0.005, 0.001]:
            step = step_mult * norm / dnorm
            a_trial = a + step * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial * norm / s_t
            score_t = fast_autocorr(a_trial).max()
            if score_t < best_score - 1e-14:
                a = a_trial
                best_score = score_t
                best_a = a_trial.copy()
                stagnation = 0
                break
        else:
            stagnation += 1

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter={iteration} ({elapsed:.0f}s): best={best_score:.10f}, dev_max={np.max(np.abs(deviation)):.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 40
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Progressive active set tightening
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 500.0)
            tol = 0.10 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(1.2 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(1.2 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 5 == 0:
                active_set = np.arange(conv_size)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.90)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average check
        if t % 40 == 0 and t > 100:
            a_damped = 0.6 * best_a + 0.4 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def dual_subgradient(a0, time_limit, solver, verbose=False, seed=42):
    """Dual-weighted subgradient with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.8 / n ** 0.3
    patience = 0
    max_patience = 500
    rng = np.random.default_rng(seed)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get active set with dual weights
        tol = 0.04
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 80):]

        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue

        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-6.0 * (M_curr - conv[active_set]) / max(M_curr * 0.015, 1e-10))
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k(a, int(k), n)

        support_thresh = 0.005 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break

        grad_mean = np.mean(grad[support])
        direction = grad - grad_mean

        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 1.5)
        else:
            patience += 1
            step *= 0.98

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualSub t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def perturb_and_reoptimize(best_seq, time_limit, solver, rng, seed):
    """Perturb and run a quick optimization cycle."""
    n = len(best_seq)
    scale = 0.008 + 0.012 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-10:
        return best_seq
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)

    # Quick LP
    t_lp = time_limit * 0.4
    a_out, _ = nesterov_lp_optimize(a_norm, t_lp, solver, verbose=False, seed=seed)

    # Quick CD
    t_cd = time_limit * 0.5
    a_out2, _ = ultra_fast_cd(a_out, t_cd, verbose=False, seed=seed + 1)
    return a_out2


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Ultra-fast CD post-LP, budget={t3:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = ultra_fast_cd(a_norm, t3, verbose=True, seed=2)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd2", t_global)

    # === Phase 4: KKT Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: KKT Polish, budget={t4:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = kkt_polish(a_norm, t4, solver, verbose=True)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"kkt1", t_global)

    # === Phase 5: LP Nesterov round 2 (tighter) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP Nesterov r2, budget={t5:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2", t_global)

    # === Phase 6: Dual subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        print(f"\n=== Phase 6: Dual Subgradient, budget={t6:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = dual_subgradient(a_norm, t6, solver, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"dual1", t_global)

    # === Phase 7: Ultra-fast CD (long, fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Ultra-fast CD final, budget={t7:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = ultra_fast_cd(a_norm, t7, verbose=True, seed=3)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd3", t_global)

    # === Phase 8: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 3
    t8_each = min(remaining / (n_restarts + 2), 80.0)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t8 = min(t8_each, remaining - 15)
        if t8 < 15:
            break
        print(f"\n=== Phase 8r{r}: Perturb+Optimize, budget={t8:.0f}s ===")
        a_out = perturb_and_reoptimize(best_seq, t8, solver, rng, seed=500 + r * 100)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}", t_global)

    # === Phase 9: Final KKT + CD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t9_kkt = min(remaining * 0.4, 50.0)
        t9_cd = min(remaining * 0.5, 60.0)
        if t9_kkt > 10:
            print(f"\n=== Phase 9a: Final KKT, budget={t9_kkt:.0f}s ===")
            n_cur = len(best_seq)
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = kkt_polish(a_norm, t9_kkt, solver, verbose=False)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"kkt_final", t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        if t9_cd > 10 and remaining > 15:
            print(f"\n=== Phase 9b: Final CD, budget={t9_cd:.0f}s ===")
            n_cur = len(best_seq)
            a_norm = normalize_to(best_seq, n_cur)
            a_out, _ = ultra_fast_cd(a_norm, t9_cd, verbose=False, seed=999)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
